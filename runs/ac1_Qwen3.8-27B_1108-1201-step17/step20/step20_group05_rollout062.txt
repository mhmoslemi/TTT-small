def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

def spectral_flatten_init(n, iterations=50, rng_seed=42):
    """Initialize a sequence with approximately flat power spectrum."""
    rng = np.random.default_rng(rng_seed)
    N = 2 * n - 1
    Nfft = next_pow2(N)
    
    # Start with a sine-squared shape
    x = np.sin(np.pi * np.linspace(0, 1, n)) ** 2
    x = x / x.sum()
    
    for _ in range(iterations):
        # Compute power spectrum
        X = np.fft.rfft(x, Nfft)
        mag = np.abs(X)
        mag = np.where(mag < 1e-30, 1e-30, mag)
        
        # Target: flat magnitude
        target_mag = np.ones_like(mag) * np.sqrt(np.mean(mag**2))
        
        # Blend current and target
        new_mag = 0.7 * mag + 0.3 * target_mag
        phase = np.angle(X)
        
        # Reconstruct
        X_new = new_mag * np.exp(1j * phase)
        # Make it conjugate symmetric for real output
        X_full = np.zeros(Nfft, dtype=complex)
        X_full[:len(X)] = X_new
        for i in range(1, len(X)):
            if Nfft - i < Nfft:
                X_full[Nfft - i] = np.conj(X_new[i])
        
        x_new = np.fft.irfft(X_full, Nfft)[:n]
        x_new = np.where(x_new > 0, x_new, 0)
        s = x_new.sum()
        if s < 1e-15:
            break
        x_new = x_new / s
        x = x_new
    
    return norm_sum1(x)

def lse_objective(x, beta, mask=None):
    """Log-sum-exp smoothed objective and gradient."""
    n = x.size
    c = fft_conv(x)
    if mask is None:
        cv = c
    else:
        cv = c[mask]
    if cv.size == 0:
        return 0.0, np.zeros(n)
    
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0 or not math.isfinite(se):
        return m, np.zeros(n)
    
    p = e / se
    pf = np.zeros(2 * n - 1, dtype=np.float64)
    if mask is None:
        pf[:] = p
    else:
        pf[mask] = p
    
    grad = 2.0 * cross_convolve_p_a(pf, x)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta

    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s

def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        ref = int(np.argmax(a)) if np.any(a > 0) else 0
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref

def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    return softmax_from_u(u)

def objgrad_v(v, beta, mask=None, reg_u=0.0, ref=0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)
    
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    
    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    
    a = softmax_from_u(u)
    f, ga = lse_objective(a, beta, mask)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n)
    
    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)
    
    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]
    
    if reg_u > 0:
        f += 0.5 * reg_u * float(np.dot(v, v))
        gv = gv + reg_u * v
    
    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size)
    return f, gv

def run_lbfgs(a0, beta, mask, maxfun=400, reg_u=1e-12, verbose=False):
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a
    if not SCIPY_OK or maxfun <= 0:
        return a
    
    v0, ref = make_v(a)
    t0 = time.monotonic()
    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun + 30),
        "ftol": 1e-12,
        "gtol": 1e-7,
        "disp": False
    }
    try:
        res = scipy.optimize.minimize(
            objgrad_v, v0,
            args=(beta, mask, reg_u, ref),
            method="L-BFGS-B", jac=True, options=options
        )
    except Exception:
        return a
    
    if res is not None and hasattr(res, "x"):
        vx = np.asarray(res.x, dtype=np.float64)
        if vx.size == v0.size and np.all(np.isfinite(vx)):
            a_out = from_v(vx, ref)
            a_out = norm_sum1(a_out)
            if verbose:
                print("    L-BFGS beta=" + repr(beta) + " n=" + str(n) +
                      " time=" + repr(time.monotonic() - t0))
            return a_out
    return a

def game_direction(f, active_idx, mw_iters=30, seed=0, init_logits=None):
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)
    
    rng = np.random.default_rng(seed)
    if init_logits is None:
        logits = np.zeros(K, dtype=np.float64)
    else:
        logits = np.asarray(init_logits, dtype=np.float64).copy()
        if logits.size != K:
            logits = np.zeros(K)
        logits = np.where(np.isfinite(logits), logits, 0.0)
        logmx = float(np.max(logits))
        if math.isfinite(logmx):
            logits = logits - logmx
    logits += rng.normal(0.0, 0.03, K)
    
    def normalize_logits_to_lambda(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        loc = loc - float(np.max(loc))
        w = np.exp(loc)
        s = float(w.sum())
        if s <= 0:
            w = np.ones(K)
            s = float(K)
        return w / s
    
    lambda_w = normalize_logits_to_lambda(logits)
    g = None
    for _ in range(max(1, int(mw_iters))):
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = lambda_w
        
        pressure = cross_convolve_p_a(pf, f)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)
        
        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            scaled = (pressure - pmn) / ptp
            w = np.exp(-10.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            g = 0.95 * g + 0.05 * (np.ones(n) / float(n))
            g = g / float(g.sum())
        
        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 15.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))
        lambda_w = normalize_logits_to_lambda(logits)
    
    return g

def lp_direction(f, active_idx, time_limit=6.0, max_k=800):
    if not SCIPY_OK or not highs_available():
        return None
    f = norm_sum1(f)
    n = f.size
    active_idx = np.asarray(active_idx, dtype=int)
    if active_idx.size == 0:
        return None
    if active_idx.size > max_k:
        active_idx = active_idx[:max_k]
    K = active_idx.size
    
    row_parts = []
    col_parts = []
    val_parts = []
    for r, k in enumerate(active_idx):
        s = max(0, int(k) - n + 1)
        e = min(n, int(k) + 1)
        if s >= e:
            continue
        cols = np.arange(s, e, dtype=np.int32)
        row_parts.append(np.full(cols.size, r, dtype=np.int32))
        col_parts.append(cols)
        val_parts.append(f[int(k) - cols].astype(np.float64))
    
    if not row_parts:
        return None
    
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0
    
    neg_one = scipy.sparse.csc_matrix(-np.ones((K, 1)), dtype=np.float64)
    A_ub = scipy.sparse.hstack((A, neg_one), format="csr")
    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, n] = 0.0
    b_eq = np.ones(1, dtype=np.float64)
    b_ub = np.zeros(K, dtype=np.float64)
    bounds = [(0.0, None)] * n + [(None, None)]
    
    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub,
            A_eq=A_eq, b_eq=b_eq, bounds=bounds,
            method="HiGHS", options={"disp": False, "time_limit": float(time_limit)}
        )
    except Exception:
        return None
    
    if res is None or not getattr(res, "success", False):
        return None
    
    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs

def line_search_mix(a, g, cf, cg, fg, current_M):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    
    alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
    
    best_alpha = 0.0
    best_m = current_M
    for alpha in alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        m = float(np.max(val))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha
    
    if best_alpha <= 0:
        return a.copy(), best_m, 0.0
    
    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha

def frank_wolfe_search(a0, budget, active_max=3000, game_iters=30,
                       exact_every=6, exact_time=6.0, verbose=False):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    active = active_top(c, None, active_max)
    
    no_improve = 0
    t = 0
    while time.monotonic() - t0 < time_limit - 3 and no_improve < 100:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 20:
            break
        
        g = None
        if highs_available() and exact_every > 0 and (t % exact_every == 0) and time_left > 50:
            g = lp_direction(a, active, exact_time, max_k=800)
        
        if g is None:
            init_log = None
            if active.size > 0:
                av = c[active]
                av = np.where(np.isfinite(av) & (av > 0.0), av, 1e-300)
                init_log = np.log(av)
                init_log -= float(np.max(init_log))
            g = game_direction(a, active, mw_iters=game_iters, seed=7919 + t, init_logits=init_log)
        
        if g is None:
            break
        
        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search_mix(a, g, c, cg, fg, M)
        
        if new_M < M - 1e-16:
            if alpha > 0:
                one = 1.0 - alpha
                c = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
                c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
            a = new_a
            M = new_M
            no_improve = 0
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            active = active_top(c, None, active_max)
            if verbose and (t <= 3 or t % 15 == 0):
                print("    FW t=" + str(t) + " proxy=" + repr(2.0 * n * best_M) +
                      " alpha=" + repr(alpha) + " active=" + str(active.size))
        else:
            no_improve += 1
            if no_improve % 12 == 0:
                rng = np.random.default_rng(4242 + t)
                scale = 0.02 + 0.01 * (no_improve // 12)
                pert = rng.normal(0.0, scale, n) * best_a
                a = best_a + pert
                np.maximum(a, 1e-12, out=a)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = float(np.max(c))
                active = active_top(c, None, active_max)
                no_improve = 0
    
    return best_a

def apply_single_move(c, a, s, d, tmove):
    a = np.asarray(a, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64).copy()
    n = a.size
    c_new = c.copy()
    # Moving tmove from index s to index d changes conv by:
    # c_new[k] = c_old[k] + tmove * (a_{k-d} - a_{k-s}) for relevant k
    # More precisely, the self-convolution changes as:
    # (x + t*(e_d - e_s)) * (x + t*(e_d - e_s))
    # = x*x + t*(x*e_d - x*e_s) + t*(e_d*x - e_s*x) + t^2*(e_d*e_d - e_d*e_s - e_s*e_d + e_s*e_s)
    # The cross terms: x*e_d shifted by d, x*e_s shifted by s
    # x * e_d has a 1 at position d, so conv(x, e_d) = x shifted by d
    # Actually let me just use the direct formula:
    # c_old[k] = sum_i x_i * x_{i+k}
    # After moving: x_s' = x_s - t, x_d' = x_d + t
    # c_new[k] = c_old[k] - t*(x_{k-s} if 0<=k-s<n) - t*(x_{k-d} if 0<=k-d<n)
    #            + t*(x'_{k-s} if 0<=k-s<n) + t*(x'_{k-d} if 0<=k-d<n)
    # Wait, this is getting complicated. Let me use the simpler approach:
    # The change in c due to x_s -= t and x_d += t:
    # dc[k] = -t * (x_{k-s} + x_{k-s}) + t * (x_{k-d} + x_{k-d}) + t^2 * (delta(k) at 2d and 2s and -(at s+d))
    # Actually for self-convolution:
    # c[k] = sum_i x_i * x_{i+k} (for k >= 0, with appropriate ranges)
    # numpy convolve: c[k] = sum_i x[i] * x[k-i] for k = 0..2n-2
    
    # Let me just recompute using the rank-1 update properly.
    # x_new = x + t*(e_d - e_s)
    # conv(x_new, x_new) = conv(x,x) + t*conv(x, e_d-e_s) + t*conv(e_d-e_s, x) + t^2*conv(e_d-e_s, e_d-e_s)
    # conv(x, e_d) = x shifted right by d (i.e., x[i-d] placed at position i)
    # In numpy convolve convention: conv(x, e_d)[k] = sum_i x[i] * e_d[k-i] = x[k-d] (if 0<=k-d<n)
    # So conv(x, e_d-e_s)[k] = x[k-d] - x[k-s] (where defined)
    # conv(e_d-x_s, x) is the same by symmetry of self-conv... wait no, x is not symmetric.
    # conv(e_d, x)[k] = sum_i e_d[i] * x[k-i] = x[k-d] (if 0<=k-d<n)
    # So it's the same! conv(x, e_d) = conv(e_d, x) = x shifted by d.
    
    # So: dc[k] = t * 2 * (x[k-d] - x[k-s]) + t^2 * (delta(k-2d) + delta(k-2s) - 2*delta(k-(s+d)))
    # where delta(k-j) = 1 if k==j else 0
    
    for k in range(0, 2 * n - 1):
        val = 0.0
        if 0 <= k - d < n:
            val += x_k_minus_d = a[k - d]  # can't do this, need different approach
    # This is too slow for large n. Let me use the vectorized approach:
    
    c_new = c.copy()
    # Add t * a shifted by d (for the +t at position d)
    d_start = d
    d_end = d + n
    c_new[d_start:min(d_end, 2*n-1)] += tmove * a[:min(n, 2*n-1-d_start)]
    # Add t * a shifted by s (for the -t at position s, but subtract)
    s_start = s
    s_end = s + n
    c_new[s_start:min(s_end, 2*n-1)] -= tmove * a[:min(n, 2*n-1-s_start)]
    # t^2 terms
    t2 = tmove * tmove
    if 2 * d < 2 * n - 1:
        c_new[2 * d] += t2
    if 2 * s < 2 * n - 1:
        c_new[2 * s] += t2
    if s + d < 2 * n - 1:
        c_new[s + d] -= 2.0 * t2
    
    return np.where(np.isfinite(c_new), c_new, 0.0)

def coordinate_polish(a0, budget, verbose=False):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = float(np.max(c))
    best_a = a.copy()
    best_M = M
    n = a.size
    stall = 0
    rnd = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit - 2 and stall < 150:
        rnd += 1
        if n <= 1:
            break
        
        # Find top peaks in convolution
        top_k = min(50, 2 * n - 1)
        top_idx = active_top(c, None, top_k)
        if top_idx.size == 0:
            break
        
        # Compute pressure (gradient-like) from top peaks
        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_idx.size)
        pressure = cross_convolve_p_a(p_top, a)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)
        
        # Candidate source: high pressure + high value (moving mass from here reduces peaks)
        # Candidate dest: low pressure (receiving mass here is "safe")
        src_score = pressure * (1.0 + float(n) * a)
        dst_score = -pressure + 1e-5 * a
        
        src_count = min(15, n)
        dst_count = min(20, n)
        src_candidates = np.argpartition(src_score, min(src_count - 1, n - 1))[:src_count]
        dst_candidates = np.argpartition(dst_score, min(dst_count - 1, n - 1))[:dst_count]
        
        best_cand = None
        improved_any = False
        
        for srr in src_candidates:
            s = int(srr)
            if a[s] < 1e-12:
                continue
            for drr in dst_candidates:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.003, 0.007, 0.015, 0.03, 0.06):
                    tmove = a[s] * frac
                    if tmove <= 1e-16:
                        continue
                    tmove = min(tmove, a[s] * 0.95)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = float(np.max(c_new))
                    if m_new < M - 1e-18:
                        if best_cand is None or m_new < best_cand[0]:
                            best_cand = (m_new, s, d, tmove)
                            improved_any = True
        
        if not improved_any:
            # Random exploratory moves
            for _ in range(40):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-12:
                    continue
                frac = float(rng.uniform(0.002, 0.04))
                tmove = a[s] * frac
                tmove = min(tmove, a[s] * 0.95)
                if tmove <= 1e-16:
                    continue
                c_new = apply_single_move(c, a, s, d, tmove)
                m_new = float(np.max(c_new))
                if m_new < M - 1e-18:
                    if best_cand is None or m_new < best_cand[0]:
                        best_cand = (m_new, s, d, tmove)
                        improved_any = True
        
        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            np.maximum(a, 0.0, out=a)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-15:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 5 or rnd % 30 == 0):
                print("    Coord rnd=" + str(rnd) + " proxy=" + repr(2.0 * n * best_M))
        else:
            stall += 1
            if stall % 10 == 9:
                # Perturb and try again
                rng2 = np.random.default_rng(999 + stall)
                scale = 0.01 + 0.005 * (stall // 10)
                pert = rng2.normal(0.0, scale, n) * best_a
                a = best_a + pert
                np.maximum(a, 1e-12, out=a)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = float(np.max(c))
                stall = 0
    
    return best_a

def lbfgs_phase(a_init, budget, cost_est):
    """Run L-BFGS with adaptive beta schedule."""
    a = norm_sum1(a_init)
    n = a.size
    size = 2 * n - 1
    
    schedules = [
        (8.0, None),
        (20.0, None),
        (50.0, min(200, size // 8)),
        (100.0, min(150, size // 12)),
        (200.0, min(100, size // 15)),
        (400.0, min(60, size // 25)),
        (800.0, min(30, size // 40)),
        (1500.0, min(15, size // 80)),
    ]
    
    total_budget = int(budget / max(2.0 * cost_est, 0.05))
    per_run = int(max(50, min(500, total_budget // max(1, len(schedules)))))
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_M = float(np.max(fft_conv(a)))
    current_a = a.copy()
    
    for beta, top_count in schedules:
        if time.monotonic() - t0 > budget - 3:
            break
        if top_count is None:
            msk = None
        else:
            c = fft_conv(current_a)
            top = active_top(c, None, top_count)
            msk = np.zeros(size, dtype=bool)
            msk[top] = True
        
        a_new = run_lbfgs(current_a, beta, msk, maxfun=per_run, verbose=False)
        c_new = fft_conv(a_new)
        m_new = float(np.max(c_new))
        
        if math.isfinite(m_new) and m_new <= best_M * 1.001 + 1e-18:
            current_a = a_new
        else:
            current_a = best_a
        
        if math.isfinite(m_new) and m_new < best_M - 1e-15:
            best_M = m_new
            best_a = a_new.copy()
            print("    L-BFGS phase beta=" + repr(beta) + " proxy=" + 
                  repr(2.0 * n * best_M) + " t=" + repr(time.monotonic() - t0))
    
    return best_a

def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:

    n = arr.size
    if n == 0:

def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

    rng = np.random.default_rng(rng_seed)
    N = next_pow2(2 * n - 1)

    x = np.sin(np.pi * np.linspace(0, 1, n)) ** 2
    x = x / x.sum()

    for it in range(iterations):
        X = np.fft.rfft(x, N)
        mag = np.abs(X)
        mag_safe = np.where(mag < 1e-30, 1e-30, mag)
        phase = np.angle(X)

        # Geometric mean of magnitudes (spectral flatness target)
        log_mag = np.log(mag_safe)
        geo_mean = np.exp(np.mean(log_mag))

        # Blend: push toward flat spectrum
        alpha = 0.3 + 0.2 * (it / max(1, iterations))
        new_mag = mag_safe ** (1 - alpha) * geo_mean ** alpha

        X_new = new_mag * np.exp(1j * phase)
        x_new = np.fft.irfft(X_new, N)[:n]
        x_new = np.where(x_new > 0, x_new, 0.0)
        s = x_new.sum()
        if s < 1e-15:
            break
        x = x_new / s

    return norm_sum1(x)


def lse_objective(x, beta, mask=None):
    n = x.size
    c = fft_conv(x)
    if mask is None:
        cv = c
    else:
        cv = c[mask]
    if cv.size == 0:
        return 0.0, np.zeros(n)

    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0 or not math.isfinite(se):
        return m, np.zeros(n)

    p = e / se
    pf = np.zeros(2 * n - 1, dtype=np.float64)
    if mask is None:
        pf[:] = p
    else:
        pf[mask] = p

    grad = 2.0 * cross_convolve_p_a(pf, x)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta
    return f, grad


def softmax_from_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.ones(n, dtype=np.float64) / float(n)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        ref = int(np.argmax(a)) if np.any(a > 0) else 0
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    return softmax_from_u(u)


def objgrad_v(v, beta, mask=None, reg_u=0.0, ref=0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)

    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1

    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    a = softmax_from_u(u)
    f, ga = lse_objective(a, beta, mask)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n)

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]

    if reg_u > 0:
        f += 0.5 * reg_u * float(np.dot(v, v))
        gv = gv + reg_u * v

    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size)
    return f, gv


def run_lbfgs(a0, beta, mask, maxfun=400, reg_u=1e-12, verbose=False):
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a
    if not SCIPY_OK or maxfun <= 0:
        return a

    v0, ref = make_v(a)
    t0 = time.monotonic()
    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun + 50),
        "ftol": 1e-13,
        "gtol": 1e-8,
        "disp": False
    }
    try:
        res = scipy.optimize.minimize(
            objgrad_v, v0,
            args=(beta, mask, reg_u, ref),
            method="L-BFGS-B", jac=True, options=options
        )
    except Exception:
        return a

    if res is not None and hasattr(res, "x"):
        vx = np.asarray(res.x, dtype=np.float64)
        if vx.size == v0.size and np.all(np.isfinite(vx)):
            a_out = from_v(vx, ref)
            a_out = norm_sum1(a_out)
            if verbose:
                print("    L-BFGS beta=" + repr(beta) + " n=" + str(n) +
                      " time=" + repr(time.monotonic() - t0))
            return a_out
    return a


def lbfgs_phase(a_init, budget, cost_est):
    """Run L-BFGS with adaptive beta schedule and active-set refinement."""
    a = norm_sum1(a_init)
    n = a.size
    size = 2 * n - 1

    # Progressive refinement: start smooth, end sharp
    schedules = [
        (6.0, None),
        (15.0, None),
        (40.0, min(300, size // 5)),
        (80.0, min(200, size // 8)),
        (150.0, min(120, size // 12)),
        (300.0, min(80, size // 20)),
        (600.0, min(40, size // 35)),
        (1200.0, min(20, size // 60)),
        (2500.0, min(10, max(5, size // 100))),
    ]

    total_eval_budget = int(budget / max(2.0 * cost_est, 0.05))
    per_run = int(max(40, min(400, total_eval_budget // max(1, len(schedules)))))

    t0 = time.monotonic()
    best_a = a.copy()
    best_M = float(np.max(fft_conv(a)))
    current_a = a.copy()

    for beta, top_count in schedules:
        if time.monotonic() - t0 > budget - 3:
            break
        if top_count is None:
            msk = None
        else:
            c = fft_conv(current_a)
            top = active_top(c, None, top_count)
            if top.size == 0:
                msk = None
            else:
                msk = np.zeros(size, dtype=bool)
                msk[top] = True

        a_new = run_lbfgs(current_a, beta, msk, maxfun=per_run)
        c_new = fft_conv(a_new)
        m_new = float(np.max(c_new))

        # Accept if not significantly worse
        if math.isfinite(m_new) and m_new <= best_M * 1.002 + 1e-18:
            current_a = a_new
        else:
            current_a = best_a

        if math.isfinite(m_new) and m_new < best_M - 1e-15:
            best_M = m_new
            best_a = a_new.copy()
            print("    L-BFGS beta=" + repr(beta) + " proxy=" +
                  repr(2.0 * n * best_M) + " t=" + repr(time.monotonic() - t0))

    return best_a
