    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_smoothed_grad(y, beta, top_k=None):
    """
    Compute smoothed gradient of max(convolve(y,y)) using softmax approximation.
    Returns (gradient, softmax_weights, autocorrelation).
    """
    n = len(y)
    b = fast_autocorr(y)
    if top_k is not None and top_k < len(b) and top_k > 0:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -400, 400)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_use)
        mask = b_max == b_use
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    # Gradient: d/dy_i max_k (y*y)[k] approx sum_k p[k] * y[k-i]
    a_rev = y[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, p, b


def fista_optimize(a0, time_limit, beta_start=20.0, beta_end=800.0,
                   beta_growth=1.003, top_k=20, restart_every=4000,
                   verbose=True):
    """FISTA with BB step size and restarts."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    avg_x = x.copy()
    avg_count = 0.0
    s_prev = None
    y_prev = None
    bb_step = 1.0 / (n ** 0.5)

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, b = compute_smoothed_grad(y, beta, top_k)

        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_candidate = sy_gy / gy_gy
                if bb_candidate > 1e-8 and bb_candidate < 10.0:
                    bb_step = max(1e-6, min(bb_candidate, 5.0 / beta))
                else:
                    bb_step = max(bb_step * 0.9, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(bb_step, 0.15)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / (n ** 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + repr(round(beta, 1)))

    J_avg = 2.0 * n * fast_autocorr(avg_x).max()
    J_best = 2.0 * n * best_max_b
    if J_avg < J_best:
        return avg_x, fast_autocorr(avg_x).max()
    return best_a, best_max_b


def lp_fw_optimize(a0, time_limit, solver, initial_K=500, max_K=6000,
                   add_per_iter=30, verbose=True):
    """
    LP-Frank-Wolfe for minimizing max(a*a) with sum(a)=1, a>=0.
    Uses large active set for strong convergence.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    # Initialize with top K constraints by value
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 1500
    best_trial_M_recent = best_m

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set size
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            # Keep tightest constraints
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix C where C[jj, i] = y[k - i] for constraint k
        # (y*g)[k] = sum_i y[k-i] * g[i] for valid indices
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # LP: min M s.t. sum(g)=1, g>=0, C@g <= M
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                ac_now = fast_autocorr(y)
                active_set = set(int(k) for k in np.argsort(ac_now)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints (y*g_dir > M_curr)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Adaptive step size: try multiple alphas
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha,
                  max(0.01, default_alpha * 0.5),
                  max(0.01, default_alpha * 1.5),
                  1.0 / (t + 1),
                  max(0.01, 3.0 / (t + 3.0))]
        alphas = [max(0.005, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)) + " M=" + repr(round(best_m * 2 * n, 6)))

    return best_a, best_m


def peak_directed_search(a0, time_limit, verbose=True):
    """
    Directly target top peaks by moving mass from peak-contributing positions
    to low-impact positions.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:50]
        improved = False

        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[rng.integers(min(20, len(top_peaks)))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])

            # Find positions that don't contribute to any top peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:25]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src:
                continue
            if a[src] < 1e-12:
                continue

            # Try multiple transfer fractions
            best_frac = -1
            best_trial_M = best_m
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt > a[src]:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_trial_M:
                    best_trial_M = M_trial
                    best_frac = frac

            if best_frac > 0:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_trial_M
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:50]

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Peak-dir (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def cyclic_shift_best(a0):
    """Find the best cyclic shift of a0."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update_best(best_seq, best_score, a, label, t_global):
    """Try to update best sequence if a is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:

        best_seq = a_norm.copy()

        print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def generate_gaussian(n, sigma_frac=0.3):
    """Generate a Gaussian-shaped sequence."""
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    g = np.maximum(g, 1e-10)
    return g / g.sum()


def generate_triangular(n):
    """Generate a triangular sequence."""
    x = np.linspace(0, 1, n)
    t = np.minimum(x, 1 - x) * 2
    t = np.maximum(t, 1e-10)
    return t / t.sum()


def generate_u_shaped(n, power=2.0):
    """Generate a U-shaped (boundary-emphasized) sequence."""
    x = np.linspace(0, 1, n)
    u = ((x - 0.5) * 2) ** power + 0.01
    return u / u.sum()

                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    # If no parent, use diverse initializations
    if best_seq is None:
        n_base = 5000
        best_score = float('inf')
        for name, init_func in [("gauss", lambda: generate_gaussian(n_base)),
                                ("tri", lambda: generate_triangular(n_base)),
                                ("u", lambda: generate_u_shaped(n_base))]:
            c = init_func()
            sc = evaluate_safe(c)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = c
                print("Init " + name + ": score=" + repr(sc))

    n0 = len(best_seq)

    # === Phase 1: FISTA (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60:
        a_out, _ = fista_optimize(norm_sum1(best_seq), t1,
                                  beta_start=15.0, beta_end=600.0,
                                  beta_growth=1.003, top_k=15,
                                  restart_every=3000, verbose=True)
        best_seq, best_score = try_update_best(best_seq, best_score, a_out, "fista1", t_global)

    # === Phase 2: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Cyclic shift (n=" + str(len(best_seq)) + ") ===")
    if remaining > 60:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update_best(best_seq, best_score, a_shifted, "cyclic1", t_global)

    # === Phase 3: LP-FW (45% time - main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: LP-FW (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.45, 700.0)
    if t3 > 60 and n_best <= 10000:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t3, solver,
                                  initial_K=600, max_K=8000,
                                  add_per_iter=40, verbose=True)
        best_seq, best_score = try_update_best(best_seq, best_score, "lp1", t_global) if False else try_update_best(best_seq, best_score, a_out, "lp1", t_global)

    # === Phase 4: Peak-directed local search (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Peak-directed (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 30:
        a_out, _ = peak_directed_search(norm_sum1(best_seq), t4, verbose=True)
        best_seq, best_score = try_update_best(best_seq, best_score, a_out, "peak1", t_global)

    # === Phase 5: Cyclic shift after LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update_best(best_seq, best_score, a_shifted, "cyclic2", t_global)

    # === Phase 6: Second FISTA with higher smoothing (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: FISTA v2 (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 40:
        a_out, _ = fista_optimize(norm_sum1(best_seq), t6,
                                  beta_start=50.0, beta_end=1500.0,
                                  beta_growth=1.005, top_k=10,
                                  restart_every=2000, verbose=True)
        best_seq, best_score = try_update_best(best_seq, best_score, a_out, "fista2", t_global)

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    # Try slightly longer sequences
    longer_lengths = [n_best + 500, n_best + 1000]
    longer_lengths = [l for l in longer_lengths if l <= 12000 and l > n_best]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print("\n=== Phase 7: Scale n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.30, 180.0)
        if t7a > 40:
            a_out, _ = fista_optimize(a_long, t7a,
                                      beta_start=20.0, beta_end=500.0,
                                      beta_growth=1.004, top_k=12,
                                      restart_every=2500, verbose=True)
            best_seq, best_score = try_update_best(best_seq, best_score, a_out, "fista_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and len(best_seq) <= 10000:
            t7b = min(remaining * 0.40, 200.0)
            if t7b > 40:
                a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t7b, solver,
                                          initial_K=500, max_K=5000,
                                          add_per_iter=30, verbose=True)
                best_seq, best_score = try_update_best(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

    # === Phase 8: Final LP burst ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final LP (n=" + str(n_best) + ") ===")
    if remaining > 60 and n_best <= 10000:
        t8 = min(remaining * 0.35, 200.0)
        if t8 > 40:
            a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t8, solver,
                                      initial_K=800, max_K=10000,
                                      add_per_iter=50, verbose=True)
            best_seq, best_score = try_update_best(best_seq, best_score, a_out, "lp_final", t_global)

    # === Phase 9: Final peak-directed polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Phase 9: Final peak-directed ===")
        t9 = min(remaining * 0.30, 100.0)
        if t9 > 20:
            a_out, _ = peak_directed_search(norm_sum1(best_seq), t9, verbose=True)
            best_seq, best_score = try_update_best(best_seq, best_score, a_out, "peak_final", t_global)

    # === Final: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Final: Cyclic shift ===")
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update_best(best_seq, best_score, a_shifted, "cyclic_final", t_global)

    # === Final: FISTA micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print("\n=== Final: FISTA micro-polish ===")
        t_f = min(remaining * 0.50, 100.0)
        if t_f > 20:
            a_out, _ = fista_optimize(norm_sum1(best_seq), t_f,
                                      beta_start=100.0, beta_end=2000.0,
                                      beta_growth=1.01, top_k=8,
                                      restart_every=1500, verbose=True)
            best_seq, best_score = try_update_best(best_seq, best_score, a_out, "fista_final", t_global)

    # Final cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update_best(best_seq, best_score, a_shifted, "cyclic_very_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
