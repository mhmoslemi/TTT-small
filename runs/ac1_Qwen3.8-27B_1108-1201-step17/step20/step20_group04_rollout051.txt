"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer on softmax (log) parameterization with adaptive top-K log-sum-exp smoothing
2. Gradient-based peak surgery using full convolution gradient structure
3. Multiresolution: optimize at increasing lengths (6700 -> 9000 -> 12000)
4. Multi-phase search with diverse restarts and periodic structure exploration
Key insight: longer sequences can achieve lower constants; Adam's per-coordinate
adaptive learning rates escape saddle points better than Nesterov for this landscape.
The gradient has closed form: dJ/dw[j] = 4n*a[j]/beta * (corr[j] - p_dot_c)
"""

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear convolution (autocorrelation) of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def softmax(w):
    """Stable softmax."""
    w_shifted = w - w.max()
    e = np.exp(w_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(w) / len(w)
    return e / s


def compute_grad_softmax(a, beta, K):
    """
    Compute gradient of J_smooth = 2*n/beta * log(sum_{topK} exp(beta*c[k])) w.r.t. w (log-parameterization).
    Returns grad_w of shape (n,).
    Formula: dJ/dw[j] = 4*n*a[j]/beta * (corr[j] - p_dot_c)
    where corr[j] = sum_k p[k]*a[k-j], p_dot_c = sum_k p[k]*c[k]
    """
    n = len(a)
    c = fast_autocorr(a)
    K = min(K, 2 * n - 1)
    top_idx = np.argpartition(c, -K)[-K:]
    c_top = c[top_idx]
    # Softmax weights over top-K
    c_shifted = beta * (c_top - c_top.max())
    c_shifted = np.clip(c_shifted, -300, 300)
    exp_vals = np.exp(c_shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p_top = np.zeros(K)
        p_top[np.argmax(c_top)] = 1.0
    else:
        p_top = exp_vals / total

    # Build full p vector
    p = np.zeros(2 * n - 1)
    p[top_idx] = p_top

    # Compute corr[j] = sum_k p[k]*a[k-j] via FFT
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_rev = np.fft.rfft(a[::-1], fft_size)
    conv = np.fft.irfft(P * A_rev, fft_size)
    corr = conv[n - 1:2 * n - 1]

    p_dot_c = np.dot(p_top, c_top)

    grad_w = (4.0 * n * a / beta) * (corr - p_dot_c)

    return grad_w, c, c.max()


def adam_optimize(a0, time_limit, beta_start=50.0, beta_end=5000.0,
                  beta_growth=1.005, lr=0.005, K_ratio=0.02,
                  warmup=200, verbose=True, seed=42, clip_grad=1.0):
    """
    Adam optimizer on softmax parameterization to minimize 2*n*max(conv(a,a)).
    a0: initial sequence (will be normalized to sum=1)
    Returns (best_a, best_J) where best_J = 2*n*max_conv.
    """
    n = len(a0)
    a = norm_sum1(a0)
    w = np.log(np.maximum(a, 1e-30))
    best_a = a.copy()
    best_c = fast_autocorr(a)
    best_J = 2.0 * n * best_c.max()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t0 = time.monotonic()
    beta = beta_start
    K = max(20, int(n * K_ratio))
    no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Learning rate schedule: warmup then slow decay
        progress = min(t / max(time_limit * 2000, 1000), 1.0)
        lr_eff = lr * (1.0 - 0.3 * progress)

        # Compute gradient
        grad_w, c, c_max = compute_grad_softmax(a, beta, K)

        # Gradient clipping
        grad_norm = np.sqrt(np.mean(grad_w ** 2))
        if grad_norm > clip_grad:
            grad_w = grad_w * (clip_grad / grad_norm)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w = w - lr_eff * m_hat / (np.sqrt(v_hat) + eps)

        # Update a
        a = softmax(w)

        # Periodic beta increase
        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)
            K = max(20, int(n * K_ratio * min(3.0, t / 2000.0)))

        # Check improvement every 10 iterations

            J_check = 2.0 * n * c_check.max()
            if J_check < best_J - 1e-12:
                best_J = J_check
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Restart if stuck
            if no_improve > 5000 and t > 1000:
                no_improve = 0
                t = max(t // 2, 100)
                m = np.zeros(n)
                v = np.zeros(n)
                beta = max(beta * 0.5, beta_start)
                # Perturb from best
                noise = rng.normal(0, 0.02, n) * best_a
                a_try = best_a + noise
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                a = a_try
                w = np.log(np.maximum(a, 1e-30))

            print("    Adam t=%d (%.1fs): J=%.6f, beta=%.1f, K=%d" % (t, elapsed, best_J, beta, K))

def gradient_peak_surgery(a0, time_limit, verbose=True, seed=99):
    """
    Gradient-based peak surgery: use the gradient structure to make targeted moves.
    Focus on reducing the top peaks by moving mass from high-gradient positions
    to low-gradient positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_c = fast_autocorr(a)
    best_J = 2.0 * n * best_c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 2000:
        c = fast_autocorr(a)
        c_max = c.max()
        J_cur = 2.0 * n * c_max
        if J_cur < best_J - 1e-12:
            best_J = J_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top peaks
        n_top = min(50, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:n_top]

        # For each top peak, try to reduce it
        improved = False
        for ki in range(min(10, n_top)):
            k = int(top_idx[ki])
            if c[k] < c_max * 0.5:
                break

            # Identify contributing pairs
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k - idx]
            total_contrib = products.sum()
            if total_contrib < 1e-20:
                continue

            # Find the largest contributor
            src_i = int(idx[np.argmax(products)])
            partner = k - src_i
            if partner < 0 or partner >= n:
                continue

            # Choose which element to reduce (the one that appears in more top peaks)
            # Count how many top peaks involve src_i and partner
            src_count = 0
            par_count = 0
            for k2 in top_idx[:20]:
                k2i = int(k2)
                if 0 <= k2i - src_i < n:
                    src_count += 1
                if 0 <= k2i - partner < n:
                    par_count += 1

            if src_count >= par_count:
                reduce_i = src_i
            else:
                reduce_i = partner

            if a[reduce_i] < 1e-14:
                continue

            # Compute "damage" for each destination: how much would it increase top peaks?
            # approx: for dest d, the new contribution to peak k2 is a[reduce_i_frac]*a[k2-d]
            # We want to minimize this
            top_peaks_arr = top_idx[:15]
            n_try = min(200, n // 10)
            # Sample candidate destinations
            candidates = rng.choice(n, size=n_try, replace=False)

            # Score each destination by sum of a[k2 - dst] over top peaks
            scores = np.zeros(len(candidates))
            for di, dst in enumerate(candidates):
                for k2 in top_peaks_arr:
                    k2i = int(k2)
                    off = k2i - dst
                    if 0 <= off < n:
                        scores[di] += a[off]
                # Penalize if dst is close to reduce_i (to spread mass)
                if abs(dst - reduce_i) < n // 50:
                    scores[di] *= 2.0

            best_dst = int(candidates[np.argmin(scores)])
            if best_dst == reduce_i:
                continue

            # Try different fractions
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[reduce_i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_i] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                J_trial = 2.0 * n * M_trial
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break

            if improved:
                break

        if not improved:
            # Random exploration
            if no_improve > 500:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            J_new = 2.0 * n * fast_autocorr(a).max()
                            if J_new < best_J:
                                best_J = J_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve % 500 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            print("    GPeakSurgery (%.1fs): J=%.6f stall=%d" % (elapsed, best_J, no_improve))

    return best_a, best_J


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def periodic_init(n, freq=3, seed=0):
    """Create a periodic-like initialization with some randomness."""
    rng = np.random.default_rng(seed)
    x = np.arange(n) / n
    # Multi-frequency pattern
    a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x) + 0.2 * np.sin(2 * np.pi * (freq + 1) * x)
    a += 0.1 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update the best sequence if a is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a_norm), elapsed))
    return best_seq, best_score

                best_seq = norm_sum1(parent)

    return [float(x) for x in best_seq]
