def fast_autocorr(a):

def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:

    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_shaping(a, n_iters=50, alpha=0.3, verbose=False):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a_cur = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a_cur).max()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        # Zero-pad and compute FFT
        fft_size = 1 << ((2 * n - 1) - 1).bit_length()
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a_cur
        A = np.fft.rfft(a_padded)
        power = np.abs(A) ** 2
        
        # Target: flatten the power spectrum
        target = np.mean(power)
        power_new = power + alpha * (target - power)
        power_new = np.maximum(power_new, 0)
        
        # Reconstruct with original phase
        A_new = np.sqrt(power_new) * np.exp(1j * np.angle(A))
        
        # Inverse FFT
        a_recon = np.fft.irfft(A_new, fft_size)[:n]
        a_recon = np.maximum(a_recon, 0.0)
        s = a_recon.sum()
        if s < 1e-15:
            break
        a_recon = a_recon / s
        
        m = fast_autocorr(a_recon).max()
        if m < best_m:
            best_m = m
            best_a = a_recon.copy()
            a_cur = a_recon
        else:
            # Reduce alpha if not improving
            alpha *= 0.9
        
        if verbose and it % 10 == 0:
            J = 2.0 * n * best_m
            print(f"    Spectral iter={it}: J={J:.6f}, alpha={alpha:.4f}")
    
    return best_a, best_m
