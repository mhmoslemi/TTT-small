def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_corr(f, y):
    """Compute z[j] = sum_m f[m] * y[j+m] for j in 0..n-1 where n=len(f)."""
    n = len(f)
    Ny = len(y)
    fft_size = 1 << (n + Ny - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    Y = np.fft.rfft(y, fft_size)
    # cross_corr[m] = sum_k f[k] * y[k+m]
    cc = np.fft.irfft(np.conj(F) * Y, fft_size)
    return cc[:n]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def mw_best_response(f, eta, inner_iters=30):
    """
    Find g ≈ argmax {sum(g) : g >= 0, conv(f,g) <= M}
    using multiplicative weights, where M = max(conv(f,f)).
    Returns g normalized to sum 1.
    """
    n = len(f)
    M = fast_autocorr(f).max()
    g = np.ones(n) / n  # Start uniform
    
    for _ in range(inner_iters):
        cross = fast_autocorr_combined(f, g)  # conv(f, g), length 2n-1
        y = np.maximum(cross - M, 0.0)
        if y.max() < 1e-12:
            break
        z = fast_cross_corr(f, y)
        log_update = -eta * z
        log_update -= log_update.max()
        log_update = np.clip(log_update, -50, 50)
        g = g * np.exp(log_update)
        g = np.maximum(g, 0.0)
        s = g.sum()
        if s < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / s
    return g

def fast_autocorr_combined(f, g):
    """Compute conv(f, g) using FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]

def line_search_mix(cf, cg, cgg):
    """
    Find t in [0,1] that minimizes max((1-t)^2*cf + 2t(1-t)*cg + t^2*cgg).
    """
    best_t = 0.0
    best_val = cf.max()
    
    # Coarse search
    for t in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]:
        combo = (1-t)**2 * cf + 2*t*(1-t) * cg + t**2 * cgg
        val = combo.max()
        if val < best_val - 1e-15:
            best_val = val
            best_t = t
    
    # Fine search around best_t
    for _ in range(5):
        dt = 0.01 * (0.5 ** _)
        for t_new in [best_t - dt, best_t + dt]:
            if 0 < t_new < 1:
                combo = (1-t_new)**2 * cf + 2*t_new*(1-t_new) * cg + t_new**2 * cgg
                val = combo.max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_t = t_new
    
    return best_t, best_val
