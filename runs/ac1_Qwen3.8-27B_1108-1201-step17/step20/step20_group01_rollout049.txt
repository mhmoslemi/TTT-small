def interp_to(a, new_n):

def build_lp_coeff(y, clist, n):
    K = len(clist)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(clist[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[ki - idx]
    return coeffs

def lp_fw_step(y, clist, n, solver):

    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1.0, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    gv = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    gs = gv.sum()
    if gs < 1e-15:
        return None
    return gv / gs

                ix = np.arange(s2, e2 + 1)
                pressure[ix] += a[kk_i - ix]
            pressure[src] = 1e15
            
            cold = np.argsort(pressure)[:40]
            for dst in cold:
                dst_i = int(dst)
                if dst_i == src:
                    continue
                for fr in fracs:
                    amt = sv * fr
                    if amt < 1e-18:
                        break
                    at = a.copy()
                    at[src] -= amt
                    at[dst_i] += amt
                    at = np.maximum(at, 0.0)
                    st = at.sum()
                    if st < 1e-15:
                        continue
                    at = at / st
                    mt = fast_conv(at).max()
                    if mt < best_m - 1e-16:
                        a = at
                        best_m = mt
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 200:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    fr = rng.uniform(0.002, 0.03)
                    amt = a[i] * fr
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        mn = fast_conv(a).max()
                        if mn < best_m:
                            best_m = mn
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 500 == 0 and stall > 0:
            print("    Coord t={:.1f}s J={:.8f} stall={}".format(
                time.monotonic() - t0, 2.0 * n * best_m, stall))
    
    return best_a, 2.0 * n * best_m

def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    t_budget = min(float(budget_s) - 45.0, 1740.0)
    solver = get_solver()
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = norm_seq(p)
            best_score = eval_wrap(best_seq)
            print("Parent: n={} score={}".format(len(p), repr(best_score)))
    except Exception:
        best_seq = None
    
    if best_seq is None:
        n_init = 8000
        x = np.linspace(0, 1, n_init)
        best_seq = norm_seq(np.exp(2.0 * x) * np.sin(np.pi * x) ** 0.5)
        best_score = eval_wrap(best_seq)
        print("Init: n={} score={}".format(n_init, repr(best_score)))
    
    n0 = len(best_seq)
    print("Starting: n={} score={:.8f}".format(n0, best_score))
    
    # Phase 1: Try multiple initializations at same length
    t_rem = t_budget - (time.monotonic() - t_start)
    t1 = min(20.0, t_rem * 0.03)
    if t1 > 5:
        print("\n=== P1: Shape search (n={}) ===".format(n0))
        sh, sh_sc = try_shapes(n0, t1)
        sh_real = eval_wrap(sh)
        if np.isfinite(sh_real) and sh_real < best_score:
            best_score = sh_real
            best_seq = sh
            print("  P1 improved: {:.8f}".format(best_score))
    
    # Phase 2: Smoothed mirror descent
    t_rem = t_budget - (time.monotonic() - t_start)
    t2 = min(180.0, t_rem * 0.12)
    if t2 > 30:
        print("\n=== P2: Smooth Mirror (n={}) ===".format(len(best_seq)))
        a_out, v_out = smooth_mirror(best_seq, t2, beta_start=5.0, beta_end=150.0,
                                      lr0=0.06, top_k_init=20, top_k_max=250, verbose=True)
        sc = eval_wrap(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: {:.8f}".format(best_score))
    
    # Phase 3: LP-FW main
    t_rem = t_budget - (time.monotonic() - t_start)
    t3 = min(450.0, t_rem * 0.30)
    if t3 > 60:
        print("\n=== P3: LP-FW main (n={}) ===".format(len(best_seq)))
        a_out, v_out = lp_fw(best_seq, t3, solver, init_K=150, max_K=4000, verbose=True)
        sc = eval_wrap(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 improved: {:.8f}".format(best_score))
    
    # Phase 4: Coordinate refinement
    t_rem = t_budget - (time.monotonic() - t_start)
    t4 = min(150.0, t_rem * 0.10)
    if t4 > 20:
        print("\n=== P4: Coord refine (n={}) ===".format(len(best_seq)))
        a_out, v_out = coord_refine(best_seq, t4, n_targets=12, seed=7, verbose=True)
        sc = eval_wrap(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 improved: {:.8f}".format(best_score))
    
    # Phase 5: Second LP-FW with larger active set
    t_rem = t_budget - (time.monotonic() - t_start)
    t5 = min(350.0, t_rem * 0.22)
    if t5 > 40:
        print("\n=== P5: LP-FW round 2 (n={}) ===".format(len(best_seq)))
        a_out, v_out = lp_fw(best_seq, t5, solver, init_K=300, max_K=6000, verbose=True)
        sc = eval_wrap(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P5 improved: {:.8f}".format(best_score))
    
    # Phase 6: Multi-length exploration
    t_rem = t_budget - (time.monotonic() - t_start)
    if t_rem > 100:
        print("\n=== P6: Multi-length ===")
        n_cur = len(best_seq)
        n_options = []
        for dn in [-2000, -1000, 1000, 2000, 3000]:
            nn = n_cur + dn
            if 2000 <= nn <= 20000 and nn != n_cur:
                n_options.append(nn)
        
        for nn in n_options[:3]:
            t_rem = t_budget - (time.monotonic() - t_start)
            if t_rem < 60:
                break
            t_nl = min(80.0, t_rem * 0.3)
            a_interp = interp_to(best_seq, nn)
            a_out, v_out = smooth_mirror(a_interp, t_nl, beta_start=20.0, beta_end=100.0,
                                          lr0=0.04, top_k_init=30, top_k_max=200, verbose=False)
            sc = eval_wrap(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P6 n={} improved: {:.8f}".format(nn, best_score))
    
    # Phase 7: Final coordinate polish
    t_rem = t_budget - (time.monotonic() - t_start)
    t7 = min(100.0, t_rem * 0.15)
    if t7 > 15:
        print("\n=== P7: Final coord (n={}) ===".format(len(best_seq)))
        a_out, v_out = coord_refine(best_seq, t7, n_targets=10, seed=999, verbose=False)
        sc = eval_wrap(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P7 improved: {:.8f}".format(best_score))
    
    # Phase 8: Final short LP-FW if time remains
    t_rem = t_budget - (time.monotonic() - t_start)
    if t_rem > 60:
        print("\n=== P8: Final LP-FW (n={}) ===".format(len(best_seq)))
        t8 = min(100.0, t_rem * 0.8)
        a_out, v_out = lp_fw(best_seq, t8, solver, init_K=400, max_K=7000, verbose=True)
        sc = eval_wrap(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P8 improved: {:.8f}".format(best_score))
    
    # Final validation
    if best_seq is not None:
        fs = eval_wrap(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print("\nFINAL: score={:.10f} n={} t={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_seq(np.exp(2.0 * x))
        best_score = eval_wrap(best_seq)
        print("\nFALLBACK: score={}".format(repr(best_score)))
    
    return [float(x) for x in best_seq]
