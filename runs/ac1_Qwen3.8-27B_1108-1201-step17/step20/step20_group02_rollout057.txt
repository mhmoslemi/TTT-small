def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def project_nonneg_sum(a, target_sum):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    a = np.asarray(a, dtype=np.float64)
    # Binary search for lambda
    lo = a.min() - abs(a).max() - 1
    hi = target_sum / len(a) + abs(a).max() + 1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.maximum(a - mid, 0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if abs(s - target_sum) < 1e-12:
            break
    
    projected = np.maximum(a - mid, 0)
    # Adjust for numerical error
    diff = target_sum - projected.sum()
    if abs(diff) > 1e-15 and projected.sum() > 0:
        projected += diff / len(projected)
        projected = np.maximum(projected, 0)
    return projected

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral gradient descent with log-sum-exp smooth approximation (FFT-based gradient)
2. LP-Frank-Wolfe with adaptive active set and warm starts
3. Direct peak-shaving heuristic
4. Multi-n exploration with interpolation
5. Coordinate descent polish
6. Nesterov-accelerated mirror descent as complementary optimizer

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _MODULE_SOLVER

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def hetero_convolve(a, b):
    size = len(a) + len(b) - 1
    L = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def project_nonneg_sum(a, target_sum):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    lo = a.min() - 1.0
    hi = target_sum / n + abs(a).max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.maximum(a - mid, 0.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.maximum(a - (lo + hi) / 2.0, 0.0)
    s = proj.sum()
    if s > 1e-15:
        proj *= target_sum / s
    return proj

def compute_lse_gradient(a, p, beta):
    """Compute gradient of LSE(beta * conv(a,a)) at a, given weights p over convolution lags."""
    n = len(a)
    # grad_raw[i] = sum_{j=0}^{n-1} a[j] * p[i+j]
    # Using FFT: conv_full = conv(a, p_rev), grad_raw[i] = conv_full[2n-2-i]
    p_rev = p[::-1]
    conv_full = hetero_convolve(a, p_rev)
    grad_raw = conv_full[2 * n - 2 : n - 2 : -1]
    return 2.0 * beta * grad_raw

def spectral_gd(a0, time_limit, max_iter=100000, beta_start=30.0, beta_end=800.0, lr_start=0.008, verbose=True):
    """Spectral gradient descent with log-sum-exp smoothing."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    ac = fast_convolve(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    vel = np.zeros(n)
    momentum = 0.85
    stall = 0
    it = 0

    while it < max_iter and time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / (max_iter * 0.7))
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.7)
        lr = lr_start / (1.0 + 0.005 * it)

        ac = fast_convolve(a)
        M = ac.max()

        if M < best_m - 1e-13:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            pert = 0.02 * rng.standard_normal(n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            vel = np.zeros(n)
            momentum = 0.8
            stall = 0
            continue

        # Compute LSE weights
        ac_shift = ac - ac.max()
        log_w = beta * ac_shift
        p = np.exp(log_w - log_w.max())
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(len(ac)) / len(ac)
        else:
            p = p / p_sum

        # Compute gradient
        grad = compute_lse_gradient(a, p, beta)

        # Nesterov momentum
        vel = momentum * vel + grad
        a_new = a - lr * vel
        a_new = project_nonneg_sum(a_new, 1.0)
        a = a_new

        if verbose and it % 5000 == 0:
            ac_check = fast_convolve(a)
            elapsed = time.monotonic() - t0
            print("  SpectralGD it=" + str(it) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2.0 * n * ac_check.max()))

    if verbose:
        elapsed = time.monotonic() - t0
        print("  SpectralGD done (" + str(round(elapsed, 1)) + "s) J=" + repr(2.0 * n * best_m))
    return best_a, 2.0 * n * best_m

def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        ki = int(constraint_list[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[ki - idx]
    return coeffs

def lp_fw(a0, time_limit, solver, initial_K=300, max_K=3000, add_per_iter=40, verbose=True):
    """LP-Frank-Wolfe with adaptive active set."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 500
    iter_count = 0
    stagnation = 0
    prev_M = None

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation > 5:
                break
            ac_curr = fast_convolve(y)
            n_pert = max(30, len(active_set) // 4)
            rng_local = np.random.default_rng(stagnation * 1000)
            keep_idx = rng_local.choice(len(active_set), size=len(active_set) - n_pert, replace=False)
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 3]
            new_set = set(int(active_set[i]) for i in keep_idx)
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
            no_improve = 0
            stagnation += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation = 0
        else:
            no_improve += 1

        # Adaptive active set management
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[int(k)] for k in active_set])
            keep_count = int(max_K * 0.6)
            keep_idx = np.argsort(margins)[:keep_count]
            active_set_base = [active_set[i] for i in keep_idx]
            top_peaks = np.argsort(ac)[::-1][:100]
            for v in top_peaks:
                if int(v) not in set(active_set_base):
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))[:max_K]

        K = len(active_set)
        if K < 5:
            active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
            K = len(active_set)

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations and add to active set
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(viol) > 0:
            viol_amt = conv_ys[viol] - M_curr
            top_v = viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(int(k) for k in active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Backtracking line search
        best_alpha = 0.1
        best_trial_M = fast_convolve(y).max()
        alphas = [0.9, 0.7, 0.5, 0.3, 0.15, 0.1, 0.05, 0.02,
                  max(0.02, 2.0 / (iter_count + 3)), max(0.01, 1.0 / (iter_count + 1))]
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 3:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("  LP-FW it=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2.0 * n * best_m) + " |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("  LP-FW done (" + str(round(elapsed, 1)) + "s) J=" + repr(2.0 * n * best_m) + " iters=" + str(iter_count))
    return best_a, 2.0 * n * best_m

def peak_shave(a0, time_limit, verbose=True):
    """Direct peak shaving: transfer mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(55)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 400:
        ac = fast_convolve(a)
        k_peak = int(np.argmax(ac))

        # Find contributors to the peak
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        idx = np.arange(i_start, i_end)
        if len(idx) == 0:
            break
        contrib = a[idx] * a[k_peak - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-20:
            break

        # Top contributors
        top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

        # Find low-impact positions (positions that don't contribute much to top peaks)
        top_peaks = np.argsort(ac)[::-1][:10]
        impact = np.zeros(n)
        for kp in top_peaks[:6]:
            kp_i = int(kp)
            s = max(0, kp_i - n + 1)
            e = min(n, kp_i + 1)
            if s < e:
                ii = np.arange(s, e)
                impact[ii] += a[kp_i - ii]

        # Candidates: positions with low impact and some mass available
        candidates = np.where((impact < np.percentile(impact, 20)) & (a > 1e-8))[0]
        if len(candidates) == 0:
            candidates = np.ndarray.argsort(impact)[:n // 4]

        improved = False
        for src in top_contrib_idx[:3]:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-10:
                continue
            # Pick destination from candidates
            for _ in range(5):
                dst = int(rng.integers(len(candidates)))
                dst_i = candidates[dst]
                if dst_i == src_i:
                    continue
                for frac in [0.05, 0.03, 0.02, 0.01, 0.005]:
                    amt = a[src_i] * frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_convolve(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                    else:
                        a = a_trial  # roll back
            if not improved:
                pass

        if not improved:
            stall += 1
            # Random perturbation
            if stall % 100 == 0:
                noise = rng.normal(0, 0.01, n) * best_a
                a_p = best_a + noise
                a_p = np.maximum(a_p, 1e-12)
                a = a_p / a_p.sum()

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            print("  PeakShave (" + str(round(elapsed, 1)) + "s) J=" + repr(2.0 * n * best_m) + " stall=" + str(stall))

    if verbose:
        elapsed = time.monotonic() - t0
        print("  PeakShave done (" + str(round(elapsed, 1)) + "s) J=" + repr(2.0 * n * best_m))
    return best_a, 2.0 * n * best_m

def mirror_descent(a0, time_limit, verbose=False):
    """Mirror descent (softmax) with subgradient of max-conv."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(33)
    eta0 = 0.02
    momentum = 0.7
    vel = np.zeros(n)
    stall = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 4000:
            break

        eta = eta0 / (np.sqrt(t) + 1.0)

        ac = fast_convolve(a)
        k_max = int(np.argmax(ac))

        # Subgradient at the max
        grad = np.zeros(n)
        i_start = max(0, k_max - n + 1)
        i_end = min(n, k_max + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] = 2.0 * a[k_max - idx]
            # Correct the diagonal
            mid = k_max // 2
            if i_start <= mid < i_end and k_max % 2 == 0:
                grad[mid] -= a[mid]

        # Small random perturbation
        grad += 0.005 * rng.standard_normal(n) * a

        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        vel = momentum * vel + (a_new - a)
        a = a_new

        M_new = fast_convolve(a).max()
        if M_new < best_m - 1e-13:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 800 and stall % 800 == 0:
            noise = rng.normal(0, 0.02, n) * best_a
            a_p = best_a + noise
            a_p = np.maximum(a_p, 1e-12)
            a = a_p / a_p.sum()
            vel = np.zeros(n)
            eta0 *= 1.3

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("  MirrorDesc t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2.0 * n * best_m))

    if verbose:
        elapsed = time.monotonic() - t0
        print("  MirrorDesc done (" + str(round(elapsed, 1)) + "s) J=" + repr(2.0 * n * best_m))
    return best_a, 2.0 * n * best_m

def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent polish: transfer mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(44)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        top_k = np.argsort(ac)[::-1][:20]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            # Impact map
            impact = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    ii = np.arange(s2, e2 + 1)
                    impact[ii] += a[k2i - ii]

            for src_pos in idx[np.argsort(contrib)[::-1][:3]]:
                src_i = int(src_pos)
                if a[src_i] < 1e-12:
                    continue
                # Find best destination
                low_impact = np.where(impact < np.percentile(impact, 15))[0]
                if len(low_impact) == 0:
                    low_impact = np.argsort(impact)[:n // 5]

                for _ in range(3):
                    dst_i = int(rng.integers(len(low_impact)))
                    dst = low_impact[dst_i]
                    if dst == src_i:
                        continue
                    for frac in [0.02, 0.01, 0.005, 0.002]:
                        amt = a[src_i] * frac
                        a_trial = a.copy()
                        a_trial[src_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_convolve(a_trial).max()
                        if M_trial < best_m - 1e-14:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

        if not improved:
            stall += 1
            if stall % 100 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            print("  CoordPolish (" + str(round(elapsed, 1)) + "s) J=" + repr(2.0 * n * best_m) + " stall=" + str(stall))

    return best_a, best_m

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

def get_initial_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + ", score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass
    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence()
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Spectral GD (main optimizer, different from previous) ===
    print("\n=== Phase 1: Spectral GD (n=" + str(n0) + ") ===")
    t1 = min(remaining_time() * 0.25, 350.0)
    if t1 > 50:
        a_out, _ = spectral_gd(norm_sum1(best_seq), t1, max_iter=100000, beta_start=30.0, beta_end=600.0, lr_start=0.008, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral_n" + str(n0), t_global)

    # === Phase 2: LP-FW for peak reduction ===
    print("\n=== Phase 2: LP-FW (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining_time() * 0.20, 280.0)
    if t2 > 40 and len(best_seq) <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t2, solver, initial_K=400, max_K=3500, add_per_iter=50, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n" + str(len(best_seq)), t_global)

    # === Phase 3: Peak shaving ===
    print("\n=== Phase 3: Peak Shave (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.08, 100.0)
    if t3 > 20:
        a_out, _ = peak_shave(norm_sum1(best_seq), t3, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakshave_n" + str(len(best_seq)), t_global)

    # === Phase 4: Mirror descent ===
    print("\n=== Phase 4: Mirror Descent (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining_time() * 0.08, 100.0)
    if t4 > 20:
        a_out, _ = mirror_descent(norm_sum1(best_seq), t4, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "mirror_n" + str(len(best_seq)), t_global)

    # === Phase 5: LP-FW second pass with different params ===
    print("\n=== Phase 5: LP-FW v2 (n=" + str(len(best_seq)) + ") ===")
    t5 = min(remaining_time() * 0.18, 250.0)
    if t5 > 40 and len(best_seq) <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t5, solver, initial_K=500, max_K=4000, add_per_iter=60, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(len(best_seq)), t_global)

    # === Phase 6: Multi-n exploration ===
    print("\n=== Phase 6: Multi-n ===")
    n_best = len(best_seq)
    candidate_ns = []
    for delta in [-600, -300, 300, 600, 1000, -1000]:
        nn = n_best + delta
        if 3000 <= nn <= 10000 and nn != n_best:
            candidate_ns.append(nn)
    candidate_ns = sorted(set(candidate_ns), key=lambda x: abs(x - n_best))

    for new_n in candidate_ns[:3]:
        rt = remaining_time()
        if rt < 100:
            break
        print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_interp = interpolate_seq(best_seq, new_n)

        # Spectral GD on new n
        t6 = min(rt * 0.15, 120.0)
        if t6 > 25:
            a_out, _ = spectral_gd(a_interp, t6, max_iter=50000, beta_start=40.0, beta_end=500.0, lr_start=0.006, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral_n" + str(new_n), t_global)

        # Quick LP-FW
        rt = remaining_time()
        if rt > 60 and len(best_seq) <= 9000:
            t6b = min(rt * 0.12, 100.0)
            if t6b > 25:
                a_out, _ = lp_fw(norm_sum1(best_seq), t6b, solver, initial_K=300, max_K=2500, add_per_iter=40, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # === Phase 7: Spectral GD with different params ===
    print("\n=== Phase 7: Spectral GD v2 (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining_time() * 0.12, 150.0)
    if t7 > 30:
        a_out, _ = spectral_gd(norm_sum1(best_seq), t7, max_iter=80000, beta_start=100.0, beta_end=2000.0, lr_start=0.004, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral2_n" + str(len(best_seq)), t_global)

    # === Phase 8: Final LP-FW ===
    print("\n=== Phase 8: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining_time() * 0.12, 150.0)
    if t8 > 30 and len(best_seq) <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t8, solver, initial_K=600, max_K=5000, add_per_iter=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Phase 9: Coordinate polish ===
    print("\n=== Phase 9: Coordinate Polish (n=" + str(len(best_seq)) + ") ===")
    t9 = min(remaining_time() * 0.10, 100.0)
    if t9 > 15:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t9, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # === Final validation ===
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
