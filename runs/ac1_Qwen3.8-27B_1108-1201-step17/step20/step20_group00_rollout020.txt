Define the smoothed objective: `f_T(a) = T * log(sum_k exp(conv(a,a)[k] / T))`

`grad[i] = sum_k w_k * a[k-i]` where `w_k = exp(conv(a,a)[k]/T) / sum_l exp(conv(a,a)[l]/T)`

This can be computed efficiently:
1. Compute ac = conv(a,a) via FFT: O(n log n)
2. Compute weights w = softmax(ac / T): O(n)
3. Compute grad = conv(w, a): O(n log n) — wait, this is a cross-correlation, not autocorrelation.

Actually, `grad[i] = sum_k w_k * a[k-i]` (for valid k-i). This is the cross-correlation of w and a, which we can compute via FFT:
`grad = ifft(fft(w) * conj(fft(a)))` — need to be careful about the exact formula.

Actually, `(w * a)[i] = sum_k w[k] * a[i-k]` is the convolution. But we want `sum_k w[k] * a[k-i]` which is the cross-correlation = `conv(w, flip(a))[i]`.

Or equivalently: `grad = convolve(w, a_reversed)` reversed, or we can use the FFT formula for cross-correlation.

Let me use numpy: `np.correlate(w, a, mode='full')` but this is O(n^2). Better to use FFT.

Cross-correlation via FFT:
`correlation = ifft(fft(w, n=2N) * conj(fft(a, n=2N)))`

This gives the circular cross-correlation, which for zero-padded sequences equals the linear cross-correlation.

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix

            prob.solve(solver=s, verbose=False)

                print("  Using solver: " + str(s))
                return s
        except Exception:
            continue
    print("  Fallback to SCS")

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_xcorr(w, a):
    """Cross-correlation: result[i] = sum_k w[k] * a[k-i] (linear)"""
    n = max(len(w), len(a))
    size = len(w) + len(a) - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, n=fft_size)
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(W * np.conj(A), fft_size)[:size]

    s = a.sum()

def smooth_gradient_descent(a0, time_limit, T_start=None, T_min=1e-6, 
                            eta_max=1.0, eta_min=1e-6, verbose=True):
    """
    Multiplicative (mirror) gradient descent on the log-sum-exp smoothed objective.
    f_T(a) = T * log(sum_k exp(ac[k]/T)) where ac = conv(a,a), sum(a)=1.
    grad[i] = sum_k w[k] * a[k-i] where w = softmax(ac/T)
    Update: a_new = a * exp(-eta * grad), then normalize.
    """
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = objective(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    ac = fast_autocorr(x)
    peak = ac.max()
    
    if T_start is None:
        T = peak / 3.0
    else:
        T = T_start
    
    stall = 0
    eta = eta_max
    n_iter = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 50000:
            break
        
        n_iter += 1
        # Compute softmax weights
        # ac / T could have large values, use stable softmax
        ac_scaled = ac / T
        ac_max = ac_scaled.max()
        w = np.exp(ac_scaled - ac_max)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(ac)
            w[ac.argmax()] = 1.0
        else:
            w = w / w_sum
        
        # Compute gradient: grad[i] = sum_k w[k] * a[k-i] = xcorr(w, x)
        # xcorr(w, x)[i] = sum_k w[k] * x[k-i]
        # Using FFT: ifft(FFT(w) * conj(FFT(x)))
        Wf = np.fft.rfft(w, n=fft_size)
        Xf = np.fft.rfft(x, n=fft_size)
        grad = np.fft.irfft(Wf * np.conj(Xf), fft_size)[:n]
        # Clamp gradient to be non-negative (it should be since w>=0, x>=0)
        grad = np.maximum(grad, 0.0)
        
        # Normalize gradient so that its max is 1 (for stable updates)
        gmax = grad.max()

            x_trial = np.maximum(x_trial, 1e-30)

            if s < 1e-15:

                # Accept

                if val < best_val:
                    best_val = val
                    best_a = x.copy()
                stall = 0
                improved = True
                break
            else:
                eta *= 0.7
        
        if not improved:
            stall += 1
            if stall > 100:
                eta = max(eta * 0.9, eta_min)
        
        # Decrease temperature
        T = max(T * 0.9995, T_min)
        
        # Periodic restart
        if stall > 3000 and stall % 3000 == 0:
            noise_scale = min(0.05, 0.005 * (stall / 3000.0))
            noise = rng.standard_normal(n) * noise_scale * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-15)
            x = x_r / x_r.sum()
            stall = 0
            eta = eta_max * 0.5

            print("    SGD t=" + str(t) + " (" + str(round(elapsed,1)) + "s) val=" + repr(best_val) + " T=" + repr(T) + " eta=" + repr(eta))

def coord_descent(a0, time_limit, verbose=True):
    """Vectorized coordinate descent: find best pairwise mass moves."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for iteration in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 2000:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Top 15 peak lags
        top_k = np.argsort(ac)[::-1][:15]
        found = False
        
        for k in top_k:
            ki = int(k)
            # For lag ki, contrib[i] = a[i] * a[ki-i] for valid i
            i_s = max(0, ki - n + 1)
            i_e = min(n - 1, ki)
            idx = np.arange(i_s, i_e + 1)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Top 5 contributing sources
            top_srcs = idx[np.argsort(contrib)[::-1][:5]]
            
            for src in top_srcs:
                src = int(src)
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                # Score: how much would moving mass from src to dst increase peaks?
                # Moving delta from src to dst changes ac[k] by:
                #   -delta * a[k-src] + delta * a[k-dst] (approximately, for small delta)
                # More precisely, for the target lag ki:
                #   change in ac[ki] = -delta*a[ki-src] + delta*a[ki-dst] (if valid)
                
                # Compute score: sum over top lags of the increase
                scores = np.zeros(n)
                for k2 in top_k[:10]:
                    k2i = int(k2)
                    # Change in ac[k2] if we move delta from src to dst:
                    # ac[k2] gets -delta * a[k2-src] (if 0<=k2-src<n)
                    #           + delta * a[k2-dst] (if 0<=k2-dst<n)
                    # plus cross terms (delta^2) which we ignore
                    # So the net change for dst is: sum_k2 [a[k2-dst] - a[k2-src]] * delta
                    # We want to minimize the max, so score[dst] = max over k2 of (a[k2-dst])
                    # (since a[k2-src] is fixed for a given src)
                    ks = max(0, k2i - n + 1)
                    ke = min(n - 1, k2i)
                    for i in range(ks, ke + 1):
                        dst_candidate = k2i - i
                        if 0 <= dst_candidate < n and dst_candidate != src:
                            scores[dst_candidate] += a[i]
                
                scores[src] = 1e10
                
                # Try best destinations
                candidates = np.argsort(scores)[:20]
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.001, 0.003, 0.005, 0.01]:
                        delta = src_val * frac
                        a_t = a.copy()
                        a_t[src] -= delta
                        a_t[dst] += delta
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < M_curr - 1e-14:
                            a = a_t
                            found = True
                            break
                    if found:
                        break
                
                if found:
                    break
            
            if found:
                break
        
        if not found:
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.005)
                delta = a[i] * frac
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    Mt = fast_autocorr(a).max()
                    if Mt < M_curr:
                        stall = 0
                        found = True
    
    if verbose:
        print("    CoordDescent done (" + str(round(time.monotonic()-t0,1)) + "s) val=" + repr(2*n*best_m))
    return best_a, 2.0 * n * best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.1, 1.0 - 1.0/old_n, old_n)  # avoid endpoints
    new_x = np.linspace(0.1, 1.0 - 1.0/new_n, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-30:
        return np.ones(new_n) / new_n
    return f_interp / ss
