    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def make_geometric(n, delta):
    """Create a geometric sequence a[i] = c * (1-delta)^i normalized to sum=1."""
    r = 1.0 - delta
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_power_law(n, alpha, r):
    """a[i] = (i+1)^(-alpha) * r^i, normalized."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + 1.0, -alpha) * np.power(r, i)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_sine(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)


def make_bump(n, center_frac, width):
    """Asymmetric bump: peak at center_frac*n, Gaussian width."""
    i = np.arange(n, dtype=np.float64)
    c = center_frac * n
    a = np.exp(-((i - c) ** 2) / (2.0 * width * width))
    return norm_sum1(a)


def compute_hinge_grad(y, top_k, temperature=1.0):
    """
    Compute a hinge/smoothed-max gradient targeting the top_k autocorrelation peaks.
    Returns (grad, b, max_b).
    """
    n = len(y)
    b = fast_autocorr(y)
    b_max = b.max()

    # Select top-k peaks
    if top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
    else:
        idx = np.arange(len(b))

    # Softmax weights over top peaks (sharper = more focus on worst)
    vals = b[idx]
    shifted = (vals - vals.max()) / temperature
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()

    # Gradient: for each peak at position k, d(b[k])/d(a[i]) = a[k-i] (for valid range)
    grad = np.zeros(n, dtype=np.float64)
    for jj, k in enumerate(idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx_i = np.arange(i_start, i_end)
            grad[idx_i] += w[jj] * y[ki - idx_i]

    return grad, b, b_max


def fista_adaptive(a0, time_limit, beta_start=10.0, beta_end=1000.0,
                   beta_growth=1.002, top_k_start=1, top_k_end=30,
                   top_k_growth=1.0005, restart_every=3000,
                   temp_decay=0.999, verbose=True, seed=42):
    """
    FISTA with adaptive top-k and temperature for hinge gradient.
    top_k anneals from small (focus on worst peak) to large (flatten spectrum).
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    top_k = top_k_start
    temperature = 1.0
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b, b_max = compute_hinge_grad(y, top_k, temperature)

        # Adaptive step size
        eta = 1.0 / (beta_start * (1.0 + 0.01 * t))
        eta = min(eta, 0.5 / (1.0 + np.linalg.norm(grad, ord=np.inf) if np.linalg.norm(grad, ord=np.inf) > 1e-10 else 1.0))
        eta = max(eta, 1e-8)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Anneal top_k and temperature
        if t % 50 == 0:
            top_k = min(int(top_k * top_k_growth), top_k_end)
            temperature = max(temperature * temp_decay, 0.01)

        # Restart
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            top_k = max(top_k_start, top_k // 2)
            temperature = max(temperature * 2.0, 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t={} ({}s): J={}".format(t, round(elapsed, 1), round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t={} ({}s): J={} top_k={} temp={}".format(
                t, round(elapsed, 1), round(J, 6), top_k, round(temperature, 4)))

    return best_a, best_max_b


def peak_targeted_local_search(a0, time_limit, verbose=True, seed=99):
    """
    Local search that specifically targets the top autocorrelation peaks.
    For each peak, identifies the contributing pairs and redistributes mass.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        # Get top peaks
        top_peaks = np.argsort(ac)[::-1][:20]
        M_curr = ac.max()
        improved = False

        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick a peak to target
            k_int = int(top_peaks[rng.integers(min(10, len(top_peaks)))])
            js = max(0, k_int - n + 1)
            je = min(n, k_int + 1)
            if js >= je:
                continue
            idx = np.arange(js, je)
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find the element contributing most to this peak
            src = int(idx[np.argmax(contrib)])

            # Find a good destination: low contribution to all top peaks
            in_peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                in_peak_mask[s2:e2] = True

            candidates = np.where(~in_peak_mask)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src or a[src] < 1e-12:
                continue

            # Try several transfer fractions
            for frac in [0.001, 0.003, 0.01, 0.03, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.001, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSearch ({}s): J={} stall={}".format(round(elapsed, 1), round(J, 6), stall))

    return best_a, best_m


def cyclic_shift_best(a0):
    """Find the best cyclic shift."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_flatten(a0, time_limit, n_iters=200, verbose=True):
    """
    Iterate in frequency domain to flatten the power spectrum,
    which reduces peak autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        power = np.maximum(power, 1e-30)
        # Flatten: reduce high-power frequencies
        p_max = power.max()
        if p_max < 1e-15:
            break
        # Gentle flattening
        A_new = A * np.sqrt(1.0 / (1.0 + power / (p_max * 0.5)))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 50 == 0:
            J = 2.0 * n * best_m
            print("    Spectral it={}: J={}".format(it, round(J, 6)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {} (n={}, t={:.1f}s)".format(
            label, repr(round(sc, 6)), len(a), elapsed))
    return best_seq, best_score

            parent = norm_sum1(p)
            inits['parent'] = parent
            sc = evaluate_safe(parent)
            if np.isfinite(sc):
                best_seq = parent.copy()
                best_score = sc
                print("Parent: n={}, score={}".format(len(parent), repr(round(sc, 6))))

        pass
