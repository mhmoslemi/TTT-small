def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_lse_grad_and_obj(y, beta, top_k=None):
    """
    Compute (1/beta)*log(sum(exp(beta*b))) and its gradient, where b = convolve(y,y).
    Returns (objective_value, gradient, b, b_max).
    Uses FFT for the gradient computation.
    """
    n = len(y)
    size = 2 * n - 1
    b = fast_autocorr(y)
    b_max = b.max()

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
    else:
        b_use = b.copy()

    # Log-sum-exp objective (shifted for numerical stability)
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
        obj_val = b_max
    else:
        p = exp_vals / s
        obj_val = b_max + np.log(s) / beta

    # Gradient: grad[j] = 2 * sum_i p[j+i] * y[i]
    # = 2 * cross_correlation(p, y)[j]
    # Computed via FFT: ifft(fft(y) * conj(fft(p)))
    L = 1 << (size + n - 1).bit_length()  # >= size + n - 1
    Yf = np.fft.rfft(y, L)
    Pf = np.fft.rfft(p, L)
    corr = np.fft.irfft(Yf * np.conj(Pf), L)
    grad = 2.0 * corr[:n]

    return obj_val, grad, b, b_max


def nesterov_lse(a0, time_limit, beta_start=50.0, beta_end=2000.0,
                 top_k_ratio_start=0.10, top_k_ratio_end=0.005,
                 eta_init=0.8, restart_every=800, verbose=True, seed=42):
    """
    Nesterov-accelerated gradient descent on log-sum-exp smoothed max autocorrelation.
    Adaptive beta (cosine annealing), top-k focusing, backtracking step size.
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    t_nest = 1.0
    eta = eta_init
    stall = 0
    rng = np.random.default_rng(seed)
    total_iters = max(1, int(time_limit / 0.05))  # rough estimate

    for t in range(1, total_iters + 100):
        if time.monotonic() - t0 > time_limit:
            break

        # Cosine annealing for beta and top_k
        progress = min(1.0, t / total_iters)
        cos_val = 0.5 * (1 + np.cos(np.pi * progress))
        beta = beta_start + (beta_end - beta_start) * (1 - cos_val)
        top_k_ratio = top_k_ratio_start + (top_k_ratio_end - top_k_ratio_start) * progress
        top_k = max(5, int(n * top_k_ratio))

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute gradient
        obj_val, grad, b, b_max = compute_lse_grad_and_obj(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x.copy()
            max_new = b_max
        else:
            # Backtracking line search
            eta_trial = eta
            y_max = fast_autocorr(y).max()
            found = False
            x_new = x.copy()
            max_new = y_max
            for _ in range(8):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                max_new = fast_autocorr(x_new).max()
                if max_new <= y_max + 1e-12:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                eta = max(eta * 0.7, 0.01)
                x_new = x
                max_new = fast_autocorr(x).max()

        # Update best
        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 3.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Restart if stalled
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                noise = 0.02 * rng.standard_normal(n)
                x_r = best_a + noise * best_a
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.2)
            elif strategy == 2:
                x_r = best_a * np.exp(0.04 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= rng.uniform(0.5, 0.8, mask.sum())
                x_r[~mask] *= rng.uniform(1.2, 1.5, (~mask).sum())
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 15)))
                x_r = np.roll(best_a, shift)
            else:
                # Flatten peaks: reduce top contributors, boost bottom
                ac = fast_autocorr(best_a)
                top_pos = np.argsort(ac)[::-1][:max(3, n // 100)]
                x_r = best_a.copy()
                for k in top_pos:
                    ki = int(k)
                    lo = max(0, ki - n + 1)
                    hi = min(n - 1, ki)
                    for ii in range(lo, hi + 1):
                        x_r[ii] *= 0.95
                x_r += 0.05 * best_a
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            eta = eta_init

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    NLSE t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" +
                  str(round(J, 6)) + " beta=" + str(int(beta)) + " top_k=" + str(top_k))

    return best_a, best_max_b


def peak_peeling(a0, time_limit, verbose=True, seed=555):
    """
    Direct peak elimination: repeatedly find the max autocorrelation peak,
    identify contributing pairs, and redistribute mass to reduce it.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        peak_idx = int(np.argmax(ac))
        peak_val = ac[peak_idx]

        if peak_val < best_m * 0.999:
            break

        # Find contributing pairs for this peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        idx = np.arange(lo, hi + 1)
        partners = peak_idx - idx
        contrib = a[idx] * a[partners]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            break

        # Identify the top contributors
        top_contrib_count = max(2, int(0.1 * len(idx)))
        top_local = np.argsort(contrib)[::-1][:top_contrib_count]

        # Redistribute: reduce mass at top contributors, add to random other positions
        reduction = 0.02 * total_contrib
        for li in top_local:
            ii = int(idx[li])
            partner = int(partners[li])
            amt = reduction * contrib[li] / total_contrib
            if amt > a[ii] * 0.5:
                amt = a[ii] * 0.3
            if amt < 1e-15:
                continue
            a[ii] -= amt
            # Add to a random position far from current peaks
            for _ in range(3):
                dst = int(rng.integers(n))
                # Check if this reduces other peaks
                a_trial = a.copy()
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < peak_val:
                    a = a_trial
                    break
            else:
                a[lo + int(rng.integers(min(5, n - lo)))] += amt

        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            break
        a = a / s_a

        m_curr = fast_autocorr(a).max()
        if m_curr < best_m:
            best_m = m_curr
            best_a = a.copy()
        elif verbose and (int((time.monotonic() - t0) * 10) % 20 == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakPeel (" + str(round(elapsed, 1)) + "s): J=" + str(round(J, 6)))

        # Periodic random perturbation to escape local minima
        if rng.random() < 0.05:
            pert = 0.01 * rng.standard_normal(n) * a
            a_trial = a + pert
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                if fast_autocorr(a_trial).max() < best_m:
                    a = a_trial
                    best_m = fast_autocorr(a).max()
                    best_a = a.copy()

    return best_a, best_m


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000, verbose=True):
    """LP-Frank-Wolfe for structured improvement."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 500

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check for new violations
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:20]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        for alpha in [best_alpha, best_alpha * 0.5, 0.05, 0.1, 0.15, 0.2]:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = y.copy()
        else:
            y = y / s_y

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" +
                  str(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def spectral_init(n, shape='cosine'):
    """Create spectral-domain-based initializations."""
    if shape == 'cosine':
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif shape == 'sinc':
        x = np.linspace(-1, 1, n)
        a = (np.sinc(0.8 * x) ** 2)
        a = np.maximum(a, 0)
    elif shape == 'optimal':
        # Try to design a sequence with flat |FFT|^2
        # Start with random phases, optimize amplitudes
        a = np.ones(n)
        a[:n//4] = 0.5
        a[-(n//4):] = 0.5
    else:
        a = np.ones(n)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) +
              " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv)/(sum)^2.
    Uses log-sum-exp Nesterov, peak peeling, LP-FW, and spectral methods.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(round(time_budget, 1)) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        best_seq = spectral_init(n_base, 'cosine')
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + repr(round(best_score, 6)))

    # === Phase 1: Main Nesterov optimization on parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: NLSE main (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_out, _ = nesterov_lse(norm_sum1(best_seq), t1,
                                beta_start=30.0, beta_end=1500.0,
                                top_k_ratio_start=0.15, top_k_ratio_end=0.003,
                                eta_init=0.8, restart_every=600,
                                verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nlse1", t_global)

    # === Phase 2: Peak peeling ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P2: PeakPeel (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 15:
        a_out, _ = peak_peeling(norm_sum1(best_seq), t2, verbose=True, seed=555)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peel1", t_global)

    # === Phase 3: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: LP-FW (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20 and n_best <= 10000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t3, solver,
                         initial_K=400, max_K=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # === Phase 4: Second NLSE with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: NLSE round 2 (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 25:
        a_out, _ = nesterov_lse(norm_sum1(best_seq), t4,
                                beta_start=100.0, beta_end=3000.0,
                                top_k_ratio_start=0.05, top_k_ratio_end=0.002,
                                eta_init=0.4, restart_every=500,
                                verbose=True, seed=999)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nlse2", t_global)

    # === Phase 5: Longer sequence exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: Longer sequences ===")

    candidate_lengths = []
    for delta in [500, 1000, 2000]:
        new_n = n_best + delta
        if new_n <= 12000:
            candidate_lengths.append(new_n)

    for new_n in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("  -> n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.35, 150.0)
        if t5a > 20:
            a_out, _ = nesterov_lse(a_long, t5a,
                                    beta_start=50.0, beta_end=1200.0,
                                    top_k_ratio_start=0.10, top_k_ratio_end=0.003,
                                    eta_init=0.6, restart_every=500,
                                    verbose=True, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nlse_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.20, 80.0)
            if t5b > 10:
                a_out, _ = peak_peeling(norm_sum1(best_seq), t5b, verbose=False, seed=new_n % 3163)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "peel_n" + str(new_n), t_global)

    # === Phase 6: Alternating NLSE + PeakPeel on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P6: Alternating NLSE+Peel (n=" + str(len(best_seq)) + ") ===")
    round_num = 0
    while remaining > 40:
        round_num += 1
        t6a = min(remaining * 0.40, 120.0)
        if t6a > 15:
            a_out, _ = nesterov_lse(norm_sum1(best_seq), t6a,
                                    beta_start=200 + 100 * round_num,
                                    beta_end=3000 + 500 * round_num,
                                    top_k_ratio_start=0.03, top_k_ratio_end=0.001,
                                    eta_init=0.3, restart_every=400,
                                    verbose=True, seed=1000 + round_num * 7)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "alt_nlse_" + str(round_num), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.30, 80.0)
            if t6b > 10:
                a_out, _ = peak_peeling(norm_sum1(best_seq), t6b, verbose=True, seed=2000 + round_num * 11)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "alt_peel_" + str(round_num), t_global)

        remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: Final LP (n=" + str(n_best) + ") ===")
    if remaining > 20 and n_best <= 10000:
        t7 = min(remaining * 0.60, 150.0)
        if t7 > 10:
            a_out, _ = lp_fw(norm_sum1(best_seq), t7, solver,
                             initial_K=500, max_K=3000, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # === Phase 8: Final peak peeling ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t8 = min(remaining * 0.80, 100.0)
        if t8 > 5:
            a_out, _ = peak_peeling(norm_sum1(best_seq), t8, verbose=True, seed=7777)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peel_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) +
              ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        best_seq = spectral_init(n_fb, 'cosine')
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
