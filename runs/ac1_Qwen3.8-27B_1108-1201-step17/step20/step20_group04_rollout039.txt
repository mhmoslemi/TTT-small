import numpy as np
import time
import cvxpy as cp

def solve_lp_direction(f, M, top_k_positions):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in top_k_positions"""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Objective: maximize sum(g)
    objective = cp.Maximize(cp.sum(g))
    
    constraints = [g >= 0]
    
    # Convolution constraints
    for k in top_k_positions:
        # (f*g)[k] = sum_{j} f[j] * g[k-j] where k-j in [0, n-1]
        # = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        # (f*g)[k] = sum_{j=j_min}^{j_max} f[j] * g[k-j]
        # Let i = k-j, so i ranges from k-j_max to k-j_min
        # = sum_{i=k-j_max}^{k-j_min} f[k-i] * g[i]
        # So the constraint is: f[k-idx] @ g[idx] <= M
        # where idx ranges over valid values
        
        # More directly: (f*g)[k] = sum_{i=0}^{n-1} f[k-i]*g[i] (with f out of range = 0)
        # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[k-i]*g[i]
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        coeffs = f[k - idx]  # f[k-i] for i in idx
        constraints.append(coeffs @ g <= M)
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CLARABEL, verbose=False)
    
    if g.value is None:
        return None
    return np.maximum(g.value, 0)

import numpy as np
import time
from scipy.optimize import linprog

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def lp_direction(f, M, top_k, solver='highs'):
    """
    Solve LP to find descent direction.
    Primal: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in top_k
    Returns: (g, opt_val) or (None, None) if infeasible
    """

    K = len(top_k)
    
    # Build constraint matrix T_K (K x n)
    # T_K[k_idx, j] = f[k - j] if 0 <= k-j < n, else 0
    T = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        for j in range(n):

def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def build_conv_matrix_row(f, k):
    """Build the k-th row of the convolution matrix T. T[k,j] = f[k-j] if valid."""
    n = len(f)
    row = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    for j in range(j_min, j_max + 1):
        row[j] = f[k - j]
    return row


def lp_solve_dual(f, M, peak_indices):
    """
    Solve the dual LP: min M*sum(y) s.t. T^T y >= 1, y >= 0
    where T[k,j] = f[k-j] for valid k-j.
    Returns dual optimal value and dual variables.
    """
    n = len(f)
    K = len(peak_indices)
    
    # Build T as list of rows
    T_rows = []
    for k in peak_indices:
        row = np.zeros(n)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            row[idx] = f[k - idx]
        T_rows.append(row)
    T = np.array(T_rows)  # K x n
    
    # Dual: min M*sum(y) s.t. T^T y >= 1, y >= 0
    # In linprog: minimize c @ y s.t. A_ub @ y <= b_ub
    c = M * np.ones(K)
    A_ub = -T.T  # n x K
    b_ub = -np.ones(n)
    bounds = [(0, None)] * K
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if not res.success:
        return None, None
    return res.fun, res.x

            T[idx, jidx] = f[k - jidx]

    c_p = -np.ones(n)
    A_ub_p = T
    b_ub_p = M * np.ones(K)
    bounds_p = [(0, None)] * n
    
    res = linprog(c_p, A_ub=A_ub_p, b_ub=b_ub_p, bounds=bounds_p, method='highs',
                  options={'preset': True, 'time_limit': 60})
    
    if not res.success:
        return None, None
    g = np.maximum(res.x, 0)
    val = -res.fun
    return g, val


def lp_descent_step(f, M, K_constraint=30, max_violation_rounds=3):
    """
    LP-based descent direction with constraint generation.
    Returns a direction g (sum=1) or None.
    """
    n = len(f)
    c = fast_autocorr(f)
    
    # Start with top K peaks
    top_k = np.argsort(c)[::-1][:K_constraint]
    peak_set = set(int(x) for x in top_k)
    
    for round_num in range(max_violation_rounds):
        peak_list = sorted(peak_set)
        
        # Solve primal LP
        g, val = lp_solve_primal(f, M, peak_list)
        if g is None:
            break
        
        sg = g.sum()
        if sg < 1e-10:
            break
        
        g_norm = g / sg
        
        # Check for violations outside the constraint set
        cross_corr = np.convolve(f, g_norm)
        violations = np.where(cross_corr > M + 1e-10)[0]
        violations = violations[~np.isin(violations, list(peak_set))]
        
        if len(violations) == 0:
            return g_norm, val
        
        # Add worst violations
        worst_viol = violations[np.argsort(cross_corr[violations])[::-1]][:5]
        for v in worst_viol:
            peak_set.add(int(v))
    
    # Final solve with all accumulated constraints
    peak_list = sorted(peak_set)
    g, val = lp_solve_primal(f, M, peak_list)
    if g is not None and g.sum() > 1e-10:
        return g / g.sum(), val
    return None, None


def lp_optimize(f0, time_limit, K_start=20, K_end=80, K_growth=1.02,
                step_decay=0.9995, verbose=True, seed=42):
    """
    Main LP-based optimization loop.
    """
    global _t_global_start
    t0 = time.monotonic()
    n = len(f0)
    f = norm_sum1(f0).copy()
    best_f = f.copy()
    best_M = fast_autocorr(f).max()
    best_obj = 2.0 * n * best_M
    rng = np.random.default_rng(seed)
    
    step_size = 0.15
    no_improve = 0
    max_no_improve = 200
    K_current = K_start
    lp_success_count = 0
    lp_fail_count = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        
        # Get current max
        c = fast_autocorr(f)
        M = c.max()
        
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_f = f.copy()
            best_obj = 2.0 * n * M
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Restart with perturbation
            noise_scale = 0.05 / (1.0 + no_improve / 100.0)
            pert = rng.normal(0, noise_scale, n) * best_f
            f_try = best_f + pert
            f_try = np.maximum(f_try, 1e-12)
            f_try = f_try / f_try.sum()
            M_try = fast_autocorr(f_try).max()
            if M_try < best_M:
                f = f_try
                best_M = M_try
                best_f = f.copy()
                best_obj = 2.0 * n * M_try
                no_improve = 0
            else:
                f = best_f + 0.01 * rng.normal(0, 1, n) * best_f
                f = np.maximum(f, 1e-12)
                f = f / f.sum()
            K_current = K_start
            step_size = 0.15
            continue
        
        # Solve LP for descent direction
        c = fast_autocorr(f)
        M = c.max()
        g_dir, lp_val = lp_descent_step(f, M, K_constraint=K_current,
                                         max_violation_rounds=2)
        
        if g_dir is not None:
            lp_success_count += 1
            # Take step with line search
            improved = False
            for t in [step_size, step_size * 0.5, step_size * 0.25,
                      step_size * 0.1, 0.05, 0.02]:
                f_try = (1 - t) * f + t * g_dir
                f_try = np.maximum(f_try, 0)
                s = f_try.sum()
                if s < 1e-10:
                    continue
                f_try = f_try / s
                M_try = fast_autocorr(f_try).max()
                if M_try < M - 1e-12:
                    f = f_try
                    improved = True
                    break
            
            if not improved:
                no_improve += 1
            else:
                no_improve = 0
            
            # Adapt step size
            if improved:
                step_size = min(step_size * 1.02, 0.5)
            else:
                step_size = max(step_size * 0.7, 0.01)
            
            # Grow K slowly
            K_current = min(int(K_current * K_growth), K_end)
        else:
            lp_fail_count += 1
            no_improve += 1
            # Fall back to gradient step
            step_size = max(step_size * 0.8, 0.01)
            # Simple gradient: move mass away from peak contributors
            c = fast_autocorr(f)
            k_peak = int(np.argmax(c))
            i_start = max(0, k_peak - n + 1)
            i_end = min(n - 1, k_peak)
            idx = np.arange(i_start, i_end + 1)
            products = f[idx] * f[k_peak - idx]
            # Reduce the largest product
            max_pi = int(np.argmax(products))
            src = int(idx[max_pi])
            partner = k_peak - src
            if partner < 0 or partner >= n:
                partner = src
            # Move small amount from larger to random position
            if f[src] >= f[partner]:
                donor = src
            else:
                donor = partner
            if f[donor] > 1e-10:
                amt = f[donor] * 0.01
                dst = int(rng.integers(n))
                f[donor] -= amt
                f[dst] += amt
                f = np.maximum(f, 0)
                f = f / f.sum()
        
        # Periodic verbose
        if verbose and (lp_success_count + lp_fail_count) % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-opt t={}s: obj={:.6f}, K={}, step={:.3f}, "
                  "lp_suc={}, lp_fail={}, stall={}".format(
                      round(elapsed, 1), best_obj, K_current, step_size,
                      lp_success_count, lp_fail_count, no_improve))
    
    return best_f, best_obj


def nesterov_opt(a0, time_limit, beta_start=30.0, beta_end=3000.0,
                 beta_growth=1.007, eta_base=0.5, top_k_ratio=0.02,
                 restart_every=1500, verbose=True, seed=42):
    """Nesterov accelerated projected gradient on log-sum-exp smoothed objective."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute smoothed gradient
        c = fast_autocorr(y)
        if top_k < len(c):
            idx_top = np.argsort(c)[::-1][:top_k]
            c_use = np.zeros(len(c))
            c_use[idx_top] = c[idx_top]
            c_max = c[idx_top].max()
        else:
            c_use = c
            c_max = c.max()
        
        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros_like(c_use)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total
        
        # Gradient via FFT cross-correlation
        fft_size = 1 << (2 * n - 1).bit_length()
        A_freq = np.fft.rfft(y, fft_size)
        P_freq = np.fft.rfft(p, fft_size)
        a_rev = y[::-1]
        AR_freq = np.fft.rfft(a_rev, fft_size)
        Conv = np.fft.irfft(P_freq * AR_freq, fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
        
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)
        
        z = y - eta * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart ({:.1f}s): J={:.6f}".format(elapsed, J))
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t={} ({:.1f}s): J={:.6f}, beta={:.1f}".format(
                t, elapsed, J, beta))
    
    return best_a, best_max_b
