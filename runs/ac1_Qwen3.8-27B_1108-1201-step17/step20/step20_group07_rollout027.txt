def get_solver():
    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return cp.ECOS

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), N)
    return np.fft.irfft(A * A, N)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def convolve_fft(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:size]

def compute_smoothed_gradient(a, beta, top_k=None):
    n = len(a)
    b = fast_autocorr(a)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros_like(b)
        b_use[idx] = b[idx]
    else:
        b_use = b
    shifted = beta * (b_use - b_use.max())
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.ones_like(b_use) / len(b_use)
    else:
        p = exp_vals / total
    rev_p = p[::-1]
    conv = convolve_fft(a, rev_p)
    c = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * c

def mirror_descent(a0, time_limit, beta_start=10.0, beta_end=800.0,
                   beta_growth=1.003, top_k=None, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    beta = beta_start
    t = 0
    stall = 0
    max_stall = 800
    rng = np.random.default_rng(999)
    eta_base = 0.1

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)
        m_curr = b.max()
        if m_curr < best_m - 1e-15:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        g = compute_smoothed_gradient(a, beta, top_k)
        g_norm = np.abs(g).max()
        if g_norm < 1e-30:
            g_norm = 1e-30
        eta = eta_base / g_norm
        eta = min(eta, 0.5)

        log_update = -eta * g
        log_update = np.clip(log_update, -100, 100)
        log_a = np.log(np.maximum(a, 1e-300)) + log_update
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-30:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
            a_new = np.maximum(a_new, 0.0)
            s2 = a_new.sum()
            if s2 > 1e-15:
                a_new = a_new / s2

        a = a_new

        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > max_stall:
            stall = 0
            restart_count = t // max_stall
            noise_scale = 0.05 / (1.0 + 0.5 * restart_count)
            pert = rng.normal(0, noise_scale, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-12)
            a_r = a_r / a_r.sum()
            m_r = fast_autocorr(a_r).max()
            if m_r < best_m:
                best_m = m_r
                best_a = a_r.copy()
            a = a_r
            beta = max(beta * 0.5, beta_start)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    MD t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J) + " beta=" + str(round(beta, 1)))

    return best_a, best_m

                reduce_idx = src
            else:
                reduce_idx = partner
            if a[reduce_idx] < 1e-14:
                continue
            safe_mask = ~dangerous
            if safe_mask.sum() < 50:
                safe_mask = pressure < np.percentile(pressure, 50)
            if safe_mask.sum() < 20:
                safe_mask = np.ones(n, dtype=bool)
            safe_pressure = pressure.copy()
            safe_pressure[~safe_mask] = np.max(pressure) + 1.0
            safe_pressure[reduce_idx] = np.max(pressure) + 2.0
            dst = int(np.argmin(safe_pressure))
            if dst == reduce_idx:
                continue
            found = False
            for frac in [0.003, 0.008, 0.02, 0.05, 0.1]:
                amt = a[reduce_idx] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            if not found and stall > 150:
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-12:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        if not improved:
            stall += 1
            if stall % 100 == 0:
                noise_scale = min(0.04, 0.005 * stall / 100)
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r if m_r < best_m + 0.01 else best_a.copy()

    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    b = fast_autocorr(a)
    active_set = set(int(k) for k in np.argsort(b)[::-1][:600])
    no_improve = 0
    t = 0

    while time.monotonic() - t0 < time_limit and no_improve < 1000:
        t += 1
        b = fast_autocorr(a)
        M_curr = b.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > 2000:
            margins = np.array([M_curr - b[k] for k in constraint_list])
            keep = np.argsort(margins)[:2000]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        Af = np.fft.rfft(a, 1 << (size - 1).bit_length())
        Gf = np.fft.rfft(g_dir, 1 << (size - 1).bit_length())
        conv_ag = np.fft.irfft(Af * Gf, 1 << (size - 1).bit_length())[:size]
        violations = np.where(conv_ag > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ag[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:30]]
            for k in top_v:
                active_set.add(int(k))

        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.01, default_alpha * 0.5), 1.0 / (t + 1), max(0.05, default_alpha * 2)]
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            alpha = min(alpha, 1.0)
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-15:
            a_new = a_new / st
        a = a_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, best_m

def parametric_init(n, time_limit):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    best_a = None
    best_m = float('inf')

    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0]:
        for beta_p in [0.5, 0.8, 1.0, 1.5, 2.0]:
            if time.monotonic() - t0 > time_limit:
                break
            a_raw = (x + 0.01) ** alpha * (1.0 - x + 0.01) ** beta_p
            a = norm_sum1(a_raw)
            m = fast_autocorr(a).max()
            if m < best_m:
                best_m = m
                best_a = a.copy()

    rng = np.random.default_rng(42)
    for _ in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        p = rng.uniform(0.5, 3.0)
        a = norm_sum1(rng.random(n) ** p)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
    return best_a, best_m

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        a_init, _ = parametric_init(n_base, 20.0)
        best_seq = a_init
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Mirror descent (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Mirror Descent (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60:
        a_out, _ = mirror_descent(best_seq, t1, beta_start=8.0, beta_end=600.0,
                                  beta_growth=1.003, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "MD_n" + str(n0), t_global)

    # Phase 2: Projected subgradient with momentum
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Projected Subgradient (n=" + str(n0) + ") ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 60:
        a_out, _ = projected_subgradient(best_seq, t2, beta_start=5.0, beta_end=400.0,
                                         beta_growth=1.004, momentum=0.85, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "PSG_n" + str(n0), t_global)

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Peak Shaving (n=" + str(n0) + ") ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_out, _ = peak_shaving(best_seq, t3, top_k=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "PS_n" + str(n0), t_global)

    # Phase 4: Mirror descent round 2 (with higher beta)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 4: Mirror Descent 2 (n=" + str(n0) + ") ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 40:
        a_out, _ = mirror_descent(best_seq, t4, beta_start=50.0, beta_end=2000.0,
                                  beta_growth=1.005, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "MD2_n" + str(n0), t_global)

    # Phase 5: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: LP-Frank-Wolfe (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 40 and n_best <= 8000:
        a_out, _ = lp_frank_wolfe(best_seq, t5, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "LP_n" + str(n_best), t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby = []
    for dn in [-300, -100, 100, 200, 400, 600]:
        nl = n_best + dn
        if 5000 < nl <= 8000 and nl != n_best:
            nearby.append(nl)

    for new_n in nearby[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print("\n=== Phase 6: n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.3, 150.0)
        if t6a > 30:
            a_out, _ = mirror_descent(a_long, min(t6a * 0.6, 100.0),
                                      beta_start=10.0, beta_end=400.0,
                                      beta_growth=1.003, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "MD_n" + str(new_n), t_global)
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.3, 120.0)
        if t6b > 30:
            a_out, _ = peak_shaving(best_seq, t6b, top_k=15, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "PS_n" + str(new_n), t_global)

    # Phase 7: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t7a = min(remaining * 0.3, 120.0)
        if t7a > 20:
            a_out, _ = mirror_descent(best_seq, t7a, beta_start=100.0, beta_end=3000.0,
                                      beta_growth=1.005, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_MD", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 50 and n_best <= 8000:
        t7b = min(remaining * 0.5, 150.0)
        if t7b > 30:
            a_out, _ = lp_frank_wolfe(best_seq, t7b, solver, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "final_LP", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
