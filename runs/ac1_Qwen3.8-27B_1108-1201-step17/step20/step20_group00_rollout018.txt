    fft_size = 1 << (2*n-2).bit_length()

            break

        coeffs = build_lp_matrix(y, clist, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K:
                active_set = set(int(k) for k in clist[:init_K])
                continue

        if not lp_ok:
            if no_improve > stall_limit // 2:
                break
            if len(clist) > 100:
                active_set = set(int(k) for k in clist[:200])
                continue
            break

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Update active set: find violations
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, n=fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))

        # Also add currently top peaks not in active set (periodic refresh)
        if iter_count % 50 == 0:
            top_peaks = set(int(k) for k in np.argsort(ac)[::-1][:max_K // 2])
            active_set = active_set.union(top_peaks)
            if len(active_set) > max_K:
                ac2 = fast_autocorr(y)
                margins = np.array([ac2.max() - ac2[k] for k in active_set])
                keep = np.argsort(margins)[:max_K]
                active_set = set(int(list(active_set)[i]) for i in keep)

        # Backtracking Armijo line search
        f_curr = M_curr
        alpha = 1.0
        c_armijo = 1e-4
        rho = 0.5
        grad_dir = np.zeros(n, dtype=np.float64)
        # Approximate gradient at y in direction g_dir
        # d/dt max_k ((1-t)*y + t*g_dir conv (1-t)*y + t*g_dir) at t=0
        # This is complex; use simple backtracking on the actual objective
        for _ in range(30):
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                alpha *= rho
                continue
            trial = trial / ts
            f_trial = fast_autocorr(trial).max()
            if f_trial <= f_curr - c_armijo * alpha * (f_curr - M_curr * 0.5):
                y_new = trial
                break
            alpha *= rho
        else:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                alpha = 0.01
                trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                y_new = trial / ts
            else:
                y_new = trial / ts

        y = y_new

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)))

    if verbose:
        print("    LP-FW done (" + str(round(time.monotonic() - t0, 1)) + "s) iter=" + str(iter_count) + " J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def mirror_descent(a0, time_limit, top_k=50, temp_init=30.0, momentum_beta=0.9, verbose=True):
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    v = np.zeros(n, dtype=np.float64)  # momentum
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 8000:
            break

        # Nesterov lookahead
        y = x + momentum_beta * v
        y = np.maximum(y, 1e-18)
        y = y / y.sum()

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        cur_temp = temp_init * max(0.2, 1.0 - t / 30000.0)
        w = np.exp(cur_temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        # Compute gradient: d/dc max_k autocorr[k] projected to y
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * y[ki - idx_arr]

        # Mirror descent step (multiplicative updates)
        eta = 3.0 / (np.sqrt(max(t, 1)) + 1.0)
        log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        v = a_new - x
        x = a_new

        val = obj_val(x)
        if val < best_val - 1e-12:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 4000 and stall % 4000 == 0:
            noise_scale = 0.01 + 0.02 * (stall / 16000.0)
            noise = noise_scale * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            v = np.zeros(n, dtype=np.float64)
            stall = 0

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    Mirror t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val))

    return best_a, best_val

def peak_shave(a0, time_limit, n_targets=6, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False

        for k in targets:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            top_srcs = idx[np.argsort(contrib)[::-1][:5]]
            found_move = False

            for src in top_srcs:
                src = int(src)
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Score destinations by potential peak increase
                scores = np.zeros(n)
                for k2 in targets:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10

                candidates = np.argsort(scores)[:80]
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.003, 0.007, 0.015, 0.03]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            found_move = True
                            break
                    if found_move:
                        break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Random perturbation
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.005)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            no_improve = 0
                            break

        if verbose and no_improve % 100 == 0:
            print("    Shave (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m) + " noimp=" + str(no_improve))

    return best_a, 2.0 * n * best_m

def coord_polish(a0, time_limit, seed=42, verbose=False):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 1000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:

            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            top_srcs = idx[np.argsort(contrib)[::-1][:3]]
            for src in top_srcs:
                src = int(src)
                if a[src] < 1e-12:
                    continue

                scores = np.zeros(n)
                for k2 in top_k[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10

                candidates = np.argsort(scores)[:40]
                found = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break

        if not improved:
            stall += 1
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.008)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 200 == 0:
            print("    Coord (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Total budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent loaded: n=" + str(len(p)) + " score=" + repr(best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 8000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Default init n=" + str(n_init) + " score=" + repr(best_score))

    n_parent = len(best_seq)

    # Phase 1: Diverse initialization at working length
    n_work = min(n_parent, 8000)
    a_work = interpolate_seq(best_seq, n_work) if n_work != n_parent else best_seq.copy()
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Diverse Init (n=" + str(n_work) + ") ===")
    t1 = min(remaining * 0.03, 40.0)
    if t1 > 5:
        a_p, v_p = diverse_init(n_work, a_work, obj_val(a_work))
        sc_p = eval_safe(a_p)
        if sc_p < best_score:
            best_score = sc_p
            best_seq = interpolate_seq(a_p, n_parent) if n_parent > n_work else a_p
            a_work = a_p
            print("  Init improved: " + repr(best_score))
        else:
            a_work = a_p  # still use it as starting point for optimization

    # Phase 2: LP-FW at working length
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P2: LP-FW (n=" + str(n_work) + ") ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 60:
        a_out, v_out = lp_fw_best(normalize(a_work), t2, solver,
                                  init_K=400, max_K=4000, add_per=50, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_parent) if n_parent > n_work else a_out
            a_work = a_out
            print("  LP-FW improved: " + repr(best_score))
        else:
            a_work = a_out  # use as next starting point regardless

    # Phase 3: Mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P3: Mirror Descent (n=" + str(n_work) + ") ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30:
        a_out, v_out = mirror_descent(normalize(a_work), t3, top_k=60, temp_init=40.0,
                                      momentum_beta=0.85, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_parent) if n_parent > n_work else a_out
            a_work = a_out
            print("  Mirror improved: " + repr(best_score))
        else:
            a_work = a_out

    # Phase 4: Peak shaving at working length
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: Peak Shave (n=" + str(n_work) + ") ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, v_out = peak_shave(normalize(a_work), t4, n_targets=8, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_parent) if n_parent > n_work else a_out
            a_work = a_out
            print("  Shave improved: " + repr(best_score))
        else:
            a_work = a_out

    # Phase 5: Work at parent length if it's not too large
    remaining = time_budget - (time.monotonic() - t_global)
    n_full = n_parent
    if n_full > 15000:
        n_full = 15000
    a_full = interpolate_seq(best_seq, n_full)
    print("\n=== P5: Optimize at n=" + str(n_full) + " ===")
    if n_full <= 10000:
        t5 = min(remaining * 0.20, 300.0)
        if t5 > 60:
            a_out, v_out = lp_fw_best(normalize(a_full), t5, solver,
                                      init_K=500, max_K=5000, add_per=50, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Full LP improved: " + repr(best_score))
            else:
                a_full = a_out
    else:
        t5 = min(remaining * 0.15, 200.0)
        if t5 > 30:
            a_out, v_out = mirror_descent(normalize(a_full), t5, top_k=80, temp_init=50.0,
                                          momentum_beta=0.85, verbose=True)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  Full Mirror improved: " + repr(best_score))
            else:
                a_full = a_out

    # Phase 6: Coordinate descent polish at the best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_pol = min(len(best_seq), 10000)
    a_pol = interpolate_seq(best_seq, n_pol) if n_pol != len(best_seq) else best_seq.copy()
    print("\n=== P6: Coord Polish (n=" + str(n_pol) + ") ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 20:
        a_out, v_out = coord_polish(normalize(a_pol), t6, seed=42, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, len(best_seq)) if len(best_seq) > n_pol else a_out
            print("  Polish improved: " + repr(best_score))

    # Phase 7: Final LP burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_fin = min(len(best_seq), 8000)
    a_fin = interpolate_seq(best_seq, n_fin) if n_fin != len(best_seq) else best_seq.copy()
    if remaining > 40 and n_fin <= 8000:
        print("\n=== P7: Final LP (n=" + str(n_fin) + ") ===")
        t7 = min(remaining - 10.0, 200.0)
        a_out, v_out = lp_fw_best(normalize(a_fin), t7, solver,
                                  init_K=600, max_K=5000, add_per=60, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, len(best_seq)) if len(best_seq) > n_fin else a_out
            print("  Final LP improved: " + repr(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " t=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: using fallback")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
