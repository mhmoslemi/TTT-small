"""

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _eval_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]

def _norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / max(len(a), 1)
    return a / s


def _simplex_proj(z):
    n = len(z)
    if n == 0:
        return z.copy()
    u = np.sort(z)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def _smoothed_grad(a, beta, top_k=None):
    n = len(a)
    b = _fast_autocorr(a)
    if top_k is not None and 0 < top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_top = b[idx]
        b_shift = np.zeros(len(b))
        b_shift[idx] = b_top
        b_max = b_top.max()
    else:
        b_top = b
        b_shift = b
        b_max = b.max()
    shifted = beta * (b_shift - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        w = np.zeros_like(b_shift)
        mask = b_shift == b_max
        w[mask] = 1.0 / max(1, mask.sum())
    else:
        w = exp_vals / p_total
    # Grad[i] = sum_{m=0}^{n-1} w[m+i] * a[m]
    # Use FFT: grad = conv of w (length 2n-1) reversed with a (length n)
    # grad[i] = sum_m w[m+i]*a[m]
    # Let's compute via: pad w to length L, pad a to length L
    # cross-correlation style
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    w_pad = np.zeros(fft_size)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    # We want grad[i] = sum_{m=0}^{n-1} w[m+i] * a[m]
    # This is (a * flip(w))[i + (2n-2)] in linear convolution
    # flip(w) has length 2n-1, a has length n
    # conv(a, flip(w)) has length n + (2n-1) - 1 = 3n-2
    # grad[i] = conv(a, flip(w))[i + 2n - 2]
    w_flip = w[::-1].copy()
    wf_pad = np.zeros(fft_size)
    wf_pad[:len(w_flip)] = w_flip
    conv = np.fft.irfft(np.fft.rfft(wf_pad) * np.fft.rfft(a_pad), fft_size)
    offset = 2 * n - 2
    grad = conv[offset:offset + n]
    return grad, b, w


def fista_optimize(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   beta_growth=1.004, top_k=30, restart_every=4000,
                   verbose=True, seed=42):
    n = len(a0)
    x = _norm1(a0).copy()
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = _fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    s_prev = None
    y_prev = None
    bb_step = 1.0 / np.sqrt(n)

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b, w = _smoothed_grad(y, beta, top_k)

        # Barzilai-Borwein step
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_cand = sy_gy / gy_gy
                if bb_cand > 1e-8 and bb_cand < 20.0:
                    bb_step = max(1e-6, min(bb_cand, 10.0 / beta))
                else:
                    bb_step = max(bb_step * 0.85, 1e-7)
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(bb_step, 0.5)
        z = y - eta * grad
        x_new = _simplex_proj(z)

        b_new = _fast_autocorr(x_new)
        max_new = b_new.max()

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.02 / (1.0 + 0.03 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 8.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / np.sqrt(n)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print(f"    FISTA restart t={t} ({elapsed:.1f}s): J={J:.6f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print(f"    FISTA t={t} ({elapsed:.1f}s): J={J:.6f} beta={beta:.1f}")

    return best_a, best_max_b


def lp_fw_optimize(a0, time_limit, solver, initial_K=2000, max_K=5000,
                   add_per_iter=30, verbose=True):
    n = len(a0)
    a = _norm1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = _fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:min(initial_K, 2 * n - 1)])
    no_improve = 0
    max_no_improve = 1000

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = _fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Prune active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_list = list(active_set)
            active_set = set(int(active_list[i]) for i in keep)

        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = set(int(k) for k in np.argsort(_fast_autocorr(y))[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check for violations, add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search for step size
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.01, default_alpha * 0.7),
                  max(0.01, default_alpha * 1.5)]
        alphas = list(set(max(0.01, min(1.0, a)) for a in alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = _fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.1f}s): J={J:.6f} |S|={len(active_set)}")

    return best_a, best_m


def lbfgs_smoothed(a0, time_limit, beta=50.0, verbose=True):
    n = len(a0)
    a_init = _norm1(a0).copy()

    def obj_and_grad(flat):
        a = _norm1(flat.reshape(-1))
        grad, b, w = _smoothed_grad(a, beta, top_k=None)
        obj = (1.0 / beta) * np.log(np.sum(np.exp(beta * (b - b.max()))) + 1e-300) + beta * b.max()
        return obj, grad

    # Use scipy L-BFGS-B on the normalized coordinates
    # Parameterize with log to maintain positivity: a_i = exp(x_i) / sum(exp(x_j))
    x0 = np.log(a_init + 1e-30)

    def softobj(flat):
        x = flat
        e = np.exp(x - x.max())
        a = e / e.sum()
        b = _fast_autocorr(a)
        b_max = b.max()
        shifted = beta * (b - b_max)
        shifted = np.clip(shifted, -500, 500)
        obj = (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
        return obj

    x_start = x0.copy()
    t0 = time.monotonic()
    result = scipy_minimize(softobj, x_start, method='L-BFGS-B',
                            options={'maxiter': 500, 'maxfun': 30000,
                                     'maxls': 50, 'gtol': 1e-7,
                                     'ftol': 1e-12, 'disp': (verbose and 0)})
    elapsed = time.monotonic() - t0

    if result.success or result.fun < softobj(x_start):
        x_best = result.x
    else:
        x_best = x_start

    e = np.exp(x_best - x_best.max())
    a_out = e / e.sum()
    m_out = _fast_autocorr(a_out).max()
    if verbose:
        J = 2.0 * n * m_out
        print(f"    L-BFGS: J={J:.6f} ({elapsed:.1f}s) fun={result.fun:.6f}")

def local_search(a0, time_limit, verbose=True, seed=999):
    n = len(a0)
    a = _norm1(a0).copy()
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_autocorr(a)
        # Focus on top peaks
        top_peaks = np.argsort(ac)[::-1][:50]
        improved = False

        # Systematic: for each peak, try redistributing from contributing elements
        for k_int in top_peaks[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Try moving mass from the dominant contributor
            for attempt in range(10):
                src = int(idx[np.argmax(contrib)])
                if a[src] < 1e-12:
                    break
                # Find a good destination (away from peaks)
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_peaks[:25]:
                    s2 = max(0, int(k2) - n + 1)
                    e2 = min(n - 1, int(k2))
                    in_peak[s2:e2 + 1] = True

                low_mask = ~in_peak
                if low_mask.sum() < 5:
                    low_mask = np.ones(n, dtype=bool)
                low_idx = np.where(low_mask)[0]
                # Pick the one with smallest value
                dst = int(low_idx[np.argmin(a[low_idx])])
                if dst == src:
                    continue

                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = _fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        ac = _fast_autocorr(a)
                        top_peaks = np.argsort(ac)[::-1][:50]
                        break

        # Random perturbations
        for _ in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = float(rng.uniform(0.001, 0.02))
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = _fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        improved = True

        if not improved:
            stall += 1
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Local ({elapsed:.1f}s): J={J:.6f} stall={stall}")

    return best_a, best_m


def cyclic_shift_best(a0):
    n = len(a0)
    best_m = _fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a0, fft_size)
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m - 1e-15:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def peak_defocus(a0, time_limit, n_iters=200, verbose=True):
    """Iteratively reduce the peak autocorrelation by defocusing contributing elements."""
    n = len(a0)
    a = _norm1(a0).copy()
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b = _fast_autocorr(a)
        b_max = b.max()
        k_peak = int(np.argmax(b))

        # Find contributing elements
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        total = contrib.sum()
        if total < 1e-15:
            break

        # Proportional reduction: scale down contributors, scale up non-contributors
        step = 0.05
        a_new = a.copy()
        a_new[idx] *= (1.0 - step)
        # Redistribute to non-contributing elements
        non_contrib = np.setdiff1d(np.arange(n), idx)
        if len(non_contrib) == 0:
            break
        removed = a_new.sum() - a.sum()
        if removed < 0:
            add_per = -removed / len(non_contrib)
            a_new[non_contrib] += add_per

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        m_new = _fast_autocorr(a_new).max()

        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        # Accept partially
        a = 0.7 * a + 0.3 * a_new
        a = _norm1(a)

        if verbose and it % 50 == 0:
            J = 2.0 * n * best_m
            print(f"    Defocus it={it}: J={J:.6f}")

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = _eval_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = _eval_safe(parent)
            if np.isfinite(p_score):
                best_seq = _norm1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6000
        x = np.linspace(0, 1, n_base)
        best_seq = _norm1(np.sin(np.pi * x) ** 2)
        best_score = _eval_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: FISTA warm start ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FISTA (n={n0}) ===")
    t1 = min(remaining * 0.15, 250.0)
    if t1 > 60:
        try:
            a_out, _ = fista_optimize(best_seq, t1, beta_start=15.0, beta_end=500.0,
                                      beta_growth=1.004, top_k=25, restart_every=3000)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)
        except Exception as e:
            print(f"  FISTA error: {e}")

    # === Phase 2: Cyclic shift ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_sh, _ = cyclic_shift_best(best_seq)
        best_seq, best_score = try_update(best_seq, best_score, a_sh, "cyclic1", t_global)

    # === Phase 3: LP-FW main ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.30, 500.0)
    if t3 > 60 and n_best <= 10000:
        try:
            a_out, _ = lp_fw_optimize(best_seq, t3, solver,
                                      initial_K=3000, max_K=6000, add_per_iter=40)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)
        except Exception as e:
            print(f"  LP-FW error: {e}")

    # === Phase 4: L-BFGS-B ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: L-BFGS-B (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 40:
        try:
            a_out, _ = lbfgs_smoothed(best_seq, t4, beta=80.0)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs", t_global)
        except Exception as e:
            print(f"  L-BFGS error: {e}")

    # === Phase 5: Local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Local search (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30:
        try:
            a_out, _ = local_search(best_seq, t5, seed=123)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "local1", t_global)
        except Exception as e:
            print(f"  Local search error: {e}")

    # === Phase 6: Peak defocus ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Peak defocus (n={len(best_seq)}) ===")
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 20:
        try:
            a_out, _ = peak_defocus(best_seq, t6, n_iters=300)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "defocus", t_global)
        except Exception as e:
            print(f"  Defocus error: {e}")

    # === Phase 7: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Multi-length (base n={n_best}) ===")
    lengths_to_try = [n_best + 500, n_best + 1500, n_best + 3000]
    lengths_to_try = [l for l in lengths_to_try if l <= 12000 and l > n_best]

    for new_n in lengths_to_try[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n}")
        a_long = interpolate_seq(best_seq, new_n)

        t7a = min(remaining * 0.25, 150.0)
        if t7a > 30:
            try:
                a_out, _ = fista_optimize(a_long, t7a, beta_start=25.0, beta_end=600.0,
                                          beta_growth=1.005, top_k=20, restart_every=2500)
                best_seq, best_score = try_update(best_seq, best_score, a_out, f"fista_n{new_n}", t_global)
            except Exception as e:
                print(f"  FISTA n={new_n} error: {e}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_sh, _ = cyclic_shift_best(best_seq)
            best_seq, best_score = try_update(best_seq, best_score, a_sh, f"cyc_n{new_n}", t_global)

    # === Phase 8: FISTA polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: FISTA polish (n={n_best}) ===")
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 30:
        try:
            a_out, _ = fista_optimize(best_seq, t8, beta_start=100.0, beta_end=2000.0,
                                      beta_growth=1.008, top_k=15, restart_every=2000, seed=777)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista2", t_global)
        except Exception as e:
            print(f"  FISTA2 error: {e}")

    # === Phase 9: LP-FW final ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: LP-FW final (n={n_best}) ===")
    t9 = min(remaining * 0.25, 300.0)
    if t9 > 40 and n_best <= 10000:
        try:
            a_out, _ = lp_fw_optimize(best_seq, t9, solver,
                                      initial_K=4000, max_K=7000, add_per_iter=50)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)
        except Exception as e:
            print(f"  LP-FW2 error: {e}")

    # === Phase 10: Final local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 10: Final local search ===")
    t10 = min(remaining * 0.30, 200.0)
    if t10 > 20:
        try:
            a_out, _ = local_search(best_seq, t10, seed=456)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "local_final", t_global)
        except Exception as e:
            print(f"  Final local error: {e}")

    # === Final: Cyclic shift + tiny FISTA ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_sh, _ = cyclic_shift_best(best_seq)
        best_seq, best_score = try_update(best_seq, best_score, a_sh, "cyc_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        try:
            a_out, _ = fista_optimize(best_seq, min(remaining * 0.6, 80.0),
                                      beta_start=200.0, beta_end=3000.0,
                                      beta_growth=1.01, top_k=10, restart_every=1500, seed=2024)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_final", t_global)
        except Exception:
            pass

    # === Final local micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        try:
            a_out, _ = local_search(best_seq, min(remaining * 0.5, 40.0), seed=31415)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "local_micro", t_global)
        except Exception:
            pass

    # Final validation
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = _norm1(np.sin(np.pi * x) ** 2)
        best_score = _eval_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
