def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_autocorr(a):

    size = 2 * n - 1

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_gradient(a, p_weights):
    """Compute gradient of smoothed objective. p_weights is length 2n-1."""
    n = len(a)
    L = next_pow2(3 * n)
    P = np.zeros(L)
    P[:2 * n - 1] = p_weights
    A = np.zeros(L)
    A[:n] = a
    # Cross-correlation: (R_p_a)[i] = sum_j p[i+j] * a[j]
    # = IFFT(FFT(P) * conj(FFT(A)))
    P_fft = np.fft.rfft(P)
    A_fft = np.fft.rfft(A)
    R = np.fft.irfft(P_fft * np.conj(A_fft), L)
    return 2.0 * R[:n]


def smooth_gradient_descent(a0, time_limit, verbose=True,
                            beta_start=50.0, beta_end=3000.0,
                            beta_growth=1.003, lr_start=0.05,
                            top_k_frac=0.05):
    """Adam optimizer on simplex with log-sum-exp smoothed max."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    beta = beta_start
    lr = lr_start
    stall = 0
    max_stall = 800
    top_k = max(20, int(n * top_k_frac))

    b_cache = fast_autocorr(x)

    while time.monotonic() - t0 < time_limit and stall < max_stall * 4:
        t += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break

        # Compute autocorrelation
        b = b_cache
        # Select top-k for smoothing
        idx_top = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[idx_top]
        b_max = b_top.max()

        # Softmax weights over top-k (expanded to full for gradient)
        shifted = beta * (b_top - b_max)
        shifted = np.clip(shifted, -200, 200)
        exp_vals = np.exp(shifted)
        p_top = exp_vals / exp_vals.sum()

        # Build full p_weights (length 2n-1)
        p_weights = np.zeros(2 * n - 1)
        p_weights[idx_top] = p_top

        # Compute gradient via FFT
        grad = compute_gradient(x, p_weights)

        # Adam update
        t_f = float(t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_f)
        v_hat = v / (1 - beta2 ** t_f)
        update = m_hat / (np.sqrt(v_hat) + eps)

        # Step (on simplex, use projected step)
        x_new = x - lr * update
        x_new = simplex_project(x_new)

        # Evaluate
        b_new = fast_autocorr(x_new)
        m_new = b_new.max()

        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
            x = x_new
        else:
            stall += 1
            # Revert
            x_new = x

        b_cache = fast_autocorr(x)

        # Adaptive beta (increase smoothing sharpness)
        if t % 30 == 0:
            beta = min(beta * beta_growth, beta_end)
            lr = max(lr_start * 0.97 ** (t // 200), lr_start * 0.01)
            beta1 = min(0.9, 0.8 + 0.001 * (t // 500))

        # Restart if stalled
        if stall == max_stall:
            stall = 0
            noise_scale = 0.015
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-14)
            x_r = x_r / x_r.sum()
            m_r = fast_autocorr(x_r).max()
            if m_r < best_m:
                best_m = m_r
                best_a = x_r.copy()
            x = x_r
            m = np.zeros(n)
            v = np.zeros(n)
            beta = max(beta * 0.5, beta_start)
            lr = lr_start
            if verbose:
                J = 2.0 * n * best_m
                print("    AdamOGD restart t={} ({}s): J={}".format(t, round(elapsed, 1), J))

        if verbose and t % 500 == 0:
            J = 2.0 * n * best_m
            print("    AdamOGD t={} ({}s): J={}".format(t, round(elapsed, 1), J))

    return best_a, best_m

def lp_refinement(a0, time_limit, solver, verbose=True):

        t += 1
        if time.monotonic() - t0 > time_limit - 10:
            break

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix A_con @ g <= M
        # A_con[k, i] = a[k-i] if 0 <= k-i < n, else 0
        A_con = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                A_con[jj, idx] = a[k_int - idx]

        # Solve LP: minimize M s.t. A_con @ g <= M, sum(g)=1, g>=0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, A_con @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            no_improve += 10
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 5
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-12:
            no_improve += 5
            continue
        g_dir = g_val / g_sum

        # Convolve a and g_dir to find violations
        fft_size = next_pow2(2 * n - 1)
        Af = np.fft.rfft(a, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)

        violations = np.where(conv_ag > M_curr + 1e-8)[0]

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(label, sc, len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: Adam-based optimization + peak targeting + LP + spectral perturb + multi-length."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: {:.1f}s".format(time_budget))

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={}, score={}".format(len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={}".format(len(best_seq), best_score))

    # Phase 1: Parametric sweep (brief)
    remaining = time_budget - (time.monotonic() - t_global)
    n0 = len(best_seq)
    print("\n=== Phase 1: Parametric sweep (n={}) ===".format(n0))
    t1 = min(remaining * 0.02, 25.0)
    if t1 > 8:
        a_param, _ = parametric_sweep(n0, t1, best_seq, best_score)
        best_seq, best_score = try_update(best_seq, best_score, a_param, "sweep", t_global)

    # Phase 2: Adam smoothed gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Adam OGD (n={}) ===".format(n_best))
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 30:
        a_out, _ = smooth_gradient_descent(best_seq, t2, verbose=True,
                                           beta_start=40.0, beta_end=2500.0,
                                           beta_growth=1.004, lr_start=0.04)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "adam1", t_global)

    # Phase 3: Peak targeting
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Peak targeting (n={}) ===".format(n_best))
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_out, _ = peak_targeting(best_seq, t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)

    # Phase 4: LP refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP (n={}) ===".format(n_best))
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 30 and n_best <= 8000:
        a_out, _ = lp_refinement(best_seq, t4, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 5: Spectral perturbation + re-optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Spectral perturbation (n={}) ===".format(n_best))
    t5_spec = min(remaining * 0.05, 60.0)
    if t5_spec > 10:
        a_out, _ = spectral_perturb(best_seq, t5_spec, num_trials=20, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec", t_global)

    # Phase 6: Adam round 2 (with higher beta)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Adam OGD round 2 (n={}) ===".format(n_best))
    t6 = min(remaining * 0.18, 280.0)
    if t6 > 30:
        a_out, _ = smooth_gradient_descent(best_seq, t6, verbose=True,
                                           beta_start=200.0, beta_end=5000.0,
                                           beta_growth=1.003, lr_start=0.02)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "adam2", t_global)

    # Phase 7: Peak targeting round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Peak targeting 2 (n={}) ===".format(n_best))
    t7 = min(remaining * 0.08, 120.0)
    if t7 > 15:
        a_out, _ = peak_targeting(best_seq, t7, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak2", t_global)

    # Phase 8: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby_lengths = []
    for dn in [-400, -200, 200, 400, 600, -600, 800]:
        nl = n_best + dn
        if 5000 < nl <= 9000 and nl not in nearby_lengths:
            nearby_lengths.append(nl)

    for new_n in nearby_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("\n=== Phase 8: Multi-length n={} ===".format(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t8a = min(remaining * 0.20, 120.0)
        if t8a > 15:
            a_out, _ = smooth_gradient_descent(a_long, t8a, verbose=False,
                                               beta_start=100.0, beta_end=2000.0,
                                               beta_growth=1.003, lr_start=0.03)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_n{}".format(new_n), t_global)

    # Phase 9: LP final
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: LP final (n={}) ===".format(n_best))
    t9 = min(remaining * 0.15, 200.0)
    if t9 > 25 and n_best <= 8500:
        a_out, _ = lp_refinement(best_seq, t9, solver, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 10: Final polish - coordinate descent + peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 10: Final polish (n={}) ===".format(n_best))
    if remaining > 40:
        t10a = min(remaining * 0.5, 150.0)
        if t10a > 15:
            a_out, _ = coordinate_descent(best_seq, t10a, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t10b = min(remaining * 0.5, 100.0)
        if t10b > 15:
            a_out, _ = peak_targeting(best_seq, t10b, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
