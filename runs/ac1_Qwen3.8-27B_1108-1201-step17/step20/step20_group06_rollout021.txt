def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    sum_a = a.sum()
    score = 2.0 * n * max_b / (sum_a ** 2)
    return score, max_b, sum_a, b

def nesterov_optimize(a0, time_limit, beta_start=20.0, beta_end=800.0,
                       beta_growth=1.003, eta_init=0.5, top_k_ratio=0.03,

        Pr = np.fft.rfft(p, fft_size)
        conv = np.fft.irfft(Ar * Pr, fft_size)
        grad = conv[n - 1:2 * n - 1] / beta_val
        return grad, b

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

                eta = eta_trial
            else:
                score_new, max_new, _, _ = compute_score_components(x_new)

        if score_new < best_score - 1e-12:

            best_a = x_new.copy()

def spectral_optimize(a0, time_limit, verbose=True, seed=777):
    """Spectral domain optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score, best_m, _, _ = compute_score_components(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.02
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        target = power[1:].mean() if len(power) > 1 else power[0]

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * (ratio - 1.0)
                A_new[k] *= max(0.3, np.sqrt(red))
            elif ratio < 0.5:
                inc = 1.0 + alpha * 0.3 * (0.5 - ratio)
                A_new[k] *= min(2.0, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        score_new, m_new, _, _ = compute_score_components(a_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        elif it % 200 == 0:
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.02
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                sc_t, _, _, _ = compute_score_components(a_trial)
                if sc_t < best_score:
                    best_score = sc_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 500 == 0:
            alpha = max(0.005, alpha * 0.95)

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True, seed=42):
    """
    Coordinate descent focusing on reducing the autocorrelation peak.
    For non-negative sequences, max_b = sum(a^2), so we want to
    minimize sum(a^2) subject to sum(a) = 1, which means making
    the sequence more uniform.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score, _, _, _ = compute_score_components(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # The key insight: for non-negative sequences, score = 2n*sum(a^2)/(sum(a))^2
    # To minimize this, we want to make a as uniform as possible.
    # Coordinate descent: pick a pair (i,j) where a[i] > a[j] and move
    # mass from i to j to reduce sum(a^2).
    
    stall = 0
    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break

        # Pick the largest and smallest elements
        max_idx = int(np.argmax(a))
        min_idx = int(np.argmin(a))
        
        if a[max_idx] - a[min_idx] < 1e-12:
            break  # Already uniform
        
        # Try moving mass from max to min
        delta = (a[max_idx] - a[min_idx]) * 0.1
        a_trial = a.copy()
        a_trial[max_idx] -= delta
        a_trial[min_idx] += delta
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        a_trial = a_trial / s
        
        score_trial, _, _, _ = compute_score_components(a_trial)
        if score_trial < best_score - 1e-14:
            a = a_trial
            best_score = score_trial
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Try random pair
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if a[i] > a[j]:
                delta = (a[i] - a[j]) * rng.uniform(0.01, 0.2)
            else:
                delta = (a[j] - a[i]) * rng.uniform(0.01, 0.2)
                i, j = j, i
            a_trial = a.copy()
            a_trial[i] -= delta
            a_trial[j] += delta
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial / s
                score_trial, _, _, _ = compute_score_components(a_trial)
                if score_trial < best_score - 1e-14:
                    a = a_trial
                    best_score = score_trial
                    best_a = a.copy()
                    stall = 0

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print("    CoordDescent it={} ({}s): J={}".format(it, round(elapsed, 1), round(best_score, 6)))

    return best_a, best_score

def simulated_annealing(a0, time_limit, verbose=True, seed=123):
    """SA with diverse moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score, _, _, _ = compute_score_components(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T = 0.002
    T_min = 0.00001
    T_decay = 0.9998
    cur_score, _, _, _ = compute_score_components(a)

    for it in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        if T < T_min:
            break

        move_type = rng.integers(3)
        if move_type == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = rng.uniform(0.001, 0.02)
            amt = min(a[i], a[j]) * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
        elif move_type == 1:
            # Swap
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            a_trial = a.copy()
            a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
        else:
            # Gaussian perturbation
            center = int(rng.integers(n))
            width = int(rng.integers(5, min(50, n // 4)))
            mask = np.zeros(n, dtype=bool)
            lo = max(0, center - width)
            hi = min(n, center + width)
            mask[lo:hi] = True
            pert = rng.normal(0, 0.005, n) * mask * a
            a_trial = a + pert

        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        score_trial, _, _, _ = compute_score_components(a_trial)

        if score_trial < cur_score:
            a = a_trial
            cur_score = score_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a_trial.copy()
        elif rng.random() < np.exp(-(score_trial - cur_score) / T):
            a = a_trial
            cur_score = score_trial

        T *= T_decay

        if it % 10000 == 0 and T < 0.001:
            T = 0.003

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print("    SA it={} T={} ({}s): J={}".format(it, round(T, 6), round(elapsed, 1), round(best_score, 6)))

    return best_a, best_score

import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_score(a):
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf'), max_b, sum_a
    score = 2.0 * n * max_b / (sum_a ** 2)
    return score, max_b, sum_a


def smooth_max_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max of convolution via softmax weighting."""
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()

    if b_max <= 0:
        return np.zeros(n), b

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    # Gradient of sum_k p_k * c_k where c_k = sum_j y[j]*y[k-j]
    # d/dy[i] of c_k = sum_j delta(i,j)*y[k-j] + delta(i,k-j)*y[j]
    # = y[k-i] + y[i-k] (where valid)
    # More efficiently: grad[i] = sum_k p_k * (y[k-i] + y[i-k]) if valid
    # This is equivalent to: grad = p * y (circular) + p * y[::-1] (circular) projected
    # Using FFT: convolve p with y, take appropriate slice
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Pf = np.fft.rfft(p, fft_size)
    Yf = np.fft.rfft(y, fft_size)
    # c_k = (y*y)[k], dc_k/dy[i] = y[k-i] + y[i-k] (for valid indices)
    # grad[i] = sum_k p_k * (y[k-i] + wrap)
    # = (p*y)[i] + (p*y_rev)[n-1-i] where * is linear convolution
    conv1 = np.fft.irfft(Pf * Yf, fft_size)[:size]
    y_rev = y[::-1]
    Yrf = np.fft.rfft(y_rev, fft_size)
    conv2 = np.fft.irfft(Pf * Yrf, fft_size)[:size]
    # grad[i] = conv1[i] + conv2[n-1-i] for i in [0, n-1]
    grad = conv1[:n] + conv2[n - 1::-1]
    grad = grad / beta
    return grad, b


def nesterov_optimize(a0, time_limit, beta_start=20.0, beta_end=800.0,
                       beta_growth=1.003, eta_init=0.5, top_k_ratio=0.03,
                       restart_every=1500, verbose=True, seed=42):
    n = len(a0)

    x_prev = x.copy()

    best_a = x.copy()

    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)

    eta = eta_init

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = smooth_max_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x
            score_new, _, _ = compute_score(x_new)
        else:
            eta_trial = eta
            found = False
            y_max = fast_autocorr(y).max()
            for _ in range(10):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                max_new = fast_autocorr(x_new).max()
                if max_new <= y_max + 1e-15:
                    found = True
                    break
                eta_trial *= 0.5

                x_new = x

                eta = max(eta_trial, 0.001)
            else:
                score_new, _, _ = compute_score(x_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 2.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 60 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:

                x_r = best_a + rng.normal(0, noise, n) * best_a

            elif strategy == 4:

            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                print("    Nest restart t={} ({}s): J={}".format(t, round(elapsed, 1), round(best_score, 6)))

def spectral_optimize(a0, time_limit, verbose=True, seed=777):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score, _, _ = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    alpha = 0.02

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        target = power[1:].mean() if len(power) > 1 else power[0]

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * (ratio - 1.0)
                A_new[k] *= max(0.3, np.sqrt(red))
            elif ratio < 0.5:
                inc = 1.0 + alpha * 0.3 * (0.5 - ratio)
                A_new[k] *= min(2.0, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        score_new, _, _ = compute_score(a_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        elif it % 200 == 0:
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.02
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() > 1e-15:
                a_trial = a_trial / a_trial.sum()
                sc_t, _, _ = compute_score(a_trial)
                if sc_t < best_score:
                    best_score = sc_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 500 == 0:
            alpha = max(0.005, alpha * 0.95)

    return best_a, best_score


def peak_targeting_cd(a0, time_limit, verbose=True, seed=42):
    """
    Coordinate descent targeting the peak of the convolution.
    For np.convolve(a,a), the peak is often at lag 2 for peaked sequences.
    We directly reduce the peak by redistributing mass.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score, _, _ = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        b = fast_autocorr(a)
        size = 2 * n - 1
        peak_lag = int(np.argmax(b))
        peak_val = b[peak_lag]

        # Find positions contributing to the peak
        # b[peak_lag] = sum_j a[j] * a[peak_lag - j] for valid j
        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        if js > je:
            stall += 1
            continue

        idx = np.arange(js, je + 1)
        partners = peak_lag - idx
        contrib = a[idx] * a[partners]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            break

        # Target: reduce peak to average level
        avg_level = b.mean()
        target_reduction = peak_val - avg_level
        if target_reduction <= 0:
            break

        # Find the pair with largest contribution and reduce it
        max_pi = int(np.argmax(contrib))
        i_src = int(idx[max_pi])
        i_partner = int(partners[max_pi])

        # Try moving mass from i_src to a position that doesn't contribute to the peak
        # Find a good destination: one where a[dst] * a[peak_lag - dst] is small
        # and dst is not a partner of any significant contributor
        move_amt = a[i_src] * min(0.1, target_reduction / max(peak_val, 1e-15))

        if move_amt < 1e-15:
            stall += 1
            continue

        # Try several destinations
        improved = False
        candidates = list(range(n))
        rng.shuffle(candidates)
        for dst in candidates[:min(20, n)]:
            if dst == i_src or (peak_lag >= 0 and peak_lag < size and 0 <= dst <= n - 1 and 0 <= peak_lag - dst < n and abs(dst - i_partner) < 2):
                continue

            a_trial = a.copy()
            a_trial[i_src] -= move_amt
            a_trial[dst] += move_amt
            if a_trial[i_src] < 0 or a_trial[dst] < 0:
                continue
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t

            score_trial, max_trial, _ = compute_score(a_trial)
            if score_trial < best_score - 1e-14:
                a = a_trial
                best_score = score_trial
                best_a = a.copy()
                stall = 0
                improved = True
                break

        if not improved:
            stall += 1
            # Random exploration
            if stall > 100:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    if a_trial.sum() > 1e-15:
                        a_trial = a_trial / a_trial.sum()
                        sc_t, _, _ = compute_score(a_trial)
                        if sc_t < best_score:
                            a = a_trial
                            best_score = sc_t
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print("    PeakCD ({}s): J={} stall={}".format(round(elapsed, 1), round(best_score, 6), stall))

    return best_a, best_score


def simulated_annealing(a0, time_limit, verbose=True, seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_score, _, _ = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    T = 0.005
    T_min = 0.00002
    T_decay = 0.9997
    cur_score, _, _ = compute_score(a)

    for it in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        if T < T_min:
            break

        move_type = rng.integers(4)
        if move_type == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = rng.uniform(0.001, 0.02)
            amt = min(a[i], a[j]) * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
        elif move_type == 1:
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            a_trial = a.copy()
            a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
        elif move_type == 2:
            center = int(rng.integers(n))
            width = int(rng.integers(5, min(50, n // 4)))
            mask = np.zeros(n, dtype=bool)
            lo = max(0, center - width)
            hi = min(n, center + width)
            mask[lo:hi] = True
            pert = rng.normal(0, 0.005, n) * mask * a
            a_trial = a + pert
        else:
            # Block move
            blk = int(rng.integers(5, min(30, n // 2)))
            start = int(rng.integers(max(1, n - blk)))
            a_trial = a.copy()
            block = a_trial[start:start + blk].copy()
            a_trial[start:start + blk] = a_trial[start + blk - 1:start - 1 + blk + 1 - blk + blk] if start + blk <= n else block
            # Simpler: shift a block
            a_trial = a.copy()
            a_trial[start:start+blk] = a_trial[start+1:start+blk+1] if start+blk < n else a_trial[start:start+blk] * 0.9

        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        score_trial, _, _ = compute_score(a_trial)

        if score_trial < cur_score:
            a = a_trial
            cur_score = score_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a_trial.copy()
        elif rng.random() < np.exp(-(score_trial - cur_score) / T):
            a = a_trial
            cur_score = score_trial

        T *= T_decay
        if it % 10000 == 0 and T < 0.001:
            T = 0.003

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print("    SA it={} T={} ({}s): J={}".format(it, round(T, 6), round(elapsed, 1), round(best_score, 6)))

    return best_a, best_score


def build_flat_core(n, tail_width, tail_power=2.0):
    """Build a sequence with a flat core and smooth tails to minimize convolution peak."""
    a = np.ones(n)
    tw = int(tail_width)
    if tw > 0 and tw < n // 2:
        t = np.arange(tw) / max(tw - 1, 1)
        envelope = t ** tail_power
        a[:tw] *= envelope
        a[-tw:] *= envelope[::-1]
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_length_sweep(a_best, time_limit, verbose=True):
    """Try different sequence lengths."""
    t0 = time.monotonic()
    base_n = len(a_best)
    best_a = a_best.copy()
    best_score, _, _ = compute_score(a_best)

    # Interpolated lengths
    lengths = [base_n, base_n + 200, base_n + 500, base_n + 1000,
               base_n - 500, base_n - 1000,
               3000, 5000, 7000, 9000, 11000, 13000, 15000, 20000, 25000]
    for new_n in lengths:
        if time.monotonic() - t0 > time_limit:
            break
        if new_n < 100 or new_n > 50000:
            continue
        a_try = interpolate_sequence(a_best, new_n)
        sc, _, _ = compute_score(a_try)
        if sc < best_score:
            best_score = sc
            best_a = a_try.copy()
            if verbose:
                print("    ML n={}: J={}".format(new_n, round(sc, 6)))

    # Flat-core constructions
    for fc_n in [5000, 8000, 10000, 15000, 20000, 30000]:
        if time.monotonic() - t0 > time_limit:
            break
        for tw in [50, 100, 200, 500]:
            if tw >= fc_n // 4:
                continue
            a_fc = build_flat_core(fc_n, tw)
            sc, _, _ = compute_score(a_fc)
            if sc < best_score:
                best_score = sc
                best_a = a_fc.copy()
                if verbose:
                    print("    FlatCore n={}, tw={}: J={}".format(fc_n, tw, round(sc, 6)))

    return best_a, best_score

                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={}, score={}".format(len(parent), round(p_score, 6)))
    except Exception:
        pass

    # Try flat-core constructions as initial candidates
    for n_try in [5000, 8000, 10000, 12000, 15000, 20000]:
        for tw in [50, 100, 200]:
            if tw >= n_try // 4:
                continue
            a_fc = build_flat_core(n_try, tw)
            sc = evaluate_safe(a_fc)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_fc.copy()
                print("  FlatCore n={}, tw={}: J={}".format(n_try, tw, round(sc, 6)))

    # Try uniform
    a_uni = np.ones(10000) / 10000
    sc_uni = evaluate_safe(a_uni)
    if np.isfinite(sc_uni) and sc_uni < best_score:
        best_score = sc_uni
        best_seq = a_uni.copy()
        print("  Uniform n=10000: J={}".format(round(sc_uni, 6)))

    if best_seq is None:
        best_seq = build_flat_core(8000, 100)
        best_score = evaluate_safe(best_seq)

    print("After init: n={}, J={}".format(len(best_seq), round(best_score, 6)))

    # Phase 1: Peak-targeting coordinate descent (fast)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: PeakCD (n={}) ===".format(len(best_seq)))
    if remaining > 15:
        t1 = min(remaining * 0.10, 120.0)
        a_out, _ = peak_targeting_cd(norm_sum1(best_seq), t1, verbose=True, seed=42)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  P1 improved: J={}".format(round(sc, 6)))

    # Phase 2: Nesterov (main optimizer)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P2: Nesterov (n={}) ===".format(len(best_seq)))
    if remaining > 30:
        t2 = min(remaining * 0.25, 350.0)
        a_out, _ = nesterov_optimize(norm_sum1(best_seq), t2,
                                     beta_start=20.0, beta_end=600.0,
                                     beta_growth=1.003, eta_init=0.5,
                                     top_k_ratio=0.03, restart_every=1500,
                                     verbose=True, seed=42)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  P2 improved: J={}".format(round(sc, 6)))

    # Phase 3: Spectral
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P3: Spectral (n={}) ===".format(len(best_seq)))
    if remaining > 20:
        t3 = min(remaining * 0.12, 150.0)
        a_out, _ = spectral_optimize(norm_sum1(best_seq), t3, verbose=True, seed=777)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print("  P3 improved: J={}".format(round(sc, 6)))
