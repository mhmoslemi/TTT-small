"""
Search for a nonnegative sequence minimizing
    2*n*max(np.convolve(a,a))/(sum(a)^2)
using smooth Frank-Wolfe-style optimization on the simplex, with:
  - FFT-based convolution and exact first-order gradients
  - top-band active set weights
  - multiplicative direction refinement approximating the LP-FW vertex
  - adaptive step sizes, temperature/active-set schedules
  - random smooth restarts and low-dimensional Fourier/parametric seeds
"""

import numpy as np
import time

try:
    from scipy import optimize as _scipy_opt
except Exception:
    _scipy_opt = None


def _nxt_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        n = max(1, len(a))
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _interp(a, new_n):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    old_n = len(a)
    new_n = int(max(1, new_n))
    if old_n == new_n:
        return a.copy()
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    a = a / s
    old_x = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    new_x = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    b = np.interp(new_x, old_x, a)
    b = np.maximum(b, 0.0)
    bs = float(np.sum(b))
    if bs <= 1e-300:
        return np.ones(new_n, dtype=np.float64) / new_n
    return b / bs


def _conv_max_N(a, N, n):
    A = np.fft.rfft(a, N)
    y = np.fft.irfft(A * A, N)
    m = float(np.max(y[:2 * n - 1]))
    if not np.isfinite(m):
        return 0.0
    return max(0.0, m)


def _score_fast_N(a, n, N):
    a = np.asarray(a, dtype=np.float64)
    return 2.0 * n * _conv_max_N(a, N, n)


def _score_fast(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    n = len(a)
    if n <= 0:
        return float("inf")
    N = _nxt_pow2(2 * n - 1)
    return _score_fast_N(a, n, N)


def _update_best(best_a, best_score, a):
    if a is None:
        return best_a, best_score
    a = _normalize(a)
    sc = _score_fast(a)
    if np.isfinite(sc) and sc < best_score:
        return a.copy(), sc
    return best_a, best_score


def _top_band_weights(y, m, theta, n):
    L = len(y)
    if L <= 0 or m <= 1e-300:
        q = np.ones(L, dtype=np.float64)
        if L:
            q /= L
        return q

    band = theta * m
    if band <= 1e-300:
        K = max(1, min(L, 16))
        idx = np.argpartition(y, -K)[-K:]
        q = np.zeros(L, dtype=np.float64)
        q[idx] = 1.0 / K
        return q

    lo = m - band
    q = np.maximum(y - lo, 0.0)
    q /= band
    s = float(np.sum(q))
    if s <= 1e-300:
        K = max(1, min(L, max(10, int(0.002 * max(1, n)))))
        idx = np.argpartition(y, -K)[-K:]
        q = np.zeros(L, dtype=np.float64)
        q[idx] = 1.0 / K
        return q
    q /= s
    return q


def _smooth_perturb(a, rng, rel_scale=1.0, K=4):
    a = _normalize(a)
    n = len(a)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    p = np.zeros(n, dtype=np.float64)
    base = max(1.0 / n, 0.25 * float(np.max(a)))
    for k in range(1, int(max(1, K)) + 1):
        amp = rng.normal(0.0, rel_scale * base / max(1.0, float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        p += amp * np.cos(2.0 * np.pi * k * x + ph)
    out = a + p
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s <= 1e-300:
        return a.copy()
    return out / s


def _linear_update(a, v, mu):
    if mu <= 0.0:
        return a.copy()
    if mu >= 1.0:
        return v.copy()
    out = (1.0 - mu) * a + mu * v
    out = np.maximum(out, 0.0)
    s = float(np.sum(out))
    if s <= 1e-300:
        return v.copy()
    return out / s


def _smooth_frank_wolfe(a0, time_limit, theta_start=0.10, theta_min=0.004,
                        mu_start=0.05, max_mu=0.30, focus=1.5,

    n = len(a)
    size = 2 * n - 1
    N_y = _nxt_pow2(size)
    N_g = _nxt_pow2(3 * n - 2)

    phase = np.exp((-2j * np.pi / N_g) * (np.arange(N_g // 2 + 1) * (n - 1)))

    best_a = a.copy()
    best_m = _conv_max_N(a, N_y, n)
    best_score = 2.0 * n * best_m

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    theta = float(theta_start)
    mu = float(mu_start)
    no_current = 0
    global_stall = 0
    it = 0

    tol = 1e-11

    while True:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        if it > 2000000:
            break

        A_y = np.fft.rfft(a, N_y)
        y = np.fft.irfft(A_y * A_y, N_y)[:size]
        m = float(np.max(y))
        if not np.isfinite(m) or m <= 1e-300:
            break
        cur_score = 2.0 * n * m

        if cur_score < best_score - tol:
            best_score = cur_score
            best_a = a.copy()
            global_stall = 0
        else:
            global_stall += 1

        A_g = np.fft.rfft(a, N_g)
        AR = phase * np.conj(A_g)

        q = _top_band_weights(y, m, theta, n)
        Wq = np.fft.rfft(q, N_g)
        convq = np.fft.irfft(Wq * AR, N_g)
        g = 2.0 * convq[n - 1:n + n]
        g = np.maximum(g, 0.0)
        gmax = float(np.max(g))

        if not np.isfinite(gmax) or gmax <= 1e-300:
            theta = max(theta_min, theta * 0.7)
            mu = mu_start
            no_current = 0
            it += 1
            continue

        g = g / gmax
        ev = np.exp(-focus * g)
        ev = np.maximum(ev, 1e-300)
        es = float(np.sum(ev))
        if es <= 1e-300 or not np.isfinite(es):
            ev = np.ones(n, dtype=np.float64) / n
        else:
            ev = ev / es

        g_dir = ev.copy()

        if dir_refine > 0 and refine_every > 0 and (it % max(1, refine_every) == 0):
            theta_dir = max(theta_min, min(0.15, theta))
            dir_eta = 1.0
            for rr in range(int(dir_refine)):
                G_y = np.fft.rfft(g_dir, N_y)
                h = np.fft.irfft(A_y * G_y, N_y)[:size]
                hm = float(np.max(h))
                if not np.isfinite(hm) or hm <= 1e-300:
                    break
                qd = _top_band_weights(h, hm, theta_dir, n)
                Wd = np.fft.rfft(qd, N_g)
                convd = np.fft.irfft(Wd * AR, N_g)
                d = np.maximum(convd[n - 1:n + n], 0.0)
                dmax = float(np.max(d))
                if not np.isfinite(dmax) or dmax <= 1e-300:
                    break
                d = d / dmax
                gd = g_dir * np.exp(-dir_eta * d)
                gd = _normalize(gd)
                g_dir = gd

        v = _normalize((1.0 - vertex_blend) * a + vertex_blend * g_dir)

        trial_mu = min(max_mu, max(1e-4, mu))
        a_t = _linear_update(a, v, trial_mu)
        sc_t = _score_fast_N(a_t, n, N_y)
        accepted = False

        if sc_t < cur_score - tol:
            accepted = True
        else:
            hmu = max(1e-5, 0.5 * trial_mu)
            if hmu < trial_mu:
                a_t2 = _linear_update(a, v, hmu)
                sc_t2 = _score_fast_N(a_t2, n, N_y)
                if sc_t2 < cur_score - tol:
                    a_t = a_t2
                    sc_t = sc_t2
                    trial_mu = hmu
                    accepted = True

        if accepted:
            a = a_t
            if sc_t < cur_score - 1e-10:
                mu = min(max_mu, mu * 1.12)
            else:
                mu = max(1e-5, mu * 0.9)
            no_current = 0

            if sc_t < best_score - tol:
                best_score = sc_t
                best_a = a.copy()
                global_stall = 0

            if (it + 1) % 25 == 0 and time.monotonic() - t0 < time_limit:
                big_mu = min(max_mu, 2.0 * mu)
                if big_mu > trial_mu + 1e-5:
                    a_big = _linear_update(a, v, big_mu)
                    sc_big = _score_fast_N(a_big, n, N_y)
                    if sc_big < sc_t - tol:
                        a = a_big
                        sc_t = sc_big
                        mu = big_mu
                        if sc_big < best_score - tol:
                            best_score = sc_big
                            best_a = a.copy()
                            global_stall = 0
        else:
            no_current += 1
            if no_current > 10:
                mu = max(1e-5, mu * 0.5)
            if no_current > 90:
                if theta < 0.07 and global_stall < 180:
                    theta = min(theta_start * 0.9, max(theta_min, 1.8 * theta))
                elif theta > theta_min * 1.02:
                    theta = max(theta_min, theta * 0.7)
                else:
                    a = _smooth_perturb(best_a, rng, rel_scale=2.0, K=5)
                    theta = max(theta_min, min(theta_start * 0.6, 0.10))
                mu = mu_start
                no_current = 0
                global_stall = 0

        if global_stall > 700:
            if theta > theta_min * 1.05:
                theta = max(theta_min, theta * 0.7)
            else:
                a = _smooth_perturb(best_a, rng, rel_scale=2.5, K=6)
                theta = max(theta_min, min(theta_start * 0.5, 0.09))
            mu = mu_start
            no_current = 0
            global_stall = 0

        if verbose and (it == 0 or (it % 100 == 0)):
            print("  {} it={} t={:.0f}s best={:.7f} cur={:.7f} theta={:.4f} mu={:.4f}".format(
                prefix, it, time.monotonic() - t0, best_score, cur_score, theta, mu))

        it += 1

    if np.isfinite(best_score):
        return best_a, best_score
    return a, 2.0 * n * _conv_max_N(a, N_y, n)


def _build_fourier_basis(n, L):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    basis = [np.ones(n, dtype=np.float64)]
    for k in range(1, int(max(1, L)) + 1):
        ang = 2.0 * np.pi * k * x
        basis.append(np.cos(ang))
        basis.append(np.sin(ang))
    return basis


def _fourier_params_to_a(p, basis):
    p = np.asarray(p, dtype=np.float64)
    if len(p) == 0 or len(p) != len(basis):
        n = len(basis[0]) if len(basis) else 1
        return np.ones(n, dtype=np.float64) / max(1, n)
    phi = p[0] * basis[0]
    for i in range(1, len(p)):
        phi = phi + p[i] * basis[i]
    if not np.all(np.isfinite(phi)):
        n = len(basis[0])
        return np.ones(n, dtype=np.float64) / max(1, n)
    phi = phi - float(np.max(phi))
    if phi.min() < -700.0:
        phi = np.maximum(phi, -700.0)
    a = np.exp(phi)
    s = float(np.sum(a))
    if s <= 1e-300:
        n = len(basis[0])
        return np.ones(n, dtype=np.float64) / max(1, n)
    return a / s


def _fourier_obj_params(p, basis, N, theta):
    p = np.asarray(p, dtype=np.float64)
    if p.ndim != 1 or len(p) != len(basis):
        return 1e100
    if not np.all(np.isfinite(p)):
        return 1e100

    phi = p[0] * basis[0]
    for i in range(1, len(p)):
        phi = phi + p[i] * basis[i]

    dy = float(np.max(phi) - np.min(phi))
    if dy > 90.0 or not np.isfinite(dy):
        return 1e80

    phi = phi - float(np.max(phi))
    a = np.exp(phi)
    s = float(np.sum(a))
    if s <= 1e-300 or not np.isfinite(s):
        return 1e100
    a = a / s
    n = len(a)

    A = np.fft.rfft(a, N)
    y = np.fft.irfft(A * A, N)[:2 * n - 1]
    m = float(np.max(y))
    if not np.isfinite(m) or m <= 1e-300:
        return 1e100

    tau = theta * m
    if tau <= 1e-300:
        return 2.0 * n * m

    z = (y - m) / tau
    z = np.clip(z, -50.0, 0.0)
    s2 = float(np.sum(np.exp(z)))
    if s2 <= 1e-300 or not np.isfinite(s2):
        s2 = 1.0
    return float(2.0 * n * (m + tau * np.log(s2)))


def _run_fourier_opt(n_eval, L, num_restarts, time_limit, seed, verbose=False):
    if _scipy_opt is None:
        return None, float("inf")

    t0 = time.monotonic()
    basis = _build_fourier_basis(n_eval, L)
    N = _nxt_pow2(2 * n_eval - 1)
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = float("inf")
    last_res_fun = float("inf")

    for r in range(int(max(1, num_restarts))):
        if time.monotonic() - t0 > time_limit:
            break

        if r == 0:
            p0 = np.zeros(len(basis), dtype=np.float64)
            p0[0] = 0.0
        else:
            scale = 0.35 / (0.5 + 0.5 * float(r))
            p0 = rng.normal(0.0, scale, size=len(basis))
            p0 = np.clip(p0, -2.0, 2.0)
            p0[0] = rng.normal(0.0, 0.2)

        try:
            res = _scipy_opt.minimize(
                _fourier_obj_params,
                p0,
                args=(basis, N, 0.05),
                method="Nelder-Mead",
                options={"maxiter": 350, "xatol": 1e-5, "fatol": 1e-9, "adaptive": True}
            )
        except Exception:
            continue

        if res is None:
            continue

        a = _fourier_params_to_a(res.x, basis)
        sc = _score_fast(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            last_res_fun = float(res.fun if hasattr(res, "fun") else np.nan)

        if verbose:
            print("    FourOpt r={} fun={} maxscore={:.7f}".format(
                r,
                last_res_fun if np.isfinite(last_res_fun) else float("nan"),
                best_score))

    return best_a, best_score


def _random_fourier_a(n, rng, L=5, amp=0.8, offset_scale=0.15):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    phi = np.full(n, rng.normal(0.0, offset_scale), dtype=np.float64)
    for k in range(1, int(max(1, L)) + 1):
        ang = 2.0 * np.pi * k * x
        c = rng.normal(0.0, amp / max(1.0, float(k)))
        s = rng.normal(0.0, amp / max(1.0, float(k)))
        phi = phi + c * np.cos(ang) + s * np.sin(ang)
    phi = phi - float(np.max(phi))
    if phi.min() < -700.0:
        phi = np.maximum(phi, -700.0)
    a = np.exp(phi)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _generate_initial(parent, n, seed, random_count=80, verbose=False):
    n = int(max(2, n))
    uniform = np.ones(n, dtype=np.float64) / n
    best_a, best_score = uniform, _score_fast(uniform)
    if verbose:
        print("  init uniform: {0:.7f}".format(best_score))

    if parent is not None:
        a = _interp(parent, n)
        best_a, best_score = _update_best(best_a, best_score, a)
        if verbose:
            print("  init parent: {0:.7f}".format(best_score))

        if n >= 512:
            half = max(64, n // 4)
            a2 = _interp(_interp(a, half), n)
            best_a, best_score = _update_best(best_a, best_score, a2)
            if verbose:
                print("  init smoothed parent: {0:.7f}".format(best_score))

    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    xp = np.maximum(x, 1e-12)
    xm = np.maximum(1.0 - x, 1e-12)

    alphas = [0.2, 0.35, 0.5, 0.75, 1.0, 1.5, 2.2, 3.0]
    for aa in alphas:
        for bb in alphas:
            a = (xp ** aa) * (xm ** bb)
            a = float(np.sum(a))
            if not np.isfinite(a) or a <= 1e-300:
                continue
            a = (xp ** aa) * (xm ** bb)
            a = a / float(np.sum(a))
            best_a, best_score = _update_best(best_a, best_score, a)

    for c in [-1.2, -0.7, -0.35, 0.35, 0.7, 1.2, 2.0, 3.0]:
        a = 1.0 + c * x
        a = np.maximum(a, 1e-6)
        a = a / float(np.sum(a))
        best_a, best_score = _update_best(best_a, best_score, a)

    rng = np.random.default_rng(seed)
    for i in range(int(max(0, random_count))):
        L = 3 + (i % 4)
        amp = 0.5 + 0.15 * float(i % 6)
        a = _random_fourier_a(n, rng, L=L, amp=amp)
        best_a, best_score = _update_best(best_a, best_score, a)

    if verbose:
        print("  init best: {0:.7f}".format(best_score))

    return best_a, best_score
