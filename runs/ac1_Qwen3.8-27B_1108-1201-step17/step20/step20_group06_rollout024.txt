        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgrad_exact(y, peak_set, fft_size):
    """Compute subgradient of max using explicit peak set."""
    n = len(y)
    size = 2 * n - 1
    K = len(peak_set)
    if K == 0:
        return np.zeros(n)
    weights = np.ones(K) / K
    ar = np.asarray(y[::-1], dtype=np.float64)
    pr = np.zeros(size, dtype=np.float64)
    for i, k in enumerate(peak_set):
        pr[int(k)] = weights[i]
    Ar = np.fft.rfft(ar, fft_size)
    Pr = np.fft.rfft(pr, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    return conv[n - 1:2 * n - 1] / K


def compute_smoothed_grad_v2(y, beta, top_k=None, fft_size=None):
    """Compute gradient of smoothed max with better numerical stability."""
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1
    if fft_size is None:
        fft_size = 1 << (size - 1).bit_length()

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()

    if b_max <= 0:
        return np.zeros(n), b, b_max

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -800, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    ar = np.asarray(y[::-1], dtype=np.float64)
    Ar = np.fft.rfft(ar, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]
    return grad, b, b_max


def lp_fw_v3(a0, time_limit, solver, initial_K=400, max_K=3000,
             add_per_iter=30, reselect_every=200, verbose=True):
    """
    Improved LP-Frank-Wolfe with:
    - Periodic full reselection of active set
    - Entropic step size option
    - Better violation detection
    - Multi-candidate line search
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    y = a.copy()
    ac_y = fast_autocorr(y)
    M_curr = ac_y.max()
    active_set = list(np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 800
    t_fw = 1.0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic full reselection
        if t % reselect_every == 0:
            active_set = list(np.argsort(ac)[::-1][:max_K])
            if len(active_set) > max_K:
                active_set = active_set[:max_K]

        # Trim active set if too large
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            order = np.argsort(margins)
            active_set = [active_set[i] for i in order[:max_K]]

        K = len(active_set)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        valid_rows = 0
        for jj, k in enumerate(active_set):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
                valid_rows += 1

        if valid_rows == 0:
            break

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs[:valid_rows] @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = list(np.argsort(ac)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violations and add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_k = set(int(k) for k in top_v) - set(active_set)
            active_set.extend(list(new_k))

        # Line search with multiple candidates
        best_alpha = 2.0 / (t_fw + 2.0)
        best_trial_M = M_curr
        candidate_alphas = [best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                           1.0 / (t_fw + 1), 0.03, 0.05, 0.1, 0.15, 0.2, 0.3]
        # Entropic step size
        pos_y = y[y > 1e-15]
        pos_g = g_dir[g_dir > 1e-15]
        if len(pos_y) > 1 and len(pos_g) > 1:
            ent_y = -np.sum(pos_y * np.log(pos_y))
            ent_g = -np.sum(pos_g * np.log(pos_g))
            ent_alpha = max(0.01, min(1.0, (ent_g - ent_y) / 2.0 + 0.1))
            candidate_alphas.append(ent_alpha)

        for alpha in candidate_alphas:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new
        t_fw += 2.0 / (t_fw)  # slowly increase t_fw

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LPv3 it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " |S|=" + str(len(active_set)) + " alpha=" + str(round(best_alpha, 4)))

    return best_a, best_m


def nesterov_v4(a0, time_limit, beta_start=30.0, beta_end=2000.0,
                beta_growth=1.004, eta_init=0.4, top_k_ratio=0.04,
                restart_every=1200, verbose=True, seed=42):
    """
    Nesterov with:
    - Adaptive beta (faster growth when stalled)
    - Subgradient extraction for final polishing
    - Better restart strategies
    - Fogel cooldown
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    top_k = max(20, int(n * top_k_ratio))
    eta = eta_init

    # Phase 1: Smoothed gradient descent
    phase1_end = time_limit * 0.7
    phase2_start = time_limit * 0.7

    for t in range(1, 100000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        if elapsed > phase2_start:
            # Phase 2: Exact subgradient
            b = fast_autocorr(x)
            b_max = b.max()
            threshold = b_max - (b_max - b.min()) * 0.02
            peak_set = list(np.where(b >= threshold)[0])
            if len(peak_set) < 5:
                peak_set = list(np.argsort(b)[::-1][:20])
            grad = compute_subgrad_exact(x, peak_set, fft_size)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                for step in [eta, eta * 0.5, eta * 0.25]:
                    z = x - step * grad
                    x_new = simplex_project(z)
                    M_new = fast_autocorr(x_new).max()
                    if M_new < best_max_b:
                        best_max_b = M_new
                        best_a = x_new.copy()
                        x = x_new
                        x_prev = x_new.copy()
                        break
            if t % 500 == 0:
                x = 0.95 * x + 0.05 * rng.dirichlet(np.ones(n) * 0.5)
                x = x / x.sum()
            continue

        # Phase 1: Smoothed
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b, b_max = compute_smoothed_grad_v2(y, beta, top_k, fft_size)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x
            max_new = b_max
        else:
            found = False
            eta_trial = eta
            y_max = b_max
            for _ in range(8):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                max_new = fast_autocorr(x_new).max()
                if max_new <= y_max + 1e-14:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                max_new = b_max
                eta = max(0.01, eta_trial)

        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 1.5)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Adaptive beta growth
        if t % 40 == 0:
            if stall > 200:
                beta = min(beta * 1.02, beta_end)
            else:
                beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                noise_level = 0.015 + 0.01 * rng.random()
                pert = rng.normal(0, noise_level, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.2)
            elif strategy == 2:
                x_r = best_a * np.exp(0.02 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= (0.6 + 0.2 * rng.random())
                x_r[~mask] *= (1.2 + 0.2 * rng.random())
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 15)))
                x_r = np.roll(best_a, shift)
            else:
                # Split and recombine
                half = n // 2
                x_r = best_a.copy()
                x_r[:half] *= rng.uniform(0.5, 1.5)
                x_r[half:] *= rng.uniform(0.5, 1.5)
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 15.0)
            eta = eta_init
            if verbose:
                J = 2.0 * n * best_max_b
                print("    Nest v4 restart (" + str(round(time.monotonic() - t0, 1)) +
                      "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            J = 2.0 * n * best_max_b
            print("    Nest v4 t=" + str(t) + " (" + str(round(time.monotonic() - t0, 1)) +
                  "s): J=" + repr(round(J, 6)) + " beta=" + str(int(beta)) + " stall=" + str(stall))

    return best_a, best_max_b


def spectral_v3(a0, time_limit, verbose=True, seed=777):
    """Spectral refinement with adaptive threshold and multi-scale approach."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    alpha = 0.015
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        target = np.median(power[1:]) if len(power) > 1 else power[0]

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * min(ratio - 1.0, 5.0)
                A_new[k] *= max(0.2, np.sqrt(red))
            elif ratio < 0.3:
                inc = 1.0 + alpha * 0.5 * (0.3 - ratio)
                A_new[k] *= min(3.0, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 300 == 0:
            # Phase randomization
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.03
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 400 == 0:
            alpha = max(0.003, alpha * 0.96)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral v3 it=" + str(it) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def mass_redistribute_v2(a0, time_limit, verbose=True, seed=456):
    """
    Local search focused on reducing the top peaks by redistributing mass
    from contributing positions to less impactful ones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    size = 2 * n - 1
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:min(30, size)]
        improved = False

        for k in top_k[:15]:
            if elapsed > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            # Find the position contributing most to this peak
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:5]]

            for src in top_contrib:
                src = int(src)
                # Try moving mass from src to random positions
                for _ in range(8):
                    if elapsed > time_limit:
                        break
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    # Also check if dst would create a new peak
                    fracs = [0.02, 0.05, 0.1]
                    for frac in fracs:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Broader random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * rng.uniform(0.005, 0.03)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 500 == 0 and stall > 0:
            J = 2.0 * n * best_m
            print("    MassRedist (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def coordinate_descent_v2(a0, time_limit, verbose=True, seed=789):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes max_b given all other coordinates fixed.
    Uses randomized coordinate ordering and batch updates.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    ac = fast_autocorr(a)
    M_curr = ac.max()

    # Focus on coordinates that contribute to top peaks
    top_peaks = np.argsort(ac)[::-1][:30]
    # Find which coordinates contribute to these peaks
    important_coords = set()
    for k in top_peaks:
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(n, k_int + 1)
        for j in range(js, je):
            important_coords.add(j)
    important_list = list(important_coords)
    if len(important_list) > 500:
        important_list = list(rng.choice(len(important_list), 500, replace=False))

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Randomly shuffle important coordinates
        rng.shuffle(important_list)

        for i in important_list[:100]:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(i)
            if a[i] < 1e-15:
                continue

            # Binary search for optimal value of a[i]
            # We want to minimize max_b when a[i] changes
            # b[k] changes by: delta * (a[k-i] + a[i_old]) for relevant k
            # This is complex, so use a simple try-and-accept
            old_val = a[i]
            candidates = [old_val * 0.5, old_val * 0.7, old_val * 1.3, old_val * 1.5]
            best_trial_m = M_curr
            best_trial_val = old_val

            for trial_val in candidates:
                trial_val = max(0.0, trial_val)
                a_trial = a.copy()
                a_trial[i] = trial_val
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_trial_m:
                    best_trial_m = M_trial
                    best_trial_val = trial_val

            if best_trial_val != old_val:
                a[i] = best_trial_val
                a = a / a.sum()
                ac = fast_autocorr(a)
                M_curr = ac.max()
                if M_curr < best_m - 1e-15:
                    best_m = M_curr
                    best_a = a.copy()

        # Periodically re-select important coordinates
        if it % 50 == 0:
            top_peaks = np.argsort(ac)[::-1][:30]
            important_coords = set()
            for k in top_peaks:
                k_int = int(k)
                js = max(0, k_int - n + 1)
                je = min(n, k_int + 1)
                important_coords.update(range(js, je))
            important_list = list(important_coords)
            if len(important_list) > 500:
                rng.shuffle(important_list)
                important_list = important_list[:500]
            else:
                # Add some random coordinates for exploration
                extra = list(rng.choice(n, min(100, n - len(important_list)), replace=False))
                important_list.extend([int(x) for x in extra if x not in important_coords])

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordDesc it=" + str(it) + " (" + str(round(elapsed, 1)) +
                  "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) +
              " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization with emphasis on LP-FW and adaptive Nesterov.
    Uses multi-length exploration and coordinated local search.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 7000
        x = np.linspace(0, 1, n_base)
        # Initialize with a shape that has low autocorrelation peak
        # Use a "triangle" shape (concentrated in middle) or "U-shape"
        # U-shape tends to have lower cross-correlation peak
        init = 0.5 * (1 + np.cos(2 * np.pi * x))
        init = init ** 1.5  # Sharpen the U-shape
        best_seq = norm_sum1(init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + repr(round(best_score, 6)))

    # Phase 1: Quick LP-FW to establish good starting point
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: LP-FW init (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 20:
        a_out, _ = lp_fw_v3(norm_sum1(best_seq), t1, solver,
                            initial_K=300, max_K=2000,
                            add_per_iter=25, reselect_every=150, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_init", t_global)

    # Phase 2: Nesterov with aggressive parameters
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: Nesterov v4 (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.20, 320.0)
    if t2 > 40:
        a_out, _ = nesterov_v4(norm_sum1(best_seq), t2,
                               beta_start=30.0, beta_end=1500.0,
                               beta_growth=1.004, eta_init=0.4,
                               top_k_ratio=0.04, restart_every=1200,
                               verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: LP-FW main (most important phase)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: LP-FW main (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.22, 380.0)
    if t3 > 30:
        a_out, _ = lp_fw_v3(norm_sum1(best_seq), t3, solver,
                            initial_K=500, max_K=3500,
                            add_per_iter=35, reselect_every=100, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_main", t_global)

    # Phase 4: Nesterov with high beta (polishing)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: Nesterov polish (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 25:
        a_out, _ = nesterov_v4(norm_sum1(best_seq), t4,
                               beta_start=100.0, beta_end=3000.0,
                               beta_growth=1.005, eta_init=0.2,
                               top_k_ratio=0.02, restart_every=800,
                               verbose=True, seed=999)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_polish", t_global)

    # Phase 5: Coordinate descent on the best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: CoordDesc (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        a_out, _ = coordinate_descent_v2(norm_sum1(best_seq), t5, verbose=True, seed=789)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord1", t_global)

    # Phase 6: Mass redistribution
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: MassRedist (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.07, 80.0)
    if t6 > 15:
        a_out, _ = mass_redistribute_v2(norm_sum1(best_seq), t6, verbose=True, seed=456)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "mass1", t_global)

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: Multi-length (base n=" + str(n_best) + ") ===")

    # Try a few different lengths around the current best
    length_candidates = []
    for delta in [-1000, -500, 0, 500, 1000, 2000]:
        candidate_n = n_best + delta
        if 3000 <= candidate_n <= 15000 and candidate_n != n_best:
            length_candidates.append(candidate_n)

    for new_n in length_candidates[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print("  -> Trying n=" + str(new_n) + " (remaining=" + str(round(remaining, 1)) + "s)")

        a_long = interpolate_sequence(best_seq, new_n)

        # LP-FW on new length
        t7a = min(remaining * 0.35, 200.0)
        if t7a > 20 and new_n <= 12000:
            a_out, _ = lp_fw_v3(a_long, t7a, solver,
                                initial_K=300, max_K=2000,
                                add_per_iter=20, reselect_every=150, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

        # Quick Nesterov
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.30, 120.0)
            if t7b > 15:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = nesterov_v4(a_long2, t7b,
                                       beta_start=50.0, beta_end=1000.0,
                                       beta_growth=1.005, eta_init=0.3,
                                       top_k_ratio=0.03, restart_every=1000,
                                       verbose=False, seed=new_n % 7919)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

    # Phase 8: Final LP-FW on best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P8: LP-FW final (n=" + str(n_best) + ") ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 20 and n_best <= 12000:
        a_out, _ = lp_fw_v3(norm_sum1(best_seq), t8, solver,
                            initial_K=600, max_K=4000,
                            add_per_iter=40, reselect_every=80, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 9: Final Nesterov polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P9: Nesterov final (n=" + str(n_best) + ") ===")
    t9 = min(remaining * 0.20, 150.0)
    if t9 > 15:
        a_out, _ = nesterov_v4(norm_sum1(best_seq), t9,
                               beta_start=200.0, beta_end=5000.0,
                               beta_growth=1.006, eta_init=0.15,
                               top_k_ratio=0.01, restart_every=600,
                               verbose=True, seed=12345)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 10: Quick spectral + mass redistribute
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P10: Final touches (n=" + str(n_best) + ") ===")
    t10a = min(remaining * 0.25, 60.0)
    if t10a > 10:
        a_out, _ = spectral_v3(norm_sum1(best_seq), t10a, verbose=False, seed=888)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t10b = min(remaining * 0.40, 60.0)
        if t10b > 8:
            a_out, _ = mass_redistribute_v2(norm_sum1(best_seq), t10b, verbose=False, seed=321)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "mass_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) +
              ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        init = 0.5 * (1 + np.cos(2 * np.pi * x))
        best_seq = norm_sum1(init ** 1.5)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
