"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using projected subgradient
with Nesterov acceleration, exact active set, and peak shaving.
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_subgrad(y, top_k=20):
    n = len(y)
    b = fast_autocorr(y)
    idx = np.argsort(b)[::-1][:top_k]
    b_max = b[idx[0]]
    # Inverse-gap weights (emphasize most active lags)
    gaps = b_max - b[idx] + 1e-12
    w = 1.0 / gaps
    w /= w.sum()
    grad = np.zeros(n, dtype=np.float64)
    for jj in range(len(idx)):
        k = int(idx[jj])
        wj = w[jj]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i_vals = np.arange(i_start, i_end + 1)
            grad[i_vals] += 2.0 * wj * y[k - i_vals]
    return grad, b, b_max

def nesterov_subgrad(a0, time_limit, top_k=20, verbose=True, seed=42):
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()  # Nesterov extrapolation point
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    t = 0
    t_nest = 1.0
    m = np.zeros(n, dtype=np.float64)  # momentum
    beta = 0.9  # momentum coefficient
    step = 1.0 / (n ** 0.5)
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - y)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute gradient
        g, b, b_max = compute_subgrad(y, top_k)
        
        # Momentum update
        m = beta * m + (1.0 - beta) * g
        
        # Backtracking line search
        alpha = step
        found = False
        for _ in range(20):
            z = y - alpha * m
            x_new = simplex_project(z)
            m_new = fast_autocorr(x_new).max()
            if m_new < b_max - 1e-15 * alpha * (m @ m):
                found = True
                break
            alpha *= 0.5
        
        if not found:
            # Accept anyway if it's not too bad
            x_new = simplex_project(y - alpha * m)
            m_new = fast_autocorr(x_new).max()
        
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_x = x_new.copy()
            stall = 0
            # Slowly decrease momentum
            beta = max(0.7, beta * 0.999)
        else:
            stall += 1
        
        x = x_new
        t_nest = t_new
        
        # Periodic restart
        if stall > max_stall:
            stall = 0
            t_nest = 1.0
            m = np.zeros(n, dtype=np.float64)
            beta = 0.9
            noise_scale = 0.01 / (1.0 + 0.1 * (t // max_stall))
            pert = rng.normal(0, noise_scale, n)
            x_r = best_x * (1.0 + pert)
            x_r = np.maximum(x_r, 0.0)
            s_r = x_r.sum()
            if s_r > 1e-15:
                x = x_r / s_r
            y = x.copy()
            step = 1.0 / (n ** 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    NS restart t={t} ({elapsed:.1f}s): J={J:.6f}")
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NS t={t} ({elapsed:.1f}s): J={J:.6f} stall={stall}")
    
    return best_x, best_m

def peak_shaving(a0, time_limit, verbose=True, seed=99):
    """Direct peak shaving: move mass away from pairs contributing to peak lag."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        b_peak = b[k_peak]
        
        # Find pairs contributing to k_peak
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        i_vals = np.arange(i_start, i_end + 1)
        contrib = a[i_vals] * a[k_peak - i_vals]
        
        if contrib.max() < 1e-20:
            break
        
        # Identify top contributors
        top_contrib_idx = i_vals[np.argsort(contrib)[::-1][:10]]
        
        # Find target positions (those not contributing to any top-5 lag)
        top_lags = np.argsort(b)[::-1][:5]
        in_peak = np.zeros(n, dtype=bool)
        for k in top_lags:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            in_peak[s:e+1] = True
        
        candidates = np.where(~in_peak)[0]
        if len(candidates) < 5:
            candidates = np.where(a < np.median(a))[0]
        
        if len(candidates) == 0:
            stall += 1
            continue
        
        improved = False
        for src in top_contrib_idx:
            if a[src] < 1e-12:
                continue
            for _ in range(3):
                dst = int(candidates[rng.integers(len(candidates))])
                if dst == src:
                    continue
                frac = float(rng.choice([0.002, 0.005, 0.01, 0.02, 0.05]))
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = float(rng.uniform(0.001, 0.02))
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.1f}s): J={J:.6f} stall={stall}")
    
    return best_a, best_m

def cyclic_shift_best(a0):
    n = len(a0)
    a = norm_sum1(a0)
    best_m = fast_autocorr(a).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a, -s)
        m = fast_autocorr(a_shift).max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a, -best_shift), best_m
    return a.copy(), best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated projected subgradient with exact active-set gradient
2. Armijo backtracking line search
3. Peak-shaving local search with contribution-aware moves
4. Multi-length exploration with interpolation
5. Cyclic shift optimization
6. Periodic noise-based restarts to escape local minima

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_active_subgrad(y, top_k=20):
    n = len(y)
    b = fast_autocorr(y)
    idx = np.argsort(b)[::-1][:top_k]
    b_max = b[idx[0]]
    gaps = b_max - b[idx] + 1e-12
    w = 1.0 / gaps
    w /= w.sum()
    grad = np.zeros(n, dtype=np.float64)
    for jj in range(len(idx)):
        k = int(idx[jj])
        wj = w[jj]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i_vals = np.arange(i_start, i_end + 1)
            grad[i_vals] += 2.0 * wj * y[k - i_vals]
    return grad, b, b_max


def nesterov_subgrad(a0, time_limit, top_k=20, verbose=True, seed=42,
                     beta_mom=0.85, initial_step=None):
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best_x = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    t = 0
    t_nest = 1.0
    m = np.zeros(n, dtype=np.float64)
    beta = beta_mom
    if initial_step is None:
        step = 5.0 / (n ** 0.8)
    else:
        step = initial_step
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit:
        t += 1
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - y)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        g, b, b_max = compute_active_subgrad(y, top_k)
        m = beta * m + (1.0 - beta) * g

        alpha = step
        x_new = None
        m_new = None
        for _ in range(15):
            z = y - alpha * m
            x_trial = simplex_project(z)

            best_x = x_new.copy()
            stall = 0
        else:

        t_nest = t_new

    return best_x, best_m


def peak_shaving(a0, time_limit, verbose=True, seed=99):

    t0 = time.monotonic()

    stall = 0
    max_stall = 800

        b = fast_autocorr(a)
        top_lags = np.argsort(b)[::-1][:8]
        k_peak = int(top_lags[0])

        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        i_vals = np.arange(i_start, i_end + 1)
        contrib = a[i_vals] * a[k_peak - i_vals]

        if contrib.max() < 1e-20:
            break

        in_peak = np.zeros(n, dtype=bool)

            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            in_peak[s:e + 1] = True

def cyclic_shift_best(a0):
    n = len(a0)
    a = norm_sum1(a0)
    best_m = fast_autocorr(a).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A_base = np.fft.rfft(a, fft_size)
    for s in range(1, n):
        phase = np.exp(-2j * np.pi * s * np.arange(len(A_base)) / fft_size)
        A_shift = A_base * phase
        b = np.fft.irfft(A_shift * A_shift, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m - 1e-15:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a, -best_shift), best_m
    return a.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
