def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    sz = 2 * n - 1
    F = 1 << (sz - 1).bit_length()
    A = np.fft.rfft(a, F)
    return np.fft.irfft(A * A, F)[:sz]

def norm1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Use the sorting algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def build_coeff(y, peaks, n):
    """Build constraint matrix rows for given peaks."""
    K = len(peaks)
    C = np.zeros((K, n))
    for jj in range(K):
        k = int(peaks[jj])
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        if i_lo < i_hi:
            idx = np.arange(i_lo, i_hi)
            C[jj, idx] = y[k - idx]
    return C


def lp_fw_opt(a0, budget, solver, K_init=300, K_max=2500, verbose=False):
    n = len(a0)
    y = norm1(a0)
    best = y.copy()
    ac = fft_conv(y)
    best_M = ac.max()
    t0 = time.monotonic()
    active = sorted(np.argsort(ac)[::-1][:K_init].tolist())
    no_imp = 0
    stag = 0
    it = 0

    while time.monotonic() - t0 < budget:
        no_imp_limit = 600 + 200 * stag
        if no_imp > no_imp_limit:
            if stag > 5:
                break
            # Perturb active set
            ac_cur = fft_conv(y)
            n_pert = max(30, len(active) // 4)
            rng = np.random.default_rng(stag * 137)
            keep = list(rng.choice(len(active), size=len(active) - n_pert, replace=False))
            new_adds = np.argsort(ac_cur)[::-1][:n_pert * 2]
            s = set(active[i] for i in keep)
            for k in new_adds:
                s.add(int(k))
            active = sorted(s)[:K_max]
            no_imp = 0
            stag += 1
            continue

        it += 1
        ac = fft_conv(y)
        M_cur = ac.max()

        if M_cur < best_M - 1e-14:
            best_M = M_cur
            best = y.copy()
            no_imp = 0
            stag = 0
        else:
            no_imp += 1

        # Trim active set
        if len(active) > K_max:
            margins = np.array([M_cur - ac[k] for k in active])
            keep_idx = np.argsort(margins)[:int(K_max * 0.7)]
            base = [active[i] for i in keep_idx]
            viol = np.where(ac > M_cur - 1e-8)[0]
            for v in viol[:80]:
                if int(v) not in base:
                    base.append(int(v))
            active = sorted(set(base))[:K_max]

        K = len(active)
        if K < 5:
            active = sorted(np.argsort(fft_conv(y))[::-1][:K_init].tolist())
            continue

        C = build_coeff(y, active, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        cons = [cp.sum(g_var) == 1, C @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), cons)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active = active[:max(50, K // 4)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        gv = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = gv.sum()
        if gs < 1e-14:
            continue
        gd = gv / gs

        # Add violations to active set
        F = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, F)
        Gf = np.fft.rfft(gd, F)
        cy = np.fft.irfft(Yf * Gf, F)[:2 * n - 1]
        viol = np.where(cy > M_cur + 1e-10)[0]
        if len(viol) > 0:
            va = cy[viol] - M_cur
            top = viol[np.argsort(va)[::-1][:50]]
