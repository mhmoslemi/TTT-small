def topk_gradient(c, a, K, temp=0.01):
    """Compute gradient of soft-top-K of c w.r.t. a"""
    n = len(a)
    # Soft assignment: w_k = sigmoid((c_k - threshold)/temp)
    # threshold chosen so that sum(w) ≈ K
    # Simple approximation: use the K-th largest value as threshold
    sorted_c = np.sort(c)[::-1]
    threshold = sorted_c[min(K, len(c)-1)]
    w = 1.0 / (1.0 + np.exp(-(c - threshold) / temp))
    w = w / w.sum() * K  # normalize to sum to K
    
    # grad_j = (1/K) * sum_k w_k * 2*a[k-j]
    # This is (2/K) * cross-correlation of w and a
    fft_size = 1 << (2*n - 1).bit_length()

    A_rev = np.fft.rfft(a[::-1], fft_size)
    Conv = np.fft.irfft(W * A_rev, fft_size)
    grad = (2.0 / K) * Conv[n-1:2*n-1]
    return grad

        if a.sum() < 0.01:
            return float('inf')
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def logistic(x):
    """Numerically stable sigmoid."""
    x = np.clip(x, -500, 500)
    out = np.zeros_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    exp_neg = np.exp(x[~pos])
    out[~pos] = exp_neg / (1.0 + exp_neg)
    return out


def topk_gradient_fft(c, a, K, temp):
    """
    Compute gradient of soft-top-K of c w.r.t. a.
    c = full autocorrelation of a (length 2n-1), a is length n.
    grad_j = (1/K) * sum_k w_k * 2 * a[k-j]
    """
    n = len(a)
    if K >= len(c):
        K = len(c) - 1
    K = max(1, int(K))
    
    sorted_c = np.sort(c)[::-1]
    threshold = sorted_c[K - 1]
    
    z = (c - threshold) / temp
    w = logistic(z)
    w_sum = w.sum()
    if w_sum < 1e-15:
        # Fallback: hard top-K
        top_idx = np.argsort(c)[::-1][:K]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / K
    else:
        w = w / w_sum * K
    
    # grad_j = (2/K) * sum_{m=0}^{n-1} a[m] * w[m+j]
    # = (2/K) * conv(a, w_rev)[n-1+j]
    fft_size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    W_rev = np.fft.rfft(w[::-1], fft_size)
    Conv = np.fft.irfft(A * W_rev, fft_size)
    grad = (2.0 / K) * Conv[(n - 1):(2 * n - 1)]
    return grad


def softmax_assign(b):
    """Convert log-parameters to simplex vector via softmax."""
    b = b - b.max()
    e = np.exp(np.clip(b, -500, 500))
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(b)) / len(b)
    return e / s


def logspace_nesterov(a0, time_limit, K_start=5, K_end=50,
                      temp_start=0.001, temp_end=0.0001,
                      lr_base=0.4, restart_every=1500,
                      K_growth=1.002, verbose=True, seed=42):
    """
    Nesterov accelerated gradient in log-space using softmax parameterization.
    a = softmax(b), optimize b to minimize 2n * max(conv(a,a)).
    """
    n = len(a0)
    a0 = norm_sum1(a0)
    # Initialize log-space: b = log(a + eps)
    eps = 1e-30
    b = np.log(a0 + eps)
    b = b - b.mean()  # center for numerical stability
    
    best_a = a0.copy()
    best_max_b = fast_autocorr(a0).max()
    t0 = time.monotonic()
    
    b_prev = b.copy()
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    K = float(K_start)
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        b_y = b + theta * (b - b_prev)
        
        # Convert to simplex
        a_y = softmax_assign(b_y)
        a_y = np.maximum(a_y, 1e-15)
        a_y = a_y / a_y.sum()
        
        # Compute gradient in a-space
        c = fast_autocorr(a_y)
        c_max = c.max()
        # Scale temp relative to c_max
        temp = max(temp_end, temp_start * (c_max / best_max_b) * (best_max_b / max(c_max, 1e-10)))
        temp = max(temp, 1e-5)
        
        grad_a = topk_gradient_fft(c, a_y, int(K), temp)
        
        # Convert gradient to log-space (b-space)
        # db_j = a_j * (grad_a_j - sum_i grad_a_i * a_i)
        dot = np.dot(a_y, grad_a)
        grad_b = a_y * (grad_a - dot)
        
        # Step size (adaptive)
        lr = lr_base / max(1.0, np.abs(grad_a).max() + 1e-10)
        lr = min(lr, 0.05)
        
        b_new = b_y - lr * grad_b
        # Keep b centered
        b_new = b_new - np.mean(b_new)
        
        a_new = softmax_assign(b_new)
        
        # Evaluate
        c_new = fast_autocorr(a_new)
        max_new = c_new.max()
        if max_new < best_max_b - 1e-14:
            best_max_b = max_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        b_prev = b
        b = b_new
        t_nest = t_new
        
        # Gradually increase K
        if t % 50 == 0:
            K = min(K * K_growth, K_end)
        
        # Restart if stuck
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Restart with perturbation in log-space
            noise_scale = 0.05 / (1.0 + 0.1 * (t // restart_every))
            noise = rng.normal(0, 1, n) * noise_scale
            b_base = np.log(best_a + eps)
            b_base = b_base - b_base.mean()
            b = b_base + noise
            b = b - b.mean()
            b_prev = b.copy()
            K = max(K * 0.5, K_start)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    LogNest restart t=%d (%.1fs): J=%.6f K=%.0f" % (t, elapsed, J, K))
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    LogNest t=%d (%.1fs): J=%.6f K=%.0f" % (t, elapsed, J, K))
    
    return best_a, best_max_b


def peak_shaving(a0, time_limit, verbose=True, seed=77):
    """
    Pressure-based peak shaving: identify top peaks, compute pressure map,
    redistribute mass from high-pressure to low-pressure positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 5000
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top peaks
        n_top = min(50, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:n_top]
        
        # Compute pressure map from top peaks
        n_pressure = min(20, n_top)
        pressure = np.zeros(n)
        for k in top_idx[:n_pressure]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            # Contribution to pressure from this peak
            partner = k_int - idx
            pressure[idx] += 2.0 * a[partner] * (c[k_int] / c_max)
        
        # Find positions to reduce (high pressure, high mass) and add (low pressure)
        if a.max() < 1e-15:
            break
        
        mass_score = pressure * a  # positions with both high pressure and mass
        high_mask = mass_score > np.percentile(mass_score, 90)
        low_mask = pressure < np.percentile(pressure, 10)
        
        if not high_mask.any() or not high_mask.any():
            high_mask = mass_score > np.percentile(mass_score, 75)
            low_mask = pressure < np.percentile(pressure, 25)
        
        n_moves = min(20, int(high_mask.sum() // 2))
        if n_moves < 1:
            n_moves = max(1, int(n * 0.005))
        
        high_positions = np.where(high_mask)[0]
        low_positions = np.where(low_mask)[0]
        
        if len(high_positions) == 0 or len(low_positions) == 0:
            no_improve += 10
            continue
        
        # For each high position, try moving mass to a low position
        improved = False
        rng.shuffle(high_positions)
        rng.shuffle(low_positions)
        
        for src in high_positions[:n_moves]:
            if improved or time.monotonic() - t0 > time_limit:
                break
            for dst in low_positions[:n_moves]:
                if dst == src:
                    continue
                # Try different fractions
                for frac in [0.002, 0.005, 0.01, 0.02, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            # Random perturbation when stuck
            if no_improve > 500:
                n_rand = max(5, n // 100)
                idx = rng.integers(0, n, 2 * n_rand)
                for _ in range(n_rand):
                    i, j = idx[2*_], idx[2*_+1]
                    if i == j or a[i] < 1e-12:
                        continue
                    amt = a[i] * rng.uniform(0.001, 0.02)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
        
        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave (%.1fs): J=%.6f stall=%d" % (elapsed, J, no_improve))
    
    return best_a, best_m


def local_polish(a0, time_limit, verbose=True, seed=123):
    """
    Fine-grained coordinate descent: for each peak, try moving mass from
    the top contributing pair to random safe positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_autocorr(a)
        # Focus on top peaks
        n_peaks = min(15, 2 * n - 1)
        peak_pos = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        for k in peak_pos:
            if improved or time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            
            # Top contributing pair
            top_pi = np.argmax(products)
            src_i = int(idx[top_pi])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue
            if src_i == partner:
                # Diagonal: try splitting
                for frac in [0.001, 0.003, 0.01]:
                    amt = a[src_i] * frac / 2
                    if amt < 1e-15:
                        continue
                    for dst in rng.integers(0, n, 3):
                        if dst == src_i:
                            continue
                        a_trial = a.copy()
                        a_trial[src_i] -= 2 * amt
                        a_trial[dst] += amt
                        # Divide by something... actually just move
                        a_trial[src_i] += 2 * amt  # revert
                        a_trial[src_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                continue
            
            # Off-diagonal: move from larger to a safe position
            larger_i = src_i if a[src_i] >= a[partner] else partner
            if a[larger_i] < 1e-14:
                continue
            
            for frac in [0.001, 0.002, 0.005, 0.01]:
                if improved:
                    break
                amt = a[larger_i] * frac
                if amt < 1e-16:
                    continue
                for _ in range(5):
                    dst = int(rng.integers(n))
                    if dst == larger_i:
                        continue
                    a_trial = a.copy()
                    a_trial[larger_i] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
        
        if not improved:
            stall += 1
            # Structured nudge
            if stall % 50 == 0:
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.005 / (1.0 + stall / 50.0)
                a_try = a * (1.0 + eps * anti)
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Polish (%.1fs): J=%.6f stall=%d" % (elapsed, J, stall))
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initialization(n, shape='power', gamma=1.5):
    """Create an initial sequence with a given shape."""
    x = np.linspace(0.0, 1.0, n)
    if shape == 'power':
        a = (1.0 - x) ** gamma + 0.1
    elif shape == 'cosine':
        a = 0.5 * (1.0 + np.cos(np.pi * x)) ** 2 + 0.05
    elif shape == 'exponential':
        a = np.exp(-gamma * x) + 0.05
    elif shape == 'bimodal':
        a = np.exp(-20.0 * x ** 2) + np.exp(-20.0 * (x - 1.0) ** 2) + 0.02
    elif shape == 'flat_u':
        a = 1.0 / (0.1 + x) + 1.0 / (0.1 + (1.0 - x)) + 0.1
    else:
        a = np.ones(n)
    return norm_sum1(np.maximum(a, 0.0))


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best if candidate is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.8f (n=%d, t=%.1fs)" % (label, sc, len(a_norm), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search function. Uses log-space Nesterov + peak shaving + local polish
    with multiple initializations and lengths.
    """
    t0 = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent loaded: n=%d, score=%.8f" % (len(parent), p_score))
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 6000
        best_seq = make_initialization(n_base, 'power', 1.5)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=%d, score=%.8f" % (len(best_seq), best_score))
    
    n0 = len(best_seq)
    print("Starting: n=%d, score=%.8f" % (n0, best_score))
    
    # === Phase 1: Log-space Nesterov on parent ===
    remaining = time_budget - (time.monotonic() - t0)
    print("\n=== Phase 1: Log-space Nesterov (n=%d) ===" % n0)
    t1 = min(remaining * 0.22, 380.0)
    if t1 > 60:
        a_out, _ = logspace_nesterov(best_seq, t1,
                                     K_start=3, K_end=80,
                                     temp_start=0.002, temp_end=0.00005,
                                     lr_base=0.35, restart_every=2000,
                                     K_growth=1.0015, verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lognest1", t0)
    
    # === Phase 2: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(best_seq)
    print("\n=== Phase 2: Peak Shaving (n=%d) ===" % n_best)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40:
        a_out, _ = peak_shaving(best_seq, t2, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "shave1", t0)
    
    # === Phase 3: Local polish ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(best_seq)
    print("\n=== Phase 3: Local Polish (n=%d) ===" % n_best)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_out, _ = local_polish(best_seq, t3, verbose=True, seed=22)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "polish1", t0)
    
    # === Phase 4: Log-space Nesterov with higher K (refine all peaks) ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(best_seq)
    print("\n=== Phase 4: Log-space Nesterov refine (n=%d) ===" % n_best)
    t4 = min(remaining * 0.18, 320.0)
    if t4 > 60:
        a_out, _ = logspace_nesterov(best_seq, t4,
                                     K_start=10, K_end=200,
                                     temp_start=0.001, temp_end=0.00002,
                                     lr_base=0.3, restart_every=1500,
                                     K_growth=1.001, verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lognest2", t0)
    
    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length exploration ===")
    
    lengths_to_try = [n_best + 800, n_best + 2000, max(2000, n_best - 1200)]
    lengths_to_try = [l for l in lengths_to_try if 2000 <= l <= 12000]
    # Avoid duplicates
    lengths_to_try = list(dict.fromkeys(lengths_to_try))
    
    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 150:
            break
        print("  Trying n=%d (best so far: %.8f)" % (new_n, best_score))
        a_long = interpolate_sequence(best_seq, new_n)
        
        t5a = min(remaining * 0.30, 180.0)
        if t5a > 40:
            a_out, _ = logspace_nesterov(a_long, t5a,
                                         K_start=3, K_end=60,
                                         temp_start=0.002, temp_end=0.0001,
                                         lr_base=0.4, restart_every=1500,
                                         K_growth=1.002, verbose=False, seed=3)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_n%d" % new_n, t0)
        
        remaining = time_budget - (time.monotonic() - t0)
        if remaining > 80:
            t5b = min(remaining * 0.25, 100.0)
            if t5b > 30:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = peak_shaving(a_try, t5b, verbose=False, seed=33)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "mlps_n%d" % new_n, t0)
    
    # === Phase 6: Alternative initialization and optimization ===
    remaining = time_budget - (time.monotonic() - t0)
    print("\n=== Phase 6: Alternative shapes ===")
    
    if remaining > 200:
        n_alt = len(best_seq)
        shapes = [('cosine', 2.0), ('bimodal', 1.0), ('flat_u', 1.0)]
        t_per = min(remaining / (len(shapes) + 1), 120.0)
        
        for shape_name, gamma in shapes:
            remaining = time_budget - (time.monotonic() - t0)
            if remaining < 80:
                break
            a_init = make_initialization(n_alt, shape_name, gamma)
            # Mix with current best
            mix = 0.5 * best_seq + 0.5 * a_init
            mix = norm_sum1(mix)
            
            a_out, _ = logspace_nesterov(mix, t_per,
                                         K_start=5, K_end=100,
                                         temp_start=0.002, temp_end=0.0001,
                                         lr_base=0.5, restart_every=1000,
                                         K_growth=1.003, verbose=False, seed=42)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "alt_%s" % shape_name, t0)
    
    # === Phase 7: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Peak Shaving (n=%d) ===" % n_best)
    t7 = min(remaining * 0.25, 250.0)
    if t7 > 30:
        a_out, _ = peak_shaving(best_seq, t7, verbose=True, seed=44)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "shave_final", t0)
    
    # === Phase 8: Final log-space Nesterov (very high K, low temp) ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Log-space Nesterov (n=%d) ===" % n_best)
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 30:
        a_out, _ = logspace_nesterov(best_seq, t8,
                                     K_start=5, K_end=50,
                                     temp_start=0.0005, temp_end=0.00001,
                                     lr_base=0.25, restart_every=1000,
                                     K_growth=1.001, verbose=True, seed=5)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lognest_final", t0)
    
    # === Phase 9: Final local polish ===
    remaining = time_budget - (time.monotonic() - t0)
    n_best = len(best_seq)
    print("\n=== Phase 9: Final Polish (n=%d) ===" % n_best)
    t9 = min(remaining * 0.40, 150.0)
    if t9 > 20:
        a_out, _ = local_polish(best_seq, t9, verbose=True, seed=66)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "polish_final", t0)
    
    # === Phase 10: Another round if time remains ===
    remaining = time_budget - (time.monotonic() - t0)
    if remaining > 100:
        n_best = len(best_seq)
        print("\n=== Phase 10: Extra round (n=%d) ===" % n_best)
        t10a = min(remaining * 0.4, 80.0)
        if t10a > 30:
            a_out, _ = logspace_nesterov(best_seq, t10a,
                                         K_start=3, K_end=30,
                                         temp_start=0.001, temp_end=0.00002,
                                         lr_base=0.2, restart_every=800,
                                         K_growth=1.001, verbose=False, seed=7)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "extra1", t0)
        
        remaining = time_budget - (time.monotonic() - t0)
        if remaining > 40:
            t10b = min(remaining * 0.6, 60.0)
            if t10b > 20:
                a_out, _ = local_polish(best_seq, t10b, verbose=False, seed=88)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "extra2", t0)
    
    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t0
        print("\nFINAL: score=%.8f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence! Using fallback.")
        n_fb = 5000
        best_seq = make_initialization(n_fb, 'power', 1.5)
        best_score = evaluate_safe(best_seq)
    
    # Ensure valid output
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)
    
    return [float(x) for x in best_seq]
