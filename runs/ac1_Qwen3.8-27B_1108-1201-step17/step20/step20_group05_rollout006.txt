def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def fft_conv(a):
    """FFT-based self-convolution."""
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.where(np.isfinite(c), c, 0.0)

def fft_conv2(a, b):
    """FFT-based cross-convolution."""
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return np.where(np.isfinite(c), c, 0.0)

def normalize(a):

def interpolate(a, new_n):

    if a.size == new_n:
        return normalize(a)
    old_x = np.linspace(0, 1, a.size)
    new_x = np.linspace(0, 1, new_n)
    y = np.interp(new_x, old_x, a)
    return normalize(y)

def lse_objective(a, beta, top_lags=None):
    """Log-sum-exp smoothed objective and its gradient."""
    n = a.size
    c = fft_conv(a)
    if top_lags is not None:
        cv = c[top_lags]
    else:
        cv = c
    m = float(np.max(cv))
    beta_cv = beta * (cv - m)
    beta_cv = np.clip(beta_cv, -500, 500)
    e = np.exp(beta_cv)
    se = float(e.sum())
    if se < 1e-300:
        return m, np.zeros(n)
    p = e / se
    # Back to full vector
    pf = np.zeros_like(c)
    if top_lags is not None:
        pf[top_lags] = p
    else:
        pf = p
    # Gradient: sum_k pf_k * 2*a_{k-j} for each j
    # This is 2 * cross_convolution of pf (reversed) with a
    N = next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:3*n-2]
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    obj = m + math.log(se) / beta
    return obj, grad

def softmax_params(a, ref_idx):
    """Convert probability vector to softmax parameters (log-ratios to ref)."""
    n = a.size
    ref_idx = int(ref_idx)
    a_ref = max(float(a[ref_idx]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref_idx == 0:
        v = loga[1:]
    elif ref_idx == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref_idx], loga[ref_idx+1:]))
    return v, ref_idx

def softmax_from_params(v, ref_idx):
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    if ref_idx > 0:
        u[:ref_idx] = v[:ref_idx]
    if ref_idx < n - 1:
        u[ref_idx+1:] = v[ref_idx:]
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones(n) / n
    return e / s

def lbfegs_warm(a0, beta, top_lags, maxfun, ref_idx=None):

    n = a0.size

        obj, ga = lse_objective(a, beta, top_lags)
        mean = float(np.dot(a, ga))
        gu = a * (ga - mean)
        gv = np.empty(v.size)

            gv[:ref_idx] = gu[:ref_idx]

            gv[ref_idx:] = gu[ref_idx+1:]

        if res.x is not None and res.x.size == v0.size:
            a_out = softmax_from_params(res.x, ref_idx)
            return normalize(a_out)
    except:
        pass
    return a0

def pressure_vector(a, top_lags, c=None):
    """Compute the pressure on each coordinate from top lags."""
    n = a.size
    if c is None:
        c = fft_conv(a)
    # For each top lag k, the "pressure" on coordinate j is a_{k-j} (for valid k-j)
    # Total pressure = sum over top lags of a_{k-j}
    # This equals the cross-convolution of the indicator of top lags with a
    pf = np.zeros(2*n-1)
    pf[top_lags] = 1.0
    N = next_pow2(3*n-2)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:3*n-2]
    pressure = conv[n-1:2*n-1][::-1]
    return np.where(np.isfinite(pressure), pressure, 0.0)

def rank1_move(c, a, s, d, delta):
    """Apply a rank-1 move: a[s]-=delta, a[d]+=delta. Returns new convolution."""
    n = a.size
    c_new = c.copy()
    # conv_k changes by: -delta*a_{k-s} + delta*a_{k-d} for the linear part
    # In the convolution array, this means:
    # Adding -delta*a to positions s..s+n-1
    # Adding +delta*a to positions d..d+n-1
    c_new[s:s+n] -= delta * a
    c_new[d:d+n] += delta * a
    # Second-order terms: delta^2 at positions 2s, 2d, and -2delta^2 at s+d
    if 2*s < c_new.size:
        c_new[2*s] += delta*delta
    if 2*d < c_new.size:
        c_new[2*d] += delta*delta
    if s+d < c_new.size:
        c_new[s+d] -= 2.0*delta*delta
    return np.where(np.isfinite(c_new), c_new, 0.0)
