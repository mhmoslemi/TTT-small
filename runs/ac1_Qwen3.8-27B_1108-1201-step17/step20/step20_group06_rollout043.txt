import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad_and_probs(a, beta, top_k=None):
    n = len(a)
    b = fast_autocorr(a)
    size = 2 * n - 1

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()

    if b_max <= 0:
        return np.zeros(n), b, np.ones(size) / size

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]

    return grad, b, p


def subgradient_method(a0, time_limit, beta=50.0, verbose=True, seed=42):
    """Hybrid: smooth gradient when many near-peaks, exact subgradient when few."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    eta = 1.0 / (2.0 * np.sqrt(n))
    top_k = max(5, int(n * 0.02))
    stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(x)
        b_max = b.max()
        size = 2 * n - 1

        # Determine active set: peaks within tolerance of max
        tol = 1e-4 * b_max
        active = np.where(b >= b_max - tol)[0]
        n_active = len(active)

        if n_active <= max(3, top_k // 5):
            # Use exact subgradient: average gradient of active peaks
            grad = np.zeros(n)
            for k in active:
                k_int = int(k)
                # Gradient of (a*a)[k] w.r.t. a
                # d/da[i] (a*a)[k] = a[k-i] + a[i] (with bounds, handling k=2i case)
                js = max(0, k_int - n + 1)
                je = min(n - 1, k_int)
                idx = np.arange(js, je + 1)
                # a[k - j] for j in idx
                a_kj = a = x[k_int - idx]
                # a[j] for j in idx
                a_j = x[idx]
                # For the case k - j == j (i.e., j = k/2), the derivative is 2*a[j]
                # Otherwise it's a[k-j] + a[j]
                # Actually: d/da[j](a*a)[k] = a[k-j] + (j != k-j ? a[j] : 0)
                # Wait: (a*a)[k] = sum_j a[j]*a[k-j]
                # d/da[i] = a[k-i] (from j=i) + a[i] (from k-j=i, i.e., j=k-i, if k-i != i)
                # So d/da[i](a*a)[k] = a[k-i] + (k-i != i ? a[i] : 0)
                # For i in [js, je]: k-i is in [0, n-1]
                # The term a[k-i] is always present
                # The term a[i] is present only if k-i != i, i.e., i != k/2
                mid = k_int / 2.0
                for j in range(js, je + 1):
                    if j == mid:
                        grad[j] += x[j]  # 2*a[j] * a[j]... wait
                    else:
                        grad[j] += x[k_int - j] + x[j]
            # Normalize by number of active peaks
            grad /= n_active
        else:
            # Use smoothed gradient
            grad, _, _ = compute_smoothed_grad_and_probs(x, beta, top_k)

        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-30:
            break

        # Line search
        best_alpha = 0.1
        best_val = b_max
        base_b_max = fast_autocorr(x).max()
        for alpha in [eta, eta * 2.0, eta * 0.5, 0.01, 0.005, 0.1]:
            z = x - alpha * grad
            x_trial = simplex_project(z)
            m_trial = fast_autocorr(x_trial).max()
            if m_trial < best_val:
                best_val = m_trial
                best_alpha = alpha

        x_new = simplex_project(x - best_alpha * grad)
        m_new = fast_autocorr(x_new).max()

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.02, 1.0)
        else:
            stall += 1
            if stall > 500:
                eta *= 0.8
                if eta < 1e-5:
                    # Restart
                    x_new = norm_sum1(best_a + 0.02 * rng.standard_normal(n) * best_a)
                    eta = 1.0 / (2.0 * np.sqrt(n))
                    stall = 0

        x = x_new

        if t % 2000 == 0:
            beta = min(beta * 1.01, 500.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SubGrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " eta=" + str(round(eta, 6)))

def nesterov_improved(a0, time_limit, beta_start=15.0, beta_end=800.0,
                      beta_growth=1.004, eta_init=0.3, top_k_ratio=0.04,
                      restart_every=1200, verbose=True, seed=42):
    """Nesterov with improved line search and adaptive top-k."""
    n = len(a0)
    x = norm_sum1(a0)

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b, _ = compute_smoothed_grad_and_probs(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x
            max_new = best_max_b
        else:
            # Backtracking with Armijo condition
            eta_trial = eta
            y_max = fast_autocorr(y).max()
            found = False
            for _ in range(8):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                max_new = fast_autocorr(x_new).max()
                if max_new <= y_max - 1e-3 * eta_trial * gnorm * gnorm or max_new <= y_max + 1e-14:
                    found = True
                    break
                eta_trial *= 0.5

                x_new = x
                max_new = y_max
                eta = max(eta_trial, 1e-6)

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 1.5)

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 40 == 0:

            # Adaptive top-k: increase when stalled
            if stall > 200:

            elif stall < 50:
                top_k = min(size if (size := 2 * n - 1) else n, int(n * top_k_ratio * 2))

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                pert = rng.normal(0, 0.015, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.7 * best_a + 0.3 * rng.dirichlet(np.ones(n) * 0.5)
            elif strategy == 2:
                x_r = best_a * np.exp(0.04 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5

                x_r[mask] *= 0.6
                x_r[~mask] *= 1.4

def peak_surgery(a0, time_limit, verbose=True, seed=111):
    """Directly reduce the largest autocorrelation peak via coordinate moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        # Focus on top peaks
        top_count = min(5, size)
        top_idx = np.argsort(ac)[::-1][:top_count]
        improved = False

        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            # Find the pair contributing most to this peak
            contrib = a[idx] * a[k_int - idx]
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            if not (0 <= partner < n):
                continue

            # Try moving mass from src to reduce this peak
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                # Try several destinations
                for _ in range(8):
                    dst = int(rng.integers(n))
                    if dst == src or dst == partner:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 100:
                # Broader random exploration
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.04)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 1000 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurg (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def lbfgs_softmax(a0, time_limit, beta=50.0, top_k=None, verbose=True, maxiter=2000):
    """L-BFGS-B with softmax parameterization."""
    n = len(a0)
    a_start = norm_sum1(a0)
    # Softmax parameterization: a = softmax(x)
    x0 = np.log(a_start + 1e-30)
    x0 -= x0.max()  # numerical stability

    if top_k is None:
        top_k = max(10, int(n * 0.03))

    def objective(xv):
        ex = np.exp(xv - xv.max())
        a = ex / ex.sum()
        b = fast_autocorr(a)
        if top_k < len(b):
            idx = np.argsort(b)[::-1][:top_k]
            b_sel = b[idx]
        else:
            b_sel = b
        m = b_sel.max()
        shifted = beta * (b_sel - m)
        shifted = np.clip(shifted, -500, 0)
        log_sum = np.log(np.exp(shifted).sum() / len(b_sel))
        return m + log_sum / beta

    def gradient(xv):
        ex = np.exp(xv - xv.max())
        a = ex / ex.sum()
        grad_a, _, _ = compute_smoothed_grad_and_probs(a, beta, top_k)
        # Chain rule: d/dx_i = sum_j da_j/dx_i * g_j = a_i * (g_i - sum_j a_j g_j)
        ag = a @ grad_a
        return a * (grad_a - ag)

    result = scipy_minimize(objective, x0, method='L-BFGS-B',
                            jac=gradient, options={'maxiter': maxiter,
                                                   'maxfun': maxiter * 2,
                                                   'ftol': 1e-12, 'gtol': 1e-8})
    x_opt = result.x
    ex = np.exp(x_opt - x_opt.max())
    a_opt = ex / ex.sum()
    return a_opt


def spectral_refine(a0, time_limit, verbose=True, seed=777):

            target = power[1:].mean()
        else:
            target = power[0]

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.5:
                red = 1.0 - alpha * min(2.0, (ratio - 1.5) * 0.5)
                A_new[k] *= max(0.3, np.sqrt(max(0.1, red)))
            elif ratio < 0.3:
                inc = 1.0 + alpha * 0.3 * (0.3 - ratio)
                A_new[k] *= min(2.0, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new

        if it % 1000 == 0:
            alpha = max(0.005, alpha * 0.92)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        sc = evaluate_sequence([float(x) for x in a_norm])
    except Exception:
        sc = float('inf')
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

                best_seq = norm_sum1(parent)

    except Exception:
        parent = None

    # Phase 3: L-BFGS-B with softmax parameterization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: L-BFGS-B (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.15, 250.0)

        t_start = time.monotonic()

        best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs1", t_global)

    # Phase 4: Subgradient method
    remaining = time_budget - (time.monotonic() - t_global)
