def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

def norm_sum1(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad_fft(a, beta, top_k):
    """Compute smoothed gradient using FFT. Returns (grad, peaks, autocorr)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]

    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()

    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total

    # Gradient: (2/beta) * sum_k p[k] * a[k-j]
    P = np.fft.rfft(p, fft_size)
    a_rev = a[::-1]
    Ar = np.fft.rfft(a_rev, fft_size)
    Conv = np.fft.irfft(P * Ar, fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c

    momentum = 0.7

    reeval_interval = 15

        valid = (kmj >= 0) & (kmj < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * a[kmj[valid]]

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad

        # Projected step
        z = a - eta * v
        a_new = simplex_project(z)

        # Quick check: did the targeted peak decrease? (O(n))
        # c_new[k_star] = sum_i a_new[i] * a_new[k_star - i]
        i_valid = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        c_target = np.sum(a_new[i_valid] * a_new[k_star - i_valid])

        # Evaluate every reeval_interval steps
        if step % reeval_interval == 0:
            c_full = fast_autocorr(a_new)
            c_max_full = c_full.max()
            if c_max_full < best_m - 1e-14:
                best_m = c_max_full
                best_a = a_new.copy()
                stall = 0
                k_star = int(np.argmax(c_full))
                # Adaptive: if we're making progress, keep going
                # If stuck, try restart
            else:
                stall += 1

            # If stuck, try a restart
            if stall > 300:
                # Try different peaks
                c_full = fast_autocorr(a)
                top_peaks = np.argsort(c_full)[::-1][:5]
                k_star = int(top_peaks[rng.integers(len(top_peaks))])
                v = np.zeros(n)
                eta = 0.8 / n
                stall = 0
                # Perturb
                pert = rng.normal(0, 0.005, n) * a
                a = a + pert
                a = np.maximum(a, 1e-10)
                a = a / a.sum()

        else:
            # Just check if targeted peak decreased
            c_current = np.sum(a[i_valid] * a[k_star - i_valid])
            if c_target < c_current * 0.999:
                a = a_new
                if step % 50 == 0:
                    c_full = fast_autocorr(a)
                    c_max_full = c_full.max()
                    if c_max_full < best_m - 1e-14:
                        best_m = c_max_full
                        best_a = a.copy()
                        k_star = int(np.argmax(c_full))
                        stall = 0
                    else:
                        stall += 1
            else:
                # Step too large, reduce
                eta *= 0.5
                z = a - eta * v
                a_new = simplex_project(z)
                a = a_new

        # Adaptive step size
        if step % 100 == 0:
            eta = min(eta * 1.02, 2.0 / n)
        eta = max(eta, 1e-10 / n)

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakChase step=%d (%.1fs): J=%.6f, eta=%.2e, stall=%d" % (step, elapsed, J, eta, stall))

    return best_a, best_m


def local_peak_search(a0, time_limit, verbose=True, seed=99):
    """
    Systematic local search: identify top peaks and reduce them via
    pressure-based mass redistribution.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-14:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top peaks
        n_top = min(20, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:n_top]

        # Compute pressure at each position
        pressure = np.zeros(n)
        for k2 in top_idx[:10]:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            if s2 <= e2:
                idx2 = np.arange(s2, e2 + 1)
                pressure[idx2] += a[k2i - idx2]
                pressure[k2i - idx2] += a[idx2]

        improved = False

        for k in top_idx[:8]:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            # Get top contributing pairs
            top_pairs = np.argsort(products)[::-1][:5]

            for pi in top_pairs:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue

                # Determine which position to reduce
                if a[src_i] >= a[partner]:
                    reduce_i = src_i
                else:
                    reduce_i = partner

                if a[reduce_i] < 1e-12:
                    continue

                # Find low-pressure destinations
                candidates = np.where(pressure < np.median(pressure))[0]
                candidates = candidates[candidates != reduce_i]
                if len(candidates) == 0:
                    candidates = np.arange(n)
                    candidates = candidates[candidates != reduce_i]

                # Sort by pressure (lowest first)
                cand_pressure = pressure[candidates]
                n_try = min(8, len(candidates))
                try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]

                for dst in try_dsts:
                    dst_int = int(dst)
                    if improved:
                        break
                    for frac in [0.003, 0.005, 0.01, 0.02, 0.03]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break

        if not improved:
            # Random exploration when stuck
            if no_improve > 200:
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break
                if not improved and no_improve % 500 == 0:
                    # Bigger perturbation
                    pert = rng.normal(0, 0.01, n) * a
                    a_try = a + pert
                    a_try = np.maximum(a_try, 1e-10)
                    a_try = a_try / a_try.sum()
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        no_improve = 0

        if verbose and no_improve % 1000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LocalPS (%.1fs): J=%.6f stall=%d" % (elapsed, J, no_improve))

    return best_a, best_m


def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    """Fine-grained pairwise coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 800:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 97)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:15]

        improved = False

        for k in peak_positions[:15]:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            top_pairs = np.argsort(products)[::-1][:3]
            for pi in top_pairs:
                if improved:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue

                larger_i = src_i if a[src_i] >= a[partner] else partner
                if a[larger_i] < 1e-14:
                    continue

                for frac in [0.001, 0.002, 0.003, 0.005]:
                    if improved:
                        break
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(5):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

        if not improved:
            stall += 1
            if stall % 200 == 0:
                # Structured anti-symmetric nudge
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.005 / (1.0 + stall / 200.0)
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 400 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (%.1fs): J=%.6f stall=%d" % (elapsed, J, stall))

    return best_a, best_m


def build_structured_init(n, seed=0):
    """Build a structured initialization with good autocorrelation properties."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)

    # Try several structured forms and pick the best
    candidates = []

    # Hann window
    a_hann = np.sin(np.pi * x) ** 2
    a_hann += 0.01
    candidates.append(norm_sum1(a_hann))

    # Cosine squared
    a_cos = np.cos(np.pi * (x - 0.5)) ** 2
    a_cos += 0.01
    candidates.append(norm_sum1(a_cos))

    # Raised cosine with parameter
    for p in [1.5, 2.0, 3.0]:
        a_raised = (0.5 * (1 + np.cos(2 * np.pi * x))) ** p
        a_raised += 0.01
        candidates.append(norm_sum1(a_raised))

    # Two-lobe: high in middle, low at edges
    a_two = np.exp(-8.0 * (x - 0.5) ** 2) + 0.1 * np.exp(-2.0 * (x - 0.25) ** 2) + 0.1 * np.exp(-2.0 * (x - 0.75) ** 2)
    candidates.append(norm_sum1(a_two))

    # Flat with noise
    a_flat = np.ones(n)
    a_flat += rng.normal(0, 0.1, n)
    a_flat = np.maximum(a_flat, 0)
    candidates.append(norm_sum1(a_flat))

    # Power law decay from center
    a_pl = 1.0 / (0.1 + np.abs(x - 0.5) * 5.0)
    candidates.append(norm_sum1(a_pl))

    # Evaluate all
    best_c = None
    best_score = float('inf')
    for c in candidates:
        sc = evaluate_safe(c)
        if sc < best_score:
            best_score = sc
            best_c = c

    return best_c, best_score


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update the best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a_norm), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent loaded: n=%d, score=%.6f" % (len(parent), p_score))
    except Exception:
        pass

    # If no parent, use structured initialization
    if best_seq is None:
        n_base = 6700
        best_seq, best_score = build_structured_init(n_base, seed=42)
        print("Structured init: n=%d, score=%.6f" % (len(best_seq), best_score))

    n0 = len(best_seq)
    print("Starting: n=%d, score=%.6f" % (n0, best_score))

    # Phase 1: Nesterov with moderate beta (coarse optimization)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Nesterov coarse (n=%d, t=%.0fs) ===" % (n0, remaining))
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 50:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t1,
                                beta_start=30.0, beta_end=2000.0,
                                beta_growth=1.006, eta_base=0.6,
                                top_k_ratio=0.03, restart_every=2000,
                                verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 2: Peak chasing (exact subgradient, O(n) per step)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Peak Chase (n=%d, t=%.0fs) ===" % (n_best, remaining))
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 50:
        a_out, _ = peak_chase(norm_sum1(best_seq), t2, verbose=True, seed=7)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)

    # Phase 3: Local peak search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Local Peak Search (n=%d, t=%.0fs) ===" % (n_best, remaining))
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, _ = local_peak_search(norm_sum1(best_seq), t3, verbose=True, seed=99)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lps1", t_global)

    # Phase 4: Nesterov with high beta (refinement)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Nesterov fine (n=%d, t=%.0fs) ===" % (n_best, remaining))
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 50:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t4,
                                beta_start=200.0, beta_end=8000.0,
                                beta_growth=1.012, eta_base=0.4,
                                top_k_ratio=0.01, restart_every=1500,
                                verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 5: Peak chasing again (with refined starting point)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Peak Chase 2 (n=%d, t=%.0fs) ===" % (n_best, remaining))
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_out, _ = peak_chase(norm_sum1(best_seq), t5, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak2", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Multi-length (n_best=%d, t=%.0fs) ===" % (n_best, remaining))

    # Try a few nearby lengths
    lengths_to_try = sorted(set([
        n_best + 200, n_best + 500, n_best + 1000,
        max(4000, n_best - 500), max(5000, n_best - 1000)
    ]))
    lengths_to_try = [l for l in lengths_to_try if 3000 <= l <= 12000]

    for new_n in lengths_to_try[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=%d" % new_n)
        a_long = interpolate_sequence(best_seq, new_n)

        t_ml = min(remaining * 0.30, 150.0)
        if t_ml > 30:
            a_out, _ = nesterov_opt(a_long, t_ml * 0.5,
                                    beta_start=100.0, beta_end=4000.0,
                                    beta_growth=1.01, eta_base=0.5,
                                    top_k_ratio=0.02, restart_every=1000,
                                    verbose=False, seed=3)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_n%d" % new_n, t_global)

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = peak_chase(a_try, min(remaining * 0.4, 80.0), verbose=False, seed=13)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "mlpc_n%d" % new_n, t_global)

    # Phase 7: Pairwise CD for final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Pairwise CD (n=%d, t=%.0fs) ===" % (n_best, remaining))
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 30:
        a_out, _ = pairwise_cd(norm_sum1(best_seq), t7, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd1", t_global)

    # Phase 8: Final peak chase + local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final refinement (n=%d, t=%.0fs) ===" % (n_best, remaining))

    if remaining > 60:
        t8a = min(remaining * 0.4, 150.0)
        if t8a > 20:
            a_out, _ = peak_chase(norm_sum1(best_seq), t8a, verbose=False, seed=21)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t8b = min(remaining * 0.4, 100.0)
            if t8b > 20:
                a_out, _ = local_peak_search(norm_sum1(best_seq), t8b, verbose=False, seed=33)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lps_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8c = min(remaining * 0.5, 80.0)
            if t8c > 10:
                a_out, _ = pairwise_cd(norm_sum1(best_seq), t8c, verbose=False, seed=66)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd_final", t_global)

    # Phase 9: One more Nesterov burst if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print("\n=== Phase 9: Final Nesterov (n=%d, t=%.0fs) ===" % (n_best, remaining))
        t9 = min(remaining * 0.7, 150.0)
        if t9 > 30:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t9,
                                    beta_start=500.0, beta_end=15000.0,
                                    beta_growth=1.015, eta_base=0.3,
                                    top_k_ratio=0.005, restart_every=800,
                                    verbose=True, seed=5)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_ult", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_out, _ = peak_chase(norm_sum1(best_seq), min(remaining * 0.6, 60.0), verbose=False, seed=25)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_ult", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2 + 0.01)
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
