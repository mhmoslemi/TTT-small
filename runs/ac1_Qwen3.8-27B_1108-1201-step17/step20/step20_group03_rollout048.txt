        # Compute b and gradient
        b = fast_autocorr(y)
        
        if top_k is not None and 0 < top_k < L:
            idx = np.argsort(b)[::-1][:top_k]
            b_use = np.zeros(L)
            b_use[idx] = b[idx]
            b_max = b[idx].max()
        else:
            b_use = b
            b_max = b.max()
        
        # Softmax weights
        shifted = beta * (b_use - b_max)
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()
        
        # Gradient
        grad = compute_grad_fft(y, p)
        
        # Optional variance penalty
        if use_variance:
            # Penalize deviation from uniform
            var_grad = var_weight * compute_grad_fft(y, (b - uniform_b) / b_max)
            grad = grad + var_grad
        
        # Gradient step
        z = y - step * grad
        # Project onto simplex {a >= 0, sum(a) = 1}
        x_new = simplex_project(z)
        
        # Update best
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        if max_new < best_m - 1e-15:
            best_m = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Adaptive beta
        if t % 50 == 0:
            beta = min(beta * 1.005, beta_end)
        
        # Restart
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, beta_start * 0.5)
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("  FISTA restart t=%d (%.1fs): J=%.6f" % (t, elapsed, J))
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  FISTA t=%d (%.1fs): J=%.6f beta=%.0f" % (t, elapsed, J, beta))
    
    return best_a, best_m


def simplex_project(z):
    """Project z onto {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def cyclic_shift_best(a0, verbose=False):
    """Try all cyclic shifts and return the best one."""
    n = len(a0)
    a = normalize(a0)
    best_m = fast_autocorr(a).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for s in range(1, n):
        a_shift = np.roll(a, -s)
        A = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    
    if best_shift > 0:
        return np.roll(a, -best_shift), best_m
    return a.copy(), best_m


def local_search_1move(a0, time_limit, verbose=True):
    """1-move local search: move mass from one position to another."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:20]
        improved = False
        
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            
            k_int = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            if a[src] < 1e-12:
                continue
            
            # Find a good destination: position with low "engagement"
            # (low sum of a[j]*a[k-j] over all k)
            engagement = np.zeros(n)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                engagement[s2:e2+1] += a[s2:e2+1] * a[k2i - (s2:e2+1)]
            
            candidates = np.argsort(engagement)[:n // 2 + 1]
            candidates = candidates[candidates != src]
            if len(candidates) == 0:
                candidates = np.arange(n)
                candidates = candidates[candidates != src]
            dst = int(candidates[rng.integers(len(candidates))])
            
            for frac in [0.01, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.01, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
    
    return best_a, best_m


def interpolate_to_length(f, new_n):
    """Interpolate sequence f to new length new_n."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
