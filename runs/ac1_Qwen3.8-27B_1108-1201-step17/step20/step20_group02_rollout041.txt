```
def project_simplex(v):
    v=np.asarray(v,float)
    n=len(v)
    if n==1: return np.ones(1)
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    cond = u * np.arange(1,n+1) > (cssv - 1.0)
    idx = np.flatnonzero(cond)
    if idx.size == 0:
        rho=0
    else:
        rho=int(idx[-1])
    theta = (cssv[rho] - 1.0)/(rho+1.0)
    w = np.maximum(v - theta, 0.0)
    s=w.sum()
    if s <= 0: return np.ones(n)/n
    return w/s
```

```
       return M, w, M
    w = ew / s
    soft = M + gamma * float(np.log(s))
    if not np.isfinite(soft): soft = M
    return soft, w, M
```

`project_simplex`: Ensure input finite.
```
def project_simplex(v):
    v = np.where(np.isfinite(v), v, 0.0)
    n = v.size
    if n == 1:
       return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    idx = np.flatnonzero(cond)
    if idx.size == 0:
       rho = 0
    else:
       rho = int(idx[-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-15:
       return np.ones(n)/float(n)
    return w / s
```

```
def resample_shape(a, new_n):
    a=np.asarray(a,float)
    if new_n <=0: return np.array([])
    if a.size==0: return np.ones(new_n)/new_n
    a=np.maximum(a,0)
    s=a.sum()

    xo=np.linspace(0.0,1.0,old_n)
    xn=np.linspace(0.0,1.0,new_n)
    out=np.interp(xn,xo,a)
    out=np.maximum(out,0)
    s=out.sum()
    if s<=1e-15: return np.ones(new_n)/new_n
    return out/s
```

```
def gaussian_smooth(a, sigma):
    n=a.size
    half = max(1, int(4*sigma+1))
    x = np.arange(-half, half+1, dtype=float)
    kern = np.exp(-(x*x)/(2.0*sigma*sigma))
    ks = kern.sum()
    if ks <= 0: return a.copy()
    kern = kern/ks
    out = np.convolve(a, kern, mode='same')
    out = np.maximum(out,0)
    s=out.sum()
    if s <=1e-15: return a.copy()
    return out/s
```

```python
"""
Search for low-height step functions minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

The code starts from the supplied parent sequence if available, tries
re-sampled lengths and simple edge-weighted priors, then uses a new
smoothing/max-gradient method.  The main optimizer computes a soft maximum
of the convolution coefficients, differentiates it exactly through FFT
convolution, and uses alternating multiplicative / clipped-additive
moves with adaptive step sizes.  A gradient-informed coordinate polish is
used to remove residual discrete artifacts.
"""

import numpy as np
import time


def next_pow2(value):
    if value <= 1:
        return 1
    size = 1
    while size < value:
        size <<= 1
    return size


def normalize_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.array([1.0])
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def prepare_positive(a, floor=1e-14):
    a = normalize_sum1(a)
    a = np.maximum(a, 0.0)
    if a.size > 0:
        a = a + float(floor)
        s = float(a.sum())
        if s > 1e-15 and np.isfinite(s):
            a = a / s
    return a


def resample_shape(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([], dtype=np.float64)
    if a.size == 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    a = a / s
    if a.size == new_n:
        return a.copy()
    old_n = int(a.size)
    xo = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    xn = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    out = np.interp(xn, xo, a)
    out = np.maximum(out, 0.0)
    s2 = float(out.sum())
    if s2 <= 1e-15 or not np.isfinite(s2):
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return out / s2


def make_prior(n, kind):
    n = int(n)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    if kind == 0:
        f = np.ones(n, dtype=np.float64)
    elif kind == 1:
        f = x * x + (1.0 - x) * (1.0 - x) + 0.08
    elif kind == 2:
        f = x ** 2 + (1.0 - x) ** 2 + 0.2 * x * (1.0 - x) + 0.02
    elif kind == 3:
        f = (x + 0.05) ** 2 + ((1.0 - x) + 0.05) ** 2 + 0.01
    elif kind == 4:
        f = np.exp(-6.0 * x) + np.exp(-6.0 * (1.0 - x)) + 0.02
    else:
        f = np.ones(n, dtype=np.float64)
    return normalize_sum1(f)


def gaussian_smooth(a, sigma):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    half = max(1, int(4.0 * float(sigma) + 1.0))
    x = np.arange(-half, half + 1, dtype=np.float64)
    kern = np.exp(-(x * x) / (2.0 * float(sigma) * float(sigma)))
    ks = float(kern.sum())
    if ks <= 1e-15 or not np.isfinite(ks):
        return a.copy()
    kern = kern / ks
    out = np.convolve(a, kern, mode="same")
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return a.copy()
    return out / s


def random_perturb(a, scale, rng):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a
    noise = rng.standard_normal(n)
    fac = np.exp(np.clip(float(scale) * noise, -3.0, 3.0))
    b = a * fac
    b = np.maximum(b, 0.0)
    b = b + 1e-14
    s = float(b.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return a.copy()
    return b / s


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    n = v.size
    if n == 0:
        return v
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    idx = np.flatnonzero(cond)
    if idx.size == 0:
        rho = 0
    else:
        rho = int(idx[-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return w / s


def convolve_full_fft(a, fsize):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    out_len = 2 * n - 1
    if fsize < out_len:
        fsize = next_pow2(out_len)
    af = np.fft.rfft(a, fsize)
    c = np.fft.irfft(af * af, fsize)[:out_len]
    c = np.maximum(c, 0.0)
    c = np.where(np.isfinite(c), c, 0.0)
    return c


def fast_score_sequence(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-12 or not np.isfinite(s):
        return float("inf")
    a = a / s
    n = int(a.size)
    fsize = next_pow2(2 * n - 1)
    c = convolve_full_fft(a, fsize)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def soft_information(c, frac):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([], dtype=np.float64), 0.0
    M = float(np.max(c))
    if not np.isfinite(M):
        M = 0.0
    if M <= 1e-300:
        return 0.0, np.zeros_like(c), 0.0
    gamma = float(max(M * float(frac), 1e-15))
    delta = (c - M) / gamma
    delta = np.clip(delta, -50.0, 0.0)
    ew = np.exp(delta)
    s = float(ew.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return M, w, M
    w = ew / s
    soft = M + gamma * float(np.log(s))
    if not np.isfinite(soft):
        soft = M
    return soft, w, M


def gradient_from_weights(weights, a, fsize):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    w = np.asarray(weights, dtype=np.float64).ravel()
    if w.size != L:
        tmp = np.zeros(L, dtype=np.float64)
        take = min(L, w.size)
        if take > 0:
            tmp[:take] = w[:take]
        w = tmp
    need = L + n - 1
    if fsize < need:
        fsize = next_pow2(need)
    ar = np.fft.rfft(a[::-1], fsize)
    wf = np.fft.rfft(w, fsize)
    gfull = np.fft.irfft(wf * ar, fsize)
    g = 2.0 * gfull[n - 1:n - 1 + n]
    g = np.maximum(g, 0.0)
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def make_mult_candidate(a, h_scaled, move):
    h_scaled = np.where(np.isfinite(h_scaled), h_scaled, 0.0)
    e = -float(move) * h_scaled
    e = np.clip(e, -40.0, 40.0)
    b = a * np.exp(e)
    b = np.maximum(b, 0.0)
    b = b + 1e-14
    s = float(b.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return a.copy()
    return b / s


def make_add_candidate(a, h_scaled, move):
    h_scaled = np.where(np.isfinite(h_scaled), h_scaled, 0.0)
    x = a - float(move) * h_scaled
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    x = x + 1e-14
    s = float(x.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return a.copy()
    return x / s


def smooth_multiplicative_optimize(a0, time_limit, verbose=True):
    t0 = time.monotonic()
    a = prepare_positive(a0, 1e-14)
    n = int(a.size)
    if n == 0:
        return a0, fast_score_sequence(a0)

    fsize = next_pow2(max(1, 3 * n - 2))
    c = convolve_full_fft(a, fsize)
    M = float(np.max(c))
    current_score = float(2.0 * n * M)
    best_a = a.copy()
    best_score = current_score

    frac = 0.30
    min_frac = 0.015
    soft, w, _ = soft_information(c, frac)

    log_move = 0.08
    add_move = 0.004
    mode = 0
    h_vel = None
    rng = np.random.default_rng(555)

    soft_stall = 0
    score_since_improve = 0
    drift = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < 1000000:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        if (not np.isfinite(current_score)) or (current_score > best_score + 0.004):
            drift += 1
            if drift > 200:
                a = best_a.copy()
                c = convolve_full_fft(a, fsize)
                M = float(np.max(c))
                current_score = float(2.0 * n * M)
                soft, w, _ = soft_information(c, frac)
                h_vel = None
                log_move = 0.08
                add_move = 0.004
                soft_stall = 0
                drift = 0
                it += 1
                continue
        else:
            drift = 0

        M = float(np.max(c))
        if M <= 1e-300:
            break

        soft, w, M = soft_information(c, frac)
        g = gradient_from_weights(w, a, fsize)
        mu = float(np.dot(a, g))
        h = g - mu
        h = np.where(np.isfinite(h), h, 0.0)

        if h_vel is None:
            h_vel = h.copy()
            h_use = h
        else:
            h_vel = 0.7 * h + 0.3 * h_vel
            h_use = h_vel

        h_absmax = float(np.max(np.abs(h_use)))
        if not np.isfinite(h_absmax) or h_absmax < 1e-18:
            base = best_a if np.isfinite(best_score) else a
            a = random_perturb(base, 0.02, rng)
            c = convolve_full_fft(a, fsize)
            M = float(np.max(c))
            current_score = float(2.0 * n * M)
            soft, w, _ = soft_information(c, frac)
            h_vel = None
            it += 1
            continue

        h_scaled = h_use / h_absmax
        h_scaled = np.where(np.isfinite(h_scaled), h_scaled, 0.0)

        accepted = False
        trial_score = None

        attempts = 0
        while attempts < 6:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 0.5:
                break

            if mode == 0:
                if log_move < 1e-7:
                    mode = 1
                    attempts += 1
                    continue
                trial = make_mult_candidate(a, h_scaled, log_move)
            else:
                if add_move < 1e-8:
                    mode = 0
                    attempts += 1
                    continue
                trial = make_add_candidate(a, h_scaled, add_move)

            trial_c = convolve_full_fft(trial, fsize)
            trial_soft, _, trial_M = soft_information(trial_c, frac)
            trial_score = float(2.0 * n * trial_M)

            if np.isfinite(trial_score) and trial_soft < soft - 1e-16:
                a = trial
                c = trial_c
                M = trial_M
                soft = trial_soft
                current_score = trial_score
                accepted = True
                if mode == 0:
                    log_move = min(0.25, log_move * 1.15)
                else:
                    add_move = min(0.05, add_move * 1.15)
                soft_stall = 0
                break
            else:
                if mode == 0:
                    log_move = max(1e-8, log_move * 0.5)
                else:
                    add_move = max(1e-9, add_move * 0.5)

            mode = 1 - mode
            attempts += 1

        if accepted:
            if trial_score is not None and trial_score < best_score - 1e-14:
                best_score = trial_score
                best_a = a.copy()
                score_since_improve = 0
            else:
                score_since_improve += 1

            if score_since_improve > 3000:
                frac = max(min_frac, frac * 0.8)
                score_since_improve = 0
        else:
            soft_stall += 1
            if soft_stall % 100 == 0:
                frac = max(min_frac, frac * 0.8)
                log_move = max(1e-7, log_move * 0.7)
                add_move = max(1e-8, add_move * 0.7)
                h_vel = None

                if soft_stall % 500 == 0:
                    trial = gaussian_smooth(a, 2.0)
                    trial_c = convolve_full_fft(trial, fsize)
                    trial_M = float(np.max(trial_c))
                    trial_score_trial = float(2.0 * n * trial_M)
                    if np.isfinite(trial_score_trial) and (
                        trial_score_trial < current_score - 1e-14
                        or trial_score_trial < best_score - 1e-14
                    ):
                        a = trial
                        c = trial_c
                        M = trial_M
                        current_score = trial_score_trial
                        soft, w, _ = soft_information(c, frac)

                if soft_stall > 3000:
                    a = random_perturb(best_a, 0.03, rng)
                    c = convolve_full_fft(a, fsize)
                    M = float(np.max(c))
                    current_score = float(2.0 * n * M)
                    soft, w, _ = soft_information(c, frac)
                    h_vel = None
                    log_move = 0.08
                    add_move = 0.004
                    soft_stall = 0
                    drift = 0

        it += 1
        if it % 2000 == 0:
            frac = max(min_frac, frac * 0.97)

        if verbose and it % 1000 == 0:
            print(
                "    smooth t=" + str(round(time.monotonic() - t0, 1)) +
                " iter=" + str(it) +
                " curr=" + repr(current_score) +
                " best=" + repr(best_score) +
                " frac=" + repr(frac)
            )

    if np.isfinite(current_score) and current_score < best_score:
        best_score = current_score
        best_a = a.copy()

    return best_a, best_score


def coordinate_gradient_polish(a0, time_limit, seed=777, verbose=True):
    t0 = time.monotonic()
    a = prepare_positive(a0, 1e-14)
    n = int(a.size)
    if n == 0:
        return a0, fast_score_sequence(a0)

    fsize_score = next_pow2(2 * n - 1)
    fsize_grad = next_pow2(max(1, 3 * n - 2))

    c = convolve_full_fft(a, fsize_score)
    M = float(np.max(c))
    current_score = float(2.0 * n * M)
    best_a = a.copy()
    best_score = current_score

    rng = np.random.default_rng(seed)
    pool_size = 150
    stall = 0
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        c = convolve_full_fft(a, fsize_score)
        M = float(np.max(c))
        current_score = float(2.0 * n * M)

        if np.isfinite(current_score) and current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stall = 0

        soft, w, M = soft_information(c, 0.03)
        g = gradient_from_weights(w, a, fsize_grad)
        mu = float(np.dot(a, g))
        h = g - mu
        h = np.where(np.isfinite(h), h, 0.0)

        hmax = float(np.max(np.abs(h)))
        if hmax > 1e-18:
            top_pos = np.argsort(h)[::-1][:pool_size]
            top_neg = np.argsort(h)[:pool_size]
        else:
            top_pos = rng.integers(0, n, size=pool_size)
            top_neg = rng.integers(0, n, size=pool_size)

        improved = False

        for _ in range(70):
            rem = time_limit - (time.monotonic() - t0)
            if rem < 0.5:
                break
            s_idx = int(top_pos[rng.integers(len(top_pos))])
            d_idx = int(top_neg[rng.integers(len(top_neg))])
            if s_idx == d_idx or a[s_idx] <= 1e-12:
                continue

            fr_ok = False
            for fr in (0.02, 0.005, 0.05):
                amt = float(a[s_idx] * fr)
                if amt < 1e-15:
                    break
                new = a.copy()
                new[s_idx] -= amt
                new[d_idx] += amt
                new = np.maximum(new, 0.0)
                ns = float(new.sum())
                if ns <= 1e-15 or not np.isfinite(ns):
                    break
                new = new / ns
                nc = convolve_full_fft(new, fsize_score)
                nm = float(np.max(nc))
                nsc = float(2.0 * n * nm)
                if np.isfinite(nsc) and nsc < current_score - 1e-15:
                    a = new
                    c = nc
                    current_score = nsc
                    improved = True
                    fr_ok = True
                    break
            if improved:
                break

        if not improved:
            for _ in range(20):
                rem = time_limit - (time.monotonic() - t0)
                if rem < 0.5:
                    break
                s_idx = int(rng.integers(n))
                d_idx = int(rng.integers(n))
                if s_idx == d_idx or a[s_idx] <= 1e-12:
                    continue
                fr = float(rng.uniform(0.002, 0.03))
                amt = float(a[s_idx] * fr)
                if amt < 1e-15:
                    continue
                new = a.copy()
                new[s_idx] -= amt
                new[d_idx] += amt
                new = np.maximum(new, 0.0)
                ns = float(new.sum())
                if ns <= 1e-15:
                    continue
                new = new / ns
                nc = convolve_full_fft(new, fsize_score)
                nm = float(np.max(nc))
                nsc = float(2.0 * n * nm)
                if np.isfinite(nsc) and nsc < current_score - 1e-15:
                    a = new
                    c = nc
                    current_score = nsc
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall % 15 == 0:
                trial = gaussian_smooth(a, 2.0)
                tc = convolve_full_fft(trial, fsize_score)
                tm = float(np.max(tc))
                tsc = float(2.0 * n * tm)
                if np.isfinite(tsc) and tsc < current_score - 1e-14:
                    a = trial
                    c = tc
                    current_score = tsc
                    stall = 0

            if stall > 300:
                break
        else:
            stall = 0

        epoch += 1
        if verbose and epoch % 20 == 0:
            print(
                "    coord t=" + str(round(time.monotonic() - t0, 1)) +
                " epoch=" + str(epoch) +
                " curr=" + repr(current_score) +
                " best=" + repr(best_score)
            )

    if np.isfinite(current_score) and current_score < best_score:
        best_score = current_score
        best_a = a.copy()

    return best_a, best_score


def valid_score(a):
    try:
        vals = [float(x) for x in a]
        sc = evaluate_sequence(vals)
        if np.isfinite(sc):
            return float(sc)
    except Exception:
        pass
    return float("inf")


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    total_budget = min(float(budget_s) - 45.0, 1740.0)
    if total_budget < 120.0:
        total_budget = 120.0

    print("Total time budget: " + repr(round(total_budget, 3)) + "s")

    cand_seqs = []
    cand_scores = []
    parent_seq = None

    try:
        par = globals().get("height_sequence_1", None)
        if par is not None:
            p = np.asarray(par, dtype=np.float64)
            p = np.clip(p, 0.0, 1e9)
            if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and float(p.sum()) > 1e-10:
                p = normalize_sum1(p)
                sc = fast_score_sequence(p)
                cand_seqs.append(p)
                cand_scores.append(sc)
                parent_seq = p
                print("Parent fast score: " + repr(sc) + ", n=" + str(p.size))
    except Exception:
        pass

    if not cand_seqs:
        fb_n = 8192
        fb = np.ones(fb_n, dtype=np.float64) / float(fb_n)
        sc = fast_score_sequence(fb)
        cand_seqs.append(fb)
        cand_scores.append(sc)
        print("No parent usable, fallback uniform n=8192: " + repr(sc))

    base_min = min(cand_scores)
    base_idx = int(np.argmin(cand_scores))
    base_seq = cand_seqs[base_idx]
    base_n = int(base_seq.size)

    base_smooth = gaussian_smooth(base_seq, 1.5)
    base_smooth_score = fast_score_sequence(base_smooth)
    if base_smooth_score < base_min + 1e-9:
        cand_seqs.append(base_smooth)
        cand_scores.append(base_smooth_score)

    cand_n_set = set()
    for f in (0.8, 1.25, 1.5, 1.75, 2.0):
        nn = int(round(base_n * f))
        if 3000 <= nn <= 18000:
            cand_n_set.add(nn)
    for fixed_n in (4096, 6144, 8192, 10240, 12288, 14336, 16384, 18000):
        if 3000 <= fixed_n <= 18000:
            cand_n_set.add(fixed_n)

    seen_ns = set()
    for nn in sorted(cand_n_set):
        if nn in seen_ns:
            continue
        seen_ns.add(nn)
        seq = resample_shape(base_seq, nn)
        sc = fast_score_sequence(seq)
        cand_seqs.append(seq)
        cand_scores.append(sc)

    prior_n = int(min(16384, max(8192, base_n)))
    for kind in (1, 2, 4, 0):
        seq = make_prior(prior_n, kind)
        sc = fast_score_sequence(seq)
        cand_seqs.append(seq)
        cand_scores.append(sc)

    if not cand_seqs:
        fb = np.ones(4096, dtype=np.float64) / 4096.0
        return [float(x) for x in fb]

    best_idx = 0
    best_score = cand_scores[0]
    best_seq = cand_seqs[0]
    for i, sc in enumerate(cand_scores):
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand_seqs[i]
            best_idx = i

    ev = valid_score(best_seq)
    if np.isfinite(ev):
        best_score = min(best_score, ev)
        print("Initial selected fast=" + repr(cand_scores[best_idx]) +
              " validated=" + repr(ev) + " n=" + str(best_seq.size))

    remaining = total_budget - (time.monotonic() - t_start)
    phase1 = min(remaining * 0.55, 1000.0)
    if phase1 > 90.0:
        print("\n=== Phase 1: smooth max-gradient optimizer ===")
        a_out, sc_out = smooth_multiplicative_optimize(
            np.asarray(best_seq, dtype=np.float64), float(phase1), verbose=True
        )
        if np.isfinite(sc_out) and sc_out < best_score:
            best_score = sc_out
            best_seq = np.asarray(a_out, dtype=np.float64)
            print("*** NEW BEST after smooth: " + repr(best_score) +
                  ", n=" + str(best_seq.size))

    remaining = total_budget - (time.monotonic() - t_start)
    phase2 = min(remaining * 0.25, 300.0)
    if phase2 > 40.0:
        print("\n=== Phase 2: gradient-informed coordinate polish ===")
        a_out, sc_out = coordinate_gradient_polish(
            np.asarray(best_seq, dtype=np.float64), float(phase2), seed=101, verbose=True
        )
        if np.isfinite(sc_out) and sc_out < best_score:
            best_score = sc_out
            best_seq = np.asarray(a_out, dtype=np.float64)
            print("*** NEW BEST after coord: " + repr(best_score) +
                  ", n=" + str(best_seq.size))

    target_set = set()
    cur_n = int-best_seq.size if False else int(best_seq.size)
    for f in (1.25, 1.5, 1.8, 2.0):
        nn = int(round(cur_n * f))
        if cur_n < nn <= 18000:
            target_set.add(nn)
    for fixed_n in (10240, 12288, 14336, 16384, 18000):
        if cur_n < fixed_n <= 18000:
            target_set.add(fixed_n)
    targets = sorted(target_set)

    for nn in targets[:2]:
        remaining = total_budget - (time.monotonic() - t_start)
        if remaining < 260.0:
            break
        print("\n=== Larger-n attempt: n=" + str(nn) + " ===")
        cand = resample_shape(np.asarray(best_seq, dtype=np.float64), nn)
        sc = fast_score_sequence(cand)
        if np.isfinite(sc) and sc < best_score + 0.008:
            t_phase = min(remaining * 0.35, 420.0)
            if t_phase > 60.0:
                a_out, sc_out = smooth_multiplicative_optimize(cand, float(t_phase), verbose=False)
                if np.isfinite(sc_out) and sc_out < best_score:
                    best_score = sc_out
                    best_seq = np.asarray(a_out, dtype=np.float64)
                    print("*** NEW BEST after larger-n smooth: " + repr(best_score))

            remaining = total_budget - (time.monotonic() - t_start)
            tc = min(remaining * 0.20, 180.0)
            if tc > 40.0:
                a_out, sc_out = coordinate_gradient_polish(
                    np.asarray(best_seq, dtype=np.float64), float(tc), seed=300, verbose=False
                )
                if np.isfinite(sc_out) and sc_out < best_score:
                    best_score = sc_out
                    best_seq = np.asarray(a_out, dtype=np.float64)
                    print("*** NEW BEST after larger-n coord: " + repr(best_score))

    remaining = total_budget - (time.monotonic() - t_start)
    if remaining > 50.0:
        tc = min(remaining * 0.85, 400.0)
        print("\n=== Final coordinate polish ===")
        a_out, sc_out = coordinate_gradient_polish(
            np.asarray(best_seq, dtype=np.float64), float(tc), seed=800, verbose=True
        )
        if np.isfinite(sc_out) and sc_out < best_score:
            best_score = sc_out
            best_seq = np.asarray(a_out, dtype=np.float64)
            print("*** NEW BEST final coord: " + repr(best_score))

    final_arr = np.asarray(best_seq, dtype=np.float64)
    final_arr = np.where(np.isfinite(final_arr), final_arr, 0.0)
    final_arr = np.maximum(final_arr, 0.0)
    s = float(final_arr.sum())
    if s <= 1e-10 or not np.isfinite(s):
        if parent_seq is not None:
            final_arr = np.asarray(parent_seq, dtype=np.float64)
            s = float(final_arr.sum())
        if s <= 1e-10 or not np.isfinite(s):
            final_arr = np.ones(4096, dtype=np.float64) / 4096.0
            s = float(final_arr.sum())
    final_arr = final_arr / s

    final_list = [float(x) for x in final_arr]
    final_score = valid_score(final_list)

    if not np.isfinite(final_score):
        fb = np.ones(4096, dtype=np.float64) / 4096.0
        final_list = [float(x) for x in fb]
        final_score = valid_score(final_list)

    print("\nFINAL internal fast score: " + repr(best_score))
    print("FINAL evaluated score: " + repr(final_score))
    print("FINAL n: " + str(final_list.size if hasattr(final_list, "size") else len(final_list)))

    return final_list
```
