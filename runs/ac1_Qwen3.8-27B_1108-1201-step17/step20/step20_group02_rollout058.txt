def fc(a):
    """Fast convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def norm1(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interp_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

    if skew != 0:

def compute_grad(a, c, top_k):
    """Compute gradient of top-k weighted peaks."""
    n = len(a)
    K = np.argsort(c)[::-1][:top_k]
    cK = c[K]
    w = cK - cK.max()
    w = np.exp(w)
    w = w / w.sum()
    g = np.zeros(n)
    for j in range(len(K)):
        k = int(K[j])
        wj = w[j]
        if wj < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            g[idx] += wj * a[k - idx]
    return 2.0 * g

    a = norm1(a0)
    c = fc(a)
    best_a = a.copy()
    best_val = 2.0 * n * c.max()
    t0 = time.monotonic()

    # Annealing schedule
    c_max_init = c.max()
    tau = c_max_init * 0.3
    tau_min = c_max_init * 0.02
    tau_decay = 0.97
    top_k = 30

    iter_count = 0
    stall = 0
    max_stall = 500
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        c = fc(a)
        cmax = c.max()
        obj_val = 2.0 * n * cmax

        if cmax < best_val / (2.0 * n) - 1e-15:
            best_val = obj_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_stall:
            # Restart with perturbation
            rng_r = np.random.default_rng(hash((n, no_improve)) % (2**31))
            noise = rng_r.normal(0, 0.05, n) * best_a
            a = np.maximum(best_a + noise, 1e-15)
            a = a / a.sum()
            no_improve = 0
            top_k = max(20, top_k - 5)
            tau = c.max() * 0.3
            if no_improve > 2000:
                break

        # Adaptive top_k
        if iter_count % 100 == 0:
            top_k = min(80, max(15, top_k + 2))

        # Compute gradient (weighted top-k subgradient)
        g = compute_grad(a, c, top_k)

        # Mirror descent with backtracking line search
        best_alpha = 0.02 / (1.0 + iter_count / 200.0)
        best_trial = cmax

        # Try several step sizes
        base_eta = 0.1 / (1.0 + iter_count / 500.0)
        etas = [base_eta * f for f in [4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.5]]
        etas = sorted(set(etas), reverse=True)

        for eta in etas:
            log_mult = -eta * g
            log_mult -= log_mult.max()
            mult = np.exp(log_mult)
            a_trial = a * mult
            a_trial = np.maximum(a_trial, 1e-20)
            st = a_trial.sum()
            if st < 1e-15:

        print("    SMMD done (" + str(round(elapsed, 1)) + "s): best=" + repr(best_val) + " iters=" + str(iter_count))
    return best_a, best_val


def lp_fw(a0, time_limit, solver, initial_K=500, max_K=4000, add_per_iter=50, verbose=True):
    """LP-Frank-Wolfe with adaptive active set."""
    n = len(a0)
    y = norm1(a0)
    ac_y = fc(y)
    best_m = ac_y.max()
    best_a = y.copy()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 600
    iter_count = 0
    stagnation_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation_count > 4:
                if verbose:
                    print("    LP-FW: persistent stagnation, stopping")
                break
            ac_curr = fc(y)
            n_pert = max(50, len(active_set) // 5)
            rng_s = np.random.default_rng(stagnation_count * 42 + n)
            keep = list(rng_s.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 2]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)
            no_improve = 0
            stagnation_count += 1
            continue

        iter_count += 1
        ac = fc(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation_count = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            n_keep = int(max_K * 0.7)
            keep_idx = np.argsort(margins)[:n_keep]
            active_set_base = [active_set[i] for i in keep_idx]
            violations = np.where(ac > M_curr - 1e-8)[0]
            for v in violations[:100]:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))

        K = min(len(active_set), max_K)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj in range(K):
            k_int = int(active_set[jj])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs[:K] @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K // 3)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add to active set
        fs = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fs)
        Gf = np.fft.rfft(g_dir, fs)
        conv_ys = np.fft.irfft(Yf * Gf, fs)[:2 * n - 1]

        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Backtracking line search
        best_trial_M = fc(y).max()
        best_alpha = 1.0 / (iter_count + 2.0)

        alphas = [1.0, 0.7, 0.5, 0.3, 0.15, 0.08, 0.04]
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fc(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = (1.0 - best_alpha) * y + best_alpha * g_dir
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " |S|=" + str(len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m) + " iters=" + str(iter_count))
    return best_a, best_m


def coord_polish(a0, time_limit, rng_seed=77, verbose=False):
    """Coordinate descent polish focusing on peak reduction."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_m = fc(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 400
    step_scales = [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fc(a)
        n_top = min(20, len(c))
        top_k = np.argsort(c)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-impact destination
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            candidates = np.where(scores < scores.mean())[0]
            if len(candidates) < n // 5:
                candidates = np.arange(n)

            dst = int(candidates[np.argmin(scores[candidates])])
            if dst == src:
                continue

            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fc(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.008)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fc(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fc(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = safe_eval(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score
