            x_new = x
            max_new = fast_autocorr(x_new).max()
        else:
            # Backtracking line search
            found = False
            y_max = fast_autocorr(y).max()
            for _ in range(8):
                z = y - eta * grad
                x_new = simplex_project(z)
                max_new = fast_autocorr(x_new).max()
                if max_new <= y_max + 1e-14:
                    found = True
                    break
                eta *= 0.5
            if not found:
                x_new = x
                max_new = fast_autocorr(x).max()
        
        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 5.0)
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Increase beta periodically
        if it % 100 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart if stalled
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            eta = eta_init
            beta = max(beta_start * 2, beta * 0.5)
            
            strategy = rng.integers(6)
            if strategy == 0:
                noise = rng.normal(0, 0.01 + 0.02 * rng.random(), n) * best_a
                x_r = best_a + noise
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.2)
            elif strategy == 2:
                x_r = best_a * np.exp(0.05 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= 0.5 + 0.5 * rng.random(len(np.where(mask)[0]))
                x_r[~mask] *= 1.0 + 0.5 * rng.random(len(np.where(~mask)[0]))
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 20)))
                x_r = np.roll(best_a, shift)
            else:
                # Gaussian bump perturbation
                center = int(rng.integers(n))
                width = int(rng.integers(n // 50, n // 10))
                bump = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
                x_r = best_a * (1 + 0.1 * bump * rng.normal(0, 1))
            
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest it=%d (%.1fs): J=%.6f beta=%.0f" % (it, elapsed, J, beta))
    
    return best_a, best_max_b

def compute_grad_topk(a, beta, top_k):
    """Gradient using only top-k autocorrelation values in softmax."""
    n = len(a)
    size = 2 * n - 1
    b = fast_autocorr(a)
    
    idx = np.argsort(b)[::-1][:top_k]
    b_use = np.zeros(size)
    b_use[idx] = b[idx]
    b_max = b[idx].max()
    
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -100, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.zeros(size)
        p[idx] = 1.0 / top_k
    else:
        p = exp_vals / s
    
    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pf = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pf, fft_size)[:conv_len]
    
    grad = (2.0 / beta) * conv[n-1:n-1+n]
    return grad, b, b_max

def peak_shaving(a0, time_limit, verbose=True, seed=789):
    """
    Targeted local search: identify top critical lags and try to reduce them
    by moving mass from contributing positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 5000:
        b = fast_autocorr(a)
        size = 2 * n - 1
        # Get top critical lags
        top_lags = np.argsort(b)[::-1][:min(30, size)]
        improved = False
        
        for lag in top_lags[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            lag_int = int(lag)
            # Positions i where a[i] * a[lag_int - i] contributes to b[lag_int]
            i_lo = max(0, lag_int - n + 1)
            i_hi = min(n - 1, lag_int)
            if i_lo > i_hi:
                continue
            idx = np.arange(i_lo, i_hi + 1)
            contribs = a[idx] * a[lag_int - idx]
            
            # Find the dominant contributor
            src = int(idx[np.argmax(contribs)])
            partner = lag_int - src
            if not (0 <= partner < n):
                continue
            src_val = a[src]
            partner_val = a[partner]
            
            if src_val < 1e-12:
                continue
            
            # Try reducing src (or partner) and moving mass elsewhere
            # Move a small fraction from the larger of (src, partner) to a random other position
            if src_val >= partner_val:
                move_from = src
                other = partner
            else:
                move_from = partner
                other = src
            
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[move_from] * frac
                if amt < 1e-15:
                    continue
                # Try several random destinations
                for _ in range(8):
                    dst = int(rng.integers(n))
                    if dst == move_from or dst == other:
                        continue
                    a_trial = a.copy()
                    a_trial[move_from] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-17:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random exploration
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if verbose and stall % 1000 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakShave (%.1fs): J=%.6f stall=%d" % (elapsed, J, stall))
    
    return best_a, best_m

def spectral_flatten(a0, time_limit, verbose=True, seed=321):
    """
    Spectral domain: iteratively flatten |A(f)|^2 while maintaining non-negativity.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    alpha = 0.05
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Target: flat spectrum above DC
        target = power[1:].mean() if len(power) > 1 else power[0]
        if target < 1e-30:
            break
        
        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / target
            if ratio > 1.0:
                red = 1.0 - alpha * min(ratio - 1.0, 5.0)
                A_new[k] *= max(0.3, np.sqrt(red))
            elif ratio < 0.3:
                inc = 1.0 + alpha * 0.5 * (0.3 - ratio)
                A_new[k] *= min(2.0, np.sqrt(inc))
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 500 == 0:
            # Phase perturbation
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.03
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial
        
        if it % 1000 == 0:
            alpha = max(0.01, alpha * 0.97)
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=%d (%.1fs): J=%.6f" % (it, elapsed, J))
    
    return best_a, best_m
