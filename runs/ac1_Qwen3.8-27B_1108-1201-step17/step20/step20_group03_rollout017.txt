def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad_lse(a, beta, top_k=None):
    """Compute gradient of log-sum-exp smoothed max autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    if top_k is not None and 0 < top_k < len(b):
        idx_all = np.argsort(b)[::-1]
        idx = idx_all[:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        mask = (b_use == b_max)
        p = np.zeros_like(b_use)
        p[mask] = 1.0 / max(1, mask.sum())
    else:
        p = exp_vals / p_total
    # Gradient: convolve reversed a with p, take middle part
    a_rev = a[::-1]
    conv_len = n + len(p) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1]
    return grad, b, b.max()


def compute_grad_power(a, alpha):
    """Compute gradient of power-smoothed max autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    b_clipped = np.maximum(b, 1e-30)
    weights = b_clipped ** (alpha - 1)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.ones_like(b) / len(b)
    w = weights / w_sum
    a_rev = a[::-1]
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Wr = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(Ar * Wr, fft_size)
    grad = conv[n - 1:2 * n - 1]
    # Scale: d/da (sum b^alpha)^(1/alpha) = (sum b^alpha)^(1/alpha - 1) * sum b^(alpha-1) * db/da
    g_val = (b_clipped ** alpha).sum() ** (1.0 / alpha - 1.0)
    grad = grad * g_val * alpha
    return grad, b, b.max()


def gradient_descent_smoothed(a0, time_limit, method='lse', beta_start=20.0,
                               beta_end=1000.0, beta_growth=1.004,
                               top_k=None, restart_every=3000, verbose=True):
    """Gradient descent with smoothed max, adaptive step, and restarts."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    step = 1.0 / (n ** 0.5)

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        if method == 'lse':
            grad, b, b_max = compute_grad_lse(y, beta, top_k)
        else:
            grad, b, b_max = compute_grad_power(y, alpha=max(beta, 4.0))

        # Backtracking line search
        eta = step
        best_val = fast_autocorr(x).max()
        for attempt in range(15):
            z = y - eta * grad
            x_new = simplex_project(z)
            m_new = fast_autocorr(x_new).max()
            if m_new < best_val - 1e-15 * best_val:
                break
            eta *= 0.7
        else:
            x_new = simplex_project(y - eta * grad)
            m_new = fast_autocorr(x_new).max()

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
            step = min(step * 1.05, 2.0 / beta)
        else:
            stall += 1
            step = max(step * 0.95, 1e-8)

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.01 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            step = 1.0 / (n ** 0.5)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    GD restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    GD t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + repr(round(beta, 1)) + " step=" + repr(round(eta, 6)))

    return best_a, best_m


def spectral_flatten(a0, time_limit, n_iters=500, verbose=True):
    """Flatten the power spectrum to reduce peak autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        b = np.fft.irfft(power, fft_size)[:2 * n - 1]
        b_max = b.max()
        if b_max < 1e-15:
            break

        # Try to flatten spectrum: reduce peaks in power spectrum
        power_mean = power[1:].mean()  # exclude DC
        power_std = power[1:].std()
        if power_std < 1e-15:
            break

        # Soft threshold: reduce power above mean
        threshold = power_mean + 0.5 * power_std
        mask = power > threshold
        # Attenuate high-power bins
        attenuation = np.ones(len(A))
        attenuation[mask] = 1.0 - 0.1 * (power[mask] - threshold) / (power[mask] - threshold + 1e-10)
        attenuation = np.maximum(attenuation, 0.1)

        A_new = A * attenuation
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        m_new = fast_autocorr(a_new).max()

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try boosting low-power bins instead
            mask_low = power < power_mean - 0.5 * power_std
            attenuation2 = np.ones(len(A))
            attenuation2[mask_low] = 1.0 + 0.05
            A_new2 = A * attenuation2
            a_new2 = np.fft.irfft(A_new2, fft_size)[:n]
            a_new2 = np.maximum(a_new2, 0.0)
            s_new2 = a_new2.sum()
            if s_new2 > 1e-15:
                a_new2 = a_new2 / s_new2
                m_new2 = fast_autocorr(a_new2).max()
                if m_new2 < best_m - 1e-15:
                    best_m = m_new2
                    best_a = a_new2.copy()
                    a = a_new2

        if verbose and it % 100 == 0:
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + ": J=" + repr(round(J, 6)))

    return best_a, best_m


def local_search_multi(a0, time_limit, verbose=True):
    """Aggressive multi-point local search with SA acceptance."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(1234)
    temp = 0.01
    temp_min = 0.0001
    temp_decay = 0.9997
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:20]
        improved = False

        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick a peak to attack
            k_int = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue

            # Pick num_points coordinates to modify
            num_pts = int(rng.choice([1, 2, 3]))
            # Source: top contributors to this peak
            top_src = idx[np.argsort(contrib)[::-1][:5]]
            srcs = [int(top_src[rng.integers(min(5, len(top_src)))]) for _ in range(num_pts)]
            # Remove duplicates
            srcs = list(set(srcs))[:num_pts]

            # Destinations: positions with low contribution to all peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            dsts = [int(candidates[rng.integers(len(candidates))]) for _ in range(num_pts)]
            dsts = list(set(dsts))[:num_pts]

            # Try the move
            frac = float(rng.choice([0.005, 0.01, 0.02, 0.05, 0.1, 0.2]))
            amt_total = min(a[s] for s in srcs) * frac if srcs else 0
            if amt_total < 1e-12:
                continue
            amt_each = amt_total / len(srcs)

            a_trial = a.copy()
            for si, di in zip(srcs, dsts):
                move_amt = min(amt_each, a_trial[si])
                a_trial[si] -= move_amt
                a_trial[di] += move_amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()

            # SA acceptance
            if M_trial < best_m:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:20]
            elif M_trial < a.max() * 0 and rng.random() < np.exp(-(M_trial - best_m) / max(temp, 1e-10)):
                a = a_trial
                improved = True
                stall = 0

        if not improved:
            stall += 1
            temp = max(temp * temp_decay, temp_min)
            # Random perturbation
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.05))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LS-multi (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall) + " temp=" + repr(round(temp, 6)))

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with optimal 1D line search on peak-contributing coords."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for iteration in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:10]

        # For each peak, find the most contributing coordinate and try to optimize it
        improved = False
        for k_int in top_peaks[:5]:
            k_i = int(k_int)
            js = max(0, k_i - n + 1)
            je = min(n - 1, k_i)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_i - idx]

            # Try reducing the top contributor
            src = int(idx[np.argmax(contrib)])
            if a[src] < 1e-15:
                continue

            # Find best amount to move: try several fractions
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = a[src] * frac
                # Find best destination
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_peaks:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True
                candidates = np.where(~in_peak)[0]
                if len(candidates) == 0:
                    candidates = np.arange(n)

                # Sample a few destinations
                for _ in range(3):
                    dst = int(candidates[rng.integers(len(candidates))])
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try random pair
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                frac = float(rng.uniform(0.005, 0.1))
                amt = min(a[i], a[j]) * frac
                if amt < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordDesc it=" + str(iteration) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def lp_fw(a0, time_limit, solver, initial_K=300, max_K=2000, add_per_iter=20, verbose=True):
    """LP-Frank-Wolfe method."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 600

    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5), max(0.02, default_alpha * 1.5)]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def cyclic_shift_best(a0):
    """Try all cyclic shifts and return the best one."""
    n = len(a0)
    best_m = fast_autocorr(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(best_score))

    n0 = len(best_seq)

    # Phase 1: Gradient descent with LSE smoothing
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: GD-LSE (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.20, 350.0)
    if t1 > 60:
        a_out, _ = gradient_descent_smoothed(norm_sum1(best_seq), t1,
                                             method='lse', beta_start=15.0,
                                             beta_end=800.0, beta_growth=1.003,
                                             top_k=15, restart_every=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "gd_lse1", t_global)

    # Phase 2: Gradient descent with power smoothing
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: GD-Power (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 40:
        a_out, _ = gradient_descent_smoothed(norm_sum1(best_seq), t2,
                                             method='power', beta_start=8.0,
                                             beta_end=50.0, beta_growth=1.005,
                                             top_k=None, restart_every=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "gd_power", t_global)

    # Phase 3: Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Cyclic shift ===")
    if remaining > 30:
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic1", t_global)

    # Phase 4: Local search multi-point
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Local search multi (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_out, _ = local_search_multi(norm_sum1(best_seq), t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ls_multi1", t_global)

    # Phase 5: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Coordinate descent (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord1", t_global)

    # Phase 6: Spectral flattening
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Spectral flatten (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 20:
        a_out, _ = spectral_flatten(norm_sum1(best_seq), t6, n_iters=300, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral1", t_global)

    # Phase 7: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: LP-FW (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.15, 250.0)

        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 8: Multi-scale - try larger n
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    target_ns = [n_best + 500, n_best + 1500]
    target_ns = [l for l in target_ns if l <= 12000 and l > n_best]

        if remaining < 100:
            break
        print("\n=== Phase 8: Scale n=" + str(new_n) + " ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t8a = min(remaining * 0.30, 150.0)
        if t8a > 40:
            a_out, _ = gradient_descent_smoothed(a_long, t8a,
                                                 method='lse', beta_start=20.0,
                                                 beta_end=600.0, beta_growth=1.004,
                                                 top_k=12, restart_every=2000, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "gd_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_n" + str(new_n), t_global)

    # Phase 9: Second GD pass with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: GD-LSE v2 (n=" + str(n_best) + ") ===")
    t9 = min(remaining * 0.12, 200.0)
    if t9 > 30:
        a_out, _ = gradient_descent_smoothed(norm_sum1(best_seq), t9,
                                             method='lse', beta_start=50.0,
                                             beta_end=1500.0, beta_growth=1.005,
                                             top_k=10, restart_every=2000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "gd_lse2", t_global)

    # Phase 10: Final local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 10: Final LS (n=" + str(n_best) + ") ===")
    t10 = min(remaining * 0.10, 120.0)
    if t10 > 20:
        a_out, _ = local_search_multi(norm_sum1(best_seq), t10, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ls_final", t_global)

    # Final: Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print("\n=== Final: Cyclic shift ===")
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)

    # Final: Quick coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print("\n=== Final: Coord descent ===")
        a_out, _ = coordinate_descent(norm_sum1(best_seq), remaining * 0.5, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
