```python
def optimize_freq_domain(p_init, N, n, max_iters=10000, beta=50, lr_initial=0.01, lr_decay=0.999):
    p = p_init.copy()
    p = np.maximum(p, 0)
    s = p.sum()
    if s > 0:
        p = p / s
    
    lr = lr_initial
    best_M = float('inf')
    best_p = p.copy()
    
    for it in range(max_iters):
        # Compute circular autocorrelation
        p_ext = np.zeros(N)
        p_ext[:n] = p
        P = np.fft.rfft(p_ext, N)
        r = np.fft.irfft(P * P, N)
        r = r[:2*n-1]  # valid linear convolution part
        M = r.max()
        
        if M < best_M:
            best_M = M
            best_p = p.copy()
        
        # Compute pressure (softmax over lags)
        r_shifted = r - r.max()
        r_shifted = np.clip(r_shifted, -50, 50)
        w = np.exp(beta * r_shifted)
        w = w / w.sum()
        
        # Compute gradient
        w_ext = np.zeros(N)
        w_ext[:2*n-1] = w
        W = np.fft.rfft(w_ext, N)
        cc = np.fft.irfft(np.conj(W) * P, N)
        grad_full = cc + cc[np.mod(-np.arange(N), N)]
        grad = grad_full[:n]
        
        # Projected gradient step
        p_new = p - lr * grad
        # Project onto simplex: clip to 0 and normalize
        p_new = np.maximum(p_new, 0)
        s = p_new.sum()
        if s < 1e-15:
            p_new = np.ones(n) / n
        else:
            p_new = p_new / s
        
        p = p_new
        lr *= lr_decay
        
        if it % 100 == 0:
            print(f"Iter {it}: M={M:.6f}, best_M={best_M:.6f}, lr={lr:.6f}")
    
    return best_p, best_M
```

This is O(N log N) per iteration, which for N ~ 16384 is very fast. We could do thousands of iterations in a few seconds.

The key hyperparameters are:
- beta: controls how "peaky" the pressure is. Higher beta focuses more on the actual peak.
- lr_initial and lr_decay: learning rate schedule
- max_iters: number of iterations

Let me also think about the relationship between the frequency domain approach and Frank-Wolfe:

In Frank-Wolfe, we solve: min_{g in simplex} max_k (f * g)_k
This is a zero-sum game where the "row player" chooses g and the "column player" chooses k.

In the frequency domain approach, we're doing:
p <- project(p - lr * grad of smooth_max(r))

These are related but different. The FW approach is a "majorize-minimize" scheme where we solve the linear subproblem exactly (or approximately), while the frequency domain approach is a gradient-based method on the smoothed objective.

The advantage of the frequency domain approach is speed (FFT-based), and the advantage of FW is that it doesn't require a learning rate and is more robust.

I think the best strategy is to combine both:
1. Use the frequency domain method for fast local optimization
2. Use FW for more robust convergence
3. Alternate between them
4. Add random perturbations to escape local minima

def next_pow2(x):
    x = int(max(1, x))
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return c


def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return c

def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)

    lr_schedule: list of (start_iter, end_iter, lr) tuples
    """
    p = norm_sum1(p_init)
    best_p = p.copy()
    best_M = float('inf')
    
    p_ext = np.zeros(N, dtype=np.float64)
    p_ext[:n] = p
    P = np.fft.rfft(p_ext)
    
    no_improve = 0
    t0 = time.monotonic()
    
    for it in range(max_iters):
        # Get current beta and lr
        beta = _get_param_from_schedule(beta_schedule, it)
        lr = _get_param_from_schedule(lr_schedule, it)
        
        # Compute autocorrelation
        r = np.fft.irfft(P * P, N)
        r_valid = r[:2*n-1]
        M = float(np.max(r_valid))
        
        if M < best_M - 1e-15:
            best_M = M
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Compute pressure (softmax)
        r_max = float(np.max(r_valid))
        r_shifted = r_valid - r_max
        r_shifted = np.clip(r_shifted, -100.0, 100.0)
        w = np.exp(beta * r_shifted)
        w_sum = float(np.sum(w))
        if w_sum < 1e-300:
            w = np.zeros(2*n-1)
            w[int(np.argmax(r_valid))] = 1.0

        cc = np.fft.irfft(np.conj(W) * P, N)

        # Project onto simplex (clip + normalize)
        p_new = np.maximum(p_new, 0.0)
        s = float(p_new.sum())
        if s < 1e-15:
            p_new = np.ones(n) / float(n)
        else:
            p_new = p_new / s
        
        # Update

        p_ext[:n] = p
        P = np.fft.rfft(p_ext)

            elapsed = time.monotonic() - t0
            print("    FDO iter=" + str(it) + " M=" + repr(M) +
                  " best_M=" + repr(best_M) + " beta=" + repr(beta) +
                  " lr=" + repr(lr) + " t=" + repr(elapsed))
    
    return best_p, best_M


def _get_param_from_schedule(schedule, it):
    """Get parameter value at iteration it from a schedule of (start, end, value)."""
    for (start, end, val) in schedule:
        if start <= it < end:
            return val
    # Default to last value
    if schedule:
        return schedule[-1][2]
    return 1.0
