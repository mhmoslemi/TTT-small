def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]) + 1
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient_fft(a, active_indices, n):
    """Compute subgradient of max at active_indices using FFT."""
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    # Build indicator of active set
    s_vec = np.zeros(fft_size)
    s_vec[active_indices] = 1.0
    S = np.fft.rfft(s_vec)
    A = np.fft.rfft(np.pad(a, (0, fft_size - n)))
    # Cross-correlation: g[i] = sum_k s[k] * a[k-i]
    # This is conv(s, flip(a)) evaluated at appropriate offset

    conv = np.fft.irfft(S * A_flip, fft_size)
    # Extract relevant portion
    offset = n - 1
    g = conv[offset:offset + n]
    return g

            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Subgrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active)) + " eta=" + str(round(eta, 6)))

    return best_a, best_m


def spectral_refine(a0, time_limit, verbose=True, seed=777):
    """Spectral domain: flatten the convolution spectrum iteratively."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.03
    for it in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Target: reduce peaks in the convolution = flatten A^2
        # We want IFFT(A^2) to be flat, so A^2 should be concentrated at DC
        # Reduce energy at non-DC frequencies
        for k in range(1, len(A)):
            if power[k] > 1e-30:
                A[k] *= (1.0 - alpha * 0.5)

        # Add small random phase perturbation for exploration
        if it % 100 == 0:
            phase_pert = rng.standard_normal(len(A)) * 0.01
            A[1:] *= np.exp(1j * phase_pert[1:])

        a_new = np.fft.irfft(A, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-18:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 500 == 0:
            # Periodic restart with current best
            a = best_a + 0.02 * rng.standard_normal(n) * best_a
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a

        if it % 2000 == 0:
            alpha = max(0.003, alpha * 0.9)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " a=" + str(round(alpha, 5)))

    return best_a, best_m


def nesterov_optimize(a0, time_limit, beta_start=30.0, beta_end=800.0,
                      beta_growth=1.004, eta_init=0.5, top_k_ratio=0.02,
                      restart_every=1000, verbose=True, seed=42):
    """Nesterov accelerated with smoothed max and backtracking."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(20, int(n * top_k_ratio))
    eta = eta_init

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Smoothed grad computation
        b = fast_autocorr(y)
        size = 2 * n - 1
        if top_k < size:
            idx = np.argsort(b)[::-1][:top_k]
            b_use = np.zeros(size)
            b_use[idx] = b[idx]
            b_max = b[idx].max()
        else:
            b_use = b
            b_max = b.max()

        if b_max <= 0:
            grad = np.zeros(n)
        else:
            shifted = beta * (b_use - b_max)
            shifted = np.clip(shifted, -500, 0)
            exp_vals = np.exp(shifted)
            s_exp = exp_vals.sum()
            if s_exp < 1e-300:
                p = np.ones(size) / size
            else:
                p = exp_vals / s_exp

            y_rev = y[::-1]
            conv_len = n + size - 1
            fft_sz = 1 << (conv_len - 1).bit_length()
            Ar = np.fft.rfft(y_rev, fft_sz)
            Pr = np.fft.rfft(p, fft_sz)
            conv = np.fft.irfft(Ar * Pr, fft_sz)
            grad = conv[n - 1:2 * n - 1] / beta

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            x_new = x
        else:
            found = False
            eta_trial = eta
            y_max = b.max()
            for _ in range(12):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                x_new = x_new / max(x_new.sum(), 1e-15)
                M_new = fast_autocorr(x_new).max()
                if M_new <= y_max + 1e-16:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                eta = eta_trial

        M_new = fast_autocorr(x_new).max()
        if M_new < best_max_b - 1e-16:
            best_max_b = M_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.1, 3.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 40 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = int(rng.integers(6))
            if strategy == 0:
                pert = rng.normal(0, 0.015, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.5)
            elif strategy == 2:
                x_r = best_a * np.exp(0.025 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= 0.75
                x_r[~mask] *= 1.25
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 20)))
                x_r = np.roll(best_a, shift)
            else:
                # Spectral perturbation
                fft_sz = 1 << (2 * n - 2).bit_length()
                Af = np.fft.rfft(best_a, fft_sz)
                Af[1:] *= np.exp(1j * rng.standard_normal(len(Af)-1) * 0.05)
                x_r = np.fft.irfft(Af, fft_sz)[:n]
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 15.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nesterov restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nesterov t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + str(int(beta)))

    return best_a, best_max_b


def peak_directed_repair(a0, time_limit, verbose=True, seed=321):
    """Directly target the peak of conv(a,a) and redistribute mass."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        peak_idx = int(np.argmax(ac))
        peak_val = ac[peak_idx]

        # Find contributing pairs for this peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        if js > je:
            stall += 100
            continue
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            stall += 100
            continue

        # Top contributors
        top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]
        improved = False

        for src in top_contrib_idx:
            src = int(src)
            partner = peak_idx - src
            if not (0 <= partner < n):
                continue

            # Try moving mass from src to random safe locations
            for _ in range(8):
                dst = int(rng.integers(n))
                if dst == src or dst == partner:
                    continue
                # Check if dst would create a new peak
                frac = rng.uniform(0.005, 0.04)
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-17:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Broader random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakRepair (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall) + " peak=" + str(peak_idx))

    return best_a, best_m


def simulated_annealing(a0, time_limit, verbose=True, seed=123):
    """SA with diverse moves for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T = 0.005
    T_min = 0.00002
    T_decay = 0.9997
    cur_m = best_m

    for it in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        if T < T_min:
            break

        move_type = int(rng.integers(4))
        if move_type == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = rng.uniform(0.001, 0.01)
            amt = a[i] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
        elif move_type == 1:
            i, j = int(rng.integers(n)), int(rng.integers(n))
            k, l = int(rng.integers(n)), int(rng.integers(n))
            if len(set([i,j,k,l])) < 4:
                continue
            amt = min(a[i], a[k]) * rng.uniform(0.01, 0.04)
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial[k] -= amt
            a_trial[l] += amt
        elif move_type == 2:
            # Local Gaussian
            center = int(rng.integers(n))
            width = int(rng.integers(3, min(30, n//10)))
            mask = np.zeros(n, dtype=bool)
            lo = max(0, center - width)
            hi = min(n, center + width)
            mask[lo:hi] = True
            a_trial = a + rng.normal(0, 0.008, n) * mask * a
        else:
            # Push-pull between neighboring positions
            i = int(rng.integers(n - 1))
            j = i + 1
            amt = a[i] * rng.uniform(0.005, 0.03)
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt

        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        m_trial = fast_autocorr(a_trial).max()

        if m_trial < cur_m:
            a = a_trial
            cur_m = m_trial
            if m_trial < best_m:
                best_m = m_trial
                best_a = a_trial.copy()
        elif rng.random() < np.exp(-(m_trial - cur_m) / max(T, 1e-10)):
            a = a_trial
            cur_m = m_trial

        T *= T_decay
        if T < T_min * 2 and it % 3000 == 0:
            T = 0.003

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SA it=" + str(it) + " T=" + str(round(T, 7)) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(time_budget=1800.0):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines active-set subgradient, spectral refinement, Nesterov,
    peak-directed repair, and simulated annealing with multi-length search.
    """
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 60.0, 1740.0)
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + repr(round(best_score, 6)))

    # Phase 1: Active-set subgradient (replaces LP-FW, much faster)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: ActiveSet Subgrad (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.18, 280.0)
    if t1 > 40:
        a_out, _ = active_set_subgrad(norm_sum1(best_seq), t1, K_init=250, K_max=2500, verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sub1", t_global)

    # Phase 2: Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P2: Nesterov (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 40:
        a_out, _ = nesterov_optimize(norm_sum1(best_seq), t2,
                              beta_start=25.0, beta_end=700.0,
                              beta_growth=1.004, eta_init=0.5,
                              top_k_ratio=0.025, restart_every=1200,
                              verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: Second active-set subgradient (different params)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P3: ActiveSet Subgrad r2 (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining * 0.15, 230.0)
    if t3 > 30:
        a_out, _ = active_set_subgrad(norm_sum1(best_seq), t3, K_init=400, K_max=3000, verbose=True, seed=99)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sub2", t_global)

    # Phase 4: Spectral refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: Spectral (n=" + str(len(best_seq)) + ") ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 15:
        a_out, _ = spectral_refine(norm_sum1(best_seq), t4, verbose=True, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 5: Peak-directed repair
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P5: PeakRepair (n=" + str(len(best_seq)) + ") ===")
    t5 = min(remaining * 0.10, 130.0)
    if t5 > 20:
        a_out, _ = peak_directed_repair(norm_sum1(best_seq), t5, verbose=True, seed=321)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Multi-length (base n=" + str(n_best) + ") ===")

    longer_lengths = [n_best + 500, n_best + 1200, n_best + 2500]
    longer_lengths = [l for l in longer_lengths if l <= 12000]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print("  -> n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.30, 160.0)
        if t6a > 25:
            a_out, _ = active_set_subgrad(a_long, t6a, K_init=300, K_max=2000, verbose=True, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sub_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t6b = min(remaining * 0.40, 180.0)
            if t6b > 25:
                a_out, _ = nesterov_optimize(norm_sum1(best_seq), t6b,
                                   beta_start=30.0, beta_end=600.0,
                                   beta_growth=1.004, eta_init=0.4,
                                   top_k_ratio=0.02, restart_every=1000,
                                   verbose=True, seed=new_n % 5431)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6c = min(remaining * 0.30, 80.0)
            if t6c > 15:
                a_out, _ = spectral_refine(norm_sum1(best_seq), t6c, verbose=False, seed=new_n % 3163)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_n" + str(new_n), t_global)

    # Phase 7: Nesterov with high beta (fine-tuning)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P7: Nesterov r2 (n=" + str(len(best_seq)) + ") ===")
    if remaining > 60:
        t7 = min(remaining * 0.20, 220.0)
        if t7 > 30:
            a_out, _ = nesterov_optimize(norm_sum1(best_seq), t7,
                               beta_start=60.0, beta_end=1200.0,
                               beta_growth=1.005, eta_init=0.3,
                               top_k_ratio=0.015, restart_every=800,
                               verbose=True, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 8: Another subgradient round
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P8: ActiveSet r3 (n=" + str(len(best_seq)) + ") ===")
    if remaining > 40:
        t8 = min(remaining * 0.20, 180.0)
        if t8 > 25:
            a_out, _ = active_set_subgrad(norm_sum1(best_seq), t8, K_init=500, K_max=3000, verbose=True, seed=2024)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sub3", t_global)

    # Phase 9: SA + peak repair for final polish
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P9: SA + PeakRepair (n=" + str(len(best_seq)) + ") ===")
    if remaining > 40:
        t9a = min(remaining * 0.35, 100.0)
        if t9a > 20:
            a_out, _ = simulated_annealing(norm_sum1(best_seq), t9a, verbose=True, seed=456)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "sa1", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t9b = min(remaining * 0.35, 80.0)
            if t9b > 15:
                a_out, _ = peak_directed_repair(norm_sum1(best_seq), t9b, verbose=True, seed=654)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "peak2", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t9c = min(remaining * 0.30, 60.0)
            if t9c > 10:
                a_out, _ = active_set_subgrad(norm_sum1(best_seq), t9c, K_init=600, K_max=3000, verbose=False, seed=7777)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "sub_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
