            break
        
        # Target: reduce power at frequencies that contribute to the peak
        # The DC component (k=0) is fixed by the sum constraint
        # We want to flatten power[1:]
        
        # Compute the contribution of each frequency to the convolution peak
        # The convolution peak is related to the integral of |A(f)|^2 * cos(2πf*k_peak)
        # For simplicity, just flatten the power spectrum
        
        target = np.median(power[1:]) if len(power) > 1 else power[0]
        
        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                reduction = 1.0 - alpha * min(ratio - 1.0, 3.0)
                A_new[k] *= np.sqrt(max(0.1, reduction))
            elif ratio < 0.3:
                boost = 1.0 + alpha * 0.2 * (0.3 - ratio)
                A_new[k] *= np.sqrt(min(2.0, boost))
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Try with smaller alpha
            if stall > 100:
                alpha *= 0.9
                if alpha < 0.001:
                    break
        
        # Periodic random phase perturbation
        if it % 500 == 0:
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.01
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SpectShave it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
    
    return best_a, best_m

def direct_peak_reduction(a0, time_limit, verbose=True, seed=555):
    """
    Directly target the peak of the convolution by redistributing mass
    from the pairs that contribute most to the peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 5000:
        b = fast_autocorr(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        if peak_val < best_m * (1 - 0.001):
            best_m = peak_val
            best_a = a.copy()
        
        # Find the top contributing pairs for the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            stall += 100
            continue
        
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()
        
        if total_contrib < 1e-15:
            stall += 100
            continue
        
        # Try to reduce the peak by moving mass from the top contributing pair
        top_idx = int(idx[np.argmax(contrib)])
        top_partner = peak_k - top_idx
        
        improved = False
        for frac in [0.005, 0.01, 0.02, 0.05]:
            amt = a[top_idx] * frac
            if amt < 1e-15:
                continue
            
            # Try multiple destinations
            for _ in range(20):
                dst = int(rng.integers(n))
                if dst == top_idx or dst == top_partner:
                    continue
                
                a_trial = a.copy()
                # Move mass from top_idx to dst
                a_trial[top_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random exploration
            if stall > 200:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
        
        if verbose and stall % 1000 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakReduce (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))
    
    return best_a, best_m

def coordinate_descent_refine(a0, time_limit, verbose=True, seed=777):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed. Uses the fact that the objective is a
    ratio of the max of quadratics to a constant.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Work in log-space to maintain positivity
    # Actually, let's work in the constraint sum(a) = 1 directly
    
    b = fast_autocorr(a)
    peak_vals = np.sort(b)[::-1][:min(50, len(b))]
    
    for epoch in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Random order
        order = rng.permutation(n)
        
        any_improve = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Fix all coordinates except j
            # c[k] = a[j]*a[k-j] + rest[k] where rest[k] doesn't depend on a[j]
            # We want to minimize max_k c[k]
            # c[k] is linear in a[j] (for fixed other coordinates)
            # So min max_k c[k] is a 1D LP: minimize t subject to c[k] <= t, a[j] >= 0, sum(a) = 1
            
            # c[k] = a[j]*a[k-j] + r[k] where r[k] = c[k] -
