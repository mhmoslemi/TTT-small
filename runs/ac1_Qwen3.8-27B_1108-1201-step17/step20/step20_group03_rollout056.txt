import numpy as np
import time
import cvxpy as cp

def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_local(a):
    """Local evaluation (same as evaluate_sequence but for numpy arrays)."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Key improvements: larger active sets, exact subgradient polishing, peak-targeted local search
"""

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None

def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    return _MODULE_SOLVER

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_local(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000.0)
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_constraint_matrix(y, active_set, n):
    K = len(active_set)
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(active_set):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            coeffs[jj, idx] = y[k - idx]
    return coeffs

def lp_frank_wolfe(a0, time_limit, solver, initial_K=800, max_K=5000, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_conv(y)
    top_idx = np.argsort(ac_y)[::-1][:min(initial_K, 2 * n - 1)]
    active_list = list(top_idx)
    no_improve = 0
    max_no_improve = 300
    t_iter = 0

    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        ac = fast_conv(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            # Rebuild active set aggressively
            top_idx = np.argsort(fast_conv(y))[::-1][:min(max_K, 2 * n - 1)]
            active_list = list(top_idx)
            no_improve = 0
            if t_iter > 50:
                break

        # Trim active set if too large
        if len(active_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_list])
            keep = np.argsort(margins)[:max_K]
            active_list = [active_list[i] for i in keep]

        K = len(active_list)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, active_list, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_list = list(np.argsort(fast_conv(y))[::-1][:min(initial_K // 2, 2 * n - 1)])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations with full convolution
        conv_ys = np.fft.irfft(np.fft.rfft(y, 1 << (2 * n - 1).bit_length()) *
                               np.fft.rfft(g_dir, 1 << (2 * n - 1).bit_length()),
                               1 << (2 * n - 1).bit_length())[:2 * n - 1]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            # Add top violations to active set
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:50]]
            for k in top_v:
                if k not in active_list:
                    active_list.append(int(k))

        # Backtracking line search
        alphas = [2.0 / (t_iter + 2.0)]
        base = 2.0 / (t_iter + 2.0)
        alphas.extend([max(0.01, base * 0.3), max(0.01, base * 0.5),
                        max(0.01, base * 0.7), max(0.01, base * 1.5),
                        max(0.01, base * 3.0), 0.05, 0.1])
        alphas = list(set([min(1.0, max(0.005, a)) for a in alphas]))

        best_alpha = base
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t_iter % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter=%d (%.1fs): J=%.6f |S|=%d alpha=%.4f" %
                  (t_iter, elapsed, J, len(active_list), best_alpha))

    return best_a, best_m

def epsilon_subgradient(a0, time_limit, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    t_nest = 1.0
    base_eta = 1.0 / np.sqrt(n)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t_nest_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_nest_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        c = fast_conv(y)
        c_max = c.max()

        # Find epsilon-active set
        eps = max(0.01 * c_max, (c_max - np.sort(c)[::-1][min(50, len(c)-1)]) + 1e-15)
        active = np.where(c >= c_max - eps)[0]

        # Compute average subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                grad[idx] += y[k - idx]
        grad /= max(1, len(active))

        # Adaptive step size based on gap
        sorted_c = np.sort(c)[::-1]
        if len(sorted_c) > 1:
            gap = c_max - sorted_c[min(10, len(sorted_c)-1)]
            eta = min(base_eta / (1 + 0.01 * stall), max(1e-4, 0.5 * gap / (np.linalg.norm(grad) + 1e-15)))
        else:
            eta = base_eta

        z = y - eta * grad
        x_new = simplex_project(z)

        c_new = fast_conv(x_new)
        max_new = c_new.max()

        if max_new < best_m - 1e-14:
            best_m = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_nest_new

        if stall > 2000:
            break

        if verbose and (t_nest_new // 100) % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    EpsSubg t=%.0f (%.1fs): J=%.6f stall=%d eps=%.4f" %
                  (t_nest_new, elapsed, J, stall, eps))

    return best_a, best_m

def local_search(a0, time_limit, verbose=True):
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 100:
        c = fast_conv(a)
        top_peaks = np.argsort(c)[::-1][:20]
        improved = False

        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_peaks[rng.integers(len(top_peaks))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])

            # Find good destination: low value, not contributing to any top peak
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            # Prefer low-value candidates
            cand_vals = a[candidates]
            if len(candidates) > 100:
                thresh = np.percentile(cand_vals, 25)
                low_cands = candidates[cand_vals <= thresh]
                if len(low_cands) > 0:
                    candidates = low_cands
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src or a[src] < 1e-12:
                continue

            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.01)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LocalSearch (%.1fs): J=%.6f stall=%d" % (elapsed, J, stall))

    return best_a, best_m

def cyclic_shift_best(a0):
    n = len(a0)
    best_m = fast_conv(a0).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    A0 = np.fft.rfft(a0, fft_size)
    for s in range(1, n):
        a_shift = np.roll(a0, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m - 1e-15:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a0, -best_shift), best_m
    return a0.copy(), best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_local(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: %.1fs" % time_budget)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_local(parent)
            if np.isfinite(p_score):
                best_seq = normalize(parent)
                best_score = p_score
                print("Parent: n=%d, score=%.6f" % (len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 8000
        x = np.linspace(0, 1, n_base)
        # Use a skewed shape as initial guess (asymmetric helps)
        best_seq = normalize(np.sin(np.pi * x) ** 2 * (1 + 0.5 * x))
        best_score = evaluate_local(best_seq)
        print("Default init: n=%d, score=%.6f" % (len(best_seq), best_score))

    n0 = len(best_seq)

    # Phase 1: LP-Frank-Wolfe (main optimizer) - 45% of time
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: LP-FW (n=%d) ===" % n0)
    t1 = min(remaining * 0.45, 700.0)
    if t1 > 60:
        a_out, _ = lp_frank_wolfe(best_seq, t1, solver,
                                   initial_K=800, max_K=5000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 2: Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        a_shifted, _ = cyclic_shift_best(best_seq)
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyc1", t_global)

    # Phase 3: Epsilon-subgradient polish - 20% of time
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Eps-Subgradient (n=%d) ===" % len(best_seq))
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        a_out, _ = epsilon_subgradient(best_seq, t3, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "subg1", t_global)

    # Phase 4: Local search - 15% of time
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 4: Local Search (n=%d) ===" % len(best_seq))
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_out, _ = local_search(best_seq, t4, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "loc1", t_global)

    # Phase 5: Second LP-FW pass with tighter parameters
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: LP-FW v2 (n=%d) ===" % n_best)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 40 and n_best <= 12000:
        a_out, _ = lp_frank_wolfe(best_seq, t5, solver,
                                   initial_K=1200, max_K=6000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp2", t_global)

    # Phase 6: Multi-n exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    n_options = []
    for delta in [500, 1000, 2000, -500, -1000]:
        nn = n_best + delta
        if 2000 <= nn <= 15000:
            n_options.append(nn)
    n_options = list(set(n_options))

    for new_n in n_options[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("\n=== Phase 6: Multi-n (n=%d) ===" % new_n)
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick LP-FW
        t6a = min(remaining * 0.35, 150.0)
        if t6a > 30 and new_n <= 12000:
            a_out, _ = lp_frank_wolfe(a_long, t6a, solver,
                                       initial_K=600, max_K=4000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lpn%d" % new_n, t_global)

        # Quick subgradient
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.30, 100.0)
        if t6b > 20:
            a_out, _ = epsilon_subgradient(best_seq, t6b, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "subgn%d" % new_n, t_global)

        # Cyclic shift
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_shifted, _ = cyclic_shift_best(best_seq)
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cycn%d" % new_n, t_global)

    # Phase 7: Final local search
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 7: Final Local Search (n=%d) ===" % len(best_seq))
    t7 = min(remaining * 0.50, 150.0)
    if t7 > 20:
        a_out, _ = local_search(best_seq, t7, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "loc_final", t_global)

    # Phase 8: Final subgradient micro-polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print("\n=== Phase 8: Final Subgradient Polish ===")
        t8 = min(remaining * 0.60, 120.0)
        if t8 > 15:
            a_out, _ = epsilon_subgradient(best_seq, t8, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "subg_final", t_global)

    # Final cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        a_shifted, _ = cyclic_shift_best(best_seq)
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyc_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=%.6f, n=%d, time=%.1fs" % (best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence! Using fallback.")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = evaluate_local(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
