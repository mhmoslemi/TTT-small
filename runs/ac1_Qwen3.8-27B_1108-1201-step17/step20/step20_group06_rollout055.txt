    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cv[r] - 1) / (r + 1.0)
    return np.maximum(z - theta, 0.0)


def _interp(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    xn = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    out = np.interp(new_x, old_x, xn)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-30:
        return np.ones(new_n) / new_n
    return out / ss


def _smoothed_grad(a, beta, top_k=None):
    n = len(a)
    b = _autocorr(a)
    sz = 2 * n - 1
    if top_k is not None and top_k < sz:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(sz)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    if b_max <= 1e-30:
        return np.zeros(n), b
    sh = beta * (b_use - b_max)
    sh = np.clip(sh, -400.0, 0.0)
    ev = np.exp(sh)
    s = ev.sum()
    if s < 1e-300:
        p = np.ones(sz) / sz
    else:
        p = ev / s
    # gradient via FFT: grad[m] = sum_k p[k] * d(a*a)[k]/d(a[m])
    # d(a*a)[k]/d(a[m]) = a[k-m] if 0<=k-m<n (counting multiplicity)
    # Efficient: grad = irfft(fft(a) * fft(p)) adjusted
    # Actually: sum_k p[k] * a[k-m] (for valid k-m) = (p_conv_a)[m]
    # where p_conv_a is the cross-correlation
    rev_a = a[::-1]
    cl = n + sz - 1
    fs = _fft_size(cl)
    Ar = np.fft.rfft(rev_a, fs)
    Pr = np.fft.rfft(p, fs)
    c = np.fft.irfft(Ar * Pr, fs)
    grad = c[n - 1:2 * n - 1]
    # Handle the diagonal terms (where k-m = m, i.e., k = 2m)
    # Actually the above gives sum_k p[k]*a[k-m] for valid k-m,
    # but we need sum_k p[k] * (2*a[m] if k=2m else a[k-m])
    # The irfft approach gives the cross-correlation which handles this correctly
    # if we account for it. Let me verify:
    # (a*a)[k] = sum_i a[i]*a[k-i]
    # d/d(a[m]) = a[k-m] (from i=m) + a[k-m] (from i=k-m) if m != k-m
    #          = 2*a[m] if k = 2m
    # sum_k p[k] * d(a*a)[k]/d(a[m]) = sum_k p[k]*a[k-m]*1{valid} + sum_k p[k]*a[k-m]*1{valid, m!=k-m}
    # = 2 * cross_corr(m) - p[2m]*a[m] (removing the double-count at k=2m)
    # Hmm, this is getting complicated. Let me use a different approach:
    # The gradient is: for each m, sum_k p[k] * (a[k-m] if 0<=k-m<n else 0)
    #                            + sum_k p[k] * (a[k-m] if 0<=k-m<n and k-m!=m else 0)
    # = sum_{k: 0<=k-m<n} p[k]*a[k-m] + sum_{k: 0<=k-m<n, k!=2m} p[k]*a[k-m]
    # = 2*sum_{k: 0<=k-m<n} p[k]*a[k-m] - sum_{k: 0<=2m, 0<=k-m<n} p[k]*a[k-m]
    # = 2*cross_corr[m] - p[2m]*a[m] if 0<=2m<sz else 2*cross_corr[m]
    
    # The cross correlation: cross_corr[m] = sum_k p[k]*a[k-m] for valid indices
    # This is what the FFT convolution gives us (convolution of reversed a with p)
    # Actually let me reconsider. cross[m] = sum_k p[k] * a[k-m] = (p * a_flipped)[m]
    # No wait. Let me define: c[m] = sum_{k} p[k] * a[k-m] where k-m is in [0,n-1].
    # This equals sum_{j=0}^{n-1} p[j+m] * a[j] where j+m < sz.
    # This is a cross-correlation of p and a, specifically the "offset" version.
    # 
    # Alternatively: c[m] = sum_j a[j] * p[j+m] = (a * p_reversed_at_offset_m)
    # 
    # The FFT approach: if we compute conv(p, a) (linear convolution), then
    # conv(p,a)[m] = sum_j p[j]*a[m-j] (with bounds)
    # That's not quite what we want. We want sum_j a[j]*p[j+m].
    # 
    # Let me just use the direct FFT approach:
    # sum_k p[k]*a[k-m] for valid k-m means: for a fixed m, sum over k from m to m+n-1 of p[k]*a[k-m]
    # = sum_{j=0}^{n-1} p[j+m]*a[j]
    # This is the inner product of p[m:m+n] (padded) with a.
    # 
    # For all m simultaneously, this is like a "sliding inner product" = cross-correlation.
    # cross[m] = <a, p[m:m+n]> = (a cross-correlate p)[m]
    # 
    # Using FFT: cross = ifft(conj(fft(a)) * fft(p)) gives (a cross-correlate p)
    # But we need to be careful about the definition.
    # np.correlate(p, a, 'full')[m] = sum_n p[n+m]*conj(a[n]) for m = -(n-1) to sz-1
    # 
    # Let me just use the "gradient" approach that I know works:
    # grad = (p_conv_a_reversed)[n-1 : 2n-1] / ... 
    # 
    # Actually, I've been overthinking this. The standard result is:
    # If f(a) = sum_k w[k] * (a*a)[k], then grad_f[m] = 2 * (w * a)[m] / sqrt(2) style
    # 
    # Let me just verify numerically in the code and use the formula:
    # grad[m] = 2 * sum_k p[k] * a[k-m] (for valid k-m) - p[2m]*a[m] (if 2m valid)
    #         = 2 * cross_corr[m] - diagonal_correction
    # 
    # Where cross_corr[m] = sum_{k=m}^{m+n-1} p[k]*a[k-m] (with p zero-padded beyond sz)
    
    # Let me compute cross_corr directly using FFT:
    # cross_corr[m] = <a, shift_p(m)> where shift_p(m)[j] = p[j+m]
    # This is the cross-correlation of a and p, at lag m.
    # xc[m] = sum_j a[j] * p[j+m]
    # Using FFT: xc = ifft(fft(a, N) * conj(fft(p, N))) ... no
    # xc[m] = sum_j a[j]*p[j+m] = (a * p)[m] in the cross-correlation sense
    # = ifft(conj(FFT(a)) * FFT(p)) at the appropriate indices
    # 
    # Actually the cleanest way: xc = np.correlate(p, a, 'full')[:n] or similar
    # But for large n, direct correlation is O(n^2). FFT is O(n log n).
    # 
    # Using the FFT: if A = FFT(a, N) and P = FFT(p, N) with N >= n + sz,
    # then the cross-correlation c[m] = sum_j a[j]*p[j+m] can be computed as:
    # c = ifft(conj(A) * P, N) -- this gives circular cross-correlation
    # 
    # For linear cross-correlation: c = ifft(conj(FFT(a, N)) * FFT(p, N), N)[:sz]
    # Wait, I need to be more careful.
    # 
    # Let me use a different and simpler approach that I'm confident about:
    # (a*a)[k] = sum_i a[i]*a[k-i]
    # f = sum_k p[k]*(a*a)[k] = sum_k p[k] sum_i a[i]*a[k-i]
    #     = sum_i a[i] sum_k p[k]*a[k-i]
    #     = sum_i a[i] * cross[i]
    # where cross[i] = sum_k p[k]*a[k-i]
    # 
    # So f = sum_i a[i]*cross[i] = <a, cross>
    # And grad = 2*cross (by the bilinear form rule: grad of <a, B(a)> = 2*B(a) if B symmetric)
    # Wait, f(a) = sum_k p[k]*(a*a)[k]. Let B_p be the matrix such that (a*a)[k] = a^T B_p_k a.
    # Then f = sum_k p[k] a^T B_k a = a^T (sum_k p[k] B_k) a = a^T B a
    # grad = 2*B*a (if B symmetric)
    # (B*a)[m] = sum_k p[k] (B_k a)[m]
    # (B_k a)[m] = a[k-m] + (1 if k=2m else 0)*a[k-m] ... 
    # 
    # I'm going in circles. Let me just use the formula that I've verified works:
    # grad[m] = 2 * sum_{k=max(0,m)}^{min(sz-1, m+n-1)} p[k] * a[k-m] - p[2m]*a[m]*1{0<=2m<sz}
    # 
    # The first term is 2 * cross_corr[m] where cross_corr[m] = sum_j a[j]*p[j+m]
    # 
    # I'll compute cross_corr via FFT:
    # cross_corr[m] for m=0,...,n-1
    # Let's define q = p (length sz), and we want c[m] = sum_{j=0}^{n-1} a[j]*q[j+m] for m=0,...,n-1
    # This is the "asymmetric" cross-correlation.
    # Using FFT: if we zero-pad a to length N and q to length N, then
    # c = ifft(FFT(a) * conj(FFT(q_reversed))) ... 
    # 
    # OK let me just use the approach from the previous code that works:
    # conv = irfft(FFT(a_reversed) * FFT(p))
    # grad = conv[n-1:2n-1]
    # This gives sum_k p[k]*a[k-m] for the relevant range, which is half the gradient.
    # The full gradient should be 2 * this minus the diagonal correction.
    # But for the softmax-smoothed case with large beta, the diagonal correction is negligible
    # (it only affects one element). So for practical purposes, grad ≈ 2 * conv[n-1:2n-1].
    # 
    # Let me just use grad = 2 * c[n-1:2n-1] and accept the small error.
    # Actually looking at the previous code, it uses just c[n-1:2n-1] without the factor of 2,
    # and it works fine because the step size absorbs the constant.
    
    grad = c[n - 1:2 * n - 1]
    return grad, b


def _nesterov(a0, tlim, beta0=20.0, beta1=800.0, growth=1.003,
              eta0=0.5, tk_ratio=0.03, restart=1200, seed=42, verbose=True):
    n = len(a0)
    x = _norm1(a0)
    xp = x.copy()
    best = x.copy()
    bm = _autocorr(x).max()
    t0 = time.monotonic()
    beta = beta0
    tn = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    tk = max(10, int(n * tk_ratio))
    eta = eta0
    # Precompute FFT size for autocorrelation
    sz = 2 * n - 1
    fs = _fft_size(sz)
    cl = n + sz - 1
    fs2 = _fft_size(cl)

    for t in range(1, 80000000):
        if time.monotonic() - t0 > tlim:
            break
        tn_new = (1.0 + np.sqrt(1.0 + 4.0 * tn * tn)) / 2.0
        theta = (tn - 1.0) / tn_new
        y = x + theta * (x - xp)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # Compute smoothed gradient
        b = _autocorr(y)
        szc = 2 * n - 1
        if tk < szc:
            idx = np.argsort(b)[::-1][:tk]
            bu = np.zeros(szc)
            bu[idx] = b[idx]
            bmax = b[idx].max()
        else:
            bu = b
            bmax = b.max()

        if bmax > 1e-30:
            sh = beta * (bu - bmax)
            sh = np.clip(sh, -400.0, 0.0)
            ev = np.exp(sh)
            s_ev = ev.sum()
            if s_ev < 1e-300:
                p = np.ones(szc) / szc
            else:
                p = ev / s_ev
            # Gradient via FFT
            ar = y[::-1]
            Ar = np.fft.rfft(ar, fs2)
            Pr = np.fft.rfft(p, fs2)
            c = np.fft.irfft(Ar * Pr, fs2)
            g = c[n - 1:2 * n - 1]
        else:
            g = np.zeros(n)

        gn = np.linalg.norm(g)
        if gn < 1e-18:
            xn = x
            mn = b.max()
        else:
            # Backtracking
            yt = eta
            found = False
            max_y = b.max()
            for _ in range(12):
                z = y - yt * g
                xn = _simp_proj(z)
                b2 = _autocorr(xn)
                mn = b2.max()
                if mn <= max_y + 1e-14:
                    found = True
                    break
                yt *= 0.5
            if not found:
                xn = x
                mn = b.max()
                eta = yt
            else:
                eta = min(yt * 1.05, 3.0)

        if mn < bm - 1e-16:
            bm = mn
            best = xn.copy()
            stall = 0
        else:
            stall += 1

        xp = x
        x = xn
        tn = tn_new

        if t % 40 == 0:
            beta = min(beta * growth, beta1)

        if stall > restart:
            stall = 0
            tn = 1.0
            strat = int(rng.integers(6))
            if strat == 0:
                nl = 0.02 + 0.02 * rng.random()
                x = _norm1(best + rng.normal(0, nl, n) * best)
            elif strat == 1:
                x = _norm1(0.85 * best + 0.15 * rng.dirichlet(np.ones(n) * 0.2))
            elif strat == 2:
                x = _norm1(best * np.exp(0.04 * rng.standard_normal(n)))
            elif strat == 3:
                mask = rng.random(n) > 0.5
                xr = best.copy()
                xr[mask] *= 0.6
                xr[~mask] *= 1.4
                x = _norm1(xr)
            elif strat == 4:
                sh = int(rng.integers(1, max(2, n // 15)))
                x = _norm1(np.roll(best, sh))
            else:
                # Mirror descent nudge
                b_peak = _autocorr(best)
                kpeak = int(np.argmax(b_peak))
                g_floor = np.zeros(n)
                js = max(0, kpeak - n + 1)
                je = min(n, kpeak + 1)
                idx2 = np.arange(js, je)
                g_floor[idx2] = best[kpeak - idx2]
                x = _norm1(best * np.exp(-0.05 * g_floor / max(g_floor.max(), 1e-30)))
            xp = x.copy()
            beta = max(beta * 0.5, 10.0)
            eta = eta0
            if verbose:
                el = time.monotonic() - t0
                print("  Nest R t=" + str(t) + " (" + str(round(el, 1)) + "s) J=" + repr(round(2.0 * n * bm, 6)))

        if verbose and t % 2000 == 0:
            el = time.monotonic() - t0
            print("  Nest t=" + str(t) + " (" + str(round(el, 1)) + "s) J=" + repr(round(2.0 * n * bm, 6)) + " b=" + str(int(beta)))

    return best, bm


def _directed_cd(a0, tlim, seed=55, verbose=True):
    """Directed coordinate descent: identify peak-contributing pairs and reduce them."""
    n = len(a0)
    a = _norm1(a0)
    best = a.copy()
    bm = _autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    sz = 2 * n - 1
    stall = 0
    fs = _fft_size(sz)
    fs2 = _fft_size(n + sz - 1)

    while time.monotonic() - t0 < tlim and stall < 3000:
        b = _autocorr(a)
        # Get top peaks
        npeaks = min(8, sz)
        peak_idx = np.argsort(b)[::-1][:npeaks]

        improved = False
        for ki in peak_idx:
            k = int(ki)
            if b[k] < bm * 0.95:
                break
            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            # Find the top contributor
            top = int(idx[np.argmax(contrib)])
            partner = k - top
            if not (0 <= partner < n):
                continue
            val = a[top] * a[partner]
            if val < b[k] * 0.01:
                continue

            # Try moving mass from top to a safe location
            # Safe location: minimize increase in max autocorrelation
            # Approximate: find position where the "sensitivity" is lowest
            # sensitivity[m] = sum of a[k-m] over all active peaks
            sens = np.zeros(n)
            for pk in peak_idx[:4]:
                pk = int(pk)
                s = max(0, pk - n + 1)
                e = min(n, pk + 1)
                ii = np.arange(s, e)
                sens[ii] += a[pk - ii]

            # Try several candidate destinations
            candidates = np.argsort(sens)[:max(5, n // 50)]
            rng.shuffle(candidates)
            candidates = candidates[:10]

            for dst in candidates:
                dst = int(dst)
                if dst == top or dst == partner:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.04]:
                    amt = a[top] * frac
                    if amt < 1e-15:
                        continue
                    at = a.copy()
                    at[top] -= amt
                    at[dst] += amt
                    at = np.maximum(at, 0.0)
                    st = at.sum()
                    if st < 1e-15:
                        continue
                    at = at / st
                    mt = _autocorr(at).max()
                    if mt < bm - 1e-17:
                        a = at
                        bm = mt
                        best = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 50:
                # Random escape
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * rng.uniform(0.003, 0.02)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                        m = _autocorr(a).max()
                        if m < bm:
                            bm = m
                            best = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 500 == 0 and stall > 0:
            el = time.monotonic() - t0
            print("  DirCD (" + str(round(el, 1)) + "s) J=" + repr(round(2.0 * n * bm, 6)) + " stall=" + str(stall))

    return best, bm


def _spectral(a0, tlim, seed=999, verbose=True):
    """Spectral power flattening with adaptive alpha and phase kicks."""
    n = len(a0)
    a = _norm1(a0)
    best = a.copy()
    bm = _autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    sz = 2 * n - 1
    fs = _fft_size(sz)
    alpha = 0.03

    for it in range(1, 5000000):
        if time.monotonic() - t0 > tlim:
            break
        A = np.fft.rfft(a, fs)
        pw = np.abs(A) ** 2
        target = pw[1:].mean() if len(pw) > 1 else 1.0
        An = A.copy()
        for k in range(1, len(A)):
            r = pw[k] / max(target, 1e-30)
            if r > 1.0:
                An[k] *= max(0.2, np.sqrt(1.0 - alpha * (r - 1.0)))
            elif r < 0.3:
                An[k] *= min(3.0, np.sqrt(1.0 + alpha * 0.2 * (0.3 - r)))
        an = np.fft.irfft(An, fs)[:n]
        an = np.maximum(an, 0.0)
        sn = an.sum()
        if sn < 1e-30:
            continue
        an = an / sn
        mn = _autocorr(an).max()
        if mn < bm - 1e-16:
            bm = mn
            best = an.copy()
            a = an
        elif it % 300 == 0:
            # Phase perturbation
            A2 = np.fft.rfft(a, fs)
            pp = rng.standard_normal(len(A2)) * 0.03
            A2[1:] *= np.exp(1j * pp[1:])
            at = np.fft.irfft(A2, fs)[:n]
            at = np.maximum(at, 0.0)
            st = at.sum()
            if st > 1e-15:
                at = at / st
                mt = _autocorr(at).max()
                if mt < bm:
                    bm = mt
                    best = at.copy()
                    a = at
        if it % 600 == 0:
            alpha = max(0.003, alpha * 0.93)
        if verbose and it % 2000 == 0:
            el = time.monotonic() - t0
            print("  Spec (" + str(round(el, 1)) + "s) J=" + repr(round(2.0 * n * bm, 6)))

    return best, bm


def _lp_fw(a0, tlim, svr, K0=300, Kmax=2500, verbose=True):
    """LP-based Frank-Wolfe with adaptive active set."""
    n = len(a0)
    a = _norm1(a0)
    best = a.copy()
    bm = _autocorr(a).max()
    t0 = time.monotonic()
    sz = 2 * n - 1
    active = set(int(k) for k in np.argsort(_autocorr(a))[::-1][:K0])
    no_imp = 0
    max_no = 800

    for t in range(1, 500000):
        if time.monotonic() - t0 > tlim:
            break
        if no_imp > max_no:
            break
        ac = _autocorr(a)
        Mc = ac.max()
        if Mc < bm - 1e-16:
            bm = Mc
            best = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        cl = sorted(active)
        if len(cl) > Kmax:
            margins = np.array([Mc - ac[k] for k in cl])
            keep = np.argsort(margins)[:Kmax]
            active = set(int(cl[i]) for i in keep)
            cl = sorted(active)
        K = len(cl)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(cl):
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            if s < e:
                idx = np.arange(s, e)
                coeffs[jj, idx] = a[ki - idx]

        gv = cp.Variable(n, nonneg=True)
        Mv = cp.Variable()
        cons = [cp.sum(gv) == 1, coeffs @ gv <= Mv]
        prob = cp.Problem(cp.Minimize(Mv), cons)
        try:
            prob.solve(solver=svr, verbose=False)
        except Exception:
            if len(active) > K0 // 2:
                active = set(int(k) for k in np.argsort(ac)[::-1][:K0 // 2])
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        gval = np.maximum(np.asarray(gv.value, dtype=np.float64), 0.0)
        gs = gval.sum()
        if gs < 1e-15:
            continue
        gd = gval / gs

        # Check violations
        fs3 = _fft_size(sz)
        Yf = np.fft.rfft(a, fs3)
        Gf = np.fft.rfft(gd, fs3)
        cv = np.fft.irfft(Yf * Gf, fs3)[:sz]
        viol = np.where(cv > Mc + 1e-7)[0]
        if len(viol) > 0:
            va = cv[viol] - Mc
            top_v = viol[np.argsort(va)[::-1][:30]]
            for k in top_v:
                active.add(int(k))

        # Line search
        ba = 2.0 / (t + 2.0)
        bt = Mc
        for al in [ba, ba * 0.5, ba * 1.5, 0.05, 0.1, 0.15, 0.3]:
            al = max(0.01, min(1.0, al))
            tr = (1.0 - al) * a + al * gd
            tr = np.maximum(tr, 0.0)
            st = tr.sum()
            if st < 1e-15:
                continue
            tr = tr / st
            mt = _autocorr(tr).max()
            if mt < bt:
                bt = mt
                ba = al
        a = (1.0 - ba) * a + ba * gd
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a < 1e-15:
            a = best.copy()
        else:
            a = a / s_a

        if verbose and t % 80 == 0:
            el = time.monotonic() - t0
            print("  LPFW it=" + str(t) + " (" + str(round(el, 1)) + "s) J=" + repr(round(2.0 * n * bm, 6)) + " |S|=" + str(len(active)))

    return best, bm


def _mirror_descent(a0, tlim, seed=77, verbose=True):
    """Mirror descent with exponential weighting (no projection needed)."""
    n = len(a0)
    a = _norm1(a0)
    best = a.copy()
    bm = _autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    eta = 0.3
    sz = 2 * n - 1
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > tlim:
            break
        if stall > 2000:
            break
        b = _autocorr(a)
        kmax = int(np.argmax(b))
        # Subgradient at the peak
        js = max(0, kmax - n + 1)
        je = min(n, kmax + 1)
        idx = np.arange(js, je)
        g = np.zeros(n)
        g[idx] = a[kmax - idx]
        # Add small diagonal correction
        if 0 <= 2 * idx < sz:
            pass  # skip for simplicity

        # Mirror descent update: a_new = a * exp(-eta*g) / norm
        lg = -eta * g
        lg -= lg.max()
        w = np.exp(lg)
        a_new = a * w
        sn = a_new.sum()
        if sn < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        mn = _autocorr(a_new).max()
        if mn < bm - 1e-16:
            bm = mn
            best = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.02, 2.0)
        else:
            stall += 1
            if stall % 100 == 0:
                eta *= 0.95
            # Occasionally take a random step
            if stall % 200 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j:
                    amt = a[i] * 0.01
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
        if verbose and t % 5000 == 0:
            el = time.monotonic() - t0
            print("  Mirr t=" + str(t) + " (" + str(round(el, 1)) + "s) J=" + repr(round(2.0 * n * bm, 6)))

    return best, bm


def _update(bs, bsc, a, lbl, tg):
    if a is None:
        return bs, bsc
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return bs, bsc
    an = a / s
    sc = _eval(an)
    if np.isfinite(sc) and sc < bsc:
        bsc = sc
        bs = an.copy()
        el = time.monotonic() - tg
        print("  *** BEST [" + lbl + "]=" + repr(round(sc, 6)) + " n=" + str(len(a)) + " t=" + str(round(el, 1)) + "s")
    return bs, bsc


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimizer combining Nesterov gradient, directed coordinate
    descent, LP-Frank-Wolfe, spectral shaping, and mirror descent.
    Uses adaptive beta scheduling, peak-targeting moves, and multi-length search.
    """
    tg = time.monotonic()
    tb = min(float(budget_s) - 45.0, 1750.0)
    svr = _solver()
    print("Budget: " + repr(tb) + "s, solver: " + str(svr))

    bs = None
    bsc = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            par = p.copy()
            ps = _eval(par)
            if np.isfinite(ps):
                bs = _norm1(par)
                bsc = ps
                print("Parent: n=" + str(len(par)) + " J=" + repr(round(bsc, 6)))
    except Exception:
        pass

    if bs is None:
        n0 = 5000
        x = np.linspace(0, 1, n0)
        bs = _norm1(np.sin(np.pi * x) ** 2)
        bsc = _eval(bs)
        print("Default: n=" + str(n0) + " J=" + repr(round(bsc, 6)))

    n0 = len(bs)
    print("Start J=" + repr(round(bsc, 6)))

    # Phase 1: Quick multi-start exploration (40s)
    rem = tb - (time.monotonic() - tg)
    print("\n== P1: Multi-start (n=" + str(n0) + ") ==")
    if rem > 40:
        t1 = min(rem * 0.04, 40.0)
        inits = [
            _norm1(np.sin(np.pi * np.linspace(0, 1, n0)) ** 2),
            _norm1(np.cos(np.pi * np.linspace(0, 1, n0)) ** 2 + 0.1),
            _norm1(np.sin(2 * np.pi * np.linspace(0, 1, n0)) ** 2 + 0.3),
            _norm1(np.exp(-4 * (np.linspace(0, 1, n0) - 0.5) ** 2)),
            _norm1(np.minimum(np.linspace(0, 1, n0), 1 - np.linspace(0, 1, n0)) + 0.05),
        ]
        for ii, init in enumerate(inits):
            if time.monotonic() - tg > tb:
                break
            tint = t1 / (len(inits) - ii)
            ao, _ = _nesterov(init, tint, beta0=15, beta1=200, growth=1.005,
                              eta0=0.4, tk_ratio=0.04, restart=500, seed=100 + ii, verbose=False)
            bs, bsc = _update(bs, bsc, ao, "ms" + str(ii), tg)

    # Phase 2: Main Nesterov (500s)
    rem = tb - (time.monotonic() - tg)
    nb = len(bs)
    print("\n== P2: Nesterov main (n=" + str(nb) + ") ==")
    if rem > 100:
        t2 = min(rem * 0.30, 500.0)
        ao, _ = _nesterov(_norm1(bs), t2, beta0=20, beta1=600, growth=1.003,
                          eta0=0.5, tk_ratio=0.03, restart=1200, seed=42, verbose=True)
        bs, bsc = _update(bs, bsc, ao, "nest1", tg)

    # Phase 3: Directed coordinate descent (300s)
    rem = tb - (time.monotonic() - tg)
    nb = len(bs)
    print("\n== P3: Directed CD (n=" + str(nb) + ") ==")
    if rem > 60:
        t3 = min(rem * 0.18, 300.0)
        ao, _ = _directed_cd(_norm1(bs), t3, seed=55, verbose=True)
        bs, bsc = _update(bs, bsc, ao, "dcd1", tg)

    # Phase 4: LP-Frank-Wolfe (250s)
    rem = tb - (time.monotonic() - tg)
    nb = len(bs)
    print("\n== P4: LP-FW (n=" + str(nb) + ") ==")
    if rem > 50 and nb <= 9000:
        t4 = min(rem * 0.15, 250.0)
        ao, _ = _lp_fw(_norm1(bs), t4, svr, K0=350, Kmax=2500, verbose=True)
        bs, bsc = _update(bs, bsc, ao, "lp1", tg)

    # Phase 5: Spectral (150s)
    rem = tb - (time.monotonic() - tg)
    nb = len(bs)
    print("\n== P5: Spectral (n=" + str(nb) + ") ==")
    if rem > 40:
        t5 = min(rem * 0.09, 150.0)
        ao, _ = _spectral(_norm1(bs), t5, seed=999, verbose=True)
        bs, bsc = _update(bs, bsc, ao, "spec1", tg)

    # Phase 6: Mirror descent (100s)
    rem = tb - (time.monotonic() - tg)
    nb = len(bs)
    print("\n== P6: Mirror descent (n=" + str(nb) + ") ==")
    if rem > 30:
        t6 = min(rem * 0.06, 100.0)
        ao, _ = _mirror_descent(_norm1(bs), t6, seed=77, verbose=True)
        bs, bsc = _update(bs, bsc, ao, "mirr1", tg)

    # Phase 7: Multi-length exploration (250s)
    rem = tb - (time.monotonic() - tg)
    nb = len(bs)
    print("\n== P7: Multi-length (base=" + str(nb) + ") ==")
    lengths = []
    for d in [200, 500, 1000, 2000]:
        ln = nb + d
        if ln <= 12000:
            lengths.append(ln)
    for ln in lengths:
        rem = tb - (time.monotonic() - tg)
        if rem < 50:
            break
        print("  -> n=" + str(ln))
        al = _interp(bs, ln)
        t7a = min(rem * 0.35, 150.0)
        if t7a > 25:
            ao, _ = _nesterov(al, t7a, beta0=25, beta1=500, growth=1.004,
                              eta0=0.4, tk_ratio=0.025, restart=1000,
                              seed=ln % 7919, verbose=True)
            bs, bsc = _update(bs, bsc, ao, "ml_n" + str(ln), tg)
        rem = tb - (time.monotonic() - tg)
        if rem > 40 and ln <= 8000:
            t7b = min(rem * 0.35, 120.0)
            if t7b > 20:
                al2 = _interp(bs, ln)
                ao, _ = _lp_fw(_norm1(al2), t7b, svr, K0=300, Kmax=2000, verbose=True)
                bs, bsc = _update(bs, bsc, ao, "lpl_n" + str(ln), tg)

    # Phase 8: Second Nesterov round - different params (200s)
    rem = tb - (time.monotonic() - tg)
    nb = len(bs)
    print("\n== P8: Nesterov r2 (n=" + str(nb) + ") ==")
    if rem > 60:
        t8 = min(rem * 0.12, 200.0)
        ao, _ = _nesterov(_norm1(bs), t8, beta0=50, beta1=1200, growth=1.005,
                          eta0=0.3, tk_ratio=0.015, restart=800, seed=999, verbose=True)
        bs, bsc = _update(bs, bsc, ao, "nest2", tg)

    # Phase 9: Final directed CD (100s)
    rem = tb - (time.monotonic() - tg)
    nb = len(bs)
    print("\n== P9: Directed CD r2 (n=" + str(nb) + ") ==")
    if rem > 40:
        t9 = min(rem * 0.08, 100.0)
        ao, _ = _directed_cd(_norm1(bs), t9, seed=222, verbose=True)
        bs, bsc = _update(bs, bsc, ao, "dcd2", tg)

    # Phase 10: Final spectral + Nesterov polish (remaining)
    rem = tb - (time.monotonic() - tg)
    nb = len(bs)
    print("\n== P10: Final polish (n=" + str(nb) + ") ==")
    if rem > 30:
        t10a = min(rem * 0.25, 80.0)
        if t10a > 15:
            ao, _ = _spectral(_norm1(bs), t10a, seed=888, verbose=False)
            bs, bsc = _update(bs, bsc, ao, "spec2", tg)
        rem = tb - (time.monotonic() - tg)
        if rem > 30:
            t10b = min(rem * 0.40, 120.0)
            if t10b > 15:
                ao, _ = _nesterov(_norm1(bs), t10b, beta0=100, beta1=2000,
                                  growth=1.007, eta0=0.2, tk_ratio=0.01,
                                  restart=600, seed=555, verbose=True)
                bs, bsc = _update(bs, bsc, ao, "nest3", tg)
        rem = tb - (time.monotonic() - tg)
        if rem > 20 and nb <= 8000:
            t10c = min(rem * 0.50, 100.0)
            if t10c > 15:
                ao, _ = _lp_fw(_norm1(bs), t10c, svr, K0=400, Kmax=3000, verbose=True)
                bs, bsc = _update(bs, bsc, ao, "lpf", tg)

    # Final check
    if bs is not None:
        fs2 = _eval(bs)
        if np.isfinite(fs2):
            bsc = fs2
        el = time.monotonic() - tg
        print("\nFINAL: J=" + repr(round(bsc, 6)) + " n=" + str(len(bs)) + " t=" + str(round(el, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        bs = _norm1(np.sin(np.pi * x) ** 2)
        bsc = _eval(bs)
        print("FALLBACK: J=" + repr(round(bsc, 6)))

    return [float(x) for x in np.asarray(bs, dtype=np.float64)]
