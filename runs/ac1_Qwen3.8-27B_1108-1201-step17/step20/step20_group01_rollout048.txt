_SOLVER = None


def _get_solver():
    global _SOLVER
    if _SOLVER is not None:
        return _SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER = s
                print("  Solver: " + str(s))
                return s
        except Exception:
            continue
    _SOLVER = cp.ECOS
    print("  Solver: ECOS (fallback)")

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _obj_val(a):
    n = len(a)
    return 2.0 * n * _fast_autocorr(a).max()

def _build_constraint_matrix(y, active_lags, support_mask):
    n = len(y)
    n_support = int(support_mask.sum())
    K = len(active_lags)
    coeffs = np.zeros((K, n_support), dtype=np.float64)
    support_idx = np.where(support_mask)[0]
    for jj in range(K):
        ki = int(active_lags[jj])
        i_start = max(0, ki - (n - 1))
        i_end = min(n - 1, ki)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            valid = idx[support_mask[idx]]
            if len(valid) > 0:
                col_idx = np.searchsorted(support_idx, valid)
                coeffs[jj, col_idx] = y[ki - valid]
    return coeffs


def _fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    B = np.fft.rfft(b, n=fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def _lp_fw_step(y, active_lags, support_mask, solver):
    n = len(y)
    n_support = int(support_mask.sum())
    if n_support < 2:
        return None
    support_idx = np.where(support_mask)[0]
    g_var = cp.Variable(n_support, nonneg=True)
    M_var = cp.Variable()
    K = len(active_lags)
    if K == 0:
        prob = cp.Problem(cp.Minimize(cp.sum(squares(g_var))), [cp.sum(g_var) == 1])
    else:
        coeffs = _build_constraint_matrix(y, active_lags, support_mask)
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    g_sup = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    gs = g_sup.sum()
    if gs < 1e-15:
        return None
    g = np.zeros(n, dtype=np.float64)
    g[support_idx] = g_sup / gs
    return g


def lp_fw_optimize(a0, time_limit, solver, init_K=200, max_K=2000, verbose=True):

    t0 = time.monotonic()
    size = 2 * n - 1

    active_set = list(np.argsort(_fast_autocorr(y))[-init_K:])
    support_mask = np.ones(n, dtype=bool)
    away_set = set()
    no_improve = 0
    stall_limit = 1500

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = _fast_autocorr(y)
        M_curr = ac.max()
        cur_obj = 2.0 * n * M_curr

        if cur_obj < 2.0 * n * best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t % 20 == 0:
            target_K = min(max_K, init_K + t * 3)
            top_sorted = np.argsort(ac)[::-1]
            current_set = set(int(k) for k in active_set)
            for k in top_sorted[:target_K]:
                current_set.add(int(k))
            if len(current_set) > max_K:
                clist = sorted(current_set)
                margins = np.array([M_curr - ac[k] for k in clist])
                keep = np.argsort(margins)[:max_K]
                active_set = [int(clist[i]) for i in keep]
            else:
                active_set = list(current_set)
            active_set.append(int(np.argmax(ac)))

        g = _lp_fw_step(y, active_set, support_mask, solver)
        if g is None:
            if len(active_set) > init_K // 2:
                clist = sorted(set(active_set))
                ac_tmp = _fast_autocorr(y)
                margins = np.array([ac_tmp.max() - ac_tmp[k] for k in clist])
                keep = np.argsort(margins)[:init_K // 2]
                active_set = [int(clist[i]) for i in keep]
            continue

        conv_yg = _fft_convolve(y, g)
        violations = np.where(conv_yg > M_curr + 1e-9)[0]
        if len(violations) > 0:
            viol_amt = conv_yg[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:15]]

            if ts > 1e-15:
                trial_damp = trial_damp / ts

def subgrad_optimize(a0, time_limit, n_top=100, verbose=True):
    n = len(a0)
    x = _normalize(a0)
    best_a = x.copy()
    best_val = _obj_val(x)

    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break

        ac = _fast_autocorr(x)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        k_sel = min(n_top, 2 * n - 1)
        idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[idx]
        temp = 50.0
        w = np.exp(temp * (top_vals - top_vals[0]))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_sel) / k_sel
        else:
            w = w / w_sum

        grad = np.zeros(n, dtype=np.float64)
        for ii in range(k_sel):
            ki = int(idx[ii])
            contrib = 2.0 * w[ii]
            i_start = max(0, ki - (n - 1))
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                j_arr = np.arange(i_start, i_end + 1)
                grad[j_arr] += contrib * x[ki - j_arr]

        lr = 0.03 / (1.0 + 0.01 * t)
        log_x = np.log(x + 1e-30)
        log_x -= lr * grad
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        x_new = np.maximum(x_new, 1e-25)
        x_new = x_new / x_new.sum()
        x = x_new

        if stall > 2000 and stall % 500 == 0:
            noise = 0.01 * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x = x_r / x_r.sum()
            stall = 0

        if verbose and t % 200 == 0:
            print("    SubGrad t={} ({:.1f}s) obj={:.7f} stall={}".format(
                t, time.monotonic() - t0, best_val, stall))

    return best_a, best_val

def _make_init_oneside_sine(n):

def _interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

    n_cur = len(best_seq)
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 60:
        print("\n=== Phase 1: LP-FW (n={}) t={:.0f}s ===".format(n_cur, t1))
        a_out, v_out = lp_fw_optimize(best_seq, t1, solver, init_K=150, max_K=1500, verbose=True)
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1 improved: {}".format(repr(best_score)))

    # Phase 2: Subgradient polish
    remaining = _time_left()
    if remaining > 60:
        t2 = min(remaining * 0.15, 200.0)
        print("\n=== Phase 2: Subgradient (n={}) t={:.0f}s ===".format(len(best_seq), t2))
        a_out, v_out = subgrad_optimize(best_seq, t2, n_top=150, verbose=True)
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: {}".format(repr(best_score)))

    # Phase 3: Coordinate refinement
    remaining = _time_left()
    if remaining > 40:
        t3 = min(remaining * 0.15, 200.0)
        print("\n=== Phase 3: CoordRefine (n={}) t={:.0f}s ===".format(len(best_seq), t3))
        a_out, v_out = coord_refine(best_seq, t3, seed=42, verbose=True)
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 improved: {}".format(repr(best_score)))

    # Phase 4: Second LP-FW pass (different active set strategy)
    remaining = _time_left()
    if remaining > 60:
        t4 = min(remaining * 0.25, 400.0)
        n_cur = len(best_seq)
        print("\n=== Phase 4: LP-FW round 2 (n={}) t={:.0f}s ===".format(n_cur, t4))
        a_out, v_out = lp_fw_optimize(best_seq, t4, solver, init_K=300, max_K=2500, verbose=True)
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 improved: {}".format(repr(best_score)))

    # Phase 5: Try different length
    remaining = _time_left()
    n_long = len(best_seq) + 3000
    if remaining > 120 and n_long <= 20000:
        t5 = min(remaining * 0.20, 250.0)
        print("\n=== Phase 5: Long n={} t={:.0f}s ===".format(n_long, t5))
        a_long = _interpolate_to_length(best_seq, n_long)
        a_out, v_out = subgrad_optimize(a_long, t5 * 0.5, n_top=200, verbose=False)
        a_out2, v_out2 = coord_refine(a_out, t5 * 0.5, seed=99, verbose=False)
        sc1 = _eval_safe(a_out)
        sc2 = _eval_safe(a_out2)
        if np.isfinite(min(sc1, sc2)) and min(sc1, sc2) < best_score:
            best_score = min(sc1, sc2)
            best_seq = a_out if sc1 < sc2 else a_out2
            print("  P5 improved: {}".format(repr(best_score)))

    # Phase 6: Final subgradient + coord polish
    remaining = _time_left()
    if remaining > 30:
        t6 = min(remaining * 0.5, 200.0)
        print("\n=== Phase 6: Final polish (n={}) t={:.0f}s ===".format(len(best_seq), t6))
        a_out, v_out = subgrad_optimize(best_seq, t6 * 0.5, n_top=200, verbose=False)
        a_out2, v_out2 = coord_refine(a_out, t6 * 0.5, seed=777, verbose=True)
        sc1 = _eval_safe(a_out)
        sc2 = _eval_safe(a_out2)
        if np.isfinite(min(sc1, sc2)) and min(sc1, sc2) < best_score:
            best_score = min(sc1, sc2)
            best_seq = a_out if sc1 < sc2 else a_out2
            print("  P6 improved: {}".format(repr(best_score)))

    # Final validation
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} elapsed={:.1f}s".format(repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: using fallback")
        best_seq = _make_init_decreasing(10000)
        best_score = _eval_safe(best_seq)

    return [float(x) for x in best_seq]
