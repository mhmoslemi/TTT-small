    n = len(f)

    # Actually, we need to think about which constraints will be binding for g
    # Use the top k values of conv(f,f) as a guide
    top_idx = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix for these k values
    # A[k_idx, j] = f[k_idx - j] if 0 <= k_idx - j < n, else 0
    from scipy.sparse import csr_matrix
    rows, cols, vals = [], [], []
    for i, k in enumerate(top_idx):

        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            vals.append(f[k_int - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(top_k, n))
    
    # LP: max 1^T g s.t. A @ g <= M * ones, g >= 0
    # In scipy: minimize -1^T g s.t. A @ g <= M, g >= 0
    c_obj = -np.ones(n)
    b = np.full(top_k, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, method='highs', bounds=(0, None))
    
    if res.success:
        return res.x
    else:
        return None

    f = f0.copy()
    n = len(f)
    best_f = f.copy()
    best_M = fast_autocorr(f).max()
    t0 = time.time()
    
    for it in range(100000):
        if time.time() - t0 > time_limit:
            break
        
        c = fast_autocorr(f)
        M = c.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_f = f.copy()
        
        # Solve LP for best response
        g = lp_best_response(f, M, top_k=top_k)
        if g is None or g.sum() < 1e-15:

        f_new = (1 - eta) * f + eta * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        f = f_new
        
        if verbose and it % 100 == 0:
            print(f"  MW it={it}, M={M:.8f}, best_M={best_M:.8f}")
    
    return best_f, best_M

def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

                    pressure[idx2] += f[k2i - idx2]
                    pressure[k2i - idx2] += f[idx2]

                continue

def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max of autocorrelation."""
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    Conv = np.fft.irfft(P * np.fft.rfft(a_rev, fft_size), fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c


def nesterov_opt(f0, time_limit, beta_start=50.0, beta_end=3000.0,
                 beta_growth=1.008, eta_base=0.5, top_k_ratio=0.015,
                 restart_every=1500, verbose=True, seed=1):
    """Nesterov accelerated projected gradient on log-sum-exp smoothed objective."""
    n = len(f0)
    x = norm_sum1(f0)
    x_prev = x.copy()
    best_f = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    
    for t in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        grad, p, c = compute_smoothed_grad(y, beta, top_k)
        
        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)
        
        z = y - eta * grad
        x_new = simplex_project(z)
        
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_M - 1e-15:
            best_M = max_new
            best_f = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.03 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_f
            x_r = best_f + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.6, 10.0)
            if verbose:
                el = time.monotonic() - t0
                J = 2.0 * n * best_M
                print(f"    Nest restart t={t} ({el:.1f}s): J={J:.6f}")
        
        if verbose and t % 2000 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Nest t={t} ({el:.1f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_f, best_M


def pairwise_cd(f0, time_limit, verbose=True, seed=123):
    """Fine-grained pairwise coordinate descent for final polish."""
    n = len(f0)
    f = norm_sum1(f0)
    best_f = f.copy()
    best_M = fast_autocorr(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 400:
        c = fast_autocorr(f)
        threshold = np.percentile(c, 95)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:20]
        
        improved = False
        
        for k in peak_positions[:15]:
            if improved and (time.monotonic() - t0 > time_limit):
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = f[idx] * f[k_int - idx]
            
            top_pairs = np.argsort(products)[::-1][:3]
            for pi in top_pairs:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if f[src_i] < 1e-14 and f[partner] < 1e-14:
                    continue
                
                larger_i = src_i if f[src_i] >= f[partner] else partner
                if f[larger_i] < 1e-14:
                    continue
                
                for frac in [0.001, 0.002, 0.005]:
                    amt = f[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(3):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        f_trial = f.copy()
                        f_trial[larger_i] -= amt
                        f_trial[dst] += amt
                        f_trial = np.maximum(f_trial, 0.0)
                        s_t = f_trial.sum()
                        if s_t < 1e-15:
                            continue
                        f_trial = f_trial / s_t
                        M_trial = fast_autocorr(f_trial).max()
                        if M_trial < best_M - 1e-17:
                            f = f_trial
                            best_M = M_trial
                            best_f = f.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall % 100 == 0:
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.01 / (1.0 + stall / 100.0)
                f_try = f + eps * anti * f
                f_try = np.maximum(f_try, 0.0)
                s_t = f_try.sum()
                if s_t > 1e-15:
                    f_try = f_try / s_t
                    M_try = fast_autocorr(f_try).max()
                    if M_try < best_M:
                        f = f_try
                        best_M = M_try
                        best_f = f.copy()
                        stall = 0
        else:
            stall = 0
        
        if verbose and stall > 0 and stall % 200 == 0:
            el = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    PairCD ({el:.1f}s): J={J:.6f} stall={stall}")
    
    return best_f, best_M


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.1f}s)")
    return best_seq, best_score
