    # corr[i] = sum_j p[i+j] * a[j] for j=0..n-1
    # slide[i] = sum_j p[i+j] for j=0..n-1
    
    # corr via FFT: cross-correlation
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size)[:n]
    
    # slide via cumsum
    p_cumsum = np.concatenate([[0.0], np.cumsum(p)])
    slide = p_cumsum[n:2*n] - p_cumsum[0:n]
    
    grad = beta * (corr + a * slide)
    return grad, b

def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update on the simplex."""
    log_step = -eta * grad
    log_step -= log_step.max()  # numerical stability
    weights = a * np.exp(log_step)
    s = weights.sum()
    if s < 1e-300:
        return a.copy()
    return weights / s

def peak_shavity(a, max_time, top_k=20, rng=None):
    """Aggressive peak shaving: move mass from peak contributors to safe positions."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    s = a.sum()

        return a

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < max_time and stall < max_stall:
        b = fast_autocorr(a)
        n_peaks = min(top_k, len(b))
        peak_indices = np.argsort(b)[::-1][:n_peaks]
        
        improved = False

            if time.monotonic() - t0 > max_time:
                break
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_i - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Find the top contributors
            top_contrib = np.argsort(contrib)[::-1][:5]
            for tc in top_contrib:
                src = int(idx[tc])
                partner = k_i - src
                if partner < 0 or partner >= n:
                    continue

                        pressure[idx2] += a[k2i - idx2]
                        pressure[k2i - idx2] += a[idx2]
                
                pressure[src] = np.inf
                pressure[partner] = np.inf
                dst = int(np.argmin(pressure))
                
                for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:

                        continue

                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
                    else:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros_like(b)
        b_use[idx] = b[idx]
    else:
        b_use = b
    
    b_max = b_use.max()
    shifted = np.clip(beta * (b_use - b_max), -400, 400)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.ones(len(b)) / len(b)
    else:
        p = exp_vals / total
    
    # grad[i] = beta * (corr[i] + a[i] * slide[i])
    # corr[i] = sum_j p[i+j] * a[j] for j=0..n-1
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size)[:n]
    
    # slide[i] = sum_{j=0}^{n-1} p[i+j]
    p_cumsum = np.concatenate([[0.0], np.cumsum(p)])
    slide = p_cumsum[n:2 * n] - p_cumsum[0:n]
    
    grad = beta * (corr + a * slide)
    return grad, b, p

def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update on the simplex (log-form for stability)."""
    log_weights = np.log(a + 1e-300) - eta * grad
    log_weights -= log_weights.max()
    weights = np.exp(log_weights)
    s = weights.sum()
    if s < 1e-300:
        return a.copy()
    return weights / s

def peak_shaving(a0, max_time, top_k=20, rng=None):
    """Aggressive peak shaving: move mass from peak contributors to low-pressure positions."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < max_time and stall < max_stall:
        b = fast_autocorr(a)
        n_peaks = min(top_k, len(b))
        peak_indices = np.argsort(b)[::-1][:n_peaks]
        
        # Compute pressure map
        pressure = np.zeros(n)
        for k_int in peak_indices[:10]:
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[k_i - idx]
                pressure[k_i - idx] += a[idx]
        
        improved = False
        for k_int in peak_indices[:10]:
            if time.monotonic() - t0 > max_time:
                break
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_i - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Top contributors to this peak
            top_contrib = np.argsort(contrib)[::-1][:6]
            for tc in top_contrib:
                src = int(idx[tc])
                partner = k_i - src
                if partner < 0 or partner >= n:
                    continue
                if a[src] < 1e-14:
                    continue
                
                # Find destination with lowest pressure
                p_copy = pressure.copy()
                p_copy[src] = np.inf
                if 0 <= partner < n:
                    p_copy[partner] = np.inf
                dst = int(np.argmin(p_copy))
                
                # Try moving mass
                found = False
                for frac in [0.02, 0.05, 0.1, 0.2, 0.35, 0.5]:
                    amt = a[src] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                
                if found:
                    break
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                noise_scale = min(0.03, 0.005 * (stall / 20.0))
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_autocorr(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r if m_r < best_m * 1.02 else best_a.copy()
                if m_r < best_m:
                    stall = 0
    
    return best_a, best_m

def mirror_descent_optimize(a0, max_time, beta_start=10.0, beta_end=300.0,
                           top_k_start=40, use_full_end=True,
                           annealing_temp=0.001, verbose=True, seed=12345):
    """Mirror descent with adaptive beta and top-k annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    t = 0
    stall = 0
    best_eta = 1.0 / (2.0 * beta_start)
    
    while time.monotonic() - t0 < max_time:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max_time, 1.0)
        
        # Adaptive beta: geometric increase
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Adaptive top-k: start focused, end full
        if use_full_end:
            top_k = int(top_k_start + (2 * n - 1 - top_k_start) * (progress ** 0.7))
        else:
            top_k = top_k_start
        top_k = max(10, min(top_k, 2 * n - 1))
        
        grad, b, p = compute_smoothed_gradient(a, beta, top_k=top_k)
        
        g_max = np.abs(grad).max()
        if g_max < 1e-15:
            break
        
        # Adaptive step size
        eta = 1.0 / (g_max * 1.2)
        eta = min(eta, 1.0 / (beta * 1.5))
        eta = max(eta, 1e-6)
        
        # Mirror descent step
        a_new = mirror_descent_step(a, grad, eta)
        m_new = fast_autocorr(a_new).max()
        
        # Simulated annealing acceptance
        temp = annealing_temp * np.exp(-4.0 * progress)
        accept = False
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
            a = a_new
            accept = True
        elif m_new < best_m * (1.0 + temp) and rng.random() < np.exp(-(m_new / best_m - 1.0) / max(temp, 1e-10)):
            a = a_new
            stall += 1
            accept = True
        
        if not accept:
            # Try smaller step
            a_new2 = mirror_descent_step(a, grad, eta * 0.3)
            m_new2 = fast_autocorr(a_new2).max()
            if m_new2 < m_new:
                a = a_new2
                if m_new2 < best_m - 1e-15:
                    best_m = m_new2
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        
        # Periodic perturbation to escape local minima
        if stall > 80:
            noise_scale = 0.015 * min(1.5, stall / 100.0)
            noise = rng.normal(0, noise_scale, n) * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            stall = 0
        
        # Entropy check: if sequence becomes too concentrated, push toward uniform
        if t % 50 == 0:
            entropy = -np.sum(a * np.log(a + 1e-300))
            max_entropy = np.log(n)
            if entropy < 0.3 * max_entropy:
                a = 0.5 * a + 0.5 * (1.0 / n)
                a = a / a.sum()
        
        if verbose and t % 300 == 0:
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f} beta={beta:.1f} top_k={top_k} stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def diverse_init(n, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0, 1, n)
    candidates = []
    
    # Power law profiles
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        for beta_p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
            a_raw = (x + 0.01) ** alpha * (1.0 - x + 0.01) ** beta_p
            candidates.append(norm_sum1(a_raw))
    
    # Gaussian mixtures
    for c1 in [0.3, 0.4, 0.5]:
        for c2 in [0.5, 0.6, 0.7]:
            for s in [0.08, 0.12, 0.18]:
                a_raw = np.exp(-((x - c1) ** 2) / (2 * s ** 2)) + np.exp(-((x - c2) ** 2) / (2 * s ** 2))
                candidates.append(norm_sum1(a_raw))
    
    # Random power
    for _ in range(10):
        a = norm_sum1(rng.random(n) ** rng.uniform(0.3, 4.0))
        candidates.append(a)
    
    return candidates

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    elif np.isfinite(sc) and sc < best_score + 0.005:
        elapsed = time.monotonic() - t_global
        print(f"  [near-best {label}]: {sc:.6f} (n={len(a)}, t={elapsed:.1f}s)")
    return best_seq, best_score
