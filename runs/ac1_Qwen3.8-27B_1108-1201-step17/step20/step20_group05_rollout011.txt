def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm1(a):
    """Normalize a to be non-negative with sum 1."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 0.0, out=a)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def fft_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)


def fft_conv2(a, b):
    """Full linear convolution a*b via FFT."""
    n = a.size
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    return np.where(np.isfinite(c) & (c > 0.0), c, 0.0)


def cmax(c, mask=None):
    """Max of c, optionally masked."""
    if mask is None:
        return float(np.max(c))
    return float(np.max(c[mask]))


def active_top(c, mask, k):
    """Indices of top-k values in c, respecting mask."""
    k = min(k, c.size)
    if k <= 0:
        return np.array([], dtype=int)
    if mask is None:
        return np.argsort(c)[::-1][:k]
    v = c.copy()
    v[~mask] = -np.inf
    return np.argsort(v)[::-1][:k]


def top_mask(c, k, base=None):
    """Boolean mask for top-k entries of c."""
    size = c.size
    if k >= size:
        m = np.ones(size, dtype=bool)
    else:
        idx = active_top(c, None, k)
        m = np.zeros(size, dtype=bool)
        m[idx] = True
    if base is not None:
        m[~base] = False
    return m

def interp_seq(a, new_n):
    """Interpolate sequence to new length."""
    a = np.asarray(a, dtype=np.float64)
    new_n = max(1, int(new_n))
    if a.size == new_n:
        return norm1(a)
    x = np.linspace(0.0, 1.0, a.size)
    y = np.linspace(0.0, 1.0, new_n)
    v = np.interp(y, x, a)
    v = np.where(np.isfinite(v), v, 0.0)
    np.maximum(v, 0.0, out=v)
    return norm1(v)

def softmax_u(u):
    u = np.asarray(u, dtype=np.float64)
    n = u.size
    if n == 0:
        return np.ones(1)
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s < 1e-15:
        return np.ones(n) / n
    return e / s


def make_v(a, ref=None):
    """Convert a to log-coordinates relative to a[ref]."""
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        ref = int(np.argmax(a)) if np.any(a > 0) else 0
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref+1:]))
    return np.where(np.isfinite(v), v, -700.0), ref


def from_v(v, ref):
    """Convert log-coordinates back to normalized sequence."""
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref+1:] = v[ref:]
    return softmax_u(u)


def lse_objgrad(a, beta, mask):
    """Log-sum-exp smoothed max-convolution objective and gradient."""
    n = a.size
    if n == 0:
        return 0.0, np.array([])
    size = 2 * n - 1
    c = fft_conv(a)
    
    if mask is None:
        cv = c
        pf = np.empty(size, dtype=np.float64)
    else:
        cv = c[mask]
        pf = np.zeros(size, dtype=np.float64)
        pf[mask] = 1.0  # placeholder
    
    if cv.size == 0:
        return 0.0, np.zeros(n)
    
    m = float(np.max(cv))
    shifted = np.clip(beta * (cv - m), -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se < 1e-15:
        return m, np.zeros(n)
    p = e / se
    
    # Build full-length probability vector
    pfull = np.zeros(size, dtype=np.float64)
    if mask is None:
        pfull[:] = p
    else:
        pfull[mask] = p
    
    # Gradient: 2 * cross_convolve(pfull, a)
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(pfull[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    
    f = m + math.log(se) / beta
    return f, grad


def objgrad_v(v, beta, mask, ref):
    """Objective and gradient in log-coordinates."""
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([])
    
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref+1:] = v[ref:]
    
    a = softmax_u(u)
    f, ga = lse_objgrad(a, beta, mask)
    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)
    
    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref+1:]
    
    # Small regularization
    gv = gv + 1e-12 * v
    f += 5e-13 * float(np.dot(v, v))
    
    return f, gv


def run_lbfegs(a0, beta, mask, maxfun=400, verbose=True):
    """Run L-BFGS-B on the smoothed objective."""
    a = norm1(a0)
    n = a.size
    if n <= 1 or not SCIPY_OK:
        return a
    
    v0, ref = make_v(a)
    t0 = time.monotonic()
    
    try:
        res = scipy.optimize.minimize(
            objgrad_v, v0,
            args=(beta, mask, ref),
            method='L-BFGS-B', jac=True,
            options={'maxfun': int(maxfun), 'ftol': 1e-14, 'gtol': 1e-8, 'disp': False}
        )
        if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            a_out = from_v(np.asarray(res.x, dtype=np.float64), ref)
            a_out = norm1(a_out)
            if verbose:
                print("    L-BFGS b={} n={} t={:.1f}s ok={}".format(
                    beta, n, time.monotonic()-t0, res.success))
            return a_out
    except Exception as e:
        print("    L-BFGS err: {}".format(e))
    
    if verbose:
        print("    L-BFGS b={} n={} t={:.1f}s (fallback)".format(beta, n, time.monotonic()-t0))
    return a

# --- Frank-Wolfe ---

def cross_pa(p, a, factor=1.0):
    """Cross-convolve p (length 2n-1) with a (length n), return length-n vector."""
    n = a.size
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n-1:2*n-1][::-1]
    return factor * np.where(np.isfinite(out), out, 0.0)


def game_dir(f, act_idx, iters=20, seed=0):
    """Solve min_g max_k (f*g)_k approximately via zero-sum game."""
    f = norm1(f)
    n = f.size
    size = 2 * n - 1
    act_idx = np.asarray(act_idx, dtype=int)
    K = act_idx.size
    if K == 0:
        return np.ones(n) / n
    
    rng = np.random.default_rng(seed)
    logits = rng.normal(0, 0.1, K)
    
    for _ in range(max(1, int(iters))):
        # Player 1: choose g to minimize max (f*g)_k
        p = np.zeros(size)
        lm = logits - np.max(logits)
        w = np.exp(lm)
        w = w / w.sum()
        p[act_idx] = w
        
        press = cross_pa(p, f)
        press = np.where(np.isfinite(press) & (press > 0), press, 0.0)
        pmx, pmn = float(press.max()), float(press.min())
        if pmx - pmn < 1e-30:
            g = np.ones(n) / n
        else:
            sc = (press - pmn) / (pmx - pmn)
            g = np.exp(-6.0 * sc)
            g = np.where(np.isfinite(g), g, 0.0)
            g += 1e-10
            g = g / g.sum()
            g = 0.98 * g + 0.02 / n
        
        # Player 2: update lambda to maximize min_k (g*h)_k... 
        # Actually, update to push weight toward peaks of f*g
        h = fft_conv2(f, g)
        ha = h[act_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hm = float(ha.max())
        if hm > 1e-30:
            eta = 10.0 / (hm + 1e-30)
            logits += eta * (ha - ha.mean())
    
    return g


def line_search(a, g, cf, cg, fg, mask, M, step_fracs):
    """Find best mixing alpha."""
    best_a, best_M, best_al = a.copy(), M, 0.0
    for al in step_fracs:
        v = (1-al)**2 * cf + 2*al*(1-al) * fg + al**2 * cg
        if mask is None:
            m = float(np.max(v))
        else:
            m = float(np.max(v[mask]))
        if math.isfinite(m) and m < best_M:
            best_M = m
            best_al = al
    if best_al > 0:
        na = (1 - best_al) * a + best_al * g
        na = np.maximum(na, 0.0)
        s = na.sum()
        if s > 1e-15:
            best_a = na / s
    return best_a, best_M, best_al

    t0 = time.monotonic()

def cmove(c, a, s, d, t):
    """Apply single transfer s->d with amount t to convolution c. Returns new c."""
    n = a.size
    cn = c.copy()
    cn[d:d+n] += 2.0 * t * a
    cn[s:s+n] -= 2.0 * t * a
    t2 = t * t
    if 2*d < cn.size:
        cn[2*d] += t2
    if 2*s < cn.size:
        cn[2*s] += t2
    if s+d < cn.size:
        cn[s+d] -= 2.0 * t2
    return np.where(np.isfinite(cn), cn, 0.0)

Actually, the pressure at position i is:
`p[i] = sum over top k of (a[k-i] + a[i-k]) / 2` (for valid indices)

Wait, the derivative of c[k] = sum_j a[j] * a[k-j] with respect to a[i] is:

`q` is a vector of length 2n-1 with q[k] = 1 for top peaks, 0 otherwise.

Then `p[i] = sum_k q[k] * a[k-i]` (for valid k-i) + `sum_k q[k] * a[i-k]` (for valid i-k)

The first sum is a cross-correlation: `(q * a)[i] = sum_k q[k] a[k-i]` (this is the convolution of q[::-1] with a, or cross-correlation).

The second sum: `sum_k q[k] a[i-k] = sum_j q[i+j] a[j]` = `(a * q)[i]` (convolution).
