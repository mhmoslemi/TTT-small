    """Find direction g using multiplicative weights on the dual."""
    S = sorted(active_set)
    m = len(S)
    
    # Initialize lambda uniformly
    lam = np.ones(m) / m
    
    # Precompute: for each k in S, the vector A_k (row of A)
    # A_k[i] = y[k-i] if 0 <= k-i < n, else 0
    # We need: v[i] = sum_{k in S} lam[k] * y[k-i]
    # This is cross-correlation of y and w (where w[k] = lam for k in S)
    
    # Build w (length 2n-1)
    w = np.zeros(2*n-1)
    for idx, k in enumerate(S):
        w[k] = lam[idx]
    
    # FFT size
    fft_size = 1 << (3*n - 3).bit_length()
    
    for t in range(T_inner):
        # Compute v = cross_corr(y, w)
        Y = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        v = np.fft.irfft(np.conj(Y) * W, fft_size)[:n]
        
        # Find bottleneck
        i_star = int(np.argmin(v))
        nu = v[i_star]

        # Update: lam[k] *= (1 + eta * A_{k, i_star})

            j = k - i_star
            if 0 <= j < n:
                updates[idx] = y[j]
        
        lam = lam * (1.0 + eta * updates)
        lam = lam / lam.sum()
        
        # Update w
        w = np.zeros(2*n-1)
        for idx, k in enumerate(S):

    g = np.fft.irfft(np.conj(Y) * W, fft_size)[:n]
    g = np.maximum(g, 0)
    sg = g.sum()
    if sg < 1e-15:
        return None
    return g / sg

import numpy as np
import time
import cvxpy as cp

# === Helper functions ===

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_val(a):
    """Compute 2*n*max(autocorr) for normalized a."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except:
        return float('inf')

def cross_corr_fft(y, w, n):
    """
    Compute v[i] = sum_j y[j] * w[i+j] for i=0,...,n-1
    where y has length n and w has length 2n-1.
    Uses FFT.
    """
    fft_size = 1 << (3 * n - 3).bit_length()
    Y = np.fft.rfft(y, fft_size)
    W = np.fft.rfft(w, fft_size)
    return np.fft.irfft(np.conj(Y) * W, fft_size)[:n]

def mwa_direction(y, active_set, n, T_inner=300, eta=1.0, warm_start=None):
    """
    Find direction g using multiplicative weights on the dual LP.
    
    The dual of (min M s.t. A@G <= M, 1^T G = 1, G >= 0) is:
    (max nu s.t. A^T lambda >= nu, 1^T lambda = 1, lambda >= 0)
    
    We use MWA to solve this dual approximately.
    Returns: direction g (normalized, nonnegative) or None
    """
    S = np.array(sorted(active_set), dtype=np.int64)
    m = len(S)
    if m == 0:
        return None
    
    fft_size = 1 << (3 * n - 3).bit_length()
    
    # Initialize lambda
    if warm_start is not None and len(warm_start) == m:
        lam = warm_start.copy()
        lam = np.maximum(lam, 1e-30)
        lam = lam / lam.sum()
    else:
        lam = np.ones(m) / m
    
    # Build w vector
    w = np.zeros(2 * n - 1, dtype=np.float64)
    w[S] = lam
    
    # Y (precompute)
    Y = np.fft.rfft(y, fft_size)
    
    # Track frequencies of i*
    freq = np.zeros(n, dtype=np.float64)
    
    for t in range(T_inner):
        # Compute v = cross_corr(y, w)
        W = np.fft.rfft(w, fft_size)
        v = np.fft.irfft(np.conj(Y) * W, fft_size)[:n]
        
        # Find bottleneck
        i_star = int(np.argmin(v))
        freq[i_star] += 1.0
        
        # Update lambda: lam[k] *= (1 + eta * A_{k, i_star})
        # A_{k, i_star} = y[k - i_star] if 0 <= k - i_star < n
        j = S - i_star
        mask = (j >= 0) & (j < n)
        updates = np.zeros(m, dtype=np.float64)
        if mask.any():
            updates[mask] = y[j[mask]]
        
        lam = lam * (1.0 + eta * updates)
        lam_sum = lam.sum()
        if lam_sum < 1e-30:
            lam = np.ones(m) / m
        else:
            lam = lam / lam_sum
        
        # Update w
        w[:] = 0.0
        w[S] = lam
    
    # Compute direction g from empirical frequencies
    g = freq
    g = np.maximum(g, 0.0)
    sg = g.sum()
    if sg < 1e-15:
        # Fallback: use A^T @ lam
        W = np.fft.rfft(w, fft_size)
        g = np.fft.irfft(np.conj(Y) * W, fft_size)[:n]
        g = np.maximum(g, 0.0)
        sg = g.sum()
        if sg < 1e-15:
            return None
    
    g = g / sg
    return g, lam

def smooth_gradient_step(a, K_temp, lr, n_top=200):
    """
    One step of smooth mirror descent.
    Returns updated a.
    """
    n = len(a)
    ac = fast_autocorr(a)
    
    # Temperature weighting (softmax over top lags)
    idx = np.argsort(ac)[::-1][:min(n_top, len(ac))]
    top_vals = ac[idx]
    # Softmax with temperature
    scaled = K_temp * (top_vals - top_vals[0])
    p = np.exp(scaled)
    p = p / p.sum()
    
    # Build weight vector
    w = np.zeros(2 * n - 1, dtype=np.float64)
    w[idx] = p
    
    # Gradient = 2 * cross_corr(a, w)
    grad = 2.0 * cross_corr_fft(a, w, n)
    
    # Mirror descent update
    log_a = np.log(a + 1e-30) - lr * grad
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    
    return a_new

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_seq(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

    Y = np.fft.rfft(y, fft_size)

def mwa_direction(y, S_arr, n, T_inner, eta, warm_lam=None, fft_size=None):
    m = len(S_arr)
    if m == 0:
        return None, None
    if fft_size is None:
        fft_size = 1 << (3 * n - 3).bit_length()
    if warm_lam is not None and len(warm_lam) == m:
        lam = np.maximum(warm_lam.copy(), 1e-30)
        lam = lam / lam.sum()
    else:
        lam = np.ones(m, dtype=np.float64) / m
    w = np.zeros(2 * n - 1, dtype=np.float64)
    w[S_arr] = lam
    Y = np.fft.rfft(y, fft_size)
    freq = np.zeros(n, dtype=np.float64)
    for t in range(T_inner):
        W = np.fft.rfft(w, fft_size)
        v = np.fft.irfft(np.conj(Y) * W, fft_size)[:n]
        i_star = int(np.argmin(v))
        freq[i_star] += 1.0
        j = S_arr - i_star
        mask = (j >= 0) & (j < n)
        updates = np.zeros(m, dtype=np.float64)
        if mask.any():
            updates[mask] = y[j[mask]]
        lam = lam * (1.0 + eta * updates)
        lam_sum = lam.sum()
        if lam_sum < 1e-30:
            lam = np.ones(m, dtype=np.float64) / m
        else:
            lam = lam / lam_sum
        w[:] = 0.0
        w[S_arr] = lam
    g = freq.copy()
    sg = g.sum()
    if sg < 1e-15:
        w[S_arr] = lam
        W = np.fft.rfft(w, fft_size)
        g = np.fft.irfft(np.conj(Y) * W, fft_size)[:n]
        g = np.maximum(g, 0.0)
        sg = g.sum()
        if sg < 1e-15:
            return None, None
    g = g / sg
    return g, lam
