Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:

import numpy as np

import scipy.sparse as sp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, n=fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def lp_direction(f, lags, M, n):
    """Solve LP to find FW direction. Returns g or None."""
    K = len(lags)
    if K == 0:
        return None
    
    # Build constraint matrix
    rows_list = []
    cols_list = []
    vals_list = []
    for jj, k in enumerate(lags):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            v = f[k - idx]
            mask = v > 1e-12
            if mask.any():
                rows_list.append(np.full(idx[mask].size, jj, dtype=np.int32))
                cols_list.append(idx[mask].astype(np.int32))
                vals_list.append(v[mask])
    
    if not rows_list:
        return None
    
    rows_arr = np.concatenate(rows_list)
    cols_arr = np.concatenate(cols_list)
    vals_arr = np.concatenate(vals_list)
    
    A_conv = sp.csr_matrix((vals_arr, (rows_arr, cols_arr)), shape=(K, n))
    
    # Add the sum(g) <= 1 constraint
    ones_row = sp.csr_matrix(np.ones((1, n)))
    A_full = sp.vstack([A_conv, ones_row], format='csr')
    b_full = np.concatenate([np.full(K, M), [1.0]])
    
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_full, b_ub=b_full, 
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    
    return None

    iter_count = 0

        ac = fast_autocorr(f)
        M = ac.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val - 1e-10:
            best_val = cur_val
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update active set: add top violations, remove least constraining
        # Add new peaks that are close to M
        close_peaks = np.where(ac > M * 0.95)[0]
        for k in close_peaks:
            if k not in active:
                active.append(int(k))
        
        # Keep only the most constraining ones
        if len(active) > max_K:
            # Remove lags with smallest "margin" (M - ac[k])
            margins = np.array([M - ac[k] for k in active])
            keep = np.argsort(-margins)[:max_K]  # keep those with largest margin (most constraining)

                continue
