gradient of (a*a)[k] w.r.t. a[i] = a[k-i] + a[i-k] (for valid indices)

So the subgradient at k* is:
∂f/∂a[i] = a[k*-i] + a[i-k*] (where defined)

We can do projected subgradient descent:
a_new = project_simplex(a - eta * subgradient)

The key question is: how to handle the case where multiple lags are near the maximum? We can use the softmax weighted average of subgradients (which is what the previous code does with the smoothed gradient).

But there's a subtlety: the "projected subgradient" method is known to converge slowly. Nesterov acceleration helps, which is what the previous code uses.

**Alternative key idea: "Exact line search along subgradient directions"**

Instead of using a fixed or adaptive step size, perform exact line search:
1. Compute the subgradient g at the current peak lag k*
2. Consider the line a(t) = project(a - t*g)
3. Find t that minimizes max_k (a(t)*a(t))[k]
4. This is a 1D optimization problem that can be solved efficiently

This is different from the previous approach which uses backtracking.

**Another idea: "Lag-interleaved optimization"**

Instead of always focusing on the global maximum lag, cycle through the top-K lags:
1. For k = argmax (a*a): reduce (a*a)[k]
2. For k = 2nd argmax: reduce (a*a)[k]
3. ...
This helps because reducing one lag might increase another.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()

    if b_max <= 0:
        return np.zeros(n), b

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, b


def nesterov_improved(a0, time_limit, beta_start=15.0, beta_end=1200.0,
                       beta_growth=1.004, eta_init=0.4, top_k_ratio=0.025,
                       restart_every=1200, verbose=True, seed=42):
    """Nesterov with exact line search on smoothed objective."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = compute_smoothed_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)
        y_max_b = b.max()

        if gnorm < 1e-15:
            x_new = x.copy()
        else:
            # Exact line search over a range of step sizes
            best_trial_max = y_max_b
            best_x_new = x.copy()
            candidates = [eta, eta * 0.7, eta * 0.5, eta * 0.3,
                          eta * 1.3, eta * 1.7, eta * 2.0, 0.01, 0.1]
            for et in candidates:
                if et <= 0 or et > 5.0:
                    continue
                z = y - et * grad
                trial = simplex_project(z)
                m_trial = fast_autocorr(trial).max()
                if m_trial < best_trial_max:
                    best_trial_max = m_trial
                    best_x_new = trial
            x_new = best_x_new

        max_new = fast_autocorr(x_new).max()
        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 3.0)
        else:
            stall += 1
            if stall % 50 == 0:
                eta = max(eta * 0.95, 0.05)

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 40 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                noise_level = 0.015 + 0.01 * rng.random()
                pert = rng.normal(0, noise_level, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.2)
            elif strategy == 2:
                x_r = best_a * np.exp(0.025 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= (0.6 + 0.4 * rng.random())
                x_r[~mask] *= (1.0 + 0.6 * rng.random())
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 15)))
                x_r = np.roll(best_a, shift)
            else:
                # Asymmetric tilt perturbation
                tilt = rng.uniform(-0.3, 0.3)
                tilt_vec = 1.0 + tilt * np.linspace(-1, 1, n)
                x_r = best_a * tilt_vec
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " b=" + str(int(beta)))

    return best_a, best_max_b


def lp_fw_aggressive(a0, time_limit, solver, initial_K=200, max_K=1500,
                     add_per_iter=30, verbose=True):
    """LP-Frank-Wolfe with aggressive cutting-plane active set and extrapolation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 400

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check for violations and add to active set
        size = 2 * n - 1
        fft_size_chk = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size_chk)
        Gf = np.fft.rfft(g_dir, fft_size_chk)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size_chk)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Adaptive line search with extrapolation
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                  1.0 / (t + 1), 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
        for alpha in alphas:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def peak_chasing_moves(a0, time_limit, verbose=True, seed=99):
    """Targeted moves that specifically reduce the top lags of the convolution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 5000:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        top_lags = np.argsort(ac)[::-1][:min(10, size)]
        improved = False

        for k in top_lags[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find the source index that contributes most to this lag
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            if not (0 <= partner < n):
                continue

            # Try moving mass from src to various destinations
            # that don't contribute to this lag
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue

                # Try destinations that are "safe" (don't add to top lags)
                n_tries = 8
                for trial_i in range(n_tries):
                    if rng.random() < 0.3:
                        # Pick destination far from src and partner
                        dst = int(rng.integers(n))
                    else:
                        # Pick destination in a "quiet" region
                        dst = int(rng.integers(n))

                    if dst == src or dst == partner:
                        continue

                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-18:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 100 and stall % 10 == 0:
                # Random exploration
                n_moves = rng.integers(2, 5)
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = rng.uniform(0.003, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if verbose and stall % 1000 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakChase (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def spectral_pinch(a0, time_limit, verbose=True, seed=555):
    """Iterative spectral pinching: reduce dynamic range of |A(f)|^2."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.03
    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Compute target: geometric mean of non-DC power
        non_dc = power[1:]
        if len(non_dc) == 0:
            break
        log_power = np.log(np.maximum(non_dc, 1e-300))
        target = np.exp(np.mean(log_power))

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * min(ratio - 1.0, 2.0)
                A_new[k] *= max(0.2, np.sqrt(red))
            elif ratio < 0.3:
                inc = 1.0 + alpha * 0.2 * (0.3 - ratio)
                A_new[k] *= min(1.5, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 300 == 0:
            # Random phase perturbation to escape local minima
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.03
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 400 == 0:
            alpha = max(0.003, alpha * 0.94)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SpectralPinch it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def make_asymmetric_init(n, tilt=0.7, seed=0):
    """Create an asymmetric initialization with a tilted window."""
    x = np.linspace(0.0, 1.0, n)
    # Base: sine window
    base = np.sin(np.pi * x) ** 2
    # Tilt: linear ramp
    ramp = 1.0 + tilt * (2.0 * x - 1.0)
    ramp = np.maximum(ramp, 0.01)
    a = base * ramp
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization with asymmetric initialization,
    aggressive LP-FW, peak chasing, and spectral pinching.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        pass

    # Also try asymmetric initializations
    if best_seq is None:
        n_base = 6000
        best_seq = norm_sum1(np.sin(np.pi * np.linspace(0, 1, n_base)) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(round(best_score, 6)))

    # Try asymmetric variants of parent (or default)
    n0 = len(best_seq)
    best_base = best_seq.copy()

    # Create and evaluate asymmetric variants
    for tilt in [0.3, 0.5, 0.7, 1.0, 1.3]:
        if tilt == 0.3 and parent is not None:
            # Use parent as-is (already evaluated)
            continue
        a_asym = make_asymmetric_init(n0, tilt=tilt)
        sc = evaluate_safe(a_asym)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_base = a_asym.copy()
            print("Asymmetric init tilt=" + str(tilt) + ": J=" + repr(round(sc, 6)))

    # Also try parent with small asymmetric perturbation
    if parent is not None:
        x = np.linspace(0, 1, n0)
        for tilt in [0.2, 0.4, -0.3, -0.5]:
            ramp = 1.0 + tilt * (2.0 * x - 1.0)
            ramp = np.maximum(ramp, 0.1)
            a_pert = norm_sum1(best_base * ramp)
            sc = evaluate_safe(a_pert)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_base = a_pert.copy()
                print("Perturbed parent tilt=" + str(tilt) + ": J=" + repr(round(sc, 6)))

    best_seq = best_base.copy()
    print("Start: n=" + str(n0) + ", J=" + repr(round(best_score, 6)))

    # Phase 1: Quick spectral pinch
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Spectral Pinch (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 15:
        a_out, _ = spectral_pinch(norm_sum1(best_seq), t1, verbose=True, seed=111)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 2: Nesterov (main optimizer)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: Nesterov (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 50:
        a_out, _ = nesterov_improved(norm_sum1(best_seq), t2,
                                     beta_start=15.0, beta_end=1000.0,
                                     beta_growth=1.004, eta_init=0.4,
                                     top_k_ratio=0.025, restart_every=1200,
                                     verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: Aggressive LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: LP-FW Aggressive (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 30 and n_best <= 9000:
        a_out, _ = lp_fw_aggressive(norm_sum1(best_seq), t3, solver,
                                    initial_K=250, max_K=1500,
                                    add_per_iter=30, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 4: Peak chasing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: PeakChase (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 15:
        a_out, _ = peak_chasing_moves(norm_sum1(best_seq), t4, verbose=True, seed=99)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "peak1", t_global)

    # Phase 5: Second Nesterov round
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: Nesterov r2 (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t5 = min(remaining * 0.20, 250.0)
        if t5 > 30:
            a_out, _ = nesterov_improved(norm_sum1(best_seq), t5,
                                         beta_start=30.0, beta_end=1500.0,
                                         beta_growth=1.005, eta_init=0.3,
                                         top_k_ratio=0.015, restart_every=1000,
                                         verbose=True, seed=777)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Multi-length (base n=" + str(n_best) + ") ===")

    longer_lengths = [n_best + 300, n_best + 600, n_best + 1200]
    longer_lengths = [l for l in longer_lengths if l <= 10000 and l > n_best]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print("  -> n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick Nesterov on longer
        t6a = min(remaining * 0.35, 150.0)
        if t6a > 25:
            a_out, _ = nesterov_improved(a_long, t6a,
                                         beta_start=20.0, beta_end=800.0,
                                         beta_growth=1.004, eta_init=0.35,
                                         top_k_ratio=0.02, restart_every=1000,
                                         verbose=True, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        # LP on longer
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 8500:
            t6b = min(remaining * 0.40, 130.0)
            if t6b > 20:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = lp_fw_aggressive(norm_sum1(a_long2), t6b, solver,
                                            initial_K=200, max_K=1200,
                                            add_per_iter=25, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

        # Spectral on longer
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6c = min(remaining * 0.25, 60.0)
            if t6c > 10:
                a_out, _ = spectral_pinch(norm_sum1(best_seq), t6c, verbose=False, seed=new_n % 3163)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_n" + str(new_n), t_global)

    # Phase 7: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: Final LP (n=" + str(n_best) + ") ===")
    if remaining > 30 and n_best <= 9000:
        t7 = min(remaining * 0.30, 150.0)
        if t7 > 20:
            a_out, _ = lp_fw_aggressive(norm_sum1(best_seq), t7, solver,
                                        initial_K=350, max_K=2000,
                                        add_per_iter=35, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 8: Final Nesterov polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P8: Final Nest (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t8 = min(remaining * 0.35, 120.0)
        if t8 > 15:
            a_out, _ = nesterov_improved(norm_sum1(best_seq), t8,
                                         beta_start=100.0, beta_end=2000.0,
                                         beta_growth=1.006, eta_init=0.2,
                                         top_k_ratio=0.01, restart_every=800,
                                         verbose=True, seed=333)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 9: Final peak chasing
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t9 = min(remaining * 0.30, 60.0)
        if t9 > 10:
            a_out, _ = peak_chasing_moves(norm_sum1(best_seq), t9, verbose=False, seed=222)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "peak_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
