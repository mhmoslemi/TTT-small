def fast_conv(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def simplex_project(z):
    """Project z onto the probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Let me also think about a key improvement: **the LP should use the CURRENT M (max of current convolution) as the threshold, not a fixed M**. This way, as we reduce the peaks, the LP becomes easier and can make finer adjustments.

Actually, looking at the previous code more carefully, it does use M_curr = b.max() as the threshold. Good.

Let me think about another approach: **mirror descent / entropic optimization**.

Instead of projecting onto the simplex, use the multiplicative weights update:
a_new[i] = a[i] * exp(-eta * grad[i]) / Z

where grad[i] is the subgradient of max(autocorr) and Z is the normalization.

This naturally keeps a >= 0 and can be more efficient than simplex projection.

The subgradient of max_k (a*a)[k] with respect to a[i] is:
If k* = argmax_k (a*a)[k], then:
grad[i] = a[k*-i] (if 0 <= k*-i < n) [for the term a[i]*a[k*-i]]
         + a[i] (if k* - i == i, i.e., the quadratic term) [actually this is already counted]

More carefully: (a*a)[k] = sum_j a[j]*a[k-j]. The derivative w.r.t. a[i] is:
d(a*a)[k]/da[i] = a[k-i] (if 0 <= k-i < n) + (a[k-i] if i != k-i, else a[k-i]) 

Wait, let me be more careful:
(a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]

d/d(a[i]) of this: the term a[i]*a[k-i] appears when j=i (giving a[k-i]) and when k-j=i i.e. j=k-i (giving a[i]).
If i != k-i, these are two different terms, so the derivative is a[k-i] + a[k-i] = 2*a[k-i].
If i == k-i (i.e., k = 2i), it's the same term a[i]^2, so the derivative is 2*a[i].

So d(a*a)[k]/da[i] = 2*a[k-i] (when 0 <= k-i < n), with the understanding that if k=2i, it's 2*a[i].

The subgradient of max_k (a*a)[k] is the gradient of (a*a)[k*] where k* is an argmax.
For a smoothed version: grad[i] = sum_k p[k] * 2*a[k-i] (where p is the softmax of beta*(a*a)[k]).

This is a convolution! grad = 2 * (p * a_reversed) restricted to [0, n-1].

This can be computed efficiently with FFT.

So the mirror descent update would be:
1. Compute b = a*a
2. Compute p = softmax(beta * b) (or top-k version)
3. Compute grad via FFT
4. Update: a_new = a * exp(-eta * grad) (normalized)
5. Project to simplex (or just normalize)

This is essentially what the previous subgradient approach does, but with the mirror descent step instead of the Euclidean step. Mirror descent is better for the simplex constraint because the Bregman divergence for the simplex is the entropy, which gives the multiplicative update.

Let me combine both approaches:
- **Phase 1**: Frank-Wolfe (LP-based) for global progress
- **Phase 2**: Mirror descent / multiplicative weights for smooth refinement
- **Phase 3**: Surgical peak reduction for final improvements

Let me now write the complete code:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Refined Frank-Wolfe (LP-based conditional gradient) with adaptive active set
2. Entropic/mirror descent with smoothed max for smooth refinement  
3. Direct peak surgery (mass redistribution from top peaks)
4. Multi-n exploration and interpolation
5. Final coordinate-wise clipping polish
"""
import numpy as np
import time
import cvxpy as cp

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z.copy()
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def frank_wolfe_refine(a0, time_limit, solver, verbose=True):
    """Refined Frank-Wolfe with adaptive active set and backtracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    t = 0
    no_improve = 0
    
    # Initial active set: top constraints
    b = fast_conv(a)
    K_init = min(2 * n - 1, 4000)
    active_set = set(int(k) for k in np.argsort(b)[::-1][:K_init])
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        
        b = fast_conv(a)
        M_curr = b.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Update active set
        # Add violated constraints
        violations = np.where(b > M_curr - 1e-3)[0]  # near-violated
        if len(violations) > 0:
            # Add the worst violations
            worst = violations[np.argsort(b[violations])[::-1][:min(50, len(violations))]]
            for k in worst:
                active_set.add(int(k))
        
        # Keep active set size bounded
        if len(active_set) > 6000:
            margins = np.array([M_curr - b[k] for k in active_set])
            # Keep the tightest (smallest margin) constraints
            keep_idx = np.argsort(margins)[:5000]
            active_set = set(int(list(active_set)[i]) for i in keep_idx)
        
        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix (sparse would be better but let's use dense for simplicity)
        # A[k, i] = a[k-i] if 0 <= k-i < n
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = k
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                coeffs[jj, idx] = a[k_int - idx]
        
        # Solve LP: max sum(g) s.t. coeffs @ g <= M_curr, g >= 0, sum(g) = 1
        g_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if no_improve > 100:
                break
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-10:
            continue
        g_dir = g_val / g_sum
        
        # Backtracking line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = float('inf')
        alphas = [2.0 / (t + 2.0), 1.0 / (t + 1), 0.5 * 2.0 / (t + 2.0), 0.1]
        for alpha in alphas:
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        st = a.sum()
        if st > 1e-15:
            a = a / st
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  FW t=%d (%.1fs): J=%.6f |S|=%d no_imp=%d" % (t, elapsed, J, len(active_set), no_improve))
    
    return best_a, best_m

def mirror_descent(a0, time_limit, beta_start=30.0, beta_end=500.0, verbose=True):
    """Entropic/mirror descent with smoothed max autorelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    t = 0
    beta = beta_start
    stall = 0
    rng = np.random.default_rng(999)
    top_k = max(50, n // 10)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_conv(a)
        b_max = b.max()
        
        if b_max < best_m - 1e-14:
            best_m = b_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute smoothed weights
        idx_top = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(2 * n - 1)
        b_use[idx_top] = b[idx_top]
        shifted = np.clip(beta * (b_use - b_max), -300, 300)
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()
        
        # Gradient: conv of p with reversed a
        conv_len = n + len(p) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        a_rev = a[::-1]
        Ar = np.fft.rfft(a_rev, fft_size)
        Pr = np.fft.rfft(p, fft_size)
        conv = np.fft.irfft(Ar * Pr, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        # Mirror descent update: a_new = a * exp(-eta * grad) / Z
        eta = 1.0 / (beta * (np.max(grad) - np.min(grad) + 1e-10))
        eta = min(eta, 0.1)
        log_w = -eta * grad
        log_w -= log_w.max()  # numerical stability
        w = np.exp(log_w)
        a_new = a * w
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        b_new = fast_conv(a_new)
        m_new = b_new.max()
        
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a  # keep old
            
            # Try smaller step
            for eta_try in [eta * 0.5, eta * 0.1]:
                log_w = -eta_try * grad
                log_w -= log_w.max()
                w = np.exp(log_w)
                a_try = a * w
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    m_try = fast_conv(a_try).max()
                    if m_try < best_m:
                        best_m = m_try
                        best_a = a_try.copy()
                        a = a_try
                        break
            
        # Update beta
        if t % 100 == 0:
            beta = min(beta * 1.02, beta_end)
        
        # Restart if stalled
        if stall > 500:
            stall = 0
            noise = rng.normal(0, 0.01, n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-12)
            a_r = a_r / a_r.sum()
            a = a_r
            beta = max(beta * 0.7, 10.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("  MD restart t=%d (%.1fs): J=%.6f" % (t, elapsed, J))
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  MD t=%d (%.1fs): J=%.6f beta=%.1f stall=%d" % (t, elapsed, J, beta, stall))
    
    return best_a, best_m

def peak_surgery(a0, time_limit, verbose=True):
    """Directly reduce the top autocorrelation peaks by surgical mass moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        b = fast_conv(a)
        m_curr = b.max()
        
        if m_curr < best_m - 1e-14:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find top peaks
        top_peaks = np.argsort(b)[::-1][:15]
        
        improved = False
        for k_int in top_peaks[:8]:
            k_i = int(k_int)
            i_start = max(0, k_i - n + 1)
            i_end = min(n - 1, k_i)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_i - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the pair with maximum contribution
            src = int(idx[np.argmax(contrib)])
            partner = k_i - src
            
            if partner < 0 or partner >= n:
                continue
            
            # Decide which to reduce (the larger one)
            if a[src] >= a[partner]:
                reduce_idx = src
            else:
                reduce_idx = partner
            
            if a[reduce_idx] < 1e-14:
                continue
            
            # Find safe destination: minimize increase in all peaks
            # Approximate: look at the "pressure" on each position
            dst_candidates = []
            for d in range(n):
                if d == reduce_idx:
                    continue
                # Estimate the increase in max if we move mass to d
                # (crude: check a few peak positions)
                max_increase = 0.0
                for kp in top_peaks[:5]:
                    kp_i = int(kp)
                    if 0 <= kp_i - d < n:
                        max_increase = max(max_increase, a[kp_i - d])
                dst_candidates.append((max_increase, d))
            
            dst_candidates.sort()
            
            found = False
            for _, dst in dst_candidates[:20]:
                for frac in [0.01, 0.03, 0.06, 0.1, 0.15]:
                    amt = a[reduce_idx] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_conv(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
            
            if not improved and stall > 100:
                # Random move
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-12:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        m_new = fast_conv(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        
        if not improved:
            stall += 1
            if stall % 100 == 0 and stall < 300:
                # Periodic perturbation
                noise_scale = min(0.02, 0.003 * stall)
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_conv(a_r).max()
                if m_r < best_m:
                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r
            
            if verbose and stall % 100 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("  PS (%.1fs): J=%.6f stall=%d" % (elapsed, J, stall))
    
    return best_a, best_m

def coord_clipping(a0, time_limit, verbose=True):
    """Iteratively clip each coordinate to its maximum allowed value."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    for iteration in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_m - 1e-14:
            best_m = M
            best_a = a.copy()
        
        # For each i, compute max allowed a[i] given current M
        # (a*a)[k] = sum_j a[j]*a[k-j]
        # Terms involving a[i]: a[i]*a[k-i] for valid k-i
        # For k=2i: a[i]^2 + rest <= M => a[i] <= sqrt(M - rest)
        # For k != 2i: a[i]*a[k-i] + rest <= M => a[i] <= (M-rest)/a[k-i]
        
        changed = False
        for i in range(n):
            a_i_max = a[i]  # current value is always feasible
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                if k >= 2 * n - 1:
                    break
                j = k - i
                if j < 0 or j >= n:
                    continue
                b_k = b[k]
                term = a[i] * a[j]
                rest = b_k - term
                if rest > M:
                    # Already violated (shouldn't happen)
                    continue
                if k == 2 * i:
                    # a[i]^2 + rest <= M
                    if rest >= M:
                        a_i_max = 0.0
                    else:
                        a_i_max = min(a_i_max, np.sqrt(M - rest))
                else:
                    if a[j] > 1e-15:
                        if rest >= M:
                            a_i_max = 0.0
                        else:
                            a_i_max = min(a_i_max, (M - rest) / a[j])
            
            if a[i] > a_i_max + 1e-12:
                a[i] = max(a_i_max, 0.0)
                changed = True
        
        if changed:
            s = a.sum()
            if s < 1e-15:
                break
            # Don't normalize here - we want to preserve the feasibility
            # But we need sum to stay reasonable for the evaluation
            # Actually, we should normalize to keep J the same
            # Wait - clipping reduces a[i], which reduces sum and max_b
            # J = 2n*max_b/sum^2. If both decrease, the ratio might go either way.
            # Let's just check.
            b_new = fast_conv(a)
            m_new = b_new.max()
            s_new = a.sum()
            if s_new > 1e-15:
                J_new = 2 * n * m_new / (s_new ** 2)
                J_best = 2 * n * best_m / (norm_sum1(best_a).sum() ** 2)  # best_a is sum=1
                # Actually best_m is max of conv of norm_sum1(a), so J_best = 2n*best_m
                if J_new < 2 * n * best_m - 1e-10:
                    # This means the un-normalized version is better
                    # But we should normalize for consistency
                    a_norm = a / s_new
                    m_norm = fast_conv(a_norm).max()
                    if m_norm < best_m:
                        best_m = m_norm
                        best_a = a_norm.copy()
                    a = a_norm
    
        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  CC iter=%d (%.1fs): J=%.6f" % (iteration, elapsed, J))
    
    return best_a, best_m

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score
