Actually, the standard LP-FW (Frank-Wolfe) for this problem works as follows:
- At the current point $y$ (on the simplex), we want to find a direction $g$ on the simplex that decreases the objective.
- The linear minimization oracle for $f(y) = \max_k b_k$ at point $y$ is:
  $\min_g \max_k [b_k(y) + \nabla b_k(y) \cdot (g - y)]$
  $= \min_g \max_k [\nabla b_k(y) \cdot g + (b_k(y) - \nabla b_k(y) \cdot y)]$

  Since $b_k$ is quadratic, $\nabla b_k(y) \cdot y = 2b_k(y)$ (by Euler's theorem for homogeneous functions: $b_k$ is degree 2, so $\nabla b_k \cdot y = 2b_k$).

  So the oracle problem is:
  $\min_g \max_k [\nabla b_k(y) \cdot g - b_k(y)]$

  Which can be written as:
  $\min_{g, M} M$
  s.t. $\nabla b_k(y) \cdot g - b_k(y) \leq M$ for all $k$
  $\sum g_i = 1$, $g_i \geq 0$

This is what the code does. The active set contains the $k$ values where the constraint is (nearly) binding.

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

    if s < 1e-15:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_coeff_rows(y, active_set, n):

        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = 2.0 * y[k_int - idx]
    return coeffs

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_n)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(55)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        k_star = int(np.argmax(ac))
        M_peak = ac[k_star]

        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start >= i_end:
            break

        idx = np.arange(i_start, i_end + 1)
        products = a[idx] * a[k_star - idx]
        max_prod_idx = int(idx[np.argmax(products)])
        max_prod_val = products.max()
        if max_prod_val < 1e-20:
            break

        partner = k_star - max_prod_idx

        scores = np.zeros(n)
        top_k = np.argsort(ac)[::-1][:12]
        for kk in top_k:
            k2 = int(kk)
            s2 = max(0, k2 - n + 1)
            e2 = min(n - 1, k2)
            if s2 <= e2:
                idx2 = np.arange(s2, e2 + 1)
                scores[idx2] += a[k2 - idx2]

        candidates = np.where(scores < scores.mean())[0]
        if len(candidates) < n // 4:
            candidates = np.arange(n)
        if max_prod_idx in candidates:
            cand_mask = candidates != max_prod_idx
            candidates = candidates[cand_mask]
        if len(candidates) == 0:
            candidates = np.arange(n)

        dst = int(candidates[np.argmin(scores[candidates])])
        if dst == max_prod_idx:
            stall += 1
            continue

        found = False
        for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]:
            amt = a[max_prod_idx] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[max_prod_idx] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_convolve(a_trial).max()
            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                found = True
                stall = 0
                break

        if not found:
            j = int(rng.integers(n))
            if j != max_prod_idx and a[max_prod_idx] > 1e-10:
                amt = a[max_prod_idx] * rng.uniform(0.002, 0.01)
                a[max_prod_idx] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_convolve(a).max()
                    if M_new < best_m - 1e-15:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        stall += 1
            else:
                stall += 1

        if verbose and stall % 400 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print("    PeakShave (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(2.0 * n * best_m) + ", stall=" + str(stall))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    PeakShave done (" + str(round(elapsed, 1)) + "s): J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) +
              ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score
