def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        # Manage active set
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)
        elif len(constraint_list) < min(initial_K, 2 * n - 1) and t > 100:
            # Add some new constraints
            all_indices = np.arange(2 * n - 1)
            remaining = np.setdiff1d(all_indices, np.array(list(active_set)))
            if len(remaining) > 0:
                top_candidates = remaining[np.argsort(ac[remaining])[::-1][:20]]
                for k in top_candidates[:5]:
                    active_set.add(int(k))
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Fallback: reduce active set and retry
            active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            g_val = np.ones(n) / n
            g_sum = 1.0
        g_dir = g_val / g_sum
        g_warm = g_dir.copy()

        # Add violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0 and len(active_set) < max_K:
            viol_amt = conv_ys[violations] - M_curr
            n_add = min(30, len(violations))
            top_v = violations[np.argsort(viol_amt)[::-1][:n_add]]
            for k in top_v:
                active_set.add(int(k))

        # Line search: try multiple step sizes
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                  1.0 / (t + 1), 0.02, 0.05, 0.1, 0.2, 0.3]
        for alpha in alphas:
            alpha = max(0.005, min(0.8, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)) + " alpha=" + str(round(best_alpha, 4)))

    return best_a, best_m


def peak_coordinate_descent(a0, time_limit, verbose=True, seed=42):
    """
    Targeted coordinate descent: for each top peak, find the best
    single-coordinate adjustment to reduce it.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:min(30, size)]

        improved = False
        for k in top_peaks[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            # For this peak, try adjusting individual coordinates
            idx = np.arange(js, je + 1)
            # Contribution of a[i] to peak k: a[i] * sum_{j: k-j valid, j != i} a[j]
            # + 2*a[i]^2 if k=0 (but k=some lag, so it's a[i]*a[k-i])
            
            # Try moving mass from each position in the peak to a random position
            for i in idx[:min(20, len(idx))]:
                i_int = int(i)
                if a[i_int] < 1e-12:
                    continue
                partner = k_int - i_int
                if partner == i_int or not (0 <= partner < n):
                    continue

                # Try several fractions
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                    amt = a[i_int] * frac
                    if amt < 1e-15:
                        continue

                    # Move to a position that doesn't contribute to this peak
                    # and doesn't create new peaks
                    for _ in range(8):
                        dst = int(rng.integers(n))
                        # Check if dst would contribute to another peak
                        a_trial = a.copy()
                        a_trial[i_int] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-18:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Random perturbation to escape
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial / s_t
                            M_trial = fast_autocorr(a_trial).max()
                            if M_trial < best_m:
                                best_m = M_trial
                                best_a = a_trial.copy()
                                a = a_trial
                                stall = 0
                                improved = True
                                break
                if not improved:
                    # Accept a slightly worse move to escape
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * 0.005
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                    if stall > 500:
                        break

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordDescent (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m
