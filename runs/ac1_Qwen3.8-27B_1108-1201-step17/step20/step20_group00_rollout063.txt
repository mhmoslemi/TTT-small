def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    """Objective: 2*n*max_b (when sum=1)."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, lags, n):
    """Build the LP constraint matrix for given lags."""
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def spectral_equalize(a0, time_limit, n_iter=5000, gamma=0.3, verbose=True):
    """
    Flatten the power spectrum in frequency domain to reduce autocorrelation peaks.
    Works by reducing the magnitude of high-frequency components that cause peaks.

    a = normalize(a0)

    best_val = obj_val(a)

        if stall > 200:

        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_val / (2.0 * n) - 1e-14:
            best_val = obj_val(a)
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Get the DFT
        A = np.fft.rfft(a, n=fft_size)
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        
        # The autocorrelation peaks correspond to specific frequency content
        # Strategy: reduce the overall "peakedness" of |A|^2
        # By slightly compressing the dynamic range of |A|^2
        S = A_mag ** 2
        S_mean = S.mean()
        if S_mean < 1e-20:
            break
        S_norm = S / S_mean

        A_new = A_mag_new * np.exp(1j * A_phase)
        
        # Inverse transform
        a_new = np.fft.irfft(A_new, n=fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        # Check improvement
        val_new = obj_val(a_new)
        if val_new < best_val + 0.01:  # Accept if not much worse
            a = a_new
        else:
            # Try smaller gamma
            a_trial = a.copy()
            break
        
        if verbose and it % 200 == 0:
            print("    SpecEq it=" + str(it) + " (" + str(round(time.monotonic()-t0, 1)) + "s) J=" + repr(best_val))
    
    return best_a, best_val

def spectral_shaping_v2(a0, time_limit, verbose=True):
    """
    More sophisticated spectral shaping:
    - Identify which frequency bins contribute most to autocorrelation peaks
    - Selectively reduce those while maintaining non-negativity
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    
    stall = 0
    gamma = 0.15  # initial compression strength
    for it in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 500:
            # Reset with perturbation
            rng = np.random.default_rng(it * 7 + 42)
            a = a + 0.01 * rng.standard_normal(n) * a
            a = normalize(a)
            stall = 0
            gamma = 0.15
            if time.monotonic() - t0 > time_limit:
                break
            continue
        
        # Compute DFT
        A = np.fft.rfft(a, n=fft_size)
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        
        # Compute autocorrelation to find peaks
        ac = fast_autocorr(a)
        M = ac.max()
        current_m = M
        
        if obj_val(a) < best_val - 1e-12:
            best_val = obj_val(a)
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Identify top peak lags
        top_lags = np.argsort(ac)[::-1][:20]
        
        # For each top lag, determine which frequency bins contribute most
        # conv(a,a)[k] = IDFT(|A|^2)[k]
        # Contribution of bin f to lag k: |A[f]|^2 * exp(-2j*pi*f*k/fft_size)
        # We want to reduce |A[f]|^2 for bins where the phase alignment causes peaks
        
        # Simpler approach: just compress the overall spectrum
        S = A_mag ** 2
        S_max = S.max()
        if S_max < 1e-20:
            break
        
        # Adaptive gamma: stronger compression when stalled
        eff_gamma = gamma * (1.0 + 0.1 * min(stall / 100.0, 2.0))
        eff_gamma = min(eff_gamma, 0.6)
        
        S_ref = S_max
        S_new = S_ref * np.power(S / S_ref, 1.0 - eff_gamma)
        
        A_mag_new = np.sqrt(S_new)
        A_new = A_mag_new * np.exp(1j * A_phase)
        a_new = np.fft.irfft(A_new, n=fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)
        
        val_new = obj_val(a_new)
        if val_new < current_m * 2 * n:  # accept
            a = a_new
            # Decrease gamma as we converge
            gamma = max(0.03, gamma * 0.995)
        else:
            stall += 5  # extra penalty for rejection
    
    return best_a, best_val

def grad_mult_v2(a0, time_limit, top_k=80, temp=50.0, c0=3.0, verbose=True):
    """Enhanced multiplicative gradient descent."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    x_prev = x.copy()
    stall = 0
    cur_temp = temp
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 15000:
            break
        
        # Nesterov-like momentum
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute top peaks
        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        
        # Adaptive temperature
        cur_temp = temp * max(0.2, 1.0 - t / 80000.0)
        w = np.exp(cur_temp * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        # Compute gradient contribution
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * y[ki - idx_arr]
        
        # Adaptive step size with backtracking
        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)
        
        # Try a few step sizes
        best_step_val = best_val
        best_new_x = x
        for step_mult in [1.0, 0.5, 2.0, 0.25]:
            eta_trial = eta_t * step_mult
            log_a = np.log(y) - eta_trial * grad
            max_log = log_a.max()
            a_trial = np.exp(log_a - max_log)
            a_trial = np.maximum(a_trial, 1e-30)
            a_trial = a_trial / a_trial.sum()
            val_trial = obj_val(a_trial)
            if val_trial < best_step_val:
                best_step_val = val_trial
                best_new_x = a_trial
        
        x_prev = x
        x = best_new_x
        
        if obj_val(x) < best_val - 1e-12:
            best_val = obj_val(x)
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000 and stall % 5000 == 0:
            noise_scale = 0.01 + 0.03 * (stall / 25000.0)
            noise = noise_scale * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            stall = 0
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    GradMult2 t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val))
    
    return best_a, best_val

def lp_fw_v2(a0, time_limit, solver, init_K=500, max_K=5000, add_per=50, verbose=True):
    """LP-based Frank-Wolfe with improved active set management."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    # Initialize active set with top peaks
    ac_init = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_init)[::-1][:init_K])
    no_improve = 0
    stall_limit = 2000
    
    size = 2 * n - 1
    fft_size_lp = 1 << (size - 1).bit_length()
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Manage active set size
        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)
        
        # Also ensure we have the current top peaks
        current_top = set(int(k) for k in np.argsort(ac)[::-1][:min(200, max_K // 10)])
        active_set = active_set | current_top
        
        K = len(clist) if len(clist) > 0 else 1
        if K == 0:
            break
        
        coeffs = build_lp_matrix(y, clist, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K:
                active_set = set(int(k) for k in clist[:init_K])
        
        if not lp_ok:
            if no_improve > stall_limit // 2:
                break
            if len(clist) > 200:
                active_set = set(int(k) for k in clist[:300])
                continue
            break
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check violations and expand active set
        Yf = np.fft.rfft(y, fft_size_lp)
        Gf = np.fft.rfft(g_dir, fft_size_lp)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size_lp)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))
        
        # Golden section line search
        lo = 0.01
        hi = 1.0
        if t <= 3:
            best_alpha = 1.0
        else:
            c1 = lo + inv_phi * (hi - lo)
            c2 = hi - inv_phi * (hi - lo)
            
            def trial_obj(alpha):
                trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
                ts = trial.sum()
                if ts < 1e-15:
                    return 1e10
                trial = trial / ts
                return fast_autocorr(trial).max()
            
            f1 = trial_obj(c1)
            f2 = trial_obj(c2)
            for _ in range(25):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_phi * (hi - lo)
                    f1 = trial_obj(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = hi - inv_phi * (hi - lo)
                    f2 = trial_obj(c2)
            
            best_alpha = (lo + hi) / 2.0
        
        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW2 t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)))
    
    if verbose:
        print("    LP-FW2 done (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def fast_coord_descent(a0, time_limit, verbose=True):
    """
    Fast coordinate descent using FFT to evaluate moves.
    Instead of full autocorrelation after each move, use the fact that
    moving a small amount from i to j changes the autocorrelation in a
    predictable way.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    max_stall = 2000
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        # Focus on top peaks
        n_top = min(40, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False
        
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Top contributing sources
            top_srcs = idx[np.argsort(contrib)[::-1][:5]]
            for src in top_srcs:
                src = int(src)
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                # Score destinations based on their "exposure" to all top peaks
                scores = np.zeros(n)
                for k2 in top_k[:10]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10
                
                # Try best destinations with multiple fractions
                candidates = np.argsort(scores)[:40]
                found = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break
                if found:
                    break
        
        if not improved:
            stall += 1
            if stall > 100:
                # Random exploration
                for _ in range(3):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        st = a.sum()
                        if st > 1e-15:
                            a = a / st
                            Mn = fast_autocorr(a).max()
                            if Mn < best_m:
                                best_m = Mn
                                best_a = a.copy()
                                stall = 0
                                break
        
        if verbose and stall % 500 == 0:
            print("    CoordFD (" + str(round(time.monotonic()-t0, 1)) + "s) J=" + repr(2*n*best_m) + " stall=" + str(stall))
    
    return best_a, 2.0 * n * best_m

def diverse_init(n, parent_seq=None):
    """Generate diverse initial candidates and return the best."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    candidates = []
    
    if parent_seq is not None:
        candidates.append(normalize(parent_seq))
    
    # Sine-based
    for p in [1, 2, 3, 4, 6]:
        candidates.append(normalize(np.sin(np.pi * x) ** p))
    
    # Gaussian bumps
    for c in [0.3, 0.5, 0.7]:
        for sg in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))
    
    # Double bumps
    for c1, c2 in [(0.25, 0.75), (0.3, 0.7), (0.35, 0.65)]:
        for sg in [0.08, 0.12, 0.18]:
            cands = np.exp(-((x - c1) ** 2) / (2 * sg * sg)) + np.exp(-((x - c2) ** 2) / (2 * sg * sg))
            candidates.append(normalize(cands))
    
    # Power-law tapered
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        cands = x ** p * (1 - x) ** p
        cands = np.maximum(cands, 1e-10)
        candidates.append(normalize(cands))
    
    # Cosine windows
    for f in [1, 2, 3, 4]:
        cands = np.maximum(1.0 + np.cos(2 * np.pi * f * x), 0.01)
        candidates.append(normalize(cands))
    
    # Tukey windows
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        cands = np.ones(n)
        m = int(alpha * n)
        if m > 0:
            taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
            cands[:m] = taper
            cands[-m:] = taper[::-1]
        candidates.append(normalize(cands))
    
    best_a = candidates[0]
    best_val = obj_val(best_a)
    for c in candidates[1:]:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()
    
    return best_a, best_val

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs


def trial_obj_fn(y, g_dir, alpha):
    trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
    ts = trial.sum()
    if ts < 1e-15:
        return 1e10
    trial = trial / ts
    return fast_autocorr(trial).max()


def spectral_shaping(a0, time_limit, verbose=True):
    """
    Flatten the power spectrum in frequency domain.
    Iteratively compresses |A|^2 dynamic range to reduce autocorrelation peaks.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    stall = 0
    gamma = 0.12
    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 800:
            rng = np.random.default_rng(it * 13 + 7)
            a = a + 0.02 * rng.standard_normal(n) * a
            a = normalize(a)
            stall = 0
            gamma = 0.12
            if time.monotonic() - t0 > time_limit:
                break
            continue

        A = np.fft.rfft(a, n=fft_size)
        A_mag = np.abs(A)
        A_phase = np.angle(A)

        S = A_mag ** 2
        S_max = S.max()
        if S_max < 1e-20:
            break

        # Adaptive compression
        eff_gamma = gamma * (1.0 + 0.05 * min(stall / 200.0, 3.0))
        eff_gamma = min(eff_gamma, 0.5)

        S_new = S_max * np.power(S / S_max, 1.0 - eff_gamma)
        A_mag_new = np.sqrt(S_new)
        A_new = A_mag_new * np.exp(1j * A_phase)
        a_new = np.fft.irfft(A_new, n=fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        val_new = obj_val(a_new)
        if val_new < best_val * 1.01:
            a = a_new
            gamma = max(0.02, gamma * 0.997)
            if val_new < best_val - 1e-12:
                best_val = val_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 3

        if verbose and it % 500 == 0:
            print("    SpecShap it=" + str(it) + " (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(best_val))

    return best_a, best_val


def grad_mult(a0, time_limit, top_k=80, temp=50.0, c0=3.0, verbose=True):
    """Enhanced multiplicative gradient descent with Nesterov momentum."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    x_prev = x.copy()
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 15000:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]

        cur_temp = temp * max(0.2, 1.0 - t / 100000.0)
        w = np.exp(cur_temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * y[ki - idx_arr]

        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)

        # Try multiple step sizes
        best_step_val = best_val
        best_new_x = x
        for step_mult in [1.0, 0.5, 2.0, 0.25, 4.0]:
            eta_trial = eta_t * step_mult
            log_a = np.log(y) - eta_trial * grad
            max_log = log_a.max()
            a_trial = np.exp(log_a - max_log)
            a_trial = np.maximum(a_trial, 1e-30)
            a_trial = a_trial / a_trial.sum()

            if val_trial < best_step_val:
                best_step_val = val_trial

        x = best_new_x

        if obj_val(x) < best_val - 1e-12:
            best_val = obj_val(x)
            best_a = x.copy()
            stall = 0

            print("    GradMult t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val))

    return best_a, best_val


def lp_fw(a0, time_limit, solver, init_K=500, max_K=5000, add_per=50, verbose=True):

    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0

        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)

        # Ensure current top peaks are in active set

        active_set = active_set | current_top
        clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        g_var = cp.Variable(n, nonneg=True)

        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True

                continue

            continue

        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))

        # Golden section line search
        lo = 0.01
        hi = 1.0

            best_alpha = 1.0
        else:

                    f2 = f1
                    c1 = lo + inv_phi * (hi - lo)

                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = hi - inv_phi * (hi - lo)
                    f2 = trial_obj_fn(y, g_dir, c2)
            best_alpha = (lo + hi) / 2.0

        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()

            y = y_new / sy
        else:
            y = g_dir.copy()

        if verbose and t % 100 == 0:

            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)))

    if verbose:
        print("    LP-FW done (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m))

            best_score = sc
            best_seq = interpolate_seq(a_out, n_parent) if n_parent > n_lp else a_out
            print("  LP-FW improved: " + repr(best_score))

    # === Phase 3: Gradient at full length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: GradMult (n=" + str(n_cur) + ") ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_out, v_out = grad_mult(normalize(best_seq), t3, top_k=100, temp=60.0, c0=3.0, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  GradMult improved: " + repr(best_score))

    # === Phase 4: LP-FW second pass (warm start) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_lp2 = min(len(best_seq), 10000)
    a_lp2 = interpolate_seq(best_seq, n_lp2) if n_lp2 != len(best_seq) else best_seq.copy()
    print("\n=== P4: LP-FW2 (n=" + str(n_lp2) + ") ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 60:
        a_out, v_out = lp_fw(normalize(a_lp2), t4, solver,
                             init_K=800, max_K=6000, add_per=80, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_parent) if n_parent > n_lp2 else a_out
            print("  LP-FW2 improved: " + repr(best_score))

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_pol = min(len(best_seq), 12000)
    a_pol = interpolate_seq(best_seq, n_pol) if n_pol != len(best_seq) else best_seq.copy()
    print("\n=== P5: CoordDescent (n=" + str(n_pol) + ") ===")
    t5 = min(remaining * 0.10, 170.0)
    if t5 > 20:
        a_out, v_out = coord_descent(normalize(a_pol), t5, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_parent) if n_parent > n_pol else a_out
            print("  Coord improved: " + repr(best_score))

    # === Phase 6: Alternate length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_alt = n_parent + 2000
        if n_alt <= 12000:
            a_alt = interpolate_seq(best_seq, n_alt)
            print("\n=== P6: AltLen LP (n=" + str(n_alt) + ") ===")
            t6 = min(remaining * 0.25, 200.0)
            if t6 > 30:
                a_out, v_out = lp_fw(normalize(a_alt), t6, solver,
                                     init_K=500, max_K=4000, add_per=50, verbose=False)
                sc = eval_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out
                    n_parent = n_alt
                    print("  AltLen improved: " + repr(best_score))
        else:
            a_alt = interpolate_seq(best_seq, max(n_parent - 2000, 8000))
            n_alt2 = len(a_alt)
            print("\n=== P6: AltLen Grad (n=" + str(n_alt2) + ") ===")
            t6 = min(remaining * 0.25, 200.0)
            if t6 > 30:
                a_out, v_out = grad_mult(normalize(a_alt), t6, top_k=80, temp=50, c0=3, verbose=False)
                sc = eval_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out
                    n_parent = n_alt2
                    print("  AltLen grad improved: " + repr(best_score))

    # === Phase 7: Final short LP burst ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40 and best_seq is not None:
        n_fin = min(len(best_seq), 9000)
        a_fin = interpolate_seq(best_seq, n_fin) if n_fin != len(best_seq) else best_seq.copy()
        print("\n=== P7: Final LP (n=" + str(n_fin) + ") ===")
        t7 = max(remaining - 20, 25)
        a_out, v_out = lp_fw(normalize(a_fin), t7, solver,
                             init_K=800, max_K=5000, add_per=60, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  Final LP: " + repr(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " t=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: fallback")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
