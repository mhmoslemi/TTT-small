        total_iters += 1

        if stall > 1500 and stall % 500 == 0:
            noise_lvl = 0.005 * (1.0 + 0.5 * rng.standard_normal(n)) * best_a
            a = simplex_project(best_a + noise_lvl)
            v = np.zeros(n)
            eta = eta0
            stall = 0

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("  SMGD t={:5d} ({:.1f}s) obj={:.7f} T={:.1f} eta={:.1e} stall={}".format(
                t, elapsed, 2.0 * n * best_obj, T, eta, stall))

    return best_a, 2.0 * n * best_obj

def coord_descent(a0, time_limit, n_targets=15, verbose=True, seed=7):
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000

    fracs = [0.0005, 0.001, 0.002, 0.004, 0.008, 0.015, 0.03, 0.05]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        top_k = np.argsort(conv)[::-1][:n_targets]
        improved = False

        # Compute pressure
        pr = np.zeros(n, dtype=np.float64)
        for k in top_k[:8]:
            ki = int(k)
            ws = max(0, ki - n + 1)
            we = min(n - 1, ki)
            if ws <= we:
                idx = np.arange(ws, we + 1)
                pr[idx] += a[ki - idx]

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ws = max(0, ki - n + 1)
            we = min(n - 1, ki)
            if ws > we:
                continue
            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue

            pr_safe = pr.copy()
            pr_safe[src] = 1e10
            cold = np.argsort(pr_safe)[:50]

            for dst in cold:
                dst = int(dst)
                if dst == src:
                    continue
                found = False
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_test = a.copy()
                    a_test[src] -= amt
                    a_test[dst] += amt
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test = a_test / st
                    obj_t = obj_val(a_test)
                    if obj_t < best_obj - 1e-14:
                        a = a_test
                        best_obj = obj_t
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

            if not improved and stall > 200:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        obj_n = obj_val(a)
                        if obj_n < best_obj:
                            best_obj = obj_n
                            best_a = a.copy()
                            stall = 0
                            improved = True
                if not improved:
                    break

        if not improved:
            stall += 1
            if stall > 500 and stall % 200 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        obj_n = obj_val(a)
                        if obj_n < best_obj:
                            best_obj = obj_n
                            best_a = a.copy()
                            stall = 0
        else:
            stall = 0

        if verbose and stall % 1000 == 0 and stall > 0:
            print("  CD ({:.1f}s) obj={:.7f} stall={}".format(
                time.monotonic() - t0, 2.0 * n * best_obj, stall))

    return best_a, obj_val(best_a)

def targeted_refinement(a0, time_limit, verbose=True):
    """Targeted refinement: focus on reducing the single worst lag."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 5000:
        conv = fast_conv(a)
        k_worst = int(np.argmax(conv))
        M_curr = conv[k_worst]
        cur_obj = 2.0 * n * M_curr

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        ws = max(0, k_worst - n + 1)
        we = min(n - 1, k_worst)
        if ws > we:
            break
        idx = np.arange(ws, we + 1)
        contrib = a[idx] * a[k_worst - idx]
        total = contrib.sum()
        if total < 1e-20:
            break

        # Find the pair contributing most to the worst lag
        src = int(idx[np.argmax(contrib)])
        partner = k_worst - src
        if partner < 0 or partner >= n:
            partner = int(idx[np.argmax(contrib)])
            src = k_worst - partner
        if src < 0 or src >= n:
            break

        # Find cold positions (low pressure across all top lags)
        top_lags = np.argsort(conv)[::-1][:10]
        pr = np.zeros(n, dtype=np.float64)
        for k in top_lags:
            ki = int(k)
            wss = max(0, ki - n + 1)
            wee = min(n - 1, ki)
            if wss <= wee:
                ii = np.arange(wss, wee + 1)
                pr[ii] += a[ki - ii]
        pr[src] = 1e10
        pr[partner] = 1e10 if partner != src else 1e10
        cold = np.argsort(pr)[:100]

        improved_this_round = False
        for dst in cold:
            dst = int(dst)
            if dst == src or dst == partner:
                continue
            # Try moving fraction of a[src] to a[dst]
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                if amt < 1e-16:
                    break
                a_test = a.copy()
                a_test[src] -= amt
                a_test[dst] += amt
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                obj_t = obj_val(a_test)
                if obj_t < best_obj - 1e-14:
                    a = a_test
                    best_obj = obj_t
                    best_a = a.copy()
                    improved_this_round = True
                    break
            if improved_this_round:
                break

        if not improved_this_round:
            # Random perturbation to escape
            if stall > 500:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a /= st
                        obj_n = obj_val(a)
                        if obj_n < best_obj:
                            best_obj = obj_n
                            best_a = a.copy()
                            stall = 0
        else:
            stall = 0

        if verbose and stall % 2000 == 0 and stall > 0:
            print("  TR ({:.1f}s) obj={:.7f} stall={}".format(
                time.monotonic() - t0, 2.0 * n * best_obj, stall))

    return best_a, obj_val(best_a)

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = obj_val(best_seq)
            print("Parent: n={} score={:.7f}".format(len(p), best_score))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 10000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = obj_val(best_seq)
        print("Default init n={} score={:.7f}".format(n_init, best_score))

    n0 = len(best_seq)
    print("Starting optimization from n={}".format(n0))

    # Phase 1: Smoothed max gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    print("\n=== P1: SmoothedMax GradDesc (n={}) t={:.0f}s ===".format(n0, t1))
    if t1 > 30:
        a_out, v_out = smooth_max_descent(normalize(best_seq), t1,
                                          T_start=3.0, T_end=60.0,
                                          eta0=0.004, momentum=0.9,
                                          verbose=True, seed=42)
        sc = obj_val(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1 improved: {:.7f}".format(best_score))

    # Phase 2: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    print("\n=== P2: CoordDescent (n={}) t={:.0f}s ===".format(len(best_seq), t2))
    if t2 > 20:
        a_out, v_out = coord_descent(normalize(best_seq), t2,
                                     n_targets=15, verbose=True, seed=7)
        sc = obj_val(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: {:.7f}".format(best_score))

    # Phase 3: Targeted refinement (focus on worst lag)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print("\n=== P3: TargetedRefine (n={}) t={:.0f}s ===".format(len(best_seq), t3))
    if t3 > 20:
        a_out, v_out = targeted_refinement(normalize(best_seq), t3, verbose=True)
        sc = obj_val(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 improved: {:.7f}".format(best_score))

    # Phase 4: Multi-start with perturbations
    remaining = time_budget - (time.monotonic() - t_global)
    n_starts = 3
    t4_each = min(remaining / (n_starts + 1), 120.0)
    print("\n=== P4: MultiStart (n={}) {} restarts ===".format(len(best_seq), n_starts))
    rng_main = np.random.default_rng(999)
    for s in range(n_starts):
        if time.monotonic() - t_global > time_budget - 30:
            break
        # Create perturbation
        noise = rng_main.standard_normal(len(best_seq)) * 0.008 * best_seq
        a_start = simplex_project(best_seq + noise)
        # Short gradient descent
        a_out, v_out = smooth_max_descent(a_start, t4_each,
                                          T_start=10.0, T_end=50.0,
                                          eta0=0.003, momentum=0.85,
                                          verbose=False, seed=100 + s)
        # Short coord descent
        a_out2, v_out2 = coord_descent(a_out, t4_each * 0.5,
                                       n_targets=10, verbose=False, seed=200 + s)
        sc1 = obj_val(a_out)
        sc2 = obj_val(a_out2)
        best_local = min(sc1, sc2)
        best_local_seq = a_out if sc1 < sc2 else a_out2
        if best_local < best_score:
            best_score = best_local
            best_seq = best_local_seq
            print("  P4 start {} improved: {:.7f}".format(s, best_score))

    # Phase 5: Try a slightly different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_try = len(best_seq) + 1000
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 40 and n_try <= 15000:
        print("\n=== P5: LengthExplore n={} t={:.0f}s ===".format(n_try, t5))
        a_long = interpolate_seq(best_seq, n_try)
        a_out, v_out = smooth_max_descent(a_long, t5 * 0.5,
                                          T_start=5.0, T_end=40.0,
                                          eta0=0.003, momentum=0.9,
                                          verbose=False, seed=555)
        a_out2, v_out2 = coord_descent(a_out, t5 * 0.5,
                                       n_targets=10, verbose=False, seed=556)
        sc1 = obj_val(a_out)
        sc2 = obj_val(a_out2)
        if min(sc1, sc2) < best_score:
            best_score = min(sc1, sc2)
            best_seq = a_out if sc1 < sc2 else a_out2
            print("  P5 improved: {:.7f}".format(best_score))

    # Phase 6: Final intensive coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.25, 300.0)
    print("\n=== P6: FinalCoord (n={}) t={:.0f}s ===".format(len(best_seq), t6))
    if t6 > 20:
        a_out, v_out = coord_descent(normalize(best_seq), t6,
                                     n_targets=20, verbose=True, seed=777)
        sc = obj_val(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P6 improved: {:.7f}".format(best_score))

    # Phase 7: Final targeted refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.20, 200.0)
    print("\n=== P7: FinalTarget (n={}) t={:.0f}s ===".format(len(best_seq), t7))
    if t7 > 20:
        a_out, v_out = targeted_refinement(normalize(best_seq), t7, verbose=True)
        sc = obj_val(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P7 improved: {:.7f}".format(best_score))

    # Final validation with evaluate_sequence
    if best_seq is not None:
        final_list = [float(x) for x in best_seq]
        try:
            fs = float(evaluate_sequence(final_list))
            if np.isfinite(fs):
                best_score = fs
        except Exception:
            pass
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={:.7f} n={} t={:.1f}s".format(
            best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: falling back")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = obj_val(best_seq)

    return [float(x) for x in best_seq]
