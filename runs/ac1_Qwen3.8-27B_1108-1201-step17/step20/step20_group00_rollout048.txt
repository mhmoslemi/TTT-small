def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_constraint_matrix(a, lags, n):
    """Build the constraint matrix for the LP.
    Row j corresponds to lag lags[j]: (a_conv) * g at that lag = sum_i a[k-i]*g[i].
    """
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        k = int(lags[jj])
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k - idx]
    return coeffs

def lp_dir_step(y, active_lags, n, solver):
    """Solve the LP to find the Frank-Wolfe direction.
    Returns (g, M_opt) or (None, None) if infeasible.
    """
    K = len(active_lags)
    if K == 0:
        return None, None
    
    coeffs = build_constraint_matrix(y, active_lags, n)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):

            gs = g_val.sum()
            if gs > 1e-15:
                g_val = g_val / gs
            return g_val, float(M_var.value)
    except Exception:
        pass
    return None, None

def spectral_smooth(a, sigma=0.5, strength=0.3):
    """Apply spectral smoothing: blend a with a version whose spectrum is smoothed.
    This can help reduce sharp autocorrelation peaks.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = np.fft.rfft(a, n=fft_size)
    power = np.abs(A) ** 2
    
    # Smooth the power spectrum
    kernel_size = max(3, int(2 * sigma * fft_size // 2))
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.exp(-0.5 * (np.arange(-kernel_size // 2, kernel_size // 2 + 1) / sigma) ** 2)
    kernel = kernel / kernel.sum()
    
    # Convolve power spectrum with kernel (circular)
    power_smooth = np.fft.irfft(np.fft.rfft(power, n=fft_size) * np.fft.rfft(kernel, n=fft_size), fft_size)
    power_smooth = np.maximum(power_smooth, 1e-20)
    
    # New amplitude: blend
    new_power = (1 - strength) * power + strength * power_smooth
    
    # Get new amplitude (keep phase)
    new_amp = np.sqrt(new_power + 1e-30)
    new_A = new_amp * (A / (np.abs(A) + 1e-30))
    
    # Inverse FFT
    a_smooth = np.fft.irfft(new_A, fft_size)[:n]
    a_smooth = np.maximum(a_smooth, 0.0)
    ss = a_smooth.sum()
    if ss < 1e-15:
        return a.copy()
    a_smooth = a_smooth / ss
    
    return a_smooth
