def simplex_project(z):
    """Project onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def smooth_max_grad(a, beta, top_k=None):
    """
    Compute gradient of smooth max of self-convolution.
    Returns (gradient, smooth_max_value).
    """
    n = len(a)
    C = fast_self_conv(a)
    if top_k is not None and top_k < len(C):
        idx = np.argpartition(C, -top_k)[-top_k:]
        C_use = np.zeros_like(C)
        C_use[idx] = C[idx]
    else:
        C_use = C.copy()
    C_max = C_use.max()
    # Softmax weights
    log_w = beta * (C_use - C_max)
    log_w = np.clip(log_w, -500.0, 500.0)
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(C)
        w[C_max == C_use.max()] = 1.0
        w /= len(np.where(C_max == C_use.max())[0])
    else:
        w /= w_sum
    # Gradient: grad[i] = 2 * (a * p_rev)[2n-2-i]
    p = w  # length 2n-1
    p_rev = p[::-1]  # length 2n-1
    # convolve a (length n) with p_rev (length 2n-1) -> length 3n-2
    conv_len = n + (2 * n - 1) - 1  # 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    Af = np.fft.rfft(a, fft_size)
    PRf = np.fft.rfft(p_rev, fft_size)
    conv = np.fft.irfft(Af * PRf, fft_size)[:conv_len]
    # grad[i] = 2 * conv[2n-2-i] for i = 0, ..., n-1
    # indices: 2n-2, 2n-3, ..., n-1
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    smooth_val = C_max + np.log(w_sum) / beta if w_sum > 0 else C_max
    return grad, smooth_val, C.max(), w

**Alternative: Direct frequency domain optimization**

The self-convolution `C = IFFT(|FFT(a)|^2)`. We want to minimize `max(C)` subject to `a >= 0, sum(a) = 1`.

In the frequency domain, this is: minimize `max(IFFT(P))` subject to `IFFT(A) >= 0, sum(A) = n` (where `A = FFT(a)` and `P = |A|^2`).

This is still non-convex due to the `|A|^2` term. But we can parameterize `A` as `A = |A| * exp(i*phase)` and optimize over `|A|` and `phase` separately.

For a given phase, optimizing over `|A|` is a quadratic program (minimize `max(IFFT(|A|^2))` subject to `IFFT(|A|*exp(i*phase)) >= 0`). This is still non-convex.

I think the gradient-based approach is the way to go. Let me focus on making it as efficient and effective as possible.

Let me also think about the "pressure" concept for peak shaving. The idea is:
- For each top peak at lag k, the "pressure" on position i is how much a[i] contributes to that peak.
- `pressure[i] = sum_{k in top_peaks} p[k] * a[k-i]` (weighted by peak importance)
- To reduce a peak, move mass from high-pressure positions to low-pressure positions.

This is essentially what the gradient already computes! The gradient IS the pressure (up to a factor of 2). So the peak shaving step is redundant with the Nesterov step. The difference is that Nesterov moves mass in proportion to the gradient (smooth), while peak shaving is more "directed" (focusing on the single worst peak).

For the peak shaving, I'll use a more aggressive approach:
- Identify the single worst peak.
- Find the position that most contributes to it.
- Move a fraction of its mass to the position that least contributes to ALL top peaks.
- This is a "directed" move that targets the specific peak.

Actually, let me also consider **coordinate descent** more seriously. The idea is:
- At each step, pick one coordinate a[i] and update it optimally (given all other coordinates).
- The objective as a function of a[i] (with all else fixed) is:

"""
Minimizes J = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated gradient with FFT-based smooth max (adaptive beta annealing)
2. Direct peak shaving with pressure-based destination selection
3. Frank-Wolfe with greedy coordinate direction
4. Multi-length exploration with interpolation
5. Final high-beta polish with coordinate descent

import numpy as np
import time

def fast_self_conv(a):
    """Compute self-convolution a*a using FFT (full mode)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(z):
    """Project onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def smooth_max_and_grad(a, beta, top_k=None):
    """
    Compute smooth max of self-convolution and its gradient.
    Returns (smooth_max_value, gradient_array, true_max_value).
    """
    n = len(a)
    C = fast_self_conv(a)
    true_max = C.max()
    if top_k is not None and top_k < len(C):
        idx = np.argpartition(C, -top_k)[-top_k:]
        C_use = np.zeros_like(C)
        C_use[idx] = C[idx]
    else:
        C_use = C.copy()
    C_max = C_use.max()
    log_w = beta * (C_use - C_max)
    log_w = np.clip(log_w, -500.0, 500.0)
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-300:
        max_indices = np.where(C_use == C_max)[0]
        w = np.zeros_like(C)
        w[max_indices] = 1.0
        w /= len(max_indices)
        smooth_val = C_max
    else:
        w /= w_sum
        smooth_val = C_max + np.log(w_sum) / beta
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (with bounds)
    # = 2 * (a * w_rev)[2n-2-i]
    w_rev = w[::-1]
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Af = np.fft.rfft(a, fft_size)
    WRf = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(Af * WRf, fft_size)[:conv_len]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1].copy()
    return smooth_val, grad, true_max

def nesterov_optimize(a0, time_limit, best_score, beta_start=10.0, beta_end=500.0,
                      top_k_ratio=0.05, verbose=True):
    """Nesterov accelerated gradient with smooth max and adaptive beta."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_m = fast_self_conv(a).max()
    best_J = 2.0 * n * best_m
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    y = a.copy()  # Nesterov's y
    t_nesterov = 1.0
    beta = beta_start
    lr = 0.1
    t = 0
    stall = 0
    max_stall = 300
    top_k = max(20, int(n * top_k_ratio))

    while time.monotonic() - t0 < time_limit:
        t += 1
        t_nesterov = (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov ** 2)) / 2.0
        momentum = (t_nesterov - 1.0) / t_nesterov

        # Compute gradient at y
        smooth_val, grad, true_max = smooth_max_and_grad(y, beta, top_k)

        # Nesterov update
        y_new = y + momentum * (y - a)
        y_new = y_new - lr * grad
        a_new = simplex_project(y_new)
        y = a_new.copy()
        a = a_new

        # Check improvement
        m_curr = fast_self_conv(a).max()
        if m_curr < best_m - 1e-15:
            best_m = m_curr
            best_a = a.copy()
            stall = 0
            best_J = 2.0 * n * best_m
        else:
            stall += 1

        # Adaptive beta
        if t % 200 == 0:
            beta = min(beta * 1.05, beta_end)

        # Adaptive learning rate
        if t % 100 == 0:
            if stall > 100:
                lr *= 0.7
            elif stall == 0:
                lr = min(lr * 1.1, 0.5)
            lr = max(lr, 1e-5)

        # Top-k annealing (increase over time)
        if t % 500 == 0:
            top_k = min(int(top_k * 1.3), n)

        # Restart
        if stall > max_stall:
            stall = 0
            noise_scale = 0.03 / (1.0 + 0.5 * (t // max_stall))
            pert = rng.normal(0, noise_scale, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-12)
            a_r = a_r / a_r.sum()
            m_r = fast_self_conv(a_r).max()
            if m_r < best_m:
                best_m = m_r
                best_a = a_r.copy()
                best_J = 2.0 * n * best_m
            a = a_r
            y = a_r.copy()
            t_nesterov = 1.0
            lr = max(lr, 0.05)
            beta = max(beta * 0.5, beta_start)
            if verbose:
                elapsed = time.monotonic() - t0
                print("    Nesterov restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(best_J))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    Nesterov t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(best_J) + " beta=" + repr(round(beta, 1)))

    # Final evaluation
    final_m = fast_self_conv(best_a).max()
    final_J = 2.0 * n * final_m
    return best_a, final_J

def peak_shaving(a0, time_limit, best_score, verbose=True):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    if s < 1e-15:

    max_stall = 1000
    top_k = 15

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        C = fast_self_conv(a)
        peak_indices = np.argsort(C)[::-1][:top_k]

        # Compute pressure
        pressure = np.zeros(n)
        for k_i in peak_indices[:8]:
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                pressure[idx] += a[k_int - idx]
                pressure[k_int - idx] += a[idx]

        improved = False
        for k_i in peak_indices[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:

            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            if partner < 0 or partner >= n:

            if a[src] >= a[partner]:
                reduce_idx = src
            else:
                reduce_idx = partner
            if a[reduce_idx] < 1e-14:
                continue
            # Find safe destination
            safe_pressure = pressure.copy()
            safe_pressure[reduce_idx] = np.max(pressure) + 1.0
            # Also penalize positions involved in other top peaks
            for k2 in peak_indices[:5]:
                k2i = int(k2)
                if abs(safe_pressure[int(min(k2i, n-1))]) < safe_pressure.max():

            dst = int(np.argmin(safe_pressure))
            if dst == reduce_idx:

            found = False
            for frac in [0.01, 0.03, 0.06, 0.1, 0.15, 0.25]:
                amt = a[reduce_idx] * frac
                if amt < 1e-16:

                a_trial = a.copy()

                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:

                a_trial = a_trial / st
                m_trial = fast_self_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
            if not found and stall > 300:
                j = int(rng.integers(n))
                if j != reduce_idx and a[reduce_idx] > 1e-12:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[reduce_idx] * frac
                    a[reduce_idx] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        m_new = fast_self_conv(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
        if not improved:
            stall += 1
            if stall % 100 == 0:
                noise_scale = min(0.05, 0.01 * stall / 100)
                pert = rng.normal(0, noise_scale, n) * best_a
                a_r = best_a + pert
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
                m_r = fast_self_conv(a_r).max()
                if m_r < best_m:
                    best_m = m_r

                a = a_r
            if verbose and stall % 500 == 0:

                J = 2.0 * n * best_m

def frank_wolfe_greedy(a0, time_limit, best_score, n_constraints=500, verbose=True):
    """Frank-Wolfe with greedy coordinate direction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_m = fast_self_conv(a).max()
    t0 = time.monotonic()
    stall = 0

    t = 0
    while time.monotonic() - t0 < time_limit and stall < 500:
        t += 1
        C = fast_self_conv(a)
        M_curr = C.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find top constraints
        top_peaks = np.argsort(C)[::-1][:n_constraints]

        # For each peak, find the best coordinate to move mass from
        best_direction = None
        best_improvement = 0.0
        for k_i in top_peaks[:50]:
            k_int = int(k_i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            # Coefficient of a[i] in C[k_int] is a[k_int - i]
            # To reduce C[k_int], we want to reduce a[i] where a[k_int - i] is large
            # Or increase a[j] where a[k_int - j] is small (but this increases C[k_int])
            # Actually, moving mass from i to j changes C[k] by:
            #   -amt * a[k-i] + amt * a[k-j] (approximately, for the linearized version)
            # So we want to move from i (where a[k-i] is large) to j (where a[k-j] is small)
            coeffs = a[k_int - idx]  # coefficient for each source position
            src = int(idx[np.argmax(coeffs)])
            if a[src] < 1e-14:
                continue
            # Find destination: minimize total impact on all top peaks
            dst_pressure = np.zeros(n)
            for k2 in top_peaks[:20]:
                k2i = int(k2)
                j_start = max(0, k2i - n + 1)
                j_end = min(n - 1, k2i)
                if j_start <= j_end:
                    j_idx = np.arange(j_start, j_end + 1)
                    dst_pressure[j_idx] += a[k2i - j_idx]
            dst_pressure[src] = dst_pressure.max() + 1.0
            dst = int(np.argmin(dst_pressure))
            if dst == src:
                continue
            # Try moving
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_self_conv(a_trial).max()
                if m_trial < M_curr - 1e-16:
                    # Accept with Frank-Wolfe step size
                    alpha = 2.0 / (t + 2.0)
                    a_new = (1.0 - alpha) * a + alpha * a_trial
                    a_new = np.maximum(a_new, 0.0)
                    st = a_new.sum()
                    if st > 1e-15:
                        a_new = a_new / st
                    m_new = fast_self_conv(a_new).max()
                    if m_new < M_curr:
                        a = a_new
                        best_direction = (src, dst)
                        break
                    else:
                        a = a_trial
                        break
        if best_direction is None and stall > 100:
            break
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, 2.0 * n * best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coordinate_descent_final(a0, time_limit, best_score, verbose=True):
    """Final polish: cycle through coordinates and try small adjustments."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_m = fast_self_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    # Focus on positions that contribute to top peaks
    C = fast_self_conv(a)
    top_peaks = np.argsort(C)[::-1][:20]
    important = set()
    for k_i in top_peaks:
        k_int = int(k_i)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:5]]
            for tc in top_contrib:
                important.add(int(tc))

    important_list = list(important) if important else list(range(n))
    rng.shuffle(important_list)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in important_list:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            # Try moving half of a[i] to a random other position
            for frac in [0.1, 0.3, 0.5]:
                j = int(rng.integers(n))
                if j == i:
                    continue
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_self_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            # Try random pairs
            for _ in range(100):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                amt = a[i] * rng.uniform(0.05, 0.3)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_self_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        if not improved and rng.random() < 0.01:
            break

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print("    CoordDescent (" + str(round(elapsed, 1)) + "s): J=" + repr(J))

    return best_a, 2.0 * n * best_m

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):

                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))

        n_base = 6700
