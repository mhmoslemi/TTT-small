import numpy as np

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def mirror_descent(a0, time_limit, n_top=50, lr0=0.1, verbose=True):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent (multiplicative weights) with adaptive top-k subgradient
2. Backtracking line search and periodic restarts
3. Coordinate descent for fine-tuning
4. Multi-start exploration with perturbed initializations

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def obj_val(a):
    """Compute objective 2*n*max_autocorr (for normalized a)."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def eval_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_subgradient(a, top_lags, top_vals, n_top):
    """Compute weighted subgradient of max_k (a*a)[k]."""
    n = len(a)
    s = np.zeros(n, dtype=np.float64)
    k_sel = min(n_top, len(top_lags))
    for ii in range(k_sel):
        ki = int(top_lags[ii])
        if top_vals[0] > 0:
            w = top_vals[ii] / top_vals[0]
        else:
            w = 1.0 / k_sel
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start < j_end:
            indices = np.arange(j_start, j_end)
            s[indices] += 2.0 * w * a[ki - indices]
    return s


def mirror_descent(a0, time_limit, n_top=50, lr0=0.05, restart_interval=2000,
                   noise_scale=0.01, verbose=True):
    """
    Mirror descent (multiplicative weights) with adaptive top-k subgradient,
    backtracking, and periodic restarts.
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Select top-k lags
        k_sel = min(n_top, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[top_idx]

        # Compute subgradient
        s = compute_subgradient(a, top_idx, top_vals, k_sel)

        # Adaptive learning rate with backtracking
        lr = lr0 / (1.0 + 0.005 * t)
        max_backtracks = 8
        accepted = False

        for bt in range(max_backtracks):
            log_a = np.log(a + 1e-30) - lr * s
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-25)
            a_new = a_new / a_new.sum()

            val_new = 2.0 * n * fast_autocorr(a_new).max()
            if val_new <= cur_val * 1.0001 + 1e-18:
                a = a_new
                accepted = True
                break
            else:
                lr *= 0.5

        if not accepted:
            a = a_new  # accept anyway to avoid stalling

        # Periodic restart with noise
        if stall > restart_interval and stall % (restart_interval // 4) == 0:
            noise = noise_scale * rng.standard_normal(n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-12)
            a = a_r / a_r.sum()
            stall = 0
            lr = lr0  # reset lr
            continue

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    MD t={:5d} ({:.1f}s) J={:.7f} stall={} lr={:.5f}".format(
                t, elapsed, best_val, stall, lr))

    return best_a, best_val

        # Normalize pressure to have mean 1 on the support
        support_mask = a > 1e-10
        if support_mask.sum() > 0:
            mean_p = pressure[support_mask].mean()
            if mean_p > 1e-20:
                pressure_norm = pressure / mean_p
            else:
                pressure_norm = np.ones(n)
        else:
            pressure_norm = np.ones(n)

        # Update: reduce where pressure > 1, increase where pressure < 1
        lr = lr0 / (1.0 + 0.003 * t)
        log_a = np.log(a + 1e-30) - lr * (pressure_norm - 1.0)
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-25)
        a_new = a_new / a_new.sum()

        # Check improvement
        val_new = 2.0 * n * fast_autocorr(a_new).max()
        if val_new > cur_val * 1.01:
            # Backtrack
            lr *= 0.25
            log_a = np.log(a + 1e-30) - lr * (pressure_norm - 1.0)
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-25)
            a_new = a_new / a_new.sum()

        a = a_new

        # Periodic restart
        if stall > 3000 and stall % 500 == 0:
            noise = 0.005 * rng.standard_normal(n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-12)
            a = a_r / a_r.sum()
            stall = 0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    PE t={:5d} ({:.1f}s) J={:.7f} stall={}".format(t, elapsed, best_val, stall))

    return best_a, best_val


def symmetric_refine(a0, time_limit, verbose=True):
    """
    Enforce approximate symmetry and optimize under that constraint.
    Try a = (a + reverse(a)) / 2 as a starting point, then optimize.
    """
    n = len(a0)
    a = normalize(a0).copy()
    
    # Try symmetric version
    a_sym = (a + a[::-1]) / 2.0
    a_sym = normalize(a_sym)
    val_sym = obj_val(a_sym)
    val_orig = obj_val(a)
    
    if val_sym < val_orig:
        base = a_sym
    else:
        base = a
    
    # Run mirror descent on the base
    result, val = mirror_descent(base, time_limit, n_top=40, lr0=0.03,
                                  restart_interval=3000, noise_scale=0.005, verbose=verbose)
    return result, val


def propose_candidate(budget_s=1800.0):
    """Main entry point: search for best sequence within time budget."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n={} score={}".format(len(p), repr(best_score)))
    except Exception:
        best_seq = None

    if best_seq is None:
        n_init = 10000
        x = np.linspace(0, 1, n_init)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Default init n={} score={}".format(n_init, repr(best_score)))

    n0 = len(best_seq)

    # Phase 1: Mirror descent with moderate settings
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Mirror Descent (n={}, n_top=30, lr=0.05) ===".format(n0))
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_out, v_out = mirror_descent(normalize(best_seq), t1, n_top=30, lr0=0.05,
                                       restart_interval=2000, noise_scale=0.01, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1 improved: {}".format(repr(best_score)))

    # Phase 2: Pressure equalization
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P2: Pressure Equalization (n={}) ===".format(len(best_seq)))
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_out, v_out = pressure_equalization(normalize(best_seq), t2, n_top=40, lr0=0.03, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: {}".format(repr(best_score)))

    # Phase 3: Mirror descent with wider top-k
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P3: Mirror Descent wide (n_top=80, lr=0.02) ===".format(len(best_seq)))
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_out, v_out = mirror_descent(normalize(best_seq), t3, n_top=80, lr0=0.02,
                                       restart_interval=3000, noise_scale=0.008, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 improved: {}".format(repr(best_score)))

    # Phase 4: Coordinate descent for fine-tuning
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: Coordinate Descent (n={}) ===".format(len(best_seq)))
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 20:
        a_out, v_out = coord_descent(normalize(best_seq), t4, n_targets=8, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 improved: {}".format(repr(best_score)))

    # Phase 5: Final mirror descent polish (very small lr)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P5: Final MD polish (n_top=15, lr=0.01) ===".format(len(best_seq)))
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_out, v_out = mirror_descent(normalize(best_seq), t5, n_top=15, lr0=0.01,
                                       restart_interval=5000, noise_scale=0.003, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P5 improved: {}".format(repr(best_score)))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} t={:.1f}s".format(repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: using fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
