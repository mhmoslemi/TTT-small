def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    s = a.sum()

    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_grad_topk(a, peak_indices, weights):
    """
    Compute gradient of sum_k w_k * b[k] w.r.t. a, where b = autocorr(a).
    peak_indices: array of lag indices
    weights: array of same length, the softmax weights
    """
    n = len(a)
    grad = np.zeros(n)
    for ki in range(len(peak_indices)):
        k = int(peak_indices[ki])
        w = weights[ki]
        if w < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        partners = k - idx
        # db[k]/da[i] = a[k-i] + a[i] (with proper handling)
        contrib = a[partners] + a[idx]
        # Check for double-counting when i == k-i
        diag_mask = (idx == partners)
        contrib[diag_mask] = a[idx[diag_mask]]  # 2*a[i] becomes a[i] since we already have a[i]+a[i]
        # Actually: when i == k-i, the derivative is 2*a[i], not a[i]+a[i]=2*a[i]. So it's fine.
        # Wait: b[k] = sum_j a[j]*a[k-j]. d/da[i]:
        #   term j=i: a[k-i]
        #   term j=k-i: a[i] (if k-i != i)
        #   if k-i == i: only one term, derivative is 2*a[i]
        # So: if i != k-i: grad = a[k-i] + a[i]
        #     if i == k-i: grad = 2*a[i]
        # My formula gives a[k-i] + a[i] = a[i] + a[i] = 2*a[i] when i==k-i. Correct!
        grad[idx] += w * contrib
    return grad

def nesterov_optimize(a0, time_limit, top_k=30, beta_start=20.0, beta_end=500.0,
                       verbose=True, seed=42):
    """
    Nesterov-accelerated projected gradient on top-k smoothed max of autocorrelation.
    """
    n = len(a0)
    x = norm_sum1(a0).copy()
    y = x.copy()  # Nesterov lookahead
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    beta = beta_start
    momentum = 0.9
    t = 0
    stall = 0
    stall_limit = 800
    eta = 0.05  # initial step size

    while time.monotonic() - t0 < time_limit:
        t += 1
        # Nesterov step
        y = (1.0 + momentum) * x - momentum * x * 0 + momentum * (x - x)  # placeholder
        # Actually: y = x + momentum * (x - x_prev)
        # Let's use the standard: y_{k+1} = x_{k+1} + beta_k * (x_{k+1} - x_k)
        # Or: y = (1+mu)*x - mu*x_prev
        # Simpler: just use heavy-ball
        y = x + momentum * (x - getattr(nesterov_optimize, '_x_prev', x))
        nesterov_optimize._x_prev = x

        b = fast_autocorr(y)
        idx_top = np.argsort(b)[::-1][:top_k]
        b_top = b[idx_top]
        b_max = b_top[0]

        # Softmax weights (sharp)
        shifted = b_top - b_max
        exp_vals = np.exp(np.clip(shifted * beta, -50, 50))
        weights = exp_vals / exp_vals.sum()

        # Gradient
        grad = compute_grad_topk(y, idx_top, weights)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        eta = min(eta * 1.001, 0.08)  # very slow growth
        # Decay
        eta = max(0.001, eta * (1.0 - 0.0005))

        x_new = x - eta * grad / (gnorm + 1e-10)
        x_new = simplex_project(x_new)

        # Evaluate
        b_new = fast_autocorr(x_new)
        m_new = b_new.max()

        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0
            x = x_new
            eta = min(eta * 1.02, 0.08)  # encourage larger steps
        else:
            stall += 1
            # Backtrack
            if stall % 10 == 0:
                eta *= 0.7
            # Keep x unchanged
            x = x

        # Adapt beta
        if t % 100 == 0:
            beta = min(beta * 1.02, beta_end)
            momentum = min(0.95, 0.9 + 0.0001 * t)

        # Restart
        if stall > stall_limit:
            stall = 0
            stall_limit = max(400, stall_limit * 0.9)
            noise_scale = 0.03 / (1.0 + 0.1 * (t // 1000))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            m_r = fast_autocorr(x_r).max()
            if m_r < best_m:
                best_m = m_r
                best_a = x_r.copy()
            x = x_r if m_r < best_m + 0.005 else best_a.copy()
            eta = 0.03
            beta = max(beta * 0.7, 10.0)
            nesterov_optimize._x_prev = x.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    Nesterov restart t={0} ({1:.1f}s): J={2:.6f}".format(t, elapsed, J))

        if verbose and t % 2000 == 0:

    return best_a, best_m

def surgical_peak_shaving(a0, time_limit, top_k=15, verbose=True, seed=777):
    """
    Targeted peak shaving: for each top peak, find the best mass transfer
    that reduces it most while minimizing damage to other peaks.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 1000:
        b = fast_autocorr(a)
        peak_indices = np.argsort(b)[::-1][:top_k]

        improved = False
        # Process peaks from highest to lowest
        for k_idx in range(min(8, len(peak_indices))):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(peak_indices[k_idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            partners = k - idx
            contrib = a[idx] * a[partners]

            if contrib.max() < 1e-20:
                continue

            # Find the pair with maximum contribution
            max_pos = int(np.argmax(contrib))
            src_i = int(idx[max_pos])
            src_partner = int(partners[max_pos])

            # Try reducing the larger of the pair
            if a[src_i] >= a[src_partner]:
                reduce_idx = src_i
            else:
                reduce_idx = src_partner

            if a[reduce_idx] < 1e-14:
                continue

            # Compute "pressure" at each position: how much increasing a[j]
            # would increase the top peaks
            pressure = np.zeros(n)
            for pk in peak_indices[:6]:
                pk_i = int(pk)
                ps = max(0, pk_i - n + 1)
                pe = min(n - 1, pk_i)
                if ps <= pe:
                    idx_p = np.arange(ps, pe + 1)
                    partners_p = pk_i - idx_p
                    # Increasing a[j] increases b[pk] by a[pk-j] (if valid)
                    for j_pos in range(len(idx_p)):
                        j = int(idx_p[j_pos])
                        p = int(partners_p[j_pos])
                        if 0 <= p < n:
                            if j == p:
                                pressure[j] += 2.0 * a[j]
                            else:
                                pressure[j] += a[p]

            pressure[reduce_idx] = np.inf  # Don't move to self

            # Find the destination with minimum pressure
            # Also prefer destinations that are "far" from peak-reducing positions
            dst = int(np.argmin(pressure))

            if dst == reduce_idx or np.isinf(pressure[dst]):
                # Try second best
                pressure_sorted = np.sort(pressure)
                if pressure_sorted[-1] == np.inf:
                    continue
                # Mask out inf and retry
                valid = np.isfinite(pressure)
                if valid.sum() < 10:
                    valid = np.ones(n, dtype=bool)
                    valid[reduce_idx] = False
                dst = int(np.argmin(np.where(valid, pressure, np.inf)))
                if np.isinf(pressure[dst]):
                    continue

            # Try different fractions
            found = False
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.4]:
                amt = a[reduce_idx] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_autocorr(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                # Try random destination
                for _ in range(3):
                    j = int(rng.integers(n))
                    if j == reduce_idx:
                        continue
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[reduce_idx] * frac
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break

        if not improved:
            stall += 1
            # Structured perturbation
            if stall % 100 == 0:
                noise_scale = min(0.05, 0.01 * stall / 100)
                pert = rng.normal(0, noise_scale, n) * best_a
                # Add some "shape" perturbation

                shape = rng.normal(0, noise_scale * 0.5) * np.sin(2 * np.pi * rng.uniform(1, 5) * x_grid)
                a_r = best_a + pert + shape * best_a
                a_r = np.maximum(a_r, 1e-12)

                m_r = fast_autocorr(a_r).max()

                    best_m = m_r
                    best_a = a_r.copy()
                a = a_r if m_r < best_m + 0.002 else best_a.copy()

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SurgShave ({0:.1f}s): J={1:.6f} stall={2}".format(elapsed, J, stall))

    return best_a, best_m

def spectral_init(n, seed=0):

    rng = np.random.default_rng(seed)

    phases = rng.uniform(0, 2 * np.pi, n_freqs)
    amp = np.ones(n_freqs) / np.sqrt(n)
    amp[0] = 1.0 / np.sqrt(n)
    # Make it conjugate symmetric
    A = np.zeros(fft_size, dtype=complex)
    A[:n_freqs] = amp * np.exp(1j * phases)
    A[n_freqs:] = np.conj(A[fft_size - np.arange(1, fft_size - n_freqs + 1)])[::-1]

    a = np.real(np.fft.ifft(A))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def parametric_init(n, seed=0):
    """Generate diverse parametric initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    candidates = []

    # Power laws
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        for beta_p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            a = (x + 0.01) ** alpha * (1 - x + 0.01) ** beta_p
            candidates.append(a)

    # Gaussians
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for s in [0.1, 0.15, 0.2]:
            a = np.exp(-((x - c) ** 2) / (2 * s ** 2))
            candidates.append(a)

    # Double Gaussian
    for w in [0.5, 1.0, 2.0]:
        a = np.exp(-((x - 0.3) ** 2) / 0.02) + w * np.exp(-((x - 0.7) ** 2) / 0.02)
        candidates.append(a)

    # Random smooth
    for _ in range(5):
        coeffs = rng.normal(0, 1, 10)
        a = np.zeros(n)
        for k, c in enumerate(coeffs):
            a += c * np.cos(2 * np.pi * k * x)
        a = np.maximum(a - a.min() + 0.01, 0)
        candidates.append(a)

    best_a = None
    best_m = np.inf
    for a in candidates:
        a = norm_sum1(a)
        m = fast_autocorr(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    return best_a, best_m

def lp_polish(a0, time_limit, verbose=True):
    """Frank-Wolfe LP polish for final convergence."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    t = 0
    no_improve = 0

    # Get a solver
    solver = None
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    active_set = set(int(k) for k in np.argsort(fast_autocorr(a))[::-1][:500])

    while time.monotonic() - t0 < time_limit and no_improve < 300:
        t += 1
        b = fast_autocorr(a)
        M_curr = b.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set
        constraint_list = sorted(active_set)
        if len(constraint_list) > 1500:
            margins = np.array([M_curr - b[k] for k in constraint_list])
            keep = np.argsort(margins)[:1500]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0 or K > 3000:
            constraint_list = sorted(int(k) for k in np.argsort(b)[::-1][:500])
            K = len(constraint_list)

        # Build constraint matrix A @ g <= M
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_i = int(k)
            i_start = max(0, k_i - n + 1)
            i_end = min(n, k_i + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = a[k_i - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Af = np.fft.rfft(a, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ag = np.fft.irfft(Af * Gf, fft_size)[:size]
        violations = conv_ag[conv_ag > M_curr + 1e-8]
        if len(violations) > 0:
            viol_idx = np.where(conv_ag > M_curr + 1e-8)[0]
            top_v = viol_idx[np.argsort(conv_ag[viol_idx])[::-1][:20]]
            for k in top_v:
                active_set.add(int(k))

        # Line search
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5), 1.0 / (t + 1), default_alpha * 2]
        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            alpha = min(max(alpha, 0.01), 1.0)
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * g_dir
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st > 1e-15:
            a_new = a_new / st
        a = a_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP t={0} ({1:.1f}s): J={2:.6f} |S|={3}".format(t, elapsed, J, len(active_set)))

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{0}]: {1:.6f} (n={2}, t={3:.1f}s)".format(label, sc, len(a), elapsed))
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    """
    Main search combining Nesterov gradient, surgical shaving, spectral init,
    and LP polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print("Time budget: {0:.1f}s".format(time_budget))

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={0}, score={1:.6f}".format(len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 7000
        best_seq, _ = parametric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print("Default init: n={0}, score={1:.6f}".format(len(best_seq), best_score))

    n0 = len(best_seq)

    # Phase 1: Quick surgery + Nesterov on parent's length
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Nesterov + Surgery (n={0}) ===".format(n0))
    t1 = min(remaining * 0.25, 300.0)
    if t1 > 30:
        # First quick surgery
        t1a = min(t1 * 0.3, 80.0)
        if t1a > 10:
            a_out, _ = surgical_peak_shaving(best_seq, t1a, top_k=15, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "surg_n{0}".format(n0), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        t1b = min(remaining * 0.2, 200.0)
        if t1b > 20:
            a_out, _ = nesterov_optimize(best_seq, t1b, top_k=30, beta_start=20.0, beta_end=400.0,
                                         verbose=True, seed=42)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n{0}".format(n0), t_global)

    # Phase 2: Spectral initialization + Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Spectral init + Nesterov ===")
    if remaining > 100:
        a_spec = spectral_init(n0, seed=1)
        a_spec, m_spec = parametric_init(n0, seed=1)
        # Combine: try spectral
        a_spec2 = spectral_init(n0, seed=2)
        m_spec2 = fast_autocorr(a_spec2).max()
        best_init = a_spec if m_spec < m_spec2 else a_spec2

        remaining = time_budget - (time.monotonic() - t_global)
        t2 = min(remaining * 0.2, 250.0)
        if t2 > 30:
            a_out, _ = nesterov_optimize(best_init, t2, top_k=25, beta_start=15.0, beta_end=300.0,
                                         verbose=True, seed=100)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral_n{0}".format(n0), t_global)

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_out, _ = surgical_peak_shaving(best_seq, min(remaining * 0.15, 80.0), top_k=10, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "surg_spectral", t_global)

    # Phase 3: Nesterov with different params on current best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Nesterov v2 (n={0}) ===".format(n_best))
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        a_out, _ = nesterov_optimize(best_seq, t3, top_k=20, beta_start=50.0, beta_end=1500.0,
                                     verbose=True, seed=200)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2_n{0}".format(n_best), t_global)

    # Phase 4: Surgery round
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Surgery (n={0}) ===".format(n_best))
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, _ = surgical_peak_shaving(best_seq, t4, top_k=10, verbose=True, seed=555)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "surg2_n{0}".format(n_best), t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length exploration ===")
    nearby_lengths = []
    for dn in [-500, -300, -100, 100, 300, 500, 800, 1200]:
        nl = n_best + dn
        if 5000 < nl <= 9000 and nl not in nearby_lengths:
            nearby_lengths.append(nl)

    for new_n in nearby_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("  Trying n={0}".format(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        remaining = time_budget - (time.monotonic() - t_global)
        t5 = min(remaining * 0.2, 150.0)
        if t5 > 20:
            a_out, _ = nesterov_optimize(a_long, min(t5 * 0.5, 80.0), top_k=25,
                                         beta_start=20.0, beta_end=300.0, verbose=False, seed=new_n % 1000)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n{0}".format(new_n), t_global)

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = surgical_peak_shaving(best_seq, min(remaining * 0.15, 60.0), top_k=10, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "surg_n{0}".format(new_n), t_global)

    # Phase 6: Final Nesterov with high beta (sharp)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final Nesterov (n={0}) ===".format(n_best))
    t6 = min(remaining * 0.3, 300.0)
    if t6 > 20:
        a_out, _ = nesterov_optimize(best_seq, t6, top_k=15, beta_start=100.0, beta_end=3000.0,
                                     verbose=True, seed=999)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 7: Final surgery
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Surgery (n={0}) ===".format(n_best))
    t7 = min(remaining * 0.2, 150.0)
    if t7 > 15:
        a_out, _ = surgical_peak_shaving(best_seq, t7, top_k=8, verbose=True, seed=1234)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "surg_final", t_global)

    # Phase 8: LP Polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: LP Polish (n={0}) ===".format(n_best))
    t8 = min(remaining * 0.5, 300.0)
    if t8 > 30 and n_best <= 8000:
        a_out, _ = lp_polish(best_seq, t8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 9: Final Nesterov burst
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print("\n=== Phase 9: Final burst (n={0}) ===".format(n_best))
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 10:
            a_out, _ = nesterov_optimize(best_seq, t9, top_k=10, beta_start=200.0, beta_end=5000.0,
                                         verbose=False, seed=7777)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "burst_final", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={0:.6f}, n={1}, time={2:.1f}s".format(best_score, len(best_seq), elapsed))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 7000
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1 - x + 0.01) ** 1.0
        best_seq = norm_sum1(a_raw)
        best_score = evaluate_safe(best_seq)

    # Ensure we return a valid list
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
