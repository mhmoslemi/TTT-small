def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]

def compute_smoothed_grad(y, beta, top_k):
    """Compute gradient of (1/beta)*log(sum top_k exp(beta*c)) w.r.t. a elements."""
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    Y_rev = np.fft.rfft(a_rev, fft_size)
    Conv = np.fft.irfft(P * Y_rev, fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c


def nesterov_opt(a0, time_limit, beta_start=30.0, beta_end=3000.0,
                 beta_growth=1.006, eta_base=0.6, top_k_ratio=0.02,
                 restart_every=2000, verbose=True, seed=42):

    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()

    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        eta = eta_base / max(1.0, beta * p.max() * 2.0)
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0

            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.04 / (1.0 + 0.08 * (t // restart_every))

            rand = rng.normal(0, 1, n)

            x = x_r

            for k2 in peak_idx[:min(6, n_pk)]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True
            candidates = np.where(~in_peak)[0]
            if len(candidates) < 20:
                candidates = np.arange(n)
            mask = candidates != from_i
            if not mask.any():
                continue
            dst = int(candidates[mask][int(rng.integers(mask.sum()))])
            trial[from_i] -= amt
            trial[dst] += amt
        elif move_type < 0.6:
            # Pair swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                trial[i], trial[j] = trial[j], trial[i]
        elif move_type < 0.8:
            # Block shift
            blk_len = int(rng.integers(3, 20))
            if blk_len < n:
                start = int(rng.integers(n - blk_len + 1))
                direction = 1 if rng.random() < 0.5 else -1
                shift = int(rng.integers(1, 5))
                if direction == 1:
                    if start + blk_len + shift <= n:
                        block = trial[start:start + blk_len].copy()
                        trial[start:start + blk_len + shift] = np.concatenate([trial[start + blk_len:start + blk_len + shift], block])
                else:
                    if start - shift >= 0:
                        block = trial[start:start + blk_len].copy()
                        trial[start - shift:start + blk_len] = np.concatenate([block, trial[start:start - shift + blk_len]])
        elif move_type < 0.9:
            # Proportional transfer between two random positions
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                trial[i] -= amt
                trial[j] += amt
        else:
            # Gaussian perturbation on a random subset
            n_pert = int(rng.integers(max(5, n // 100), max(10, n // 20)))
            positions = rng.choice(n, size=n_pert, replace=False)
            noise_scale = rng.uniform(0.01, 0.05)
            trial[positions] += noise_scale * rng.normal(0, 1, n_pert) * a[positions].mean()

        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-15:
            continue
        trial = trial / s_t

        trial_m = fast_autocorr(trial).max()
        eval_count += 1

        # Metropolis acceptance
        delta = trial_m - c_max
        if delta < 0:
            a = trial
        else:
            if T > 1e-15:
                # P = exp(-delta / T), but cap to avoid underflow
                ratio = -min(delta / T, 50.0)
                if rng.random() < np.exp(ratio):
                    a = trial

        # Cool down
        T = T * cooling

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SA it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", T=" + repr(round(T, 8)) + ", accept_rate=" + repr(round(accept_count / max(1, it), 3)))
            accept_count = 0

    return best_a, best_m


def spectral_flatten(a0, time_limit, verbose=True, seed=555):
    """
    Iterative spectral peak reduction.
    In frequency domain, reduce amplitude at frequencies corresponding to peak lags.
    """
    n = len(a0)
    fft_size = 1 << (2 * n - 1).bit_length()
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_png = np.random.default_rng(seed)

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_autocorr(a)
        M = c.max()
        if M < best_m - 1e-16:
            best_m = M
            best_a = a.copy()

        # Find top peaks
        n_pk = min(8, len(c))
        peak_idx = np.argsort(c)[::-1][:n_pk]

        # Get frequency representation
        A = np.fft.rfft(a, fft_size)
        A_phase = np.angle(A)
        A_mag = np.abs(A)

        # For each top peak at lag k, the corresponding frequency is 2*pi*k/(2*n)
        # Reduce amplitude at those frequencies
        for k in peak_idx[:5]:
            k_int = int(k)
            # The frequency index in the rfft array corresponding to lag k
            f_idx = int(round(k_int * fft_size / (2 * n)))
            for df in range(-3, 4):
                fi = f_idx + df
                if 0 <= fi < len(A_mag):
                    reduction = 0.02 * (1 - abs(df) / 4.0)
                    A_mag[fi] *= (1.0 - reduction)
                    # Also reduce the symmetric frequency
                    fi_sym = len(A_mag) - 1 - fi
                    if 0 <= fi_sym < len(A_mag) and fi_sym != fi:
                        A_mag[fi_sym] *= (1.0 - reduction * 0.5)

        # Add a small random perturbation to avoid local minima
        if rng.random() < 0.1:
            A_mag += rng.uniform(0, 0.001, len(A_mag)) * A_mag

        # Reconstruct
        A_new = A_mag * np.exp(1j * A_phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
            a = a_new
        else:
            break

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def peak_surgery(a0, time_limit, verbose=True, seed=77):
    """Targeted local search: identify top peaks and reduce via mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 4000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_top = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]

        improved = False
        found_move = False

        for k in top_idx[:12]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue

            src_i = int(idx[np.argmax(products)])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue

            if a[src_i] >= a[partner]:
                reduce_i = src_i
            else:
                reduce_i = partner

            if a[reduce_i] < 1e-14:
                continue

            # Compute pressure
            pressure = np.zeros(n)
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 50:
                candidates = np.arange(n)
            mask = candidates != reduce_i
            if not mask.any():
                continue
            candidates = candidates[mask]

            cand_pressure = pressure[candidates]
            n_try = min(4, len(candidates))
            try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]

            for dst in try_dsts:
                dst_int = int(dst)
                for frac in [0.01, 0.02, 0.03, 0.05]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        break
                if found_move:
                    break

        if not improved:
            if no_improve > 500:
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.06)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgery (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    """Fine-grained pairwise coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 95)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:15]

        improved = False

        for k in peak_positions[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            top_pairs = np.argsort(products)[::-1][:3]
            for pi in top_pairs:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue

                larger_i = src_i if a[src_i] >= a[partner] else partner
                if a[larger_i] < 1e-14:
                    continue

                for frac in [0.001, 0.003, 0.005]:
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(4):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 150 == 0:
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.005 / (1.0 + stall / 150.0)
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(repr(sc)) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """Main search: multi-start + Nesterov + SA + spectral + peak surgery + pairwise CD."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6500
        x = np.linspace(0, 1, n_base)
        a_init = (1.0 - x) ** 1.2 + 0.05
        best_seq = norm_sum1(a_init)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(best_score))

    # Phase 0: Multi-start exploration with different shapes
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 0: Multi-start exploration ===")
    candidate_shapes = [
        "power1.5", "gauss", "exp", "tri", "sinc_like"
    ]
    for shape_name in candidate_shapes:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 10:
            break
        x = np.linspace(0, 1, n0)
        if shape_name == "power1.5":
            a_test = (1.0 - x) ** 1.5 + 0.03
        elif shape_name == "gauss":
            a_test = np.exp(-4.0 * (x - 0.3) ** 2) + 0.02
        elif shape_name == "exp":
            a_test = np.exp(-3.0 * x) + 0.02
        elif shape_name == "tri":
            a_test = np.maximum(1.0 - np.abs(2.0 * x - 0.5), 0.0) + 0.02
        else:  # sinc_like
            t_sinc = np.pi * (2 * x - 1)
            with np.errstate(divide='ignore', invalid='ignore'):
                a_test = np.sin(t_sinc) / t_sinc
            a_test = np.nan_to_num(a_test, 0.0)
            a_test = np.maximum(a_test, 0.0) + 0.02

        a_test = norm_sum1(a_test)
        sc = evaluate_safe(a_test)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_test.copy()
            print("  Multi-start [" + shape_name + "]: " + repr(sc))
        # Short Nesterov on this shape
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and sc < best_score * 1.05:
            t_ms = min(remaining * 0.15, 120.0)
            if t_ms > 30:
                a_out, _ = nesterov_opt(a_test, t_ms,
                                        beta_start=30.0, beta_end=2000.0,
                                        beta_growth=1.005, eta_base=0.5,
                                        top_k_ratio=0.02, restart_every=1500,
                                        verbose=False, seed=hash(shape_name) % 10000)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "ms_" + shape_name, t_global)

    # Phase 1: Main Nesterov (high beta, focus on max)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 1: Nesterov main (n=" + str(n_best) + ") ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t1,
                                beta_start=40.0, beta_end=4000.0,
                                beta_growth=1.007, eta_base=0.5,
                                top_k_ratio=0.015, restart_every=1500,
                                verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 2: Simulated Annealing (NEW - key differentiator)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Simulated Annealing (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 60:
        a_out, _ = simulated_annealing(norm_sum1(best_seq), t2,
                                       T0_frac=0.04, cooling=0.9996,
                                       n_top_peaks=10, verbose=True, seed=99)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sa1", t_global)

    # Phase 3: Nesterov refine (moderate beta)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Nesterov refine (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 60:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t3,
                                beta_start=100.0, beta_end=6000.0,
                                beta_growth=1.009, eta_base=0.4,
                                top_k_ratio=0.01, restart_every=1200,
                                verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 4: Spectral flattening (NEW)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Spectral Flattening (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 30:
        a_out, _ = spectral_flatten(norm_sum1(best_seq), t4, verbose=True, seed=555)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 5: Peak surgery
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Peak Surgery (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 40:
        a_out, _ = peak_surgery(norm_sum1(best_seq), t5, verbose=True, seed=22)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps1", t_global)

    # Phase 6: Second SA pass (different seed, higher temperature)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: SA Pass 2 (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 40:
        a_out, _ = simulated_annealing(norm_sum1(best_seq), t6,
                                       T0_frac=0.03, cooling=0.9997,
                                       n_top_peaks=8, verbose=False, seed=200)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sa2", t_global)

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Multi-length (n_best=" + str(n_best) + ") ===")
    lengths_to_try = [n_best + 800, n_best + 2000, max(4000, n_best - 1500)]
    lengths_to_try = [l for l in lengths_to_try if l <= 12000 and l >= 3000]

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.3, 150.0)
        if t7a > 30:
            a_out, _ = nesterov_opt(a_long, t7a,
                                    beta_start=30.0, beta_end=2000.0,
                                    beta_growth=1.006, eta_base=0.5,
                                    top_k_ratio=0.015, restart_every=1500,
                                    verbose=False, seed=3)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t7b = min(remaining * 0.3, 100.0)
            if t7b > 20:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = peak_surgery(a_try, t7b, verbose=False, seed=33)

        a_out, _ = pairwise_cd(norm_sum1(best_seq), t8, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd1", t_global)

    # Phase 9: Final Nesterov burst
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t9 = min(remaining * 0.35, 200.0)
        if t9 > 30:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t9,
                                    beta_start=200.0, beta_end=10000.0,
                                    beta_growth=1.012, eta_base=0.3,
                                    top_k_ratio=0.005, restart_every=1000,
                                    verbose=True, seed=4)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Phase 10: Final pairwise CD + SA micro
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 10: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 40:
        t10a = min(remaining * 0.4, 120.0)
        if t10a > 20:
            a_out, _ = pairwise_cd(norm_sum1(best_seq), t10a, verbose=False, seed=66)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t10b = min(remaining * 0.5, 80.0)
            if t10b > 20:
                a_out, _ = simulated_annealing(norm_sum1(best_seq), t10b,
                                               T0_frac=0.015, cooling=0.9999,
                                               n_top_peaks=5, verbose=False, seed=999)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "sa_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t10c = min(remaining * 0.6, 60.0)
            if t10c > 15:
                a_out, _ = nesterov_opt(norm_sum1(best_seq), t10c,
                                        beta_start=500.0, beta_end=15000.0,
                                        beta_growth=1.015, eta_base=0.2,
                                        top_k_ratio=0.004, restart_every=800,
                                        verbose=False, seed=7)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_ultimate", t_global)

    # Validation and cleanup
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6500
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
