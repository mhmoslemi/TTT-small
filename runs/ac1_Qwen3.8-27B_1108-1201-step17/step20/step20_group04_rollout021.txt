def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

    n = len(a)

def fourier_shake(a, n_modes=5, eps_base=0.03, rng=None):
    """Apply a random Fourier mode perturbation."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    modes = rng.choice(np.arange(1, n//2+1), size=min(n_modes, n//2), replace=False)
    phases = rng.uniform(0, 2*np.pi, size=len(modes))
    amps = rng.uniform(0.5, 1.0, size=len(modes))
    pert = np.zeros(n)
    for i, m in enumerate(modes):
        pert += amps[i] * np.sin(2*np.pi*m*np.arange(n)/n + phases[i])
    pert -= pert.mean()
    pert = eps_base * pert * a
    a_new = a + pert
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
