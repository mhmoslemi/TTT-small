over non-negative sequences.

Uses exact 1D coordinate optimization with fast rank-1 convolution updates,
supplemented by log-sum-exp smoothed L-BFGS for escaping local minima,

import numpy as np

def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s

def compute_score(a, c, n, S):
    """Compute 2*n*max_c/S^2 quickly."""
    if S < 1e-15:
        return 1e100
    max_c = float(c.max())
    if max_c < 0:
        max_c = 0.0
    return 2.0 * n * max_c / (S * S)


def logsumexp_score(a, c, beta, n, S):
    """Compute log-sum-exp smoothed score."""
    if S < 1e-15:
        return 1e100
    m = float(c.max())
    if m < 0:
        return 0.0
    shifted = beta * (c - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0:
        se = 1.0
    lse = m + math.log(se) / beta
    return 2.0 * n * lse / (S * S)


def optimize_coord(a, c, s, n, S, max_fun=30):
    """
    Find optimal value for a[s] given all else fixed.
    Uses bounded scalar minimization of the score function.
    Returns (new_value, new_score, old_score).
    """
    old = float(a[s])
    S_old = S
    
    k_lo = s
    k_hi = s + n - 1  # always <= 2n-2 since s <= n-1
    
    # Precompute unaffected max
    max_unaff = 0.0
    if k_lo > 0:
        # c[0:k_lo] is unaffected
        if k_lo > 0:
            max_unaff = float(c[:k_lo].max())
    if k_hi < 2 * n - 2:
        max_unaff = max(max_unaff, float(c[k_hi + 1:].max()))
    
    is_2s_in = (k_lo <= 2 * s <= k_hi)
    pos_2s = 2 * s - k_lo if is_2s_in else -1
    c_2s = float(c[2 * s])
    
    # The affected convolution values at current v=old:
    # c_aff[k] + 2*(v-old)*a[k-s] for k in [k_lo, k_hi]
    # At v=old: just c_aff
    c_aff_base = c[k_lo:k_hi+1].copy()  # length n
    
    def score_at(v):
        S_new = S_old - old + v
        if S_new <= 1e-15:
            return 1e100
        delta = v - old
        if abs(delta) < 1e-14:
            max_c_aff = float(c_aff_base.max())

                max_c_aff = max(max_c_aff, c_2s + v * v - old * old)
        else:
            # Vectorized: c_aff_base + 2*delta*a

                val_2s = c_2s + v * v - old * old
                max_c_aff = max(max_c_aff, val_2s)
        max_c = max(max_c_aff, max_unaff)
        if max_c < 0:
            max_c = 0.0
        return 2.0 * n * max_c / (S_new * S_new)
    
    old_score = score_at(old)
    
    # Search bounds
    a_max = float(a.max())
    lo = 0.0
    hi = max(3.0 * old, 1.5 * a_max) if old > 1e-10 else max(0.5 * a_max, 1e-6)
    if hi < 1e-8:
        hi = 0.1
    
    best_v = old
    best_score = old_score
    
    if SCIPY_OK:

                best_score = sc
                best_v = float(v)

        if 0 < best_v < hi:
            idx = int((best_v - lo) / (hi - lo) * 19)
            local_lo = max(lo, grid[max(0, idx-1)])
            local_hi = min(hi, grid[min(19, idx+1)])
            fine_grid = np.linspace(local_lo, local_hi, 15)
            for v in fine_grid:
                sc = score_at(float(v))
                if sc < best_score:
                    best_score = sc
                    best_v = float(v)
    
    return best_v, best_score, old_score


def coord_descent(a, c, n, S, order=None, max_sweeps=50, verbose=False, t_start=None, time_limit=None):
    """
    Perform coordinate descent sweeps.
    Returns improved (a, c, n, S).
    """
    if order is None:
        order = np.arange(n)
    
    best_score = compute_score(a, c, n, S)
    best_a = a.copy()
    no_improve = 0
    sweep = 0
    
    while sweep < max_sweeps:
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit - 10:
                break
        
        sweep += 1
        sweep_improved = False
        current_score = best_score
        
        for s in order:
            new_val, new_score, old_score = optimize_coord(a, c, s, n, S)
            delta = new_val - a[s]
            if abs(delta) > 1e-12:
                # Update a
                a[s] = new_val
                S += delta
                # Update c
                if s + n - 1 <= 2 * n - 2:
                    c[s:s+n] += 2.0 * delta * a  # a[k-s] for k in [s, s+n-1]
                else:
                    end = 2 * n - 1  # exclusive
                    c[s:end] += 2.0 * delta * a[:end-s]
                c[2*s] += delta * delta
                c = np.where(c > 0, c, 0)  # numerical safety
                
                if new_score < current_score - 1e-14:
                    current_score = new_score
                    sweep_improved = True
        
        if sweep_improved and current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            a = best_a.copy()
            c = fft_conv(a)
            S = float(a.sum())
            no_improve += 1
            if no_improve >= 5:
                break
        
        if verbose and (sweep <= 3 or sweep % 10 == 0):
            print(f"    CD sweep {sweep}: score={best_score:.10f}, S={S:.4f}")
    
    return best_a, best_score

    # = (2/beta) * cross[j]  where cross[j] = sum_i a_i * p_{i+j}  (i from 0 to n-1, i+j from j to j+n-1)
    
    # And cross[j] = sum_i a_i * p_{i+j} can be computed as:
    # Take the FFT of a (padded to N), take the FFT of p (padded to N)
    # Then cross-correlation = IFFT(FFT(a) * conj(FFT(p)))
    # But I need to verify the convention.
    
    # Let me just test with a small example in my head:
    # a = [a0, a1], p = [p0, p1, p2]
    # cross[0] = a0*p0 + a1*p1
    # cross[1] = a0*p1 + a1*p2
    # 
    # Circular cross-correlation: IFFT(FFT_a * conj(FFT_p))
    # With N=4 (padded): a_p = [a0, a1, 0, 0], p_p = [p0, p1, p2, 0]
    # FFT_a * conj(FFT_p), then IFFT gives:
    # corr[k] = sum_i a_p[i] * p_p[(i-k) mod 4]
    # corr[0] = a0*p0 + a1*p1 = cross[0] ✓
    # corr[3] = a0*p1 + a1*p2 ... let's check: (i-3) mod 4 for i=0 gives 1, for i=1 gives 2
    # corr[3] = a0*p1 + a1*p2 = cross[1] ✓
    # So cross[j] = corr[N-1-j] for j=1, and cross[j] = corr[j] for j=0?
    # Actually corr[0] = cross[0], corr[3] = cross[1]
    # In general: cross[j] = corr[(N-j) mod N] = corr[N-j] for j > 0, corr[0] for j=0.
    
    # So: cross = [corr[0], corr[N-1], corr[N-2], ..., corr[N-n+1]]
    
    # Great, this is efficient!
    
    # But wait, I need N >= n + (2n-1) - 1 = 3n - 2 for the linear cross-correlation
    # to be correct (no wrap-around).
    
    a_padded = np.zeros(N, dtype=np.float64)
    a_padded[:n] = a
    p_padded = np.zeros(N, dtype=np.float64)
    p_padded[:2*n-1] = p
    
    A_fft = np.fft.rfft(a_padded, N)
    P_fft = np.fft.rfft(p_padded, N)
    corr = np.fft.irfft(A_fft * np.conj(P_fft), N)
    
    # cross[j] = corr[j] for j=0, corr[N-j] for j=1,...,n-1
    cross = np.zeros(n, dtype=np.float64)
    cross[0] = corr[0]
    if n > 1:
        cross[1:] = corr[N - 1:N - n:-1]  # corr[N-1], corr[N-2], ..., corr[N-n+1]
    
    grad_a = (2.0 / beta) * cross
    if not np.all(np.isfinite(grad_a)):
        grad_a = np.zeros(n, dtype=np.float64)
    
    return f, grad_a


def lbfgs_optimize(a_init, beta, maxfun=200):
    """L-BFGS optimization in softmax coordinates."""
    if not SCIPY_OK:
        return a_init
    
    a = norm_sum1(a_init)
    n = a.size
    if n <= 2:
        return a
    
    log_a = np.log(np.maximum(a, 1e-30))
    
    def obj_grad(u_reduced):
        u = np.empty(n, dtype=np.float64)
        u[:n-1] = u_reduced
        u[n-1] = 0.0
        u_max = float(np.max(u))
        e = np.exp(u - u_max)
        s = float(e.sum())
        if s <= 0 or not np.isfinite(s):
            return 1e100, np.zeros(n-1, dtype=np.float64)
        a_local = e / s
        
        f, grad_a = lse_obj_grad(a_local, beta)
        
        # Convert grad_a to grad_u
        # da_i/du_j = a_i*(delta_ij - a_j)  (for the reduced parameterization with u[n-1]=0)
        # d(f)/du_j = sum_i grad_a_i * da_i/du_j
        # = sum_i grad_a_i * a_i * (delta_ij - a_j)
        # = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i  (for j < n-1)
        # Note: for j = n-1 (the reference), we don't optimize, so we only need j in [0, n-2]
        
        mean_ga = float(np.dot(grad_a, a_local))
        grad_u = a_local[:n-1] * (grad_a[:n-1] - mean_ga)
        
        if not np.all(np.isfinite(grad_u)):
            grad_u = np.zeros(n-1, dtype=np.float64)
        if not np.isfinite(f):
            f = 1e100
        return f, grad_u
    
    u0 = log_a[:n-1].copy()
    
    try:
        res = scipy.optimize.minimize(
            obj_grad, u0,
            jac=True, method='L-BFGS-B',
            options={'maxfun': maxfun, 'maxiter': maxfun + 50,
                     'ftol': 1e-14, 'gtol': 1e-9, 'disp': False}
        )
        if hasattr(res, 'fun') and np.isfinite(res.fun):
            u_new = np.empty(n, dtype=np.float64)
            u_new[:n-1] = res.x
            u_new[n-1] = 0.0
            u_max = float(np.max(u_new))
            e = np.exp(u_new - u_max)
            a_new = e / float(e.sum())
            a_new = np.where(np.isfinite(a_new) & (a_new > 0), a_new, 0.0)
            s = float(a_new.sum())
            if s > 1e-15:
                return a_new / s
    except Exception as e:
        print(f"    L-BFGS exception: {e}")
    
    return a_init
