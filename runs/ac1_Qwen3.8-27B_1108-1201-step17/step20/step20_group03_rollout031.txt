def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_grad(a, beta, top_k=None):
    """Compute gradient of log-sum-exp smoothed max autocorrelation.
    
    f_beta(a) = (1/beta) * log(sum_k exp(beta * b_k))
    grad[j] = (2/beta) * sum_m p[j+m] * a[m]
    
    This equals (2/beta) * convolve(p[::-1], a)[2n-2 : 3n-2]
    """
    n = len(a)
    b = fast_autocorr(a)
    
    if top_k is not None and top_k < len(b) and top_k > 0:
        # Focus on top peaks for better conditioning
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(len(b))
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()
    
    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        # Uniform over the selected peaks
        p = np.zeros(len(b_use))
        peak_mask = (b_use == b_max)
        p[peak_mask] = 1.0 / max(1, peak_mask.sum())
    else:
        p = exp_vals / p_total
    
    # grad[j] = (2/beta) * convolve(p[::-1], a)[2n-2 + j]
    fft_size = 1 << (len(p) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p[::-1], fft_size)
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(P * A, fft_size)
    grad = (2.0 / beta) * conv[2*n-2 : 3*n-2]
    
    return grad, p, b

def fista_optimize(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   beta_growth=1.003, top_k=20, restart_every=4000, verbose=False, seed=42):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    avg_x = x.copy()
    avg_count = 0.0
    s_prev = None
    y_prev = None
    bb_step = 1.0 / max(1.0, np.sqrt(n))
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov acceleration
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y
        
        # Compute gradient
        grad, p, b = compute_grad(y, beta, top_k)
        
        # Barzilai-Borwein step size
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_candidate = sy_gy / gy_gy
                if bb_candidate > 1e-8 and bb_candidate < 10.0:
                    bb_step = max(1e-6, min(bb_candidate, 5.0 / beta))
                else:
                    bb_step = max(bb_step * 0.9, 1e-6)
            s_prev = grad.copy()
            y_prev = y.copy()
        
        # Gradient step + projection
        eta = min(bb_step, 0.2)
        z = y - eta * grad
        x_new = simplex_project(z)
        
        # Evaluate
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        # Nesterov averaging
        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count
        
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        t_nest = t_new
        
        # Adaptive beta
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        # Restart
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise = 0.02 / (1.0 + 0.1 * (t // restart_every))
            pert = rng.normal(0, noise, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 5.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / max(1.0, np.sqrt(n))
            if verbose:
                print(f"    FISTA restart t={t} beta={beta:.1f} J={2*n*best_max_b:.6f}")
        
        if verbose and t % 10000 == 0:
            print(f"    FISTA t={t} ({time.monotonic()-t0:.1f}s) J={2*n*best_max_b:.6f} beta={beta:.1f}")
    
    # Check averaged solution
    J_avg = 2.0 * n * fast_autocorr(avg_x).max()
    J_best = 2.0 * n * best_max_b
    if J_avg < J_best * 0.999:
        return avg_x, fast_autocorr(avg_x).max()
    return best_a, best_max_b

def lp_fw_optimize(a0, time_limit, solver, initial_K=300, max_K=2000, add_per_iter=20, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 500
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:

def two_exchange_search(a0, time_limit, verbose=False, seed=999):

def cyclic_shift_best(a0):
    n = len(a0)
    a = norm_sum1(a0)
    best_m = fast_autocorr(a).max()
    best_shift = 0
    b = fast_autocorr(a)
    
    # Try all shifts using FFT (fast)
    fft_size = 1 << (2*n-1).bit_length()
    A_base = np.fft.rfft(a, fft_size)
    
    for s in range(1, n):
        a_shift = np.roll(a, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b_s = np.fft.irfft(AS * AS, fft_size)[:2*n-1]
        m = b_s.max()
        if m < best_m:
            best_m = m
            best_shift = s
    
    if best_shift > 0:
        return np.roll(a, -best_shift), best_m
    return a.copy(), best_m

def spectral_reshape(a0, time_limit, n_iters=300, verbose=False):
    """Iteratively shape the power spectrum to flatten the autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2*n-1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        b = np.fft.irfft(power, fft_size)[:2*n-1]
        b_max = b.max()
        
        if b_max < 1e-15:
            break
        
        # Compute target: reduce peaks in autocorrelation
        # Modify power spectrum at frequency bins that contribute to peaks
        b_norm = b / b_max
        # The autocorrelation peak at lag k corresponds to modulation at frequency 2πk/(2n)
        # Reduce power at frequencies that create the peaks
        freqs = np.arange(len(power))
        # Reduce power proportional to how much each frequency contributes to peaks
        # Simple approach: reduce power where autocorrelation is high
        reduction = 1.0 - 0.5 * b_norm[freqs % len(b_norm)]
        reduction = np.maximum(reduction, 0.1)
        
        A_new = A * np.sqrt(reduction)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        # Soft update (don't fully commit)
        a = 0.7 * a + 0.3 * a_new
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        if verbose and it % 50 == 0:
            print(f"    Spectral it={it}: J={2*n*best_m:.6f}")
    
    return best_a, best_m

def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update on simplex."""
    z = a * np.exp(-eta * np.clip(grad, -50, 50))
    s = z.sum()
    if s < 1e-15:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
