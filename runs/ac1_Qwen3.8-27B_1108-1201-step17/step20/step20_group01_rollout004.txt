```python
def fast_conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smoothed_obj_and_grad(a, T):
    """Compute smoothed objective (1/T) * log(sum exp(T*c)) and its gradient."""
    n = len(a)

    Tc = T * c
    Tc_shifted = Tc - Tc.max()  # for numerical stability
    log_sum = Tc.max() + np.log(np.sum(np.exp(Tc_shifted)))
```

```python
def lp_fw_step(a, active_lags, c):
    n = len(a)
    M_curr = c.max()
    
    # Build constraint matrix A (K x n): A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    K = len(active_lags)
    A = np.zeros((K, n))
    for ii, ki in enumerate(active_lags):
        ki = int(ki)
        # A[ii, j] = a[ki - j] if 0 <= ki-j < n
        # j ranges from max(0, ki-n+1) to min(n-1, ki)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            A[ii, j_start:j_end+1] = a[ki - np.arange(j_start, j_end+1)]
    
    # LP: min M
    # s.t. sum(g) = 1, g >= 0, A@g - M <= 0
    # Variables: [g_0, ..., g_{n-1}, M]
    # c_obj = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: A@g - M <= 0 → [A, -1] @ [g, M] <= 0
    A_ub = np.hstack([A, np.ones((K, 1)) * -1])
    b_ub = np.zeros(K)
    
    # Equality: sum(g) = 1 → [1, ..., 1, 0] @ [g, M] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_i in [0, inf), M in (-inf, inf)
    bounds = [(0, None)] * n + [(None, None)]
    
    result = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        g = np.maximum(result.x[:n], 0)
        g = g / g.sum()
        return g
    return None
```
