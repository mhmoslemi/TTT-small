def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def simplex_project(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_topk_grad(y, beta, top_k):
    """Compute gradient using exact top-K selection + softmax for the rest."""
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1
    
    if top_k >= size:
        # Just use full softmax
        shifted = beta * (b - b.max())
        shifted = np.clip(shifted, -500, 0)
        exp_vals = np.exp(shifted)
        s = exp_vals.sum()
        p = exp_vals / s
    else:
        # Use top-K with softmax weighting
        idx = np.argsort(b)[::-1]
        top_idx = idx[:top_k]
        rest_idx = idx[top_k:]
        
        b_max = b[top_idx[0]]
        
        # For top K, use softmax
        top_shifted = beta * (b[top_idx] - b_max)
        top_exp = np.exp(top_shifted)
        top_s = top_exp.sum()
        
        # For rest, use a single weight
        rest_shifted = beta * (b[rest_idx] - b_max)
        rest_exp = np.exp(rest_shifted)
        rest_s = rest_exp.sum()
        
        p = np.zeros(size)
        p[top_idx] = top_exp / (top_s + rest_s)
        if len(rest_idx) > 0:
            p[rest_idx] = rest_exp / (top_s + rest_s)
    
    # Gradient via FFT
    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, b

    stall = 0

            break
        
        # Compute gradient
        grad, b = compute_topk_grad(x, beta, top_k)
        
        # Adam update
        t_adam += 1
        m_v = momentum * m_v + (1 - momentum) * grad
        v_v = momentum * v_v + (1 - momentum) * grad * grad
        m_hat = m_v / (1 - momentum ** t_adam)
        v_hat = v_v / (1 - momentum ** t_adam)
        
        # Project gradient step
        z = x - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        x_new = simplex_project(z)
        
        # Evaluate
        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        
        if max_new < best_m - 1e-15:

            stall = 0

                lr = max(lr * 0.95, 0.001)

                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contrib = best_a[idx] * best_a[peak_k - idx]
                # Reduce the contributing elements
                x_r = best_a.copy()
                for j in idx[np.argsort(contrib)[::-1][:5]]:
                    x_r[j] *= 0.9
                # Redistribute
                x_r[js:je+1] = x_r[js:je+1] * 1.1  # boost neighbors

def spectral_refine_v3(a0, time_limit, verbose=True, seed=777):
    """Spectral domain optimization with adaptive flattening and phase fixing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    alpha = 0.03
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Compute target: geometric mean of non-DC power
        if len(power) > 1:
            log_powers = np.log(power[1:] + 1e-30)
            target_log = np.median(log_powers)
            target = np.exp(target_log_powers)
        else:
            target = power[0] if power[0] > 0 else 1.0
        
        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * min(ratio - 1.0, 3.0)
                A_new[k] *= max(0.2, np.sqrt(red))
            elif ratio < 0.3:
                inc = 1.0 + alpha * 0.2 * (0.3 - ratio)
                A_new[k] *= min(1.5, np.sqrt(inc))
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 100 == 0:
            # Random phase perturbation
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.01
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial
        
        if it % 300 == 0:
            alpha = max(0.005, alpha * 0.97)
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  SpecV3 it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
    
    return best_a, best_m

def lp_fw_v3(a0, time_limit, solver, initial_K=200, max_K=3000,
             add_per_iter=20, verbose=True):
    """LP-Frank-Wolfe with improved active set management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
    
    no_improve = 0
    max_no_improve = 800
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Manage active set
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_list = sorted(active_set)
            active_set = set(int(active_list[i]) for i in keep)
        
        # Ensure all near-max are in active set
        near_max = np.where(ac >= M_curr * 0.95)[0]
        for k in near_max:
            active_set.add(int(k))
        
        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build coefficient matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check for violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))
        
        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        for alpha in [best_alpha, best_alpha * 0.5, best_alpha * 1.5, 0.05, 0.1, 0.15, 0.2, 0.3]:
            alpha = max(0.005, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP-FW3 it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))
    
    return best_a, best_m

def peak_shaving(a0, time_limit, verbose=True, seed=42):
    """Targeted peak shaving: directly reduce the largest autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        top_ks = np.argsort(ac)[::-1][:15]
        improved = False
        
        for k in top_ks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue
            
            # Find positions with high contribution
            top_contrib = idx[np.argsort(contrib)[::-1][:3]]
            
            for src in top_contrib:
                if time.monotonic() - t0 > time_limit:
                    break
                src_int = int(src)
                partner = k_int - src_int
                if not (0 <= partner < n):
                    continue
                
                # Try moving mass from src to a position that doesn't create peaks
                frac = 0.02
                amt = a[src_int] * frac
                if amt < 1e-15:
                    continue
                
                # Try several destinations
                for _ in range(8):
                    dst = int(rng.integers(n))
                    if dst == src_int or dst == partner:
                        continue
                    
                    a_trial = a.copy()
                    a_trial[src_int] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                # More aggressive random moves
                for _ in range(10):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                                break
        
        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  PeakShave (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))
    
    return best_a, best_m

def try_update(best_seq, best_score, a, label, t_global):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_grad_exact_topk(y, beta, top_k):
    """Compute gradient using exact top-K peaks with softmax blending."""
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1

    if top_k >= size:
        shifted = beta * (b - b.max())
        shifted = np.clip(shifted, -600, 0)
        exp_vals = np.exp(shifted)
        s = exp_vals.sum()
        if s < 1e-300:
            p = np.ones(size) / size
        else:
            p = exp_vals / s
    else:
        idx = np.argsort(b)[::-1]
        top_idx = idx[:top_k]
        rest_idx = idx[top_k:]

        b_max = b[top_idx[0]]

        # Softmax over top K
        top_shifted = beta * (b[top_idx] - b_max)
        top_exp = np.exp(top_shifted)
        top_s = top_exp.sum()

        # Uniform weight for rest (avoids computation)
        rest_weight = min(len(rest_idx) * 0.1, top_s * 0.5)

        p = np.zeros(size)
        total = top_s + rest_weight
        p[top_idx] = top_exp / total
        if len(rest_idx) > 0:
            p[rest_idx] = rest_weight / (len(rest_idx) * total)

    # Gradient via FFT: grad_j = (1/beta) * sum_k p_k * (a_rev convolved with p)[...]
    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, b


def adam_optimize(a0, time_limit, beta_start=100.0, beta_end=3000.0,
                  beta_growth=1.004, lr_init=0.008, top_k=20,
                  restart_every=3000, verbose=True, seed=42,
                  momentum1=0.9, momentum2=0.999, epsilon=1e-8):
    """Adam-based optimization on the simplex with smoothed max objective."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    lr = lr_init
    rng = np.random.default_rng(seed)

    # Adam state
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t_adam = 0

    stall = 0
    step_count = 0

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient
        grad, b = compute_grad_exact_topk(x, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-14:
            # Try small perturbation
            x = x + 0.001 * rng.standard_normal(n)
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            continue

        # Adam update
        t_adam += 1
        m_v = momentum1 * m_v + (1 - momentum1) * grad
        v_v = momentum2 * v_v + (1 - momentum2) * (grad * grad)
        m_hat = m_v / (1.0 - momentum1 ** t_adam)
        v_hat = v_v / (1.0 - momentum2 ** t_adam)

        # Backtracking line search
        found = False
        lr_trial = lr
        b_y_max = b.max()
        for _ in range(8):
            z = x - lr_trial * m_hat / (np.sqrt(v_hat) + epsilon)
            x_new = simplex_project(z)
            b_new = fast_autocorr(x_new)
            max_new = b_new.max()
            if max_new <= b_y_max + 1e-14:
                found = True
                break
            lr_trial *= 0.5

        if not found:
            lr = max(lr * 0.7, 0.0005)
            x_new = x
            max_new = best_m
        else:
            lr = min(lr * 1.03, 0.05)

        x = x_new

        if max_new < best_m - 1e-15:
            best_m = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        step_count += 1

        # Increase beta periodically
        if step_count % 20 == 0:
            beta = min(beta * beta_growth, beta_end)

        # Restart
        if stall > restart_every:
            stall = 0
            t_adam = 0
            m_v = np.zeros(n)
            v_v = np.zeros(n)
            lr = lr_init
            beta = max(beta * 0.4, 30.0)

            strategy = rng.integers(7)
            if strategy == 0:
                noise = 0.015 + 0.015 * rng.random()
                pert = rng.normal(0, noise, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.15)
            elif strategy == 2:
                x_r = best_a * np.exp(0.025 * rng.standard_normal(n))
            elif strategy == 3:
                # Peak-directed: reduce top contributing elements
                b_best = fast_autocorr(best_a)
                peak_k = int(np.argmax(b_best))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contrib = best_a[idx] * best_a[peak_k - idx]
                x_r = best_a.copy()
                top_contrib = idx[np.argsort(contrib)[::-1][:8]]
                for j in top_contrib:
                    x_r[int(j)] *= 0.85
                # Redistribute to less contributing positions
                bot_contrib = idx[np.argsort(contrib)[:8]]
                for j in bot_contrib:
                    x_r[int(j)] *= 1.15
            elif strategy == 4:
                # Periodic shift
                shift = int(rng.integers(1, max(2, n // 30)))
                x_r = np.roll(best_a, shift)
            elif strategy == 5:
                # Mirror blend with random ratio
                ratio = rng.uniform(0.3, 0.7)
                x_r = ratio * best_a + (1 - ratio) * best_a[::-1]
            else:
                # Block redistribution
                block_size = n // 10
                x_r = best_a.copy()
                for i in range(0, n, block_size):
                    end_i = min(i + block_size, n)
                    x_r[i:end_i] = x_r[i:end_i] * rng.uniform(0.8, 1.2)

            x_r = np.maximum(x_r, 1e-10)
            x = x_r / x_r.sum()

            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("  Adam restart step=" + str(step_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and step_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  Adam step=" + str(step_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + str(int(beta)) + " lr=" + str(round(lr, 6)))

    return best_a, best_m


def spectral_refine_v3(a0, time_limit, verbose=True, seed=777):
    """Spectral domain optimization with adaptive median-based flattening."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.025
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Use median as target (more robust than mean)
        if len(power) > 1:
            target = np.median(power[1:])
        else:
            target = power[0] if power[0] > 1e-30 else 1.0

        if target < 1e-30:
            target = 1.0

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / target
            if ratio > 1.1:
                red = 1.0 - alpha * min(ratio - 1.0, 4.0)
                A_new[k] *= max(0.15, np.sqrt(red))
            elif ratio < 0.4:
                inc = 1.0 + alpha * 0.3 * (0.5 - ratio)
                A_new[k] *= min(2.0, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 50 == 0:
            # Random phase perturbation every 50 failed iters
            A = np.fft.rfft(a, fft_size)
            perturb_amount = 0.01 + 0.01 * (it % 500) / 500.0
            phase_pert = rng.standard_normal(len(A)) * perturb_amount
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 200 == 0:
            alpha = max(0.003, alpha * 0.96)

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  SpecV3 it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def lp_fw_v3(a0, time_limit, solver, initial_K=200, max_K=2500,
             add_per_iter=15, verbose=True):
    """LP-Frank-Wolfe with careful active set management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 500
    iter_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set size
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K]
            active_list = sorted(active_set)
            active_set = set(int(active_list[i]) for i in keep)

        # Add near-max constraints
        threshold = M_curr * 0.92 + M_curr * 0.02
        near_max = np.where(ac >= threshold)[0]
        for k in near_max[:50]:
            active_set.add(int(k))

        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K == 0:
            break

        # Build coefficient matrix (sparse structure)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                ac_now = fast_autocorr(y)
                active_set = set(int(k) for k in np.argsort(ac_now)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Adaptive line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 2.0,
                  0.03, 0.05, 0.1, 0.15, 0.2]
        for alpha in alphas:
            alpha = max(0.005, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new
        iter_count += 1

        if verbose and iter_count % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  LP-FW3 it=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True, seed=42):
    """Targeted peak shaving with smart mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 4000:
        ac = fast_autocorr(a)
        size = 2 * n - 1
        top_ks = np.argsort(ac)[::-1][:12]
        improved = False

        for k in top_ks:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue

            # Find top contributors
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:4]]

            for src in top_contrib_idx:
                if time.monotonic() - t0 > time_limit:
                    break
                src_int = int(src)
                partner = k_int - src_int
                if not (0 <= partner < n):
                    continue

                fracs = [0.015, 0.025, 0.04, 0.06]
                for frac in fracs:
                    amt = a[src_int] * frac
                    if amt < 1e-15:
                        continue

                    # Try multiple destinations prioritizing low-contribution areas
                    n_tries = 10
                    for _ in range(n_tries):
                        dst = int(rng.integers(n))
                        if dst == src_int or dst == partner:
                            continue

                        # Check if dst would create new peaks
                        a_trial = a.copy()
                        a_trial[src_int] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break

                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 150:
                # More aggressive exploration
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0
                                improved = True
                                break

        if verbose and stall % 600 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("  PeakShave (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score
