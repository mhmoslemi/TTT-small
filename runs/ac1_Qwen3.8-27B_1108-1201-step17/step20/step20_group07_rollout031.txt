def directed_exchange(c, time_limit, verbose=True):
    """
    Directed pair-exchange: identify top peaks, find optimal source-target
    pairs considering multi-peak interactions, and execute moves.
    """

    c = normalize(c)
    best_c = c.copy()
    best_score = eval_score(c)
    b = fast_autocorr(c)
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    stall = 0
    max_stall = 300

    # We'll work with the top K peaks
    K_peaks = min(10, n)  # number of peaks to consider
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(c)
        b_max = b.max()
        curr_score = 2.0 * n * b_max
        
        if curr_score < best_score - 1e-10:
            best_score = curr_score
            best_c = c.copy()
            stall = 0
        else:
            stall += 1

        peak_lags = np.argsort(b)[::-1][:K_peaks]
        peak_vals = b[peak_lags]
        b_max_idx = peak_lags[0]

        best_delta = 0  # best improvement (negative = improvement)
        best_move = None
        
        for k_star in peak_lags[:5]:  # Only top 5 peaks
            k_int = int(k_star)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            partners = k_int - idx
            contrib = c[idx] * c[partners]
            
            if contrib.max() < 1e-15:
                continue
            
            src = int(idx[np.argmax(contrib)])
            src_val = c[src]
            if src_val < 1e-12:
                continue
            
            # Find target: position q that minimizes impact on all top peaks
            # Change in b[k] from moving eps from src to q:
            # db[k] ≈ 2*eps*(c[k-q] - c[k-src])
            # We want db[k*] < 0 (reduce the peak) and db[k'] <= 0 for other peaks
            
            # For the target peak k*: we want c[k*-q] < c[k*-src]
            # For other peaks k': we want c[k'-q] <= c[k'-src] (ideally)
            
            # Compute "cost" of each candidate target
            # Cost[q] = sum over top peaks of max(0, c[k-q] - c[k-src]) * weight[k]
            
            # This is O(n * K_peaks) per source, which could be slow for n=7000
            # Use a smarter approach: only consider a subset of targets
            
            # Candidate targets: positions where c is small (low "energy")
            # and that are far from the peak lags
            small_c = np.argsort(c)[:n//4]  # bottom quartile of c values
            small_c = small_c[small_c != src]
            
            if len(small_c) == 0:
                continue
            
            # Score each candidate target
            target_scores = np.zeros(len(small_c))
            for j, q in enumerate(small_c):
                score = 0.0
                for k_prime in peak_lags[:5]:
                    kp = int(k_prime)
                    # c[kp - q] if valid, else 0
                    val_q = c[kp - q] if 0 <= kp - q < n else 0.0
                    val_src = c[kp - src] if 0 <= kp - src < n else 0.0
                    score += max(0.0, val_q - val_src) * (b[k_prime] / b_max)
                target_scores[j] = score
            
            best_target_idx = int(np.argmin(target_scores))
            q = int(small_c[best_target_idx])
            
            # Verify: does moving to q actually reduce the target peak?
            val_q = c[k_int - q] if 0 <= k_int - q < n else 0.0
            val_src = c[k_int - src] if 0 <= k_int - src < n else 0.0
            if val_q >= val_src:
                # Moving to q won't help this peak, try next
                continue
            
            best_move = (src, q)
            break
        
        if best_move is None:
            # No directed move found, try random
            src = int(rng.integers(n))
            if c[src] < 1e-12:
                continue
            # Target: random from low-c positions
            candidates = np.argsort(c)[:n//2]
            candidates = candidates[candidates != src]
            if len(candidates) == 0:
                continue
            q = int(candidates[rng.integers(len(candidates))])
            best_move = (src, q)
        
        src, q = best_move

        improved = False
        for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
            eps = c[src] * frac
            if eps < 1e-15:
                continue
            c_trial = c.copy()
            c_trial[src] -= eps
            c_trial[q] += eps
            c_trial = np.maximum(c_trial, 0.0)
            s_t = c_trial.sum()
            if s_t < 1e-15:
                continue
            c_trial = c_trial / s_t
            b_trial = fast_autocorr(c_trial)
            m_trial = b_trial.max()
            if m_trial < b_max:
                c = c_trial
                b_max = m_trial
                improved = True
                break
        
        if not improved:
            # Add small random perturbation to escape
            if stall > 50:
                pert_pos = rng.integers(0, 5)
                j = int(rng.integers(n))
                k = int(rng.integers(n))
                if j != k and c[j] > 1e-10:

                    c[j] -= eps
                    c[k] += eps
                    c = normalize(c)
                    b = fast_autocorr(c)
                    if b.max() < best_score / (2.0 * n):
                        best_c = c.copy()
                        best_score = 2.0 * n * b.max()
                        stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DirectedExchange stall={stall} ({elapsed:.1f}s): J={best_score:.6f}")
    
    return best_c, best_score


def simulated_annealing(c0, time_limit, verbose=True):
    """
    Simulated annealing with structured moves:
    - Single position move
    - Block scaling
    - Mirror perturbation
    """
    n = len(c0)
    c = normalize(c0)
    best_c = c.copy()
    best_score = eval_score(c)
    b = fast_autocorr(c)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    T = 0.01 * b.max()  # Initial temperature
    T_min = 1e-8 * b.max()
    cooling = 0.9995
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_autocorr(c)
        b_max = b.max()
        curr_score = 2.0 * n * b_max
        
        if curr_score < best_score:
            best_score = curr_score
            best_c = c.copy()
        
        # Choose move type
        move_type = rng.integers(0, 3)
        c_trial = c.copy()
        
        if move_type == 0:
            # Single position move: transfer mass from i to j
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or c[i] < 1e-12:
                continue
            frac = rng.uniform(0.01, 0.2)
            eps = c[i] * frac
            c_trial[i] -= eps
            c_trial[j] += eps
        
        elif move_type == 1:
            # Block scaling: scale a block up and another block down
            block_size = int(rng.integers(max(10, n//100), max(20, n//20)))
            i = int(rng.integers(0, n - block_size))
            scale_up = rng.uniform(0.9, 1.3)
            mass_moved = c_trial[i:i+block_size].sum() * (scale_up - 1.0)
            j = int(rng.integers(0, n - block_size))
            while j == i:
                j = int(rng.integers(0, n - block_size))
            c_trial[i:i+block_size] *= scale_up
            # Distribute the mass change to the other block
            c_trial[j:j+block_size] -= mass_moved / block_size
        
        else:
            # Mirror perturbation: add a scaled mirrored version
            center = n // 2
            mirror = c[::-1]
            alpha = rng.uniform(-0.05, 0.05)
            c_trial = c + alpha * (mirror - c)
        
        c_trial = np.maximum(c_trial, 0.0)
        s_t = c_trial.sum()
        if s_t < 1e-15:
            continue
        c_trial = c_trial / s_t
        
        b_trial = fast_autocorr(c_trial)
        m_trial = b_trial.max()
        new_score = 2.0 * n * m_trial
        delta = new_score - curr_score
        
        # Metropolis acceptance
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            c = c_trial
        
        # Cool
        T = max(T * cooling, T_min)
        
        if verbose and iteration % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA iter={iteration} ({elapsed:.1f}s): J={best_score:.6f} T={T:.2e}")
        
        # Warm restart if stuck
        if iteration % 20000 == 0 and T < T_min * 10:
            T = 0.005 * best_score / (2.0 * n)
            # Perturb from best
            pert = rng.normal(0, 0.02, n) * best_c
            c = best_c + pert
            c = normalize(c)
    
    return best_c, best_score


def spectral_redistribute(c0, time_limit, verbose=True):
    """
    Use spectral analysis to identify and fix problematic frequency components.
    The idea: peaks in the autocorrelation correspond to spectral features.
    By analyzing the spectrum of the autocorrelation, we can identify which
    time-domain features cause the peaks and modify them.
    """
    n = len(c0)
    c = normalize(c0)
    best_c = c.copy()
    best_score = eval_score(c)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(c)
        b_max = b.max()
        curr_score = 2.0 * n * b_max
        
        if curr_score < best_score:
            best_score = curr_score
            best_c = c.copy()
        
        # Analyze the spectrum of the autocorrelation
        # The "excess" autocorrelation (above mean) has a spectrum that tells us
        # what time-domain features are causing the peaks
        b_mean = b.sum() / len(b)  # = 1/(2n-1) approximately
        b_excess = b - b_mean
        
        # FFT of the excess gives us the "problem spectrum"
        # High values in this spectrum indicate frequencies in c that
        # create structured autocorrelation patterns
        fft_size = 1 << (len(b_excess) - 1).bit_length()
        B = np.fft.rfft(b_excess, fft_size)
        spectrum = np.abs(B)
        
        # The peak frequencies in the spectrum correspond to "wavelengths"
        # in c that are causing autocorrelation structure
        # We want to add "anti-correlation" at these frequencies
        
        # Find top spectral peaks (excluding DC)
        spec_no_dc = spectrum.copy()
        spec_no_dc[0] = 0
        top_freqs = np.argsort(spec_no_dc)[::-1][:3]
        
        # For each top frequency, add a perturbation that cancels it
        improved = False
        for freq_idx in top_freqs:
            if time.monotonic() - t0 > time_limit:
                break
            
            freq = freq_idx / fft_size  # Normalized frequency
            
            # Create a perturbation that has energy at this frequency
            x = np.arange(n)
            pert = np.cos(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
            
            # Scale the perturbation
            pert_scale = 0.01 * c.std()
            for scale_factor in [0.5, 1.0, 2.0, 4.0]:
                c_trial = c + pert_scale * scale_factor * pert
                c_trial = np.maximum(c_trial, 0.0)
                st = c_trial.sum()
                if st < 1e-15:
                    continue
                c_trial = c_trial / st
                b_trial = fast_autocorr(c_trial)
                m_trial = b_trial.max()
                if m_trial < b_max:
                    c = c_trial
                    b_max = m_trial
                    improved = True
                    break
        
        if not improved:
            # Try a random spectral perturbation
            num_harmonics = rng.integers(1, 5)
            pert = np.zeros(n)
            x = np.arange(n)
            for h in range(num_harmonics):
                f = rng.uniform(0.01, 0.4)
                phase = rng.uniform(0, 2*np.pi)
                amp = rng.uniform(0.005, 0.02)
                pert += amp * np.cos(2*np.pi*f*x + phase)
            c_trial = c + pert
            c_trial = np.maximum(c_trial, 0.0)
            st = c_trial.sum()
            if st > 1e-15:
                c_trial = c_trial / st
                b_trial = fast_autocorr(c_trial)
                if b_trial.max() < b_max:
                    c = c_trial
                    b_max = b_trial.max()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Spectral ({elapsed:.1f}s): J={best_score:.6f}")
        
        # Break if no improvement in a while
        elapsed = time.monotonic() - t0
        if elapsed > time_limit * 0.8:
            break
    
    return best_c, best_score

                c_trial = np.maximum(c_trial, 0.0)
                st = c_trial.sum()
                if st > 1e-15:
                    c_trial = c_trial / st
                    b_trial = fast_autocorr(c_trial)
                    if b_trial.max() < best_score / (2.0 * n):
                        c = c_trial
                        b = fast_autocorr(c)
                        b_max = b.max()
                        best_score = 2.0 * n * b_max
                        best_c = c.copy()
                        stall = 0
                else:
                    c = best_c.copy()
                    b = fast_autocorr(c)
                    b_max = b.max()

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DirectedExch stall={stall} ({elapsed:.1f}s): J={best_score:.6f}")

    return best_c, best_score


def simulated_annealing(c0, time_limit, verbose=True):
    n = len(c0)
    c = normalize(c0)
    best_c = c.copy()
    best_score = eval_j(c)
    b_max = fast_autocorr(c).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    T = 0.005 * b_max
    T_min = 1e-10 * b_max
    cooling = 0.9997
    iter_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        curr_score = 2.0 * n * b_max

        if curr_score < best_score:
            best_score = curr_score
            best_c = c.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Choose move
        mt = rng.integers(0, 4)
        c_trial = c.copy()

        if mt == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and c[i] > 1e-12:
                frac = rng.uniform(0.005, 0.15)
                eps = c[i] * frac
                c_trial[i] -= eps
                c_trial[j] += eps
        elif mt == 1:
            # Pair swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                amt = min(c[i], c[j]) * rng.uniform(0.1, 0.8)
                c_trial[i] -= amt
                c_trial[j] += amt
        elif mt == 2:
            # Block perturbation
            bs = int(rng.integers(max(10, n // 200), max(20, n // 50)))
            i = int(rng.integers(0, max(1, n - bs)))
            c_trial[i:i+bs] *= rng.uniform(0.9, 1.1)
        else:
            # Spectral nudge
            x = np.arange(n)
            f = rng.uniform(0.005, 0.5)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.002, 0.01) * c.std()
            c_trial = c + amp * np.cos(2 * np.pi * f * x + phase)

        c_trial = np.maximum(c_trial, 0.0)
        st = c_trial.sum()
        if st < 1e-15:
            continue
        c_trial = c_trial / st

        b_trial = fast_autocorr(c_trial)
        m_trial = b_trial.max()
        new_score = 2.0 * n * m_trial
        delta = (new_score - curr_score) / (2.0 * n)

        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            c = c_trial
            b_max = m_trial

        T = max(T * cooling, T_min)

        # Restarts
        if no_improve > 5000 and T < T_min * 5:
            T = 0.003 * best_score / (2.0 * n)
            pert = rng.normal(0, 0.015, n) * best_c
            c = best_c + pert
            c = normalize(c)
            b_max = fast_autocorr(c).max()
            no_improve = 0

        if verbose and iter_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA iter={iter_count} ({elapsed:.1f}s): J={best_score:.6f} T={T:.2e} noimp={no_improve}")

    return best_c, best_score


def spectral_optimize(c0, time_limit, verbose=True):
    n = len(c0)
    c = normalize(c0)
    best_c = c.copy()
    best_score = eval_j(c)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    x = np.arange(n)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        b = fast_autocorr(c)
        b_max = b.max()
        curr_score = 2.0 * n * b_max

        if curr_score < best_score - 1e-12:
            best_score = curr_score
            best_c = c.copy()
            stall = 0
        else:
            stall += 1

        # Find the lag with maximum autocorrelation
        k_peak = int(np.argmax(b))

        # Compute the "excess" autocorrelation
        b_mean = b.sum() / len(b)
        b_excess = b - b_mean

        # Get spectral content of the excess
        fft_sz = 1 << (len(b_excess) - 1).bit_length()
        B = np.fft.rfft(b_excess, fft_sz)
        spec = np.abs(B)
        spec[0] = 0

        # Try to cancel top spectral components
        top_freqs = np.argsort(spec)[::-1][:2]
        improved = False

        for fi in top_freqs:
            freq = int(fi) / fft_sz
            if freq < 1e-6 or freq > 0.49:
                continue

            for sign in [1, -1]:
                for amp in [0.005, 0.01, 0.02, 0.04]:
                    c_trial = c + sign * amp * c.std() * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
                    c_trial = np.maximum(c_trial, 0.0)
                    st = c_trial.sum()
                    if st < 1e-15:
                        continue
                    c_trial = c_trial / st
                    b_trial = fast_autocorr(c_trial)
                    if b_trial.max() < b_max * 0.999:
                        c = c_trial
                        b_max = b_trial.max()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random spectral perturbation with multiple harmonics
            num_har = rng.integers(1, 4)
            pert = np.zeros(n)
            for h in range(num_har):
                f = rng.uniform(0.01, 0.4)
                ph = rng.uniform(0, 2 * np.pi)
                a = rng.uniform(0.003, 0.015) * c.std()
                pert += a * np.cos(2 * np.pi * f * x + ph)
            c_trial = c + pert
            c_trial = np.maximum(c_trial, 0.0)
            st = c_trial.sum()
            if st > 1e-15:
                c_trial = c_trial / st
                b_trial = fast_autocorr(c_trial)
                if b_trial.max() < b_max:
                    c = c_trial
                    b_max = b_trial.max()

        if verbose and stall % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral stall={stall} ({elapsed:.1f}s): J={best_score:.6f}")

    return best_c, best_score


def fine_tune(c0, time_limit, verbose=True):
    """Very fine coordinate descent for final polish."""
    n = len(c0)
    c = normalize(c0)
    best_c = c.copy()
    best_score = eval_j(c)
    b = fast_autocorr(c)
    b_max = b.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        curr_score = 2.0 * n * b_max
        if curr_score < best_score - 1e-12:
            best_score = curr_score
            best_c = c.copy()
            stall = 0
        else:
            stall += 1

        # Find the peak lag
        k_peak = int(np.argmax(b))
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        idx = np.arange(i_start, i_end + 1)
        partners = k_peak - idx
        contrib = c[idx] * c[partners]

        # Find the most contributing position
        src = int(idx[np.argmax(contrib)])
        if c[src] < 1e-13:
            break

        # Try small moves to nearby positions
        search_radius = max(10, n // 50)
        lo = max(0, src - search_radius)
        hi = min(n, src + search_radius + 1)
        candidates = np.arange(lo, hi)
        candidates = candidates[candidates != src]

        best_move = None
        best_new_max = b_max

        for q in candidates:
            q = int(q)
            # Quick check: does this reduce the peak?
            vq = c[k_peak - q] if 0 <= k_peak - q < n else 0.0
            vs = c[k_peak - src] if 0 <= k_peak - src < n else 0.0
            if vq >= vs:
                continue
            # Try tiny move
            for frac in [0.001, 0.003, 0.005]:
                eps = c[src] * frac
                c_trial = c.copy()
                c_trial[src] -= eps
                c_trial[q] += eps
                c_trial = np.maximum(c_trial, 0.0)
                st = c_trial.sum()
                if st < 1e-15:
                    continue
                c_trial = c_trial / st
                b_trial = fast_autocorr(c_trial)
                if b_trial.max() < best_new_max:
                    best_new_max = b_trial.max()
                    best_move = (src, q, frac)
                    break

        if best_move is not None:
            src, q, frac = best_move
            eps = c[src] * frac
            c[src] -= eps
            c[q] += eps
            c = normalize(c)
            b = fast_autocorr(c)
            b_max = b.max()
        else:
            # Try a wider random search
            q = int(rng.integers(n))
            if q != src and c[src] > 1e-12:
                for frac in [0.002, 0.005, 0.01]:
                    eps = c[src] * frac
                    c_trial = c.copy()
                    c_trial[src] -= eps
                    c_trial[q] += eps
                    c_trial = np.maximum(c_trial, 0.0)
                    st = c_trial.sum()
                    if st < 1e-15:
                        continue
                    c_trial = c_trial / st
                    b_trial = fast_autocorr(c_trial)
                    if b_trial.max() < b_max:
                        c = c_trial
                        b = fast_autocorr(c)
                        b_max = b.max()
                        break
            else:
                stall += 10

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineTune stall={stall} ({elapsed:.1f}s): J={best_score:.7f}")

    return best_c, best_score


def parametric_init(n, best_c, best_score):
    """Try various parametric families for initialization."""
    x = np.linspace(0.0, 1.0, n)
    best_a = best_c.copy()
    best_val = best_score

    # Power laws
    for alpha in [0.3, 0.5, 0.7, 0.8, 1.0, 1.2, 1.5, 2.0]:
        for beta in [0.3, 0.5, 0.7, 0.8, 1.0, 1.2, 1.5, 2.0]:
            a_raw = (x + 0.005) ** alpha * (1.0 - x + 0.005) ** beta
            a = normalize(a_raw)
            sc = eval_j(a)
            if sc < best_val:
                best_val = sc
                best_a = a.copy()

    # Bimodal
    for c1 in [0.15, 0.2, 0.25, 0.3]:
        for c2 in [0.7, 0.75, 0.8, 0.85]:
            for s in [0.03, 0.05, 0.08, 0.12]:
                for w in [0.5, 1.0, 2.0]:
                    a_raw = np.exp(-((x - c1) ** 2) / (2 * s ** 2)) + w * np.exp(-((x - c2) ** 2) / (2 * s ** 2))
                    a = normalize(a_raw)
                    sc = eval_j(a)
                    if sc < best_val:
                        best_val = sc
                        best_a = a.copy()

    # U-shape (low center, high edges)
    for center_depth in [0.1, 0.3, 0.5, 0.7]:
        a_raw = 1.0 - center_depth * np.exp(-((x - 0.5) ** 2) / 0.05)
        a_raw = np.maximum(a_raw, 0.01)
        a = normalize(a_raw)
        sc = eval_j(a)
        if sc < best_val:
            best_val = sc
            best_a = a.copy()

    # Cosine-based
    for k in [1, 2, 3, 4]:
        for phase in [0, 0.3, 0.5, 0.7]:
            a_raw = 1.0 + 0.3 * np.cos(2 * np.pi * k * x + phase * np.pi)
            a_raw = np.maximum(a_raw, 0.01)
            a = normalize(a_raw)
            sc = eval_j(a)
            if sc < best_val:
                best_val = sc
                best_a = a.copy()

    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.1f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            parent_score = evaluate_safe(parent)
            if np.isfinite(parent_score):
                best_seq = normalize(parent)
                best_score = parent_score
                print(f"Parent loaded: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        parent = None
        print("No parent available, using parametric init.")

    if best_seq is None:
        n_init = 6700
        x = np.linspace(0, 1, n_init)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = normalize(a_raw)
        best_score = eval_j(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def try_update(c, label):
        nonlocal best_seq, best_score
        if c is None:
            return
        c = normalize(c)
        sc = eval_j(c)
        # Also verify with the actual evaluator
        sc_real = evaluate_safe(c)
        if np.isfinite(sc_real) and sc_real < best_score:
            best_score = sc_real
            best_seq = c.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={best_score:.6f} (n={len(c)}, t={elapsed:.1f}s)")

    n0 = len(best_seq)

    # Phase 1: Parametric sweep (short)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Parametric sweep (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.02, 30.0)
    if t1 > 5:
        a_param, _ = parametric_init(n0, best_seq, best_score)
        try_update(a_param, f"param_n{n0}")

    # Phase 2: Directed exchange (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Directed exchange (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_out, _ = directed_exchange(best_seq, t2, verbose=True)
        try_update(a_out, f"dir1_n{len(best_seq)}")

    # Phase 3: Simulated annealing
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Simulated annealing (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        a_out, _ = simulated_annealing(best_seq, t3, verbose=True)
        try_update(a_out, f"sa_n{len(best_seq)}")

    # Phase 4: Spectral optimization
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Spectral optimize (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, _ = spectral_optimize(best_seq, t4, verbose=True)
        try_update(a_out, f"spec_n{len(best_seq)}")

    # Phase 5: Directed exchange round 2
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Directed exchange 2 (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 20:
        a_out, _ = directed_exchange(best_seq, t5, verbose=True)
        try_update(a_out, f"dir2_n{len(best_seq)}")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-length (remaining={remaining:.0f}s) ===")
    n_best = len(best_seq)
    candidate_lengths = []
    for dn in [-300, -100, 100, 300, 500, 800, 1000]:
        nl = n_best + dn
        if 5000 < nl <= 9000:
            candidate_lengths.append(nl)

    for new_n in candidate_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_interp = interpolate_to_length(best_seq, new_n)
        try_update(a_interp, f"interp_n{new_n}")

        t_cl = min(remaining * 0.35, 200.0)
        if t_cl > 20:
            a_out, _ = directed_exchange(a_interp, t_cl, verbose=False)
            try_update(a_out, f"dir_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        t_sa = min(remaining * 0.25, 150.0)
        if t_sa > 15:
            a_out, _ = simulated_annealing(a_interp, t_sa, verbose=False)
            try_update(a_out, f"sa_n{new_n}")

    # Phase 7: Fine tuning
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Fine tune (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.4, 300.0)
    if t7 > 15:
        a_out, _ = fine_tune(best_seq, t7, verbose=True)
        try_update(a_out, f"fine_n{len(best_seq)}")

    # Phase 8: Final directed exchange
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final directed (remaining={remaining:.0f}s) ===")
    t8 = min(remaining * 0.6, 200.0)
    if t8 > 15:
        a_out, _ = directed_exchange(best_seq, t8, verbose=True)
        try_update(a_out, f"final_dir_n{len(best_seq)}")

    # Phase 9: Final fine tune
    remaining = time_budget - (time.monotonic() - t_global)
    t9 = min(remaining * 0.7, 150.0)
    if t9 > 10:
        a_out, _ = fine_tune(best_seq, t9, verbose=True)
        try_update(a_out, f"final_fine_n{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = (x + 0.01) ** 1.0 * (1.0 - x + 0.01) ** 1.0
        best_seq = normalize(a_raw)
        best_score = eval_j(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
