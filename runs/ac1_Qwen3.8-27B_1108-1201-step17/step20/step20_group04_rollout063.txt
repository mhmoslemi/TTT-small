"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exponential parameterization (log-space optimization on simplex)
2. Adaptive log-sum-exp smoothed objective with Nesterov acceleration
3. Simulated annealing with structured mutations for global exploration
4. Peak surgery and coordinate descent for local exploitation
5. Multi-length interpolation and length adaptation
6. Frequency-domain insights for initialization

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    # Sort in descending order
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_gradient_logspace(theta, beta, top_k=None):
    """
    Compute gradient of smoothed max in log-space parameterization.
    a[i] = exp(theta[i]) / Z, Z = sum(exp(theta))
    dJ/dtheta[i] = a[i]*(1-a[i]) * dJ/da[i]
    """
    n = len(theta)
    # Compute a from theta
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    Z = exp_theta.sum()
    a = exp_theta / Z
    
    # Compute convolution
    c = fast_autocorr(a)
    
    # Soft max
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c_use[idx].max()
    else:
        c_use = c
        c_max = c.max()
    
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    
    # d(c_k)/d(a_j) = 2*a[k-j] (with a[x]=0 outside [0,n-1])
    # So dJ/da[j] = (2/beta) * sum_k p[k] * a[k-j]
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_rev = np.fft.rfft(a[::-1], fft_size)
    Conv = np.fft.irfft(P * A_rev, fft_size)
    grad_a = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    
    # Chain rule: dJ/dtheta[i] = a[i]*(1-a[i]) * dJ/da[i]
    grad_theta = a * (1 - a) * grad_a
    
    return grad_theta, p, c, a

def nesterov_logspace(theta0, time_limit, beta_start=50.0, beta_end=5000.0,
                       beta_growth=1.005, lr_base=1.0, top_k_ratio=0.02,
                       restart_every=3000, verbose=True, seed=42):
    """Nesterov accelerated method in log-space parameterization."""
    n = len(theta0)
    theta = theta0.copy()
    theta_prev = theta.copy()
    
    # Compute best
    theta_shifted = theta - theta.max()
    a = np.exp(theta_shifted) / np.exp(theta_shifted).sum()
    best_a = a.copy()
    best_max_b = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        mu = (t_nest - 1.0) / t_new
        theta_y = theta + mu * (theta - theta_prev)
        
        grad, p, c, a_y = compute_gradient_logspace(theta_y, beta, top_k)
        
        # Adaptive learning rate
        lr = lr_base / (beta * max(p.max(), 1e-10) * 2.0)
        lr = min(lr, 0.05)
        
        theta_new = theta_y - lr * grad
        # No projection needed (log-space is unconstrained)
        
        # Compute a from new theta
        theta_new_shifted = theta_new - theta_new.max()
        a_new = np.exp(theta_new_shifted) / np.exp(theta_new_shifted).sum()
        
        max_new = fast_autocorr(a_new).max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        theta_prev = theta
        theta = theta_new
        t_nest = t_new
        
        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            # Restart with perturbation
            noise_scale = 0.05 / (1.0 + 0.03 * (t // restart_every))
            noise = rng.normal(0, 1, n) * noise_scale
            # Also add anti-symmetric component
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            theta_r = np.log(best_a + 1e-10) + noise + 0.3 * noise_scale * anti
            theta = theta_r - theta_r.max()  # normalize for numerical stability
            theta = theta - theta.mean()
            theta_prev = theta.copy()
            beta = max(beta * 0.5, 20.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    NestLog restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))
        
        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    NestLog t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))
    
    return best_a, best_max_b

def simulated_annealing(a0, time_limit, verbose=True, seed=99):
    """Simulated annealing with structured mutations."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    temperature = 0.1
    cool_rate = 0.9995
    no_improve = 0
    max_no_improve = 5000
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Choose mutation type
        r = rng.random()
        if r < 0.3:
            # Single position move: move mass from one position to another
            # Prefer moving from high "pressure" positions
            c = fast_autocorr(a)
            top_idx = np.argsort(c)[::-1][:5]
            src = int(rng.choice(top_idx))
            # Find contributing positions for this peak
            k = int(top_idx[rng.integers(len(top_idx))])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # Move mass from a position in this peak's support
            candidates_src = list(range(i_start, i_end + 1))
            candidates_src = [x for x in candidates_src if a[x] > 1e-6]
            if not candidates_src:
                continue
            src = int(rng.choice(candidates_src))
            dst = int(rng.integers(n))
            if dst == src:
                continue
            frac = rng.uniform(0.01, 0.1) * temperature
            amt = a[src] * frac
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            
        elif r < 0.5:
            # Block move: swap a block of mass
            block_size = int(rng.integers(5, min(50, n // 10)))
            start = int(rng.integers(n - block_size))
            # Move the block's mass to a different location
            src_start = start
            # Find a "quiet" destination
            c = fast_autocorr(a)
            # Low convolution positions
            low_positions = np.argsort(c)[:n // 2]
            dst_start = int(rng.choice(low_positions))
            dst_start = int(np.clip(dst_start, 0, n - block_size))
            
            amt_block = a[src_start:src_start + block_size].copy()
            a[src_start:src_start + block_size] = 0
            a[dst_start:dst_start + block_size] += amt_block
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            
        elif r < 0.7:
            # Gaussian perturbation
            scale = temperature * 0.01
            noise = rng.normal(0, scale, n)
            a_new = a + noise * a  # multiplicative noise
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a = a_new / s
            
        elif r < 0.85:
            # Anti-symmetric perturbation
            anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            scale = temperature * 0.02
            a_new = a + scale * anti * a
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a = a_new / s
        else:
            # Pair swap: swap mass between two positions
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            amt = min(a[i], a[j]) * rng.uniform(0.1, 0.5)
            a[i] -= amt
            a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        
        # Accept/reject
        new_m = fast_autocorr(a).max()
        if new_m < c_max:
            # Always accept improvement
            pass
        else:
            # Metropolis criterion
            delta = new_m - c_max
            if delta > 0:
                accept_prob = np.exp(-delta / (temperature * 1e-4))
                if rng.random() > accept_prob:
                    # Reject: revert
                    # We need to undo the mutation, which is complex

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

def build_initial(n, seed=0):
    """Build a structured initial guess."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Try several structured inits and pick the best
    candidates = []
    
    # Power law
    candidates.append((1.0 - x) ** 1.5 + 0.05)
    # Quadratic bump
    candidates.append(1.0 - (2*x - 1)**2)
    # Exponential
    candidates.append(np.exp(-2.0 * x))
    # Triangular
    candidates.append(1.0 - np.abs(2*x - 1))
    # Cosine bump
    candidates.append(0.5 + 0.5 * np.cos(np.pi * x))
    # Anti-symmetric (high on one side)
    candidates.append(np.linspace(1.0, 0.1, n))
    # Reverse
    candidates.append(np.linspace(0.1, 1.0, n))
    # Two bumps
    candidates.append(np.exp(-5*(x-0.3)**2) + np.exp(-5*(x-0.7)**2) + 0.05)
    
    best = None
    best_score = float('inf')
    for c in candidates:
        c = np.maximum(c, 0.01)
        c = c / c.sum()
        sc = evaluate_safe(c)
        if sc < best_score:
            best_score = sc
            best = c.copy()
    
    return best, best_score

import numpy as np

def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_grad_logspace(theta, beta, top_k=None):
    """
    Compute gradient of smoothed max in log-space (softmax) parameterization.
    a[i] = exp(theta[i]) / Z
    dF/dtheta[i] = a[i] * (g[i] - sum_j a[j]*g[j])
    where g[j] = (2/beta) * sum_k p[k] * a[k-j]
    """
    n = len(theta)
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    Z = exp_theta.sum()
    a = exp_theta / Z

    c = fast_autocorr(a)

    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c_use[idx].max()
    else:
        c_use = c
        c_max = c.max()

    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total

    # g[j] = (2/beta) * sum_k p[k] * a[k-j]
    # = (2/beta) * conv(p, a_rev)[n-1+j]
    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_rev = np.fft.rfft(a[::-1], fft_size)
    Conv = np.fft.irfft(P * A_rev, fft_size)
    g = (2.0 / beta) * Conv[n - 1:2 * n - 1]

    # IAPD correction
    h = np.dot(a, g)
    grad_theta = a * (g - h)

    return grad_theta, p, c, a

                       restart_every=3000, verbose=True, seed=42):
    """Nesterov accelerated method in log-space with adaptive step size."""
    n = len(theta0)
    theta = theta0.copy().astype(np.float64)
    theta_prev = theta.copy()

    theta_shifted = theta - theta.max()
    a_curr = np.exp(theta_shifted) / np.exp(theta_shifted).sum()
    best_a = a_curr.copy()
    best_max_b = fast_autocorr(a_curr).max()

    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    lr = lr_base / max(1.0, beta * 2.0)

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        mu = (t_nest - 1.0) / t_new
        theta_y = theta + mu * (theta - theta_prev)

        grad, p, c, a_y = compute_grad_logspace(theta_y, beta, top_k)

        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            lr_new = lr_base / max(1.0, beta * 2.0)
        else:
            lr_new = lr_base / max(1.0, beta * max(p.max(), 1e-10) * 2.0)
        lr_new = min(lr_new, 0.05)
        lr = 0.9 * lr + 0.1 * lr_new

        theta_new = theta_y - lr * grad

        # Compute a and check improvement
        theta_new_shifted = theta_new - theta_new.max()
        exp_tn = np.exp(theta_new_shifted)
        a_new = exp_tn / exp_tn.sum()

        max_new = fast_autocorr(a_new).max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        theta_prev = theta
        theta = theta_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.05 / (1.0 + 0.03 * (t // restart_every))
            noise = rng.normal(0, 1, n) * noise_scale
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            theta_r = np.log(best_a + 1e-10) + noise + 0.5 * noise_scale * anti
            theta = theta_r - theta_r.mean()
            theta_prev = theta.copy()
            beta = max(beta * 0.5, 20.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    NestLog restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    NestLog t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def optimal_peak_move(a, top_peaks, max_fraction=0.1):
    """
    Find the optimal single move (src -> dst, delta) that reduces the max peak.
    Uses sensitivity analysis: dc[k]/da[j] = 2*a[k-j]
    """
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    c_top = np.argsort(c)[::-1][:min(20, len(c))]

    best_result = None
    best_improvement = 0.0

    for k_t in c_top[:8]:
        k_int = int(k_t)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue

        idx = np.arange(i_start, i_end + 1)
        products = a[idx] * a[k_int - idx]
        if products.max() < 1e-20:
            continue

        # Try top 3 contributing pairs for this peak
        top_pairs = np.argsort(products)[::-1][:3]

        for pi in top_pairs:
            src_i = int(idx[pi])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue

            # Determine which to reduce
            if a[src_i] >= a[partner]:
                reduce_i = src_i
            else:
                reduce_i = partner

            if a[reduce_i] < 1e-14:
                continue

            # Max possible delta
            max_delta = max_fraction * a[reduce_i]

            # Find best destination using sensitivity
            # For destination j, the change in peak k' is:
            #   delta * (2*a[k'-j] - 2*a[k'-reduce_i])
            # We want: c[k'] + delta*(2*a[k'-j] - 2*a[k'-reduce_i]) <= c_max for all top k'
            # => delta <= (c_max - c[k']) / (2*a[k'-reduce_i] - 2*a[k'-j]) when denom > 0

            # Try candidate destinations: positions with low a values
            # (which tend to have low sensitivity a[k'-j])
            n_candidate = min(50, n)
            cand_positions = np.argsort(a)[:n_candidate]
            cand_positions = [int(j) for j in cand_positions if j != reduce_i]

            for j in cand_positions[:20]:
                max_d = max_delta
                # Check constraint for each top peak
                for k_p in c_top:
                    k_p_int = int(k_p)
                    ki = k_p_int - reduce_i
                    ai_val = a[ki] if 0 <= ki < n else 0.0
                    kj = k_p_int - j
                    aj_val = a[kj] if 0 <= kj < n else 0.0
                    denom = 2.0 * (ai_val - aj_val)
                    if denom > 1e-15:
                        d_limit = (c_max - c[k_p_int]) / denom
                        if d_limit < max_d:
                            max_d = d_limit

                if max_d > 1e-13 and max_d > best_improvement:
                    # Execute and verify
                    delta = min(max_d, 0.05 * a[reduce_i])
                    a_trial = a.copy()
                    a_trial[reduce_i] -= delta
                    a_trial[j] += delta
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    m_trial = fast_autocorr(a_trial).max()
                    if m_trial < c_max - 1e-16:
                        improvement = c_max - m_trial
                        if improvement > best_improvement:
                            best_improvement = improvement
                            best_result = a_trial

    return best_result


def peak_surgery_optimal(a0, time_limit, verbose=True, seed=77):
    """Peak surgery using optimal move computation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 8000:
        a_new = optimal_peak_move(a, None, max_fraction=0.1)
        if a_new is not None:
            m_new = fast_autocorr(a_new).max()
            if m_new < best_m - 1e-16:
                a = a_new
                best_m = m_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # If stuck, try random moves
        if no_improve > 200:
            improved = False
            for _ in range(30):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            no_improve = 0
                            improved = True
                            break
            if not improved and no_improve > 2000:
                # Larger perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * 0.2
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    m_new = fast_autocorr(a).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a.copy()
                        no_improve = 0

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    OptPS (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def spectral_local_search(a0, time_limit, verbose=True, seed=55):
    """
    Local search that considers all peaks simultaneously.
    At each step, find the move (i->j, delta) that minimizes the max convolution
    considering the top 30 peaks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        c = fast_autocorr(a)
        c_max = c.max()
        c_top = np.argsort(c)[::-1][:30]

        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find the source position that appears in the most top peaks
        src_counts = np.zeros(n)
        for k_t in c_top[:15]:
            k_int = int(k_t)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                prods = a[idx] * a[k_int - idx]
                # Count contribution
                for ii in idx[np.argsort(prods)[::-1][:3]]:
                    src_counts[int(ii)] += 1

        # Try top source positions
        top_srcs = np.argsort(src_counts)[::-1][:10]
        top_srcs = [int(s) for s in top_srcs if a[s] > 1e-8]

        improved = False
        for src in top_srcs[:5]:
            if improved:
                break

            # Find best destination
            # For each candidate j, compute the worst-case new peak value
            n_cand = min(80, n)
            cands = np.argsort(a)[:n_cand]
            cands = [int(j) for j in cands if j != src]

            best_j = -1
            best_delta = 0
            best_new_max = c_max

            for j in cands[:25]:
                # Compute max delta allowed
                max_d = 0.05 * a[src]
                for k_t in c_top[:15]:
                    k_int = int(k_t)
                    ki = k_int - src
                    ai_val = a[ki] if 0 <= ki < n else 0.0
                    kj = k_int - j
                    aj_val = a[kj] if 0 <= kj < n else 0.0
                    denom = 2.0 * (ai_val - aj_val)
                    if denom > 1e-15:
                        d_lim = (c_max - c[k_int]) / denom
                        if d_lim < max_d:
                            max_d = d_lim

                if max_d < 1e-13:
                    continue

                delta = min(max_d, 0.03 * a[src])
                if delta < 1e-14:
                    continue

                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[j] += delta
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial /= s_t
                m_trial = fast_autocorr(a_trial).max()

                if m_trial < best_new_max:
                    best_new_max = m_trial
                    best_j = j
                    best_delta = delta

            if best_j >= 0 and best_new_max < best_m - 1e-16:
                a[src] -= best_delta
                a[best_j] += best_delta
                a = np.maximum(a, 0.0)
                a /= a.sum()
                best_m = best_new_max
                best_a = a.copy()
                no_improve = 0
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 500:
                # Random exploration
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        m_new = fast_autocorr(a).max()
                        if m_new < best_m:
                            best_m = m_new
                            best_a = a.copy()
                            no_improve = 0
                            break

        if verbose and no_improve % 1500 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def build_initial(n, seed=0):
    """Build a structured initial guess by trying several shapes."""
    x = np.linspace(0, 1, n)
    candidates = [
        (1.0 - x) ** 1.5 + 0.05,
        1.0 - (2.0 * x - 1.0) ** 2,
        np.exp(-2.0 * x),
        1.0 - np.abs(2.0 * x - 1.0),
        0.5 + 0.5 * np.cos(np.pi * x),
        np.linspace(1.0, 0.1, n),
        np.linspace(0.1, 1.0, n),
        np.exp(-5.0 * (x - 0.3) ** 2) + np.exp(-5.0 * (x - 0.7) ** 2) + 0.05,
        (1.0 + 0.5 * np.sin(2.0 * np.pi * x * 3)) * (1.0 - x),
        (1.0 - x) ** 0.8 + 0.1 * x,
    ]

    best = None
    best_score = float('inf')
    for c in candidates:
        c = np.maximum(c, 0.01)
        c = c / c.sum()
        sc = evaluate_safe(c)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = c.copy()

    if best is None:
        best = np.ones(n) / n
        best_score = evaluate_safe(best)

    return best, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Main search combining:
    1. Log-space Nesterov (fast convergence, stable)
    2. Optimal peak move surgery (principled redistribution)
    3. Spectral-aware local search (multi-peak consideration)
    4. Multi-length exploration
    5. Progressive refinement with increasing beta
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        best_seq, best_score = build_initial(n_base, seed=0)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(best_score))

    # Phase 1: Nesterov log-space (fast initial convergence)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Nesterov log-space (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.20, 330.0)
    if t1 > 60:
        theta0 = np.log(best_seq + 1e-10)
        theta0 -= theta0.mean()
        a_out, _ = nesterov_logspace(theta0, t1,
                                     beta_start=40.0, beta_end=2000.0,
                                     beta_growth=1.005, lr_base=0.9,
                                     top_k_ratio=0.02, restart_every=2500,
                                     verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 2: Optimal peak surgery
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Optimal Peak Surgery (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40:
        a_out, _ = peak_surgery_optimal(norm_sum1(best_seq), t2, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ops1", t_global)

    # Phase 3: Nesterov refine (higher beta)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Nesterov refine (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 60:
        theta0 = np.log(best_seq + 1e-10)
        theta0 -= theta0.mean()
        a_out, _ = nesterov_logspace(theta0, t3,
                                     beta_start=100.0, beta_end=5000.0,
                                     beta_growth=1.007, lr_base=0.5,
                                     top_k_ratio=0.01, restart_every=2000,
                                     verbose=True, seed=2)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 4: Spectral local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Spectral Local Search (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 40:
        a_out, _ = spectral_local_search(norm_sum1(best_seq), t4, verbose=True, seed=22)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n
