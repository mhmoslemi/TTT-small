def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_final = np.max(rho[cond])
    theta = (cssv[rho_final - 1] - 1) / rho_final
    return np.maximum(v - theta, 0.0)

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_final = rho[np.max(np.where(cond)[0])]
    theta = (cssv[rho_final - 1] - 1.0) / rho_final
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, p):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(A * P, fft_size)

    if len(crosscorr) < n:
        crosscorr = np.pad(crosscorr, (0, n - len(crosscorr)))
    crosscorr = crosscorr[:n]
    
    return crosscorr + a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)

import numpy as np

def fast_conv(a):
    """Full convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(np.where(cond)[0])
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def norm_sum1(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_gradient(a, p):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*conv(a,a)[k])).
    grad[i] = sum_k p[k]*a[k-i] + a[i]  (a[m]=0 outside [0,n-1])
    The sum_k p[k]*a[k-i] term is crosscorr[i] = sum_j a[j]*p[i+j].
    Computed via FFT: crosscorr = conv(a_rev, p)[n-1 : 2n-1]
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(A * P, fft_size)
    crosscorr = conv_result[n - 1:2 * n - 1]
    if len(crosscorr) < n:
        crosscorr = np.pad(crosscorr, (0, n - len(crosscorr)))
    crosscorr = crosscorr[:n]
    return crosscorr + a

    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            for s in [0.08, 0.12, 0.18]:
                if time.monotonic() - t0 > time_budget:
                    break
                a_raw = np.exp(-((x - c1) ** 2) / (2.0 * s * s)) + np.exp(-((x - c2) ** 2) / (2.0 * s * s))
                a = norm_sum1(a_raw)
                m = fast_conv(a).max()
                if m < best_m:
                    best_m = m
                    best_a = a.copy()

    return best_a, best_m


def interpolate_seq(f, new_n):
    """Resample sequence to new length preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization combining Nesterov LSE and peak shaving."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.1f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = eval_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        x = np.linspace(0, 1, n_base)
        a_raw = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(a_raw)
        best_score = eval_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Helper to update best
    def _update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.1f}s)")

    # Phase 1: Parametric initialization
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.02, 25.0)
    if t1 > 5:
        print("\n=== Phase 1: Parametric init (n={}) ===".format(n0))
        a_param, _ = parametric_init(n0, t1)
        _update(a_param, "param_n" + str(n0))

    # Phase 2: Nesterov LSE - main optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Nesterov LSE (n={}) ===".format(n_best))
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        a_out, _ = nesterov_lse(best_seq, t2, beta_start=3.0, beta_end=400.0,
                                beta_growth=1.005, verbose=True, seed=42)
        _update(a_out, "nest_n" + str(n_best))

    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Peak shaving (n={}) ===".format(n_best))
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 30:
        a_out, _ = peak_shaving(best_seq, t3, top_k=30, verbose=True, seed=99)
        _update(a_out, "psav_n" + str(n_best))

    # Phase 4: Nesterov LSE round 2 (higher beta for finer tuning)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Nesterov LSE round 2 (n={}) ===".format(n_best))
    t4 = min(remaining * 0.22, 300.0)
    if t4 > 50:
        a_out, _ = nesterov_lse(best_seq, t4, beta_start=30.0, beta_end=1500.0,
                                beta_growth=1.004, verbose=True, seed=123)
        _update(a_out, "nest2_n" + str(n_best))

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby = []
    for dn in [-400, -200, -100, 100, 200, 400, 600]:
        nl = n_best + dn
        if 4000 < nl <= 9000 and nl not in nearby:
            nearby.append(nl)

    for new_n in nearby[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("\n=== Phase 5: Exploring n={} ===".format(new_n))
        a_long = interpolate_seq(best_seq, new_n)

        t5a = min(remaining * 0.45, 200.0)
        if t5a > 30:
            a_out, _ = nesterov_lse(a_long, t5a, beta_start=5.0, beta_end=500.0,
                                     beta_growth=1.005, verbose=False, seed=new_n % 997)
            _update(a_out, "nest_n" + str(new_n))

        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.3, 80.0)
        if t5b > 20:
            # Peak shave on the best at this length
            current_best_at_len = None
            if len(best_seq) == new_n:
                current_best_at_len = best_seq
            else:
                current_best_at_len = interpolate_seq(best_seq, new_n)
            a_out, _ = peak_shaving(current_best_at_len, t5b, top_k=25, verbose=False, seed=new_n % 997)
            _update(a_out, "psav_n" + str(new_n))

    # Phase 6: Final peak shaving polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Final peak shaving (n={}) ===".format(n_best))
    if remaining > 40:
        t6 = min(remaining * 0.35, 120.0)
        if t6 > 20:
            a_out, _ = peak_shaving(best_seq, t6, top_k=20, verbose=True, seed=777)
            _update(a_out, "final_psav")

    # Phase 7: Final Nesterov with very high beta
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Final Nesterov high-beta (n={}) ===".format(n_best))
    if remaining > 40:
        t7 = min(remaining * 0.4, 100.0)
        if t7 > 20:
            a_out, _ = nesterov_lse(best_seq, t7, beta_start=200.0, beta_end=5000.0,
                                     beta_growth=1.003, verbose=True, seed=555)
            _update(a_out, "final_nest")

    # Final check
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.1f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        a_raw = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(a_raw)
        best_score = eval_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
