def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:

    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c

def _line_search(a, g, c, cg, fg, mask, current_M):
    """Backtracking line search for FW step."""
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    n = a.size
    
    alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
    best_alpha = 0.0
    best_m = current_M
    
    for alpha in alphas:
        one = 1.0 - alpha
        val = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        if mask is not None:
            m = float(np.max(val[mask]))
        else:
            m = float(np.max(val))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha
    
    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0
    
    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha

def _rank1_move_eval(c, a, s, d, tmove):

        vals = c[top_idx]
        w = vals / float(vals.sum())
        p_top[top_idx] = w
        
        pressure = _cross_pressure(p_top, a)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)
        
        # Find best source (high pressure + high mass) and destination (low pressure)
        src_score = pressure * a  # We want to move FROM high-pressure-high-mass
        # Actually we want to move mass from high-pressure positions to low-pressure positions
        # Source: high pressure positions (reducing their contribution to peaks)
        # Dest: low pressure positions (increasing their contribution to fill in valleys)
        
        src_candidates = np.argpartition(-src_score, min(10, n - 1))[:10]
        dst_score = -pressure  # Low pressure is good destination
        dst_candidates = np.argpartition(dst_score, min(15, n - 1))[:15]
        
        best_cand = None
        best_cand_M = M
        
        for srr in src_candidates:
            s = int(srr)
            if a[s] < 1e-10:
                continue
            for drr in dst_candidates:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.01, 0.02, 0.05, 0.1):
                    tmove = a[s] * frac
                    if tmove <= 1e-14:
                        continue
                    tmove = min(tmove, a[s] * 0.95)
                    c_new = _rank1_move_eval(c, a, s, d, tmove)
                    m_new = float(np.max(c_new))
                    if math.isfinite(m_new) and m_new < best_cand_M - 1e-16:
                        best_cand_M = m_new
                        best_cand = (m_new, s, d, tmove)
        
        # Also try some random moves
        for _ in range(20):
            s = int(rng.integers(n))
            d = int(rng.integers(n))
            if s == d or a[s] < 1e-10:
                continue
            frac = float(rng.uniform(0.005, 0.08))
            tmove = a[s] * frac
            tmove = min(tmove, a[s] * 0.95)
            if tmove <= 1e-14:
                continue
            c_new = _rank1_move_eval(c, a, s, d, tmove)
            m_new = float(np.max(c_new))
            if math.isfinite(m_new) and m_new < best_cand_M - 1e-16:
                best_cand_M = m_new
                best_cand = (m_new, s, d, tmove)
        
        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = _fft_conv(a)
            M = float(np.max(c))
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 3 or rnd % 30 == 0):
                print("    Polish rnd=" + str(rnd) + " proxy=" + repr(2.0 * n * best_M))
        else:
            stall += 1
            if stall % 10 == 8:
                # Try a bigger random perturbation
                rng2 = np.random.default_rng(55555 + rnd)
                pert = rng2.normal(0.0, 0.03, n) * best_a
                a_try = best_a + pert
                a_try = np.maximum(a_try, 1e-12)
                a_try = a_try / float(a_try.sum())
                c_try = _fft_conv(a_try)
                M_try = float(np.max(c_try))
                if M_try < best_M:
                    a = a_try
                    c = c_try
                    M = M_try
                    stall = 0
                else:
                    a = best_a.copy()
                    c = _fft_conv(a)
                    M = best_M
    
    return best_a


def _lse_objgrad(v, beta, top_count, ref, a_cache):
    """Log-sum-exp smoothed objective in softmax coordinates. Returns (f, grad)."""
    n = v.size + 1
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    
    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = np.maximum(v[:ref], -100.0)
    if ref < n - 1:
        u[ref + 1:] = np.maximum(v[ref:], -100.0)
    
    u_max = float(np.max(u))
    e = np.exp(u - u_max)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = e / s
    
    # Compute convolution
    c = _fft_conv(a)
    size = 2 * n - 1
    
    # Select top peaks for the smoothed max
    if top_count is not None and top_count < size:
        top = np.argsort(c)[::-1][:top_count]
        cv = c[top]
        c_full = c  # still need full conv for gradient
    else:
        cv = c
        c_full = c
    
    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    se = float(exp_vals.sum())
    if se <= 0.0 or not math.isfinite(se):
        exp_vals = np.zeros(cv.size, dtype=np.float64)
        exp_vals[0] = 1.0
        se = 1.0
    p = exp_vals / se
    
    # Gradient: grad_a = 2 * sum_k p_k * (shifted_a for each k)
    # We need: for each k in top, the gradient of c[k] w.r.t. a
    # c[k] = sum_i a_i * a_{i+k}, grad_i c[k] = a_{i-k} + a_{i+k}
    # So grad_a[i] = 2 * sum_k p_k * (a_{i-k} + a_{i+k})
    
    if top_count is not None and top_count < size:
        p_full = np.zeros(size, dtype=np.float64)
        p_full[top] = p
    else:
        p_full = p
    
    pressure = _cross_pressure(p_full, a)
    grad_a = 2.0 * pressure
    grad_a = np.where(np.isfinite(grad_a), grad_a, 0.0)
    
    f = m + math.log(se) / beta
    
    # Transform to v coordinates
    mean = float(np.dot(a, grad_a))
    gu = a * (grad_a - mean)
    
    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]
    
    gv = np.where(np.isfinite(gv), gv, 0.0)
    
    if not math.isfinite(f):
        f = 1e100
        gv = np.zeros(v.size, dtype=np.float64)
    
    return f, gv


def _lbfegs_phase(a0, budget, verbose=True):
    """L-BFGS on smoothed objective with adaptive beta."""
    if not SCIPY_OK:
        return a0
    
    a = _norm1(a0)
    n = a.size
    if n <= 1:
        return a
    
    # Set reference index (use argmax for better conditioning)
    ref = int(np.argmax(a))
    
    # Build v from a
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -100.0)
    
    t0 = time.monotonic()
    best_v = v.copy()
    best_a_out = a.copy()
    
    # Schedule of (beta, top_count, maxfun)
    schedules = [
        (10.0, None, 150),
        (30.0, None, 150),
        (60.0, max(100, (2 * n - 1) // 10), 120),
        (120.0, max(50, (2 * n - 1) // 20), 100),
        (250.0, max(30, (2 * n - 1) // 40), 80),
        (500.0, max(20, (2 * n - 1) // 80), 60),
    ]
    
    for beta, top_count, maxfun in schedules:
        elapsed = time.monotonic() - t0
        if elapsed > budget - 30:
            break
        
        # Define objective function for scipy
        def _objfun(vx):
            f, g = _lse_objgrad(vx, beta, top_count, ref, None)
            return f, g
        
        options = {
            "maxfun": int(maxfun),
            "maxiter": int(maxfun + 20),
            "ftol": 1e-12,
            "gtol": 1e-7,
            "disp": False,
        }
        
        try:
            res = scipy.optimize.minimize(
                _objfun, v, method="L-BFGS-B", jac=True, options=options
            )
        except Exception:
            break
        
        if res is not None and hasattr(res, "x"):
            vx = np.asarray(res.x, dtype=np.float64)
            if vx.size == v.size and np.all(np.isfinite(vx)):
                # Convert back to a
                u = np.empty(n, dtype=np.float64)
                u[ref] = 0.0
                if ref > 0:
                    u[:ref] = vx[:ref]
                if ref < n - 1:
                    u[ref + 1:] = vx[ref:]
                umx = float(np.max(u))
                e = np.exp(u - umx)
                ss = float(e.sum())
                if ss > 0 and math.isfinite(ss):
                    a_new = e / ss
                    a_new = _norm1(a_new)
                    M_new = float(np.max(_fft_conv(a_new)))
                    M_best = float(np.max(_fft_conv(best_a_out)))
                    if M_new < M_best:
                        best_v = vx.copy()
                        best_a_out = a_new.copy()
                        v = vx.copy()
                    if verbose:
                        print("    L-BFGS beta=" + str(beta) + " M=" + repr(M_new))
    
    return best_a_out


def _interpolate(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(2, new_n))
    if a.size == new_n:
        return _norm1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return _norm1(y)


def _default_seq(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return _norm1(np.sin(np.pi * x) ** 2)


def propose_candidate(budget_s=1800.0):
    """Main search function. Returns best sequence found within budget."""
    t_start = time.monotonic()
    total_budget = max(60.0, min(float(budget_s) - 60.0, 1740.0))
    
    print("Starting: budget=" + repr(total_budget) + "s")
    
    # Initialize from parent if available
    seq_best = None
    try:
        parent_list = height_sequence_1
        p = np.asarray(parent_list, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        np.clip(p, 0.0, 1000.0, out=p)
        if p.ndim != 1:
            p = p.ravel()
        if p.size > 0 and float(p.sum()) > 1e-10:
            seq_best = _norm1(p)
    except Exception:
        seq_best = None
    
    if seq_best is None:
        seq_best = _default_seq(5000)
    
    best_score = _safe_eval(seq_best)
    if not math.isfinite(best_score):
        best_score = _proxy_obj(seq_best)
    print("Initial: n=" + str(seq_best.size) + " score=" + repr(best_score))
    
    # Working sequence (compressed if too long)
    if seq_best.size > 9000:
        work_seq = _interpolate(seq_best, 8000)
    else:
        work_seq = seq_best.copy()
    
    # Check if work_seq is better
    ws_score = _safe_eval(work_seq)
    if math.isfinite(ws_score) and ws_score < best_score:
        best_score = ws_score
        seq_best = work_seq.copy()
        work_seq = seq_best.copy()
    
    # Phase 1: Super Frank-Wolfe (main workhorse)
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 300:
        phase_budget = min(time_left * 0.40, 650.0)
        print("\n=== Phase 1: Super FW (budget=" + repr(phase_budget) + "s) ===")
        a_fw = _fw_search(work_seq, phase_budget, active_max=2500, game_iters=28, verbose=True)
        sc = _safe_eval(a_fw)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            seq_best = a_fw.copy()
            work_seq = seq_best.copy() if seq_best.size <= 9000 else _interpolate(seq_best, 8000)
            print("  NEW BEST after FW: " + repr(best_score))
    
    # Phase 2: Coordinate polish
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        phase_budget = min(time_left * 0.28, 480.0)
        print("\n=== Phase 2: Coordinate polish (budget=" + repr(phase_budget) + "s) ===")
        a_cp = _coord_polish(work_seq, phase_budget, max_rounds=800, verbose=True)
        sc = _safe_eval(a_cp)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            seq_best = a_cp.copy()
            work_seq = seq_best.copy() if seq_best.size <= 9000 else _interpolate(seq_best, 8000)
            print("  NEW BEST after polish: " + repr(best_score))
    
    # Phase 3: L-BFGS refinement
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if SCIPY_OK and time_left > 150:
        phase_budget = min(time_left * 0.20, 300.0)
        print("\n=== Phase 3: L-BFGS (budget=" + repr(phase_budget) + "s) ===")
        a_lb = _lbfegs_phase(work_seq, phase_budget, verbose=True)
        sc = _safe_eval(a_lb)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            seq_best = a_lb.copy()
            work_seq = seq_best.copy() if seq_best.size <= 9000 else _interpolate(seq_best, 8000)
            print("  NEW BEST after L-BFGS: " + repr(best_score))
    
    # Phase 4: Another FW round (finer)
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 200:
        phase_budget = min(time_left * 0.25, 350.0)
        print("\n=== Phase 4: FW round 2 (budget=" + repr(phase_budget) + "s) ===")
        a_fw2 = _fw_search(work_seq, phase_budget, active_max=1500, game_iters=20, verbose=True)
        sc = _safe_eval(a_fw2)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            seq_best = a_fw2.copy()
            work_seq = seq_best.copy() if seq_best.size <= 9000 else _interpolate(seq_best, 8000)
            print("  NEW BEST after FW2: " + repr(best_score))
    
    # Phase 5: Final polish
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 80:
        phase_budget = min(time_left * 0.5, 200.0)
        print("\n=== Phase 5: Final polish (budget=" + repr(phase_budget) + "s) ===")
        a_cp2 = _coord_polish(work_seq, phase_budget, max_rounds=500, verbose=True)
        sc = _safe_eval(a_cp2)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            seq_best = a_cp2.copy()
            print("  NEW BEST after final polish: " + repr(best_score))
    
    # Phase 6: Length probes
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 250:
        print("\n=== Phase 6: Length probes ===")
        base_n = seq_best.size if seq_best.size <= 9000 else 8000
        try_lengths = [base_n + 500, base_n - 500, base_n + 1000, base_n - 1000]
        tried = set()
        for target in try_lengths:
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 120:
                break
            target = int(max(2000, min(9000, target)))
            if target in tried:
                continue
            tried.add(target)
            print("  Trying n=" + str(target))
            
            a0 = _interpolate(seq_best, target)
            # Quick FW
            a_try = _fw_search(a0, min(60.0, time_left * 0.3), active_max=1000, game_iters=15, verbose=False)
            sc = _safe_eval(a_try)
            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                seq_best = a_try.copy()
                print("  NEW BEST at n=" + str(target) + ": " + repr(best_score))
            
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left > 80:
                a_try2 = _coord_polish(a_try, min(40.0, time_left * 0.3), max_rounds=200, verbose=False)
                sc = _safe_eval(a_try2)
                if math.isfinite(sc) and sc < best_score:
                    best_score = sc
                    seq_best = a_try2.copy()
                    print("  NEW BEST (polished) at n=" + str(target) + ": " + repr(best_score))
    
    # Final: one more polish on the best
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 60 and seq_best.size <= 9000:
        a_final = _coord_polish(seq_best, min(time_left * 0.7, 100.0), max_rounds=300, verbose=False)
        sc = _safe_eval(a_final)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            seq_best = a_final.copy()
            print("  Final polish improvement: " + repr(best_score))
    
    # Validate and return
    final_val = _safe_eval(seq_best)
    if math.isfinite(final_val):
        if final_val < best_score:
            best_score = final_val
    
    arr = np.asarray(seq_best, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = _default_seq(5000)
    else:
        arr = arr / s
    
    elapsed = time.monotonic() - t_start
    final_score = _safe_eval(arr)
    print("\nFINAL: score=" + repr(final_score) + " n=" + str(arr.size) + " elapsed=" + repr(elapsed) + "s")
    
    if not math.isfinite(final_score):
        # Fallback
        arr = _default_seq(5000)
        print("FALLBACK to default: " + repr(_safe_eval(arr)))
    
    return [float(x) for x in arr]
