    """Smoothed objective optimization via L-BFGS with continuation on tau."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    tau = tau_start
    n_phases = 0

    while time.monotonic() - t0 < time_limit and tau <= tau_max:
        n_phases += 1

        def fun_grad(x):
            x_clip = np.maximum(x, 0.0)
            s = x_clip.sum()
            if s < 1e-12:
                return 2.0 * n * n, np.zeros(n)
            x_n = x_clip / s
            c = fast_convolve(x_n)
            cm = c.max()
            shifted = tau * (c - cm)
            exp_shifted = np.exp(shifted)
            p_sum = exp_shifted.sum()
            val = 2.0 * n * (cm + np.log(p_sum) / tau)
            p = exp_shifted / p_sum
            fft_size = 1 << ((2 * n - 1) + n - 1 - 1).bit_length()
            Pf = np.fft.rfft(p, fft_size)
            Af = np.fft.rfft(x_n, fft_size)
            corr = np.fft.irfft(np.conj(Pf) * Af, fft_size)[:n]
            grad = 2.0 * corr
            grad = grad - grad.mean()
            return val, grad

        try:
            res = minimize(fun_grad, a, method='L-BFGS-B', jac=True,
                           options={'maxiter': 800, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50})
            a_new = np.maximum(res.x, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-12:
                a_new = a_new / s_new
                obj_new = obj_val(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose:
                        print("      L-BFGS phase=" + str(n_phases) + " tau=" + str(int(tau)) + " obj=" + repr(best_obj) + " t=" + str(round(time.monotonic() - t0, 1)) + "s")
                a = a_new
        except Exception:
            pass

        tau *= tau_mult

    if verbose:
        print("    L-BFGS done: phases=" + str(n_phases) + " best_obj=" + repr(best_obj) + " t=" + str(round(time.monotonic() - t0, 1)) + "s")
    return best_a, best_obj


def projected_subgrad(a0, time_limit, verbose=False):
    """Projected subgradient with momentum on the simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    vel = np.zeros(n)
    momentum = 0.7
    eta0 = 0.02
    stall = 0
    t = 0

    while time.monotonic() - t0 < time_limit and stall < 8000:
        t += 1
        eta = eta0 / (1.0 + 0.02 * t)

        c = fast_convolve(a)
        k_star = int(np.argmax(c))

        grad = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] = a[k_star - idx]
            grad[k_star - idx] += a[idx]
            for j in idx:
                if k_star - j == j:
                    grad[j] = 2.0 * a[j]

        grad = grad - grad.mean() + 0.005 * a
        vel = momentum * vel - eta * grad
        a_new = a + vel
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        obj_new = obj_val(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 500 == 0:
                a = best_a + 0.02 * rng.standard_normal(n) * best_a
                a = norm_sum1(a)
                vel *= 0.5
                momentum = max(0.3, momentum * 0.95)

    if verbose:
        print("    Subgrad done: obj=" + repr(best_obj) + " iters=" + str(t) + " t=" + str(round(time.monotonic() - t0, 1)) + "s")
    return best_a, best_obj


def build_constraint_matrix(y, constraint_list, n):
    K = len(constraint_list)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_fw(a0, time_limit, solver, initial_K=400, max_K=3000, verbose=False):
    """LP Frank-Wolfe with adaptive active set."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_convolve(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac_y)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 600
    iter_count = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation > 4:
                break
            ac_curr = fast_convolve(y)
            n_pert = max(30, len(active_set) // 4)
            rng_s = np.random.default_rng(stagnation * 7)
            keep = list(rng_s.choice(len(active_set), size=len(active_set) - n_pert, replace=False))
            new_adds = np.argsort(ac_curr)[::-1][:n_pert * 2]
            new_set = set([active_set[i] for i in keep])
            for k in new_adds:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
            no_improve = 0
            stagnation += 1
            continue

        iter_count += 1
        ac = fast_convolve(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_idx = np.argsort(margins)[:int(max_K * 0.6)]
            active_set_base = [active_set[i] for i in keep_idx]
            violations = np.where(ac > M_curr - 1e-8)[0]
            for v in violations[:80]:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))[:max_K]

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_constraint_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = sorted(active_set[:max(50, K // 4)])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violations to active set
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]
        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:40]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        # Backtracking line search
        best_alpha = 1.0 / (iter_count + 2.0)
        best_trial_M = fast_convolve(y).max()
        alphas = [1.0, 0.8, 0.5, 0.3, 0.15, max(0.05, 2.0 / (iter_count + 2)), max(0.02, 1.0 / (iter_count + 1))]
        for alpha in alphas:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_convolve(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if iter_count <= 2:
            y = g_dir.copy()
        else:
            y = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        if verbose and iter_count % 200 == 0:
            print("      LP-FW iter=" + str(iter_count) + " J=" + repr(2.0 * n * best_m) + " |S|=" + str(len(active_set)) + " t=" + str(round(time.monotonic() - t0, 1)) + "s")

    if verbose:
        print("    LP-FW done: J=" + repr(2.0 * n * best_m) + " iters=" + str(iter_count) + " t=" + str(round(time.monotonic() - t0, 1)) + "s")
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False):
    """Peak-shifting coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    stall = 0
    max_stall = 2000
    step_scales = [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        n_top = min(20, 2 * n - 1)
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Find best destination (low contribution to peaks)
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            candidates = np.where(scores < scores.mean())[0]
            if len(candidates) < n // 4:
                candidates = np.arange(n)

            dst = int(candidates[np.argmin(scores[candidates])])
            if dst == src:
                dst = int(rng.integers(n))
                if dst == src:
                    continue

            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found and rng.random() < 0.3:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-12:
                    amt = a[src] * rng.uniform(0.002, 0.01)
                    a[src] -= amt
                    a[j] += amt
                    a = norm_sum1(a)
                    M_new = fast_convolve(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            if stall % 500 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-12:
                        amt = a[i] * rng.uniform(0.003, 0.02)
                        a[i] -= amt
                        a[j] += amt
                        a = norm_sum1(a)
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

    if verbose:
        print("    Coord done: obj=" + repr(2.0 * n * best_m) + " stall=" + str(stall) + " t=" + str(round(time.monotonic() - t0, 1)) + "s")
    return best_a, best_m


def get_initial_sequence():
    """Get starting sequence from parent or default."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                print("Parent: n=" + str(len(p)) + " score=" + repr(sc))
                return p_norm, sc
    except Exception:
        pass

    n = 5000
    x = np.linspace(0, 1, n)
    best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_score = evaluate_safe(best_seq)
    print("Default init: n=" + str(n) + " score=" + repr(best_score))
    return best_seq, best_score


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence()
    n0 = len(best_seq)

    def remaining():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Smoothed L-BFGS on parent ===
    print("\n=== Phase 1: L-BFGS (n=" + str(n0) + ") ===")
    t1 = min(remaining() * 0.15, 220.0)
    if t1 > 30:
        a_out, _ = smooth_lbfgs(norm_sum1(best_seq), t1, tau_start=3.0, tau_mult=2.5, tau_max=2000.0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs1", t_global)

    # === Phase 2: LP-FW ===
    print("\n=== Phase 2: LP-FW (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining() * 0.15, 220.0)
    if t2 > 40 and len(best_seq) <= 12000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t2, solver, initial_K=500, max_K=3500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1", t_global)

    # === Phase 3: L-BFGS second pass ===
    print("\n=== Phase 3: L-BFGS 2nd (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining() * 0.12, 180.0)
    if t3 > 30:
        a_out, _ = smooth_lbfgs(norm_sum1(best_seq), t3, tau_start=10.0, tau_mult=2.0, tau_max=5000.0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs2", t_global)

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n ===")
    n_best = len(best_seq)
    n_candidates = sorted(set([
        n_best - 500, n_best + 500,
        n_best - 1000, n_best + 1000,
        n_best + 2000,
        max(3000, n_best - 2000),
    ]))
    n_candidates = [x for x in n_candidates if 2500 <= x <= 15000]
    n_candidates.sort(key=lambda x: abs(x - n_best))

    for new_n in n_candidates[:5]:
        rt = remaining()
        if rt < 150:
            break
        print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_interp = interpolate_seq(best_seq, new_n)

        t4a = min(rt * 0.18, 140.0)
        if t4a > 30:
            a_out, _ = smooth_lbfgs(a_interp, t4a, tau_start=5.0, tau_mult=2.5, tau_max=3000.0, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_n" + str(new_n), t_global)

        rt = remaining()
        if rt > 80 and len(best_seq) <= 12000:
            t4b = min(rt * 0.12, 80.0)
            if t4b > 25:
                a_out, _ = lp_fw(norm_sum1(best_seq), t4b, solver, initial_K=400, max_K=2500, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_n" + str(new_n), t_global)

    # === Phase 5: Subgradient on best ===
    print("\n=== Phase 5: Subgradient (n=" + str(len(best_seq)) + ") ===")
    t5 = min(remaining() * 0.08, 100.0)
    if t5 > 30:
        a_out, _ = projected_subgrad(norm_sum1(best_seq), t5, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "subgrad", t_global)

    # === Phase 6: L-BFGS third pass ===
    print("\n=== Phase 6: L-BFGS 3rd (n=" + str(len(best_seq)) + ") ===")
    t6 = min(remaining() * 0.12, 160.0)
    if t6 > 30:
        a_out, _ = smooth_lbfgs(norm_sum1(best_seq), t6, tau_start=50.0, tau_mult=2.0, tau_max=8000.0, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs3", t_global)

    # === Phase 7: LP-FW second pass ===
    print("\n=== Phase 7: LP-FW 2nd (n=" + str(len(best_seq)) + ") ===")
    t7 = min(remaining() * 0.12, 160.0)
    if t7 > 40 and len(best_seq) <= 12000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t7, solver, initial_K=600, max_K=4000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2", t_global)

    # === Phase 8: Coordinate polish ===
    print("\n=== Phase 8: Coord polish (n=" + str(len(best_seq)) + ") ===")
    t8 = min(remaining() * 0.10, 130.0)
    if t8 > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t8, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord", t_global)

    # === Phase 9: Final L-BFGS + coord ===
    print("\n=== Phase 9: Final polish ===")
    t9 = min(remaining() * 0.10, 120.0)
    if t9 > 20:
        a_out, _ = smooth_lbfgs(norm_sum1(best_seq), t9 * 0.6, tau_start=100.0, tau_mult=2.0, tau_max=10000.0, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lbfgs_final", t_global)

    rt = remaining()
    if rt > 20:
        a_out, _ = coordinate_polish(norm_sum1(best_seq), rt * 0.7, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # === Final validation ===
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]
