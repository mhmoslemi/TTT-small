def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of (1/beta)*log(sum top_k exp(beta*c)) w.r.t. a."""
    n = len(y)
    c = fast_autocorr(y)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total

    fft_size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    a_rev = y[::-1]
    A_rev = np.fft.rfft(a_rev, fft_size)
    Conv = np.fft.irfft(P * A_rev, fft_size)
    grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]
    return grad, p, c


def compute_subgrad_at_max(a):
    """Compute exact subgradient of max_k c[k] at the argmax."""
    n = len(a)
    c = fast_autocorr(a)
    k_star = int(np.argmax(c))
    # d(c[k_star])/d(a_j) = 2*a[k_star - j] (with a[x]=0 outside [0,n-1])
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    return grad, c, k_star

            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def peak_surgery(a0, time_limit, verbose=True, seed=77):
    """Targeted local search: identify top peaks and reduce them via mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_top = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]

        improved = False
        found_move = False

        for k in top_idx[:10]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue

            # Top 3 product pairs
            top_pairs = np.argsort(products)[::-1][:3]
            pi = int(top_pairs[0])
            src_i = int(idx[pi])
            partner = k_int - src_i
            if partner < 0 or partner >= n:
                continue

            if a[src_i] >= a[partner]:
                reduce_i = src_i
            else:
                reduce_i = partner

            if a[reduce_i] < 1e-14:
                continue

            # Compute pressure
            pressure = np.zeros(n)
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    pressure[idx2] += a[k2i - idx2]
                    pressure[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_idx[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 100:
                candidates = np.arange(n)

            mask = candidates != reduce_i
            candidates = candidates[mask]
            if len(candidates) == 0:
                continue

            cand_pressure = pressure[candidates]
            n_try = min(8, len(candidates))
            try_dsts = candidates[np.argsort(cand_pressure)[:n_try]]

            for dst in try_dsts:
                dst_int = int(dst)
                if dst_int == reduce_i:
                    continue
                for frac in [0.003, 0.008, 0.015, 0.03, 0.05]:
                    amt = a[reduce_i] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_i] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        break
                if found_move:
                    break

        if not improved:
            if no_improve > 300:
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.04)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakSurgery (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(no_improve))

    return best_a, best_m


def pairwise_cd(a0, time_limit, verbose=True, seed=123):
    """Fast pairwise coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        threshold = np.percentile(c, 97)
        peak_positions = np.where(c > threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:15]

        improved = False

        for k in peak_positions[:12]:
            if improved and (time.monotonic() - t0 > time_limit):
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            top_pairs = np.argsort(products)[::-1][:2]
            for pi in top_pairs:
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n or partner == src_i:
                    continue
                if a[src_i] < 1e-14 and a[partner] < 1e-14:
                    continue

                larger_i = src_i if a[src_i] >= a[partner] else partner
                if a[larger_i] < 1e-14:
                    continue

                for frac in [0.001, 0.003, 0.005, 0.01]:
                    amt = a[larger_i] * frac
                    if amt < 1e-18:
                        continue
                    for _ in range(4):
                        dst = int(rng.integers(n))
                        if dst == larger_i:
                            continue
                        a_trial = a.copy()
                        a_trial[larger_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall % 150 == 0:
                anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                eps = 0.008 / (1.0 + stall / 100.0)
                a_try = a + eps * anti * a
                a_try = np.maximum(a_try, 0.0)
                s_t = a_try.sum()
                if s_t > 1e-15:
                    a_try = a_try / s_t
                    M_try = fast_autocorr(a_try).max()
                    if M_try < best_m:
                        a = a_try
                        best_m = M_try
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 400 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PairCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + repr(sc) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def build_asymmetric_init(n, seed=0):
    """Create an asymmetric initialization biased towards edges."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Bimodal: mass at edges
    base = np.exp(-8.0 * (x - 0.1)) + np.exp(-8.0 * (x - 0.9))
    # Add some noise
    noise = rng.normal(1.0, 0.1, n)
    a = np.maximum(base * noise, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization with Nesterov, peak surgery, pairwise CD,
    and multi-length exploration. Longer sequences are preferred when quality ratio holds."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent loaded: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    # Also create diverse initializations
    inits = []
    if best_seq is not None:
        inits.append(("parent", best_seq.copy()))

    # Asymmetric init at parent length
    if parent is not None:
        n_p = len(parent)
        inits.append(("asym", build_asymmetric_init(n_p, seed=100)))

    # Different asymmetric inits at various lengths
    for n_try, sd in [(5000, 200), (8000, 300), (12000, 400)]:
        inits.append(("asym_n" + str(n_try), build_asymmetric_init(n_try, seed=sd)))

    # Power-law decays
    for n_try, gamma, sd in [(6000, 1.0, 500), (9000, 0.7, 600)]:
        x = np.linspace(0, 1, n_try)
        a_init = (1.0 - x) ** gamma + 0.05 * np.random.default_rng(sd).normal(1, 0.1, n_try)
        a_init = np.maximum(a_init, 0)
        inits.append(("pow_n" + str(n_try), norm_sum1(a_init)))

    # Evaluate all inits
    for label, a_init in inits:
        sc = evaluate_safe(norm_sum1(a_init))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = norm_sum1(a_init).copy()
            print("  Init [" + label + "]: score=" + repr(sc) + " n=" + str(len(a_init)))

    if best_seq is None:
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)
        print("Fallback init: n=" + str(n_fb) + ", score=" + repr(best_score))

    n0 = len(best_seq)
    print("\nStarting best: n=" + str(n0) + ", score=" + repr(best_score))

    # === Phase 1: Nesterov on best sequence (high beta, focus on max peak) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: Nesterov high-beta (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 50:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t1,
                                beta_start=50.0, beta_end=3000.0,
                                beta_growth=1.008, eta_base=0.5,
                                top_k_ratio=0.015, restart_every=1500,
                                verbose=True, seed=1)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # === Phase 2: Peak surgery ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Peak Surgery (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 30:
        a_out, _ = peak_surgery(norm_sum1(best_seq), t2, verbose=True, seed=11)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps1", t_global)

    # === Phase 3: Nesterov with subgradient mixing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Nesterov + subgrad mix (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 50:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t3,
                                beta_start=100.0, beta_end=5000.0,
                                beta_growth=1.01, eta_base=0.4,
                                top_k_ratio=0.01, restart_every=1200,
                                verbose=True, seed=2, use_subgrad_mix=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_sg", t_global)

    # === Phase 4: Multi-length exploration (key for improvement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: Multi-length exploration ===")

    # Try longer sequences - the objective improves with n if quality ratio holds
    longer_lengths = []
    for delta in [1000, 2000, 3000, 5000]:
        ln = n_best + delta
        if ln > n_best and ln <= 20000:
            longer_lengths.append(ln)
    # Also try some specific lengths
    for ln in [9000, 12000, 15000]:
        if ln > n_best and ln not in longer_lengths and ln <= 20000:
            longer_lengths.append(ln)

    longer_lengths = sorted(set(longer_lengths))[:4]  # Limit to 4 tries

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n) + " (remaining=" + str(round(remaining, 1)) + "s)")

        a_long = interpolate_sequence(best_seq, new_n)
        sc_long = evaluate_safe(a_long)
        print("    Interpolated score: " + repr(sc_long))

        t4a = min(remaining * 0.4, 180.0)
        if t4a > 40:
            a_out, _ = nesterov_opt(a_long, t4a,
                                    beta_start=40.0, beta_end=2500.0,
                                    beta_growth=1.007, eta_base=0.5,
                                    top_k_ratio=0.015, restart_every=1200,
                                    verbose=False, seed=3)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t4b = min(remaining * 0.35, 100.0)
            if t4b > 25:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = peak_surgery(a_try, t4b, verbose=False, seed=33)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "mlps_n" + str(new_n), t_global)

    # === Phase 5: Nesterov refine at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Nesterov refine (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.18, 280.0)
    if t5 > 50:
        a_out, _ = nesterov_opt(norm_sum1(best_seq), t5,
                                beta_start=150.0, beta_end=6000.0,
                                beta_growth=1.01, eta_base=0.35,
                                top_k_ratio=0.008, restart_every=1000,
                                verbose=True, seed=4, use_subgrad_mix=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_refine", t_global)

    # === Phase 6: Peak surgery ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Peak Surgery 2 (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 30:
        a_out, _ = peak_surgery(norm_sum1(best_seq), t6, verbose=True, seed=22)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "ps2", t_global)

    # === Phase 7: Pairwise CD for polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: Pairwise CD (n=" + str(n_best) + ") ===")
    t7 = min(remaining * 0.25, 300.0)
    if t7 > 30:
        a_out, _ = pairwise_cd(norm_sum1(best_seq), t7, verbose=True, seed=55)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd1", t_global)

    # === Phase 8: Final Nesterov burst ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: Final Nesterov (n=" + str(n_best) + ") ===")
    if remaining > 50:
        t8 = min(remaining * 0.35, 200.0)
        if t8 > 30:
            a_out, _ = nesterov_opt(norm_sum1(best_seq), t8,
                                    beta_start=300.0, beta_end=10000.0,
                                    beta_growth=1.012, eta_base=0.25,
                                    top_k_ratio=0.005, restart_every=800,
                                    verbose=True, seed=5, use_subgrad_mix=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    remaining = time_budget - (time.monotonic() - t_global)

    if remaining > 25:
        t9 = min(remaining * 0.6, 150.0)

        target_n = min(n_best + 3000, 25000)
        if target_n > n_best:
            a_push = interpolate_sequence(best_seq, target_n)
            t10a = min(remaining * 0.5, 150.0)

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:

                if t10b > 20:
                    a_out, _ = peak_surgery(norm_sum1(best_seq), t10b, verbose=False, seed=88)
                    best_seq, best_score = try_update(best_seq, best_score, a_out, "push_ps", t_global)

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((1.0 - x) ** 1.5 + 0.1)
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
