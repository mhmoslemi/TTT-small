    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVEr = s
                return s
        except Exception:
            continue
    _SOLVEr = cp.ECOS
    return _SOLVEr

def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def score(a):
    a = normalize(a)
    return 2.0 * len(a) * fast_conv(a).max()

def subgradient(ac, a):
    """Compute subgradient of max_k (a*a)_k at point a."""
    n = len(a)
    k_star = int(np.argmax(ac))
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n, k_star + 1)
    if j_start < j_end:
        idx = np.arange(j_start, j_end)
        grad[idx] = 2.0 * a[k_star - idx]
    return grad

def multi_subgradient(ac, a, top_k=5, beta=5.0):
    """Compute weighted subgradient using top peaks."""
    n = len(a)
    # Get top k peaks
    peaks = np.argsort(ac)[::-1][:top_k]
    vals = ac[peaks]
    # Softmax weights
    w = np.exp(beta * (vals - vals[0]))
    w = w / w.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(peaks):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            idx = np.arange(j_start, j_end)
            grad[idx] += w[i] * 2.0 * a[k - idx]
    return grad

def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update on the simplex."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def nesterov_mirror(a0, time_limit, beta=5.0, top_k=5, verbose=True):
    """Nesterov-accelerated mirror descent with multi-peak subgradient."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_val = score(y)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = y.copy()
    vel = np.zeros(n)
    momentum = 0.5
    eta0 = 0.1
    stall = 0
    max_stall = 2000
    iter_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        iter_count += 1
        # Nesterov lookahead
        y_look = x + momentum * vel
        y_look = np.maximum(y_look, 1e-20)
        s_y = y_look.sum()
        if s_y < 1e-15:
            y_look = x.copy()
        else:
            y_look = y_look / s_y
        
        # Subgradient
        ac = fast_conv(y_look)
        grad = multi_subgradient(ac, y_look, top_k=top_k, beta=beta)
        
        # Step size
        eta = eta0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent update
        a_new = mirror_descent_step(x, grad, eta)
        
        # Velocity update
        vel = momentum * vel + (a_new - x)
        x = a_new
        y = x
        
        M_new = fast_conv(x).max()
        val_new = 2.0 * n * M_new
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive beta increase
        if stall > 500 and stall % 500 == 0:
            beta *= 1.5
            momentum = min(0.8, momentum * 1.05)
            eta0 *= 1.3
        
        # Random restart if very stuck
        if stall > 5000:
            noise = rng.normal(0, 0.05, n) * best_a
            x = normalize(best_a + noise)
            vel = np.zeros(n)
            stall = 0
            beta = 5.0
            eta0 = 0.1
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("  Nesterov t={:d} ({:.1f}s): J={:.6f}".format(t, elapsed, val_new))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("  Nesterov done ({:.1f}s): J={:.6f}".format(elapsed, best_val))
    return best_a, best_val

def lp_fw_full(a0, time_limit, solver, verbose=True):
    """Frank-Wolfe with full constraint set (all 2n-1 constraints)."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    ac = fast_conv(y)
    best_m = ac.max()
    t0 = time.monotonic()
    
    # Build full constraint matrix
    # (g*y)_k = sum_i g_i * y_{k-i} for k = 0, ..., 2n-2
    # For each k, the constraint is: sum_i g_i * y_{k-i} <= M
    # where y_j = 0 if j < 0 or j >= n
    
    # Build the matrix C where C[k, i] = y_{k-i} (with y_j = 0 for invalid j)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            C[k, i] = y[k - i]
    
    # LP: min M s.t. sum(g) = 1, g >= 0, C @ g <= M
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, C @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    iter_count = 0
    no_improve = 0
    max_no_improve = 200
    
    if verbose:
        print("  LP-FW full: n={}".format(n))
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        ac = fast_conv(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Solve LP (rebuild constraint matrix since y changed)
        C = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                C[k, idx] = y[k - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, C @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            break
        g_dir = g_val / g_sum
        
        # Line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_M = M_curr
        
        # Try several step sizes
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, best_alpha, 0.05]:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = best_a.copy()
        else:
            y = y / s_y
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter={:d} ({:.1f}s): obj={:.6f}".format(iter_count, elapsed, 2*n*best_m))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done ({:.1f}s): obj={:.6f}, iters={}".format(elapsed, 2*n*best_m, iter_count))
    return best_a, best_m

def lp_fw_active(a0, time_limit, solver, initial_K=1000, max_K=5000, verbose=True):
    """Frank-Wolfe with active set (for larger n)."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    ac = fast_conv(y)
    best_m = ac.max()
    t0 = time.monotonic()
    
    active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 300
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            # Perturb and continue
            noise = np.random.default_rng(iter_count).normal(0, 0.02, n) * y
            y = normalize(y + noise)
            ac = fast_conv(y)
            active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
            no_improve = 0
            continue
        
        iter_count += 1
        ac = fast_conv(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        K = len(active_set)
        if K == 0:
            break
        
        # Build constraint matrix for active set
        C = np.zeros((K, n))
        for jj in range(K):
            k_int = int(active_set[jj])
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                C[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, C @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(100, K//3)]
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Check for new violations
        # Compute g*y convolution
        fft_size = 1 << (2*n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2*n-1]
        
        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:100]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]
        
        # Line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_M = M_curr
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, best_alpha, 0.05]:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = best_a.copy()
        else:
            y = y / s_y
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW-A iter={:d} ({:.1f}s): obj={:.6f} |S|={}".format(iter_count, elapsed, 2*n*best_m, len(active_set)))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW-A done ({:.1f}s): obj={:.6f}".format(elapsed, 2*n*best_m))
    return best_a, best_m

def spectral_poocs(a0, time_limit, n_iters=1000, verbose=True):
    """POCS-style spectral diversification."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = score(a)
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Find peak of autocorrelation (equivalent to peak of |A|^2 in certain sense)
        ac = fast_conv(a)
        peak_val = ac.max()
        
        # Cap the magnitude: reduce peaks
        # The autocorrelation peak corresponds to the "energy concentration" in frequency
        # Simple approach: cap |A| at sqrt(peak_val * fft_size / n) or similar
        # Actually, let's use a different approach: 
        # We want to make the autocorrelation flatter. In frequency domain, this means
        # making |A|^2 flatter. But |A(0)| = sum(a) = 1 is fixed.
        
        # Alternative: use iterative ThresholDing
        threshold = np.sqrt(peak_val * fft_size)
        mag_capped = np.minimum(mag, threshold)
        # Keep DC component
        mag_capped[0] = mag[0]
        
        a_new = np.fft.irfft(mag_capped * np.exp(1j * phase), fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = best_a.copy()
            continue
        a_new = a_new / s
        
        a = a_new
        val = score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        
        if verbose and (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print("  POCS iter={:d} ({:.1f}s): J={:.6f}".format(it+1, elapsed, val))
    
    if verbose:
        elapsed = time.monotonic() - t0
        print("  POCS done ({:.1f}s): J={:.6f}".format(elapsed, best_val))
    return best_a, best_val

def make_initial(n, family, **kwargs):
    """Create initial sequences from different parametric families."""
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    
    if family == 'uniform':
        a = np.ones(n)
    elif family == 'raised_cosine':
        # High at edges, low in middle
        p = kwargs.get('p', 1.0)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x / (n - 1))) ** p
        a = np.maximum(a, 0.01)
    elif family == 'edge_conc':
        # Strongly edge-concentrated
        q = kwargs.get('q', 2.0)
        a = (np.abs(x - mid) / mid + 0.1) ** q
    elif family == 'u_shape':
        alpha = kwargs.get('alpha', 1.0)
        a = 1.0 / (0.1 + alpha * (x / (n-1) - 0.5)**2)
    elif family == 'gaussian_dip':
        # Gaussian peak at edges (two bumps)
        sigma = kwargs.get('sigma', 0.15)
        a = np.exp(-((x/n - 0.15)**2) / (2*sigma**2)) + np.exp(-((x/n - 0.85)**2) / (2*sigma**2))
        a = a + 0.05
    elif family == 'parent':
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            p = np.clip(p, 0.0, 1000.0)
            if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
                a = p
            else:
                a = np.ones(n)
        except Exception:
            a = np.ones(n)
    else:
        a = np.ones(n)
    
    return normalize(np.asarray(a, dtype=np.float64))

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = 2.0 * len(a_norm) * fast_conv(a_norm).max()
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [{}]: {:.6f} (n={})".format(label, sc, len(a)))
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print("Total budget: {:.1f}s".format(time_budget))
    
    solver = get_solver()
    
    def remaining_time():
        return time_budget - (time.monotonic() - t_global)
    
    # === Phase 0: Multi-start initialization ===
    print("\n=== Phase 0: Initialization ===")
    
    best_seq = None
    best_score = float('inf')
    
    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = score(p_norm)
            print("  Parent: n={}, score={:.6f}".format(len(p), sc))
            if sc < best_score:
                best_score = sc
                best_seq = p_norm.copy()
    except Exception:
        pass
    
    # Try different n values and families
    init_ns = []
    if best_seq is not None:
        n0 = len(best_seq)
        init_ns = [n0, n0 - 500, n0 + 500, n0 - 1000, n0 + 1000]
    else:
        init_ns = [3000, 4000, 5000, 6000]
    
    for n_try in init_ns:
        if n_try < 1000 or n_try > 12000:
            continue
        for family in ['raised_cosine', 'edge_conc', 'u_shape', 'gaussian_dip']:
            a_init = make_initial(n_try, family)
            # Quick subgradient to improve
            a_improved, _ = nesterov_mirror(a_init, min(remaining_time() * 0.02, 20.0), verbose=False)
            val = score(a_improved)
            if val < best_score:
                best_score = val
                best_seq = a_improved.copy()
                print("  Init [{}, n={}]: {:.6f}".format(family, n_try, val))
    
    if best_seq is None:
        n_fb = 5000
        best_seq = make_initial(n_fb, 'raised_cosine')
        best_score = score(best_seq)
        print("  Fallback init: n={}, score={:.6f}".format(n_fb, best_score))
    
    n_best = len(best_seq)
    print("  Best after init: n={}, score={:.6f}".format(n_best, best_score))
    
    # === Phase 1: Nesterov mirror descent ===
    print("\n=== Phase 1: Nesterov Mirror Descent ===")
    t1 = min(remaining_time() * 0.15, 250.0)
    if t1 > 30:
        a_out, _ = nesterov_mirror(normalize(best_seq), t1, beta=5.0, top_k=5, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nesterov", t_global)
    
    # === Phase 2: LP-FW ===
    print("\n=== Phase 2: LP-FW ===")
    t2 = min(remaining_time() * 0.35, 500.0)
    if t2 > 60:
        if len(best_seq) <= 5000:
            a_out, _ = lp_fw_full(normalize(best_seq), t2, solver, verbose=True)
        else:
            a_out, _ = lp_fw_active(normalize(best_seq), t2, solver, initial_K=2000, max_K=5000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, "lpfw", t_global)
    
    # === Phase 3: Nesterov again (different params) ===
    print("\n=== Phase 3: Nesterov (beta=10, top_k=10) ===")
    t3 = min(remaining_time() * 0.12, 200.0)
    if t3 > 30:
        a_out, _ = nesterov_mirror(normalize(best_seq), t3, beta=10.0, top_k=10, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, "nesterov2", t_global)
    
    # === Phase 4: LP-FW second pass ===
    print("\n=== Phase 4: LP-FW second pass ===")
    t4 = min(remaining_time() * 0.20, 300.0)
    if t4 > 40:
        if len(best_seq) <= 5000:
            a_out, _ = lp_fw_full(normalize(best_seq), t4, solver, verbose=True)
        else:
            a_out, _ = lp_fw_active(normalize(best_seq), t4, solver, initial_K=3000, max_K=6000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, "lpfw2", t_global)
    
    # === Phase 5: Multi-n exploration ===
    print("\n=== Phase 5: Multi-n exploration ===")
    n_best = len(best_seq)
    delta = max(200, n_best // 10)
    try_ns = sorted(set([n_best - 2*delta, n_best - delta, n_best + delta, n_best + 2*delta]))
    try_ns = [l for l in try_ns if 2000 <= l <= 12000]
    
    for new_n in try_ns:
        rt = remaining_time()
        if rt < 100:
            break
        print("  Multi-n: n={} (rt={:.1f}s)".format(new_n, rt))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start initialization with parameterized families (edge-concentrated, raised cosine)
2. Nesterov-accelerated mirror descent with multi-peak subgradient
3. Frank-Wolfe with adaptive active-set LP (sparse constraint matrices)
4. Multi-n exploration with interpolation
5. Iterated local search with randomization

import numpy as np
import time
import cvxpy as cp
from scipy import sparse

_solver_cached = None


def get_solver():
    global _solver_cached
    if _solver_cached is not None:
        return _solver_cached
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(30, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cached = s
                print("  Solver: %s" % s)
                return s
        except Exception:
            continue
    _solver_cached = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _solver_cached


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_value(a):
    a = normalize(a)
    return 2.0 * len(a) * fast_conv(a).max()


def build_sparse_constraint(y, constraint_list, n):
    K = len(constraint_list)
    rows = []
    cols = []
    vals = []
    for jj in range(K):
        k_int = int(constraint_list[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        for i in range(i_start, i_end):
            v = y[k_int - i]
            if v > 1e-15:
                rows.append(jj)
                cols.append(i)
                vals.append(v)
    if len(vals) == 0:
        return sparse.csr_matrix((K, n))
    C = sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    return C


def nesterov_mirror(a0, time_limit, beta=5.0, top_k=5, seed=42, verbose=True):
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_val = obj_value(y)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = y.copy()
    vel = np.zeros(n)
    momentum = 0.5
    eta0 = 0.08
    stall = 0
    max_stall = 3000
    iter_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break

        iter_count += 1
        y_look = x + momentum * vel
        y_look = np.maximum(y_look, 1e-20)
        s_y = y_look.sum()
        if s_y < 1e-15:
            y_look = x.copy()
        else:
            y_look = y_look / s_y

        ac = fast_conv(y_look)
        peaks = np.argsort(ac)[::-1][:top_k]
        pvals = ac[peaks]
        w = np.exp(beta * (pvals - pvals[0]))
        w = w / w.sum()

        grad = np.zeros(n)
        for ii in range(len(peaks)):
            k = int(peaks[ii])
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            if j_start < j_end:
                idx = np.arange(j_start, j_end)
                grad[idx] += w[ii] * 2.0 * y_look[k - idx]

        eta = eta0 / (np.sqrt(t) + 1.0)
        log_a = np.log(np.maximum(x, 1e-30)) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = x.copy()
        else:
            a_new = a_new / s_new

        vel = momentum * vel + (a_new - x)
        x = a_new
        y = x

        M_new = fast_conv(x).max()
        val_new = 2.0 * n * M_new
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500 and stall % 500 == 0:
            beta = min(beta * 1.4, 50.0)
            momentum = min(0.75, momentum * 1.03)
            eta0 *= 1.2

        if stall > 5000:
            noise = rng.normal(0, 0.03, n) * best_a
            x = normalize(best_a + noise)
            vel = np.zeros(n)
            stall = 0
            beta = 5.0
            eta0 = 0.08
            momentum = 0.5

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print("  Nesterov t=%d (%.1fs): J=%.6f" % (t, elapsed, val_new))

    if verbose:
        elapsed = time.monotonic() - t0
        print("  Nesterov done (%.1fs): J=%.6f" % (elapsed, best_val))
    return best_a, best_val


def lp_fw(a0, time_limit, solver, initial_K=1500, max_K=4000, add_per_iter=60, verbose=True):
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    ac = fast_conv(y)
    best_m = ac.max()
    t0 = time.monotonic()

    active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
    no_improve = 0
    max_no_improve = 400
    iter_count = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            if stagnation > 4:
                if verbose:
                    print("    LP-FW: persistent stagnation, stopping")
                break
            rng = np.random.default_rng(stagnation * 137)
            noise = rng.normal(0, 0.015, n) * y
            y = normalize(y + noise)
            ac = fast_conv(y)
            active_set = sorted(np.argsort(ac)[::-1][:initial_K].tolist())
            no_improve = 0
            stagnation += 1
            continue

        iter_count += 1
        ac = fast_conv(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
            stagnation = 0
        else:
            no_improve += 1

        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep_n = int(max_K * 0.7)
            keep_idx = np.argsort(margins)[:keep_n]
            active_set_base = [active_set[i] for i in keep_idx]
            violations = np.where(ac > M_curr - 1e-10)[0]
            for v in violations[:80]:
                if int(v) not in active_set_base:
                    active_set_base.append(int(v))
            active_set = sorted(set(active_set_base))

        K = len(active_set)
        if K == 0:
            break
        if K > max_K:
            active_set = active_set[:max_K]
            K = max_K

        C = build_sparse_constraint(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, C @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            active_set = active_set[:max(200, K // 4)]
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        fft_size = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        all_viol = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(all_viol) > 0:
            viol_amt = conv_ys[all_viol] - M_curr
            top_v = all_viol[np.argsort(viol_amt)[::-1][:add_per_iter]]
            new_set = set(active_set)
            for k in top_v:
                new_set.add(int(k))
            active_set = sorted(new_set)[:max_K]

        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_M = M_curr
        alphas = [1.0, 0.8, 0.6, 0.4, 0.25, best_alpha, 0.1, 0.05]
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_conv(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y = (1.0 - best_alpha) * y + best_alpha * g_dir
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = best_a.copy()
        else:
            y = y / s_y

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=%d (%.1fs): obj=%.6f |S|=%d" % (iter_count, elapsed, 2 * n * best_m, len(active_set)))

    if verbose:
        elapsed = time.monotonic() - t0
        print("    LP-FW done (%.1fs): obj=%.6f iters=%d" % (elapsed, 2 * n * best_m, iter_count))
    return best_a, best_m


def coordinate_polish(a0, time_limit, seed=77, verbose=True):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500
    step_scales = [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < n // 4:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = int(candidates[np.argmin(cand_scores)])

            if dst == src:
                continue

            found = False
            for frac in step_scales:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.008)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_conv(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord (%.1fs): J=%.6f stall=%d" % (elapsed, 2 * n * best_m, stall))

    return best_a, best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initial(n, family, **kwargs):
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    if family == 'uniform':
        a = np.ones(n)
    elif family == 'raised_cosine':
        p = kwargs.get('p', 1.0)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x / max(n - 1, 1))) ** p
        a = np.maximum(a, 0.01)
    elif family == 'edge_conc':
        q = kwargs.get('q', 2.0)
        a = (np.abs(x - mid) / max(mid, 1.0) + 0.05) ** q
    elif family == 'u_shape':
        alpha = kwargs.get('alpha', 2.0)
        a = 1.0 / (0.01 + alpha * (x / max(n - 1, 1) - 0.5) ** 2)
    elif family == 'gaussian_dip':
        sigma = kwargs.get('sigma', 0.12)
        a = np.exp(-((x / max(n - 1, 1) - 0.1) ** 2) / (2 * sigma ** 2))
        a += np.exp(-((x / max(n - 1, 1) - 0.9) ** 2) / (2 * sigma ** 2))
        a += 0.03
    elif family == 'power_law_edge':
        gamma = kwargs.get('gamma', 1.5)
        dist = np.minimum(x, n - 1 - x) + 1.0
        a = dist ** (-gamma)
    else:
        a = np.ones(n)

    return normalize(a)


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = obj_value(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [%s]: %.6f (n=%d, t=%.1fs)" % (label, sc, len(a), elapsed))
    return best_seq, best_score

        p = np.clip(p, 0.0, 1000.0)

            p_norm = p / p.sum()
            sc = obj_value(p_norm)
            print("  Parent: n=%d, score=%.6f" % (len(p), sc))

        init_ns = [3000, 4000, 5000, 6000, 7000]
