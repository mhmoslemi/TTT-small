def fast_autocorr(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    """Clamp to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    """Project onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def evaluate_j(a):
    """Compute J = 2*n*max(conv)/(sum)^2 directly."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_autocorr(a)
    return 2.0 * n * conv.max() / (s * s)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_flat_pass(a, n_passes=30, alpha=0.03):
    """Spectral domain: reduce peaks in |A(f)|^2 to flatten convolution."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    a = norm_sum1(a)
    for p in range(n_passes):
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        if len(power) <= 1:
            break
        target = power[1:].mean()
        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * (ratio - 1.0)
                A_new[k] *= max(0.2, np.sqrt(red))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        m_old = fast_autocorr(a).max()
        m_new = fast_autocorr(a_new).max()
        if m_new < m_old:
            a = a_new
    return a


def logsumexp_grad(a, beta, top_k=None):
    """
    Compute gradient of (1/beta) * logsumexp(beta * conv(a,a)).
    Uses FFT-based convolution for the gradient.
    Returns (gradient, conv_values, max_conv).
    """
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    max_cv = conv.max()

    if top_k is not None and top_k < size:
        idx = np.argsort(conv)[::-1][:top_k]
        conv_use = np.zeros(size)
        conv_use[idx] = conv[idx]
    else:
        conv_use = conv

    # Stable log-sum-exp
    shifted = beta * (conv_use - max_cv)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s_exp = exp_vals.sum()
    if s_exp < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s_exp

    # Gradient: (2/beta) * convolve(p_scaled, flip(a)) sliced to length n
    # conv(a,a)[k] = sum_i a[i]*a[k-i]
    # d/d a[j]: for each k, contribution is a[k-j] (if valid)
    # So grad[j] = (1/beta) * sum_k p[k] * a[k-j] * (2 or 1 for boundary)
    # Approximate as: (2/beta) * (flip(a) * p)[j] where * is cross-correlation
    # flip(a)[i] = a[n-1-i]
    # The "correlation" corr[j] = sum_k flip(a)[j+k] * p[k] = sum_k a[n-1-j-k]*p[k]
    # At the center: this gives sum_k a[k-j']*p[something]
    # Better: grad[j] = (2/beta) * sum_{k} p[k] * a[k-j] (for 0<=k-j<n)
    # This is (2/beta) * (a * p_flipped)[n-1+j] in convolution terms

    # Use FFT for efficiency
    conv_len = n + size - 1
    fft_sz = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a[::-1], fft_sz)  # flip(a)
    Pr = np.fft.rfft(p, fft_sz)
    conv_grad = np.fft.irfft(Ar * Pr, fft_sz)
    # conv(a[::-1], p)[n-1+j] for j=0,...,n-1
    grad = (2.0 / beta) * conv_grad[n - 1:2 * n - 1]

    # Boundary correction: at indices where k-j is out of range, factor is 1 not 2
    # For simplicity, keep the factor of 2 (slight overestimate at boundaries)
    # This is acceptable for optimization purposes.

    return grad, conv, max_cv

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        y = y / s_y

        # Compute smoothed gradient
        grad, conv, max_cv = logsumexp_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-18:
            x_new = x
            max_new = fast_autocorr(x_new).max()
        else:
            # Gradient step with backtracking
            eta_trial = eta
            x_new = None
            y_max = max_cv
            for bt in range(12):
                z = y - eta_trial * grad
                x_cand = simplex_project(z)
                conv_c = fast_autocorr(x_cand)
                max_c = conv_c.max()
                if max_c <= y_max + 1e-14:
                    x_new = x_cand
                    max_new = max_c
                    break
                eta_trial *= 0.5
            if x_new is None:
                x_new = x
                max_new = fast_autocorr(x).max()
                eta = eta_trial
            else:
                eta = min(eta * 1.05, 1.5)

        # Track best
        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Grow beta periodically
        if t % 80 == 0:
            beta = min(beta * beta_growth, beta_end)

        # Restart strategy
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = int(rng.integers(6))
            if strategy == 0:
                noise = (0.01 + 0.02 * rng.random())
                pert = rng.normal(0, noise, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                w = 0.6 + 0.4 * rng.random()
                x_r = w * best_a + (1 - w) * rng.dirichlet(np.ones(n) * 0.5)
            elif strategy == 2:
                x_r = best_a * np.exp(0.04 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= (0.5 + 0.5 * rng.random())
                x_r[~mask] *= (1.0 + 0.5 * rng.random())
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 20)))
                x_r = np.roll(best_a, shift)
            else:
                # Spectral perturbation
                fft_sz = 1 << (2 * n - 2).bit_length()
                A = np.fft.rfft(best_a, fft_sz)
                for k in range(1, len(A)):
                    A[k] *= np.exp(1j * rng.standard_normal() * 0.03)
                x_r = np.fft.irfft(A, fft_sz)[:n]

            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nesterov restart t={} ({}s): J={}".format(t, round(elapsed, 1), round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nesterov t={} ({}s): J={} beta={}".format(t, round(elapsed, 1), round(J, 6), int(beta)))

    return best_a, best_max_b


def lp_fw_kkt(a0, time_limit, solver, initial_K=500, max_K=3000,
              add_per_iter=30, verbose=True):
    """
    LP-based Frank-Wolfe with KKT-informed active set management.
    At each step, solves LP to find descent direction, then does line search.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_list = sorted(int(k) for k in np.argsort(ac_y)[::-1][:min(initial_K, 2 * n - 1)])

    no_improve = 0
    max_no_improve = 800
    fp_iter = 0  # function-point iteration count

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set: keep most violated constraints
        if len(active_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_list])
            keep_idx = np.argsort(margins)[:max_K]
            active_list = sorted(int(active_list[i]) for i in keep_idx)

        K = len(active_list)
        if K == 0:
            break

        # Build LP constraint matrix: (y * g)[k] <= M_curr for k in active_list
        # (y*g)[k] = sum_i g[i] * y[k-i] (valid indices only)
        # This is a linear function of g.
        # coeffs[row, col] = y[k - col] if 0 <= k-col < n else 0
        # For memory efficiency with large n, use sparse representation
        # But for cvxpy, we build the matrix directly if K*n is manageable

        if K * n > 50_000_000:
            # Too large, reduce active set
            active_list = sorted(int(k) for k in np.argsort(ac)[::-1][:initial_K])
            K = len(active_list)

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(active_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_list) > initial_K // 2:
                active_list = sorted(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Detect violated constraints and add to active set
        fft_sz = 1 << (2 * n - 2).bit_length()
        Yf = np.fft.rfft(y, fft_sz)
        Gf = np.fft.rfft(g_dir, fft_sz)
        conv_ys = np.fft.irfft(Yf * Gf, fft_sz)[:2 * n - 1]

        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                if int(k) not in set(active_list):
                    active_list.append(int(k))

        # Line search (use convexity: f(t) = max_conv((1-t)*y + t*g_dir) is convex in t)
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, 0.1, 0.2, 0.3, 0.5, 0.05, 0.8, 1.0]
        for alpha in alphas:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new
        ac_y = fast_autocorr(y)
        fp_iter += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it={} ({}s): J={} |S|={}".format(t, round(elapsed, 1), round(J, 6), len(active_list)))

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=True, seed=789):
    """
    Coordinate-wise descent: pick the element whose modification most reduces
    the peak convolution, and adjust it.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        conv = fast_autocorr(a)
        size = 2 * n - 1
        peak_lag = int(np.argmax(conv))
        peak_val = conv[peak_lag]
        improved = False

        # Find the element that contributes most to the peak
        i_start = max(0, peak_lag - n + 1)
        i_end = min(n, peak_lag + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_lag - idx]
        j_max = int(idx[np.argmax(contrib)])

        # Try reducing a[j_max] and redistributing
        for frac in [0.005, 0.01, 0.02, 0.05]:
            amt = a[j_max] * frac
            if amt < 1e-14:
                continue
            # Try moving mass to several candidate positions
            candidates = [int(rng.integers(n)) for _ in range(20)]
            # Also try the mirror position
            mirror = n - 1 - j_max
            candidates.append(mirror)
            candidates.append((j_max + n // 2) % n)

            found_better = False
            for dst in candidates:
                if dst == j_max:
                    continue
                a_trial = a.copy()
                a_trial[j_max] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-17:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found_better = True
                    break
            if found_better:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Random exploration
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall > 0 and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordPolish ({}s): J={} stall={}".format(round(elapsed, 1), round(J, 6), stall))

    return best_a, best_m


def kkt_residual(a):
    """
    Compute KKT residual: at optimum, for a>0 everywhere,
    the weighted shifted sum of a should be constant.
    Returns a measure of non-optimality.
    """
    n = len(a)
    conv = fast_autocorr(a)
    M = conv.max()
    # Softmax weights on constraints
    beta_kkt = 1.0 / (M * 0.1 + 1e-10)
    shifted = beta_kkt * (conv - M)
    shifted = np.clip(shifted, -300.0, 0.0)
    exp_v = np.exp(shifted)
    p = exp_v / exp_v.sum()

    # Compute "weighted shifted sum": w[j] = sum_k p[k] * a[k-j]
    # This is a correlation
    conv_len = n + len(p) - 1
    fft_sz = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a[::-1], fft_sz)
    Pr = np.fft.rfft(p, fft_sz)
    w = np.fft.irfft(Ar * Pr, fft_sz)[n - 1:2 * n - 1]

    # Residual: how non-constant is w (at positions where a > 0)?
    mask = a > 1e-8
    if mask.sum() < 2:
        return 0.0
    w_masked = w[mask]
    return (w_masked.std() / (w_masked.mean() + 1e-15)) if w_masked.mean() > 1e-15 else 0.0


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(label, sc, len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy convex optimization for minimizing 2*n*max(conv)/(sum)^2.
    Uses Nesterov acceleration, LP-Frank-Wolfe, spectral methods, and
    multi-length exploration within the time budget.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print("Time budget: {:.1f}s".format(time_budget))

    best_seq = None
    best_score = float('inf')

    # --- Load parent ---
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={}, score={:.6f}".format(len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        # Initialize with a raised-cosine (matches the "flat convolution" goal)
        best_seq = norm_sum1(0.5 + 0.5 * np.cos(2 * np.pi * x))
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={:.6f}".format(n_base, best_score))

    # --- Phase 0: Spectral pre-flattening ---
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P0: Spectral init (n={}) ===".format(len(best_seq)))
    if remaining > 10:
        t0_alloc = min(remaining * 0.02, 30.0)
        a_spec = spectral_flat_pass(norm_sum1(best_seq), n_passes=40, alpha=0.04)
        best_seq, best_score = try_update(best_seq, best_score, a_spec, "spec0", t_global)

    # --- Phase 1: Primary Nesterov ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P1: Nesterov main (n={}) ===".format(n_cur))
    if remaining > 50:
        t1 = min(remaining * 0.20, 320.0)
        a_out, _ = nesterov_accelerated(norm_sum1(best_seq), t1,
                                        beta_start=15.0, beta_end=3000.0,
                                        beta_growth=1.004, eta_init=0.4,
                                        top_k_ratio=0.02, restart_every=2000,
                                        verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # --- Phase 2: LP-Frank-Wolfe ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P2: LP-FW (n={}) ===".format(n_cur))
    if remaining > 40 and n_cur <= 12000:
        t2 = min(remaining * 0.15, 250.0)
        a_out, _ = lp_fw_kkt(norm_sum1(best_seq), t2, solver,
                             initial_K=400, max_K=2500,
                             add_per_iter=30, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # --- Phase 3: Nesterov round 2 (different params) ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P3: Nesterov r2 (n={}) ===".format(n_cur))
    if remaining > 40:
        t3 = min(remaining * 0.15, 250.0)
        a_out, _ = nesterov_accelerated(norm_sum1(best_seq), t3,
                                        beta_start=50.0, beta_end=5000.0,
                                        beta_growth=1.005, eta_init=0.2,
                                        top_k_ratio=0.01, restart_every=1500,
                                        verbose=True, seed=999)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # --- Phase 4: Multi-length exploration ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P4: Multi-length (base n={}) ===".format(n_cur))

    # Try a few longer sequences
    longer_lengths = [n_cur + 500, n_cur + 1200, n_cur + 2500]
    # Also try a significantly longer one
    longer_lengths.append(min(n_cur + 5000, 20000))
    longer_lengths = sorted(set([l for l in longer_lengths if 200 <= l <= 25000]))

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print("  -> Trying n={}".format(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick spectral pass
        a_long = spectral_flat_pass(a_long, n_passes=20, alpha=0.03)

        # Nesterov on longer sequence
        t4a = min(remaining * 0.40, 250.0)
        if t4a > 30:
            a_out, _ = nesterov_accelerated(a_long, t4a,
                                            beta_start=20.0, beta_end=2000.0,
                                            beta_growth=1.004, eta_init=0.3,
                                            top_k_ratio=0.015, restart_every=1500,
                                            verbose=True, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n{}".format(new_n), t_global)

        # LP-FW if time allows
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 10000:
            t4b = min(remaining * 0.45, 180.0)
            if t4b > 25:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = lp_fw_kkt(a_long2, t4b, solver,
                                     initial_K=300, max_K=2000,
                                     add_per_iter=25, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n{}".format(new_n), t_global)

    # --- Phase 5: Coordinate polish on best ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P5: Coord polish (n={}) ===".format(n_cur))
    if remaining > 20:
        t5 = min(remaining * 0.10, 120.0)
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t5, verbose=True, seed=789)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord1", t_global)

    # --- Phase 6: Final high-beta Nesterov ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P6: Final Nesterov (n={}) ===".format(n_cur))
    if remaining > 30:
        t6 = min(remaining * 0.35, 300.0)
        a_out, _ = nesterov_accelerated(norm_sum1(best_seq), t6,
                                        beta_start=100.0, beta_end=8000.0,
                                        beta_growth=1.006, eta_init=0.15,
                                        top_k_ratio=0.005, restart_every=1000,
                                        verbose=True, seed=555)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # --- Phase 7: Final LP-FW polish ---
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print("\n=== P7: Final LP-FW (n={}) ===".format(n_cur))
    if remaining > 30 and n_cur <= 10000:
        t7 = min(remaining * 0.50, 250.0)
        a_out, _ = lp_fw_kkt(norm_sum1(best_seq), t7, solver,
                             initial_K=500, max_K=4000,
                             add_per_iter=40, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # --- Phase 8: One more coordinate pass ---
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.5, 100.0)
        a_out, _ = coordinate_polish(norm_sum1(best_seq), t8, verbose=False, seed=321)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # --- Final validation ---
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J={:.6f}, n={}, time={:.1f}s".format(best_score, len(best_seq), elapsed))
        print("KKT residual: {:.6e}".format(kkt_residual(best_seq)))
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(0.5 + 0.5 * np.cos(2 * np.pi * x))
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
