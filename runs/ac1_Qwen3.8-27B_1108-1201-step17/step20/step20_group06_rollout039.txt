def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()

    if b_max <= 0:
        return np.zeros(n), b

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size)
    Pr = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(Ar * Pr, fft_size)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, b


def spectral_waterfill(a0, time_limit, verbose=True, seed=777):
    """Water-filling spectral flattening with adaptive threshold."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    alpha = 0.03
    clip_ratio = 1.3

    for it in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        if len(power) > 1:
            target = np.mean(power[1:])
        else:
            target = power[0]

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > clip_ratio:
                red = 1.0 - alpha * (ratio - 1.0)
                A_new[k] *= max(0.2, np.sqrt(red))
            elif ratio < 0.4:
                inc = 1.0 + alpha * 0.5 * (0.4 - ratio)
                A_new[k] *= min(2.5, np.sqrt(inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 300 == 0:
            # Random phase perturbation for diversity
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.03
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 800 == 0:
            alpha = max(0.003, alpha * 0.92)
            clip_ratio = max(1.1, clip_ratio * 0.98)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SpectralWf it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def nesterov_adaptive(a0, time_limit, beta_start=20.0, beta_end=800.0,
                      beta_growth=1.003, eta_init=0.5, top_k_ratio=0.025,
                      restart_every=1200, verbose=True, seed=42):
    """Nesterov with backtracking and diverse restarts."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = compute_smoothed_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x
            b_new = fast_autocorr(x_new)
            max_new = b_new.max()
        else:
            eta_trial = eta
            found = False
            y_max = fast_autocorr(y).max()
            for _ in range(8):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                b_new = fast_autocorr(x_new)
                max_new = b_new.max()
                if max_new <= y_max + 1e-15:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                b_new = fast_autocorr(x)
                max_new = b_new.max()
                eta = eta_trial

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.08, 2.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)
            top_k = max(10, int(n * max(0.005, top_k_ratio * (0.95 ** (t / 1000)))))

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                noise_level = 0.01 + 0.02 * rng.random()
                pert = rng.normal(0, noise_level, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.2)
            elif strategy == 2:
                x_r = best_a * np.exp(0.04 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= 0.6
                x_r[~mask] *= 1.4
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 15)))
                x_r = np.roll(best_a, shift)
            else:
                # Local perturbation at peak locations
                b = fast_autocorr(best_a)
                peak_idx = np.argsort(b)[::-1][:5]
                x_r = best_a.copy()
                for pk in peak_idx:
                    pk_int = int(pk)
                    js = max(0, pk_int - n + 1)
                    je = min(n, pk_int + 1)
                    for j in range(js, je):
                        if 0 <= j < n:
                            partner = pk_int - j
                            if 0 <= partner < n and j != partner:
                                amt = 0.01 * best_a[j]
                                x_r[j] -= amt
                                x_r[partner] += amt
            x_r = np.maximum(x_r, 1e-10)
            xr_sum = x_r.sum()
            if xr_sum > 1e-15:
                x_r = x_r / xr_sum
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 15.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    NestAdapt restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    NestAdapt t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " b=" + str(int(beta)))

    return best_a, best_max_b


def lp_fw_improved(a0, time_limit, solver, initial_K=400, max_K=2500,
                   add_per_iter=25, verbose=True):
    """LP-Frank-Wolfe with improved line search and active set management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])

    no_improve = 0
    max_no_improve = 800

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-16:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Multi-step line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        alphas = [best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                  1.0 / (t + 1), 0.03, 0.07, 0.15, 0.3]
        for alpha in alphas:
            alpha = max(0.01, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 150 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def peak_cd(a0, time_limit, verbose=True, seed=321):
    """Peak-targeting coordinate descent: directly reduce top peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 3000:
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        size = 2 * n - 1
        top_k = np.argsort(ac)[::-1][:min(15, size)]
        improved = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            src = int(idx[np.argmax(contrib)])
            partner = k_int - src
            if not (0 <= partner < n):
                continue

            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue

                for _ in range(8):
                    dst = int(rng.integers(n))
                    if dst == src or dst == partner:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-17:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 150:
                # Random exploration
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakCD (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def simulated_annealing_v3(a0, time_limit, verbose=True, seed=123):
    """SA with diverse move types including peak-aware moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    T = 0.006
    T_min = 0.00003
    T_decay = 0.9996
    cur_m = best_m

    for it in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        if T < T_min:
            break

        move_type = rng.integers(4)
        if move_type == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = rng.uniform(0.001, 0.012)
            amt = a[i] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
        elif move_type == 1:
            # Peak-aware: target the largest peak
            ac = fast_autocorr(a)
            k_peak = int(np.argmax(ac))
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            src = int(idx[np.argmax(contrib)])
            dst = int(rng.integers(n))
            if dst == src:
                continue
            amt = a[src] * rng.uniform(0.005, 0.03)
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
        elif move_type == 2:
            i, j = int(rng.integers(n)), int(rng.integers(n))
            k, l = int(rng.integers(n)), int(rng.integers(n))
            if i == j or k == l or i == k or i == l or j == k or j == l:
                continue
            amt = min(a[i], a[k]) * rng.uniform(0.01, 0.04)
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial[k] -= amt
            a_trial[l] += amt
        else:
            # Local Gaussian perturbation
            center = int(rng.integers(n))
            width = int(rng.integers(10, max(20, n // 100)))
            mask = np.zeros(n, dtype=bool)
            lo = max(0, center - width)
            hi = min(n, center + width)
            mask[lo:hi] = True
            pert = rng.normal(0, 0.008, n) * mask * a
            a_trial = a + pert

        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        m_trial = fast_autocorr(a_trial).max()

        if m_trial < cur_m:
            a = a_trial
            cur_m = m_trial
            if m_trial < best_m:
                best_m = m_trial
                best_a = a_trial.copy()
        elif rng.random() < np.exp(-(m_trial - cur_m) / T):
            a = a_trial
            cur_m = m_trial

        T *= T_decay

        if it % 8000 == 0 and T < 0.001:
            T = 0.004

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SA3 it=" + str(it) + " T=" + str(round(T, 6)) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization with water-filling spectral method,
    adaptive Nesterov, LP-FW, peak-targeting CD, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + repr(round(best_score, 6)))

    # Phase 1: Spectral water-filling (quick)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Spectral WaterFill (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        a_out, _ = spectral_waterfill(norm_sum1(best_seq), t1, verbose=True, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 2: Nesterov (primary)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: Nesterov (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.20, 320.0)
    if t2 > 40:
        a_out, _ = nesterov_adaptive(norm_sum1(best_seq), t2,
                                     beta_start=15.0, beta_end=700.0,
                                     beta_growth=1.003, eta_init=0.5,
                                     top_k_ratio=0.025, restart_every=1200,
                                     verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nest1", t_global)

    # Phase 3: LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: LP-FW (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 25 and n_best <= 10000:
        a_out, _ = lp_fw_improved(norm_sum1(best_seq), t3, solver,
                                  initial_K=400, max_K=2500,
                                  add_per_iter=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 4: Peak-targeting CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: PeakCD (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.07, 80.0)
    if t4 > 10:
        a_out, _ = peak_cd(norm_sum1(best_seq), t4, verbose=True, seed=321)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd1", t_global)

    # Phase 5: Simulated annealing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: SA (n=" + str(n_best) + ") ===")
    t5 = min(remaining * 0.07, 80.0)
    if t5 > 10:
        a_out, _ = simulated_annealing_v3(norm_sum1(best_seq), t5, verbose=True, seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "sa1", t_global)

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Multi-length (base n=" + str(n_best) + ") ===")

    # Try a few longer lengths
    longer_lengths = [n_best + 500, n_best + 1500, n_best + 3000]
    longer_lengths = [l for l in longer_lengths if l <= 15000]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print("  -> n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.35, 200.0)
        if t6a > 30:
            a_out, _ = nesterov_adaptive(a_long, t6a,
                                         beta_start=20.0, beta_end=600.0,
                                         beta_growth=1.003, eta_init=0.4,
                                         top_k_ratio=0.02, restart_every=1000,
                                         verbose=True, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 10000:
            t6b = min(remaining * 0.35, 150.0)
            if t6b > 20:
                a_long2 = interpolate_sequence(best_seq, new_n)
                a_out, _ = lp_fw_improved(norm_sum1(a_long2), t6b, solver,
                                          initial_K=350, max_K=2000,
                                          add_per_iter=20, verbose=True)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6c = min(remaining * 0.25, 60.0)
            if t6c > 10:
                a_out, _ = spectral_waterfill(norm_sum1(best_seq), t6c, verbose=False, seed=new_n % 3163)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_n" + str(new_n), t_global)

    # Phase 7: Second Nesterov with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: Nesterov r2 (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t7 = min(remaining * 0.25, 250.0)
        if t7 > 30:
            a_out, _ = nesterov_adaptive(norm_sum1(best_seq), t7,
                                         beta_start=50.0, beta_end=1200.0,
                                         beta_growth=1.004, eta_init=0.3,
                                         top_k_ratio=0.01, restart_every=800,
                                         verbose=True, seed=999)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest2", t_global)

    # Phase 8: LP-FW final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P8: LP-FW final (n=" + str(n_best) + ") ===")
    if remaining > 30 and n_best <= 10000:
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 15:
            a_out, _ = lp_fw_improved(norm_sum1(best_seq), t8, solver,
                                      initial_K=500, max_K=3000,
                                      add_per_iter=30, verbose=True)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Phase 9: Final short Nesterov + PeakCD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P9: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 20:
        t9a = min(remaining * 0.35, 80.0)
        if t9a > 10:
            a_out, _ = nesterov_adaptive(norm_sum1(best_seq), t9a,
                                         beta_start=100.0, beta_end=2000.0,
                                         beta_growth=1.005, eta_init=0.15,
                                         top_k_ratio=0.005, restart_every=600,
                                         verbose=True, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nest3", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t9b = min(remaining * 0.35, 60.0)
            if t9b > 5:
                a_out, _ = peak_cd(norm_sum1(best_seq), t9b, verbose=True, seed=654)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "pcd2", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t9c = min(remaining * 0.30, 40.0)
            if t9c > 5:
                a_out, _ = spectral_waterfill(norm_sum1(best_seq), t9c, verbose=False, seed=888)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "spec3", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
