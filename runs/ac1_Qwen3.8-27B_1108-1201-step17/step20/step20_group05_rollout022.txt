Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Strategy:

3. Frank-Wolfe with game-based directions

import numpy as np
import time
import math

def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def _norm1(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s

def _fft_conv(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = np.fft.rfft(a, N)

Let me also think about the "game direction" more carefully. The Frank-Wolfe direction solves:
min_g max_k (f*g)[k] subject to g >= 0, sum(g) = 1

where (f*g)[k] = sum_i f[i]*g[k-i] (cross-convolution).

This is a zero-sum game where:
- Player 1 chooses a distribution p over the peaks (indices k)
- Player 2 chooses a distribution g over the positions (indices i)
- The payoff is sum_k sum_i p[k]*g[i]*f[k-i] (or f[i]*g[k-i] depending on convention)

Wait, let me be more careful. The cross-convolution (f*g)[k] = sum_i f[i]*g[k-i]. We want to minimize max_k (f*g)[k].

Using von Neumann's minimax theorem, this equals:
max_p min_g sum_k p[k] * (f*g)[k] = max_p min_g sum_k p[k] * sum_i f[i]*g[k-i]
= max_p min_g sum_i g[i] * sum_k p[k]*f[k-i]
= max_p min_g sum_i g[i] * (pressure from p at position i)

So the game is:
- Player 1 (maximizing) chooses p (distribution over k)
- Player 2 (minimizing) chooses g (distribution over i)
- Payoff for player 1: sum_i g[i] * pressure_p[i] where pressure_p[i] = sum_k p[k]*f[k-i]

The optimal strategy for player 2 (g) given p is to put all mass on the position i with minimum pressure. The optimal strategy for player 1 (p) given g is to put all mass on the k with maximum (f*g)[k].

In a general equilibrium, both players use mixed strategies. The value of the game is the minimum max_k (f*g)[k] over all g in the simplex, which is the Frank-Wolfe step size (times the current M).

"""
Multi-phase optimizer for step function self-convolution minimization.

Minimizes f(a) = 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative
sequences a. Uses L-BFGS on log-sum-exp smoothed objectives, Frank-Wolfe
with game-based directions, and targeted coordinate moves.
"""

import numpy as np
import time
import math
from scipy.optimize import minimize
import scipy.sparse

def _npow2(x):
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def _norm(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(max(1, a.size), dtype=np.float64) / max(1, a.size)
    return a / s

def _conv(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _npow2(2*n - 1)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A*A, N)[:2*n-1]
    return np.where(np.isfinite(c) & (c > 0), c, 0.0)

def _conv2(a, b):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _npow2(2*n - 1)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A*B, N)[:2*n-1]
    return np.where(np.isfinite(c) & (c > 0), c, 0.0)

def _eval_safe(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            return float('inf')
        arr = np.clip(arr, 0.0, 1000.0)
        if not np.all(np.isfinite(arr)):

        return float(evaluate_sequence([float(x) for x in arr]))

def _lse_obj_grad(v, beta, top_k, ref):
    """Log-sum-exp objective in softmax coordinates."""
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref+1:] = v[ref:]
    
    # softmax
    um = u.max()
    e = np.exp(u - um)
    s = e.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = e / s
    
    # convolution
    c = _conv(a)
    size = 2*n - 1
    
    if top_k is not None and 0 < top_k < size:
        idx = np.argsort(c)[::-1][:top_k]
        vals = c[idx]
    else:
        vals = c
    
    m = vals.max()
    shifted = np.clip(beta * (vals - m), -500, 500)
    ex = np.exp(shifted)
    se = ex.sum()
    if se < 1e-300:
        return 1e100, np.zeros(v.size)
    p = ex / se  # distribution over top peaks
    
    # Build full pressure vector
    pf = np.zeros(size, dtype=np.float64)
    if top_k is not None and 0 < top_k < size:
        pf[idx] = p
    else:
        pf[:] = p
    
    # Gradient: 2 * cross_conv(pf, a)
    total = 3*n - 2
    N = _npow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    ga = 2.0 * conv[n-1:2*n-1][::-1]
    ga = np.where(np.isfinite(ga), ga, 0.0)
    
    f = m + math.log(se) / beta
    
    # Convert to v-space gradient
    mean = (a * ga).sum()
    gu = a * (ga - mean)
    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref+1:]
    
    if not np.all(np.isfinite(gv)):
        gv = np.zeros(v.size)
    
    return f, gv

def _v_to_a(v, ref):

        ref = 0

        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]

def _game_direction(f, active_idx, iters, seed):
    """Compute Frank-Wolfe direction via multiplicative weights."""
    f = _norm(f)

        return np.ones(n) / n

    logits = rng.normal(0, 0.05, K)

        p = np.zeros(size, dtype=np.float64)
        p[active_idx] = np.exp(logits - logits.max())
        p[active_idx] /= p[active_idx].sum()
        
        pressure = _cross_p(f, p)
        pmx = pressure.max()
        pmn = pressure.min()
        if pmx - pmn < 1e-30:
            g = np.ones(n) / n
        else:
            scaled = (pressure - pmn) / (pmx - pmn)
            w = np.exp(-6.0 * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-10
            g = w / w.sum()
            g = 0.95 * g + 0.05 * (np.ones(n) / n)
            g = g / g.sum()
        
        # Player 1: given g, find best response p
        h = _conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)

        hmin = ha.min()
        if hmax - hmin > 1e-30:

def _cross_p(f, p):
    """pressure[i] = sum_k p[k] * f[k-i] (valid range)."""
    n = f.size
    total = 3*n - 2
    N = _npow2(total)
    A = np.fft.rfft(f, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n-1:2*n-1][::-1]
    return np.where(np.isfinite(out), out, 0.0)

def _fw_step(a, c, M, active, seed, game_iters):
    """One Frank-Wolfe iteration."""
    g = _game_direction(a, active, game_iters, seed)
    cg = _conv(g)
    fg = _conv2(a, g)
    
    # Line search
    best_alpha = 0.0
    best_m = M
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        val = (1-alpha)**2 * c + 2*alpha*(1-alpha) * fg + alpha**2 * cg
        m = val.max()
        if m < best_m:
            best_m = m
            best_alpha = alpha
    
    if best_alpha <= 0:
        return a, M, False
    new_a = (1 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s < 1e-15:
        return a, M, False
    return new_a / s, best_m, True

def _coord_move(a, c, M, top_peak_idx, rng, ntries):
    """Try coordinate moves to reduce the peak."""
    n = a.size
    for _ in range(ntries):
        s = int(rng.integers(n))
        d = int(rng.integers(n))
        if s == d or a[s] < 1e-12:
            continue
        frac = float(rng.uniform(0.002, 0.04))
        t = a[s] * frac
        if t < 1e-16:
            continue
        # Rank-1 update
        c_new = c.copy()
        c_new[d:d+n] += 2*t*a
        c_new[s:s+n] -= 2*t*a
        if 2*d < c_new.size:
            c_new[2*d] += t*t
        if 2*s < c_new.size:
            c_new[2*s] += t*t
        if s+d < c_new.size:

            a[d] += t
