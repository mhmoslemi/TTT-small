_MODULE_SOLVER = None


def get_solver():

    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)

        except Exception:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_grad_softmax(y, beta, top_k):
    """Compute gradient of softmax-smoothed max autocorrelation."""
    n = len(y)
    b = fast_autocorr(y)
    b_max = b.max()
    if top_k is not None and top_k < len(b) and top_k > 0:
        idx = np.argsort(b)[::-1][:top_k]
        b_masked = np.zeros(len(b))
        b_masked[idx] = b[idx]
    else:
        b_masked = b
    shifted = beta * (b_masked - b_max)
    shifted = np.clip(shifted, -400, 400)
    exp_vals = np.exp(shifted)
    p_total = exp_vals.sum()
    if p_total < 1e-300:
        p = np.zeros_like(b_masked)
        n_max = (b_masked == b_max).sum()
        if n_max == 0:
            n_max = 1
        p[b_masked == b_max] = 1.0 / n_max
    else:
        p = exp_vals / p_total
    # grad[j] = (2/beta) * sum_k p[k] * y[k-j] (valid indices)
    # Use FFT cross-correlation
    fft_size = 1 << (4 * n - 1).bit_length()
    y_pad = np.zeros(fft_size)
    y_pad[:n] = y
    p_pad = np.zeros(fft_size)
    p_pad[:2*n-1] = p
    Yf = np.fft.rfft(y_pad)
    Pf = np.fft.rfft(p_pad)
    # cross-correlation: ifft(fft(p) * conj(fft(y)))
    cross = np.fft.irfft(Pf * np.conj(Yf), fft_size)
    grad = 2.0 * cross[:n] / beta
    return grad, p, b


def fista_adaptive(a0, time_limit, beta_start=20.0, beta_end=1000.0,
                   beta_growth=1.003, top_k_start=10, top_k_end=40,
                   restart_every=3000, verbose=True):
    """FISTA with proper BB step, adaptive top_k, and Nesterov averaging."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(42)
    avg_x = x.copy()
    avg_count = 0.0
    s_prev = None
    y_prev = None
    bb_step = 1.0 / np.sqrt(n)
    top_k = top_k_start

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive top_k: grow over iterations
        frac = min(1.0, t / (restart_every * 3.0))
        top_k = int(top_k_start + frac * (top_k_end - top_k_start))

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, p, b = compute_grad_softmax(y, beta, top_k)

        # Barzilai-Borwein step size
        if s_prev is not None and y_prev is not None:
            sy = y - y_prev
            gy = grad - s_prev
            sy_gy = sy.dot(gy)
            gy_gy = gy.dot(gy)
            if gy_gy > 1e-20:
                bb_candidate = sy_gy / gy_gy
                if bb_candidate > 1e-8 and bb_candidate < 10.0 / beta:
                    bb_step = 0.7 * bb_step + 0.3 * bb_candidate
                    bb_step = max(1e-6, min(bb_step, 5.0 / beta))
                else:
                    bb_step = max(bb_step * 0.85, 1e-7)
            s_prev = grad.copy()
            y_prev = y.copy()

        eta = min(bb_step, 0.3)
        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()

        # Nesterov running average
        avg_count += 1.0
        avg_x = avg_x + (x_new - avg_x) / avg_count

        if max_new < best_max_b - 1e-14:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 80 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.01 / (1.0 + 0.03 * (t // restart_every))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 8.0)
            s_prev = None
            y_prev = None
            bb_step = 1.0 / np.sqrt(n)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA restart t={:.0f} ({:.1f}s): J={:.6f}".format(t, elapsed, J))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA t={:.0f} ({:.1f}s): J={:.6f} beta={:.1f} tk={}".format(t, elapsed, J, beta, top_k))

    # Compare with Nesterov average
    J_avg = 2.0 * n * fast_autocorr(avg_x).max()
    J_best = 2.0 * n * best_max_b
    if J_avg < J_best:
        return avg_x, fast_autocorr(avg_x).max()
    return best_a, best_max_b


def spectral_spread(a0, time_limit, n_iters=300, verbose=True):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        b = np.fft.irfft(power, fft_size)[:2 * n - 1]
        b_max = b.max()
        if b_max < 1e-15:
            break
        b_norm = b / b_max
        # Reduce amplitude at frequencies where the autocorrelation has peaks
        # The peak positions in b correspond to "structure" in the spectrum
        rfftfreq_idx = np.arange(len(A))
        # Modulate: reduce amplitude where b_norm is high
        # This is heuristic - flatten the spectrum
        flatten_strength = 0.15 + 0.1 * (it / max(1, n_iters))
        A_new = A * np.sqrt(1.0 - flatten_strength * b_norm[rfftfreq_idx % len(b_norm)])
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        a = a_new
        if verbose and it % 100 == 0:
            J = 2.0 * n * best_m
            print("    Spectral it={}: J={:.6f}".format(it, J))

    return best_a, best_m


def lp_fw_optimize(a0, time_limit, solver, initial_K=300, max_K=2500, add_per_iter=20, verbose=True):
    """LP-Frank-Wolfe with warm-start and adaptive active set."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 500

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(fast_autocorr(y))[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations and add to active set
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                active_set.add(int(k))

        # Line search over step sizes
        default_alpha = 2.0 / (t + 2.0)
        alphas = [default_alpha, max(0.02, default_alpha * 0.5),
                  max(0.02, default_alpha * 1.5), 1.0 / (t + 1), 0.05, 0.1]
        alphas = [max(0.01, min(1.0, a)) for a in alphas]
        alphas = list(set(alphas))

        best_alpha = default_alpha
        best_trial_M = float('inf')
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        if t == 1:
            y_new = g_dir.copy()
        else:
            y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_y

        y = y_new

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW iter={} ({:.1f}s): J={:.6f} |S|={}".format(t, elapsed, J, len(active_set)))

    return best_a, best_m


def peak_pair_search(a0, time_limit, verbose=True):
    """Targeted local search: identify top contributing pairs to peaks and reduce them."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:25]
        improved = False

        for _ in range(80):
            if time.monotonic() - t0 > time_limit:
                break
            # Pick a random peak from top peaks
            k_int = int(top_peaks[rng.integers(min(15, len(top_peaks)))])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue

            # Find top contributing indices
            top_contrib = np.argsort(contrib)[::-1][:5]
            src_idx = idx[top_contrib[rng.integers(len(top_contrib))]]

            # Find destination: away from current peaks
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:12]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) < 5:
                candidates = np.arange(n)
            dst = int(candidates[rng.integers(len(candidates))])
            if dst == src_idx:
                continue
            if a[src_idx] < 1e-12:
                continue

            frac = float(rng.choice([0.003, 0.008, 0.015, 0.03, 0.05]))
            amt = a[src_idx] * frac
            a_trial = a.copy()
            a_trial[src_idx] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-14:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                stall = 0
                ac = fast_autocorr(a)
                top_peaks = np.argsort(ac)[::-1][:25]

        if not improved:
            stall += 1
            # Random exploration to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.002, 0.02))
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    PeakPair ({:.1f}s): J={:.6f} stall={}".format(elapsed, J, stall))

    return best_a, best_m


def cyclic_shift_best(a0):
    """Try all cyclic shifts and return the best one."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64)
    best_m = fast_autocorr(a).max()
    best_shift = 0
    fft_size = 1 << (2 * n - 1).bit_length()
    # Precompute FFT of original
    for s in range(1, n):
        a_shift = np.roll(a, -s)
        AS = np.fft.rfft(a_shift, fft_size)
        b = np.fft.irfft(AS * AS, fft_size)[:2 * n - 1]
        m = b.max()
        if m < best_m:
            best_m = m
            best_shift = s
    if best_shift > 0:
        return np.roll(a, -best_shift), best_m
    return a.copy(), best_m


def try_mirror_improve(a):
    """Try averaging with reversal to see if it improves."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    a_rev = a[::-1]
    a_sym = (a + a_rev) / 2.0
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-15:
        return a
    a_sym = a_sym / s
    m_orig = fast_autocorr(a).max()
    m_sym = fast_autocorr(a_sym).max()
    if m_sym < m_orig:
        return a_sym
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.6f} (n={}, t={:.1f}s)".format(label, sc, len(a), elapsed))
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print("Time budget: {:.1f}s".format(time_budget))

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n={}, score={}".format(len(parent), p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n={}, score={}".format(len(best_seq), best_score))

    # Try mirror
    best_seq = try_mirror_improve(best_seq)
    best_seq, best_score = try_update(best_seq, best_score, best_seq, "mirror_init", t_global)

    n0 = len(best_seq)

    # Phase 1: FISTA (main optimization, 40% of budget)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA (n={}) ===".format(n0))
    t1 = min(remaining * 0.38, 600.0)
    if t1 > 60:
        a_out, _ = fista_adaptive(norm_sum1(best_seq), t1,
                                  beta_start=20.0, beta_end=800.0,
                                  beta_growth=1.003, top_k_start=8, top_k_end=35,
                                  restart_every=2500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista1", t_global)
        best_seq = try_mirror_improve(best_seq)
        best_seq, best_score = try_update(best_seq, best_score, best_seq, "mirror_p1", t_global)

    # Phase 2: Cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 2: Cyclic shift ===")
    if remaining > 30:
        a_shifted, m_shift = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic1", t_global)

    # Phase 3: Spectral spread
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 3: Spectral spread ===")
    t3s = min(remaining * 0.06, 90.0)
    if t3s > 20:
        a_out, _ = spectral_spread(norm_sum1(best_seq), t3s, n_iters=200, verbose=False)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spectral", t_global)

    # Phase 4: LP-Frank-Wolfe (20% of budget)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: LP-FW (n={}) ===".format(n_best))
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 40 and n_best <= 9000:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t4, solver,
                                  initial_K=400, max_K=2500, add_per_iter=25, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 5: Peak-pair local search (12% of budget)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Peak-pair search ===")
    t5 = min(remaining * 0.12, 180.0)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    # Try nearby n values
    n_candidates = []
    for delta in [500, -500, 1000, -1000]:

        if 2000 <= new_n <= 12000:

        print("  Trying n={}...".format(new_n))

                                      beta_start=25.0, beta_end=600.0,

                                      restart_every=2000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_n{}".format(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)

            a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
            best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_n{}".format(new_n), t_global)

    # Phase 7: Second FISTA with different params
    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 8: Final FISTA + local search
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 8: Final polish ===")
    t8a = min(remaining * 0.40, 100.0)
    if t8a > 20:
        a_out, _ = fista_adaptive(norm_sum1(best_seq), t8a,
                                  beta_start=100.0, beta_end=2000.0,
                                  beta_growth=1.01, top_k_start=5, top_k_end=20,

        best_seq, best_score = try_update(best_seq, best_score, a_out, "peakpair_final", t_global)

    # Final: cyclic shift
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_shifted, _ = cyclic_shift_best(norm_sum1(best_seq))
        best_seq, best_score = try_update(best_seq, best_score, a_shifted, "cyclic_final", t_global)

    best_seq = try_mirror_improve(best_seq)

        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
