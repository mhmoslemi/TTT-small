    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

    except Exception:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def lp_fw_core(a0, time_limit, solver, init_K=300, max_K=6000, add_per=60, verbose=True, seed=0):
    """Improved LP-based Frank-Wolfe with dual-guided active set management."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    ac = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[-init_K:])
    no_improve = 0
    stall_limit = 2000

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        coeffs = build_lp_matrix(y, clist, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        lp_ok = False
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                lp_ok = True
        except Exception:
            if len(active_set) > init_K:
                active_set = set(int(clist[:init_K]))
                continue

        if not lp_ok:
            if no_improve > stall_limit // 2:
                break
            if len(clist) > 100:
                active_set = set(int(clist[:200]))
                continue
            break

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Dual-guided active set update
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        # Find violations
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))

        # Also add near-peak lags that are close to M_curr
        near_peaks = np.where(ac > 0.95 * M_curr)[0]
        for k in near_peaks:
            if int(k) not in active_set and len(active_set) < max_K:
                active_set.add(int(k))

        # Golden section line search with more iterations
        lo = 0.005
        hi = 1.0

        def trial_obj(alpha):
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                return 1e10
            trial = trial / ts
            return fast_autocorr(trial).max()

        c1 = lo + inv_phi * (hi - lo)
        c2 = hi - inv_phi * (hi - lo)
        f1 = trial_obj(c1)
        f2 = trial_obj(c2)
        for _ in range(28):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = lo + inv_phi * (hi - lo)
                f1 = trial_obj(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = hi - inv_phi * (hi - lo)
                f2 = trial_obj(c2)

        best_alpha = (lo + hi) / 2.0

        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)))

    if verbose:
        print("    LP-FW done (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def coord_descent_refine(a0, time_limit, seed=42, verbose=True):
    """Aggressive coordinate descent with multi-peak awareness."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            top_srcs = idx[np.argsort(contrib)[::-1][:4]]
            for src in top_srcs:
                src = int(src)
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                scores = np.zeros(n)
                for k2 in top_k[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10

                candidates = np.argsort(scores)[:40]
                found = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.001, 0.003, 0.006, 0.012, 0.025, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break

        if not improved:
            stall += 1
            if stall > 200:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 1000 == 0:
            print("    Coord (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def diverse_init(n, parent_seq):

    x = np.linspace(0.0, 1.0, n)

    best_val = obj_val(best_a)

        candidates.append(normalize((n - i_arr) ** p))

    # Gaussian bumps

        for sg in [0.05, 0.08, 0.12, 0.15, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))

    # Double bumps
    for c1 in [0.15, 0.25, 0.35]:
        for c2 in [0.65, 0.75, 0.85]:
            for sg in [0.05, 0.08, 0.12, 0.18]:
                cands = np.exp(-((x - c1) ** 2) / (2 * sg * sg)) + np.exp(-((x - c2) ** 2) / (2 * sg * sg))
                candidates.append(normalize(cands))

    # Triple bumps
    for sg in [0.06, 0.1, 0.15]:
        cands = np.exp(-((x - 0.2) ** 2) / (2 * sg * sg)) + np.exp(-((x - 0.5) ** 2) / (2 * sg * sg)) + np.exp(-((x - 0.8) ** 2) / (2 * sg * sg))
        candidates.append(normalize(cands))

    # Cosine windows
    for f in [0.5, 1, 2, 3, 4, 6, 8]:
        cands = np.maximum(1.0 + np.cos(2 * np.pi * f * x), 0.001)
        candidates.append(normalize(cands))

    # Tukey / cosine-tapered
    for alpha in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        cands = np.ones(n)
        m = int(alpha * n)
        if m > 0:
            taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
            cands[:m] = taper
            cands[-m:] = taper[::-1]
        candidates.append(normalize(cands))

    # Power-law tapered
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        cands = np.maximum(x, 1e-10) ** p * np.maximum(1 - x, 1e-10) ** p
        candidates.append(normalize(cands))

    # Sine-weighted
    for f in [1, 2, 3, 4]:
        candidates.append(normalize(np.sin(np.pi * x) ** (2 * f)))

    # Binomial
    from math import comb
    try:
        binom = np.array([comb(n-1, i) for i in range(n)], dtype=np.float64)
        candidates.append(normalize(binom))
    except Exception:
        pass

    # Spectral: inverse FFT of shaped power spectrum
    for shape_p in [0.5, 1.0, 2.0]:
        freqs = np.fft.rfftfreq(n)
        mask = np.maximum(1.0 - np.abs(freqs) * 2.0, 0.0) ** shape_p
        spec = mask * np.ones_like(mask)
        a_spec = np.fft.irfft(spec * n, n=n)
        a_spec = np.abs(a_spec) + 1e-10
        candidates.append(normalize(a_spec))

    # Blend with parent
    if parent_seq is not None:
        bi = normalize(parent_seq)
        for w in [0.3, 0.5, 0.7]:
            candidates.append(normalize(w * bi + (1 - w) * np.ones(n) / n))
            if len(candidates) > 3:
                candidates.append(normalize(w * bi + (1 - w) * candidates[2]))

    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()

    return best_a, best_val

    a_lp2 = interpolate_seq(best_seq, n_lp2) if n_lp2 != n_cur else best_seq.copy()

    print("\n=== P3: LP-FW fine (n=" + str(n_lp2) + ") ===")
    t3 = min(remaining * 0.35, 600.0)
    if t3 > 60:
        a_out, v_out = lp_fw_core(normalize(a_lp2), t3, solver,
                                  init_K=500, max_K=5000, add_per=80,
                                  verbose=True, seed=42)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_cur) if n_cur > n_lp2 else a_out
            print("  LP-FW fine improved: " + repr(best_score))

    # === Phase 4: Multi-restart with perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_ms = min(n_cur, 5000)
    a_ms = interpolate_seq(best_seq, n_ms) if n_ms != n_cur else best_seq.copy()

    print("\n=== P4: Multi-restart ===")
    t4_total = min(remaining * 0.2, 300.0)
    if t4_total > 60:
        n_restarts = 3
        t_per_restart = t4_total / n_restarts
        rng = np.random.default_rng(99)

        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget - 200:
                break
            # Create perturbed starting point
            noise_scale = 0.05 + 0.05 * r
            noise = noise_scale * rng.standard_normal(n_ms) * a_ms
            a_pert = a_ms + noise
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()

            a_out, v_out = lp_fw_core(a_pert, t_per_restart, solver,
                                      init_K=300, max_K=4000, add_per=60,
                                      verbose=False, seed=100 + r)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = interpolate_seq(a_out, n_cur) if n_cur > n_ms else a_out
                print("  Restart " + str(r) + " improved: " + repr(best_score))

    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_pol = min(n_cur, 6000)
    a_pol = interpolate_seq(best_seq, n_pol) if n_pol != n_cur else best_seq.copy()

    print("\n=== P5: Coord Polish (n=" + str(n_pol) + ") ===")
    t5 = min(remaining * 0.2, 300.0)
    if t5 > 30:
        a_out, v_out = coord_descent_refine(normalize(a_pol), t5, seed=77, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = interpolate_seq(a_out, n_cur) if n_cur > n_pol else a_out
            print("  Polish improved: " + repr(best_score))

    # === Phase 6: Final LP burst ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    n_final = min(n_cur, 5000)
    a_fin = interpolate_seq(best_seq, n_final) if n_final != n_cur else best_seq.copy()

    if remaining > 60:
        print("\n=== P6: Final LP (n=" + str(n_final) + ") ===")
        t6 = min(remaining * 0.5, 200.0)
        if t6 > 30:
            a_out, v_out = lp_fw_core(normalize(a_fin), t6, solver,
                                      init_K=600, max_K=5000, add_per=80,
                                      verbose=True, seed=200)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = interpolate_seq(a_out, n_cur) if n_cur > n_final else a_out
                print("  Final LP improved: " + repr(best_score))

    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        n_fin2 = min(len(best_seq), 5000)
        a_fin2 = interpolate_seq(best_seq, n_fin2) if n_fin2 != len(best_seq) else best_seq.copy()
        print("\n=== P7: Final Polish ===")
        t7 = min(remaining * 0.6, 150.0)
        if t7 > 15:
            a_out, v_out = coord_descent_refine(normalize(a_fin2), t7, seed=314, verbose=False)
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = interpolate_seq(a_out, len(best_seq)) if len(best_seq) > n_fin2 else a_out
                print("  Final polish improved: " + repr(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) + " t=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: fallback")
        n_fb = 8000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
