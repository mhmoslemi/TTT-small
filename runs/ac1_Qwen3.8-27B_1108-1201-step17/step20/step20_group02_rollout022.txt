def fast_convolve(a):
    """FFT-based full convolution of a with itself."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """FFT-based full cross-convolution of a and b (same length)."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    try:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_grad_matrix(y, active_set, n):
    """Build matrix where row k has grad of conv[k] at y.
    grad_k[i] = 2*y[k-i] if 0<=k-i<n, else 0.
    We divide by 2 since it cancels in LP."""
    K = len(active_set)
    coeffs = np.zeros((K, n))
    for jj in range(K):
        k_int = int(active_set[jj])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs


def lp_dir(y, active_set, n, solver):
    """Solve LP to find FW direction minimizing max over active constraints.
    min M s.t. sum(g)=1, g>=0, A@y_k^T @ g <= M for each active k.
    Returns the direction g (normalized to sum 1)."""
    K = len(active_set)
    if K == 0:
        return y.copy()
    coeffs = build_grad_matrix(y, active_set, n)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return y.copy()
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return y.copy()
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return y.copy()
    return g_val / g_sum


def vanilla_fw_dir(y, n):
    """Vanilla FW: pick direction from the single worst constraint."""
    ac = fast_convolve(y)
    k_star = int(np.argmax(ac))
    # Gradient of conv[k_star] at y: grad[i] = 2*y[k_star-i] if valid
    i_start = max(0, k_star - n + 1)
    i_end = min(n, k_star + 1)
    idx = np.arange(i_start, i_end)
    grad = np.zeros(n)
    grad[idx] = y[k_star - idx]
    # FW direction: e_{argmin grad}
    i_min = int(np.argmin(grad))
    e_i = np.zeros(n)
    e_i[i_min] = 1.0
    return e_i, k_star


def hybrid_fw_step(y, n, active_set, solver, use_lp, iter_count):
    """Hybrid FW step: use LP every few iterations, vanilla otherwise."""
    if use_lp:
        g_dir = lp_dir(y, active_set, n, solver)
    else:
        g_dir, _ = vanilla_fw_dir(y, n)
    return g_dir


def backtracking_line_search(y, g_dir, M_curr, n, max_alphas=12):
    """Find best step size via backtracking."""
    best_alpha = 1.0 / (iter_count + 3.0)
    best_M = M_curr
    alphas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15,
              1.0 / (iter_count + 2.0), 1.0 / (iter_count + 3.0),
              max(0.05, 2.0 / (iter_count + 3.0))]
    alphas = sorted(set(max(0.01, a) for a in alphas), reverse=True)[:max_alphas]
    for alpha in alphas:
        trial = (1.0 - alpha) * y + alpha * g_dir
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-15:
            continue
        trial = trial / s_t
        M_trial = fast_convolve(trial).max()
        if M_trial < best_M - 1e-15:
            best_M = M_trial
            best_alpha = alpha
    return best_alpha, best_M

            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = best_a.copy()
            else:
                y = y / s_y

        # Adaptive: increase LP frequency if improving
        if best_trial_M < best_m:
            lp_every = max(1, lp_every - 1)
        elif no_improve > 100:
            lp_every = min(lp_every + 1, max(5, n // 500))

        if verbose and iter_count % 200 == 0:
            print("    FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): "
                  + "J=" + repr(2.0 * n * best_m) + ", |S|=" + str(len(active_set))
                  + ", lp_every=" + str(lp_every))

    if verbose:
        print("    FW done (" + str(round(time.monotonic() - t0, 1)) + "s): "
              + "J=" + repr(2.0 * n * best_m) + ", iters=" + str(iter_count))
    return best_a, best_m


def kkt_refine(a0, time_limit, verbose=True):
    """KKT-based refinement: iteratively adjust a to satisfy KKT conditions.
    At optimum: sum_{k in S} lambda_k * grad_k(a) = -mu * 1 (on support of a)
    This means the weighted sum of shifted copies is constant on the support."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()

    for outer in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_convolve(a)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = a.copy()

        # Active set: lags close to the max
        threshold = M_curr - 1e-6
        active = np.where(ac >= threshold)[0]
        if len(active) > 50:
            active = np.argsort(ac)[::-1][:50]
        if len(active) < 3:
            active = np.argsort(ac)[::-1][:10]

        # Build gradient matrix V: V[i, j] = grad_{active[j]}[i] = a[k_j - i] if valid
        K = len(active)
        V = np.zeros((n, K))
        for jj in range(K):
            k = int(active[jj])
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            V[idx, jj] = a[k - idx]

        # Solve for lambda, mu: V @ lambda + mu = 0 (on support), V @ lambda + mu <= 0 (off support)
        # Simplified: find lambda >= 0, mu such that V @ lambda + mu * 1 is as close to 0 as possible on support
        support = a > 1e-10
        if support.sum() < 10:
            break

        # Least squares on support: minimize ||V[support] @ lambda + mu||^2 subject to lambda >= 0
        V_s = V[support]
        # Add mu as a variable with value 1: augmented system
        # [V_s | 1] @ [lambda; mu] = 0
        n_support = support.sum()
        A_aug = np.hstack([V_s, np.ones((n_support, 1))])

        # Solve non-negative least squares for lambda, allow mu to be free
        # Use a simple approach: project gradient descent on the dual
        lam = np.ones(K) / K
        mu_val = -V_s @ lam / max(n_support, 1)

        for inner in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            # Residual on support
            resid = V_s @ lam + mu_val
            # Update mu to minimize ||resid||^2: mu = -mean(V_s @ lam)
            mu_val = -np.mean(V_s @ lam)
            # Update lam via projected gradient
            # d/d lam: V_s^T @ (V_s @ lam + mu)
            grad_lam = V_s.T @ (V_s @ lam + mu_val)
            lr = 1.0 / (np.linalg.norm(V_s, 2) ** 2 + 1e-10)
            lam = lam - lr * grad_lam
            lam = np.maximum(lam, 0)
            if lam.sum() < 1e-15:
                lam = np.ones(K) / K

        # Dual values: d[i] = V[i] @ lam + mu for all i
        dual = V @ lam + mu_val

        # Primal update: a[i] should be 0 where dual[i] > 0 (KKT complementarity)
        # Blend: reduce a where dual is positive
        positive_dual = np.maximum(dual, 0)
        scale = np.exp(-positive_dual * n * 0.5)
        a_new = a * scale
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Check improvement
        M_new = fast_convolve(a_new).max()
        if M_new < M_curr - 1e-14:
            a = a_new
        elif M_new < M_curr + 1e-10:
            # Accept small changes to help escape
            a = 0.7 * a + 0.3 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()

    if verbose:
        print("    KKT done (" + str(round(time.monotonic() - t0, 1)) + "s): "
              + "J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def coordinate_swap_polish(a0, time_limit, rng_seed=42, verbose=True):
    """Pairwise swap coordinate descent: move mass between pairs to reduce peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_convolve(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:20]

        improved = False

        # For each top lag, find the pair that contributes most and try to reduce it
        for k in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]
            c_max = contrib.max()
            if c_max < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination: minimize impact on all top lags
            scores = np.zeros(n)
            for k2 in top_lags[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            # Prefer moving to low-score positions
            candidates = np.where(scores < np.median(scores))[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            dst = int(candidates[np.argmin(scores[candidates] / (1e-15 + a[candidates]))])
            if dst == src:
                continue

            # Try transfer
            found = False
            for frac in [0.01, 0.005, 0.002, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_convolve(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

            if not found:
                # Random perturbation
                j = int(rng.integers(n))
                if j != src:
                    frac = rng.uniform(0.001, 0.01)
                    amt = src_val * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_convolve(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True

        if not improved:
            stall += 1
            if stall % 100 == 0:
                # Larger random perturbations
                for _ in range(20):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0)
                        s_t = a.sum()
                        if s_t > 1e-15:
                            a = a / s_t
                            M_new = fast_convolve(a).max()
                            if M_new < best_m:
                                best_m = M_new
                                best_a = a.copy()
                                stall = 0

        if verbose and stall % 300 == 0:
            print("    Coord (" + str(round(time.monotonic() - t0, 1)) + "s): "
                  + "J=" + repr(2.0 * n * best_m) + ", stall=" + str(stall))

    if verbose:
        print("    Coord done (" + str(round(time.monotonic() - t0, 1)) + "s): "
              + "J=" + repr(2.0 * n * best_m))
    return best_a, best_m


def get_initial_sequence():
    """Get initial sequence from parent or analytical forms."""
    candidates = []

    # Try parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 100 and np.isfinite(p).all() and p.sum() > 1e-10:
            p_norm = p / p.sum()
            sc = evaluate_safe(p_norm)
            if np.isfinite(sc):
                candidates.append((sc, p_norm, "parent"))
                print("  Parent: n=" + str(len(p)) + ", score=" + repr(sc))
    except Exception:
        pass

    # Analytical initializations
    for n in [4000, 5000, 6000, 7000, 8000]:
        x = np.linspace(0, 1, n)
        # Sine squared
        f1 = np.sin(np.pi * x) ** 2
        # Skewed sine
        f2 = np.sin(np.pi * x ** 0.7) ** 2
        # Exponential decay
        f3 = np.exp(-2.0 * x)
        # Skewed: x^0.5 * (1-x)^1.5
        f4 = x ** 0.5 * (1 - x) ** 1.5
        # Double hump asymmetric
        f5 = 0.6 * np.exp(-((x - 0.3) ** 2) / 0.04) + 0.4 * np.exp(-((x - 0.7) ** 2) / 0.06)

        for f, name in [(f1, "sin2"), (f2, "skew_sin"), (f3, "exp"),
                        (f4, "asym_beta"), (f5, "hump")]:
            f = np.maximum(f, 0)
            s = f.sum()
            if s > 1e-10:
                f = f / s
                sc = evaluate_safe(f)
                if np.isfinite(sc):
                    candidates.append((sc, f, name + "_n" + str(n)))

    if candidates:
        candidates.sort(key=lambda x: x[0])
        best_sc, best_seq, best_name = candidates[0]
        print("  Best init: " + best_name + ", score=" + repr(best_sc))
        return best_seq, best_sc
    else:
        n = 5000
        x = np.linspace(0, 1, n)
        f = np.sin(np.pi * x) ** 2
        f = f / f.sum()
        sc = evaluate_safe(f)
        print("  Fallback init: n=" + str(n) + ", score=" + repr(sc))
        return f, sc


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("*** NEW BEST [" + label + "]: " + str(sc) + " (n=" + str(len(a)) +
              ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)
    print("Total budget: " + repr(time_budget) + "s")

    solver = get_solver()
    best_seq, best_score = get_initial_sequence()
    n0 = len(best_seq)

    def remaining_time():
        return time_budget - (time.monotonic() - t_global)

    # === Phase 1: Main LP-FW on parent n ===
    print("\n=== Phase 1: LP-FW main (n=" + str(n0) + ") ===")
    t1 = min(remaining_time() * 0.45, 700.0)
    if t1 > 60 and n0 <= 12000:
        a_out, _ = lp_fw_optimize(normalize_sum1(best_seq), t1, solver,
                                  initial_K=400, max_K=5000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw1_n" + str(n0), t_global)

    # === Phase 2: KKT refinement ===
    print("\n=== Phase 2: KKT refine (n=" + str(len(best_seq)) + ") ===")
    t2 = min(remaining_time() * 0.10, 150.0)
    if t2 > 20:
        a_out, _ = kkt_refine(norm_sum1(best_seq), t2, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "kkt_n" + str(len(best_seq)), t_global)

    # === Phase 3: LP-FW second pass ===
    print("\n=== Phase 3: LP-FW second (n=" + str(len(best_seq)) + ") ===")
    t3 = min(remaining_time() * 0.20, 300.0)
    if t3 > 40 and len(best_seq) <= 12000:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t3, solver,
                                  initial_K=600, max_K=6000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw2_n" + str(len(best_seq)), t_global)

    # === Phase 4: Multi-n exploration ===
    print("\n=== Phase 4: Multi-n exploration ===")
    n_best = len(best_seq)
    # Try n values around the current best, and some spread
    n_candidates = []
    for delta in [-100, -50, 50, 100, -200, 200, -300, 300]:
        nn = n_best + delta
        if 3000 <= nn <= 12000:
            n_candidates.append(nn)
    # Also try some absolute values
    for nn in [4000, 5000, 6000, 7000, 8000, 9000, 10000]:
        if nn not in n_candidates and 3000 <= nn <= 12000:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_best))

    for new_n in n_candidates[:6]:
        rt = remaining_time()
        if rt < 80:
            break
        print("\n  Multi-n: n=" + str(new_n) + " (remaining=" + str(round(rt, 1)) + "s)")
        a_interp = interpolate_seq(best_seq, new_n)

        # Quick FW
        t4 = min(rt * 0.25, 200.0)
        if t4 > 30 and new_n <= 12000:
            a_out, _ = lp_fw_optimize(a_interp, t4, solver,
                                      initial_K=300, max_K=4000, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fw_n" + str(new_n), t_global)

        # Quick coordinate polish
        rt = remaining_time()
        if rt > 30 and len(best_seq) <= 12000:
            t4b = min(rt * 0.15, 100.0)
            if t4b > 15:
                a_out, _ = coordinate_swap_polish(norm_sum1(best_seq), t4b,
                                                  rng_seed=new_n, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out,
                                                  "coord_n" + str(len(best_seq)), t_global)

    # === Phase 5: Final LP-FW on best ===
    print("\n=== Phase 5: Final LP-FW (n=" + str(len(best_seq)) + ") ===")
    t5 = min(remaining_time() * 0.30, 350.0)
    if t5 > 40 and len(best_seq) <= 12000:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t5, solver,
                                  initial_K=800, max_K=7000, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lpfw_final", t_global)

    # === Phase 6: Final KKT + coordinate polish ===
    rt = remaining_time()
    if rt > 20 and len(best_seq) <= 12000:
        t6 = min(rt * 0.30, 100.0)
        if t6 > 10:
            a_out, _ = kkt_refine(norm_sum1(best_seq), t6, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "kkt_final", t_global)

    rt = remaining_time()
    if rt > 15 and len(best_seq) <= 12000:
        t7 = min(rt * 0.50, 100.0)
        if t7 > 10:
            a_out, _ = coordinate_swap_polish(norm_sum1(best_seq), t7,
                                              rng_seed=999, verbose=False)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "coord_final", t_global)

    # Final validation
    if best_seq is not None and len(best_seq) > 0:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) +
              ", time=" + str(round(elapsed, 1)) + "s")
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print("Fallback: score=" + repr(best_score))

    return [float(x) for x in best_seq]


def normalize_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s
