```python
def compute_grad(a, beta, top_k=None):

    c = fast_autocorr(a)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
        c_max = c[idx].max()
    else:
        c_use = c
        c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros_like(c_use)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    
    # Compute grad[j] = 2 * sum_k p[k] * a[k-j]
    # = 2 * conv(p, a[::-1])[j + n - 1]
    fft_size = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_flip = np.fft.rfft(a[::-1], fft_size)
    conv_result = np.fft.irfft(P * A_flip, fft_size)
    grad = 2.0 * conv_result[n-1 : 2*n-1]
    return grad, p, c
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer on log-sum-exp smoothed max autocorrelation
2. Targeted peak surgery with pressure-based destination selection
3. Simulated annealing with structured mutations
4. Multi-parameter sweeps (beta, top_k, learning rate)
5. Periodic restarts with diverse initializations
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def simplex_project(z):
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

def compute_smoothed_grad(a, beta, top_k=None):
    n = len(a)
    c = fast_autocorr(a)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
    else:
        c_use = c
    c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    
    fft_size = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_flip = np.fft.rfft(a[::-1], fft_size)
    conv_result = np.fft.irfft(P * A_flip, fft_size)
    grad = 2.0 * conv_result[n-1 : 2*n-1]
    return grad, p, c

def adam_optimize(a0, time_limit, beta=200.0, top_k=20, lr=0.05,
                  beta1=0.9, beta2=0.999, verbose=True, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t_start = time.monotonic()
    eps = 1e-8
    rng = np.random.default_rng(seed)
    stall = 0
    stall_threshold = 200
    
    while time.monotonic() - t_start < time_limit:
        t += 1
        grad, p, c = compute_smoothed_grad(a, beta, top_k)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = simplex_project(a_new)
        
        new_m = fast_autocorr(a_new).max()
        if new_m < best_m - 1e-15:
            best_m = new_m
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        if stall > stall_threshold:
            # Restart with perturbation
            noise = rng.normal(0, 0.01, n)
            a_r = best_a + noise * best_a
            a_r = np.maximum(a_r, 0.0)
            s = a_r.sum()
            if s > 1e-15:
                a_r = a_r / s
            a = a_r
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            stall = 0
            lr = max(lr * 0.8, 0.001)
            beta = min(beta * 1.5, 5000.0)
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_m
            print("    Adam t={} ({}s): J={}, beta={}, lr={}".format(
                t, round(elapsed,1), J, round(beta,1), round(lr,5)))
    
    return best_a, best_m

def peak_surgery(a0, time_limit, verbose=True, seed=77):

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    while time.monotonic() - t_start < time_limit and no_improve < 2000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top peaks

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
```

```python
def compute_grad_logspace(x, a, beta, top_k=None):
    """Compute gradient of smoothed max autocorr w.r.t. log-space parameters x."""
    n = len(a)
    c = fast_autocorr(a)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(len(c))
        c_use[idx] = c[idx]
    else:
        c_use = c
    c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    
    # S_l = sum_k p_k * a_{k-l}
    # This is conv(p, a[::-1])[l + n - 1]
    fft_size = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_flip = np.fft.rfft(a[::-1], fft_size)
    conv_result = np.fft.irfft(P * A_flip, fft_size)
    S = conv_result[n-1:2*n-1]
    
    # C = sum_k p_k * c_k
    C = np.dot(p, c)
    
    # grad_x_l = 2 * a_l * (S_l - C)
    grad = 2.0 * a * (S - C)
    return grad
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer in log-space (softmax parameterization) - avoids projection issues
2. Targeted peak surgery with pressure-based destination selection
3. Simulated annealing with structured mutations for global exploration
4. Multi-length exploration and adaptive beta/top_k scheduling
5. Interleaved refinement: Adam bursts alternating with surgery
Key insight: log-space gradient has clean form grad_x_l = 2*a_l*(S_l - C)
where S_l = weighted correlation and C = weighted mean autocorrelation.

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safe wrapper around evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    return ex / ex.sum()


def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def compute_p_weights(c, beta, top_k=None):
    """Compute softmax weights over autocorrelation values."""
    n_c = len(c)
    if top_k is not None and top_k < n_c:
        idx = np.argsort(c)[::-1][:top_k]
        c_use = np.zeros(n_c)
        c_use[idx] = c[idx]
    else:
        c_use = c
    c_max = c.max()
    shifted = beta * (c_use - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros(n_c)
        p[np.argmax(c)] = 1.0
    else:
        p = exp_vals / total
    return p


def compute_grad_logspace(a, beta, top_k=None):
    """
    Compute gradient of smoothed max autocorrelation w.r.t. log-space params.
    grad_x_l = 2 * a_l * (S_l - C)
    where S_l = sum_k p_k * a_{k-l}, C = sum_k p_k * c_k
    """
    n = len(a)
    c = fast_autocorr(a)
    p = compute_p_weights(c, beta, top_k)

    # S_l = sum_k p_k * a_{k-l} = conv(p, a[::-1])[l + n - 1]
    fft_size = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A_flip = np.fft.rfft(a[::-1], fft_size)
    conv_result = np.fft.irfft(P * A_flip, fft_size)
    S = conv_result[n - 1:2 * n - 1]

    # C = sum_k p_k * c_k
    C = np.dot(p, c)

    # grad_x_l = 2 * a_l * (S_l - C)
    grad = 2.0 * a * (S - C)
    return grad, p, c


def adam_logspace(a0, time_limit, beta=300.0, top_k=20, lr=0.05,
                  beta1=0.9, beta2=0.999, eps=1e-8,
                  restart_every=300, verbose=True, seed=42):
    """Adam optimizer in log-space (softmax parameterization)."""
    n = len(a0)
    a = norm_sum1(a0)
    a = np.maximum(a, 1e-12)
    x = np.log(a)
    x = x - x.sum()  # center for numerical stability (doesn't change softmax)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t_start < time_limit:
        t += 1
        a = softmax(x)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        grad, p, c = compute_grad_logspace(a, beta, top_k)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step
        # Re-center for numerical stability
        if t % 100 == 0:
            x = x - x.mean()

        new_a = softmax(x)
        new_a = np.maximum(new_a, 1e-15)
        new_a = new_a / new_a.sum()

        new_m = fast_autocorr(new_a).max()
        if new_m < best_m - 1e-16:
            best_m = new_m
            best_a = new_a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive: increase beta over time to focus more
        if t % 200 == 0:
            beta = min(beta * 1.1, 5000.0)
            # Reduce top_k over time
            if top_k > 3:
                top_k = max(3, int(top_k * 0.9))

        # Restart mechanism
        if stall > restart_every:
            stall = 0
            # Perturb in log-space
            noise_scale = 0.02 + 0.01 * (t % 500) / 500.0
            x = np.log(np.maximum(best_a, 1e-12))
            x += rng.normal(0, noise_scale, n)
            x = x - x.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = max(lr * 0.85, 0.001)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_m
            print("    AdamLS t={} ({}s): J={}, beta={}, top_k={}, lr={}".format(
                t, round(elapsed, 1), round(J, 6), round(beta, 1), top_k, round(lr, 5)))

    return best_a, best_m


def peak_surgery(a0, time_limit, verbose=True, seed=77):
    """Targeted local search: identify peaks and reduce via mass redistribution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t_start < time_limit and no_improve < 3000:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-16:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_top = min(25, len(c))
        top_idx = np.argsort(c)[::-1][:n_top]

        improved = False
        found_move = False

        for k in top_idx[:12]:
            if found_move:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            if products.max() < 1e-20:
                continue

            top_pairs = np.argsort(products)[::-1][:5]

            for pi in top_pairs:
                if found_move:
                    break
                src_i = int(idx[pi])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue

                if a[src_i] >= a[partner]:
                    reduce_i = src_i
                else:
                    reduce_i = partner

                if a[reduce_i] < 1e-14:
                    continue

                # Compute pressure
                pressure = np.zeros(n)
                for k2 in top_idx[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        pressure[idx2] += a[k2i - idx2]
                        pressure[k2i - idx2] += a[idx2]

                # Find candidates outside peak regions
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_idx[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    in_peak[s2:e2 + 1] = True
                in_peak[reduce_i] = True

                candidates = np.where(~in_peak)[0]
                if len(candidates) < 50:
                    candidates = np.arange(n)
                    candidates = candidates[candidates != reduce_i]

                n_try = min(10, len(candidates))
                try_dsts = candidates[np.argsort(pressure[candidates])[:n_try]]

                for dst in try_dsts:
                    if found_move:
                        break
                    dst_int = int(dst)
                    best_frac = None
                    best_trial_m = best_m
                    for frac in [0.002, 0.005, 0.01, 0.015, 0.02, 0.03]:
                        amt = a[reduce_i] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_i] -= amt
                        a_trial[dst_int] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_trial_m:
                            best_trial_m = M_trial
                            best_frac = frac

                    if best_frac is not None and best_trial_m < best_m - 1e-16:
                        amt = a[reduce_i] * best_frac
                        a[reduce_i] -= amt
                        a[dst_int] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        best_m = best_trial_m
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        no_improve = 0
                        break

        if not improved and no_improve > 300:
            # Random exploration
            for _ in range(30):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.04)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break

        if verbose and no_improve % 1000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_m
            print("    PeakSurgery ({}s): J={} stall={}".format(
                round(elapsed, 1), round(J, 6), no_improve))

    return best_a, best_m


def simulated_annealing(a0, time_limit, verbose=True, seed=99):
    """Simulated annealing in log-space with structured mutations."""
    n = len(a0)
    a = norm_sum1(a0)
    a = np.maximum(a, 1e-12)
    x = np.log(a)
    x = x - x.mean()

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)

    T = 0.015
    T_min = 0.0002
    cooling = 0.9997
    current_m = best_m

    step_count = 0
    while time.monotonic() - t_start < time_limit:
        step_count += 1

        # Generate mutation in log-space
        scale = T * 3.0
        mutation = rng.normal(0, scale, n)
        # Add structured component
        if rng.random() < 0.3:
            # Anti-symmetric
            anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
            mutation += T * rng.uniform(0.5, 2.0) * anti
        elif rng.random() < 0.5:
            # Low-frequency sinusoidal
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            sin_comp = np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            mutation += T * rng.uniform(0.5, 2.0) * sin_comp

        x_new = x + mutation
        x_new = x_new - x_new.mean()
        a_new = softmax(x_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        m_new = fast_autocorr(a_new).max()
        delta = m_new - current_m

        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            x = x_new
            a = a_new
            current_m = m_new
            if m_new < best_m:
                best_m = m_new
                best_a = a_new.copy()

        T = max(T * cooling, T_min)

        if verbose and step_count % 20000 == 0:
            elapsed = time.monotonic() - t_start
            J = 2.0 * n * best_m
            print("    Anneal step={} ({}s): J={}, T={}".format(
                step_count, round(elapsed, 1), round(J, 6), T))

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best if candidate is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [{}]: {:.8f} (n={}, t={:.1f}s)".format(
            label, sc, len(a_norm), elapsed))
    return best_seq, best_score
