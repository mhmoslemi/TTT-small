def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1:
        arr = arr.ravel()
    n = arr.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.0
    np.maximum(arr, 0.0, out=arr)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return arr / s


def fft_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def fft_conv2(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if b.size != n:
        b = np.resize(b, n)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * B, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
    return c


def cross_convolve_p_a(p, a, factor=1.0):
    p = np.asarray(p, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    total = 3 * n - 2
    if p.size != 2 * n - 1:
        p = np.resize(p, 2 * n - 1)
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    Pr = np.fft.rfft(p[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    out = conv[n - 1:2 * n - 1][::-1]
    out = np.where(np.isfinite(out), out, 0.0)
    return factor * out


def proxy_max(c, mask=None):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0
    if mask is None:
        vals = c
    else:
        vals = c[mask]
    if vals.size == 0:
        return 0.0
    vals = vals[np.isfinite(vals)]
    if vals.size == 0:
        return 0.0
    return float(np.max(vals))


def active_top(c, mask, max_k):
    c = np.asarray(c, dtype=np.float64)
    size = c.size
    max_k = int(min(max_k, size))
    if max_k <= 0:
        return np.array([], dtype=int)
    if mask is None:
        return np.argsort(c)[::-1][:max_k]
    vals = c.copy()
    vals[~mask] = -np.inf
    return np.argsort(vals)[::-1][:max_k]


def safe_eval(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            return float("inf")
        arr = np.clip(arr, 0.0, 1000.0)
        if not np.all(np.isfinite(arr)):
            return float("inf")
        s = float(arr.sum())
        if s < 0.01:
            return float("inf")
        lst = [float(x) for x in arr]
        val = evaluate_sequence(lst)
        if val is None:
            return float("inf")
        val = float(val)
        if not math.isfinite(val):
            return float("inf")
        return val
    except Exception:
        return float("inf")


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    try:
        arr = np.asarray(a, dtype=np.float64)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        np.maximum(arr, 0.0, out=arr)
        s = float(arr.sum())
        if arr.size == 0 or s < 0.01:
            return best_seq, best_score
        arr = arr / s
        sc = safe_eval(arr)
        if math.isfinite(sc) and sc < best_score - 1e-15:
            best_seq = arr.copy()
            best_score = float(sc)
            print("  NEW BEST [" + label + "]: " + repr(sc) +
                  " n=" + str(arr.size) +
                  " t=" + repr(time.monotonic() - t_global))
    except Exception:
        pass
    return best_seq, best_score


def interpolate_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(max(1, int(new_n)), dtype=np.float64)
    new_n = int(max(1, new_n))
    if a.size == new_n:
        return norm_sum1(a)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.where(np.isfinite(y), y, 0.0)
    np.maximum(y, 0.0, out=y)
    return norm_sum1(y)


def make_v(a, ref=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64), 0
    if ref is None:
        if np.any(a > 0.0):
            ref = int(np.argmax(a))
        else:
            ref = 0
    ref = int(ref)
    a_ref = max(float(a[ref]), 1e-300)
    loga = np.log(np.maximum(a, 1e-300)) - math.log(a_ref)
    if ref == 0:
        v = loga[1:]
    elif ref == n - 1:
        v = loga[:-1]
    else:
        v = np.concatenate((loga[:ref], loga[ref + 1:]))
    v = np.where(np.isfinite(v), v, -700.0)
    return v, ref


def from_v(v, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    u = np.zeros(n, dtype=np.float64)
    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]
    um = float(np.max(u))
    e = np.exp(u - um)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def lse_a(a, beta, mask=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return 0.0, np.array([], dtype=np.float64)
    size = 2 * n - 1
    total = 3 * n - 2
    N = next_pow2(total)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:size]
    c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)

    if mask is None:
        cv = c
    else:
        cv = c[mask]

    if cv.size == 0:
        return 0.0, np.zeros(n, dtype=np.float64)

    m = float(np.max(cv))
    shifted = beta * (cv - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    e = np.exp(shifted)
    se = float(e.sum())
    if se <= 0.0 or not math.isfinite(se):
        e = np.zeros(cv.size, dtype=np.float64)
        e[0] = 1.0
        se = 1.0
    p = e / se

    pf = np.zeros(size, dtype=np.float64)
    if mask is None:
        pf[:] = p
    else:
        pf[mask] = p

    Pr = np.fft.rfft(pf[::-1], N)
    conv = np.fft.irfft(A * Pr, N)[:total]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    f = m + math.log(se) / beta

    if not math.isfinite(f):
        f = 1e100
    return f, grad


def objgrad_v(v, beta, mask, ref):
    v = np.asarray(v, dtype=np.float64)
    n = v.size + 1
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return 1e100, np.zeros(v.size, dtype=np.float64)

    ref = int(ref)
    if not (0 <= ref < n):
        ref = n - 1

    u = np.empty(n, dtype=np.float64)
    u[ref] = 0.0
    if ref > 0:
        u[:ref] = v[:ref]
    if ref < n - 1:
        u[ref + 1:] = v[ref:]

    a = from_v(v, ref)
    f, ga = lse_a(a, beta, mask)
    if not math.isfinite(f):
        f = 1e100
        ga = np.zeros(n, dtype=np.float64)

    mean = float(np.dot(a, ga))
    gu = a * (ga - mean)

    gv = np.empty(v.size, dtype=np.float64)
    if ref > 0:
        gv[:ref] = gu[:ref]
    if ref < n - 1:
        gv[ref:] = gu[ref + 1:]
    gv = np.where(np.isfinite(gv), gv, 0.0)
    return f, gv


def run_lbfegs(a0, beta, mask, maxfun=400, verbose=True):
    a = norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a, np.array([], dtype=np.float64), 0

    v0, ref = make_v(a)
    if not SCIPY_OK or maxfun <= 0:
        return a, v0, ref

    t0 = time.monotonic()
    options = {
        "maxfun": int(maxfun),
        "maxiter": int(maxfun + 50),
        "ftol": 1e-13,
        "gtol": 1e-8,
        "disp": False
    }
    res = None
    try:
        res = scipy.optimize.minimize(
            objgrad_v,
            v0,
            args=(beta, mask, ref),
            method="L-BFGS-B",
            jac=True,
            options=options
        )
    except Exception:
        res = None

    if res is not None and hasattr(res, "x"):
        vx = np.asarray(res.x, dtype=np.float64)
        if vx.size == v0.size and np.all(np.isfinite(vx)):
            a_out = from_v(vx, ref)
            a_out = norm_sum1(a_out)
            v_out, ref_out = make_v(a_out)
            if verbose:
                print("    L-BFGS beta=" + repr(beta) + " n=" + str(n) +
                      " time=" + repr(time.monotonic() - t0))
            return a_out, v_out, ref_out
    return a, v0, ref


def lbfegs_phase(a_init, budget, cost):
    a = norm_sum1(a_init)
    n = a.size
    size = 2 * n - 1
    c = fft_conv(a)
    best_M = proxy_max(c, None)
    best_a = a.copy()
    current_a = a.copy()

    t1 = int(min(800, max(50, size // 8)))
    t2 = int(min(500, max(40, size // 15)))
    t3 = int(min(300, max(30, size // 25)))
    t4 = int(max(15, size // 80))

    schedules = [
        (8.0, None),
        (20.0, None),
        (50.0, None),
        (100.0, t1),
        (200.0, t2),
        (500.0, t3),
        (1000.0, t4)
    ]

    total_eval_budget = int(budget / max(2.5 * cost, 0.03))
    per_run = int(max(100, min(800, max(1, total_eval_budget // max(1, len(schedules))))))

    t0 = time.monotonic()
    for beta, top_count in schedules:
        if time.monotonic() - t0 > budget - 5:
            break
        if top_count is None:
            msk = None
        else:
            c_cur = fft_conv(current_a)
            top = active_top(c_cur, None, top_count)
            if top.size > 0:
                msk = np.zeros(size, dtype=bool)
                msk[top] = True
            else:
                msk = None

        a_new, _, _ = run_lbfegs(current_a, beta, msk, maxfun=per_run, verbose=True)
        c_new = fft_conv(a_new)
        m_new = proxy_max(c_new, None)

        if math.isfinite(m_new) and m_new <= best_M * (1.0 + 5e-5):
            current_a = a_new
        else:
            current_a = best_a

        if math.isfinite(m_new) and m_new < best_M - 1e-16:
            best_M = m_new
            best_a = a_new.copy()

    return best_a


def game_direction(f, active_idx, mw_iters=30, seed=0, temperature=8.0):
    f = norm_sum1(f)
    n = f.size
    size = 2 * n - 1
    active_idx = np.asarray(active_idx, dtype=int)
    K = active_idx.size
    if K <= 0:
        return np.ones(n, dtype=np.float64) / float(n)

    rng = np.random.default_rng(seed)
    logits = np.zeros(K, dtype=np.float64)
    logits += rng.normal(0.0, 0.1, K)

    def norm_logits(loc):
        loc = np.where(np.isfinite(loc), loc, -1000.0)
        loc = loc - float(np.max(loc))
        w = np.exp(loc)
        s = float(w.sum())
        if s <= 0.0:
            w = np.ones(K, dtype=np.float64)
            s = float(K)
        return w / s

    lambda_w = norm_logits(logits)

    for _ in range(max(1, int(mw_iters))):
        pf = np.zeros(size, dtype=np.float64)
        pf[active_idx] = lambda_w

        pressure = cross_convolve_p_a(pf, f, 1.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        pmx = float(np.max(pressure))
        pmn = float(np.min(pressure))
        ptp = pmx - pmn
        if ptp <= 1e-30:
            g = np.ones(n, dtype=np.float64) / float(n)
        else:
            scaled = (pressure - pmn) / ptp
            w = np.exp(-temperature * scaled)
            w = np.where(np.isfinite(w), w, 0.0)
            w += 1e-12
            g = w / float(w.sum())
            g = 0.95 * g + 0.05 * (np.ones(n, dtype=np.float64) / float(n))
            g = g / float(g.sum())

        h = fft_conv2(f, g)
        ha = h[active_idx]
        ha = np.where(np.isfinite(ha), ha, 0.0)
        hmac_ = float(np.max(ha))
        hmin = float(np.min(ha))
        hptp = hmac_ - hmin
        if hptp > 1e-30:
            eta = 10.0 / hptp
            logits += eta * (ha - float(np.mean(ha)))
        lambda_w = norm_logits(logits)

    return g


def lp_direction(f, active_idx, time_limit=10.0, max_k=2000):
    if not SCIPY_OK:
        return None
    if not check_hihts():
        return None

    f = norm_sum1(f)
    n = f.size
    active_idx = np.asarray(active_idx, dtype=int)
    if active_idx.size == 0:
        return None
    if active_idx.size > max_k:
        active_idx = active_idx[:max_k]
    K = active_idx.size

    row_parts = []
    col_parts =
    val_parts = []
    for r, k in enumerate(active_idx):
        s = max(0, int(k) - n + 1)
        e = min(n, int(k) + 1)
        if s >= e:
            continue
        cols = np.arange(s, e, dtype=np.int32)
        row_parts.append(np.full(cols.size, r, dtype=np.int32))
        col_parts.append(cols)
        val_parts.append(f[int(k) - cols].astype(np.float64))

    if not row_parts:
        return None

    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[n] = 1.0

    neg_one = scipy.sparse.csc_matrix(-np.ones((K, 1)), dtype=np.float64)
    A_ub = scipy.sparse.hstack((A, neg_one), format="csr")
    A_eq = np.ones((1, n + 1), dtype=np.float64)
    A_eq[0, n] = 0.0
    b_eq = np.ones(1, dtype=np.float64)
    b_ub = np.zeros(K, dtype=np.float64)
    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
            bounds=bounds, method="HiGHS",
            options={"disp": False, "time_limit": float(time_limit)}
        )
    except Exception:
        return None

    if res is None or not getattr(res, "success", False):
        return None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0.0), g, 0.0)
    gs = float(g.sum())
    if gs < 1e-15:
        return None
    return g / gs


def line_search_mix(a, g, cf, cg, fg, current_M, t):
    n = a.size
    base_alphas = [
        max(0.01, 1.0 / (t + 1.0)),
        max(0.02, 2.0 / (t + 2.0)),
        0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.50, 0.70, 0.90
    ]
    alphas = []
    seen = set()
    for al in base_alphas:
        al = float(al)
        if not math.isfinite(al) or al <= 0.0 or al > 1.0:
            continue
        key = int(round(al * 10000.0))
        if key in seen:
            continue
        seen.add(key)
        alphas.append(al)
    if len(alphas) == 0:
        alphas = [0.1]

    best_alpha = 0.0
    best_m = current_M
    for alpha in alphas:
        one = 1.0 - alpha
        val = (one * one) * cf + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
        m = float(np.max(val))
        if math.isfinite(m) and m < best_m:
            best_m = m
            best_alpha = alpha

    if best_alpha <= 0.0:
        return a.copy(), best_m, 0.0

    new_a = (1.0 - best_alpha) * a + best_alpha * g
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s < 1e-15:
        return a.copy(), best_m, 0.0
    new_a = new_a / s
    return new_a, best_m, best_alpha


def frank_wolfe_search(a0, budget, active_max=3000, game_iters=30,
                       exact_every=5, exact_time=10.0, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    n = a.size
    c = fft_conv(a)
    M = proxy_max(c, None)
    best_a = a.copy()
    best_M = M
    active = active_top(c, None, active_max)

    no_improve = 0
    t = 0
    while time.monotonic() - t0 < time_limit - 5 and no_improve < 60:
        t += 1
        time_left = time_limit - (time.monotonic() - t0)
        if time_left < 20:
            break

        g = None
        use_lp = (check_hihts() and exact_every > 0 and
                  (t % exact_every == 0) and time_left > 80 and
                  active.size <= max_k)
        max_k = 1500

        if use_lp:
            g = lp_direction(a, active, exact_time, max_k)

        if g is None:
            temp = 6.0 + 4.0 * (t % 5)
            g = game_direction(a, active, mw_iters=game_iters,
                               seed=2000 + t, temperature=temp)

        if g is None:
            break

        cg = fft_conv(g)
        fg = fft_conv2(a, g)
        new_a, new_M, alpha = line_search_mix(a, g, c, cg, fg, M, t)

        if new_M < M - 1e-16:
            if alpha > 0:
                one = 1.0 - alpha
                c = (one * one) * c + (2.0 * alpha * one) * fg + (alpha * alpha) * cg
                c = np.where(np.isfinite(c) & (c > 0.0), c, 0.0)
            a = new_a
            M = new_M
            no_improve = 0
            if M < best_M - 1e-16:
                best_M = M
                best_a = a.copy()
            active = active_top(c, None, active_max)
            if verbose and (t == 1 or t % 5 == 0):
                print("    FW t=" + str(t) + " proxy=" + repr(2.0 * n * best_M) +
                      " alpha=" + repr(round(alpha, 4)) + " active=" + str(active.size))
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                rng = np.random.default_rng(4242 + t)
                scale = 0.02 + 0.02 * (no_improve // 10)
                pert = rng.normal(0.0, scale, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-14)
                a = a / float(a.sum())
                c = fft_conv(a)
                M = proxy_max(c, None)
                active = active_top(c, None, active_max)
                no_improve = 0

    return best_a


def apply_single_move(c, a, s, d, tmove):
    c_new = c.copy()
    n = a.size
    c_new[d:d + n] += 2.0 * tmove * a
    c_new[s:s + n] -= 2.0 * tmove * a

    t2 = tmove * tmove
    if 2 * d < c_new.size:
        c_new[2 * d] += t2
    if 2 * s < c_new.size:
        c_new[2 * s] += t2
    if s + d < c_new.size:
        c_new[s + d] -= 2.0 * t2
    c_new = np.where(np.isfinite(c_new), c_new, 0.0)
    return c_new


def coordinate_polish(a0, budget, verbose=True):
    t0 = time.monotonic()
    time_limit = max(1.0, float(budget))
    a = norm_sum1(a0)
    c = fft_conv(a)
    M = proxy_max(c, None)
    best_a = a.copy()
    best_M = M
    n = a.size
    stall = 0
    rnd = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit - 3 and stall < 200:
        rnd += 1
        if n <= 1:
            break

        top_k = min(60, 2 * n - 1)
        top_idx = active_top(c, None, top_k)
        if top_idx.size == 0:
            break

        p_top = np.zeros(2 * n - 1, dtype=np.float64)
        p_top[top_idx] = 1.0 / float(top_idx.size)
        pressure = cross_convolve_p_a(p_top, a, 2.0)
        pressure = np.where(np.isfinite(pressure) & (pressure > 0.0), pressure, 0.0)

        src_count = min(12, n)
        dst_count = min(16, n)

        src_score = pressure + 0.1 * float(n) * a
        src_idx = np.argpartition(-src_score, min(src_count - 1, n - 1))[:src_count]

        dst_score = -pressure + 1e-6 * a
        dst_idx = np.argpartition(dst_score, min(dst_count - 1, n - 1))[:dst_count]

        improved_any = False
        best_cand = None
        best_cand_M = M

        for srr in src_idx:
            s = int(srr)
            if a[s] < 1e-14:
                continue
            for drr in dst_idx:
                d = int(drr)
                if d == s:
                    continue
                for frac in (0.003, 0.007, 0.015, 0.03, 0.06):
                    tmove = a[s] * frac
                    if tmove <= 1e-18:
                        continue
                    tmove = min(tmove, a[s] * 0.95)
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new, None)
                    if m_new < best_cand_M - 1e-18:
                        best_cand = (m_new, s, d, tmove)
                        best_cand_M = m_new
                        improved_any = True

        if not improved_any:
            for _ in range(40):
                s = int(rng.integers(n))
                d = int(rng.integers(n))
                if s == d or a[s] < 1e-14:
                    continue
                frac = float(rng.uniform(0.002, 0.04))
                tmove = a[s] * frac
                tmove = min(tmove, a[s] * 0.95)
                if tmove <= 1e-18:
                    continue
                c_new = apply_single_move(c, a, s, d, tmove)
                m_new = proxy_max(c_new, None)
                if m_new < best_cand_M - 1e-18:
                    best_cand = (m_new, s, d, tmove)
                    best_cand_M = m_new
                    improved_any = True

        if best_cand is not None:
            m_new, s, d, tmove = best_cand
            a[s] -= tmove
            a[d] += tmove
            a = np.maximum(a, 0.0)
            ss = float(a.sum())
            if ss < 1e-15:
                break
            a = a / ss
            c = fft_conv(a)
            M = proxy_max(c, None)
            if M < best_M - 1e-16:
                best_M = M
                best_a = a.copy()
            stall = 0
            if verbose and (rnd <= 5 or rnd % 30 == 0):
                print("    Coord rnd=" + str(rnd) + " proxy=" + repr(2.0 * n * best_M))
        else:
            stall += 1
            if stall % 15 == 14:
                for _ in range(80):
                    s = int(rng.integers(n))
                    d = int(rng.integers(n))
                    if s == d or a[s] < 1e-14:
                        continue
                    frac = float(rng.uniform(0.001, 0.02))
                    tmove = a[s] * frac
                    if tmove <= 1e-18:
                        continue
                    c_new = apply_single_move(c, a, s, d, tmove)
                    m_new = proxy_max(c_new, None)
                    if m_new < M - 1e-18:
                        a[s] -= tmove
                        a[d] += tmove
                        a = np.maximum(a, 0.0)
                        a = a / float(a.sum())
                        c = fft_conv(a)
                        M = proxy_max(c, None)
                        stall = 0
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        break
                else:
                    a = best_a + rng.normal(0.0, 0.01, n) * best_a
                    a = np.maximum(a, 1e-14)
                    a = a / float(a.sum())
                    c = fft_conv(a)
                    M = proxy_max(c, None)
                    stall = 0

    return best_a


def default_sequence(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    return norm_sum1(np.sin(np.pi * x) ** 2)


def skewed_sequence(n=5000, alpha=1.5):
    x = np.linspace(0.0, 1.0, int(n))
    return norm_sum1(x ** alpha)


def bimodal_sequence(n=5000):
    x = np.linspace(0.0, 1.0, int(n))
    a = np.exp(-50.0 * (x - 0.3) ** 2) + np.exp(-50.0 * (x - 0.7) ** 2)
    return norm_sum1(a)


def estimate_cost(a, beta=30.0):
    if not SCIPY_OK:
        return 0.2
    v, ref = make_v(a)
    try:
        t0 = time.monotonic()
        objgrad_v(v, beta, None, ref)
        dt = time.monotonic() - t0
        if dt <= 0.0 or not math.isfinite(dt):
            dt = 0.05
        return float(dt)
    except Exception:
        return 0.2


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    try:
        total_budget = max(60.0, min(float(budget_s) - 60.0, 1740.0))
    except Exception:
        total_budget = 1740.0

    print("Start: budget=" + repr(total_budget))

    seq_best = None
    try:
        parent_list = height_sequence_1
        if parent_list is not None:
            p = np.asarray(parent_list, dtype=np.float64)
            p = np.where(np.isfinite(p), p, 0.0)
            np.clip(p, 0.0, 1000.0, out=p)
            if p.ndim != 1:
                p = p.ravel()
            if p.size > 0 and float(p.sum()) > 1e-10:
                seq_best = norm_sum1(p)
    except Exception:
        pass

    if seq_best is None:
        seq_best = default_sequence(5000)

    best_score = safe_eval(seq_best)
    if not math.isfinite(best_score):
        best_score = 2.0 * seq_best.size * proxy_max(fft_conv(seq_best), None)
    print("Initial: n=" + str(seq_best.size) + " score=" + repr(best_score))

    if seq_best.size > 8000:
        work_seq = interpolate_sequence(seq_best, 7000)
    else:
        work_seq = seq_best.copy()

    seq_best, best_score = try_update(seq_best, best_score, work_seq, "init", t_start)
    if seq_best.size > 8000:
        work_seq = interpolate_sequence(seq_best, 7000)
    else:
        work_seq = seq_best.copy()

    work_score = safe_eval(work_seq)
    lb_cost = estimate_cost(work_seq, beta=30.0)
    print("Work: n=" + str(work_seq.size) + " score=" + repr(work_score) +
          " L-BFGS cost=" + repr(round(lb_cost, 4)))

    def refresh_work():
        nonlocal work_seq
        if seq_best.size <= 8000:
            work_seq = seq_best.copy()
        else:
            work_seq = interpolate_sequence(seq_best, 7000)

    # Phase 1: L-BFGS smoothed optimization
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if SCIPY_OK and time_left > 400:
        phase_budget = min(time_left * 0.38, 660.0)
        print("\n=== Phase 1: L-BFGS (budget=" + repr(round(phase_budget, 1)) + "s) ===")
        a_lb = lbfegs_phase(work_seq, phase_budget, lb_cost)
        seq_best, best_score = try_update(seq_best, best_score, a_lb, "lbfegs", t_start)
        refresh_work()

    # Phase 2: Frank-Wolfe
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 350:
        phase_budget = min(time_left * 0.30, 520.0)
        print("\n=== Phase 2: Frank-Wolfe (budget=" + repr(round(phase_budget, 1)) + "s) ===")
        a_fw = frank_wolfe_search(work_seq, phase_budget,
                                  active_max=3000, game_iters=30,
                                  exact_every=4, exact_time=8.0, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_fw, "fw", t_start)
        refresh_work()

    # Phase 3: Coordinate polish
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 250:
        phase_budget = min(time_left * 0.25, 430.0)
        print("\n=== Phase 3: Coordinate polish (budget=" + repr(round(phase_budget, 1)) + "s) ===")
        a_cp = coordinate_polish(work_seq, phase_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_cp, "coord", t_start)
        refresh_work()

    # Phase 4: Restart with different strategy
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 250:
        phase_budget = min(time_left * 0.22, 380.0)
        print("\n=== Phase 4: Restart + FW (budget=" + repr(round(phase_budget, 1)) + "s) ===")

        rng = np.random.default_rng(999)
        a_restart = seq_best.copy().astype(np.float64)
        noise_scale = 0.03
        a_restart += rng.normal(0.0, noise_scale, a_restart.size) * a_restart
        a_restart = np.maximum(a_restart, 1e-14)
        a_restart = a_restart / float(a_restart.sum())

        sub_budget = phase_budget * 0.6
        a_fw2 = frank_wolfe_search(a_restart, sub_budget,
                                   active_max=2500, game_iters=25,
                                   exact_every=5, exact_time=6.0, verbose=False)
        seq_best, best_score = try_update(seq_best, best_score, a_fw2, "restart_fw", t_start)
        refresh_work()

        time_left = total_budget - (time.monotonic() - t_start)
        if time_left > 80:
            a_cp2 = coordinate_polish(work_seq, min(time_left * 0.5, 100.0), verbose=False)
            seq_best, best_score = try_update(seq_best, best_score, a_cp2, "restart_coord", t_start)
            refresh_work()

    # Phase 5: Length probes
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 300:
        print("\n=== Phase 5: Length probes ===")
        base_len = seq_best.size
        candidates = [base_len - 500, base_len + 500, base_len - 1000, base_len + 1000,
                      4000, 6000, 7000]
        tried = set()
        for target in candidates:
            time_left = total_budget - (time.monotonic() - t_start)
            if time_left < 120:
                break
            target = int(max(2000, min(8000, target)))
            if target in tried:
                continue
            tried.add(target)
            print("  Trying n=" + str(target))

            a0 = interpolate_sequence(seq_best, target)

            if SCIPY_OK and time_left > 180:
                per = int(max(60, min(200, int((time_left - 60) / max(4.0 * lb_cost, 0.1)))))
                a_new, _, _ = run_lbfegs(a0, 80.0, None, maxfun=per, verbose=False)
                seq_best, best_score = try_update(seq_best, best_score, a_new, "len_lb_" + str(target), t_start)
                refresh_work()

            time_left = total_budget - (time.monotonic() - t_start)
            if time_left > 80:
                a_base = seq_best.copy() if seq_best.size == target else a0
                a_fw3 = frank_wolfe_search(a_base, min(time_left * 0.5, 60.0),
                                           active_max=2000, game_iters=20,
                                           exact_every=5, exact_time=5.0, verbose=False)
                seq_best, best_score = try_update(seq_best, best_score, a_fw3, "len_fw_" + str(target), t_start)
                refresh_work()

    # Phase 6: Final intensive polish
    t0 = time.monotonic()
    time_left = total_budget - (t0 - t_start)
    if time_left > 60:
        phase_budget = min(time_left * 0.7, 150.0)
        print("\n=== Phase 6: Final polish (budget=" + repr(round(phase_budget, 1)) + "s) ===")
        a_cp3 = coordinate_polish(work_seq, phase_budget, verbose=True)
        seq_best, best_score = try_update(seq_best, best_score, a_cp3, "final", t_start)
        refresh_work()

    # Final L-BFGS if time remains
    time_left = total_budget - (time.monotonic() - t_start)
    if SCIPY_OK and time_left > 60:
        sub = min(time_left * 0.6, 80.0)
        per = int(max(40, min(150, int(sub / max(3.0 * lb_cost, 0.1)))))
        a_final, _, _ = run_lbfegs(work_seq, 200.0, None, maxfun=per, verbose=False)
        seq_best, best_score = try_update(seq_best, best_score, a_final, "final_lb", t_start)

    # Final validation
    final_val = safe_eval(seq_best)
    if math.isfinite(final_val):
        if final_val < best_score:
            best_score = final_val
    else:
        seq_best = default_sequence(5000)
        best_score = safe_eval(seq_best)

    arr = np.asarray(seq_best, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if arr.size == 0 or s < 0.01:
        arr = default_sequence(5000)
    else:
        arr = arr / s

    elapsed = time.monotonic() - t_start
    print("\nFINAL: score=" + repr(best_score) + " n=" + str(arr.size) +
          " elapsed=" + repr(round(elapsed, 2)) + "s")

    return [float(x) for x in arr]
