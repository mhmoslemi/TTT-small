```python
"""
Search for low-convolution step functions.

The objective is
    2*n*max(conv(a,a)) / sum(a)^2.
Since the formula is scale invariant, internal optimization normalizes sum(a)=1
and minimizes 2*n*max(conv(a,a)).

The main local search uses:
  - fast FFT convolution,
  - a Hankel-gradient induced by weighted convolution coefficients,
  - mirror-descent and Frank-Wolfe line searches,
  - gradient-guided pairwise mass swaps,
  - periodic restarts and multi-scale starts / length interpolation.
"""

import numpy as np
import time

ACCEPT_TOL = 1e-13
MAX_EVAL_LENGTH = 14000


def next_power_of_two(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_seq(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.array([1.0], dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(arr.shape, dtype=np.float64) / float(arr.shape[0])
    return arr / s


def fast_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    L = 2 * n - 1
    fs = next_power_of_two(L)
    A = np.fft.rfft(a, fs)
    c = np.fft.irfft(A * A, fs)[:L]
    return c


def internal_score(a):
    a = normalize_seq(a)
    n = int(a.size)
    if n == 0:
        return float("inf")
    c = fast_convolution(a)
    return 2.0 * float(n) * float(c.max())


def hankel_gradient(a, w):
    """
    For coefficients C_t = conv(a,a)_t and weights w_t,
    returns H_w a where (H_w a)_i = sum_j w_{i+j} a_j.
    This is one half of the gradient of sum_t w_t C_t.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    if len(w) != L:
        ww = np.zeros(L, dtype=np.float64)
        m = min(L, len(w))
        ww[:m] = w[:m]
        w = ww

    if n == 1:
        return np.array([float(w[0] * a[0])], dtype=np.float64)

    size = L + n - 1
    fs = next_power_of_two(size)
    A = np.fft.rfft(a, fs)
    Wr = np.fft.rfft(w[::-1], fs)
    conv = np.fft.irfft(A * Wr, fs)[:size]
    return conv[2 * n - 2 : n - 2 : -1].copy()


def weights_power(C, alpha):
    C = np.asarray(C, dtype=np.float64).ravel()
    L = int(C.size)
    if L == 0:
        return np.array([], dtype=np.float64)
    Cp = np.maximum(C, 0.0)
    cmean = float(Cp.mean())
    if not np.isfinite(cmean) or cmean <= 1e-300:
        return np.ones(L, dtype=np.float64) / float(L)
    z = Cp / cmean
    z = np.maximum(z, 0.0)
    w = z ** float(alpha)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(L, dtype=np.float64) / float(L)
    return w / s


def weights_top(C, k, beta):
    C = np.asarray(C, dtype=np.float64).ravel()
    L = int(C.size)
    if L == 0:
        return np.array([], dtype=np.float64)
    k = int(min(k, L))
    if k <= 0:
        return np.zeros(L, dtype=np.float64)

    if k >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(C, L - k)[-k:]

    w = np.zeros(L, dtype=np.float64)
    top = np.maximum(C[idx], 0.0)
    cmean = float(C.mean())
    if not np.isfinite(cmean) or cmean <= 1e-300:
        w[idx] = 1.0 / float(k)
        return w

    z = top / cmean
    zmax = float(z.max()) if z.size else 1.0
    zmin = float(z.min()) if z.size else 1.0
    spread = max(zmax - zmin, 0.05)
    temp = max(float(beta) * spread, 1e-8)
    wtop = np.exp((z - zmax) / temp)
    s = float(wtop.sum())
    if not np.isfinite(s) or s <= 1e-300:
        wtop = np.ones(len(wtop), dtype=np.float64) / float(len(wtop))
    else:
        wtop = wtop / s
    w[idx] = wtop
    return w


def eval_move(C, a, s, d, delta):
    n = int(a.size)
    c = C.copy()
    td = 2.0 * float(delta)
    c[d:d + n] += td * a
    c[s:s + n] -= td * a
    d2 = float(delta) * float(delta)
    c[2 * int(d)] += d2
    c[int(s) + int(d)] -= 2.0 * d2
    c[2 * int(s)] += d2
    return float(c.max())


def apply_move(C, a, s, d, delta):
    n = int(a.size)
    s = int(s)
    d = int(d)
    delta = float(delta)
    td = 2.0 * delta
    d2 = delta * delta

    C[d:d + n] += td * a
    C[s:s + n] -= td * a
    C[2 * d] += d2
    C[s + d] -= 2.0 * d2
    C[2 * s] += d2

    a[s] = a[s] - delta
    a[d] = a[d] + delta
    if a[s] < 0.0:
        a[s] = 0.0
    if a[d] < 0.0:
        a[d] = 0.0


def find_best_move(a, C, h, current_M):
    n = int(a.size)
    if n <= 1:
        return None, current_M

    D = min(16, n)
    if D >= n:
        dst_idx = np.arange(n)
    else:
        dst_idx = np.argpartition(h, D)[:D]

    score = h * a
    if D >= n:
        src_idx = np.argsort(score)[-min(16, n):]
    else:
        src_idx = np.argpartition(score, n - D)[n - D:]

    if D >= n:
        htop = np.argsort(h)[-min(16, n):]
    else:
        htop = np.argpartition(h, n - D)[n - D:]

    src_set = []
    for idx in src_idx:
        ii = int(idx)
        if ii not in src_set:
            src_set.append(ii)
    for idx in htop:
        ii = int(idx)
        if a[ii] > 1e-10 and ii not in src_set:
            src_set.append(ii)
    if len(src_set) > 18:
        src_set = src_set[:18]

    dst_list = [int(x) for x in dst_idx[:10]]
    pairs = []
    for dval in dst_list:
        for sval in src_set:
            if sval == dval:
                continue
            mass = float(a[sval])
            if mass <= 1e-14:
                continue
            gap = float(h[sval]) - float(h[dval])
            if gap > 0.0:
                pairs.append((gap * mass, sval, dval))

    if not pairs:
        return None, current_M

    pairs.sort(reverse=True)

    seen = set()
    top_pairs = []
    for _, sval, dval in pairs:
        key = (int(sval), int(dval))
        if key in seen:
            continue
        seen.add(key)
        top_pairs.append(key)
        if len(top_pairs) >= 5:
            break

    fracs = (0.005, 0.02, 0.05, 0.1, 0.25, 0.5)
    best_m = float(current_M)
    best_move = None

    for s, d in top_pairs:
        base_mass = float(a[s])
        for frac in fracs:
            delta = base_mass * float(frac)
            if delta <= 1e-15:
                continue
            mm = eval_move(C, a, s, d, delta)
            if mm < best_m:
                best_m = mm
                best_move = (s, d, delta)

    return best_move, best_m


def mirror_line_search(a, h):
    a = np.asarray(a, dtype=np.float64).ravel()
    h = np.asarray(h, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 1:
        return None, float("inf")

    h_shift = h - float(np.mean(h))
    h_range = float(np.max(h_shift) - np.min(h_shift))
    if not np.isfinite(h_range) or h_range <= 1e-15:
        return None, float("inf")

    base = 1.0 / h_range
    loga = np.log(np.maximum(a, 1e-300))
    best_y = None
    best_m = float("inf")
    best_mult = 0.0
    mults = (0.01, 0.05, 0.2, 1.0)

    for mult in mults:
        eta = base * float(mult)
        if eta > 1e4:
            eta = 1e4
        y = np.exp(loga - eta * h_shift)
        y = np.maximum(y, 0.0)
        s = float(y.sum())
        if not np.isfinite(s) or s <= 1e-15:
            continue
        y = y / s
        c = fast_convolution(y)
        mm = float(c.max())
        if mm < best_m:
            best_m = mm
            best_y = y.copy()
            best_mult = mult

    if best_y is not None and best_mult == mults[-1]:
        for mult in (2.0, 5.0):
            eta = base * float(mult)
            if eta > 5e4:
                eta = 5e4
            y = np.exp(loga - eta * h_shift)
            y = np.maximum(y, 0.0)
            s = float(y.sum())
            if not np.isfinite(s) or s <= 1e-15:
                continue
            y = y / s
            c = fast_convolution(y)
            mm = float(c.max())
            if mm < best_m:
                best_m = mm
                best_y = y.copy()

    if best_y is None:
        return None, float("inf")
    return best_y, best_m


def fw_line_search(a, h, current_M):
    a = np.asarray(a, dtype=np.float64).ravel()
    h = np.asarray(h, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 1:
        return None, current_M

    d = int(np.argmin(h))
    best_y = None
    best_m = float(current_M)

    alphas = (0.01, 0.05, 0.2, 0.5, 0.9)
    for alpha in alphas:
        y = (1.0 - float(alpha)) * a
        y[d] += float(alpha)
        s = float(y.sum())
        if not np.isfinite(s) or s <= 1e-15:
            continue
        y = y / s
        c = fast_convolution(y)
        mm = float(c.max())
        if mm < best_m:
            best_m = mm
            best_y = y.copy()

    if best_y is None:
        return None, current_M
    return best_y, best_m


def perturb_log(a, rng, scale):
    a = normalize_seq(a)
    loga = np.log(np.maximum(a, 1e-300))
    noise = rng.standard_normal(len(a))
    noise -= noise.mean()
    loga = loga + float(scale) * noise
    loga -= loga.max()
    y = np.exp(loga)
    return normalize_seq(y)


def random_smooth_start(n, seed):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    vals = np.ones(n, dtype=np.float64)
    for _ in range(6):
        f = int(rng.integers(1, 30))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.05, 0.35))
        vals += amp * np.cos(2.0 * np.pi * float(f) * x + ph)
    vals = np.maximum(vals, 1e-3)
    return normalize_seq(vals)


def lognormal_start(n, seed):
    rng = np.random.default_rng(seed)
    vals = np.exp(rng.normal(0.0, 0.55, size=n))
    return normalize_seq(vals)


def blend_seq(a, b, alpha):
    a = normalize_seq(a)
    b = normalize_seq(b)
    if len(a) != len(b):
        b = interpolate_seq(b, len(a))
    return normalize_seq(float(alpha) * a + (1.0 - float(alpha)) * b)


def interpolate_seq(a, new_n):
    a = normalize_seq(a)
    old_n = int(a.size)
    new_n = int(new_n)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    b = np.interp(new_x, old_x, a)
    return normalize_seq(b)


def local_search(a0, time_limit, seed=123, verbose=True, tag="LS"):
    a = normalize_seq(a0)
    n = int(a.size)
    if n < 2:
        c = fast_convolution(a)
        return a, 2.0 * float(n) * float(c.max())

    C = fast_convolution(a)
    best_a = a.copy()
    best_M = float(C.max())
    t0 = time.monotonic()
    time_limit = float(time_limit)
    if time_limit <= 0.0:
        return best_a, 2.0 * float(n) * best_M

    rng = np.random.default_rng(seed)
    stall = 0
    it = 0
    alpha = 4.0
    accept_count = 0

    if verbose:
        print("    {} start score={:.7f} n={}".format(tag, 2.0 * n * best_M, n))

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = float(C.max())

        if M < best_M - ACCEPT_TOL:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10000:
                scale = 0.02 + 0.01 * float(stall // 10000)
                scale = min(scale, 0.15)
                a = perturb_log(best_a, rng, scale)
                C = fast_convolution(a)
                stall = 1200
                if verbose:
                    print("    {} restart score={:.7f}".format(tag, 2.0 * n * float(C.max())))
                continue

        if it % 1000 == 0:
            if stall > 3000:
                alpha = min(8.0, alpha + 0.3)
            elif stall < 500:
                alpha = max(3.0, alpha - 0.2)

        if stall > 2000 and (it % 5 == 0):
            w = weights_top(C, 256, 0.1)
        else:
            w = weights_power(C, alpha)

        h = hankel_gradient(a, w)

        # Pairwise gradient-guided swap.
        move, m_move = find_best_move(a, C, h, M)
        if move is not None and m_move < M - ACCEPT_TOL:
            s, d, delta = move
            if delta > a[s]:
                delta = a[s]
            if delta > 1e-15:
                apply_move(C, a, s, d, delta)
                accept_count += 1
                stall = 0
                if accept_count % 80 == 0:
                    a = normalize_seq(a)
                    C = fast_convolution(a)
                continue

        # Mirror descent line search.
        if it % 4 == 0 or stall > 300:
            new_a, new_M = mirror_line_search(a, h)
            if new_a is not None and new_M < M - ACCEPT_TOL:
                a = new_a
                C = fast_convolution(a)
                accept_count += 1
                stall = 0
                if accept_count % 50 == 0:
                    a = normalize_seq(a)
                    C = fast_convolution(a)
                continue

        # Frank-Wolfe line search.
        if it % 12 == 0 or stall > 800:
            new_a, new_M = fw_line_search(a, h, M)
            if new_a is not None and new_M < M - ACCEPT_TOL:
                a = new_a
                C = fast_convolution(a)
                accept_count += 1
                stall = 0
                continue

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    {} iter={:6d} t={:6.1f}s score={:.7f} best={:.7f} stall={} alpha={:.2f}".format(
                tag, it, elapsed, 2.0 * n * M, 2.0 * n * best_M, stall, alpha))

    best_a = normalize_seq(best_a)
    bc = fast_convolution(best_a)
    best_score = 2.0 * float(n) * float(bc.max())
    if verbose:
        print("    {} done score={:.7f}".format(tag, best_score))
    return best_a, best_score


def load_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        return None
    if p.ndim != 1:
        p = p.ravel()
    if p.size < 100:
        return None
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return p / s


def safe_eval(a, max_len=MAX_EVAL_LENGTH):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        if arr.size > max_len:
            return float("inf")
        lst = [float(x) for x in arr]
        val = float(evaluate_sequence(lst))
        if np.isfinite(val):
            return val
    except Exception:
        pass
    return float("inf")


def update_best_with_actual(best_seq, best_internal, best_actual, best_actual_seq,
                            a, internal_score_val, actual_score_val):
    new_best_seq = best_seq
    new_best_internal = best_internal
    new_best_actual = best_actual
    new_best_actual_seq = best_actual_seq

    if a is not None:
        if np.isfinite(internal_score_val) and internal_score_val < new_best_internal - 1e-12:
            new_best_internal = float(internal_score_val)
            new_best_seq = np.array(a, dtype=np.float64).copy()
        if np.isfinite(actual_score_val) and actual_score_val < new_best_actual - 1e-12:
            new_best_actual = float(actual_score_val)
            new_best_actual_seq = np.array(a, dtype=np.float64).copy()

    return new_best_seq, new_best_internal, new_best_actual, new_best_actual_seq


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    budget = float(budget_s)
    total_budget = min(budget - 60.0, 1750.0)
    if total_budget < 120.0:
        total_budget = max(60.0, budget - 10.0)
    deadline = t_start + total_budget

    parent = load_parent()
    if parent is None:
        n0 = 10000
        parent = random_smooth_start(n0, 1)
        parent = blend_seq(parent, lognormal_start(n0, 2), 0.7)
    parent = normalize_seq(parent)
    if parent.size > 16000:
        parent = interpolate_seq(parent, 16000)

    best_seq = parent.copy()
    best_internal = internal_score(best_seq)
    best_actual = float("inf")
    best_actual_seq = None
    actual_calls = 0

    if parent.size <= MAX_EVAL_LENGTH and time.monotonic() < deadline - 60.0:
        val = safe_eval(parent)
        actual_calls += 1
        if np.isfinite(val):
            best_actual = val
            best_actual_seq = parent.copy()
        print("Parent: n={} internal={:.7f} actual={}".format(
            len(parent), best_internal, repr(val)))
    else:
        print("Parent: n={} internal={:.7f} actual=not evaluated".format(
            len(parent), best_internal))

    # Main long optimization from the parent / best known start.
    remaining = deadline - time.monotonic()
    t_main = min(remaining * 0.62, 1300.0)
    if t_main > 60.0:
        print("\n=== Phase 1: main local search, n={} budget={:.1f}s ===".format(len(best_seq), t_main))
        out_a, out_score = local_search(best_seq, t_main, seed=101, verbose=True, tag="P1-main")
        actual_val = float("inf")
        if out_a is not None and out_a.size <= MAX_EVAL_LENGTH and actual_calls < 4 and time.monotonic() < deadline - 60.0:
            actual_val = safe_eval(out_a)
            actual_calls += 1
            print("Phase 1 actual={}".format(repr(actual_val)))
        best_seq, best_internal, best_actual, best_actual_seq = update_best_with_actual(
            best_seq, best_internal, best_actual, best_actual_seq, out_a, out_score, actual_val)

    # Secondary diversified search: slightly longer if possible, or blended random start.
    remaining = deadline - time.monotonic()
    if remaining > 260.0:
        if best_seq.size < 15000:
            n_alt = min(16000, int(best_seq.size * 1.12) + 600)
            alt_start = interpolate_seq(best_seq, n_alt)
            alt_start = blend_seq(alt_start, random_smooth_start(n_alt, 7), 0.8)
        else:
            alt_start = blend_seq(best_seq, random_smooth_start(best_seq.size, 7), 0.85)

        t_alt = min(remaining * 0.25, 360.0)
        if t_alt > 60.0:
            print("\n=== Phase 2: alternated start, n={} budget={:.1f}s ===".format(len(alt_start), t_alt))
            out_a2, out_score2 = local_search(alt_start, t_alt, seed=7, verbose=True, tag="P2-alt")
            actual_val2 = float("inf")
            if out_a2 is not None and out_a2.size <= MAX_EVAL_LENGTH and actual_calls < 6 and time.monotonic() < deadline - 60.0:
                actual_val2 = safe_eval(out_a2)
                actual_calls += 1
                print("Phase 2 actual={}".format(repr(actual_val2)))
            best_seq, best_internal, best_actual, best_actual_seq = update_best_with_actual(
                best_seq, best_internal, best_actual, best_actual_seq, out_a2, out_score2, actual_val2)

    # Final polish on the current best vector.
    remaining = deadline - time.monotonic()
    if remaining > 70.0:
        t_final = remaining - 25.0
        if t_final > 30.0:
            print("\n=== Phase 3: final polish, n={} budget={:.1f}s ===".format(len(best_seq), t_final))
            out_a3, out_score3 = local_search(best_seq, t_final, seed=99, verbose=True, tag="P3-final")
            actual_val3 = float("inf")
            if out_a3 is not None and out_a3.size <= MAX_EVAL_LENGTH and actual_calls < 8 and time.monotonic() < deadline - 25.0:
                actual_val3 = safe_eval(out_a3)
                actual_calls += 1
                print("Phase 3 actual={}".format(repr(actual_val3)))
            best_seq, best_internal, best_actual, best_actual_seq = update_best_with_actual(
                best_seq, best_internal, best_actual, best_actual_seq, out_a3, out_score3, actual_val3)

    # One last actual check of the internally best sequence, if allowed.
    if best_seq is not None and best_seq.size <= MAX_EVAL_LENGTH and actual_calls < 10 and time.monotonic() < deadline - 15.0:
        actual_final = safe_eval(best_seq)
        actual_calls += 1
        best_seq, best_internal, best_actual, best_actual_seq = update_best_with_actual(
            best_seq, best_internal, best_actual, best_actual_seq, best_seq, best_internal, actual_final)
        print("Final best_seq actual={}".format(repr(actual_final)))

    # Prefer an observed actual best if one exists; otherwise internally best.
    if best_actual_seq is not None and np.isfinite(best_actual):
        final_seq = best_actual_seq
    else:
        final_seq = best_seq

    final_seq = normalize_seq(final_seq)
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    final_seq = np.nan_to_num(final_seq, nan=0.0, posinf=0.0, neginf=0.0)

    elapsed = time.monotonic() - t_start
    final_internal = internal_score(final_seq)
    print("\nFINAL: n={} internal={:.7f} actual_best={:.7f} time={:.1f}s".format(
        len(final_seq), final_internal, best_actual if np.isfinite(best_actual) else float("nan"), elapsed))

    return [float(x) for x in final_seq]
```
