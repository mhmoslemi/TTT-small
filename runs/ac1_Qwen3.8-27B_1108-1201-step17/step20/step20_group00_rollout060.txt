def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def obj_val(a):
    """Compute objective value (normalized to sum=1)."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def eval_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def build_lp_matrix(a, lags, n):
    """Build constraint matrix for LP: row k has a[k-i] at position i."""
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs


def lp_fw_fast(a0, time_limit, solver, init_K=300, max_K=2000, verbose=True):
    """
    LP-based Frank-Wolfe with:
    - Warm starts
    - Small active set (top peaks only)
    - Bisection line search (faster than golden section for this problem)
    """
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    # Initial active set: top init_K peaks
    ac = fast_autocorr(y)
    active_set = list(np.argsort(ac)[::-1][:init_K])
    no_improve = 0
    
    # Warm start variables
    prev_g = None
    iter_count = 0

    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 800:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update active set: keep top max_K peaks
        # Also ensure we include the current peak
        top_peaks = np.argsort(ac)[::-1][:max_K]
        active_set = list(top_peaks)
        
        # Also add any previously active that are still high
        for k in active_set:
            if k not in top_peaks and ac[k] > M_curr * 0.9:
                pass  # already in top_peaks since top_peaks is top max_K

        clist = sorted(set(int(k) for k in active_set))
        K = len(clist)
        if K == 0:
            break

        # Solve LP
        coeffs = build_lp_matrix(y, clist, n)
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        
        # Warm start
        if prev_g is not None:
            warm_start = {'x': prev_g, 'x0': prev_g}
        else:
            warm_start = {}
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False, warm_start=True, **warm_start)
            if prob.status in ('optimal', 'optimal_inaccurate') and g_var.value is not None:
                g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
                g_sum = g_val.sum()
                if g_sum > 1e-15:
                    g_dir = g_val / g_sum
                    prev_g = g_val.copy()
                else:
                    g_dir = y.copy()
            else:
                g_dir = y.copy()
        except Exception:
            # Fallback: use gradient direction
            g_dir = multiplicative_step(y, ac, top_k=50, temp=20.0)
            prev_g = None

        # Line search: bisection (simple and fast)
        # We want to find best alpha in [0, 1] minimizing obj((1-alpha)*y + alpha*g_dir)
        lo, hi = 0.0, 1.0
        best_alpha = 0.5
        best_f = obj_val(normalize(np.maximum((1-0.5)*y + 0.5*g_dir, 0)))
        
        # Quick bisection
        for _ in range(8):
            mid = (lo + hi) / 2.0
            trial = np.maximum((1.0 - mid) * y + mid * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                lo = mid
                continue
            trial = trial / ts
            f_mid = fast_autocorr(trial).max()
            if f_mid < best_f:
                best_f = f_mid
                best_alpha = mid
                hi = mid
            else:
                lo = mid
        
        # Also check alpha near 1
        for alpha in [0.9, 0.95, 1.0]:
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts > 1e-15:
                trial = trial / ts
                f_a = fast_autocorr(trial).max()
                if f_a < best_f:
                    best_f = f_a
                    best_alpha = alpha

        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()

        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW iter=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2*n*best_m) + " |S|=" + str(K))

    if verbose:
        print("    LP-FW done (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2*n*best_m) + " iters=" + str(iter_count))
    return best_a, 2.0 * n * best_m


def multiplicative_step(a, ac, top_k=50, temp=20.0):
    """Compute one multiplicative gradient step direction."""
    n = len(a)
    k_max = min(top_k, len(ac))
    idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[idx]
    w = np.exp(temp * (top_vals - top_vals[0]))
    w = w / w.sum()
    
    grad = np.zeros(n, dtype=np.float64)
    for wi, k in zip(w, idx):
        ki = int(k)
        i_s = max(0, ki - n + 1)
        i_e = min(n, ki + 1)
        idx_arr = np.arange(i_s, i_e)
        grad[idx_arr] += 2.0 * wi * a[ki - idx_arr]
    
    # Multiplicative update
    log_a = np.log(np.maximum(a, 1e-30)) - grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    return a_new


def grad_mult_fast(a0, time_limit, top_k=50, temp=30.0, c0=2.0, verbose=True):
    """Fast multiplicative gradient descent."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    a_prev = a.copy()
    stall = 0
    rng = np.random.default_rng(42)

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break

        beta_t = (t - 1.0) / (t + 3.0)
        y = a + beta_t * (a - a_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        a_new = multiplicative_step(y, ac, top_k=top_k, temp=temp * max(0.3, 1.0 - t/30000.0))

        # Damped step
        alpha = c0 / (np.sqrt(max(t, 1)) + 1.0)
        a_new = (1 - alpha) * y + alpha * a_new
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        a_prev = a
        a = a_new

        val = obj_val(a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000 and stall % 2000 == 0:
            # Restart with noise
            noise = 0.01 * rng.standard_normal(n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-12)
            a_r = a_r / a_r.sum()
            a = a_r
            a_prev = a_r.copy()
            stall = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    GradMult t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val))

    return best_a, best_val

def diverse_init(n, parent_seq=None):
    """Generate diverse initial candidates."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    best_a = None
    best_val = float('inf')
    
    candidates = []
    
    # Symmetric shapes
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    candidates.append(normalize(np.sin(np.pi * x) ** 4))
    candidates.append(normalize(np.cos(np.pi * (x - 0.5)) ** 2))
    
    # Asymmetric shapes (NEW!)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize((x + 0.1) ** p))
        candidates.append(normalize((1 - x + 0.1) ** p))
    
    # Skewed Gaussians
    for c in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for sg in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))
    
    # U-shaped (high at edges)
    for p in [0.5, 1.0, 2.0]:
        candidates.append(normalize((x - 0.5) ** (2 * p) + 0.1))
    
    # Multi-modal
    for c1 in [0.2, 0.3, 0.4]:
        for c2 in [0.6, 0.7, 0.8]:
            candidates.append(normalize(
                np.exp(-((x - c1)**2) / 0.02) + np.exp(-((x - c2)**2) / 0.02)))
    
    # Power taper
    for p in [1, 2, 3, 4]:
        candidates.append(normalize(x ** p * (1 - x) ** p + 0.01))
    
    # If we have a parent, add it and blends
    if parent_seq is not None and len(parent_seq) == n:
        bi = normalize(parent_seq)
        candidates.append(bi)
        for w in [0.7, 0.8, 0.9]:
            candidates.append(normalize(w * bi + (1-w) * candidates[0]))
    
    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()
    
    return best_a, best_val
