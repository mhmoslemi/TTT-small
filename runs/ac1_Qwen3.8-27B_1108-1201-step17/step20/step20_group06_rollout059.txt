def simplex_project(z):
    """Project onto probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)

            eta_trial = eta
            found = False
            best_trial = y_max
            x_best_trial = x.copy()
            for _back in range(8):
                z = y - eta_trial * grad
                x_trial = simplex_project(z)
                b_trial = fast_autocorr(x_trial)
                max_trial = b_trial.max()
                if max_trial < best_trial:
                    best_trial = max_trial
                    best_trial_x = x_trial
                if max_trial <= y_max + 1e-16:
                    found = True
                    break
                eta_trial *= 0.5

            x_new = best_trial_x if found or eta_trial < 0.01 else x.copy()
            max_new = best_trial if found else y_max
            if not found:
                eta = max(eta_trial, 0.01)

        if max_new < best_max_b - 1e-18:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.05, 5.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        # Grow beta
        if t % 40 == 0:
            beta = min(beta * beta_growth, beta_end)

        # Adaptive top_k: increase as we converge
        if stall > 200 and top_k < n // 3:
            top_k = min(top_k + 5, int(n * 0.05))

        # Restart
        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                noise = 0.01 + 0.02 * rng.random()
                pert = rng.normal(0, noise, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.2)
            elif strategy == 2:
                x_r = best_a * np.exp(0.04 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= (0.6 + 0.2 * rng.random())
                x_r[~mask] *= (1.2 + 0.3 * rng.random())
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 15)))
                x_r = np.roll(best_a, shift)
            else:
                # Blend with spectral-flattened version
                A = np.fft.rfft(best_a, 1 << (2 * n - 2).bit_length())
                power = np.abs(A) ** 2
                target = power[1:].mean() if len(power) > 1 else power[0]
                for k in range(1, len(A)):
                    ratio = power[k] / max(target, 1e-30)
                    if ratio > 1.0:
                        A[k] *= np.sqrt(max(0.3, 1.0 - 0.3 * (ratio - 1.0)))
                x_r = np.fft.irfft(A, 1 << (2 * n - 2).bit_length())[:n]
                x_r = 0.5 * best_a + 0.5 * np.maximum(x_r, 0)

            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nadv restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + str(int(beta)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nadv t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " beta=" + str(int(beta)) + " stall=" + str(stall))

    return best_a, best_max_b


def spectral_refine(a0, time_limit, verbose=True, seed=777):
    """Spectral domain peak suppression."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    alpha = 0.03

    for it in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        target = power[1:].mean() if len(power) > 1 else power[0]

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                A_new[k] *= np.sqrt(max(0.2, 1.0 - alpha * (ratio - 1.0)))
            elif ratio < 0.3:
                A_new[k] *= np.sqrt(min(3.0, 1.0 + alpha * 0.5 * (0.3 - ratio)))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-18:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 500 == 0:
            # Phase perturbation
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.03
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 800 == 0:
            alpha = max(0.003, alpha * 0.93)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def lp_fw(a0, time_limit, solver, initial_K=250, max_K=1500, verbose=True):
    """LP-Frank-Wolfe with active set management."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    ac_y = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K])
    no_improve = 0
    max_no_improve = 400

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > max_no_improve:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-18:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        constraint_list = sorted(active_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(active_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(active_set) > initial_K // 2:
                active_set = set(int(k) for k in np.argsort(ac_y)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints
        size = 2 * n - 1
        fft_sz = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_sz)
        Gf = np.fft.rfft(g_dir, fft_sz)
        conv_ys = np.fft.irfft(Yf * Gf, fft_sz)[:size]
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:20]]
            for k in top_v:
                active_set.add(int(k))

        # Adaptive line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = M_curr
        for alpha_try in [best_alpha, best_alpha * 0.5, best_alpha * 2.0,
                          0.05, 0.1, 0.15, 0.2, 0.3]:
            alpha_try = max(0.01, min(1.0, alpha_try))
            trial = (1.0 - alpha_try) * y + alpha_try * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha_try

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " |S|=" + str(len(active_set)))

    return best_a, best_m


def coordinate_local_search(a0, time_limit, num_peaks=20, verbose=True, seed=123):
    """
    NEW: Coordinate-based local search.
    For each top autocorrelation peak, identify contributing coordinates
    and try small adjustments to reduce the peak.
    Uses O(n) incremental autocorrelation updates.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    size = 2 * n - 1
    stall = 0

    # Precompute FFT size
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        max_val = ac.max()

        # Get top peaks
        top_peaks = np.argsort(ac)[::-1][:num_peaks]

        improved = False
        for p_idx in top_peaks[:min(10, len(top_peaks))]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(p_idx)
            # Coordinates that contribute to ac[k]:
            # js = max(0, k-n+1) to je = min(n-1, k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            # Contribution: a[i] * a[k-i] for each i in idx
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()

            if total_contrib < 1e-15:
                continue

            # The peak value is ac[k]. We want to reduce it.
            # Strategy: move mass from the largest contributor to a random other position.
            # Find the source with largest contribution
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])
            partner = k - src

            # Try several fractions and destinations
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue

                # Try a few random destinations (avoiding positions that create new peaks)
                for attempt in range(8):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    # Check if dst would create issues with partner
                    # (dst + partner or dst - partner in range)
                    if dst == partner:
                        continue

                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-18:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Periodic random perturbation to escape
            if stall > 200:
                n_moves = int(rng.integers(1, 4))
                for _ in range(n_moves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if verbose and stall % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CoordLS (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " stall=" + str(stall))

    return best_a, best_m


def simulated_annealing(a0, time_limit, verbose=True, seed=456):
    """SA with diverse moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    T = 0.006
    T_min = 0.00003
    T_decay = 0.9996
    cur_m = best_m

    for it in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break
        if T < T_min:
            T = 0.004  # Reheat

        move_type = rng.integers(3)
        if move_type == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = rng.uniform(0.001, 0.02)
            amt = a[i] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
        elif move_type == 1:
            # Two-pair swap
            i, j = int(rng.integers(n)), int(rng.integers(n))
            k, l = int(rng.integers(n)), int(rng.integers(n))
            if i == j or k == l or i == k or i == l or j == k or j == l:
                continue
            amt = min(a[i], a[k]) * rng.uniform(0.01, 0.04)
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial[k] -= amt
            a_trial[l] += amt
        else:
            # Local Gaussian bump
            center = int(rng.integers(n))
            width = int(rng.integers(3, 30))
            mask = np.zeros(n, dtype=bool)
            lo = max(0, center - width)
            hi = min(n, center + width)
            mask[lo:hi] = True
            pert = rng.normal(0, 0.008, n) * mask * (a + 0.001)
            a_trial = a + pert

        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        m_trial = fast_autocorr(a_trial).max()

        if m_trial < cur_m:
            a = a_trial
            cur_m = m_trial
            if m_trial < best_m:
                best_m = m_trial
                best_a = a_trial.copy()
        elif rng.random() < np.exp(-(m_trial - cur_m) / T):
            a = a_trial
            cur_m = m_trial

        T *= T_decay

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SA it=" + str(it) + " T=" + str(round(T, 6)) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best sequence if candidate is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-strategy optimization to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
    Uses Nesterov, spectral, LP-FW, coordinate local search, SA, and multi-length.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print("Time budget: " + repr(time_budget) + "s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(round(p_score, 6)))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(len(best_seq)) + ", score=" + repr(round(best_score, 6)))

    n0 = len(best_seq)
    print("Start: n=" + str(n0) + ", J=" + repr(round(best_score, 6)))

    # Phase 1: Quick spectral refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Spectral (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        a_out, _ = spectral_refine(norm_sum1(best_seq), t1, verbose=False, seed=777)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "spec1", t_global)

    # Phase 2: Main Nesterov (primary workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P2: Nesterov Adv (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40:
        a_out, _ = nesterov_adv(norm_sum1(best_seq), t2,
                                beta_start=15.0, beta_end=2000.0,
                                beta_growth=1.004, eta_init=1.0,
                                top_k=25, restart_every=600,
                                quench_every=2000,
                                verbose=True, seed=42)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nadv1", t_global)

    # Phase 3: LP-Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P3: LP-FW (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 25 and n_best <= 9000:
        a_out, _ = lp_fw(norm_sum1(best_seq), t3, solver,
                         initial_K=250, max_K=1500, verbose=True)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "lp1", t_global)

    # Phase 4: Coordinate Local Search (NEW)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P4: CoordLS (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_out, _ = coordinate_local_search(norm_sum1(best_seq), t4,
                                           num_peaks=15, verbose=True, seed=123)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cl1", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P5: Multi-length (base n=" + str(n_best) + ") ===")

    # Try several longer lengths
    longer_lengths = [n_best + 400, n_best + 1000, n_best + 2000, n_best + 3500]
    longer_lengths = [l for l in longer_lengths if l <= 12000]

    for new_n in longer_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print("  -> n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        # Nesterov on longer
        t5a = min(remaining * 0.35, 200.0)
        if t5a > 30:
            a_out, _ = nesterov_adv(a_long, t5a,
                                    beta_start=20.0, beta_end=1500.0,
                                    beta_growth=1.004, eta_init=0.8,
                                    top_k=20, restart_every=500,
                                    quench_every=2500,
                                    verbose=False, seed=new_n % 7919)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nl_" + str(new_n), t_global)

        # Coordinate LS on longer
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.30, 100.0)
            if t5b > 15:
                a_out, _ = coordinate_local_search(norm_sum1(best_seq), t5b,
                                                   num_peaks=15, verbose=False, seed=new_n % 3217)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "cl_nl" + str(new_n), t_global)

    # Phase 6: Second Nesterov round (different params)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P6: Nesterov r2 (n=" + str(n_best) + ") ===")
    if remaining > 40:
        t6 = min(remaining * 0.20, 250.0)
        if t6 > 30:
            a_out, _ = nesterov_adv(norm_sum1(best_seq), t6,
                                    beta_start=50.0, beta_end=3000.0,
                                    beta_growth=1.005, eta_init=0.5,
                                    top_k=15, restart_every=400,
                                    quench_every=1500,
                                    verbose=True, seed=999)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nadv2", t_global)

    # Phase 7: SA + Spectral combo
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P7: SA+Spec (n=" + str(n_best) + ") ===")
    if remaining > 25:
        t7a = min(remaining * 0.30, 80.0)
        if t7a > 10:
            a_out, _ = spectral_refine(norm_sum1(best_seq), t7a, verbose=False, seed=888)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "spec2", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.40, 100.0)
            if t7b > 10:
                a_out, _ = simulated_annealing(norm_sum1(best_seq), t7b, verbose=False, seed=456)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "sa1", t_global)

    # Phase 8: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== P8: Final polish (n=" + str(n_best) + ") ===")
    if remaining > 20:
        # Quick Nesterov
        t8a = min(remaining * 0.35, 100.0)
        if t8a > 10:
            a_out, _ = nesterov_adv(norm_sum1(best_seq), t8a,
                                    beta_start=100.0, beta_end=4000.0,
                                    beta_growth=1.006, eta_init=0.3,
                                    top_k=10, restart_every=300,
                                    quench_every=1000,
                                    verbose=False, seed=555)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nadv3", t_global)

        # Coordinate LS
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t8b = min(remaining * 0.35, 80.0)
            if t8b > 10:
                a_out, _ = coordinate_local_search(norm_sum1(best_seq), t8b,
                                                   num_peaks=10, verbose=False, seed=654)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "cl2", t_global)

        # Final LP
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15 and n_best <= 9000:
            t8c = min(remaining * 0.50, 100.0)
            if t8c > 10:
                a_out, _ = lp_fw(norm_sum1(best_seq), t8c, solver,
                                 initial_K=300, max_K=2000, verbose=False)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "lp_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
