import time

def solve_lp(a, lags, n):
    """Solve the LP to find the best direction g."""
    K = len(lags)
    # Build constraint matrix A (K x n): row k has a[k-i] at position i
    A = np.zeros((K, n))
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            A[jj, idx] = a[k_int - idx]
    
    # LP: min M s.t. sum(g)=1, g>=0, A@g <= M
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M → c = [0,...,0,1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Inequality: A@g - M <= 0 for each row
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(K)
    
    # Bounds: g_i >= 0, M free (or M >= -inf)
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum > 1e-15:

        return g
    return None
