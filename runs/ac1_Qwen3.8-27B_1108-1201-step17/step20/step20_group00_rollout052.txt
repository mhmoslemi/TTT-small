def soft_max_and_grad(a, T, top_k=300):
    """
    Compute soft-max value and gradient of the peak autocorrelation.
    Returns (soft_max_value, gradient_array)
    """
    n = len(a)
    size = 2 * n - 1
    ac = fast_autocorr(a)
    
    k = min(top_k, size)
    idx = np.argsort(ac)[::-1][:k]
    vals = ac[idx]
    
    # Soft-max with numerical stability
    v_max = vals[0]
    exp_vals = np.exp(T * (vals - v_max))
    weights = exp_vals / exp_vals.sum()
    
    sm_val = v_max + np.log(exp_vals.sum()) / T
    
    # Gradient computation
    # grad[j] = 2 * sum_{k in active} w_k * a[k-j] (if 0 <= k-j < n)
    grad = np.zeros(n, dtype=np.float64)
    
    for wk, ki in zip(weights, idx):
        ki = int(ki)
        # c[ki] = sum_i a[i]*a[ki-i], valid i: max(0,ki-n+1) to min(n-1,ki)
        # d/d(a[j]) c[ki] = 2*a[ki-j] if 0 <= ki-j < n
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start >= j_end:
            continue
        js = np.arange(j_start, j_end)
        partners = ki - js
        grad[js] += 2.0 * wk * a[partners]
    
    return sm_val, grad

def soft_max_mirror_descent(a0, time_limit, T_init=500.0, T_min=2.0, 
                             top_k=300, lr_init=0.05, decay=0.9995,
                             verbose=True):

    Mirror descent with soft-max objective and temperature annealing.
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    
    a_prev = a.copy()
    T = T_init
    lr = lr_init
    stall = 0
    momentum = 0.3
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            # Perturb and restart
            rng = np.random.default_rng(hash((t, n)) % 2**32)
            noise = rng.standard_normal(n) * 0.02 * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            a_prev = a.copy()
            T = max(T_min, T * 0.5)
            lr = lr_init * 0.5
            stall = 0
            continue
        
        sm_val, grad = soft_max_and_grad(a, T, top_k=top_k)
        
        # Mirror descent with momentum
        log_ratio = np.log(a / np.maximum(a_prev, 1e-15))
        log_update = -lr * grad + momentum * log_ratio
        log_update = np.clip(log_update, -40.0, 40.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        val = obj_val(a_new)
        if val < best_val - 1e-13:
            best_val = val
            best_a = a_new.copy()
            stall = 0
            # Anneal temperature
            T = max(T_min, T * decay)
        else:
            stall += 1
            if stall % 200 == 0:
                lr *= 0.85
                T = max(T_min, T * 0.9)
        
        a_prev = a
        a = a_new
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD t={t} ({elapsed:.1f}s) J={best_val:.8f} T={T:.2f} lr={lr:.5f} stall={stall}")
            if elapsed > time_limit * 0.9:
                break
    
    return best_a, best_val

def lp_refine(a0, time_limit, solver, init_K=500, max_K=4000, verbose=True):

    n = len(a0)

    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    ac = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[-init_K:])
    no_improve = 0
    stall_limit = 2000
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)
        
        K = len(clist)
        if K == 0:
            break
        
        # Build LP constraint matrix
