def fast_autocorr(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, active_list, n):
    K = len(active_list)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj in range(K):
        ki = int(active_list[jj])
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[ki - idx]
    return coeffs

def fft_gradient(a, top_lags, top_weights):
    n = len(a)
    w_padded = np.zeros(2 * n - 1)
    for i in range(len(top_lags)):
        ki = int(top_lags[i])
        if 0 <= ki < 2 * n - 1:
            w_padded[ki] = top_weights[i]
    flipped_w = w_padded[::-1]
    req_size = 3 * n - 2
    fft_size = 1 << (req_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(flipped_w, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    return conv_result[n - 1 : 2 * n - 2 + 1]

            idx = np.arange(ws, we + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-25:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-15:
                continue

            pr = np.zeros(n, dtype=np.float64)
            for kk in top_k[:10]:
                kki = int(kk)
                wsi = max(0, kki - (n - 1))
                wei = min(n - 1, kki)
                if wsi > wei:
                    continue
                idxi = np.arange(wsi, wei + 1)
                pr[idxi] += a[kki - idxi]
            pr[src] = 1e10
            quiet_positions = np.argsort(pr)[:40]

            found = False
            for dst in quiet_positions:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in fracs:
                    amt = src_val * frac
                    if amt < 1e-18:
                        break
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

            if not improved and stall > 200:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
                            improved = True
                break

        if not improved:
            stall += 1
            if stall > 800:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 1000 == 0 and stall > 0:
            print("    Peel ({:.1f}s) J={:.7f} stall={}".format(time.monotonic() - t0, 2 * n * best_m, stall))

    return best_a, 2.0 * n * best_m

def gen_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()

def gen_asymmetric(n, alpha):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x ** alpha)
    a = np.maximum(a, 0)
    return a / a.sum()

def gen_combined(n, r, alpha):
    g = gen_geometric(n, r)
    s = gen_asymmetric(n, alpha)
    a = 0.5 * g + 0.5 * s
    return a / a.sum()

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1700.0)
    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_norm = normalize(p)
            sc = eval_safe(parent_norm)
            if np.isfinite(sc):
                best_seq = parent_norm
                best_score = sc
                print("Parent: n={} score={}".format(len(p), repr(sc)))
    except Exception:
        pass

    if best_seq is None:
        n_init = 10000
        best_seq = gen_geometric(n_init, 0.999)
        best_score = eval_safe(best_seq)
        print("Default init n={} score={}".format(n_init, repr(best_score)))

    n0 = len(best_seq)

    # Phase 1: Try alternative initializations
    remaining = time_budget - (time.monotonic() - t_global)
    alt_inits = []
    if remaining > 100:
        alt_inits.append(("geo_0995", gen_geometric(n0, 0.995)))
        alt_inits.append(("geo_0999", gen_geometric(n0, 0.999)))
        alt_inits.append(("asym_05", gen_asymmetric(n0, 0.5)))
        alt_inits.append(("asym_2", gen_asymmetric(n0, 2.0)))
        alt_inits.append(("combo_0999_1", gen_combined(n0, 0.999, 1.0)))
        alt_inits.append(("combo_0995_05", gen_combined(n0, 0.995, 0.5)))
        for name, init in alt_inits:
            sc = eval_safe(init)
            print("  Init {}: n={} score={}".format(name, n0, repr(sc)))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = init
                print("  -> New best: {}".format(name))

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 2: LP-FW on best initialization
    n_cur = len(best_seq)
    print("\n=== P2: LP-FW main (n={}) ===".format(n_cur))
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 60:
        a_out, v_out = lp_fw(normalize(best_seq), t2, solver, init_K=200, max_K=3000, add_per=25, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: {}".format(repr(best_score)))

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 3: Gradient refinement
    n_cur = len(best_seq)
    print("\n=== P3: GradRefine (n={}) ===".format(n_cur))
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_out, v_out = grad_refine(normalize(best_seq), t3, n_top=150, lr0=0.04, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 improved: {}".format(repr(best_score)))

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 4: LP-FW second pass (with larger active set)
    n_cur = len(best_seq)
    print("\n=== P4: LP-FW pass 2 (n={}) ===".format(n_cur))
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 60:
        a_out, v_out = lp_fw(normalize(best_seq), t4, solver, init_K=300, max_K=4500, add_per=30, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P4 improved: {}".format(repr(best_score)))

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 5: Coordinate peeling
    n_cur = len(best_seq)
    print("\n=== P5: CoordPeel (n={}) ===".format(n_cur))
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20:
        a_out, v_out = coord_peel(normalize(best_seq), t5, n_targets=15, seed=42, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P5 improved: {}".format(repr(best_score)))

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 6: Try longer length
    n_long = min(len(best_seq) + 3000, 20000)
    if n_long != len(best_seq) and remaining > 120:
        print("\n=== P6: Long n={} ===".format(n_long))
        t6 = min(remaining * 0.18, 280.0)
        a_long = interpolate_seq(best_seq, n_long)
        a_out, v_out = grad_refine(a_long, t6 * 0.4, n_top=200, lr0=0.03, verbose=False)
        a_out2, v_out2 = coord_peel(a_out, t6 * 0.3, n_targets=12, seed=99, verbose=False)
        sc1 = eval_safe(a_out)
        sc2 = eval_safe(a_out2)
        best_sc6 = min(sc1, sc2)
        if np.isfinite(best_sc6) and best_sc6 < best_score:
            best_score = best_sc6
            best_seq = a_out if sc1 < sc2 else a_out2
            print("  P6 improved: {}".format(repr(best_score)))

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 7: Final LP-FW polish
    n_cur = len(best_seq)
    if remaining > 60 and n_cur <= 15000:
        print("\n=== P7: Final LP-FW (n={}) ===".format(n_cur))
        t7 = min(remaining * 0.20, 250.0)
        a_out, v_out = lp_fw(normalize(best_seq), t7, solver, init_K=400, max_K=5000, add_per=30, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P7 improved: {}".format(repr(best_score)))
        remaining = time_budget - (time.monotonic() - t_global)

    # Phase 8: Final coord polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print("\n=== P8: Final Peel (n={}) ===".format(n_cur))
        t8 = min(remaining * 0.30, 200.0)
        a_out, v_out = coord_peel(normalize(best_seq), t8, n_targets=20, seed=777, verbose=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P8 improved: {}".format(repr(best_score)))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} t={:.1f}s".format(repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    return [float(x) for x in best_seq]
