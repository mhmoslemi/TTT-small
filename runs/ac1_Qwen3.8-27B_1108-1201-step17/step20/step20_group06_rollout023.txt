def fast_autocorr(a):
    """Compute full autocorrelation (convolution) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def simplex_project(z):
    """Project onto the probability simplex (sum=1, non-negative)."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_smoothed_grad(y, beta, top_k=None):
    """Compute gradient of smoothed max of autocorrelation (log-sum-exp)."""
    n = len(y)
    b = fast_autocorr(y)
    size = 2 * n - 1

    if top_k is not None and top_k < size:
        idx = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros(size)
        b_use[idx] = b[idx]
        b_max = b[idx].max()
    else:
        b_use = b
        b_max = b.max()

    if b_max <= 0:
        return np.zeros(n), b

    shifted = beta * (b_use - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        p = np.ones(size) / size
    else:
        p = exp_vals / s

    # Gradient via FFT: d/dy_k max_b = sum_i p[i] * (contribution of y_k to b[i])
    # b[i] = sum_j y[j]*y[i-j], so db[i]/dy[k] = y[i-k] + y[k-i] (in convolution sense)
    # Full gradient: grad[k] = sum_i p[i] * (y[i-k] + y[k-i]) for valid indices
    # This is equivalent to convolve(p, y) + convolve(p, y[::-1]) at the right offset
    a_rev = y[::-1]
    conv_len = n + size - 1
    fft_size_g = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fft_size_g)
    Pr = np.fft.rfft(p, fft_size_g)
    conv = np.fft.irfft(Ar * Pr, fft_size_g)
    grad = conv[n - 1:2 * n - 1] / beta
    return grad, b

    """Nesterov accelerated gradient with backtracking and diverse restarts."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(10, int(n * top_k_ratio))
    eta = eta_init

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        grad, b = compute_smoothed_grad(y, beta, top_k)
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            x_new = x
            max_new = fast_autocorr(x_new).max()
        else:
            eta_trial = eta
            found = False
            y_max = fast_autocorr(y).max()
            for _ in range(12):
                z = y - eta_trial * grad
                x_new = simplex_project(z)
                max_new = fast_autocorr(x_new).max()
                if max_new <= y_max + 1e-15:
                    found = True
                    break
                eta_trial *= 0.5
            if not found:
                x_new = x
                max_new = fast_autocorr(x).max()
                eta = max(eta_trial, 0.01)

        if max_new < best_max_b - 1e-16:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
            eta = min(eta * 1.08, 2.0)
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 50 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            strategy = rng.integers(6)
            if strategy == 0:
                noise_level = 0.02 + 0.02 * rng.random()
                pert = rng.normal(0, noise_level, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                x_r = 0.85 * best_a + 0.15 * rng.dirichlet(np.ones(n) * 0.5)
            elif strategy == 2:
                x_r = best_a * np.exp(0.04 * rng.standard_normal(n))
            elif strategy == 3:
                mask = rng.random(n) > 0.5
                x_r = best_a.copy()
                x_r[mask] *= 0.6
                x_r[~mask] *= 1.4
            elif strategy == 4:
                shift = int(rng.integers(1, max(2, n // 8)))
                x_r = np.roll(best_a, shift)
            else:
                # Gaussian blur + re-normalize
                from scipy.ndimage import gaussian_filter1d
                x_r = gaussian_filter1d(best_a, sigma=3)
            x_r = np.maximum(x_r, 1e-10)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.5, 10.0)
            eta = eta_init
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    Nest restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    Nest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " b=" + str(int(beta)))

    return best_a, best_max_b


def lp_fw_opt(a0, time_limit, solver, initial_K=200, max_K=3000,
              add_per_iter=30, verbose=True, seed=1):
    """
    LP-Frank-Wolfe with aggressive constraint management.
    Solves: min M s.t. A_active @ g <= M, sum(g)=1, g>=0
    Then moves y towards g with adaptive step size.
    """
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    ac = fast_autocorr(y)
    M_curr = ac.max()
    # Initialize active set with top constraints
    top_idx = np.argsort(ac)[::-1][:initial_K]
    active_set = set(int(k) for k in top_idx)

    no_improve = 0
    max_no_improve = 800
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > max_no_improve:
            # Restart active set
            ac = fast_autocorr(y)
            top_idx = np.argsort(ac)[::-1][:initial_K]
            active_set = set(int(k) for k in top_idx)
            no_improve = 0
            t_nest = 1.0
            continue

        iter_count += 1
        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Manage active set size
        if len(active_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in active_set])
            keep = np.argsort(margins)[:max_K // 2]
            active_set = set(int(k) for i, k in enumerate(active_set) if i in set(keep.tolist()))
            # Re-add top constraints
            top_idx = np.argsort(ac)[::-1][:initial_K // 2]
            for k in top_idx:
                active_set.add(int(k))

        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix A where A[j, i] = coefficient of y[i] in conv[k_j]
        coeffs = np.zeros((K, n), dtype=np.float64)
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP: min M s.t. coeffs @ g <= M, sum(g) = 1, g >= 0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reduce active set on failure
            active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K // 2])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if len(active_set) > initial_K:
                active_set = set(int(k) for k in np.argsort(ac)[::-1][:initial_K])
            continue

        M_lp = float(M_var.value) if M_var.value is not None else M_curr
        g_val = np.asarray(g_var.value, dtype=np.float64)
        g_val = np.maximum(g_val, 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Add violated constraints
        if M_lp < M_curr:
            # Check if g_dir violates any currently inactive constraints
            size = 2 * n - 1
            fft_size = 1 << (size - 1).bit_length()
            Yf = np.fft.rfft(y, fft_size)
            Gf = np.fft.rfft(g_dir, fft_size)
            conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
            violations = np.where(conv_ys > M_curr + 1e-7)[0]
            n_add = min(len(violations), add_per_iter)
            if n_add > 0:
                viol_amt = conv_ys[violations] - M_curr
                top_v = violations[np.argsort(viol_amt)[::-1][:n_add]]
                for k in top_v:
                    active_set.add(int(k))

        # Adaptive line search
        best_alpha = 2.0 / (iter_count + 2.0)
        best_trial_M = M_curr
        alphas_to_try = [best_alpha, best_alpha * 0.5, best_alpha * 1.5,
                         1.0 / (iter_count + 1), 0.02, 0.05, 0.1]
        for alpha in alphas_to_try:
            alpha = max(0.005, min(1.0, alpha))
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha

        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW it=" + str(iter_count) + " (" + str(round(elapsed, 1)) + "s): J=" +
                  repr(round(J, 6)) + " |S|=" + str(len(active_set)) + " M_lp=" + repr(round(M_lp, 8)))

    return best_a, best_m


def subgradient_polish(a0, time_limit, verbose=True, seed=99):
    """
    Cutting-plane / subgradient method: directly target reducing the max peak.
    At each step, identify the argmax peak, compute its subgradient, and take a step.
    """
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    lr = 1.0 / (n ** 0.5)
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 500:
            break

        ac = fast_autocorr(y)
        max_val = ac.max()
        max_idx = np.argmax(ac)

        if max_val < best_m - 1e-15:
            best_m = max_val
            best_a = y.copy()
            stall = 0
            lr = min(lr * 1.1, 2.0 / np.sqrt(n))
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.7

        # Subgradient of max at the peak: grad[k] = y[max_idx - k] (for valid range)
        k_int = int(max_idx)
        grad = np.zeros(n)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        grad[idx] = y[k_int - idx]

        # Also include a small smoothing gradient to avoid oscillation
        top_k = max(5, n // 50)
        top_indices = np.argsort(ac)[::-1][:top_k]
        smooth_grad = np.zeros(n)
        for k_i in top_indices:
            ki = int(k_i)
            w = ac[k_i] / (ac[top_indices[0]] + 1e-30)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            ii = np.arange(s, e)
            smooth_grad[ii] += w * y[ki - ii]
        smooth_grad /= top_k

        grad_final = 0.5 * grad + 0.5 * smooth_grad
        gnorm = np.linalg.norm(grad_final)
        if gnorm < 1e-15:
            break

        grad_final /= gnorm
        z = y - lr * grad_final
        y_new = simplex_project(z)
        M_new = fast_autocorr(y_new).max()

        # Accept if improves, or with Stiefel-Kingsbury rule
        if M_new < max_val:
            y = y_new
        else:
            # Stiefel-Kingsbury: accept if close
            if M_new < max_val + lr * 0.1:
                y = y_new
                beta_sk = -lr
            else:
                beta_sk = 2.0 / (t + 2.0)
                y = y  # Stay put

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    SubGrad t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)) + " lr=" + repr(round(lr, 4)))

    return best_a, best_m


def spectral_smooth(a0, time_limit, verbose=True, seed=444):
    """Iteratively flatten the power spectrum to reduce peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    alpha = 0.05
    rng = np.random.default_rng(seed)

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Target: reduce peaks above median
        if len(power) > 1:
            target = np.median(power[1:])
        else:
            target = power[0]

        A_new = A.copy()
        for k in range(1, len(A)):
            ratio = power[k] / max(target, 1e-30)
            if ratio > 1.0:
                red = 1.0 - alpha * min(ratio - 1.0, 2.0)
                A_new[k] *= np.sqrt(max(0.2, red))
            elif ratio < 0.3:
                inc = 1.0 + alpha * 0.5 * (0.3 - ratio)
                A_new[k] *= np.sqrt(min(3.0, inc))

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        elif it % 100 == 0:
            # Phase randomization escape
            A = np.fft.rfft(a, fft_size)
            phase_pert = rng.standard_normal(len(A)) * 0.03
            A[1:] *= np.exp(1j * phase_pert[1:])
            a_trial = np.fft.irfft(A, fft_size)[:n]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                m_t = fast_autocorr(a_trial).max()
                if m_t < best_m:
                    best_m = m_t
                    best_a = a_trial.copy()
                    a = a_trial

        if it % 300 == 0:
            alpha = max(0.01, alpha * 0.97)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 6)))

    return best_a, best_m


def try_update(best_seq, best_score, a, label, t_global):
    """Try to update best with candidate a."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_sequence([float(x) for x in a_norm])
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 6)) + " (n=" + str(len(a)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(x) for x in parent])
            if np.isfinite(p_score):

        x = np.linspace(0, 1, n_base)

        best_seq = norm_sum1(w)

            best_seq, best_score = try_update(best_seq, best_score, a_out, "spec_final", t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:

                                        beta_start=200.0, beta_end=3000.0,
                                        beta_growth=1.006, eta_init=0.15,
                                        top_k_ratio=0.005, restart_every=500,
                                        verbose=False, seed=1234)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "nest_final", t_global)

    # Final validation
    if best_seq is not None:
        fs = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: J=" + repr(round(best_score, 6)) + ", n=" + str(len(best_seq)) +
              ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_sequence([float(xv) for xv in best_seq])

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
