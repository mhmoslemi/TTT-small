  - If i == k-i (i.e., k = 2i): change is -t*a[i] + (-t)^2 ... actually it's (a[i]-t)^2 - a[i]^2 = -2t*a[i] + t^2
  - Similarly for a[j]: +2t*a[k-j] (if j != k-j)
  
So for a given k:
- delta_b[k] ≈ 2t*(a[k-j] - a[k-i]) (first order, ignoring the t^2 term and assuming i != k-i and j != k-j)

This is linear in t for fixed k. The new b[k] = b[k] + 2t*(a[k-j] - a[k-i]).

We want to minimize max_k (b[k] + 2t*(a[k-j] - a[k-i])).

This is a 1D convex optimization problem! We can solve it via:
- Bisection: find t* where the derivative of the max changes sign
- Or more simply: try several values of t and pick the best

But the issue is that with n=6700, checking all k for each (i,j) pair is O(n) per evaluation, and we have O(n^2) pairs. So we need to be selective.

A practical approach:
1. Find top K peaks
2. For each peak k, the "hot" coordinates are those i where a[i]*a[k-i] is large
3. For each hot coordinate i (from the top peaks), find the best j to transfer to
4. Use a small number of trial t values

Let me also think about a completely different approach: **alternating projection**.

The problem is: minimize max_k b[k] subject to a >= 0, sum a = 1.

We can think of this as:

For a fixed M, we want to find if this intersection is non-empty. The minimal M is our answer.

But finding the projection onto {a : max_k (a*a)[k] <= M} is hard (it's a convex set since each (a*a)[k] is convex in a, and the max of convex is convex, so the sublevel set is convex).

Actually wait: (a*a)[k] = sum_i a[i]*a[k-i] is a quadratic form. Is it convex? The Hessian is 2 times the matrix with 1s at positions (i, k-i) and (k-i, i). This matrix is not positive semidefinite in general (it could have negative eigenvalues). So (a*a)[k] is not necessarily convex!

For example, if k=0: (a*a)[0] = a[0]^2, which is convex.
If k=1: (a*a)[1] = 2*a[0]*a[1], which is not convex (it's a rank-1 indefinite quadratic form).

So the problem is non-convex. This explains why it's hard.

def fast_autocorr(a):
    """Full self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(z):
    """Euclidean projection onto the probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1))
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(z - theta, 0.0)

def eval_j(a):
    """Compute J = 2*n*max_b / sum^2 for given sequence a."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def smooth_max_and_grad(a, beta, top_k=200):
    """
    Compute the beta-smoothed max of autocorrelation and its gradient.
    Returns (value, gradient).
    """
    n = len(a)
    b = fast_autocorr(a)
    
    # Get top-k peaks
    if top_k < len(b):
        idx_top = np.argpartition(b, -top_k)[-top_k:]
    else:
        idx_top = np.arange(len(b))
    
    b_top = b[idx_top]
    b_max = b_top.max()
    
    # Softmax weights
    shifted = beta * (b_top - b_max)
    shifted = np.clip(shifted, -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    
    # Smoothed value
    val = b_max + (1.0 / beta) * np.log(exp_vals.sum())
    
    # Gradient: grad[i] = (1/beta) * sum_k p_k * (a[k-i] + a[i]) for valid pairs
    grad = np.zeros(n)
    for ii in range(len(idx_top)):
        k = int(idx_top[ii])
        w = p[ii] / beta
        if w < 1e-18:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        grad[idx] += w * a[k - idx]
        grad[k - idx] += w * a[idx]
    
    return val, grad

def nesterov_smooth_descent(a0, time_limit, beta_start=50.0, beta_end=5000.0,
                            beta_growth=1.01, top_k=200, verbose=True):
    """
    Nesterov-accelerated gradient descent on the smooth max of autocorrelation,
    projected onto the probability simplex.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    t = 0
    beta = beta_start
    
    # Nesterov state
    x_prev = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        
        # Nesterov extrapolation
        if t == 1:
            y = a.copy()
        else:
            alpha_mom = (t - 1.0) / (t + 2.0)
            y = a + alpha_mom * (a - x_prev)
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            else:
                y = a.copy()
        
        # Compute gradient at y
        _, grad_y = smooth_max_and_grad(y, beta, top_k)
        
        # Project gradient to simplex tangent space
        grad_proj = grad_y - np.mean(grad_y)
        
        # Adaptive step size
        g_norm = np.linalg.norm(grad_proj)
        if g_norm < 1e-15:
            break
        eta = 1.0 / (g_norm * np.sqrt(t + 1.0))
        eta = min(eta, 0.05)
        eta = max(eta, 1e-8)
        
        # Update
        x_new = y - eta * grad_proj
        x_new = simplex_project(x_new)
        
        # Evaluate
        b_new = fast_autocorr(x_new)
        m_new = b_new.max()
        
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = x_new.copy()
        
        x_prev = a.copy()
        a = x_new
        
        # Adapt beta
        if t % 10 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if verbose and t % 500 == 0:
            J = 2.0 * n * best_m
            print(f"    Nesterov t={t} ({elapsed:.1f}s): J={J:.6f} beta={beta:.0f}")
    
    return best_a, best_m

        w = p[ii] / beta
        if w < 1e-18:
            continue
        i_s = max(0, k - n + 1)
        i_e = min(n - 1, k)
        if i_s > i_e:
            continue
        idx = np.arange(i_s, i_e + 1)
        grad[idx] += w * a[k - idx]
        grad[k - idx] += w * a[idx]
    return val, grad

def nesterov_descent(a0, time_limit, beta_start=50.0, beta_end=5000.0,
                     beta_growth=1.012, top_k=200, verbose=True):
    """Nesterov-accelerated gradient descent on smooth max of autocorrelation."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    t = 0
    beta = beta_start
    t_nest = 1.0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        
        # Nesterov extrapolation
        if t == 1:
            y = a.copy()
        else:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            alpha = (t_nest - 1.0) / t_new
            y = a + alpha * (a - y)
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            else:
                y = a.copy()
            t_nest = t_new
        
        # Gradient at y
        _, grad_y = smooth_max_grad(y, beta, top_k)
        
        # Project to tangent space
        grad_proj = grad_y - np.mean(grad_y)
        g_norm = np.linalg.norm(grad_proj)
        
        if g_norm < 1e-14:
            break
        
        # Step size: 1/sqrt(t) decay
        eta = 1.0 / (g_norm * (1.0 + 0.1 * np.sqrt(t)))
        eta = min(eta, 0.03)
        eta = max(eta, 1e-8)
        
        # Update
        x_new = y - eta * grad_proj
        x_new = simplex_project(x_new)
        
        # Evaluate
        b_new = fast_autocorr(x_new)
        m_new = b_new.max()
        
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = x_new.copy()
        
        a = x_new
        
        # Adapt beta
        if t % 10 == 0:
            beta = min(beta * beta_growth, beta_end)
        
        if verbose and t % 1000 == 0:
            J = 2.0 * n * best_m
            print(f"    Nesterov t={t} ({elapsed:.1f}s): J={J:.6f} beta={beta:.0f}")
    
    return best_a, best_m

def coord_refine(a0, time_limit, top_k=20, verbose=True):

                t_val = a[src] * frac

                    continue

                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

                    a = a_trial
                    improved = True
                    break

                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))

                    continue

        J = 2.0 * n * best_m
        print(f"    CoordRefine ({elapsed:.1f}s): J={J:.6f} iters={iters}")

def lp_frankwolfe(a0, time_limit, verbose=True):
    """LP-based Frank-Wolfe for the active set of top constraints."""
    solver = _get_solver()
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    t = 0
    
    b = fast_autocorr(a)
    active_set = set(int(k) for k in np.argsort(b)[::-1][:500])
    
    while time.monotonic() - t0 < time_limit and no_improve < 500:
        t += 1
        b = fast_autocorr(a)
        M_curr = b.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update active set
        active_set = set(int(k) for k in np.argsort(b)[::-1][:500])
        constraint_list = sorted(active_set)
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix (sparse-friendly)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n, k_i + 1)
            if i_s < i_e:
                idx = np.arange(i_s, i_e)
                coeffs[jj, idx] = a[k_i - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Line search
        best_alpha = 2.0 / (t + 2.0)
        best_trial_M = float('inf')
        for alpha in [best_alpha, max(0.01, best_alpha * 0.5), 1.0 / (t + 1.0)]:
            trial = (1.0 - alpha) * a + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial = trial / st
            M_trial = fast_autocorr(trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g_dir
        a = np.maximum(a, 0.0)
        st = a.sum()
        if st > 1e-15:
            a = a / st
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.6f} |S|={K}")
    
    return best_a, best_m

def interpolate(a, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a)
    return normalize(a_interp)

    for s in [cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _SOLVER = s
                return s
        except Exception:
            continue
    _SOLVER = cp.ECOS
    return _SOLVER


def fast_autocorr(a):
    """Full self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_project(z):
    """Euclidean projection onto the probability simplex."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1))
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(z - theta, 0.0)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Call the external evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def smooth_max_grad(a, beta, top_k=200):
    """
    Compute beta-smoothed max of autocorrelation and its exact gradient.
    Returns (smoothed_value, gradient_vector).
    """
    n = len(a)
    b = fast_autocorr(a)
    if top_k < len(b):
        idx_top = np.argpartition(b, -top_k)[-top_k:]
    else:
        idx_top = np.arange(len(b))
    b_top = b[idx_top]
    b_max = b_top.max()
    shifted = np.clip(beta * (b_top - b_max), -300, 300)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    val = b_max + (1.0 / beta) * np.log(exp_vals.sum())
    grad = np.zeros(n)
    for ii in range(len(idx_top)):
        k = int(idx_top[ii])
        w = p[ii] / beta
        if w < 1e-18:
            continue
        i_s = max(0, k - n + 1)
        i_e = min(n - 1, k)
        if i_s > i_e:
            continue
        idx = np.arange(i_s, i_e + 1)
        grad[idx] += w * a[k - idx]
        grad[k - idx] += w * a[idx]
    return val, grad


def nesterov_descent(a0, time_limit, beta_start=50.0, beta_end=5000.0,
                     beta_growth=1.012, top_k=200, verbose=True):
    """
    Nesterov-accelerated (FISTA) gradient descent on smooth max of autocorrelation,
    with Euclidean projection onto the probability simplex.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    t = 0
    beta = beta_start
    t_nest = 1.0
    x_prev = a.copy()

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0

        # Nesterov extrapolation
        if t == 1:
            y = a.copy()
        else:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            alpha = (t_nest - 1.0) / t_new
            y = a + alpha * (a - x_prev)
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            else:
                y = a.copy()
            t_nest = t_new

        # Gradient at y
        _, grad_y = smooth_max_grad(y, beta, top_k)

        # Project gradient to tangent space of simplex
        grad_proj = grad_y - np.mean(grad_y)
        g_norm = np.linalg.norm(grad_proj)

        if g_norm < 1e-14:
            break

        # Step size with 1/sqrt(t) decay (standard for Nesterov)
        eta = 1.0 / (g_norm * (1.0 + 0.05 * np.sqrt(t)))
        eta = min(eta, 0.04)
        eta = max(eta, 1e-9)

        # Update with projection
        x_new = y - eta * grad_proj
        x_new = simplex_project(x_new)

        # Evaluate
        b_new = fast_autocorr(x_new)
        m_new = b_new.max()

        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = x_new.copy()

        x_prev = a.copy()
        a = x_new

        # Adapt beta (grow to make smooth max closer to true max)
        if t % 8 == 0:
            beta = min(beta * beta_growth, beta_end)

        if verbose and t % 2000 == 0:
            J = 2.0 * n * best_m
            print(f"    Nesterov t={t} ({elapsed:.1f}s): J={J:.6f} beta={beta:.0f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Nesterov done t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

        m_curr = b.max()
