So the max will be at the center (conv[n-1]) or at the edges (conv[0]).

To equalize: we want conv[0] ≈ conv[n-1], i.e., a_0^2 ≈ 2*a_0*a_{n-1} + middle terms.
For a symmetric sequence (a_0 = a_{n-1}): a_0^2 ≈ 2*a_0^2 + middle, which means a_0^2 ≈ 2*a_0^2, impossible (unless a_0 = 0).

So for a symmetric U-shape, the center peak will always be larger than the edge peaks. The optimal is to make the center peak as small as possible while keeping the edge peaks from growing too large.

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient(p_w, a):
    n = len(a)
    a_flip = a[::-1]
    total_len = len(p_w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    P = np.fft.rfft(p_w, fft_size)
    Af = np.fft.rfft(a_flip, fft_size)
    conv_res = np.fft.irfft(P * Af, fft_size)
    g = conv_res[n - 1:2 * n - 1]
    return 2.0 * g


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    r = idx[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_to(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, a_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, seed=42):
    """Generate diverse parametric initializations and return the best."""
    x = np.linspace(0.0, 1.0, n)
    best_a = None
    best_m = float('inf')
    rng = np.random.default_rng(seed)

    candidates = []

    # U-shaped: (x^p + (1-x)^p)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0, 8.0]:
        a_raw = x ** p + (1.0 - x) ** p
        a_raw = np.maximum(a_raw, 1e-10)
        candidates.append(normalize(a_raw))

    # Inverse polynomial: 1/(1+c*x*(1-x))
    for c in [2, 5, 10, 20, 50]:
        a_raw = 1.0 / (1.0 + c * x * (1.0 - x))
        candidates.append(normalize(a_raw))

    # Cosine-based: 1 + cos(2*pi*x) variants
    for k in [1, 2, 3]:
        a_raw = 1.0 + 0.5 * np.cos(2.0 * np.pi * k * x)
        a_raw = np.maximum(a_raw, 1e-10)
        candidates.append(normalize(a_raw))

    # Power law asymmetric
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        for beta in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            a_raw = (x + 0.01) ** alpha * (1.0 - x + 0.01) ** beta
            candidates.append(normalize(a_raw))

    # Gaussian at edges
    for s in [0.05, 0.1, 0.15, 0.2]:
        a_raw = np.exp(-((x - 0.05) ** 2) / (2 * s ** 2)) + np.exp(-((x - 0.95) ** 2) / (2 * s ** 2))
        candidates.append(normalize(a_raw))

    # Random
    for _ in range(20):
        a_raw = rng.random(n) ** rng.uniform(0.3, 4.0)
        candidates.append(normalize(a_raw))

    for a in candidates:
        m = fast_conv(a).max()
        if m < best_m:
            best_m = m
            best_a = a.copy()

    return best_a, best_m


def adam_optimize(a0, time_limit, beta_start=30.0, beta_end=500.0, verbose=True):
    """Adam optimizer with log-sum-exp smoothing and FFT-based gradient."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    beta = beta_start
    lr = 0.01
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    stall = 0
    rng = np.random.default_rng(123)
    beta_growth = 1.003
    top_k_ratio = 0.05
    top_k = max(50, int(n * top_k_ratio))

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_conv(x)
        b_max = b.max()

        if b_max < best_m - 1e-14:
            best_m = b_max
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Compute softmax weights (focus on top peaks)
        idx_top = np.argsort(b)[::-1][:top_k]
        b_use = np.zeros_like(b)
        b_use[idx_top] = b[idx_top]
        shifted = np.clip(beta * (b_use - b_max), -300, 300)
        exp_vals = np.exp(shifted)
        p_w = exp_vals / exp_vals.sum()

        # Compute gradient via FFT
        grad = compute_gradient(p_w, x)

        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        t_cor = t
        m_unbias = m_hat / (1 - beta1 ** t_cor)
        v_unbias = v_hat / (1 - beta2 ** t_cor)
        step = lr * m_unbias / (np.sqrt(v_unbias) + eps)

        x_new = x - step
        x_new = simplex_project(x_new)

        # Check
        b_new = fast_conv(x_new)
        m_new = b_new.max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x_new.copy()
            stall = 0

        x = x_new

        # Adaptive learning rate
        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)
            lr = max(0.001, lr * 0.995)

        # Restart on stall
        if stall > 800:
            stall = 0
            noise_scale = min(0.05, 0.01 * (stall + 1))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            m_r = fast_conv(x_r).max()
            if m_r < best_m:
                best_m = m_r
                best_a = x_r.copy()
            x = x_r
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            lr = 0.01
            beta = max(beta * 0.5, 20.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print("    Adam restart t={} ({}s): J={}".format(t, round(elapsed, 1), repr(J)))

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    Adam t={} ({}s): J={}".format(t, round(elapsed, 1), repr(J)))

    return best_a, best_m
