    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

def simplex_project(z):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        # But we already have grad_scale which includes the c^p factor
        # Let me redo: grad_j = 2 * S^(1/p-1) * sum_k c_k^(p-1) * y[k-j]
        # = 2 * S^(1/p-1) * exp(log_w.max()) * sum_k exp(log_w_k - log_w.max()) * y[k-j]
        # = 2 * S^(1/p-1) * exp(log_w.max()) * cross_corr(w_norm, y)[j]

        corr = fast_cross_corr(w_norm, y)
        log_grad_full = log_grad_scale + log_w.max()
        grad = np.exp(log_grad_full) * corr

        # Adaptive step size: use backtracking-like heuristic
        grad_norm = np.sqrt(np.sum(grad ** 2))
        if grad_norm > 1e-30:
            # Step size inversely proportional to gradient magnitude
            eta = min(0.5 * n / max(1.0, grad_norm * grad_norm * n), 1.0)
            eta = max(eta, 1e-6)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_fista = t_new

        if it % 200 == 0:
            p = min(p * p_growth, p_end)

        if stall > restart_every:
            stall = 0
            t_fista = 1.0
            p = max(p * 0.5, p_start)
            noise_scale = 0.02 / (1.0 + 0.1 * (it // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    FISTA-PS restart t=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 7)))

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    FISTA-PS t=" + str(it) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 7)) + ", p=" + str(round(p, 1)))

    return best_a, best_max_b


def constrained_peak_reduction(a0, time_limit, verbose=True, seed=99):
    """Principled peak reduction: check all peaks before making a move."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 5000
    peak_count = 15

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-15:
            best_m = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find peaks above 90% of max
        threshold = 0.90 * c_max
        peak_indices = np.where(c >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.argsort(c)[::-1][:peak_count]
        peak_indices = peak_indices[:peak_count]

        improved = False
        found_move = False

        # For each peak (starting with the largest), try to reduce it
        for pk_idx in peak_indices[:8]:
            if found_move:
                break
            k = int(pk_idx)

            # Find the position contributing most to this peak
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k - idx]
            if products.max() < 1e-20:
                continue

            src_i = int(idx[np.argmax(products)])
            src_val = a[src_i]
            if src_val < 1e-14:
                continue

            # Precompute: for each candidate destination j, check safety
            # delta_c_m = 2*eps*(a[k_m - j] - a[k_m - src_i]) for each peak k_m
            # We need c[k_m] + delta_c_m <= c_max for all peaks
            # => a[k_m - j] <= a[k_m - src_i] + c_max/(2*eps) for all m
            # For small eps, this is approximately a[k_m - j] <= a[k_m - src_i]

            # Get the "partner" values for the source
            partner_vals = np.zeros(n)
            for k_m in peak_indices:
                k_mi = int(k_m)
                p_idx = k_mi - src_i
                if 0 <= p_idx < n:
                    partner_vals[p_idx] = max(partner_vals[p_idx], 0)  # just tracking

            # For each candidate j, compute the worst-case increase
            # delta_c_m(j) = 2*eps*(a[k_m - j] - a[k_m - src_i])
            # We need: for all m, c[k_m] + 2*eps*(a[k_m - j] - a[k_m - src_i]) <= c_max
            # i.e., a[k_m - j] <= a[k_m - src_i] + (c_max - c[k_m]) / (2*eps)

            eps = src_val * 0.005
            margin = (c_max - c[peak_indices]) / (2 * eps) + 1e-10

            # For each peak m, the constraint is: a[k_m - j] <= a[k_m - src_i] + margin[m]
            # Precompute the "allowance" for each position j
            # allowance[j] = min over m of (a[k_m - src_i] + margin[m] - a[k_m - j]) / (2*eps)
            # Actually we just need: a[k_m - j] - a[k_m - src_i] <= margin[m] for all m
            # Define: violation[j] = max over m of (a[k_m - j] - a[k_m - src_i] - margin[m])
            # We need violation[j] <= 0

            # Compute efficiently
            violation = np.zeros(n)
            for mi, k_m in enumerate(peak_indices):
                k_mi = int(k_m)
                # a[k_m - j] for j = 0..n-1
                # This is a "shifted" version of a
                # a[k_mi - j] is valid when 0 <= k_mi - j < n, i.e., k_mi - n + 1 <= j <= k_mi
                js = np.arange(max(0, k_mi - n + 1), min(n, k_mi + 1))
                vals = a[k_mi - js]
                # a[k_m - src_i]
                p_src = k_mi - src_i
                val_src = a[p_src] if 0 <= p_src < n else 0.0
                # violation at position j: a[k_m - j] - a[k_m - src_i] - margin[m]
                v = vals - val_src - margin[mi]
                violation[js] = np.maximum(violation[js], v)

            # Safe positions: violation[j] <= 0
            safe_mask = violation <= 1e-10
            safe_mask[src_i] = False  # Can't move to itself

            # Among safe positions, prefer those that most reduce the main peak
            # Reduction = 2*eps*(a[k - src_i] - a[k - j])
            # We want to maximize a[k - src_i] - a[k - j], i.e., minimize a[k - j]
            p_main = k - src_i
            val_main = a[p_main] if 0 <= p_main < n else 0.0

            if safe_mask.any():
                safe_positions = np.where(safe_mask)[0]
                # Score: how much this reduces the main peak
                reduction_scores = np.zeros(n)
                for j in safe_positions:
                    p_j = k - int(j)
                    val_j = a[p_j] if 0 <= p_j < n else 0.0
                    reduction_scores[j] = val_main - val_j

                # Pick the best few
                top_safe = safe_positions[np.argsort(reduction_scores[safe_positions])[::-1][:10]]

                for j in top_safe:
                    j_int = int(j)
                    # Try the move
                    a_trial = a.copy()
                    a_trial[src_i] -= eps
                    a_trial[j_int] += eps
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found_move = True
                        no_improve = 0
                        break

            if not found_move:
                # Try random safe moves with smaller eps
                for _ in range(5):
                    j = int(rng.integers(n))
                    if j == src_i:
                        continue
                    for eps_try in [src_val * 0.001, src_val * 0.002]:
                        a_trial = a.copy()
                        a_trial[src_i] -= eps_try
                        a_trial[j] += eps_try
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found_move = True
                            no_improve = 0
                            break
                    if found_move:
                        break

        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Escape local minimum with a structured perturbation
                strategy = rng.integers(3)
                if strategy == 0:
                    # Anti-symmetric nudge
                    anti = (np.arange(n, dtype=np.float64) - (n - 1) / 2.0) / n
                    eps_n = 0.005 / (1.0 + no_improve / 200.0)
                    a_try = a + eps_n * anti * a
                elif strategy == 1:
                    # Random small perturbation
                    pert = 0.01 * rng.normal(0, 1, n) * a
                    a_try = a + pert
                else:
                    # Move mass from a high-value position to a low-value one
                    i_hi = int(np.argmax(a))
                    i_lo = int(rng.integers(n))
                    if i_lo != i_hi:
                        amt = a[i_hi] * 0.01
                        a_try = a.copy()
                        a_try[i_hi] -= amt
                        a_try[i_lo] += amt
                    else:
                        a_try = a.copy()

                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                M_try = fast_autocorr(a_try).max()
                if M_try < best_m:
                    a = a_try
                    best_m = M_try
                    best_a = a.copy()
                    no_improve = 0
                else:
                    # Accept anyway with probability (simulated annealing-like)
                    if rng.random() < 0.1:
                        a = a_try

        if verbose and no_improve % 2000 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    CPR (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 7)) + " stall=" + str(no_improve))

    return best_a, best_m


def fine_tune(a0, time_limit, verbose=True, seed=200):
    """Very fine local moves for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < best_m - 1e-16:
            best_m = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        improved = False
        # Focus on the top 5 peaks
        top_peaks = np.argsort(c)[::-1][:5]

        for k in top_peaks:
            if improved:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]

            # Try top 5 contributing positions
            top_contrib = np.argsort(products)[::-1][:5]

            for ci in top_contrib:
                if improved:
                    break
                src_i = int(idx[ci])
                partner = k_int - src_i
                if partner < 0 or partner >= n:
                    continue

                # Try very small moves
                for frac in [0.0005, 0.001, 0.002]:
                    amt = a[src_i] * frac
                    if amt < 1e-18:
                        continue
                    # Try a few random destinations
                    for _ in range(5):
                        dst = int(rng.integers(n))
                        if dst == src_i:
                            continue
                        a_trial = a.copy()
                        a_trial[src_i] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-18:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Tiny random nudge to escape
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    amt = a[i] * 0.0005
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    FineTune (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 7)) + " stall=" + str(stall))

    return best_a, best_m


def nesterov_lse(a0, time_limit, beta_start=50.0, beta_end=3000.0,
                 beta_growth=1.007, eta_base=0.5, top_k_ratio=0.01,
                 restart_every=1000, verbose=True, seed=42):
    """Nesterov on log-sum-exp smoothed max (kept from previous, slightly modified)."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_max_b = fast_autocorr(x).max()
    t0 = time.monotonic()
    beta = beta_start
    t_nest = 1.0
    stall = 0
    rng = np.random.default_rng(seed)
    top_k = max(20, int(n * top_k_ratio))

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y = x + theta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        c = fast_autocorr(y)
        c = np.maximum(c, 0.0)

        if top_k < len(c):
            idx = np.argsort(c)[::-1][:top_k]
            c_use = np.zeros(len(c))
            c_use[idx] = c[idx]
            c_max = c[idx].max()
        else:
            c_use = c
            c_max = c.max()

        shifted = beta * (c_use - c_max)
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        total = exp_vals.sum()
        if total < 1e-300:
            p = np.zeros_like(c_use)
            p[np.argmax(c)] = 1.0
        else:
            p = exp_vals / total

        # Gradient: (2/beta) * conv(p, y_rev)[n-1:2n-1]
        fft_size = 1 << (2 * n - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        Yr = np.fft.rfft(y[::-1], fft_size)
        Conv = np.fft.irfft(P * Yr, fft_size)
        grad = (2.0 / beta) * Conv[n - 1:2 * n - 1]

        # Adaptive step
        grad_norm_sq = np.sum(grad ** 2)
        eta = eta_base / max(1.0, np.sqrt(grad_norm_sq) * np.sqrt(n))
        eta = min(eta, 0.1)

        z = y - eta * grad
        x_new = simplex_project(z)

        b_new = fast_autocorr(x_new)
        max_new = b_new.max()
        if max_new < best_max_b - 1e-15:
            best_max_b = max_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new
        t_nest = t_new

        if t % 100 == 0:
            beta = min(beta * beta_growth, beta_end)

        if stall > restart_every:
            stall = 0
            t_nest = 1.0
            noise_scale = 0.02 / (1.0 + 0.05 * (t // restart_every))
            anti = (np.arange(n) - (n - 1) / 2.0) / n
            anti -= anti.mean()
            rand = rng.normal(0, 1, n)
            pert = noise_scale * (0.5 * anti + 0.5 * rand) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            beta = max(beta * 0.7, 20.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_max_b
                print("    NLSE restart t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 7)))

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_max_b
            print("    NLSE t=" + str(t) + " (" + str(round(elapsed, 1)) + "s): J=" + repr(round(J, 7)) + ", beta=" + str(round(beta, 1)))

    return best_a, best_max_b


def try_update(best_seq, best_score, a, label, t_global):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print("  *** NEW BEST [" + label + "]: " + str(round(sc, 7)) + " (n=" + str(len(a_norm)) + ", t=" + str(round(elapsed, 1)) + "s)")
    return best_seq, best_score


def coarse_to_fine(coarse, fine_n, time_limit, verbose=True):
    """Optimize coarse, interpolate to fine, then refine."""
    coarse_n = len(coarse)
    fine = interpolate_sequence(coarse, fine_n)
    return fine


def propose_candidate(budget_s=1800.0):
    """Main search combining FISTA-power-sum, constrained peak reduction, Nesterov-LSE, and fine tuning."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print("Parent: n=" + str(len(parent)) + ", score=" + repr(p_score))
    except Exception:
        parent = None

    if best_seq is None:
        n_base = 6700
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print("Default init: n=" + str(n_base) + ", score=" + repr(best_score))

    n0 = len(best_seq)
    print("Starting: n=" + str(n0) + ", score=" + repr(best_score))

    # Phase 1: FISTA with power-sum (different smoothing than previous log-sum-exp)
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 1: FISTA Power-Sum (n=" + str(n0) + ") ===")
    t1 = min(remaining * 0.20, 340.0)
    if t1 > 60:
        a_out, _ = fista_power_sum(norm_sum1(best_seq), t1,
                                   p_start=50, p_end=3000, p_growth=1.01,
                                   eta_init=0.5, restart_every=600,
                                   verbose=True, seed=10)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_ps1", t_global)

    # Phase 2: Constrained peak reduction
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 2: Constrained Peak Reduction (n=" + str(n_best) + ") ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 40:
        a_out, _ = constrained_peak_reduction(norm_sum1(best_seq), t2, verbose=True, seed=20)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cpr1", t_global)

    # Phase 3: Nesterov LSE (different hyperparams from previous)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 3: Nesterov LSE (n=" + str(n_best) + ") ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 60:
        a_out, _ = nesterov_lse(norm_sum1(best_seq), t3,
                                beta_start=100.0, beta_end=5000.0,
                                beta_growth=1.009, eta_base=0.4,
                                top_k_ratio=0.008, restart_every=800,
                                verbose=True, seed=30)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "nlse1", t_global)

    # Phase 4: Constrained peak reduction again
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 4: CPR 2 (n=" + str(n_best) + ") ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 40:
        a_out, _ = constrained_peak_reduction(norm_sum1(best_seq), t4, verbose=True, seed=40)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "cpr2", t_global)

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 5: Multi-length Exploration ===")
    lengths_to_try = []
    # Try longer sequences
    for dl in [300, 800, 1500]:
        nl = n_best + dl
        if nl <= 12000:
            lengths_to_try.append(nl)
    # Try shorter
    if n_best - 500 >= 3000:
        lengths_to_try.append(n_best - 500)

    for new_n in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print("  Trying n=" + str(new_n))
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.30, 150.0)
        if t5a > 40:
            a_out, _ = fista_power_sum(a_long, t5a,
                                       p_start=40, p_end=2000, p_growth=1.008,
                                       eta_init=0.5, restart_every=500,
                                       verbose=False, seed=50)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_ps_n" + str(new_n), t_global)

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t5b = min(remaining * 0.25, 100.0)
            if t5b > 30:
                a_try = interpolate_sequence(best_seq, new_n)
                a_out, _ = constrained_peak_reduction(a_try, t5b, verbose=False, seed=60)
                best_seq, best_score = try_update(best_seq, best_score, a_out, "ml_cpr_n" + str(new_n), t_global)

    # Phase 6: Fine tuning
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 6: Fine Tuning (n=" + str(n_best) + ") ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        a_out, _ = fine_tune(norm_sum1(best_seq), t6, verbose=True, seed=70)
        best_seq, best_score = try_update(best_seq, best_score, a_out, "fine1", t_global)

    # Phase 7: Another FISTA pass with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 7: FISTA PS 2 (n=" + str(n_best) + ") ===")
    if remaining > 60:
        t7 = min(remaining * 0.20, 200.0)
        if t7 > 30:
            a_out, _ = fista_power_sum(norm_sum1(best_seq), t7,
                                       p_start=200, p_end=10000, p_growth=1.015,
                                       eta_init=0.3, restart_every=500,
                                       verbose=True, seed=80)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fista_ps2", t_global)

    # Phase 8: CPR final
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 8: CPR Final (n=" + str(n_best) + ") ===")
    if remaining > 40:
        t8 = min(remaining * 0.25, 180.0)
        if t8 > 20:
            a_out, _ = constrained_peak_reduction(norm_sum1(best_seq), t8, verbose=True, seed=90)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "cpr_final", t_global)

    # Phase 9: Final fine tuning
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print("\n=== Phase 9: Fine Tune Final (n=" + str(n_best) + ") ===")
    if remaining > 30:
        t9 = min(remaining * 0.5, 120.0)
        if t9 > 15:
            a_out, _ = fine_tune(norm_sum1(best_seq), t9, verbose=False, seed=100)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fine_final", t_global)

    # Phase 10: One last Nesterov burst if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print("\n=== Phase 10: Final NLSE (n=" + str(n_best) + ") ===")
        t10 = min(remaining * 0.6, 150.0)
        if t10 > 30:
            a_out, _ = nesterov_lse(norm_sum1(best_seq), t10,
                                    beta_start=500.0, beta_end=10000.0,
                                    beta_growth=1.02, eta_base=0.2,
                                    top_k_ratio=0.005, restart_every=600,
                                    verbose=True, seed=110)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "nlse_final", t_global)

    # Final fine tune
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t11 = min(remaining * 0.8, 80.0)
        if t11 > 10:
            a_out, _ = fine_tune(norm_sum1(best_seq), t11, verbose=False, seed=120)
            best_seq, best_score = try_update(best_seq, best_score, a_out, "fine_ultimate", t_global)

    # Validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + ", n=" + str(len(best_seq)) + ", time=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 6700
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)

    # Ensure all values are valid
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    return [float(x) for x in best_seq]
