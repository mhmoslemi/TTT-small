"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses multi-peak soft-gradient descent, LP-based refinement with geometric
interpolation, adaptive temperature, and random perturbation restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return np.ones_like(a) * (target / len(a))
    return a * (target / S)


def multi_peak_gradient(a, tau=10.0):
    """
    Compute gradient of soft-max autocorrelation using FFT-based cross-correlation.
    Uses softmax weighting over all lags with temperature tau.
    """
    n = len(a)
    c = np.convolve(a, a)

    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # Compute via cross-correlation
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]

    return grad


def hard_gradient(a):
    """Compute gradient at the single peak lag."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def lp_solve(f, M, time_limit=30):
    """
    Solve LP: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    if n > 1500:
        return None

    c_obj = -np.ones(n)

    # Build constraint matrix: A[m,j] = f[m-j] if 0 <= m-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, min(j + n, 2 * n - 1)):
            A[m, j] = f[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def geometric_interpolate(a, g, t):
    """
    Geometric interpolation between a and g (both non-negative).
    a_new[i] = a[i] * (g[i]/a[i])^t where a[i] > 0, else g[i]^t * scale
    This preserves non-negativity and makes large relative changes at small coords.
    """
    a_safe = np.maximum(a, 1e-15)
    g_safe = np.maximum(g, 1e-15)
    ratio = g_safe / a_safe
    a_new = a_safe * np.power(ratio, t)
    # Handle positions where a was zero
    zero_mask = a < 1e-10
    if np.any(zero_mask):
        a_new[zero_mask] = g[zero_mask] * t
    return a_new


def gradient_step(a, lr, momentum_buf, decay, tau=10.0):
    """One gradient descent step with momentum, projecting to sum and non-negativity."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a_curr = normalize(a, S_t)

    if tau is None:
        grad = hard_gradient(a_curr)
    else:
        grad = multi_peak_gradient(a_curr, tau)

    # Project out mean to maintain sum
    grad -= grad.mean()

    momentum_buf[:] = decay * momentum_buf - lr * grad
    a_new = a_curr + momentum_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)

    return a_new, momentum_buf


def coordinate_descent_polish(a, deadline, max_rounds=5):
    """
    Coordinate descent: for each coordinate, try perturbations guided by
    the autocorrelation structure.
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    c = np.convolve(a, a)
    patience = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        order = np.arange(n)
        np.random.shuffle(order)

        for idx, i in enumerate(order):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-15:
                # Try adding a small value
                candidates = [0.01, 0.05, 0.1]
            else:
                # Try based on autocorrelation pressure
                c = np.convolve(a, a)
                # Pressure at position i: how much does a[i] contribute to peaks?
                # a[i] contributes to C[k] for k in [i, i+n-1] as a[i]*a[k-i]
                pressure = 0.0
                for k in range(i, min(i + n, 2 * n - 1)):
                    j = k - i
                    if 0 <= j < n:
                        pressure += c[k] * a[j]

                scale = orig * pressure / max(pressure, 1e-10)
                candidates = [0.0, orig * 0.5, orig * 1.5, orig * 0.8, orig * 1.2]

            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = max(0.0, val)
                a = normalize(a, S_t)
                s = compute_obj(a)
                if s < best_s - 1e-12:
                    best_s = s
                    best_val = val
            a[i] = orig
            a = normalize(a, S_t)

            if best_val != orig:
                a[i] = max(0.0, best_val)
                a = normalize(a, S_t)
                best_score = best_s
                c = np.convolve(a, a)
                improved = True

        if not improved:
            patience += 1
            if patience >= 2:
                break
        else:
            patience = 0

    return a, best_score

            blocks = [shuffled[i:i+block_size] for i in range(0, target_n, block_size)]
            rng.shuffle(blocks)
            shuffled = np.concatenate(blocks)
            extra_starts.append(shuffled)

        for idx, a0 in enumerate(extra_starts):
            remaining = deadline - time.monotonic() - 30
            if remaining < 30:
                break
            t_alloc = min(120.0, remaining / max(1, len(extra_starts) - idx))
            if t_alloc < 20:
                break
            seq, score = optimize_from_start(a0, t_start + time.monotonic() - t_start + t_alloc)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [extra {idx}] New best: score={best_score:.6f}")
            else:
                print(f"  [extra {idx}] score={score:.6f}")

    # --- Phase D: Try a slightly different length ---
    if time.monotonic() < deadline - 60:
        alt_n = 512
        print(f"\nPhase D: Try n={alt_n}")
        a0 = np.maximum(best_seq + rng.random(alt_n - target_n) * 0.3, 0) if alt_n > target_n else best_seq[:alt_n]
        if len(a0) != alt_n:
            if len(a0) < alt_n:
                a0 = np.concatenate([a0, rng.random(alt_n - len(a0)) * 0.3 + 0.1])
            else:
                a0 = a0[:alt_n]

        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            seq, score = optimize_from_start(a0, t_start + time.monotonic() - t_start + min(remaining, 180))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Phase D (n={alt_n}): New best! score={best_score:.6f}")
            else:
                print(f"  Phase D (n={alt_n}): score={score:.6f}")

    # --- Phase E: Final intensive polish on the best ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\nPhase E: Final polish (score={best_score:.6f})")
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            # Focused gradient + LP
            n = len(best_seq)
            a = best_seq.copy()
            S_t = np.sqrt(2.0 * n)
            momentum_buf = np.zeros(n)

            # High-tau gradient to smooth out
            for it in range(5000):
                if time.monotonic() > deadline - 10:
                    break
                tau = 30.0 + 20.0 * it / 5000.0
                lr = max(0.003, 0.1 * (0.9997 ** it))
                a, momentum_buf = gradient_step(a, lr, momentum_buf, 0.85, tau=tau)
                if it % 500 == 0:
                    s = compute_obj(a)
                    if s < best_score:
                        best_score = s
                        best_seq = a.copy()

            # Quick LP
            if n <= 1024 and time.monotonic() < deadline - 10:
                c = np.convolve(best_seq, best_seq)
                M = float(c.max())
                g = lp_solve(best_seq, M * 0.99, time_limit=30)
                if g is not None and g.sum() > 1e-6:
                    g_norm = g * (S_t / g.sum())
                    for t_b in [0.1, 0.05, 0.02]:
                        a_new = (1 - t_b) * best_seq + t_b * g_norm
                        a_new = np.maximum(a_new, 0.0)
                        a_new = normalize(a_new, S_t)
                        s_new = compute_obj(a_new)
                        if s_new < best_score:
                            best_score = s_new
                            best_seq = a_new.copy()
                            print(f"  Final LP: score={best_score:.6f}")
                            break

            # Coordinate polish
            if n <= 512 and time.monotonic() < deadline - 5:
                a_pol, s_pol = coordinate_descent_polish(best_seq, deadline - 3.0, max_rounds=2)
                if s_pol < best_score:
                    best_score = s_pol
                    best_seq = a_pol.copy()
                    print(f"  Final coord: score={best_score:.6f}")

    # --- Final validation ---
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence score: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback to uniform sequence.")
    n = 384
    return [1.0] * n
