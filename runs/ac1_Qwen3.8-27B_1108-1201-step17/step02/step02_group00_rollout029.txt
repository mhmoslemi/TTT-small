For our problem (non-negative coefficients), the optimal sequence likely has a structure that makes A(x)^2 as flat as possible.

Let me try a different approach: use scipy's optimization to directly minimize the objective.

The objective is: f(a) = 2*n*max(conv(a,a))/sum(a)^2

This is not differentiable at points where the argmax changes, but we can handle this by considering active constraints.

At any point, let m* = argmax conv[m]. Then the subgradient of max(conv) with respect to a is related to the "cross-correlation" at lag m*.

Specifically, d(conv[m])/d(a[i]) = sum_j a[j] * delta(i+j, m) + sum_j a[j] * delta(j+i, m) = 2*sum_j a[j] * delta(i+j, m) (for the symmetric case) minus the diagonal term.

More precisely: conv[m] = sum_{i=0}^{min(m,n-1)} a[i]*a[m-i] (where we require m-i >= 0 and m-i <= n-1).

= a[m-i] + (a[m-i] if i < m-i else 0) + (a[m-i] if i > m-i else 0)
Actually: d(conv[m])/d(a[i]) = a[m-i] + (a[m-i] if m-2*i != 0 else 0)

Wait, let me be more careful. conv[m] = sum_{i+j=m, 0<=i,j<=n-1} a[i]*a[j].

d(conv[m])/d(a[k]) = sum_{j: k+j=m} a[j] + sum_{i: i+k=m} a[i] = a[m-k] + a[m-k] (if m-k is valid and m-2k != 0)
= 2*a[m-k] (if 0 <= m-k <= n-1 and k != m-k)
= a[m-k] (if 0 <= m-k <= n-1 and k == m-k, i.e., m = 2k)

So the gradient of max(conv) with respect to a[k] is:
- If m* is the argmax and m* - k is in [0, n-1]: 2*a[m*-k] (or a[m*-k] if m* = 2k)
- If m* is the argmax and m* - k is NOT in [0, n-1]: 0

This gives us a subgradient of the max function.

So the subgradient of the objective (with sum=1 constraint) is:
sg[k] = 2*n * (2*a[m*-k] or a[m*-k]) / sum^2 (for valid k)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smooth L-BFGS optimization
with log-sum-exp approximation of the max function, multi-start search,
and active-set refinement for final precision.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Fast auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_objective_and_grad(a, t):
    """
    Compute smooth approximation of 2*n*max(conv(a,a))/sum(a)^2 and its gradient.
    """
    n = len(a)
    conv = fast_convolve(a)
    sum_a = a.sum()
    
    # Log-sum-exp for numerical stability
    c = conv / t
    c_max = c.max()
    log_sum_exp = c_max + np.log(np.sum(np.exp(c - c_max)))
    smooth_max = t * log_sum_exp
    
    # Softmax weights
    probs = np.exp(c - c_max)
    probs /= probs.sum()
    
    # Gradient of smooth_max w.r.t. a
    # d(smooth_max)/d(a[k]) = t * sum_m probs[m] * d(conv[m])/d(a[k])
    # d(conv[m])/d(a[k]) = 2*a[m-k] (if 0<=m-k<=n-1 and m!=2k) or a[m-k] (if m==2k) or 0
    
    # Efficient computation using correlation
    # sum_m probs[m] * a[m-k] for valid m,k
    # This is the cross-correlation of probs and a
    # (probs ⋆ a)[k] = sum_m probs[m] * a[m-k]
    
    # Using FFT for cross-correlation:
    # np.correlate(probs, a, 'full') gives (a ⋆ probs) which is what we want
    # Actually np.correlate(a, b, 'full')[m] = sum_n a[n] * b[n-m+...] 
    # Let me just use the direct computation for correctness
    
    grad_smooth = np.zeros(n)
    for m in range(len(conv)):
        p = probs[m]
        if p < 1e-13:
            continue
        # For this m, valid k range: max(0, m-n+1) <= k <= min(n-1, m)
        k_start = max(0, m - n + 1)
        k_end = min(n - 1, m)
        if k_start > k_end:
            continue
        # d(conv[m])/d(a[k]) for k in [k_start, k_end]:
        # = 2*a[m-k] unless m-k == k (i.e., m == 2k), in which case = a[m-k]
        indices = np.arange(k_start, k_end + 1)
        vals = 2.0 * probs[m] * a[m - indices]
        # Fix diagonal: where m - k == k, i.e., k == m/2
        diag_pos = m // 2
        if m % 2 == 0 and k_start <= diag_pos <= k_end:
            vals[diag_pos - k_start] *= 0.5
        grad_smooth[indices] += vals
    
    # t * grad_smooth
    grad_smooth *= t
    
    # Objective and its gradient
    obj = 2.0 * n * smooth_max / (sum_a ** 2)
    grad = 2.0 * n * grad_smooth / (sum_a ** 2) - 4.0 * n * smooth_max * a / (sum_a ** 3)
    
    return obj, grad

def optimize_smooth(a0, t_schedule=None, max_iter=500, time_limit=None):
    """
    Optimize using L-BFGS with smooth approximation.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    
    # Normalize
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    if t_schedule is None:
        t_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    t0 = time.monotonic()
    
    for t in t_schedule:
        if time_limit and time.monotonic() - t0 > time_limit:
            break
        
        result = minimize(
            lambda x: smooth_objective_and_grad(x, t)[0],
            a,
            jac=lambda x: smooth_objective_and_grad(x, t)[1],
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        
        a = result.x
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        
        # Check actual objective
        actual_obj = 2.0 * n * fast_convolve(a).max()
        if actual_obj < best_obj:
            best_obj = actual_obj
            best_a = a.copy()
    
    return best_a, best_obj

def active_set_refine(a, n_peaks=5, tol=1e-4):

        a = a / s

    for _ in range(50):
        conv = fast_convolve(a)
        max_val = conv.max()

            diag_pos = m // 2

            sg[indices] += vals

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    # g1[k] = 2 * cross[k] for k = 0..n-1
    # g2[k] = probs[2k] * a[k] for k = 0..n-1
    grad_max = 2.0 * cross[:n] - probs[0::2][:n] * a
    grad_max *= t
    
    # Full objective and gradient
    # f = 2n * smooth_max / sum_a^2
    obj = 2.0 * n * smooth_max / (sum_a * sum_a)
    grad = (2.0 * n * grad_max / (sum_a * sum_a) - 
            4.0 * n * smooth_max * a / (sum_a * sum_a * sum_a))
    
    return obj, grad

    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()
