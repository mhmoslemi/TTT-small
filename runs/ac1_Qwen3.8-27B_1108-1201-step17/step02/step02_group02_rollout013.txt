- For each coordinate j, find the value that minimizes the peak (1D optimization).
- This is like coordinate descent on the peak.

For the 1D optimization at coordinate j:
- Fix all other coordinates.
- The peak as a function of a[j] is: max_k (c[k] without j's contribution + a[j] * (contribution of a[j] to c[k]))
- The contribution of a[j] to c[k] is 2*a[k-j] (from our gradient formula).
- So c[k](a[j]) = c[k]_other + 2*a[j]*a[k-j] (for the relevant k).
- The peak is max_k (c[k]_other + 2*a[j]*a[k-j]).
- This is a piecewise-linear function of a[j] (max of linear functions), so the minimum is at an intersection of two pieces or at a boundary.

This is a nice property! The 1D optimization can be solved exactly.

Let me implement this:
For coordinate j, the peak as a function of x = a[j] is:
`f(x) = max_k (alpha_k + beta_k * x)` where alpha_k = c[k] - 2*a[j]*a[k-j] (c[k] without x's contribution) and beta_k = 2*a[k-j] (the coefficient of x).

Wait, c[k] = sum_i a[i]*a[k-i]. The terms involving a[j] are:
- When i=j: a[j]*a[k-j] (if 0<=k-j<n)
- When k-i=j, i.e., i=k-j: a[k-j]*a[j] (if 0<=k-j<n)

So c[k] = (sum of terms not involving a[j]) + a[j]*a[k-j] + a[k-j]*a[j] (if k-j valid)
= c[k]_base + 2*a[j]*a[k-j] (if 0<=k-j<n)
= c[k]_base (if k-j not in [0,n-1])

where c[k]_base = c[k] - 2*a[j]*a[k-j] (if 0<=k-j<n), else c[k].

So f(x) = max_k (c[k]_base + 2*x*a[k-j] * 1_{0<=k-j<n})

This is the max of a set of linear functions of x, so it's convex and piecewise linear. The minimum is at the intersection of two adjacent pieces, or at x=0 (boundary).

To find the minimum:
1. Evaluate f at x=0.
2. The slope at any point is 2*a[k*-j] where k* is the argmax.
3. If the slope is positive at x=0, the minimum is at x=0.
4. Otherwise, find where the slope changes sign.

"""
Gradient-based optimization for minimizing peak autocorrelation of non-negative sequences.
Uses analytical subgradients of the max-convolution objective with Adam momentum,
soft-max smoothing, coordinate descent polish, and simulated annealing for escaping local minima.
Key insight: the subgradient of max_k conv[a,a][k] w.r.t. a[j] is 2*a[k*-j] (averaged over peak lags),
enabling O(n log n) iterations vs O(n^3) LP solves in the previous approach.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute the evaluation score (fast, assumes a is already valid numpy array)."""
    a = np.maximum(a, 0.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def compute_subgradient(a, peak_frac=0.99):
    """Compute subgradient of max_k conv[a,a][k] w.r.t. a.
    Uses: d/d a[j] conv[a,a][k] = 2*a[k-j] if 0<=k-j<n else 0.
    Averages over lags where conv value >= peak_frac * max.
    """
    n = len(a)
    c = fft_convolve(a, a)
    max_c = np.max(c)
    if max_c <= 0:
        return np.zeros(n), c
    threshold = peak_frac * max_c
    peak_mask = c >= threshold
    peak_indices = np.where(peak_mask)[0]
    grad = np.zeros(n)
    for k in peak_indices:
        # grad[j] += 2*a[k-j] where 0 <= k-j < n
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # j ranges from lo to hi, and k-j ranges from k-hi to k-lo
        grad[lo:hi+1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= len(peak_indices)
    return grad, c


def soft_max_score_and_grad(a, p=50.0):
    """Compute soft-max of convolution and its gradient. Returns (M_soft, grad)."""
    n = len(a)
    c = fft_convolve(a, a)
    c = np.maximum(c, 1e-30)  # avoid log(0)
    log_c = np.log(c)
    max_log_c = np.max(log_c)
    # log_sum_exp for numerical stability
    log_sum = max_log_c + np.log(np.sum(np.exp(p * (log_c - max_log_c))))
    M_soft = np.exp(log_sum / p)
    # grad[j] = M_soft^(1-p) * sum_k c[k]^(p-1) * 2*a[k-j]
    # = 2 * M_soft^(1-p) * sum_k c[k]^(p-1) * a[k-j]
    weights = np.exp((p - 1) * (log_c - max_log_c))  # c[k]^(p-1) / exp((p-1)*max_log_c)
    scale = M_soft ** (1.0 - p) * np.exp((p - 1) * max_log_c)
    # Now compute sum_k weights[k] * a[k-j] for each j
    # This is a cross-correlation
    # Use FFT: correlate(weights, a)
    size = 1
    while size < len(weights) + n - 1:
        size <<= 1
    fw = np.fft.rfft(weights, size)
    fa = np.fft.rfft(a, size)
    # cross-correlation: ifft(fft(w) * conj(fft(a)))
    corr = np.fft.irfft(fw * np.conj(fa), size)
    # corr[j] = sum_k w[k] * a[k-j] for appropriate j
    # We need corr[n-1+j] for j=0..n-1 (with proper indexing)
    # Actually for correlate(w, a) with w length 2n-1 and a length n:
    # result has length 3n-2, and result[n-1+j] = sum_k w[k+j]*a[k]
    # But we want sum_k w[k]*a[k-j] = sum_m w[m+j]*a[m] (sub m=k-j)
    # This is the cross-correlation at lag j
    # np.correlate(w, a, 'full')[n-1+j] should give this
    grad = 2.0 * scale * corr[n-1:n-1+n]
    # Clip small negative values from FFT roundoff
    grad = np.maximum(grad, 0)  # Actually grad can be negative in theory, but for non-negative a it should be >= 0
    # Wait, grad[j] = 2 * scale * sum_k weights[k] * a[k-j] which is >= 0 since all terms are >= 0
    return M_soft, grad, c


def normalize_sequence(a, target_sum=None):
    """Normalize sequence to non-negative with given target sum."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    if target_sum is None:
        return a
    return a * (target_sum / s)


def adam_update(a, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer update. Returns (new_a, updated_m, updated_v)."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v


def coordinate_descent_polish(a, n_iters=200, deadline=None):
    """Polish using coordinate descent: for each coordinate, find optimal value."""
    n = len(a)
    best = a.copy()
    best_score = score_fast(best)
    rng = np.random.default_rng(123)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        for j in order:
            if deadline is not None and time.monotonic() > deadline:
                return best, best_score
            # Try different values for a[j]
            current_val = a[j]
            # Golden section search or simple multi-point
            candidates = [0.0, current_val * 0.5, current_val * 0.8, current_val,
                          current_val * 1.2, current_val * 1.5, current_val * 2.0]
            # Add some around the current best
            candidates = [max(0, min(1000, c)) for c in candidates]
            best_val = current_val
            best_s = score_fast(a)
            for cv in candidates:
                if cv == current_val:
                    continue
                a[j] = cv
                s = score_fast(a)
                if s < best_s:
                    best_s = s
                    best_val = cv
            a[j] = best_val
            if best_val != current_val:
                improved = True
                if best_s < best_score:
                    best_score = best_s
                    best = a.copy()
        if not improved:
            break
    return best, best_score


def optimize_subgradient(a_init, time_limit, rng, adaptive_p=True, verbose=False):
    """Main optimization loop using subgradient + Adam."""
    n = len(a_init)
    target_sum = np.sqrt(2.0 * n)
    a = normalize_sequence(a_init.copy(), target_sum)
    
    current_score = score_fast(a)
    best = a.copy()
    best_score = current_score
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Adaptive parameters
    lr = 0.05
    p_soft = 20.0  # soft-max parameter
    peak_frac = 0.98
    
    t0 = time.monotonic()
    stall_count = 0
    iter_count = 0
    patience = 5000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        iter_count += 1
        
        # Compute gradient (choose between hard subgradient and soft-max)
        if p_soft >= 100 or not adaptive_p:
            # Hard subgradient
            grad, c = compute_subgradient(a, peak_frac)
        else:
            # Soft-max gradient (smoother)
            M_soft, grad, c = soft_max_score_and_grad(a, p_soft)
        
        # Project gradient to remove scaling component (since objective is scale-invariant)
        # The "wasted" direction is the all-ones vector (scaling)
        grad_proj = grad - np.mean(grad)
        
        # Adam update
        a_new, m, v = adam_update(a, grad_proj, m, v, t, lr=lr)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sequence(a_new, target_sum)
        
        new_score = score_fast(a_new)
        
        if new_score < current_score:
            a = a_new
            current_score = new_score
            stall_count = 0
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
                if verbose and iter_count % 1000 == 0:
                    print(f"      iter={iter_count}, best={best_score:.6f}")
            # Slowly increase lr on improvement
            lr = min(0.1, lr * 1.001)
        else:
            stall_count += 1
            # Reduce learning rate
            if stall_count > 100:
                lr *= 0.999
            if stall_count > 500:
                lr *= 0.99
    
        # Adaptive soft-max parameter
        if adaptive_p:
            if stall_count < 100:
                p_soft = max(10, p_soft * 0.999)  # smoother when progressing
            else:
                p_soft = min(200, p_soft * 1.001)  # sharper when stuck
            peak_frac = min(0.999, 0.95 + 0.0001 * stall_count)
        
        # Annealing: random perturbation when stuck
        if stall_count > patience:
            # Large random perturbation
            perturbation = rng.normal(0, 0.05 * np.mean(a), n)
            a_perturbed = a + perturbation
            a_perturbed = np.maximum(a_perturbed, 0.0)
            a_perturbed = normalize_sequence(a_perturbed, target_sum)
            pert_score = score_fast(a_perturbed)
            if pert_score < current_score * 1.001:  # accept if close
                a = a_perturbed
                current_score = pert_score
            else:
                # Smaller perturbation
                perturbation = rng.normal(0, 0.01 * np.mean(a), n)
                a_perturbed = a + perturbation
                a_perturbed = np.maximum(a_perturbed, 0.0)
                a_perturbed = normalize_sequence(a_perturbed, target_sum)
                a = a_perturbed
                current_score = score_fast(a)
            stall_count = 0
            patience = max(1000, patience * 0.8)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = max(0.01, lr * 0.5)
        
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, "
                  f"lr={lr:.5f}, p={p_soft:.0f}, elapsed={elapsed:.1f}s")
    
    return best, best_score


def make_init_chirp(n, alpha, rng):
    """Create a chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.01)


def make_init_multitone(n, rng, n_tones=5):
    """Create a multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, n_tones + 1):
        amp = 0.3 / k
        freq = rng.uniform(0.5, 3.0) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.maximum(a, 0.01)


def make_init_uniform_with_noise(n, rng, noise_level=0.1):
    """Uniform with noise."""
    a = np.ones(n) + noise_level * rng.standard_normal(n)
    return np.maximum(a, 0.01)


def make_init_geometric(n, ratio, rng):
    """Geometric decay initialization."""
    a = ratio ** np.arange(n)
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 0.001)


def resample_linear(seq, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    x_old = np.linspace(0, 1, len(seq), endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, seq)
