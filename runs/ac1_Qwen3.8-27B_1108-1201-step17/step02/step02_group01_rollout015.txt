Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Combines LP-based iterative refinement with

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'primal_feasibility_tolerance': 1e-10,
                               'dual_feasibility_tolerance': 1e-10})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return np.zeros(n)


def lp_refine(a, time_limit, verbose=True, max_iters=5000):
    """
    LP-based iterative refinement with careful convergence.
    Solves the LP, finds optimal blend via bisection, and iterates.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))

    for iteration in range(max_iters):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 3.0:
            break

        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        if M < best_conv_max - 1e-12:
            best_conv_max = M
            best_a = a.copy()

        # Solve LP
        g = solve_lp(a, M)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            if verbose:
                print(f"    LP iter {iteration}: g_sum too small, stopping")
            break

        g = normalize_seq(g, S)

        # Bisection for best blend parameter t
        lo, hi = 0.0, 1.0
        best_t = 0.0
        for _ in range(18):
            if time.monotonic() - t0 > time_limit - 2.0:
                break
            mid = (lo + hi) / 2.0
            b = (1.0 - mid) * a + mid * g
            b = np.clip(b, 0.0, None)
            b = normalize_seq(b, S)
            b_conv_max = np.max(fft_convolve(b, b))
            if b_conv_max <= M * (1.0 + 1e-9) + 1e-12:
                lo = mid
                best_t = mid
            else:
                hi = mid

        if best_t < 1e-7:
            if verbose and iteration % 50 == 0:
                print(f"    LP iter {iteration}: best_t={best_t:.2e}, conv_max={best_conv_max:.8f}")
            break

        a = (1.0 - best_t) * a + best_t * g
        a = np.clip(a, 0.0, None)
        a = normalize_seq(a, S)

        new_conv_max = np.max(fft_convolve(a, a))
        if new_conv_max < best_conv_max - 1e-12:
            best_conv_max = new_conv_max
            best_a = a.copy()

        if verbose and iteration % 100 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    LP iter {iteration}: conv_max={best_conv_max:.8f}, score={score:.8f}, t={best_t:.4f}, elapsed={time.monotonic()-t0:.0f}s")

        # Check for very small improvement (convergence)
        if best_t < 1e-5:
            if verbose:
                print(f"    LP iter {iteration}: converged (t={best_t:.2e})")
            break

    return best_a


def momentum_gradient(a, time_limit, verbose=True):
    """
    Gradient descent with momentum to reduce the peak of autocorrelation.
    Uses subgradient at the current peak location.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()

    lr = 2.0 / n
    momentum = 0.85
    v = np.zeros(n)
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    stall = 0
    step = 0

    while time.monotonic() - t0 < time_limit - 2.0:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        # Compute subgradient of conv[peak_idx] w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j1 = peak_idx - i
            j2 = i - peak_idx
            val = 0.0
            if 0 <= j1 < n:
                val += a[j1]
            if 0 <= j2 < n:
                val += a[j2]
            grad[i] = val

        # Also add small component from mean to avoid degenerate solutions
        grad += 0.01 * np.ones(n)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize_seq(a_new, S)

        new_conv_max = np.max(fft_convolve(a_new, a_new))

        if new_conv_max < best_conv_max:
            best_conv_max = new_conv_max
            best_a = a_new.copy()
            a = a_new
            stall = 0
        elif new_conv_max < peak_val * 1.01:
            a = a_new
            stall += 1
        else:
            v *= 0.5
            lr *= 0.95
            stall += 1

        if stall > 150:
            lr *= 0.7
            v *= 0.1
            stall = 0

        step += 1
        if verbose and step % 500 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    Grad step {step}: score={score:.8f}, lr={lr:.5f}, elapsed={time.monotonic()-t0:.0f}s")

    return best_a


def anneal_peaks(a, time_limit, rng, verbose=True):
    """
    Simulated annealing focused on reducing the current peak.
    Perturbs top contributors to the peak and compensates elsewhere.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))

    T = 0.05
    T_min = 0.0005
    cooling = 0.9995
    iteration = 0

    while time.monotonic() - t0 < time_limit - 2.0:
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        # Compute contributions to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        # Select top contributors
        top_k = max(5, n // 15)
        top_idx = np.argsort(contribs)[-top_k:]

        # Generate perturbation
        candidate = a.copy()
        # Reduce top contributors with varying strength
        for idx in top_idx:
            strength = rng.uniform(0.01, 0.06)
            candidate[idx] *= np.exp(-strength)

        # Increase random other positions
        others = np.setdiff1d(np.arange(n), top_idx)
        if len(others) > 0:
            inc_count = max(5, n // 25)
            inc_idx = rng.choice(others, size=min(inc_count, len(others)), replace=False)
            candidate[inc_idx] *= np.exp(rng.uniform(0.01, 0.05, size=len(inc_idx)))

        candidate = np.clip(candidate, 0.0, None)
        candidate = normalize_seq(candidate, S)

        new_conv_max = np.max(fft_convolve(candidate, candidate))
        delta = new_conv_max - peak_val

        # Accept if better or with Boltzmann probability
        if delta < 0 or (T > T_min and rng.random() < np.exp(-max(delta, 0) / T)):
            a = candidate
            if new_conv_max < best_conv_max:
                best_conv_max = new_conv_max
                best_a = candidate.copy()

        T = max(T * cooling, T_min)
        iteration += 1

        if verbose and iteration % 2000 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    Anneal iter {iteration}: score={score:.8f}, T={T:.6f}, elapsed={time.monotonic()-t0:.0f}s")

    return best_a


def multi_peak_gradient(a, time_limit, verbose=True):
    """
    Gradient step that considers the top 3 peaks simultaneously.
    Reduces all significant peaks at once.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()

    lr = 1.5 / n
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    stall = 0

    while time.monotonic() - t0 < time_limit - 2.0:
        conv = fft_convolve(a, a)

        # Find top 3 peaks
        sorted_peaks = np.argsort(conv)[::-1][:3]
        # Weight by proximity to maximum
        weights = 1.0 / (1.0 + 0.5 * np.abs(conv[sorted_peaks] - conv[sorted_peaks[0]]) / max(conv[sorted_peaks[0]], 1e-10))

        # Compute combined gradient
        grad = np.zeros(n)
        for pk, w in zip(sorted_peaks, weights):
            for i in range(n):
                j1 = pk - i
                j2 = i - pk
                val = 0.0
                if 0 <= j1 < n:
                    val += a[j1]
                if 0 <= j2 < n:
                    val += a[j2]
                grad[i] += w * val

        grad /= np.sum(weights)

        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize_seq(a_new, S)

        new_conv_max = np.max(fft_convolve(a_new, a_new))

        if new_conv_max < best_conv_max:
            best_conv_max = new_conv_max
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            lr *= 0.97
            stall += 1
            if stall > 100:
                lr *= 0.8
                stall = 0

    return best_a


def interpolate_to_length(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate():
    """
    Multi-phase optimization combining LP refinement, gradient descent,
    and simulated annealing. Tries the provided sequence length and nearby
    lengths, with careful time management to ensure convergence.
    """
    t_global = time.monotonic()
    deadline = t_global + 1780.0  # 20s safety margin
    rng = np.random.default_rng(123)

    best_score = np.inf
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        a = np.clip(a, 0.0, 1000.0)
        s = compute_score(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()
            print(f"  >>> NEW BEST: score={s:.8f}, n={len(a)}, time={time.monotonic()-t_global:.0f}s")
            return True
        return False

    # Get provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        prov_score = compute_score(provided)
        print(f"Provided sequence: n={len(provided)}, score={prov_score:.8f}")
        update_best(provided)

    # Determine lengths to try
    lengths_to_try = []
    if provided is not None:
        n_prov = len(provided)
        lengths_to_try.append(n_prov)
        # Try a few nearby lengths
        for offset in [100, -100, 200, -50, 50]:
            n_try = n_prov + offset
            if 128 <= n_try <= 4000 and n_try not in lengths_to_try:
                lengths_to_try.append(n_try)

    # Add some structured lengths if we have time
    for n in [512, 768, 1024, 1280, 1536]:
        if n not in lengths_to_try and 128 <= n <= 4000:
            lengths_to_try.append(n)

    # Process lengths, prioritizing the provided one
    print(f"\nPlan: trying {len(lengths_to_try)} lengths: {lengths_to_try}")

    for length_idx, n in enumerate(lengths_to_try):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            print(f"  Not enough time remaining ({remaining:.0f}s) for more lengths")
            break

        print(f"\n{'='*60}")
        print(f"Length {n} (attempt {length_idx+1}/{len(lengths_to_try)}), remaining time: {remaining:.0f}s")
        print(f"{'='*60}")

        # Get initial sequence
        if provided is not None:
            if n == len(provided):
                a0 = provided.copy()
            else:
                a0 = interpolate_to_length(provided, n)
                a0 = np.clip(a0, 0.01, None)
        else:
            t = np.arange(n, dtype=np.float64)
            # Chirp initialization
            alpha = 0.3 + 0.2 * rng.random()
            a0 = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.7))
            a0 = np.clip(a0, 0.01, None)

        a0 = normalize_seq(a0)
        initial_score = compute_score(a0)
        print(f"  Initial score for n={n}: {initial_score:.8f}")
        update_best(a0)

        # Phase 1: LP refinement (most important, give it the most time)
        alloc_lp = min(400.0, remaining * 0.45)
        if alloc_lp > 30:
            print(f"\n  Phase 1: LP refinement (time={alloc_lp:.0f}s)")
            a = lp_refine(a0, time_limit=alloc_lp, verbose=True)
            update_best(a)

        remaining = deadline - time.monotonic()
        if remaining < 30:
            continue

        # Phase 2: Multi-peak gradient
        alloc_grad = min(150.0, remaining * 0.2)
        if alloc_grad > 20:
            print(f"\n  Phase 2: Multi-peak gradient (time={alloc_grad:.0f}s)")
            a2 = multi_peak_gradient(a, time_limit=alloc_grad, verbose=True)
            if compute_score(a2) < compute_score(a):
                a = a2
                update_best(a)

        remaining = deadline - time.monotonic()
        if remaining < 30:
            continue

        # Phase 3: Simulated annealing
        alloc_ann = min(120.0, remaining * 0.2)
        if alloc_ann > 20:
            print(f"\n  Phase 3: Simulated annealing (time={alloc_ann:.0f}s)")
            a_ann = anneal_peaks(a, time_limit=alloc_ann, rng=rng, verbose=True)
            update_best(a_ann)
            if compute_score(a_ann) < compute_score(a):
                a = a_ann

        remaining = deadline - time.monotonic()
        if remaining < 30:
            continue

        # Phase 4: LP polish (second round)
        alloc_lp2 = min(200.0, remaining * 0.4)
        if alloc_lp2 > 20:
            print(f"\n  Phase 4: LP polish (time={alloc_lp2:.0f}s)")
            a = lp_refine(a, time_limit=alloc_lp2, verbose=True)
            update_best(a)

        remaining = deadline - time.monotonic()
        if remaining < 20:
            continue

        # Phase 5: Final momentum gradient
        alloc_grad2 = min(60.0, remaining * 0.3)
        if alloc_grad2 > 10:
            print(f"\n  Phase 5: Final gradient (time={alloc_grad2:.0f}s)")
            a = momentum_gradient(a, time_limit=alloc_grad2, verbose=True)
            update_best(a)

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = compute_score(np.array(final_list))
        print(f"\n{'='*60}")
        print(f"=== FINAL RESULT ===")
        print(f"    Score: {final_score:.8f}")
        print(f"    Length: {len(final_list)}")
        print(f"    Total time: {time.monotonic()-t_global:.1f}s")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    return [1.0] * 256
