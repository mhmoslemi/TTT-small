"""
Optimization of non-negative step sequences to minimize peak autocorrelation score.
Uses smoothed max (log-sum-exp) with L-BFGS in softmax parameterization,
combined with peak-targeted refinement and LP-based polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    A = rfft(a, size)
    B = rfft(b, size)
    return irfft(A * B, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def softmax(x):
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)


def smooth_objective_and_grad(x, T, entropy_weight=0.0):
    """
    Compute (1/T)*logsumexp(T*conv(softmax(x),softmax(x))) + entropy_weight * H(softmax(x))
    and its gradient w.r.t. x.
    """
    a = softmax(x)
    n = len(a)
    c = fft_convolve(a, a)
    
    # Log-sum-exp for numerical stability
    Tc = T * c
    mc = np.max(Tc)
    log_sum_exp = mc + np.log(np.sum(np.exp(Tc - mc)))
    obj = log_sum_exp / T
    
    # Entropy term
    if entropy_weight > 0:
        # H(a) = -sum a_i log(a_i)
        # In terms of x: sum a_i * x_i + log(sum exp(x_i))
        # But we want to MAXIMIZE entropy, so we subtract it
        # Gradient of -H wrt x: -1 + log(a_i) ... let me compute properly
        # dH/dx_i = -1 + log(a_i) [in the softmax parameterization]
        # Actually: H = -sum a_i log(a_i)
        # dH/dx_i = -sum_j (da_j/dx_i * (log(a_j) + 1))
        # da_j/dx_i = a_j(delta_{ji} - a_i)
        # dH/dx_i = -sum_j a_j(delta_{ji} - a_i)(log(a_j) + 1)
        # = -a_i(log(a_i) + 1) + a_i * sum_j a_j(log(a_j) + 1)
        # = -a_i(log(a_i) + 1) + a_i * (sum_j a_j log(a_j) + 1)
        # = -a_i * log(a_i)
        # So dH/dx_i = -a_i * log(a_i)
        # We want to maximize H, so we add -entropy_weight * H to minimize
        # d(-H)/dx_i = a_i * log(a_i)
        H_grad = a * (np.log(a + 1e-30))
        obj += -entropy_weight * (-(a * np.log(a + 1e-30)).sum())
        H_grad_vec = -entropy_weight * H_grad  # gradient of -entropy_weight*H
    else:
        H_grad_vec = np.zeros(n)
    
    # Gradient of (1/T)*logsumexp(T*c) w.r.t. a
    S_soft = np.exp(Tc - mc) / np.sum(np.exp(Tc - mc))
    
    grad_a = np.zeros(n)
    for j in range(n):
        # sum_k S_k * a_{k-j} for valid k
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        if k_start <= k_end:
            grad_a[j] += np.dot(S_soft[k_start:k_end+1], a[:k_end-k_start+1])
        # sum_k S_k * a_{j-k} for valid k
        k_start2 = max(0, j - n + 1)
        k_end2 = min(j, 2*n - 2)
        if k_start2 <= k_end2:
            grad_a[j] += np.dot(S_soft[k_start2:k_end2+1], a[j-k_end2:j-k_start2+1][::-1])
    grad_a /= T
    
    # Convert to gradient w.r.t. x: grad_x = a * (grad_a - dot(a, grad_a))
    grad_x = a * (grad_a - np.dot(a, grad_a))
    grad_x += H_grad_vec
    
    return obj, grad_x


def optimize_smooth(a_init, time_limit, T_schedule=None, entropy_weight=0.0, verbose=False):
    """
    Optimize using L-BFGS on smoothed max with softmax parameterization.
    """
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / S
    
    x = np.log(a + 1e-15)
    
    best_score = score_fast(a)
    best_a = a.copy()
    t0 = time.monotonic()
    
    if T_schedule is None:
        T_schedule = [10, 30, 100, 300, 500, 800]
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj_grad(x_eval):
            return smooth_objective_and_grad(x_eval, T, entropy_weight)
        
        try:
            res = minimize(obj_grad, x, jac=True, method='L-BFGS-B',
                          options={'maxiter': 500, 'gtol': 1e-6, 'maxfun': 2000})
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error at T={T}: {e}")
            break
        
        a_new = softmax(res.x)
        a_new = a_new / np.sum(a_new)
        s_new = score_fast(a_new)
        
        if verbose:
            print(f"    T={T:5.0f}: obj={res.fun:.8f}, score={s_new:.6f}, best={best_score:.6f}")
        
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
            x = res.x
        
        a = a_new
    
    return best_a, best_score


def peak_targeted_refine(a_init, time_limit, K=5, verbose=False):
    """
    Iteratively reduce the top-K peaks by adjusting contributing elements.
    """
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    it = 0
    alpha = 0.1  # step size
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        M = np.max(c)
        
        # Get top K peaks
        top_idx = np.argsort(c)[-K:]
        
        # Compute pressure on each element
        pressure = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            for j in range(n):
                idx1 = ki - j
                idx2 = j - ki
                contrib = 0.0
                if 0 <= idx1 < n:
                    contrib += a[j] * a[idx1]
                if 0 <= idx2 < n and idx2 != idx1:
                    contrib += a[j] * a[idx2]
                # Pressure is proportional to how much this element contributes to the peak
                if 0 <= ki - j < n:
                    pressure[j] += a[ki - j]
        
        # Normalize pressure
        p_max = np.max(pressure)
        if p_max < 1e-15:
            break
        
        # Reduce high-pressure elements, increase low-pressure ones
        a_new = a.copy()
        a_new *= np.exp(-alpha * pressure / p_max)
        a_new = np.clip(a_new, 1e-10, None)
        
        # Also add some random perturbation to escape local minima
        a_new *= np.exp(rng.normal(0, 0.01, n))
        
        # Renormalize
        a_new = a_new / np.sum(a_new)
        
        s_new = score_fast(a_new)
        
        if s_new < best_score - 1e-10:
            best_score = s_new
            best_a = a_new.copy()
            a = a_new
            # Adaptively decrease step size
            alpha = max(0.01, alpha * 0.98)
        else:
            a = a  # reject
            alpha = max(0.005, alpha * 0.95)
        
        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [peak {elapsed:6.1f}s] it={it:5d} score={s_new:.6f} best={best_score:.6f}")
    
    return best_a, best_score


def lp_refine(a_init, time_limit, verbose=False):

    n = len(a_init)
    a = a_init.astype(np.float64).copy()

    a = a / S

        c = fft_convolve(a, a)
        M = float(np.max(c))
        
        # Build constraint matrix: conv(a, b)_k = sum_j a_j * b_{k-j} <= M
        # This is linear in b: A_k @ b <= M for all k
        # A_k is a circulant-like matrix
        A = np.zeros((2*n-1, n))
        for j in range(n):
            # a_j * b_{k-j} contributes to position k
            # so b_j appears in positions j, j+1, ..., j+n-1 multiplied by a_{k-j}
            # Position k: sum_j a_{k-j} * b_j where k-j is in [0, n-1]
            # So b_j is multiplied by a_{k-j} at position k
            # A[k, j] = a_{k-j} if 0 <= k-j < n, else 0
            pass
        
        # Better way to build A:
        # (a*b)_k = sum_j a_j * b_{k-j}
        # So A[k, j] = a_j if k-j >= 0 and k-j < n, i.e., j <= k < j+n
        # Equivalently: A[k, j] = a_j if 0 <= k-j < n
        for j in range(n):
            A[j:j+n, j] = a

        b_ub = np.full(2*n-1, M)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n,

            if not res.success:
                break
            b_sol = res.x
        except Exception:
            break
        
        if np.sum(b_sol) < 1e-15:
            break

"""
Minimizes peak autocorrelation score of non-negative step sequences using:
1. Smoothed max (log-sum-exp) objective with L-BFGS in softmax parameterization
2. Peak-targeted multiplicative refinement
3. LP-based direction finding for final polish

Key innovation: uses the full softmax-weighted gradient (all peaks weighted by
proximity to max) rather than hard top-k selection, enabling smoother optimization.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def softmax(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)


def compute_grad_a(S_soft, a, T):
    """
    Compute grad_a[j] = (2/T) * sum_{m=0}^{n-1} S_soft[j+m] * a[m]
    where S_soft is the softmax of T*conv(a,a).
    """
    n = len(a)
    if n <= 4096:
        j_idx = np.arange(n)[:, None]
        m_idx = np.arange(n)[None, :]
        indices = j_idx + m_idx
        valid = indices < (2 * n - 1)
        S_window = np.where(valid, S_soft[indices], 0.0)
        return (2.0 / T) * (S_window @ a)
    else:
        grad = np.zeros(n)
        for j in range(n):
            m_max = min(n - 1, 2 * n - 2 - j)
            grad[j] = np.dot(S_soft[j:j + m_max + 1], a[:m_max + 1])
        return (2.0 / T) * grad


def smooth_obj_grad(x, T, entropy_w=0.0):
    """
    Objective: (1/T)*logsumexp(T*conv(softmax(x), softmax(x))) - entropy_w*H(softmax(x))
    Returns (value, gradient w.r.t. x).
    """
    a = softmax(x)
    n = len(a)
    c = fft_convolve(a, a)

    Tc = T * c
    mc = np.max(Tc)
    log_sum_exp = mc + np.log(np.sum(np.exp(Tc - mc)))
    val = log_sum_exp / T

    S_soft = np.exp(Tc - mc) / np.sum(np.exp(Tc - mc))

    grad_a = compute_grad_a(S_soft, a, T)

    # Convert to gradient w.r.t. x via softmax Jacobian
    grad_x = a * (grad_a - np.dot(a, grad_a))

    if entropy_w > 0:
        # H(a) = -sum a_i log(a_i), dH/dx_i = -a_i * log(a_i)
        # We want to maximize H, so subtract entropy_w * H from objective
        # d(entropy_w*H)/dx_i = -entropy_w * a_i * log(a_i)
        # So we add this to grad_x for minimization
        log_a = np.log(np.maximum(a, 1e-30))
        val += entropy_w * (a * log_a).sum()
        grad_x += entropy_w * a * log_a

    return val, grad_x


def optimize_smooth(a_init, time_limit, T_schedule=None, entropy_w=0.0, verbose=False):
    """
    Optimize using L-BFGS on smoothed max with softmax parameterization.
    """
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / S

    x = np.log(np.maximum(a, 1e-15))
    best_score = score_fast(a)
    best_a = a.copy()
    t0 = time.monotonic()

    if T_schedule is None:
        n = len(a)
        if n <= 320:
            T_schedule = [10, 30, 80, 200, 400, 700]
        elif n <= 512:
            T_schedule = [10, 30, 80, 200, 400]
        else:
            T_schedule = [10, 30, 80, 200]

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break

        def obj_grad(x_eval):
            return smooth_obj_grad(x_eval, T, entropy_w)

        try:
            res = minimize(obj_grad, x, jac=True, method='L-BFGS-B',
                           options={'maxiter': 300, 'gtol': 1e-7, 'maxfun': 1500})
        except Exception:
            break

        a_new = softmax(res.x)
        a_new = a_new / np.sum(a_new)
        s_new = score_fast(a_new)

        if verbose:
            print(f"    T={T:5.0f}: lse_val={res.fun:.8f}, score={s_new:.6f}, best={best_score:.6f}")

        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
            x = res.x
        else:
            # Keep previous x if this T didn't help
            pass

    return best_a, best_score


def peak_targeted_refine(a_init, time_limit, K=5, verbose=False):
    """
    Iteratively reduce top-K peaks using multiplicative (pressure-based) updates.
    """
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S
    n = len(a)

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    alpha = 0.15
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break

        c = fft_convolve(a, a)
        M = np.max(c)

        # Get top K peaks
        top_idx = np.argsort(c)[-K:]

        # Compute pressure: for each element j, how much does it contribute to top peaks?
        pressure = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            for j in range(n):
                if 0 <= ki - j < n:
                    pressure[j] += a[ki - j]

        p_max = np.max(pressure)
        if p_max < 1e-15:
            break

        # Multiplicative update: reduce high-pressure elements
        a_new = a * np.exp(-alpha * pressure / p_max)
        a_new = np.maximum(a_new, 1e-10)

        # Add small random perturbation
        a_new *= np.exp(rng.normal(0, 0.005, n))
        a_new = a_new / np.sum(a_new)

        s_new = score_fast(a_new)
        it += 1

        if s_new < best_score - 1e-12:
            best_score = s_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            alpha = min(0.2, alpha * 1.02)
        else:
            stall += 1
            alpha = max(0.02, alpha * 0.97)
            if stall > 200:
                # Restart with perturbation
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                a = a / np.sum(a)
                stall = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [peak {elapsed:6.1f}s] it={it:5d} score={s_new:.6f} best={best_score:.6f}")

    return best_a, best_score


def lp_refine(a_init, time_limit, verbose=False):
    """LP-based direction finding for final polish."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 4:
            break

        c = fft_convolve(a, a)
        M = float(np.max(c))

        # Build A matrix: A[k, j] = a_j if 0 <= k-j < n
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a

        # Solve: max sum(b) s.t. A@b <= M*ones, b >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': 3.0, 'presolve': True})
            if not res.success:
                break
            b_sol = res.x
        except Exception:
            break

        if np.sum(b_sol) < 1e-15:
            break

        b_sol = b_sol * (np.sum(a) / np.sum(b_sol))

        # Line search
        best_t = 0.05
        best_s = np.inf
        for t_try in [0.02, 0.05, 0.1, 0.15, 0.25]:
            a_cand = (1 - t_try) * a + t_try * b_sol
            a_cand = np.clip(a_cand, 0.0, None)
            if np.sum(a_cand) < 1e-10:
                continue
            s_cand = score_fast(a_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        a = (1 - best_t) * a + best_t * b_sol
        a = np.clip(a, 0, None)
        a = a / np.sum(a)
        it += 1

        if best_s < best_score - 1e-11:
            best_score = best_s
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [LP {elapsed:6.1f}s] it={it:4d} score={best_s:.6f} best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find optimal value given others fixed.
    Uses the fact that the peak is piecewise-linear in each coordinate.
    """
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S
    n = len(a)

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()

    # Precompute the "background" convolution (conv of a with a, excluding each element)
    for sweep in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        c = fft_convolve(a, a)
        M = float(np.max(c))

        # For each coordinate j, the peak as a function of a_j is:
        # c_k(a_j) = bg_k + a_j * (a_{k-j} + a_{j-k}) for the peak position
        # where bg_k = c_k - a_j*(a_{k-j} + a_{j-k}) (the part not involving a_j)
        # We want a_j >= 0 that minimizes max_k [bg_k + a_j * w_k(j)]
        # where w_k(j) = a_{k-j} + a_{j-k}

        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 1:
                break

            old_val = a[j]
            # Compute the "weight" vector: w_k = dc_k/da_j = 2*a_{k-j} if valid
            w = np.zeros(2 * n - 1)
            for k in range(j, min(2 * n - 1, j + n)):
                w[k] = 2.0 * a[k - j]

            # bg = c - old_val * w (the part of conv not involving a_j)
            bg = c - old_val * w
            bg = np.maximum(bg, 0)  # numerical safety

            # Find a_j >= 0 minimizing max_k [bg_k + a_j * w_k]
            # The peak is max_k [bg_k + a_j * w_k]
            # If w_k <= 0 for all k with bg_k > 0, the peak doesn't increase with a_j
            # Otherwise, we need to find the crossing point

            # Binary search on a_j
            # Upper bound: old_val * 2
            lo = 0.0
            hi = old_val * 3.0 + 0.01

            best_aj = old_val
            best_peak = float(np.max(bg))  # peak at a_j = 0

            for _ in range(30):
                mid = (lo + hi) / 2.0
                peak_mid = float(np.max(bg + mid * w))
                if peak_mid < best_peak:
                    best_peak = peak_mid
                    best_aj = mid
                    lo = mid
                else:
                    hi = mid

            # Also try a few specific values
            for cand in [old_val, old_val * 0.5, old_val * 1.5, best_aj]:
                if cand < 0:
                    continue
                peak_cand = float(np.max(bg + cand * w))
                if peak_cand < best_peak:
                    best_peak = peak_cand
                    best_aj = cand

            if best_aj != old_val:
                a[j] = best_aj
                improved = True

        # Renormalize
        a = np.maximum(a, 1e-12)
        a = a / np.sum(a)
        s = score_fast(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()

        if not improved and sweep > 10:
            break

    return best_a, best_score


def make_initialization(name, n, rng):
    """Generate various initializations."""
    t = np.arange(n, dtype=np.float64)
    if name == 'uniform':
        return np.ones(n) / n
    elif name == 'hann':
        w = 0.5 * (1 + np.cos(2 * np.pi * t / max(1, n - 1)))
        w = np.maximum(w, 0.01)
        return w / np.sum(w)
    elif name == 'bartlett':
        w = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
        w = np.maximum(w, 0.05)
        return w / np.sum(w)
    elif name == 'cosine':
        w = 0.5 + 0.5 * np.cos(np.pi * t / max(1, n - 1))
        w = np.maximum(w, 0.01)
        return w / np.sum(w)
    elif name == 'bump':
        center = n / 2.0
        w = np.exp(-((t - center) ** 2) / (2 * (n / 4) ** 2))
        w += 0.05
        return w / np.sum(w)
    elif name == 'random':
        r = rng.random(n) + 0.1
        return r / np.sum(r)
    elif name == 'chirp':
        w = 0.5 + 0.5 * np.cos(2 * np.pi * t * t / (2.0 * n))
        w = np.maximum(w, 0.05)
        return w / np.sum(w)
    elif name == 'two_peaks':
        c1, c2 = n * 0.3, n * 0.7
        w = np.exp(-((t - c1) ** 2) / (2 * (n / 8) ** 2)) + np.exp(-((t - c2) ** 2) / (2 * (n / 8) ** 2))
        w += 0.05
        return w / np.sum(w)
    else:
        return np.ones(n) / n
