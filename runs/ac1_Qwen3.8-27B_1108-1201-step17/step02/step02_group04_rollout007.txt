"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-guided direction finding, smooth approximation with L-BFGS-B,
active-set identification, and simulated annealing restarts.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_gradient_at_max(a, conv, top_k=10):
    """Compute gradient of max_k conv[k] w.r.t. a, using top-k approximation."""
    n = len(a)
    # Get top-k indices
    threshold = np.sort(conv)[-top_k:]
    active = conv >= threshold[0]
    
    # For each active k, gradient contribution to a[m] is 2*a[k-m] for valid m
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        k = int(k)
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            grad[m_start:m_end + 1] += 2.0 * a[k - m_end:k - m_start + 1][::-1]
    
    # Normalize by number of active constraints
    n_active = np.sum(active)
    if n_active > 0:
        grad /= n_active
    return grad

def smooth_objective(a, conv, t=50.0):
    """Smooth approximation of max(conv) using log-sum-exp."""
    m = conv.max()
    return m + np.log(np.exp((conv - m) / t).sum()) * t

def smooth_gradient(a, conv, t=50.0):
    """Gradient of smooth approximation."""
    n = len(a)
    m = conv.max()
    weights = np.exp((conv - m) / t)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(len(conv)):
        k = int(k)
        if weights[k] < 1e-10:
            continue
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            grad[m_start:m_end + 1] += weights[k] * 2.0 * a[k - m_end:k - m_start + 1][::-1]
    
    return grad

def lp_direction(f, M, n, timeout=10):
    """Solve LP to find direction g that maximizes sum(g) subject to (f*g)_k <= M."""
    rows_list = []
    cols_list = []
    vals_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
            vals_list.append(f[j])
    
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.array(vals_list, dtype=np.float64)
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def lp_optimize(f0, time_limit, verbose=False):
    """LP-based alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / f_sum  # normalize to simplex
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = lp_direction(f, M, n, timeout=min(10, time_limit - (time.monotonic() - t0)))
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-14:

        # Try different step sizes
        improved = False
        for t_try in [t, 0.8*t, 0.6*t, 0.4*t, 0.2*t, 0.1*t]:

            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:

            f_new = f_new / s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:

                M = conv_new.max()
