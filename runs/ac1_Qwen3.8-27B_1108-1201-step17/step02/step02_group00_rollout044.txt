import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        return np.zeros(n)
    
    # Softmax weights over the convolution peaks
    scaled = temperature * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # Only use indices with significant weight for efficiency
    thresh = w.max() * 0.01
    active = np.where(w > thresh)[0]
    
    sg = np.zeros(n)
    for k_int in active:
        k = int(k_int)
        wi = w[k_int] * 2.0
        if wi < 1e-14:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    
    return sg


def nesterov_subgrad(a0, time_limit, lr_init=0.05, top_temp=0.01,
                     temp_growth=1.003, perturb_every=5000, perturb_scale=0.004):
    """
    Nesterov-accelerated projected subgradient descent on the simplex.
    Minimizes max(a*a) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    a_prev = a.copy()
    t_k = 1.0
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = top_temp
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a + (t_k - 1.0) / t_next * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, temp)
        
        # Backtracking line search
        alpha = lr
        for _ in range(10):
            a_candidate = project_simplex(y - alpha * sg)
            conv_c = fast_convolve(a_candidate)
            obj_c = conv_c.max()
            # Accept if objective decreased (or we've tried enough)
            if obj_c < conv_y.max() + 1e-10 or alpha < lr * 1e-4:
                break
            alpha *= 0.5
        
        a_prev = a
        a = a_candidate
        conv = conv_c
        cur_obj = conv.max()
        
        t_k = t_next
        iteration += 1
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Schedule
        temp *= temp_growth
        
        # Periodic perturbation
        if no_improve >= perturb_every:
            rng = np.random.default_rng(iteration)
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            a_prev = a.copy()
            conv = fast_convolve(a)
            t_k = 1.0
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)
            lr *= 0.7
        
        # Adaptive learning rate
        if iteration % 1000 == 0:
            lr *= 0.95
    
    return best_a, best_obj


def nesterov_with_flatness(a0, time_limit, lr_init=0.04, temp_init=0.02,
                           flat_weight=0.1, flat_anneal=0.0001):
    """
    Nesterov subgradient with additional penalty on autocorrelation variance.
    This encourages flatter autocorrelation which can lead to lower peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    a_prev = a.copy()
    t_k = 1.0
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    iteration = 0
    flat_w = flat_weight
    
    while time.monotonic() - t0 < time_limit:
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a + (t_k - 1.0) / t_next * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        conv_y = fast_convolve(y)
        
        # Combined subgradient: max + flatness penalty
        sg_peak = compute_subgradient(y, conv_y, temp)
        
        # Flatness: penalty on variance of conv
        mean_conv = conv_y.mean()
        # Gradient of var(conv) w.r.t. a is complex; approximate with
        # gradient of sum((conv - mean)^2) which involves all conv values
        # Simplified: push towards uniform conv
        flat_thresh = 0.05
        above_mean = np.where(conv_y > mean_conv * (1 + flat_thresh))[0]
        
        sg_flat = np.zeros(n)
        for k_int in above_mean[:20]:  # Limit to top contributors
            k = int(k_int)
            excess = conv_y[k] - mean_conv
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg_flat[idx] += excess * y[k - idx]
        
        sg = sg_peak + flat_w * sg_flat
        
        alpha = lr
        for _ in range(8):
            a_candidate = project_simplex(y - alpha * sg)
            conv_c = fast_convolve(a_candidate)
            obj_c = conv_c.max()
            if obj_c < conv_y.max() + 1e-10 or alpha < lr * 1e-4:
                break
            alpha *= 0.5
        
        a_prev = a
        a = a_candidate
        conv = conv_c
        cur_obj = conv.max()
        
        t_k = t_next
        iteration += 1
        flat_w = max(0.0, flat_w - flat_anneal)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 4000:
            rng = np.random.default_rng(iteration)
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            a_prev = a.copy()
            conv = fast_convolve(a)
            t_k = 1.0
            no_improve = 0
            lr *= 0.75
        
        if iteration % 2000 == 0:
            lr *= 0.92
            temp *= 1.01
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_restarts=5):
    """
    Coordinate descent: try adjusting pairs of coordinates to reduce max autocorrelation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        # Try random pair updates
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            
            # Try moving mass from i to j (and vice versa)
            for direction in [1, -1]:
                epsilon = 0.001 * np.sqrt(a[i] * a[j] + 1e-20)
                a_try = a.copy()
                a_try[i] -= direction * epsilon
                a_try[j] += direction * epsilon
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                s_try = a_try.sum()
                a_try = a_try / s_try
                
                conv_try = fast_convolve(a_try)
                if conv_try.max() < best_obj - 1e-12:
                    a = a_try
                    conv = conv_try
                    best_obj = conv.max()
                    best_a = a.copy()
                    break
        
        # Occasional larger perturbation
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            fraction = rng.uniform(0.01, 0.1)
            target = rng.integers(0, n)
            a_try = a.copy()
            transfer = a_try[idx] * fraction
            a_try[idx] -= transfer
            a_try[target] += transfer
            if a_try[idx] >= 0:
                conv_try = fast_convolve(a_try)
                if conv_try.max() < best_obj:
                    a = a_try
                    best_obj = conv_try.max()
                    best_a = a.copy()
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc = np.maximum(rc, 0.01)
    inits['raised_cos'] = rc / rc.sum()
    
    # Hamming window
    ham = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    ham = np.maximum(ham, 0.01)
    inits['hamming'] = ham / ham.sum()
    
    # Triangular
    tri = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Bimodal (edges emphasized)
    bi = np.zeros(n)
    edge = max(2, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi = np.maximum(bi, 0.01)
    inits['bimodal'] = bi / bi.sum()
    
    # Bump
    bump = np.exp(-((x - n * 0.3) / (n * 0.2)) ** 2)
    inits['bump_left'] = bump / bump.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0.01)
            inits['parent_up'] = pp / pp.sum()
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0.01)
            inits['parent_ds'] = pp / pp.sum()
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search using Nesterov-accelerated subgradient
    with adaptive active sets, flatness regularization, and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass
    
    def try_sequence(a):
        nonlocal best_sequence, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_norm.copy()
            print(f"  New best: score={score:.6f}, n={len(a_norm)}")
        return score
    
    # Phase 1: Quick scan over lengths and initializations
    print("=== Phase 1: Quick Scan ===")
    lengths_to_try = [128, 256, 512]
    if parent is not None:
        lengths_to_try.insert(0, len(parent))
    
    scan_results = []
    phase1_deadline = time.monotonic() + 250.0
    
    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline - 20:
            break
        inits = generate_initializations(n, parent, rng)
        
        # Prioritize parent and promising inits
        priority = ['parent', 'parent_ds', 'parent_up', 'raised_cos', 'hamming',
                    'uniform', 'random', 'bimodal', 'triangle', 'bump_left']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in inits if nm not in priority]
        init_names = init_names[:5]
        
        time_per_init = max(15.0, 200.0 / (len(lengths_to_try) * 5))
        
        for name in init_names:
            if time.monotonic() > phase1_deadline - 10:
                break
            t_alloc = min(time_per_init, phase1_deadline - time.monotonic() - 5)
            if t_alloc < 8:
                break
            
            a0 = inits[name]
            t_start = time.monotonic()
            a_opt, obj_opt = nesterov_subgrad(
                a0, time_limit=t_alloc, lr_init=0.05, top_temp=0.005,
                temp_growth=1.005, perturb_every=3000, perturb_scale=0.005)
            
            score = try_sequence(a_opt)
            scan_results.append((score, n, name, a_opt.copy()))
            print(f"  Scan: n={n}, {name}, elapsed={time.monotonic()-t_start:.1f}s, score={score:.6f}")
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")
    
    # Phase 2: Deep optimization on top configs
    print("=== Phase 2: Deep Optimization ===")
    top_configs = []
    
    # Add best from scan
    for score, n, name, a_opt in scan_results[:4]:
        top_configs.append((n, name, a_opt))
    
    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))
    
    # Add parent-based configs at multiple lengths
    for n in [128, 256, 512]:
        inits_b = generate_initializations(n, parent, rng)
        for extra in ['parent', 'parent_ds', 'parent_up', 'random']:
            if extra in inits_b:
                top_configs.append((n, extra, inits_b[extra]))
    
    remaining = deadline - time.monotonic() - 80.0
    time_per_deep = max(20.0, remaining / max(1, len(top_configs) * 2))
    
    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 40)
        if t_alloc < 10:
            break
        
        # Try Nesterov subgradient
        t_single = t_alloc * 0.5
        if t_single > 5:
            a_opt, obj_opt = nesterov_subgrad(
                a0, time_limit=t_single, lr_init=0.04, top_temp=0.003,
                temp_growth=1.004, perturb_every=4000, perturb_scale=0.004)
            try_sequence(a_opt)
        
        # Try flatness-assisted optimization
        t_single = min(t_alloc * 0.5, deadline - time.monotonic() - 40)
        if t_single > 5:
            a_opt, obj_opt = nesterov_with_flatness(
                a0, time_limit=t_single, lr_init=0.03, temp_init=0.01,
                flat_weight=0.1, flat_anneal=0.00005)
            try_sequence(a_opt)
    
    print(f"Phase 2 done. Best: {best_score:.6f}")
    
    # Phase 3: Polish on the best sequence
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        n_best = len(best_sequence)
        
        # Multiple polish passes with different parameters
        polish_params = [
            {'lr_init': 0.01, 'top_temp': 0.001, 'temp_growth': 1.01},
            {'lr_init': 0.005, 'top_temp': 0.0005, 'temp_growth': 1.02},
            {'lr_init': 0.02, 'top_temp': 0.002, 'temp_growth': 1.008},
            {'lr_init': 0.008, 'top_temp': 0.001, 'temp_growth': 1.015},
        ]
        
        t_per = remaining / (len(polish_params) + 2)
        
        for cfg in polish_params:
            if remaining < 5:
                break
            a_opt, obj_opt = nesterov_subgrad(
                best_sequence, time_limit=min(t_per, deadline - time.monotonic() - 10), **cfg)
            try_sequence(a_opt)
            remaining = deadline - time.monotonic() - 10
        
        # Coordinate descent polish
        if remaining > 15:
            a_opt, _ = coordinate_descent(
                best_sequence, time_limit=min(remaining / 2, 60))
            try_sequence(a_opt)
            remaining = deadline - time.monotonic() - 10
        
        # Random perturbation + re-optimize
        if remaining > 10:
            for _ in range(4):
                if remaining < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                a_pert = project_simplex(best_sequence + noise)
                a_opt, obj_opt = nesterov_subgrad(
                    a_pert, time_limit=min(remaining / 4, 20),
                    lr_init=0.005, top_temp=0.001, temp_growth=1.01)
                try_sequence(a_opt)
                remaining = deadline - time.monotonic() - 10
        
        # Try flatness on best
        if remaining > 10:
            a_opt, obj_opt = nesterov_with_flatness(
                best_sequence, time_limit=min(remaining / 2, 40),
                lr_init=0.008, temp_init=0.005, flat_weight=0.05, flat_anneal=0.0002)
            try_sequence(a_opt)
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
