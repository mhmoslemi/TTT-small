import numpy as np
import time
from scipy.optimize import minimize, linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def compute_conv(a):
    return fft_convolve(a, a)

def local_score(a):
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * np.max(compute_conv(a)) / (sum_a ** 2)

def normalize(a, S=None):
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * S / s

def smooth_obj_and_grad(a, tau):
    """Compute smoothed objective and gradient."""
    n = len(a)
    sum_a = np.sum(a)
    conv = compute_conv(a)  # length 2n-1
    tau_conv = tau * conv
    max_tc = np.max(tau_conv)
    exp_tc = np.exp(tau_conv - max_tc)
    sum_exp = np.sum(exp_tc)
    lse = max_tc + np.log(sum_exp)
    p = exp_tc / sum_exp  # softmax weights
    
    # Objective: (1/tau) * lse / sum_a^2
    obj = lse / (tau * sum_a ** 2)
    
    # Gradient
    # d(lse)/da[j] = sum_k p[k] * 2*a[k-j] (for 0 <= k-j < n)
    # = 2 * sum_m p[m+j] * a[m]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    d_lse = 2.0 * np.fft.irfft(P * np.conj(A), N)[:n]
    
    # d(obj)/da[j] = (1/(tau * sum_a^2)) * d_lse[j] - 2 * lse / (tau * sum_a^3)
    grad = d_lse / (tau * sum_a ** 2) - 2.0 * lse / (tau * sum_a ** 3)
    
    return obj, grad

def smooth_optimize(a, time_limit, verbose=True):
    """Optimize using L-BFGS-B on the smoothed objective."""
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = local_score(a)
    
    # Animate tau from small to large
    taus = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    
    for tau in taus:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        def obj_func(x):
            x = np.maximum(x, 0.0)
            o, _ = smooth_obj_and_grad(x, tau)
            return o
        
        def grad_func(x):
            x = np.maximum(x, 0.0)
            _, g = smooth_obj_and_grad(x, tau)
            return g
        
        result = minimize(obj_func, a, jac=grad_func, method='L-BFGS-B',
                         bounds=[(0, 1000)] * len(a),
                         options={'maxiter': 200, 'maxfun': 2000,
                                  'ftol': 1e-10, 'gtol': 1e-8,
                                  'time_limit': max(1, remaining * 0.9)})
        
        if result.x is not None:
            a = np.maximum(result.x, 0.0)
            a = normalize(a)
            score = local_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if verbose:
                    print(f"    tau={tau}: score={best_score:.6f}")
    
    return best_a

def lp_refine_fast(a, time_limit, top_k=20, verbose=True):
    """LP refinement using only top K constraints."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(compute_conv(a))
    
    for iteration in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = compute_conv(a)
        M = np.max(conv_a)
        
        # Select top K constraints
        top_indices = np.argsort(conv_a)[-top_k:]
        
        # Build reduced constraint matrix
        A = np.zeros((top_k, n))
        for idx, k in enumerate(top_indices):
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    A[idx, j] = a[kj]
        
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(top_k, M),
                     bounds=[(0, None)] * n, method='highs',
                     options={'presolve': True, 'time_limit': 2.0})
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        g_norm = normalize(g)
        
        # Find best blend
        best_t = 0.0
        best_m = M
        for t_try in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            b = (1 - t_try) * a + t_try * g_norm
            b = np.maximum(b, 0.0)
            b = normalize(b, S)
            bm = np.max(compute_conv(b))
            if bm < best_m:
                best_m = bm
                best_t = t_try
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * g_norm
        a = np.maximum(a, 0.0)
        a = normalize(a, S)
        
        new_max = np.max(compute_conv(a))
        if new_max < best_conv_max - 1e-10:
            best_conv_max = new_max
            best_a = a.copy()
        
        if verbose and iteration % 50 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    LP iter {iteration}: conv_max={best_conv_max:.6f}, score={score:.6f}")
        
        if best_t < 0.01:
            # Try smaller steps
            improved = False
            for t_try in [0.005, 0.002, 0.001]:
                b = (1 - t_try) * a + t_try * g_norm
                b = np.maximum(b, 0.0)
                b = normalize(b, S)
                bm = np.max(compute_conv(b))
                if bm < best_conv_max - 1e-10:
                    a = b
                    best_conv_max = bm
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                break
    
    return best_a

def targeted_polish(a, time_limit, verbose=True):
    """Targeted local search to reduce the peak."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = local_score(a)
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]
        
        # Find elements that contribute to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        # Try reducing top contributors
        top_k = max(5, n // 10)
        top_idx = np.argsort(contribs)[-top_k:]
        
        for _ in range(10):
            candidate = a.copy()
            # Reduce some contributors
            n_reduce = max(2, top_k // 3)
            reduce_idx = rng.choice(top_idx, size=n_reduce, replace=False)
            for idx in reduce_idx:
                candidate[idx] *= np.exp(rng.normal(-0.03, 0.02))
            
            # Increase some non-contributors
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = max(3, n // 15)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.02, size=n_inc))
            
            candidate = np.maximum(candidate, 0.0)
            candidate = normalize(candidate, S)
            
            new_score = local_score(candidate)
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                a = candidate
            elif new_score < best_score * 1.001 and rng.random() < 0.2:
                a = candidate
        
        # Block perturbation
        if rng.random() < 0.3:
            width = max(2, n // 20)
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.02, size=width))
            candidate = np.maximum(candidate, 0.0)
            candidate = normalize(candidate, S)
            if local_score(candidate) < best_score:
                best_score = local_score(candidate)
                best_a = candidate.copy()
                a = candidate
    
    return best_a

Multi-phase optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Combines LP refinement, smoothed gradient

import numpy as np

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def self_conv(a):
    """Self-convolution a * a."""
    return fft_convolve(a, a)


def local_score(a):
    """Compute 2n * max(conv) / sum^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * np.max(self_conv(a)) / (s * s))


def normalize(a, S=None):
    """Normalize to have sum S."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def smooth_obj_grad(a, tau):
    """
    Compute smoothed objective and gradient.
    Objective: (1/tau) * log(sum exp(tau * conv(a,a))) / sum(a)^2
    """
    n = len(a)
    s = np.sum(a)
    conv = self_conv(a)
    tau_conv = tau * conv
    shift = np.max(tau_conv)
    exp_vals = np.exp(tau_conv - shift)
    sum_exp = np.sum(exp_vals)
    lse = shift + np.log(sum_exp)
    p = exp_vals / sum_exp  # softmax weights, length 2n-1
    
    obj = lse / (tau * s * s)
    
    # Gradient via FFT
    # d(lse)/da[j] = 2 * sum_m p[m+j] * a[m]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    d_lse = 2.0 * np.fft.irfft(P * np.conj(A), N)[:n]
    
    grad = d_lse / (tau * s * s) - 2.0 * lse / (tau * s * s * s)
    
    return obj, grad
