```
def project_simplex(y, S):
    y = np.asarray(y, dtype=np.float64).copy()
    n = y.size
    if n == 0: return y
    # If S <= 0: return zeros
    # Standard for target S:
    # v sorted descending.
    v = np.sort(y)[::-1]
    css = np.cumsum(v)
    # want rho = max index where v > (css - S)/rho? For target S.
    # For target 1: cond = v - (css - 1)/np.arange(1,n+1) >0
    # For target S: threshold tau=(css[t]-S)/t, condition v_t > tau_{t+1}? Let's use normalized to avoid errors.
    # Easiest: project to sum 1 then multiply S.
    if S == 0: return np.zeros_like(y)
    z = y / S
    v = np.sort(z)[::-1]
    cv = np.cumsum(v)
    inds = np.arange(1, n+1)
    cond = v - (cv - 1.0) / inds > 0
    if not np.any(cond):
        # all negative? return uniform?
        return np.full(n, 1.0/n) * S
    rho = np.where(cond)[0][-1] + 1
    theta = (cv[rho-1] - 1.0) / rho
    x = np.maximum(z - theta, 0.0)
    # due to numerical, renormalize
    sx = x.sum()
    if sx <= 0:
        x = np.full(n, 1.0/n)
    else:
        x = x * (1.0 / sx)
    return x * S
```

```python
"""
Search for a non-negative sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))^2.

The implementation starts from provided / constructed candidates and applies a
stack of complementary local derivatives:
  - softmax/log-sum-exp parametrization on the simplex with L-BFGS,
  - quadratic surrogate subproblems from weighted peak autocorrelation,
  - LP-based proposal blending,
  - projected soft-max gradient descent,
  - coordinate and stochastic local search.
"""

import numpy as np
import time
import math
import warnings

from scipy.optimize import minimize, linprog

warnings.filterwarnings('ignore')
np.seterr(over='ignore', under='ignore', invalid='ignore', divide='ignore')


def obj_norm(a, S):
    """Objective for an array assumed to have sum S."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0 or S is None or S <= 0.0:
        return float('inf')
    c = np.convolve(a, a)
    if c.size == 0:
        m = 0.0
    else:
        m = float(np.max(c))
    if not math.isfinite(m):
        return float('inf')
    return float(2.0 * a.size * m / (S * S))


def compute_obj(a):
    """Sanitized objective matching the evaluator."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    S = float(a.sum())
    if S <= 0.01:
        return float('inf')
    return obj_norm(a, S)


def normalize_seq(a, S=None):
    """Clip to non-negative finite values and scale to sum S."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    if S is None:
        S = math.sqrt(2.0 * float(a.size))
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    cur = float(a.sum())
    if cur <= 1e-15:
        return np.full(a.size, S / float(a.size))
    a = a * (S / cur)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    cur = float(a.sum())
    if cur <= 1e-15:
        return np.full(a.size, S / float(a.size))
    a = a * (S / cur)
    return a


def project_simplex(y, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    y = np.asarray(y, dtype=np.float64)
    n = y.size
    if n == 0:
        return y
    y = np.where(np.isfinite(y), y, 0.0)
    if S is None or S <= 0.0:
        return np.zeros(n, dtype=np.float64)

    z = y / float(S)
    v = np.sort(z)[::-1]
    cv = np.cumsum(v)
    inds = np.arange(1, n + 1, dtype=np.float64)
    cond = v - (cv - 1.0) / inds > 0.0

    if not np.any(cond):
        return np.full(n, S / float(n), dtype=np.float64)

    rho = int(np.where(cond)[0][-1]) + 1
    theta = (cv[rho - 1] - 1.0) / float(rho)
    x = np.maximum(z - theta, 0.0)
    sx = float(x.sum())
    if sx <= 1e-15:
        return np.full(n, S / float(n), dtype=np.float64)
    x = x * (1.0 / sx)
    return x * float(S)


def softmax_weights(c, tau):
    """Normalized non-negative weights concentrating on large c entries."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return c
    cmax = float(np.max(c))
    if not math.isfinite(cmax):
        cmax = 0.0

    if tau is None or tau <= 1e-12:
        return np.full(c.size, 1.0 / float(c.size))

    arg = float(tau) * (c - cmax)
    w = np.exp(arg)
    sw = float(w.sum())
    if sw <= 1e-15 or not math.isfinite(sw):
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / sw
    return w


def grad_from_weights(w, a):
    """
    Gradient of sum_k w_k * (a*a)_k with respect to a.
    w has length 2*n-1 and is aligned with the full convolution lags.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0:
        return a
    if w.size != 2 * n - 1:
        # Fallback for slightly mismatched lengths: zero out.
        return np.zeros(n, dtype=np.float64)

    a_rev = a[::-1]
    conv = np.convolve(w, a_rev, mode='full')
    start = n - 1
    out = np.zeros(n, dtype=np.float64)
    if 0 <= start and start + n <= conv.size:
        out[:] = conv[start:start + n]
    elif 0 <= start and start < conv.size:
        take = conv.size - start
        out[:take] = conv[start:start + take]
    return 2.0 * out


def z_from_a(a):
    """Map a non-negative vector to zero-mean log-probabilities."""
    p = np.asarray(a, dtype=np.float64)
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-15:
        p = np.full(p.size, 1.0 / float(p.size))
    else:
        p = p / s
    p = np.clip(p, 1e-15, None)
    s2 = float(p.sum())
    if s2 <= 1e-15:
        p = np.full(p.size, 1.0 / float(p.size))
    else:
        p = p / s2
    z = np.log(p)
    z = z - float(np.mean(z))
    return np.clip(z, -100.0, 100.0)


def a_from_z(z, S):
    """Map log-probabilities back to a non-negative vector with sum S."""
    z = np.asarray(z, dtype=np.float64)
    n = z.size
    if n == 0:
        return z
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-15 or not math.isfinite(s):
        return np.full(n, S / float(n), dtype=np.float64)
    return float(S) * (e / s)


def lbfgs_fun_grad(z, tau, reg=1e-10):
    """
    Smooth log-sum-exp objective in softmax coordinates.
    Returns (value, gradient) for SciPy L-BFGS-B with jac=True.
    """
    z = np.asarray(z, dtype=np.float64)
    n = z.size
    if n == 0:
        return 0.0, np.zeros(0, dtype=np.float64)

    S = math.sqrt(2.0 * float(n))
    mean_z = float(np.mean(z))
    z0 = z - mean_z
    z0 = z0 - float(np.max(z0))
    e = np.exp(z0)
    se = float(e.sum())
    if se <= 1e-15 or not math.isfinite(se):
        a = np.full(n, S / float(n), dtype=np.float64)
    else:
        a = S * (e / se)

    c = np.convolve(a, a)
    cmax = float(np.max(c)) if c.size else 0.0
    if not math.isfinite(cmax):
        cmax = 0.0

    tau = float(tau)
    if tau <= 1e-12:
        w = np.full(c.size, 1.0 / float(c.size), dtype=np.float64)
        val = float(np.mean(c)) if c.size else 0.0
    else:
        arg = tau * (c - cmax)
        w = np.exp(arg)
        sw = float(w.sum())
        if sw <= 1e-15 or not math.isfinite(sw):
            w = np.zeros(c.size, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
            val = float(cmax)
        else:
            w = w / sw
            val = float(cmax + math.log(sw) / tau)

    ga = grad_from_weights(w, a)
    scalar = 2.0 * float(np.dot(w, c))
    p = a / S
    gz = ga - p * scalar

    if reg > 0.0 and math.isfinite(mean_z):
        # A tiny regularization removes the all-ones null direction of softmax.
        gz = gz + (2.0 * reg * mean_z) * np.ones(n, dtype=np.float64)
        val = val + reg * mean_z * mean_z
    else:
        gz = gz - float(np.mean(gz))

    gz = np.where(np.isfinite(gz), gz, 0.0)
    if not math.isfinite(val):
        val = cmax
    return val, gz


def lbfgs_refine(a, deadline, rng=None):
    """L-BFGS on log-softmax coordinates with increasing tau."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a, float('inf')

    S = math.sqrt(2.0 * float(n))
    best = normalize_seq(a, S)
    best_score = obj_norm(best, S)

    if time.monotonic() >= deadline:
        return best, best_score

    if n <= 256:
        base_iters = 500
    elif n <= 512:
        base_iters = 300
    elif n <= 1024:
        base_iters = 150
    else:
        base_iters = 80

    tau = 1.0
    max_tau = 1024.0
    stall = 0
    stage = 0

    while time.monotonic() < deadline:
        remaining = float(deadline - time.monotonic())
        if remaining <= 10.0:
            break

        scale = remaining / 45.0
        if scale < 0.25:
            scale = 0.25
        if scale > 1.0:
            scale = 1.0
        maxiter = int(base_iters * scale)
        if maxiter < 20:
            maxiter = 20

        z0 = z_from_a(best)
        if rng is not None and stage >= 1:
            z0 = z0 + rng.normal(0.0, 0.03 * (stage + 1), n)

        try:
            res = minimize(
                lbfgs_fun_grad,
                z0,
                method='L-BFGS-B',
                jac=True,
                args=(tau,),
                options={
                    'maxiter': maxiter,
                    'ftol': 1e-12,
                    'gtol': 1e-10,
                    'maxls': 20,
                    'maxcor': 20,
                    'disp': False
                }
            )
        except Exception:
            res = None

        old_score = best_score
        if res is not None and hasattr(res, 'x') and res.x is not None:
            cand = a_from_z(res.x, S)
            s = obj_norm(cand, S)
            if math.isfinite(s) and s < best_score - 1e-12:
                best = cand
                best_score = s

        stall += 1
        if best_score < old_score - 1e-12:
            stall = 0

        if stall >= 5 or (tau >= max_tau and stall >= 2):
            break

        tau = min(max_tau, tau * 3.0)
        stage += 1

    return best, best_score


def build_hankel(w, support=None):
    """
    Build H with H[i,j] = w[i+j].
    This is the matrix of the quadratic surrogate sum_k w_k (a*a)_k.
    """
    w = np.asarray(w, dtype=np.float64)
    if support is None:
        n = (w.size + 1) // 2
        support = np.arange(n, dtype=np.int64)
    support = np.asarray(support, dtype=np.int64)
    m = support.size
    if m == 0:
        return np.empty((0, 0), dtype=np.float64)

    idx = support[:, None] + support[None, :]
    H = np.zeros((m, m), dtype=np.float64)
    mask = idx < w.size
    if np.any(mask):
        H[mask] = w[idx[mask]]
    return H


def surrogate_candidate(a, w, S):
    """
    For fixed weights w, approximately solve:
        minimize a^T H(w) a subject to sum(a)=S, a>=0.
    Uses an active-set iterative removal of negative unconstrained solutions.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return None

    threshold = max(1e-12, 1e-8 * float(S))
    support = np.where(a > threshold)[0]
    min_support = max(4, min(n, n // 4))
    if support.size < min_support:
        k = max(4, min(n, n // 2))
        support = np.argsort(a)[-k:]

    for _ in range(min(60, n + 1)):
        if support.size == 0:
            break

        H = build_hankel(w, support)
        m = H.shape[0]
        if m == 0:
            break

        scale = float(np.max(np.abs(H))) if H.size else 1.0
        if not math.isfinite(scale) or scale <= 0.0:
            scale = 1.0
        ridge = 1e-9 * scale + 1e-14
        A = H + ridge * np.eye(m)
        b = np.ones(m, dtype=np.float64)

        try:
            x = np.linalg.solve(A, b)
        except Exception:
            try:
                x = np.linalg.lstsq(A, b, rcond=None)[0]
            except Exception:
                break

        if not np.all(np.isfinite(x)):
            break

        max_abs = float(np.max(np.abs(x))) if x.size else 1.0
        if max_abs <= 0.0:
            break

        if float(np.min(x)) > -1e-6 * max_abs:
            x = np.maximum(x, 0.0)
            sx = float(x.sum())
            if sx <= 1e-14:
                break
            cand = np.zeros(n, dtype=np.float64)
            cand[support] = x * (S / sx)
            return cand

        bad_local = int(np.argmin(x))
        support = np.delete(support, bad_local)

    # Fallback: full support least-squares / pseudoinverse with positive projection.
    H = build_hankel(w, None)
    m = H.shape[0]
    if m > 0:
        scale = float(np.max(np.abs(H))) if H.size else 1.0
        if not math.isfinite(scale) or scale <= 0.0:
            scale = 1.0
        ridge = 1e-9 * scale + 1e-14
        try:
            x = np.linalg.lstsq(H + ridge * np.eye(m), np.ones(m), rcond=None)[0]
            if np.all(np.isfinite(x)):
                x = np.maximum(x, 0.0)
                sx = float(x.sum())
                if sx > 1e-14:
                    return x * (S / sx)
        except Exception:
            pass

    return None


def surrogate_refine(a, deadline, rng=None):
    """Iterate surrogate quadratic subproblems using peak-focused weights."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return a, compute_obj(a)

    S = math.sqrt(2.0 * float(n))
    best = normalize_seq(a, S)
    best_score = obj_norm(best, S)

    if time.monotonic() >= deadline:
        return best, best_score

    remaining = float(deadline - time.monotonic())
    max_iters = int(min(200, max(20, remaining * 2.0)))

    tau = 20.0
    stall = 0

    for _ in range(max_iters):
        if time.monotonic() >= deadline:
            break

        c = np.convolve(best, best)
        cmax = float(np.max(c)) if c.size else 0.0
        if not math.isfinite(cmax):
            break

        if stall == 0:
            w = softmax_weights(c, tau)
        elif stall == 1:
            top_k = min(8, max(1, c.size))
            idx = np.argsort(c)[::-1][:top_k]
            w = np.zeros_like(c)
            if idx.size:
                w[idx] = 1.0 / float(idx.size)
        elif stall == 2:
            w = softmax_weights(c, max(5.0, tau * 0.5))
        else:
            w = softmax_weights(c, tau)

        cand = surrogate_candidate(best, w, S)
        improved = False

        if cand is not None:
            for t in (1.0, 0.7, 0.4, 0.85, 0.55, 0.25, 0.1):
                trial = normalize_seq((1.0 - t) * best + t * cand, S)
                s = obj_norm(trial, S)
                if s < best_score - 1e-12:
                    best = trial
                    best_score = s
                    improved = True
                    break

        if improved:
            stall = 0
            tau = min(2000.0, tau * 1.25)
        else:
            stall += 1
            if stall == 4 and rng is not None:
                noise = rng.normal(0.0, 0.10, n)
                trial = normalize_seq(np.maximum(best + noise, 0.0), S)
                s = obj_norm(trial, S)
                if s < best_score - 1e-12:
                    best = trial
                    best_score = s
                    stall = 0
            if stall % 4 == 0:
                tau = max(5.0, tau * 0.5)
            if stall >= 9:
                break

    return best, best_score


def lp_blend_refine(a, deadline, max_solves=6):
    """
    For fixed f, solve LP:
        maximize sum(g)
        subject to f*g <= M, g >= 0.
    Then blend the LP proposal into f.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1 or n > 384:
        return a, compute_obj(a)

    S = math.sqrt(2.0 * float(n))
    best = normalize_seq(a, S)
    best_score = obj_norm(best, S)

    solves = 0
    stall = 0
    m_factor = 1.0

    while time.monotonic() < deadline and solves < max_solves:
        c = np.convolve(best, best)
        M = float(np.max(c)) if c.size else 0.0
        if not math.isfinite(M) or M <= 1e-15:
            break

        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = best

        b = np.full(2 * n - 1, M * m_factor, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(
                c_obj,
                A_ub=A,
                b_ub=b,
                bounds=bounds,
                method='highs',
                options={'presolve': True}
            )
        except Exception:
            break

        solves += 1

        if res is not None and getattr(res, 'success', False) and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.where(np.isfinite(g), g, 0.0)
            g = np.maximum(g, 0.0)
            gs = float(g.sum())
            if gs > 1e-9:
                g_norm = normalize_seq(g, S)
                improved = False
                for t in (0.3, 0.6, 0.15, 0.9, 0.45, 0.1):
                    trial = normalize_seq((1.0 - t) * best + t * g_norm, S)
                    s = obj_norm(trial, S)
                    if s < best_score - 1e-12:
                        best = trial
                        best_score = s
                        improved = True
                        break
                if improved:
                    stall = 0
                    m_factor = 1.0
                    continue

        stall += 1
        if stall == 1:
            m_factor = 0.98
        elif stall == 2:
            m_factor = 0.95
        else:
            m_factor = 1.0
            stall = 0
            # Small nudge with a peak-focused gradient.
            c = np.convolve(best, best)
            w = softmax_weights(c, 50.0)
            g = grad_from_weights(w, best)
            d = g - float(np.mean(g))
            trial = normalize_seq(best - 0.05 * d, S)
            s = obj_norm(trial, S)
            if s < best_score - 1e-12:
                best = trial
                best_score = s

    return best, best_score


def projected_gradient_refine(a, deadline, rng=None):
    """Projected gradient / momentum steps using soft-max peak weights."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return a, compute_obj(a)

    S = math.sqrt(2.0 * float(n))
    best = normalize_seq(a, S)
    best_score = obj_norm(best, S)

    v = np.zeros(n, dtype=np.float64)
    beta = 0.85
    alpha = 1.0
    tau = 10.0
    stall = 0
    v_cap = max(0.05, 10.0 * S / float(n))

    while time.monotonic() < deadline:
        c = np.convolve(best, best)
        w = softmax_weights(c, tau)
        g = grad_from_weights(w, best)
        d = g - float(np.mean(g))

        v = beta * v + d
        if not np.all(np.isfinite(v)):
            v = np.zeros(n, dtype=np.float64)
            d = np.zeros(n, dtype=np.float64)

        vn = float(np.linalg.norm(v))
        if vn > v_cap and vn > 0.0:
            v = v * (v_cap / vn)

        improved = False
        a_step = alpha
        for _ in range(8):
            y = best - a_step * v
            x = project_simplex(y, S)
            s = obj_norm(x, S)
            if s < best_score - 1e-12:
                best = x
                best_score = s
                improved = True
                break
            a_step *= 0.6

        if improved:
            stall = 0
            tau = min(2000.0, tau * 1.05)
        else:
            stall += 1
            v = 0.5 * v
            if a_step < 1e-4:
                alpha = 1.0
            elif a_step > 1.0:
                alpha = max(0.05, a_step * 0.5)
            else:
                alpha = max(0.01, a_step * 0.9)

            if stall > 8:
                beta = 0.5
                if tau < 100.0:
                    tau = min(2000.0, tau * 2.0)
                else:
                    tau = max(10.0, tau * 0.5)
                if rng is not None:
                    trial = normalize_seq(np.maximum(best + rng.normal(0.0, 0.05, n), 0.0), S)
                    s = obj_norm(trial, S)
                    if s < best_score - 1e-12:
                        best = trial
                        best_score = s
                    stall = 0
                if stall > 20:
                    break

    return best, best_score


def coordinate_refine(a, deadline, rng):
    """
    Deterministic coordinate search with candidate multipliers and a few
    random candidates. Each candidate is normalized and evaluated exactly.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return a, compute_obj(a)

    S = math.sqrt(2.0 * float(n))
    best = normalize_seq(a, S)
    best_score = obj_norm(best, S)

    if n <= 512:
        base_mults = (0.0, 0.5, 0.8, 1.25, 1.8)
        nrand = 2
        use_peer = True
    else:
        base_mults = (0.0, 0.5, 1.5)
        nrand = 1
        use_peer = False

    while time.monotonic() < deadline:
        order = rng.permutation(n)
        c = np.convolve(best, best)
        kstar = int(np.argmax(c)) if c.size else 0

        for ii in range(n):
            if time.monotonic() > deadline:
                break
            i = int(order[ii])
            orig = float(best[i])

            cands = [orig * m for m in base_mults]
            for _ in range(nrand):
                cands.append(orig * math.exp(rng.normal(0.0, 0.35 if n <= 512 else 0.5)))

            if use_peer:
                j = kstar - i
                if 0 <= j < n and j != i:
                    bv = float(best[j])
                    cands.append(0.2 * bv)
                    cands.append(0.5 * bv)
                    cands.append(0.8 * bv)

            best_val = orig
            best_local = best_score

            for val in cands:
                if not math.isfinite(val) or val < 0.0:
                    continue
                trial = best.copy()
                trial[i] = val
                trial = normalize_seq(trial, S)
                s = obj_norm(trial, S)
                if s < best_local - 1e-12:
                    best_local = s
                    best_val = float(val)

            if best_val != orig or best_local < best_score - 1e-12:
                best[i] = best_val
                best = normalize_seq(best, S)
                best_score = best_local

    return best, best_score


def stochastic_local_search(a, deadline, rng):
    """
    Fast random multiplicative perturbations.
    This gives many cheap exact evaluations and can escape coordinate minima.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 1:
        return a, compute_obj(a)

    S = math.sqrt(2.0 * float(n))
    best = normalize_seq(a, S)
    best_score = obj_norm(best, S)

    sigma = 0.20
    stall = 0
    cap = max(1, min(15, n))

    while time.monotonic() < deadline:
        k = int(rng.integers(1, cap + 1))
        idx = rng.choice(n, size=k, replace=False)
        factors = np.exp(rng.normal(0.0, sigma, size=k))

        trial = best.copy()
        trial[idx] = trial[idx] * factors
        if rng.random() < 0.03:
            trial[int(rng.integers(0, n))] = 0.0

        trial = normalize_seq(trial, S)
        s = obj_norm(trial, S)

        if s < best_score - 1e-12:
            best = trial
            best_score = s
            stall = 0
        else:
            stall += 1
            if stall > 400:
                sigma = max(0.02, sigma * 0.8)
                stall = 0
                if rng.random() < 0.2:
                    larger = normalize_seq(np.maximum(best + rng.normal(0.0, 0.25, n), 0.0), S)
                    sl = obj_norm(larger, S)
                    if sl < best_score - 1e-12:
                        best = larger
                        best_score = sl

    return best, best_score


def polish_sequence(a0, max_time, label, rng=None):
    """Run a stack of local improvements within max_time seconds."""
    max_time = float(max_time)
    a0 = np.asarray(a0, dtype=np.float64)
    if a0.size == 0:
        return a0, float('inf')
    if max_time <= 2.0:
        return normalize_seq(a0), compute_obj(a0)

    if rng is None:
        rng = np.random.default_rng(123456789)

    start = time.monotonic()
    total_end = start + max_time
    n = a0.size
    S = math.sqrt(2.0 * float(n))
    best = normalize_seq(a0, S)
    best_score = obj_norm(best, S)

    print(f"[{label}] start n={n} score={best_score:.10f} budget={max_time:.1f}s")

    now = time.monotonic()
    remaining = float(total_end - now)
    if n <= 1024 and max_time > 40.0 and remaining > 80.0:
        budget = min(0.45 * max_time, remaining - 60.0)
        if budget > 10.0:
            best, best_score = lbfgs_refine(best, now + budget, rng)
            print(f"[{label}] after L-BFGS score={best_score:.10f}")

    now = time.monotonic()
    remaining = float(total_end - now)
    if n <= 768 and remaining > 40.0:
        budget = min(0.18 * max_time, remaining - 30.0)
        if budget > 5.0:
            best, best_score = surrogate_refine(best, now + budget, rng)
            print(f"[{label}] after surrogate score={best_score:.10f}")

    now = time.monotonic()
    remaining = float(total_end - now)
    if n <= 384 and remaining > 50.0:
        budget = min(0.12 * max_time, remaining - 30.0)
        if budget > 5.0:
            max_solves = 3 if budget < 25.0 else 6
            best, best_score = lp_blend_refine(best, now + budget, max_solves=max_solves)
            print(f"[{label}] after LP score={best_score:.10f}")

    now = time.monotonic()
    remaining = float(total_end - now)
    if remaining > 20.0:
        budget = min(0.10 * max_time, remaining - 15.0)
        if budget > 2.0:
            best, best_score = projected_gradient_refine(best, now + budget, rng)
            print(f"[{label}] after projected grad score={best_score:.10f}")

    now = time.monotonic()
    remaining = float(total_end - now)
    if n <= 1024 and remaining > 15.0:
        budget = min(0.12 * max_time, remaining - 5.0)
        if budget > 2.0:
            best, best_score = coordinate_refine(best, now + budget, rng)
            print(f"[{label}] after coordinate score={best_score:.10f}")

    if time.monotonic() < total_end - 1.0:
        best, best_score = stochastic_local_search(best, total_end - 1.0, rng)
        print(f"[{label}] after stochastic score={best_score:.10f}")

    print(f"[{label}] done score={best_score:.10f}")
    return best, best_score


def update_best(best_seq, best_score, cand, label=None):
    """Validate and replace global best if candidate is better."""
    if cand is None:
        return best_seq, best_score
    arr = np.asarray(cand, dtype=np.float64)
    if arr.size == 0:
        return best_seq, best_score
    s = compute_obj(arr)
    if math.isfinite(s) and s < best_score - 1e-15:
        if label is not None:
            print(f"NEW BEST [{label}] score={s:.10f} n={arr.size}")
        return arr.copy(), s
    return best_seq, best_score


def make_candidates(rng):
    """Create a compact set of diverse initial candidates."""
    candidates = []

    h = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        h = None

    if h is not None and h.size > 10:
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.clip(h, 0.0, 1000.0)
        if float(h.sum()) > 1e-12:
            candidates.append(('height', h.copy()))

            for sig in (0.05, 0.10, 0.20):
                label = 'h_perturb_' + str(int(sig * 100))
                mult = np.exp(rng.normal(0.0, sig, h.size))
                candidates.append((label, np.maximum(h * mult, 0.0)))

            shifts = [h.size // 4, h.size // 3]
            for sh in shifts:
                candidates.append(('h_roll', np.roll(h, sh)))

            for L in (256, 320):
                if h.size >= L:
                    candidates.append(('h_trunc_' + str(L), h[:L].copy()))

            if h.size > 1:
                x_old = np.linspace(0.0, 1.0, h.size)
                for L in (320, 448, 512):
                    x_new = np.linspace(0.0, 1.0, L)
                    cand = np.interp(x_new, x_old, h)
                    candidates.append(('h_interp_' + str(L), cand))

    lengths = [256, 320, 384, 448, 512]
    for n in lengths:
        i = np.arange(n, dtype=np.float64)

        cand1 = np.abs(rng.normal(0.0, 1.0, n)) + 0.1
        candidates.append(('rand_a_' + str(n), cand1))

        cand2 = rng.random(n) * 0.5 + 0.25
        candidates.append(('rand_b_' + str(n), cand2))

        cand3 = 0.5 + 0.5 * np.cos(2.0 * np.pi * i * 3.0 / float(n))
        candidates.append(('cos3_' + str(n), cand3))

        if n in (384, 512):
            bump = (np.exp(-0.5 * ((i - 0.25 * n) / (0.15 * n)) ** 2) +
                    np.exp(-0.5 * ((i - 0.75 * n) / (0.15 * n)) ** 2) +
                    0.05)
            candidates.append(('bump_' + str(n), bump))

    return candidates


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Returns the best non-negative sequence found before
    the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    hard_deadline = start + max(60.0, float(budget_s) - 30.0)

    fallback = np.array([1.0] * 256, dtype=np.float64)
    best_seq = fallback.copy()
    best_score = compute_obj(best_seq)

    candidates = make_candidates(rng)
    scored = []
    for idx, (label, cand) in enumerate(candidates):
        arr = np.asarray(cand, dtype=np.float64)
        if arr.size == 0:
            continue
        s = compute_obj(arr)
        if math.isfinite(s):
            n = arr.size
            if n == 384:
                priority = 0
            elif n in (320, 448):
                priority = 1
            elif n in (256, 512):
                priority = 2
            else:
                priority = 3
            scored.append((float(s), int(priority), int(idx), str(label), arr.copy()))

    if not scored:
        return [float(x) for x in best_seq]

    scored.sort()

    processed = 0
    while processed < len(scored) and time.monotonic() < hard_deadline:
        s0, _priority, _idx, label, cand = scored[processed]
        remaining = float(hard_deadline - time.monotonic())
        reserve = 180.0 if processed < len(scored) - 1 else 30.0
        if remaining - reserve < 60.0:
            break

        if processed == 0:
            budget_c = min(700.0, remaining * 0.50)
        elif processed == 1:
            budget_c = min(350.0, remaining * 0.30)
        elif processed == 2:
            budget_c = min(200.0, remaining * 0.20)
        else:
            budget_c = min(120.0, remaining * 0.15)

        budget_c = max(30.0, budget_c)
        budget_c = min(budget_c, remaining - reserve)
        if budget_c < 10.0:
            break

        run_label = label + ' #d' + str(processed)
        seq, score = polish_sequence(cand, budget_c, run_label, rng)
        best_seq, best_score = update_best(best_seq, best_score, seq, run_label)
        processed += 1

    if time.monotonic() < hard_deadline - 60.0:
        remaining = float(hard_deadline - time.monotonic())
        budget_final = min(600.0, remaining * 0.70)
        if budget_final > 30.0:
            seq, score = polish_sequence(best_seq, budget_final, 'final_polish', rng)
            best_seq, best_score = update_best(best_seq, best_score, seq, 'final_polish')

    if time.monotonic() < hard_deadline - 2.0:
        seq, score = stochastic_local_search(best_seq, hard_deadline - 1.0, rng)
        best_seq, best_score = update_best(best_seq, best_score, seq, 'final_stochastic')

    final = np.asarray(best_seq, dtype=np.float64)
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0 or float(final.sum()) < 0.01:
        final = fallback.copy()
    final = normalize_seq(final)

    try:
        ev = float(evaluate_sequence([float(x) for x in final]))
        print(f"evaluate_sequence final score={ev:.10f} n={final.size}")
        if not math.isfinite(ev):
            final = fallback.copy()
    except Exception as exc:
        print(f"evaluate_sequence unavailable or failed: {exc}")
        ev = compute_obj(final)
        if not math.isfinite(ev):
            final = fallback.copy()

    return [float(x) for x in final]
```
