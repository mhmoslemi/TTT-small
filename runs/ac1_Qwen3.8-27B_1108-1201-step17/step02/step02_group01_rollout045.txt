"""
Optimized step function coefficient search using LP-based directional 
improvement, gradient refinement, and adaptive peak-spreading local search.
Improves on previous approach by using relaxed LP constraints, multi-scale 
search, and focused peak reduction.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))

def normalize(a, S=None):
    """Normalize to have sum S."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * S / s

def lp_refine_relaxed(a, time_limit, top_k_frac=0.3, verbose=False):
    """
    LP refinement with relaxed constraints.
    Only constrains the top-K lags of conv(a, g) to be <= M,
    allowing the LP more freedom at non-peak lags.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    for it in range(3000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        # Use only top-K lags as constraints
        k = max(1, int(top_k_frac * (2 * n - 1)))
        top_lags = np.argsort(conv_a)[-k:]
        
        # Build constraint matrix for only these lags
        A = np.zeros((k, n))
        for i, lag in enumerate(top_lags):
            for j in range(n):
                idx = lag - j
                if 0 <= idx < n:
                    A[i, j] = a[idx]
        
        # Solve LP: max sum(g) s.t. A[g] <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(k, M)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        # Normalize g to match a's sum
        c_norm = normalize(g, S)
        
        # Find best blend
        best_t = 0
        best_peak_t = M
        for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01, 0.005]:
            b = (1 - t) * a + t * c_norm
            b = np.clip(b, 0, None)
            b = normalize(b, S)
            p = np.max(fft_convolve(b, b))

def gradient_descent(a, time_limit, verbose=False):
    """Subgradient descent on the peak of autocorrelation."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    lr = 1.0 / n
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        
        # Subgradient w.r.t. a at the peak
        grad = np.zeros(n)
        for j in range(n):
            v = 0.0
            i1 = peak_idx - j
            if 0 <= i1 < n:
                v += a[i1]
            i2 = j - peak_idx
            if 0 <= i2 < n:
                v += a[i2]
            grad[j] = v

        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new, S)

            best_a = a_new.copy()
            a = a_new

        elif it % 100 == 0:

directional improvement, adaptive gradient descent, and multi-peak local

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * S / s


def build_sparse_conv_matrix(a, lags):
    """Build convolution constraint matrix for specific lags only."""
    n = len(a)
    k = len(lags)
    A = np.zeros((k, n))
    for i, lag in enumerate(lags):
        j_min = max(0, lag - n + 1)
        j_max = min(lag, n - 1)
        for j in range(j_min, j_max + 1):
            A[i, j] = a[lag - j]
    return A

def multi_peak_gradient(a, time_limit, verbose=False):

    Gradient descent that considers multiple peaks simultaneously.
    Uses a soft-max loss over the top autocorrelation values.

        n_top = max(5, n // 8)
        top_vals = np.sort(conv)[-n_top:]
        scale = 0.1
        weights = np.exp(top_vals / scale) / np.sum(np.exp(top_vals / scale))

        # Find which lags are in the top set
        top_lag_indices = np.argsort(conv)[-n_top:]

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, lag in zip(weights, top_lag_indices):
            for j in range(n):
                i1 = lag - j
                if 0 <= i1 < n:
                    grad[j] += w * a[i1]
                i2 = j - lag
                if 0 <= i2 < n:
                    grad[j] += w * a[i2]

        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S)

        new_peak = np.max(fft_convolve(a_new, a_new))
        if new_peak < best_peak:

        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        verify = evaluate_sequence(final_list)
        print(f"  evaluate_sequence returns: {verify:.6f}")
        print(f"  n={len(final_list)}")
        return final_list

    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    return [1.0] * 256
