For each lag k where conv(a,a)[k] is above the target, reduce the elements that contribute to it. For each lag where it's below, we can increase elements.

This is similar to the "FPTAS for discrepancy" or "multiplicative weights" framework.

Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Combines LP-based refinement, smooth-max gradient descent,
and targeted local search. Key improvements over previous approach:
- Smooth log-sum-exp gradient descent with adaptive temperature
- LP with tighter constraints and full line search

import numpy as np
import time
from scipy.optimize import linprog, minimize as scipy_minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize_seq(a, S):
    """Normalize sequence to have sum S."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * S / s


def compute_conv_max_and_idx(a):
    """Return (max_conv, argmax_idx, full_conv)."""
    conv = fft_convolve(a, a)
    idx = int(np.argmax(conv))
    return conv[idx], idx, conv


def build_lp_matrix(a):
    """Build convolution constraint matrix for LP."""
    n = len(a)
    # Use sparse-friendly construction
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    from scipy.sparse import coo_matrix
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsc()
    return A


def lp_solve(a, M):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)

def smooth_max_descent(a, time_limit, tau_start=10.0, tau_end=0.1, verbose=False):
    """
    Smooth max gradient descent using log-sum-exp trick.
    Minimizes soft approximation to max(conv(a,a)) with sum(a) fixed.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    
    # Number of iterations to decrease tau
    total_iters = 500
    step = 0
    
    while time.monotonic() - t0 < time_limit and step < total_iters:
        # Log-linear tau schedule
        frac = step / total_iters
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute gradient of smooth max
        conv = fft_convolve(a, a)
        m = np.max(conv)
        exp_vals = np.exp((conv - m) / tau)
        weights = exp_vals / np.sum(exp_vals)
        
        # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j]
        # Compute efficiently
        # T[j] = sum_{m=0}^{n-1} a[m] * weights[m+j]
        # This is cross-correlation of a and weights
        # Use FFT: pad both to size >= n + (2n-1) - 1 = 3n - 2
        N = 1
        while N < 3 * n - 1:
            N <<= 1
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(weights, N)
        # Cross-correlation: ifft(conj(fa) * fw) gives sum a[m]*w[m+k]
        # Let me verify: ifft(conj(FA) * FW)[k] = sum_m conj(a[m]) * w[(m+k)%N]
        # For real a: = sum_m a[m] * w[(m+k) mod N]
        # With zero padding, for k = 0..n-1: = sum_m a[m] * w[m+k] (linear)
        cross = np.fft.irfft(np.conj(fa) * fw, N)[:n]
        grad = 2.0 * cross
        
        # Gradient descent step with Adam-like adaptation
        # Simple: use fixed step size related to tau/n
        lr = 2.0 * tau / n
        
        a_new = a - lr * grad
        a_new = np.clip(a_new, 1e-10, None)
        a_new = normalize_seq(a_new, S)
        
        new_conv_max = np.max(fft_convolve(a_new, a_new))
        if new_conv_max < best_conv_max:
            best_conv_max = new_conv_max
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step
            lr *= 0.5
            a_new = a - lr * grad
            a_new = np.clip(a_new, 1e-10, None)
            a_new = normalize_seq(a_new, S)
            new_conv_max = np.max(fft_convolve(a_new, a_new))
            if new_conv_max < np.max(fft_convolve(a, a)):
                a = a_new
        
        step += 1
        if verbose and step % 50 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    SMD step {step}: tau={tau:.3f}, score={score:.6f}")
    
    return best_a


def block_coordinate_descent(a, time_limit, verbose=False):
    """
    Block coordinate descent: for each block of consecutive elements,
    optimize that block while keeping others fixed, to reduce the peak.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = np.max(fft_convolve(a, a))
    
    block_size = max(4, n // 50)
    n_blocks = n // block_size
    
    for iteration in range(1000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        
        # For each block, try to reduce the peak by adjusting that block
        for b in range(n_blocks):
            start = b * block_size
            end = min(start + block_size, n)
            
            # Current contribution of this block to the peak
            # Try random perturbations of this block
            for _ in range(5):
                candidate = a.copy()
                perturbation = np.exp(np.random.standard_normal(end - start) * 0.05)
                candidate[start:end] *= perturbation
                candidate = np.clip(candidate, 1e-10, None)
                candidate = normalize_seq(candidate, S)
                
                new_max = np.max(fft_convolve(candidate, candidate))

"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative
step functions. Combines LP-based refinement, smooth log-sum-exp gradient
descent with Nesterov acceleration, and targeted local search. Key
improvements: adaptive smooth-max descent, LP with multiple constraint
tightness levels, full line search, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize_seq(a, S):
    """Normalize sequence to have sum S."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * S / s


def build_lp_matrix(a):
    """Build sparse convolution constraint matrix."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsc()


def lp_solve(a, M):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)


def lp_refine_improved(a, time_limit, verbose=False):
    """LP refinement with multiple M factors and line search."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()

    best_conv_max = np.max(fft_convolve(a, a))
    
    for iteration in range(3000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        best_new_conv = np.inf
        best_a_new = None
        
        for m_factor in [1.0, 0.998, 0.995]:
            g = lp_solve(a, M * m_factor)
            g_sum = np.sum(g)
            if g_sum < 1e-10:
                continue
            c = normalize_seq(g, S)
            
            # Line search
            best_t = 0.0
            best_conv_t = np.inf
            for t_try in [0.3, 0.5, 0.2, 0.4, 0.1, 0.5, 0.7, 0.8, 1.0]:
                b = (1 - t_try) * a + t_try * c
                b = np.clip(b, 0.0, None)
                b = normalize_seq(b, S)
                cm = np.max(fft_convolve(b, b))
                if cm < best_conv_t:
                    best_conv_t = cm
                    best_t = t_try
            
            if best_t > 0 and best_conv_t < best_new_conv:
                best_new_conv = best_conv_t
                best_a_new = (1 - best_t) * a + best_t * c
                best_a_new = normalize_seq(np.clip(best_a_new, 0.0, None), S)

            best_a = best_a_new.copy()
        a = best_a_new

            print(f"    LP iter {iteration}: score={score:.6f}")

            break
    
    return best_a


def smooth_max_grad(a, tau):
    """Compute gradient of soft-max approximation to max(conv(a,a))."""
    n = len(a)
    conv = fft_convolve(a, a)
    m = np.max(conv)
    exp_vals = np.exp((conv - m) / tau)
    weights = exp_vals / np.sum(exp_vals)
    
    # T[j] = sum_{m=0}^{n-1} a[m] * weights[m+j]
    # Use FFT cross-correlation
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(weights, N)
    cross = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    grad = 2.0 * cross
    
    smooth_val = m + tau * np.log(np.sum(exp_vals) / len(conv))
    return smooth_val, grad


def smooth_max_descent(a, time_limit, tau_start=10.0, tau_end=0.05, verbose=False):
    """Smooth-max gradient descent with Nesterov momentum."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    
    # Nesterov momentum
    v = np.zeros(n)
    beta = 0.9
    lr = 1.0 / n
    step = 0
    stall = 0
    total_steps = max(100, int(time_limit * 2))
    
    while time.monotonic() - t0 < time_limit and step < total_steps:
        frac = min(step / total_steps, 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Nesterov lookahead
        a_look = a + beta * v
        a_look = np.clip(a_look, 1e-10, None)
        a_look = normalize_seq(a_look, S)
        
        smooth_val, grad = smooth_max_grad(a_look, tau)
        
        # Adaptive learning rate
        v = beta * v - lr * grad
        a_new = a + v
        a_new = np.clip(a_new, 1e-10, None)
        a_new = normalize_seq(a_new, S)
        
        new_conv_max = np.max(fft_convolve(a_new, a_new))
        
        if new_conv_max < best_conv_max:
            best_conv_max = new_conv_max
            best_a = a_new.copy()
            a = a_new
            stall = 0
        elif new_conv_max < np.max(fft_convolve(a, a)):
            a = a_new
            stall += 1
        else:
            # Revert and reduce step
            a = a - 0.5 * v
            a = np.clip(a, 1e-10, None)
            a = normalize_seq(a, S)
            v *= 0.5
            lr *= 0.9
            stall += 1
        
        if stall > 50:
            lr *= 0.5
            beta *= 0.95
            v = np.zeros(n)
            stall = 0
        
        step += 1
        if verbose and step % 100 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    SMD step {step}: tau={tau:.3f}, lr={lr:.5f}, score={score:.6f}")
    
    return best_a

                best_max = new_peak
                best_a = candidate.copy()

        f = 1 + int(rng.integers(0, 5))
        amp = 0.1 + 0.3 * rng.random()
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + rng.uniform(0, 2 * np.pi))
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for _ in range(4):
            f = 1 + int(rng.integers(0, n // 8))
            amp = rng.uniform(0.03, 0.15)
            a += amp * np.cos(2 * np.pi * f * t / n + rng.uniform(0, 2 * np.pi))
    elif init_type == 'random':
        a = 1.0 + 0.2 * rng.standard_normal(n)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)


def interpolate_sequence(a, n_new):
    """Interpolate to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP refinement, smooth-max gradient descent,
    targeted search. Explores multiple sequence lengths.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            print(f"    *** New best: score={s:.6f}, n={len(a_clip)} ***")
            return True
        return False
    
    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)
    
    # === Phase 1: LP Refinement on provided ===
    print("\n=== Phase 1: LP Refinement ===")
    if provided_seq is not None:
        n = len(provided_seq)
        t_limit = min(250.0, max(60.0, (deadline - time.monotonic()) * 0.2))
        print(f"  LP on provided (n={n}), limit={t_limit:.0f}s")
        result = lp_refine_improved(provided_seq, t_limit, verbose=True)
        update_best(result)
    
    # Try other lengths from provided
    if provided_seq is not None:
        for n_try in [512, 768, 896, 1024]:
            if time.monotonic() > deadline - 600:
                break
            a0 = np.clip(interpolate_sequence(provided_seq, n_try), 0.01, None)
            t_limit = min(40.0, (deadline - time.monotonic()) * 0.08)
            if t_limit < 20:
                break
            result = lp_refine_improved(a0, t_limit, verbose=False)
            update_best(result)
    
    # === Phase 2: Smooth-Max Gradient Descent ===
    print("\n=== Phase 2: Smooth-Max Gradient Descent ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(300.0, max(60.0, (deadline - time.monotonic()) * 0.25))
        print(f"  SMD (n={n}), limit={t_limit:.0f}s")
        result = smooth_max_descent(best_seq, t_limit, tau_start=5.0, tau_end=0.03, verbose=True)
        update_best(result)
    
    # Also try SMD from a different initialization at same length
    if best_seq is not None:
        n = len(best_seq)
        if time.monotonic() < deadline - 300:
            for init_type in ['chirp', 'multi_tone']:
                a0 = make_initialization(n, init_type, rng)
                t_limit = min(60.0, (deadline - time.monotonic()) * 0.1)
                if t_limit < 20:
                    break
                result = smooth_max_descent(a0, t_limit, tau_start=20.0, tau_end=0.1, verbose=False)
                update_best(result)
    
    # === Phase 3: Second LP Pass ===
    print("\n=== Phase 3: Second LP ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 120
        if remaining > 40:
            t_limit = min(remaining * 0.5, 200.0)
            print(f"  LP pass 2, limit={t_limit:.0f}s")
            result = lp_refine_improved(best_seq, t_limit, verbose=True)
            update_best(result)
    
    # === Phase 4: Targeted Search ===
    print("\n=== Phase 4: Targeted Search ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 20:
            t_limit = min(remaining * 0.5, 150.0)
            print(f"  Targeted, limit={t_limit:.0f}s")
            result = targeted_search(best_seq, t_limit, verbose=True)
            update_best(result)
    
    # === Phase 5: Final Polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15
        if remaining > 15:
            # Quick SMD polish
            result = smooth_max_descent(best_seq, remaining, tau_start=1.0, tau_end=0.01, verbose=False)
            update_best(result)
    
    # Final
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: eval={final_score:.6f}, n={len(final_list)}, local={best_score:.6f} ===")
        return final_list
    
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 640
