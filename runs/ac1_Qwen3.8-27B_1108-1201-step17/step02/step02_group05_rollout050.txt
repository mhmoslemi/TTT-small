def soft_max_conv(a, tau):
    """Compute soft-max approximation of max(conv(a,a)) and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    # Soft-max
    conv_shifted = conv / tau
    conv_shifted -= conv_shifted.max()  # numerical stability
    weights = np.exp(conv_shifted)
    weights /= weights.sum()
    soft_max_val = tau * (np.log(weights.sum() * np.exp(conv.max()/tau)) - np.log(weights.sum()))
    # Actually simpler: soft_max_val = tau * log(sum(exp(conv/tau)))

    # = conv.max() + tau * log(sum(exp((conv - conv.max())/tau)))

    # = sum_k weights[k] * 2*a[k-j] (if 0 <= k-j < n)
    grad = np.zeros(n)

    return soft_max_val, grad

def lp_direction(f, M):
    """LP: find g >= 0, sum(g)=1 maximizing sum(g) s.t. conv(f,g) <= M."""

            if gs > 1e-12:

    return None

        c_val = hi - gr * (hi - lo)
        d_val = lo + gr * (hi - lo)
        fc = obj_at_t(c_val)
        fd = obj_at_t(d_val)
        for _ in range(40):
            if fc < fd:
                hi = d_val
                d_val = c_val
                fd = fc
                c_val = hi - gr * (hi - lo)
                fc = obj_at_t(c_val)
            else:
                lo = c_val
                c_val = d_val
                fc = fd
                d_val = lo + gr * (hi - lo)
                fd = obj_at_t(d_val)
        t_best = (lo + hi) / 2.0
        obj_best = obj_at_t(t_best)

        if obj_best < best_obj - 1e-12:
            a_new = (1.0 - t_best) * f + t_best * g
            a_new = np.maximum(a_new, 0.0)
            ns = a_new.sum()
            if ns < 1e-12:
                break
            a_new /= ns
            f = a_new
            best_obj = obj_best
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_f, best_obj


def coordinate_descent(a, time_limit, max_rounds=200):
    """1D coordinate descent: for each j, optimize a[j] keeping sum=1."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_obj = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stuck = 0

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            # Minimize max(conv) over a[j] in [0, 1], with sum constraint
            # Fix other elements, vary a[j], rescale others to maintain sum=1
            aj = a[j]
            if aj < 1e-15:
                # Try small positive values
                for trial in [1e-4, 1e-3, 0.005]:
                    a_trial = a.copy()
                    a_trial[j] = trial
                    others = a_trial.sum() - trial
                    if others <= 0:
                        continue
                    # Scale others
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= (1.0 - trial) / others
                    o = 2.0 * n * fast_convolve(a_trial).max()
                    if o < best_obj - 1e-12:
                        best_obj = o
                        a = a_trial.copy()
                        improved = True
                        break
                continue

            # Golden section on a[j] in [0, aj*3] (capped at 1)
            hi_j = min(1.0, aj * 3.0 + 0.01)
            lo_j = max(0.0, aj * 0.3)

            def obj_1d(aj_val):
                a_trial = a.copy()
                a_trial[j] = aj_val
                others_sum = 1.0 - aj_val
                if others_sum <= 0:
                    return np.inf
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                orig_others = a[mask].sum()
                if orig_others < 1e-15:
                    return np.inf
                a_trial[mask] = a[mask] * others_sum / orig_others
                a_trial = np.maximum(a_trial, 0.0)
                return 2.0 * n * fast_convolve(a_trial).max()

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo_s, hi_s = lo_j, hi_j
            c_s = hi_s - gr * (hi_s - lo_s)
            d_s = lo_s + gr * (hi_s - lo_s)
            fc_s = obj_1d(c_s)
            fd_s = obj_1d(d_s)
            for _ in range(25):
                if fc_s < fd_s:
                    hi_s = d_s
                    d_s = c_s
                    fd_s = fc_s
                    c_s = hi_s - gr * (hi_s - lo_s)
                    fc_s = obj_1d(c_s)
                else:
                    lo_s = c_s
                    c_s = d_s
                    fc_s = fd_s
                    d_s = lo_s + gr * (hi_s - lo_s)
                    fd_s = obj_1d(d_s)

            aj_opt = (lo_s + hi_s) / 2.0
            obj_opt = obj_1d(aj_opt)
            if obj_opt < best_obj - 1e-12:
                a[j] = aj_opt
                others_sum = 1.0 - aj_opt
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                orig_others = a[mask].sum()
                if orig_others > 1e-15:
                    a[mask] = a[mask] * others_sum / orig_others
                a = np.maximum(a, 0.0)
                as_ = a.sum()
                if as_ > 1e-12:
                    a /= as_
                best_obj = obj_opt
                improved = True

        if not improved:
            stuck += 1
            if stuck >= 3:
                break
        else:
            stuck = 0

    return a, best_obj


def get_parent(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Cosine window
    cos_w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1))) if n > 1 else np.ones(n)
    cos_w = np.maximum(cos_w, 0.0)
    inits['cosine'] = cos_w / cos_w.sum()

    # Edge-concentrated
    edges = np.zeros(n)
    m = max(1, n // 3)
        edges[:m] = 1.0
        edges[-m:] = 1.0
    edges = edges / edges.sum()
    inits['edges'] = edges

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent

    # Perturbed parent
    if parent is not None:
        p_pert = parent.copy()
        p_pert += rng.normal(0, 0.05, n)
        p_pert = np.maximum(p_pert, 0)
        p_pert /= p_pert.sum()
        inits['parent_pert'] = p_pert

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining FFT-accelerated FISTA,
    LP direction finding, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a_clean = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a_clean[a_clean > 1000.0] = 1000.0
        sc = float(evaluate_sequence(a_clean.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_sequence = a_clean.copy()
            print(f"  New best: score={sc:.6f}, n={len(a_clean)}")

    lengths = [256, 512, 1024]

    # Phase 1: FISTA with soft-max (fast, many iterations)
    print("Phase 1: FISTA soft-max optimization...")
    phase1_end = time.monotonic() + (float(budget_s) - 30.0) * 0.45

    phase1_results = []
    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        inits = make_inits(n, rng)
        for name, a0 in inits.items():
            if time.monotonic() > phase1_end - 30:
                break
            remaining = phase1_end - time.monotonic()
            if remaining < 10:
                break
            t_alloc = min(remaining - 10, 60.0)
            if t_alloc < 5:
                break

            print(f"  FISTA: n={n}, {name}, alloc={t_alloc:.0f}s")
            a_opt, obj_opt = fista_with_backtrack(a0, time_limit=t_alloc, max_iters=3000)
            score = float(evaluate_sequence(np.maximum(a_opt, 0).tolist()))
            if np.isfinite(score):
                phase1_results.append((score, n, name, a_opt.copy()))
                update_best(a_opt)

    print(f"  Phase 1 done. Best so far: {best_score:.6f}")

    # Phase 2: LP-based refinement on top candidates
    print("Phase 2: LP-based refinement...")
    phase2_end = time.monotonic() + (float(budget_s) - 30.0) * 0.30

    phase1_results.sort(key=lambda x: x[0])
    seen_keys = set()
    top = []
    for score, n, name, a_opt in phase1_results:
        key = (n, name)
        if key not in seen_keys:
            top.append((n, name, a_opt))
            seen_keys.add(key)
        if len(top) >= 6:
            break

    if best_sequence is not None:
        top.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = phase2_end - time.monotonic()
    if remaining < 5:
        remaining = 5
    t_per = max(10.0, remaining / max(1, len(top)))

    for n, name, a0 in top:
        if time.monotonic() > phase2_end:
            break
        t_alloc = min(t_per, phase2_end - time.monotonic() - 10)
        if t_alloc < 5:
            break
        print(f"  LP: n={n}, {name}, alloc={t_alloc:.0f}s")
        a_opt, _ = lp_optimize(a0, time_limit=t_alloc)
        update_best(a_opt)

    print(f"  Phase 2 done. Best so far: {best_score:.6f}")

    # Phase 3: Alternate FISTA + LP + coordinate descent on best
    print("Phase 3: Intensive polish...")
    if best_sequence is not None:
        polish_deadline = deadline
        n_iters = 0
        while time.monotonic() < polish_deadline:
            remaining = polish_deadline - time.monotonic()
            if remaining < 15:
                break
            n_iters += 1

            # FISTA
            t_fista = min(remaining * 0.4, 120.0)
            if t_fista > 5:
                a_f, _ = fista_with_backtrack(best_sequence, time_limit=t_fista, max_iters=2000)
                update_best(a_f)

            # LP
            remaining = polish_deadline - time.monotonic()
            if remaining < 15:
                break
            t_lp = min(remaining * 0.3, 60.0)
            if t_lp > 5:
                a_l, _ = lp_optimize(best_sequence, time_limit=t_lp)
                update_best(a_l)

            # Coordinate descent
            remaining = polish_deadline - time.monotonic()
            if remaining < 15:
                break
            t_cd = min(remaining * 0.3, 60.0)
            if t_cd > 5:
                a_c, _ = coordinate_descent(best_sequence, time_limit=t_cd, max_rounds=50)
                update_best(a_c)

            # Random perturbation restart
            if n_iters % 5 == 3:
                remaining = polish_deadline - time.monotonic()
                if remaining < 30:
                    break
                a_pert = best_sequence + rng.normal(0, 0.02, len(best_sequence))
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
                t_pert = min(remaining * 0.2, 30.0)
                if t_pert > 5:
                    a_opt, _ = fista_with_backtrack(a_pert, time_limit=t_pert, max_iters=1000)
                    update_best(a_opt)

    # Final: try a few more lengths if time permits
    if time.monotonic() < deadline - 60:
        print("Phase 4: Extra length exploration...")
        for n in [384, 768]:
            if time.monotonic() > deadline - 30:
                break
            inits = make_inits(n, rng)
            for name in ['parent', 'gauss_0.25', 'uniform']:
                if name not in inits or time.monotonic() > deadline - 30:
                    continue
                a0 = inits[name]
                t_alloc = min(deadline - time.monotonic() - 20, 60.0)
                if t_alloc < 10:
                    break
                a_opt, _ = fista_with_backtrack(a0, time_limit=t_alloc, max_iters=2000)
                update_best(a_opt)

    # Final validation
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"FINAL: score={best_score:.6f}, length={len(best_sequence)}")
    return [float(x) for x in np.maximum(best_sequence, 0.0)]
