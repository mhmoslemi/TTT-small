import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def subgradient_descent(a0, time_limit, rng, top_m=5, verbose=False):
    """Fast subgradient descent with adaptive step size."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n)
        s = n
    a = a / s  # normalize to sum=1
    
    current_score = score(a)
    best_a = a.copy()
    best_score = current_score
    
    # Adaptive parameters
    alpha = 0.05  # step size
    alpha_decay = 0.999
    stall = 0
    
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        # Find top M peaks
        peaks = np.argsort(c)[-top_m:]
        peak_vals = c[peaks]
        
        # Compute subgradient (average of gradients at top peaks)
        grad = np.zeros(n)
        for k in peaks:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += a[idx]
        grad /= top_m
        
        # Step
        a_new = a - alpha * grad
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 0.01:
            a_new = a.copy()
            s_new = s
        else:
            a_new = a_new / s_new  # renormalize
        
        new_score = score(a_new)
        
        if new_score < current_score:
            a = a_new
            s = np.sum(a)
            current_score = new_score
            stall = 0
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            stall += 1
            if stall > 100:
                # Perturb
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.clip(a, 0, None)
                a = a / np.sum(a)
                current_score = score(a)
                stall = 0
        
        alpha *= alpha_decay
        alpha = max(alpha, 0.001)
        
        iters += 1
        if verbose and iters % 10000 == 0:
            print(f"  SGD iter={iters}, score={current_score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute 2*n*max(c)/(sum(a))^2 where c=convolve(a,a). Scale invariant."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def compute_subgradient(a, c, top_m=5):
    """Compute subgradient of max(c) at top-M peak positions. O(n*top_m)."""
    n = len(a)
    peaks = np.argsort(c)[-top_m:]
    grad = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        if hi > lo:
            # grad[j] += a[k-j] for j in [lo, hi)
            # a[k-j] for j=lo..hi-1 is a[k-lo], a[k-lo-1], ..., a[k-hi+1]
            grad[lo:hi] += a[k - lo : k - hi + 1 : -1]
    grad /= top_m
    return grad


def subgradient_descent(a0, time_limit, rng, verbose=False):
    """Fast subgradient descent with adaptive step size and top-M peaks."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n) * 0.5
        s = n * 0.5
    a = a / s  # normalize to sum=1

    current_score = fast_score(a)
    best_a = a.copy()
    best_score = current_score

    alpha = 0.08
    alpha_min = 0.002
    stall = 0
    top_m = 5
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        # Adapt top_m: use more peaks when score is close to optimal
        top_m = max(3, min(10, int(3 + 7 * (2.0 - current_score))))
        grad = compute_subgradient(a, c, top_m)

        # Multiple step sizes to try
        best_step_score = current_score
        best_step_a = a
        for trial_alpha in [alpha, alpha * 0.5, alpha * 0.25]:
            a_new = a - trial_alpha * grad
            a_new = np.clip(a_new, 0, None)
            s_new = np.sum(a_new)
            if s_new < 0.01:
                continue
            a_new = a_new / s_new
            new_score = fast_score(a_new)

        if best_step_score < current_score:
            a = best_step_a
            current_score = best_step_score
            stall = 0
            alpha = min(0.15, alpha * 1.02)

                best_score = current_score
                best_a = a.copy()
        else:

            alpha *= 0.95
            if stall > 200:
                # Perturb to escape local minimum
                perturbation_scale = 0.03 + 0.02 * min(stall / 1000, 1.0)
                a = a * np.exp(rng.normal(0, perturbation_scale, n))
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s > 0.01:
                    a = a / s
                    current_score = fast_score(a)
                else:
                    a = best_a.copy()
                    current_score = best_score

            elif stall > 50 and stall % 10 == 0:
                # Smaller perturbation

                s = np.sum(a)

        alpha = max(alpha, alpha_min)
        iters += 1
        if verbose and iters % 5000 == 0:
            elapsed = time.monotonic() - t0

                  f"alpha={alpha:.4f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def structured_local_search(a0, time_limit, rng, verbose=False):
    """Peak-targeted local search with smart move generation."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    a = a / np.sum(a)
    current_score = fast_score(a)
    best_a = a.copy()
    best_score = current_score
    temperature = 0.0005
    cooling = 0.99995
    t0 = time.monotonic()
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        max_c = np.max(c)
        peak_idx = int(np.argmax(c))
        
        # Compute contributions to the peak
        lo = max(0, peak_idx - n + 1)
        hi = min(peak_idx + 1, n)
        contribs = np.zeros(n)
        contribs[lo:hi] = a[peak_idx - lo : peak_idx - hi + 1 : -1]
        
        # Find top contributing positions
        top_k = max(5, n // 15)
        top_contrib_idx = np.argsort(contribs)[-top_k:]
        
        # Find "safe" positions (low contribution to top peaks)
        top_peaks = np.argsort(c)[-5:]
        safe_scores = np.ones(n)  # lower = safer
        for k in top_peaks:
            lo2 = max(0, k - n + 1)
            hi2 = min(k + 1, n)
            safe_scores[lo2:hi2] += a[k - lo2 : k - hi2 + 1 : -1]
        safe_scores /= len(top_peaks)
        safe_idx = np.argsort(safe_scores)[:top_k * 2]

        r = rng.random()
        candidate = a.copy()

        if r < 0.25:
            # Decrease top contributors, increase safe positions
            n_dec = max(2, top_k // 3)
            n_inc = max(3, top_k // 2)
            dec_idx = rng.choice(top_contrib_idx, size=min(n_dec, len(top_contrib_idx)), replace=False)
            inc_idx = rng.choice(safe_idx, size=min(n_inc, len(safe_idx)), replace=False)
            dec_factor = np.exp(rng.normal(-0.04, 0.015, size=len(dec_idx)))
            inc_factor = np.exp(rng.normal(0.04, 0.015, size=len(inc_idx)))
            candidate[dec_idx] *= dec_factor
            candidate[inc_idx] *= inc_factor
        elif r < 0.45:
            # Targeted pair swap: move mass from peak-contributing to safe
            i = int(rng.choice(top_contrib_idx))
            j = int(rng.choice(safe_idx))
            if i != j:
                delta = rng.uniform(0.02, 0.08) * a[i]
                candidate[i] = max(0, a[i] - delta)
                candidate[j] = a[j] + delta
        elif r < 0.60:
            # Block move near peak
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(-0.02, 0.015, size=width))
            candidate[start:start + width] *= factor
            # Compensate elsewhere
            other_start = int(rng.integers(0, max(1, n - width)))
            comp_factor = np.exp(rng.normal(0.02, 0.01, size=width))
            candidate[other_start:other_start + width] *= comp_factor
        elif r < 0.75:
            # Gradient-guided: move opposite to subgradient at peak
            peak_grad = np.zeros(n)
            pg_lo = max(0, peak_idx - n + 1)
            pg_hi = min(peak_idx + 1, n)
            peak_grad[pg_lo:pg_hi] = a[peak_idx - pg_lo : peak_idx - pg_hi + 1 : -1]
            step = rng.uniform(0.01, 0.03)
            candidate = a - step * peak_grad
            candidate = np.clip(candidate, 0, None)
            candidate = candidate / np.sum(candidate)
        elif r < 0.90:
            # Multi-position adjustment
            n_adj = int(rng.integers(3, max(4, n // 8)))
            adj_idx = rng.choice(n, size=n_adj, replace=False)
            # Decrease those that contribute to peaks
            for idx in adj_idx:
                if contribs[idx] > np.median(contribs):
                    candidate[idx] *= np.exp(rng.uniform(-0.05, -0.01))
                else:
                    candidate[idx] *= np.exp(rng.uniform(0.01, 0.05))
        else:
            # Global small perturbation
            candidate = a * np.exp(rng.normal(0, 0.008, n))
            candidate = np.clip(candidate, 0, None)
            candidate = candidate / np.sum(candidate)

        new_score = fast_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        elif temperature > 1e-8 and rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score
            no_improve += 1

        temperature *= cooling
        iters += 1
        if verbose and iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SLS: iter={iters}, curr={current_score:.6f}, best={best_score:.6f}, "
                  f"T={temperature:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def build_constraint_matrix(f):
    """Build (2n-1) x n matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j)
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = f[:hi - lo]
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return np.maximum(res.x, 0.0)
    return np.ones(n) / n


def lp_iterate(f0, time_limit, rng, verbose=False):
    """LP-based iterative optimization."""
    n = len(f0)
    S = np.sqrt(2.0 * n)
    f = f0.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    sum_f = np.sum(f)
    if sum_f < 0.01:
        f = np.ones(n) * (S / n)
    else:
        f = f * (S / sum_f)

    current_score = fast_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    t = 0.06
    stall_count = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(f, f))
        b_0 = solve_lp_direction(f, M)
        iter_count += 1

        sum_b = np.sum(b_0)
        if sum_b < 1e-10:
            break
        g = (S / sum_b) * b_0

        improved = False
        for trial_t in [t, t * 0.5, t * 0.25, t * 0.1, t * 0.05]:
            f_new = (1.0 - trial_t) * f + trial_t * g
            f_new = np.clip(f_new, 0, None)
            new_score = fast_score(f_new)
            if new_score < current_score:
                f = f_new
                current_score = new_score
                t = min(0.2, trial_t * 1.05)
                improved = True
                stall_count = 0
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
                break

        if not improved:
            stall_count += 1
            t *= 0.8
            if stall_count > 50:
                f = f * np.exp(rng.normal(0, 0.02, n))
                f = np.clip(f, 0, None)
                sf = np.sum(f)
                if sf > 0.01:
                    f = f * (S / sf)
                    current_score = fast_score(f)
                stall_count = 0
                t = 0.04

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP: iter={iter_count}, curr={current_score:.6f}, best={best_score:.6f}, "
                  f"t={t:.4f}, elapsed={elapsed:.1f}s")

    return best_f, best_score


def resample_sequence(seq, new_n):
    """Resample sequence to new length via linear interpolation."""
    x_old = np.linspace(0, 1, len(seq), endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, seq)


def make_smooth_init(n, rng, style='chirp'):
    """Generate diverse initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if style == 'chirp':
        alpha = rng.uniform(0.3, 0.7)
        a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    elif style == 'multi_tone':
        a = np.ones(n)
        for k in range(1, 8):
            amp = 0.2 / k
            freq = rng.uniform(0.5, 2.0) * k
            phase = rng.uniform(0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif style == 'gaussian':
        center = n / 2
        width = n * rng.uniform(0.2, 0.5)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / width) ** 2)
    elif style == 'u_shape':
        a = 1.0 + 0.5 * ((2 * t / n - 1) ** 2)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def coordinate_refine(a, time_limit, rng):
    """Coordinate descent: try adjusting individual positions."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / np.sum(a)
    current_score = fast_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    epochs = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_idx = int(np.argmax(c))

        # Compute which positions contribute to the peak
        lo = max(0, peak_idx - n + 1)
        hi = min(peak_idx + 1, n)
        peak_contrib = np.zeros(n)
        peak_contrib[lo:hi] = a[peak_idx - lo : peak_idx - hi + 1 : -1]

        improved = False
        for j in range(n):
            # Try several multipliers
            best_mult = 1.0
            best_local_score = current_score
            for mult in [0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15, 1.2]:
                if mult == 1.0:
                    continue
                a_trial = a.copy()
                a_trial[j] *= mult
                a_trial = np.clip(a_trial, 0, None)
                s_trial = np.sum(a_trial)
                if s_trial < 0.01:
                    continue
                a_trial = a_trial / s_trial
                trial_score = fast_score(a_trial)
                if trial_score < best_local_score:
                    best_local_score = trial_score
                    best_mult = mult

            if best_mult != 1.0:
                a[j] *= best_mult
                a = np.clip(a, 0, None)
                a = a / np.sum(a)
                current_score = best_local_score
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()

        epochs += 1
        if not improved and epochs > 3:
            # Perturb
            a = a * np.exp(rng.normal(0, 0.01, n))
            a = np.clip(a, 0, None)
            a = a / np.sum(a)
            current_score = fast_score(a)

        if time.monotonic() - t0 < time_limit and epochs % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord: epoch={epochs}, curr={current_score:.6f}, best={best_score:.6f}, "
                  f"elapsed={elapsed:.1f}s")

    return best_a, best_score
