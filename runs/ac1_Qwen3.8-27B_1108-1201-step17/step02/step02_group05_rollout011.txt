Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max gradient descent
with L-BFGS-B, progressive temperature, multiple initializations, and
local search polishing. Explores multiple sequence lengths.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_obj_and_grad(a, beta):
    """Compute soft-max objective and gradient."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    c = fast_convolve(a)
    
    # Soft-max
    beta_c = beta * c
    max_bc = beta_c.max()
    exp_bc = np.exp(beta_c - max_bc)
    Z = exp_bc.sum()
    p = exp_bc / Z
    L = (max_bc + np.log(Z)) / beta  # This is log(sum(exp(beta*c)))/beta
    
    # Objective
    f = 2.0 * n * L / (S * S)
    
    # Gradient
    # dL/da_i = 2 * sum_j p_{i+j} * a_j for j=0..n-1
    # Compute via FFT cross-correlation
    # g_i = sum_j a_j * p_{i+j} = conv(a, p_r)[2n-2-i] where p_r[m] = p[2n-2-m]
    p_r = p[::-1]  # p_r[m] = p[2n-2-m]
    
    # Need convolution of a (length n) with p_r (length 2n-1)
    # Result length: n + 2n-1 - 1 = 3n-2
    conv_size = 3 * n - 2
    fft_size = 1 << (conv_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_r, fft_size)
    conv_ap = np.fft.irfft(A * P, fft_size)[:conv_size]
    
    # g_i = conv_ap[2n-2-i]
    g = conv_ap[2*n-2:n-1:-1]  # This gives [conv_ap[2n-2], conv_ap[2n-3], ..., conv_ap[n-1]]
    # Actually for i=0: g_0 = conv_ap[2n-2], for i=n-1: g_{n-1} = conv_ap[n-1]
    # So g = conv_ap[2n-2 : n-2 : -1]
    g = conv_ap[2*n-2-n+1:2*n-1][::-1]  # conv_ap[n-1 : 2n-1][::-1]
    # Let me just do it explicitly
    g = np.array([conv_ap[2*n-2-i] for i in range(n)])
    
    # grad_i = (2n/S^2) * (2*g_i - 2L/S)
    grad = (2.0 * n / (S * S)) * (2.0 * g - 2.0 * L / S)
    
    return f, grad


def lbfgs_optimize(a0, beta, max_iter=1000, time_limit=None):
    """Run L-BFGS-B on the soft-max objective."""
    n = len(a0)
    t0 = time.monotonic()
    
    def fun(x):
        val, _ = softmax_obj_and_grad(x, beta)
        return val
    
    def grad(x):
        _, g = softmax_obj_and_grad(x, beta)
        return g
    
    bounds = [(0.0, None)] * n
    
    kwargs = {
        'maxiter': max_iter,
        'ftol': 1e-12,
        'gtol': 1e-8,
        'maxfun': 5000,
        'disp': False
    }
    
    if time_limit is not None:
        # Set a callback to stop if time is exceeded
        def callback(xk):
            return time.monotonic() - t0 > time_limit
        
        kwargs['callback'] = callback
    
    try:
        res = scipy.optimize.minimize(a0.copy(), fun, jac=grad, method='L-BFGS-B',
                                       bounds=bounds, options=kwargs)
        if res.success or res.fun < fun(a0):
            return res.x, res.fun
    except Exception as e:
        pass
    
    return a0.copy(), fun(a0)


def progressive_softmax(a0, time_limit, beta_start=10.0, beta_end=500.0, n_stages=5):
    """Progressive soft-max optimization with increasing beta."""
    a = np.maximum(a0.copy(), 0.0)
    if a.sum() < 1e-12:
        a = np.ones_like(a) / len(a)
    
    betas = np.geomspace(beta_start, beta_end, n_stages)
    t0 = time.monotonic()
    
    for beta in betas:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        a, _ = lbfgs_optimize(a, beta=beta, time_limit=remaining - 2)
        a = np.maximum(a, 0.0)
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max gradient descent
with L-BFGS-B, progressive temperature (annealing), multiple initializations,
sequence lengths, and local search polishing. Key innovations over previous
approach: (1) smooth soft-max surrogate enables fast analytical gradients,
(2) progressive beta scheduling acts as simulated annealing, (3) FFT-based
gradient computation enables handling larger n efficiently.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_obj_and_grad(a, beta):
    """Compute soft-max objective value and analytical gradient."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c = fast_convolve(a)
    beta_c = beta * c
    max_bc = beta_c.max()
    exp_bc = np.exp(beta_c - max_bc)
    Z = exp_bc.sum()
    p = exp_bc / Z
    L = (max_bc + np.log(Z)) / beta

    f = 2.0 * n * L / (S * S)

    # Gradient: dL/da_i = 2 * sum_j p_{i+j} * a_j
    # Compute g_i = sum_j a_j * p_{i+j} via FFT cross-correlation
    p_r = p[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_sz = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_sz)
    P = np.fft.rfft(p_r, fft_sz)
    conv_ap = np.fft.irfft(A * P, fft_sz)[:conv_len]

    # g_i = conv_ap[2n-2-i] for i=0..n-1
    g = conv_ap[2 * n - 2 : n - 1 : -1]

    grad = (2.0 * n / (S * S)) * (2.0 * g - 2.0 * L / S)

    return f, grad


def run_lbfgs(a0, beta, time_limit, max_iter=500):
    """Run L-BFGS-B optimization with soft-max objective."""
    n = len(a0)
    t0 = time.monotonic()

    def fun(x):
        val, _ = softmax_obj_and_grad(x, beta)
        return val

    def grad(x):
        _, g = softmax_obj_and_grad(x, beta)
        return g

    def cb(xk):
        return (time.monotonic() - t0) > time_limit

    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.minimize(
            a0.copy(), fun, jac=grad,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': 3000, 'callback': cb, 'disp': False}
        )
        if res.fun < fun(a0):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.maximum(a0.copy(), 0.0)


def progressive_optimize(a0, time_limit, beta_start=5.0, beta_end=500.0,
                         n_stages=5, restart_prob=0.3):
    """Progressive soft-max with optional random restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    t0 = time.monotonic()
    betas = np.geomspace(beta_start, beta_end, n_stages)

        s = a.sum()
        if s < 1e-12:

        else:
            a /= s

    return a

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def config_priority(n, name):
    """Priority score for sorting configs (lower = higher priority)."""
    score = 0
    if 'parent' in name:
        score -= 500
    score += n * 0.1
    if name == 'uniform':
        score += 10
    if name == 'random':
        score += 5
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses progressive soft-max gradient optimization
    with L-BFGS-B, multiple initializations and lengths, progressive
    beta annealing, and local search polishing to minimize the
    auto-correlation peak ratio.
    """
    rng = np.random.default_rng(seed)
    total = float(budget_s)
    deadline = time.monotonic() + total - 25.0

    best_sequence = None
    best_score = float('inf')

    lengths = [128, 256, 384, 512, 768, 1024]

    # Build all configs
    all_configs = []
    for n in lengths:
        inits = make_inits(n, rng)
        parent = get_parent(n)
        if parent is not None:
            inits['parent'] = parent
            noisy = parent + 0.08 * rng.standard_normal(n)
            noisy = np.maximum(noisy, 0)
            noisy /= noisy.sum()
            inits['parent_noisy'] = noisy
        for name, a0 in inits.items():
            all_configs.append((n, name, a0))

    all_configs.sort(key=lambda c: config_priority(c[0], c[1]))

    # Phase 1: Wide scan with moderate beta (45% budget)
    phase1_frac = 0.45
    phase1_end = time.monotonic() + total * phase1_frac
    print("Phase 1: Soft-max scan (%.0f configs)..." % len(all_configs))

    phase1_time = phase1_end - time.monotonic()
    time_per = max(4.0, phase1_time / max(1, len(all_configs)))

    scan_results = []
    for i, (n, name, a0) in enumerate(all_configs):
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(time_per, 90.0, rem - 20.0)
        if t_alloc < 3.0:
            break

        a_opt = progressive_optimize(a0, time_limit=t_alloc,
                                     beta_start=5.0, beta_end=200.0,
                                     n_stages=4)
        a_opt = np.maximum(a_opt, 0.0)
        s = a_opt.sum()
        if s < 1e-12:
            continue
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt.copy()))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print("  Scan [%d/%d] n=%d %s: %.6f" % (i, len(all_configs), n, name, score))

    print("  Phase 1 best: %.6f" % best_score)

    # Phase 2: Deep optimization on top candidates (35% budget)
    phase2_end = time.monotonic() + total * 0.35
    print("Phase 2: Deep optimization...")

    scan_results.sort(key=lambda x: x[0])
    top_k = min(12, len(scan_results))
    top = scan_results[:top_k]

    time_per_deep = max(15.0, (phase2_end - time.monotonic()) / max(1, top_k))

    for i, (score, n, name, a0) in enumerate(top):
        rem = deadline - time.monotonic() - 50.0
        if rem < 15:
            break
        t_alloc = min(time_per_deep, 200.0, rem)
        if t_alloc < 5.0:
            break

        # High-beta progressive
        a_opt = progressive_optimize(a0, time_limit=t_alloc * 0.6,
                                     beta_start=20.0, beta_end=2000.0,
                                     n_stages=5)
        a_opt = np.maximum(a_opt, 0.0)

        # Local search
        ls_time = min(40.0, t_alloc * 0.3)
        a_ls, _ = local_search(a_opt, time_limit=ls_time)

        for ac in [a_opt, a_ls]:
            sc = float(evaluate_sequence(ac.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_sequence = ac.copy()
                print("  Deep [%d] n=%d %s: %.6f" % (i, n, name, sc))

    # Phase 3: Intensive final polish (remaining)
    print("Phase 3: Final polish...")
    if best_sequence is not None:
        no_improve = 0
        max_no_improve = 6

        while time.monotonic() < deadline - 8:
            rem = deadline - time.monotonic() - 8
            if rem < 8:
                break

            if no_improve >= max_no_improve:
                # Inject perturbation to escape
                a_pert = best_sequence.copy()
                mask = rng.random(len(a_pert)) > 0.7
                a_pert[mask] *= (1.0 + 0.2 * rng.standard_normal(mask.sum()))
                a_pert = np.maximum(a_pert, 0)
                if a_pert.sum() > 1e-12:
                    a_pert /= a_pert.sum()
                    best_sequence = a_pert
                no_improve = 0
                continue

            old_score = best_score

            # Very high beta soft-max
            t1 = min(rem * 0.4, 150.0)
            a_opt = progressive_optimize(best_sequence, time_limit=t1,
                                         beta_start=100.0, beta_end=5000.0,
                                         n_stages=3)
            a_opt = np.maximum(a_opt, 0.0)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_sequence = a_opt.copy()

            # Local search
            rem = deadline - time.monotonic() - 8
            if rem < 5:
                break
            t2 = min(rem * 0.4, 80.0)
            a_ls, _ = local_search(best_sequence, time_limit=t2)
            sc = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_sequence = a_ls.copy()

            if best_score < old_score - 1e-8:
                no_improve = 0
                print("  Polish: %.6f" % best_score)
            else:
                no_improve += 1

    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final) and final < best_score:
        best_score = final

    print("FINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
