    """Project v onto the simplex {x: sum(x)=sigma, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - sigma))[0][-1]

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using projected gradient descent
with softmax homotopy, LP-based direction finding, and local refinement.
Multiple sequence lengths are explored to find the best candidate.
"""

import numpy as np
import time
import scipy.optimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def proj_simplex(v, sigma=1.0):
    """Project v onto {x: sum(x)=sigma, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - sigma)
    if not cond.any():
        return np.full(n, sigma / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - sigma) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_objective(a, tau):
    """Smoothed objective: 2n * soft_max(conv(a,a), tau) with sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    sm = (1.0/tau) * np.log(exp_c.sum()) + c_max
    return 2.0 * n * sm

def smooth_grad(a, tau):
    """Gradient of smoothed objective w.r.t. a (on the simplex)."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    p = exp_c / exp_c.sum()
    
    # grad[m] = 4n * sum_k p[k] * a[k-m] (with zero extension)
    # = 4n * cross_corr(p, a)[m]
    N = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(P * np.conj(A), N)
    grad = 4.0 *
