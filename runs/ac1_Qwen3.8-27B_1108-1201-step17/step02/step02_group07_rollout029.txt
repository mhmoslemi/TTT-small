"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.
Uses improved LP-based direction finding, peak-targeting perturbations,
spectral flattening, and adaptive local search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fft_convolve(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * np.max(c) / (S * S)), float(np.max(c)), int(np.argmax(c))

def build_conv_matrix(f):
    """Build matrix A where A @ b = conv(f, b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp_direction(f, M, A=None, timeout=5.0):
    """Solve: max sum(b) s.t. A@b <= M*1, b >= 0."""
    n = len(f)
    if A is None:
        A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def solve_lp_min_peak(f, A=None, timeout=5.0):
    """Solve: minimize M s.t. A@b <= M*1, b >= 0, sum(b) = 1.
    Uses epigraph formulation."""
    n = len(f)
    if A is None:
        A = build_conv_matrix(f)
    # Variables: b_0, ..., b_{n-1}, M (total n+1 variables)
    # Minimize M => c = [0,...,0,1]
    # Constraints: A@b - M <= 0 for each row => [A | -I] @ [b; M] <= 0
    # sum(b) = 1 => [1,...,1 | 0] @ [b; M] = 1
    # b >= 0, M free (but will be positive)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M
    
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def subgradient_polish(f_init, time_limit, verbose=False):
    """Subgradient descent with momentum on the peak component."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = normalize(f, 1.0)
    best_f = f.copy()
    best_score, _, _ = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta_m = 0.85
    lr = 0.01
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        c = fft_convolve(f)
        # Find top peaks
        threshold = np.max(c) * 0.99
        peak_idx = np.where(c >= threshold)[0]
        
        g = np.zeros(n)
        for k in peak_idx:
            ki = int(k)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                g[j] += 2.0 * f[ki - j]
        g /= float(len(peak_idx))
        
        # Center the gradient to maintain normalization
        g -= np.mean(g)
        
        momentum = beta_m * momentum + (1.0 - beta_m) * g
        f = f - lr * momentum
        f = np.clip(f, 0.001, 1000.0)
        f = normalize(f, 1.0)
        
        it += 1
        if it % 500 == 0 or it == 1:
            s, _, _ = score_fast(f)

                best_f = f.copy()
            if verbose:
                print(f"    [SG {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
        
        # Adaptive learning rate
        if it % 2000 == 0:
            lr *= 0.95
    
    s, _, _ = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score

def peak_target_search(f_init, time_limit, verbose=False):
    """Targeted search that specifically reduces the peak lags."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = normalize(f, 1.0)
    best_f = f.copy()
    best_score, best_M, best_peak = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        c = fft_convolve(f)
        peak = float(np.max(c))
        peak_idx = int(np.argmax(c))
        
        # Find all lags close to peak
        near_peak = np.where(c >= 0.95 * peak)[0]
        
        # Try to reduce the peak by decreasing elements that contribute most
        contribs = np.zeros(n)
        for k in near_peak:
            ki = int(k)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                contribs[j] += f[ki - j]
        
        # Top contributors
        top_k = max(5, n // 10)
        top_mask = np.zeros(n, dtype=bool)
        top_indices = np.argsort(contribs)[-top_k:]
        top_mask[top_indices] = True
        
        # Strategy: decrease top contributors, increase others
        strategy = rng.random()
        candidate = f.copy()
        
        if strategy < 0.5:
            # Decrease top, increase bottom
            dec_factor = np.exp(rng.normal(-0.05, 0.03, size=top_mask.sum()))
            candidate[top_mask] *= dec_factor
            bottom_mask = ~top_mask
            if bottom_mask.sum() > 0:
                inc_count = min(10, bottom_mask.sum())
                inc_indices = np.where(bottom_mask)[0]
                chosen = rng.choice(inc_indices, size=inc_count, replace=False)
                candidate[chosen] *= np.exp(rng.normal(0.05, 0.03, size=inc_count))
        elif strategy < 0.8:
            # Smooth the sequence slightly
            kernel_size = max(3, n // 50)
            kernel = np.ones(kernel_size) / kernel_size
            smoothed = np.convolve(f, kernel, mode='same')
            alpha = rng.uniform(0.1, 0.3)
            candidate = (1 - alpha) * f + alpha * smoothed
        else:
            # Random weighted perturbation
            candidate = f * np.exp(rng.normal(0, 0.02, n))
        
        candidate = np.clip(candidate, 0.001, 1000.0)
        candidate = normalize(candidate, 1.0)
        
        s, M_c, _ = score_fast(candidate)
        it += 1
        
        if s < best_score:
            best_score = s
            best_f = candidate.copy()
            f = candidate
            if verbose and it % 100 == 0:
                print(f"    [PT {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} peak_lag={best_peak}")
    
    return best_f, best_score

def spectral_flatten(f_init, time_limit, verbose=False):
    """Try to flatten the spectrum to reduce peak autocorrelation."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = normalize(f, 1.0)
    best_f = f.copy()
    best_score, _, _ = score_fast(f)
    t0 = time.monotonic()
    it = 0
    alpha = 0.1
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        # Compute DFT
        F = rfft(f, 2 * n)  # Use 2n points to avoid circular effects
        mag = np.abs(F)
        
        # Target: flat magnitude (geometric mean)
        target_mag = np.ones(len(mag)) * np.prod(mag) ** (1.0 / len(mag))
        
        # Phase from current sequence
        phase = np.angle(F)
        
        # New spectrum: interpolate between current and flat
        new_mag = (1 - alpha) * mag + alpha * target_mag
        new_F = new_mag * np.exp(1j * phase)
        
        # Inverse DFT
        f_new = irfft(new_F, 2 * n)[:n]
        f_new = np.clip(f_new, 0.001, 1000.0)
        f_new = normalize(f_new, 1.0)
        
        f = f_new
        it += 1
        
        if it % 500 == 0:
            s, _, _ = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"    [SPEC {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
        
        # Slowly decrease alpha
        alpha *= 0.999
    
    s, _, _ = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score

def make_initializations(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.arange(n, dtype=np.float64)
    
    # Hann window
    hann = 0.5 * (1 + np.cos(2 * np.pi * t / max(1, n - 1)))
    inits.append(np.clip(hann + 0.05 * rng.standard_normal(n), 0.01, None))
    
    # Exponential decay (one-sided)
    exp_dec = np.exp(-3.0 * t / n)
    inits.append(np.clip(exp_dec + 0.02 * rng.standard_normal(n), 0.01, None))
    
    # Two peaks
    peak1 = np.exp(-((t - 0.25 * n) ** 2) / (2 * (n / 8) ** 2))
    peak2 = np.exp(-((t - 0.75 * n) ** 2) / (2 * (n / 8) ** 2))
    inits.append(np.clip(peak1 + peak2 + 0.02 * rng.standard_normal(n), 0.01, None))
    
    # Random smooth
    rough = rng.random(n)
    kernel_size = max(3, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rough, kernel, mode='same')
    inits.append(np.clip(smooth + 0.01, 0.01, None))
    
    # Linear ramp
    ramp = t / max(1, n - 1)
    inits.append(np.clip(ramp + 0.02 * rng.standard_normal(n), 0.01, None))
    
    return inits
