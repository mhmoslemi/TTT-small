Let me also think about what kinds of sequences give low scores. 

For a symmetric sequence (a[j] = a[n-1-j]), the convolution is also symmetric, and the peak is at the center. The center value is sum_j a[j]^2. To minimize this with fixed sum, we want all a[j] equal (giving score 2). But wait, that's the minimum for symmetric sequences! So symmetric sequences can't beat 2?

No wait, that's not right. For symmetric sequences, conv[n-1] = sum_j a[j]*a[n-1-j] = sum_j a[j]^2 (by symmetry). And by Cauchy-Schwarz, sum a[j]^2 >= (sum a[j])^2/n. So conv[n-1] >= S^2/n, giving score >= 2.

But the peak might not be at n-1! For a symmetric sequence, the convolution is symmetric around (n-1), so conv[k] = conv[2(n-1)-k]. The peak could be at n-1 or elsewhere. If the sequence has a "bell" shape, the convolution will have its peak at n-1. But if the sequence is more "spread out" in frequency, the convolution peak might be lower than n-1... 

Actually no, for non-negative symmetric sequences, conv[n-1] = sum a[j]^2 >= S^2/n by Cauchy-Schwarz, and this is the maximum (since the convolution of two non-negative symmetric sequences has its peak at the center). So for symmetric non-negative sequences, the score is always >= 2.

To get below 2, we need a NON-symmetric sequence! This is a crucial insight.

For a non-symmetric sequence, the center of the convolution (at n-1) is sum_j a[j]*a[n-1-j], which is the dot product of a with its reverse. This can be much less than sum a[j]^2 if the sequence and its reverse are "misaligned."

For example, if a is concentrated at the beginning (a[0] ≈ S, rest ≈ 0), then a[n-1] ≈ 0, so conv[n-1] ≈ a[0]*a[n-1] + ... ≈ 0. But conv[0] = a[0]^2 ≈ S^2, so the peak is at 0 and the score is 2n. Bad.

The optimal sequence should balance: spread out enough to avoid a single dominant peak, but asymmetric enough to reduce the center peak below S^2/n.

This suggests that the optimal sequence has a "ramp" shape - high on one end, low on the other. Like a linear ramp from high to low!

```python
"""
Coordinate descent + LP hybrid optimizer for minimizing peak aperiodic 
autocorrelation of non-negative sequences.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def coord_descent(a_init, max_passes=200, tol=1e-10, verbose=False, t_limit=None, t_start=None):
    """Coordinate descent with exact 1D golden section search."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = float(np.sum(a))
    conv_arr = fft_convolve(a, a)
    best_score = 2.0 * n * np.max(conv_arr) / (S * S)
    best_a = a.copy()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for p in range(max_passes):
        if t_limit is not None and time.monotonic() - t_start > t_limit - 5:
            break
        
        max_improvement = 0.0
        
        for j0 in range(n):
            old = a[j0]
            S_others = S - old
            if S_others < 1e-10:
                continue
            
            # Precompute the "base" convolution without a[j0]'s contribution
            # conv_base[k] = conv[k] - contribution_of_a[j0]
            # Contribution at k: 2*old*a[k-j0] if 0<=k-j0<n and k!=2*j0, old^2 if k=2*j0
            
            k_start = max(0, j0)
            k_end = min(2*n-2, j0+n-1)
            
            # Build the linear coefficients for the affected range
            # For k in [k_start, k_end], k != 2*j0: linear coef = 2*a[k-j0]
            # For k = 2*j0: quadratic (x^2)
            
            idxs = np.arange(k_start, k_end + 1)
            j_vals = idxs - j0  # = k - j0, ranges from 0 to n-1
            lin_coefs = 2.0 * a[j_vals]
            
            has_quad = (2*j0 >= k_start and 2*j0 <= k_end)
            if has_quad:
                quad_pos = 2*j0 - k_start
                lin_coefs[quad_pos] = 0.0
            
            conv_affected = conv_arr[k_start:k_end+1].copy()
            
            # Unchanged peak: max of conv_arr outside [k_start, k_end]
            if k_start > 0:
                unchanged_left = np.max(conv_arr[:k_start])
            else:
                unchanged_left = 0.0
            if k_end < 2*n-2:
                unchanged_right = np.max(conv_arr[k_end+1:])
            else:
                unchanged_right = 0.0
            unchanged_peak = max(unchanged_left, unchanged_right)
            
            def eval_x(x):
                new_S = x + S_others
                if new_S < 0.01:
                    return np.inf
                # Peak at affected positions
                vals = conv_affected + lin_coefs * (x - old)
                if has_quad:
                    vals[quad_pos] = conv_affected[quad_pos] + x*x - old*old
                peak = max(np.max(vals), unchanged_peak)
                return 2.0 * n * peak / (new_S * new_S)
            
            # Golden section search
            x_lo = 0.0
            x_hi = max(1.0, 5.0 * max(old, float(np.max(a))))
            
            # Check endpoints
            f_lo = eval_x(x_lo)
            f_hi = eval_x(x_hi)
            
            # If function is monotonically decreasing, optimum is at x_hi
            if f_hi < f_lo:
                # Try larger x_hi
                x_hi2 = x_hi * 2
                f_hi2 = eval_x(x_hi2)
                if f_hi2 < f_hi:
                    x_hi = x_hi2
                    f_hi = f_hi2
            
            x1 = x_hi - gr * (x_hi - x_lo)
            x2 = x_lo + gr * (x_hi - x_lo)
            f1 = eval_x(x1)
            f2 = eval_x(x2)
            
            for _ in range(40):
                if f1 < f2:
                    x_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = x_hi - gr * (x_hi - x_lo)
                    f1 = eval_x(x1)
                else:
                    x_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = x_lo + gr * (x_hi - x_lo)
                    f2 = eval_x(x2)
            
            x_opt = (x_lo + x_hi) / 2
            f_opt = eval_x(x_opt)
            
            if f_opt < 2.0 * n * np.max(conv_arr) / (S * S) - 1e-12:
                # Update a and conv
                a[j0] = x_opt
                S = x_opt + S_others
                # Update convolution
                conv_arr[k_start:k_end+1] = conv_affected + lin_coefs * (x_opt - old)
                if has_quad:
                    conv_arr[2*j0] = conv_affected[quad_pos] + x_opt*x_opt - old*old
                new_score = 2.0 * n * np.max(conv_arr) / (S * S)
                max_improvement = max(max_improvement, best_score - new_score)
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
            
            old = x_opt  # for next iteration's reference
            
        if verbose and (p + 1) % 10 == 0:
            print(f"  Pass {p+1}: best_score={best_score:.8f}")
        
        if max_improvement < tol:
            if verbose:
                print(f"  Converged at pass {p+1}")
            break
    
    return best_a, best_score
```

```python
"""
Hybrid optimizer using coordinate descent (exact 1D optimization) combined 
with LP-based direction search and gradient polish for minimizing peak 
aperiodic autocorrelation of non-negative sequences.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

    a = a_init.astype(np.float64).copy()
    S = float(np.sum(a))
    conv_arr = fft_convolve(a, a).copy()
    best_score = 2.0 * n * np.max(conv_arr) / (S * S)
    best_a = a.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for p in range(max_passes):
        if t_start > 0 and time.monotonic() - t_start > t_limit - 5:
            break

        cur_score = 2.0 * n * np.max(conv_arr) / (S * S)

            old = a[j0]
            S_others = S - old
            if S_others < 1e-10 and old < 1e-10:
                continue
            
            k_start = max(0, j0)
            k_end = min(2 * n - 2, j0 + n - 1)
            
            idxs = np.arange(k_start, k_end + 1)
            j_vals = idxs - j0
            lin_coefs = 2.0 * a[j_vals]
            
            has_quad = (2 * j0 >= k_start and 2 * j0 <= k_end)
            if has_quad:
                quad_pos = 2 * j0 - k_start
                
            conv_affected = conv_arr[k_start:k_end + 1].copy()
            
            if has_quad:
                lin_coefs[quad_pos] = 0.0

            if k_end < 2 * n - 2:

                if has_quad:
                    vals = vals.copy()
                    vals[quad_pos] = conv_affected[quad_pos] + x * x - old * old
                peak = max(float(np.max(vals)), unchanged_peak)
                return 2.0 * n * peak / (new_S * new_S)

            x_lo = 0.0
            x_hi = max(1.0, 3.0 * float(np.max(a)))

def make_ramp_init(n, rng, direction=1):
    """Linear ramp initialization."""
    t = np.arange(n, dtype=np.float64)
    if direction == 1:
        a = t + 1.0  # increasing
    else:
        a = (n - t) + 1.0  # decreasing
    a += 0.1 * a.mean() * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_exp_init(n, rng, rate=0.5):
    """Exponential decay/growth initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(rate * t / n)
    a += 0.05 * a.mean() * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_hann_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_quadratic_init(n, rng):
    """Quadratic (parabola) initialization - asymmetric."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = (t - 0.3) ** 2 + 0.1  # asymmetric parabola
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def build_conv_matrix(f, n):
    """Build matrix A where A @ x = conv(f, x) for full convolution."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, n, timeout=5.0):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    A = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(f_init, t_limit, t_start, verbose=False):
    """LP-based iterative optimization."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    
    while time.monotonic() - t_start < t_limit - 5:
        c = fft_convolve(f, f)
        M = float(np.max(c))
        
        b = lp_direction(f, M, n, timeout=3.0)
        if b is None:
            break
        
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        S = float(np.sum(f))
        b = b * (S / Sb)
        
        best_t = 0.05
        best_s = np.inf
        for t_try in [0.05, 0.1, 0.2, 0.3]:
            f_cand = np.clip((1.0 - t_try) * f + t_try * b, 0.01, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try
        
        f = np.clip((1.0 - best_t) * f + best_t * b, 0.01, 1000.0)
        
        if best_s < best_score:
            best_score = best_s
            best_f = f.copy()
        elif verbose:
            print(f"    LP stall, score={best_score:.6f}")
            # Escape
            rng_esc = np.random.default_rng(42 + int(time.monotonic() * 1000))
            f += 0.05 * f * rng_esc.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
    
    return best_f, best_score


def gradient_polish(f_init, t_limit, t_start, verbose=False):
    """Subgradient descent with momentum."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    momentum = np.zeros(n)
    beta_m = 0.9
    lr = 0.002
    
    while time.monotonic() - t_start < t_limit - 2:
        c = fft_convolve(f, f)
        top_k = min(8, len(c))
        top_idx = np.argsort(c)[-top_k:]
        
        g = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                g[j] += 2.0 * f[ki - j]
        g /= float(top_k)
        g -= np.mean(g)
        
        momentum = beta_m * momentum + (1.0 - beta_m) * g
        f = f - lr * momentum
        f = np.clip(f, 0.01, None)
        f = f * (float(n) / float(np.sum(f)))
        
        s = score_fast(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
    
    return best_f, best_score
```

```python
"""
Hybrid optimizer: exact coordinate descent (1D golden section) + LP direction 
search + gradient polish for minimizing peak aperiodic autocorrelation of 
non-negative sequences. Key insight: asymmetric sequences can achieve score < 2.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def coord_descent(a_init, max_passes=2000, tol=1e-11, verbose=False, t_limit=60.0, t_start=0.0):
    """Coordinate descent with exact 1D golden section search."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = float(np.sum(a))
    conv_arr = fft_convolve(a, a).copy()
    best_score = 2.0 * n * float(np.max(conv_arr)) / (S * S)
    best_a = a.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for p in range(max_passes):
        if t_start > 0 and time.monotonic() - t_start > t_limit - 3:
            break
        max_improvement = 0.0

        for j0 in range(n):
            old = a[j0]
            S_others = S - old
            if S_others < 1e-10 and old < 1e-10:
                continue

            k_start = j0
            k_end = min(2 * n - 2, j0 + n - 1)

            j_vals = np.arange(k_start, k_end + 1) - j0
            lin_coefs = 2.0 * a[j_vals].copy()

            has_quad = (2 * j0 <= k_end)
            if has_quad:
                quad_pos = 2 * j0 - k_start
                lin_coefs[quad_pos] = 0.0

            conv_affected = conv_arr[k_start:k_end + 1].copy()

            # Unchanged peak
            unchanged_peak = 0.0
            if k_start > 0:
                unchanged_peak = max(unchanged_peak, float(np.max(conv_arr[:k_start])))
            if k_end < 2 * n - 2:
                unchanged_peak = max(unchanged_peak, float(np.max(conv_arr[k_end + 1:])))

                new_S = x + S_others
                if new_S < 0.01:
                    return np.inf

                if has_quad:
                    vals = vals.copy()
                    vals[quad_pos] = conv_affected[quad_pos] + x * x - old * old
                peak = max(float(np.max(vals)), unchanged_peak)
                return 2.0 * n * peak / (new_S * new_S)

            # Golden section search
            x_lo = 0.0
            x_hi = max(1.0, 4.0 * float(np.max(a)))

            x1 = x_hi - gr * (x_hi - x_lo)
            x2 = x_lo + gr * (x_hi - x_lo)
            f1 = eval_x(x1)
            f2 = eval_x(x2)

            for _ in range(35):
                if f1 < f2:
                    x_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = x_hi - gr * (x_hi - x_lo)
                    f1 = eval_x(x1)
                else:
                    x_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = x_lo + gr * (x_hi - x_lo)
                    f2 = eval_x(x2)

            x_opt = (x_lo + x_hi) / 2.0
            f_opt = eval_x(x_opt)

def make_ramp_init(n, rng, slope=1.0):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + slope * t / (n - 1) * (n - 1)
    a = a + 0.05 * a.mean() * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_exp_init(n, rng, rate=1.5):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(rate * t / n)
    a += 0.03 * a.mean() * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_hann_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_quad_asym_init(n, rng):
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = 0.5 + 1.5 * t + 0.5 * t * t
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_step_init(n, rng):
    a = np.ones(n)
    half = n // 2
    a[:half] = 2.0
    a[half:] = 0.5
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def build_conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_optimize(f_init, t_limit, t_start, verbose=False):
    """LP-based iterative optimization for smaller sequences."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    stall = 0

    while time.monotonic() - t_start < t_limit - 3:
        c = fft_convolve(f, f)
        M = float(np.max(c))
        A = build_conv_matrix(f, n)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': 4.0, 'presolve': True})
            if not res.success:
                break
            b = res.x
        except Exception:
            break

        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        S = float(np.sum(f))
        b = b * (S / Sb)

        best_t = 0.1
        best_s = np.inf
        for t_try in [0.03, 0.05, 0.1, 0.15, 0.25]:
            f_cand = np.clip((1.0 - t_try) * f + t_try * b, 0.01, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = np.clip((1.0 - best_t) * f + best_t * b, 0.01, 1000.0)

        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 15:
                rng_esc = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
                f += 0.08 * f * rng_esc.standard_normal(n)
                f = np.clip(f, 0.01, 1000.0)
                stall = 0

    return best_f, best_score


def gradient_polish(f_init, t_limit, t_start, verbose=False):
    """Subgradient descent with momentum on top convolution peaks."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    momentum = np.zeros(n)
    beta_m = 0.92
    lr = 0.0015

    while time.monotonic() - t_start < t_limit - 2:
        c = fft_convolve(f, f)
        top_k = min(10, len(c))
        top_idx = np.argsort(c)[-top_k:]

        g = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                g[j] += 2.0 * f[ki - j]
        g /= float(top_k)
        g -= np.mean(g)

        momentum = beta_m * momentum + (1.0 - beta_m) * g
        f = f - lr * momentum
        f = np.clip(f, 0.01, None)
        f = f * (float(n) / float(np.sum(f)))

        if verbose and (int(time.monotonic() - t_start) % 10 == 0):
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: coordinate descent + LP + gradient polish."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_fast(provided)
        print(f"Provided: n={len(provided)}, score={s:.8f}")
        best_score = s
        best_seq = provided.copy()

    # Build list of (name, n, init_array) to try
    inits = []

    if provided is not None:
        pn = len(provided)
        inits.append(('provided', pn, provided.copy()))
        # Interpolated versions
        for target_n in [int(pn * 1.5), int(pn * 2)]:
            if target_n <= 2048:
                x_old = np.linspace(0, 1, pn)
                x_new = np.linspace(0, 1, target_n)
                interp = np.interp(x_new, x_old, provided)
                inits.append((f'interp_{target_n}', target_n, np.clip(interp, 0.01, None)))

    # Asymmetric initializations (key for score < 2)
    for n in [256, 384, 512, 768, 1024]:
        inits.append((f'ramp1_{n}', n, make_ramp_init(n, rng, slope=2.0)))
        inits.append((f'ramp2_{n}', n, make_ramp_init(n, rng, slope=5.0)))
        inits.append((f'exp_{n}', n, make_exp_init(n, rng, rate=2.0)))
        inits.append((f'quad_{n}', n, make_quad_asym_init(n, rng)))
        inits.append((f'step_{n}', n, make_step_init(n, rng)))

    # Symmetric (baseline)
    for n in [256, 512]:
        inits.append((f'hann_{n}', n, make_hann_init(n, rng)))

    # Deduplicate by (n, name_prefix)
    seen = set()
    unique_inits = []
    for name, n, arr in inits:
        key = (n, name.split('_')[0])
        if key not in seen or name == 'provided':
            seen.add(key)
            unique_inits.append((name, n, arr))
    inits = unique_inits

    print(f"\nTotal initializations to try: {len(inits)}")
    print(f"Time budget: {budget_s - 15:.0f}s\n")

    # Phase 1: Coordinate descent on each initialization
    print("=" * 60)
    print("PHASE 1: Coordinate Descent")
    print("=" * 60)

    # Sort by n (smaller first for more time per run)
    inits.sort(key=lambda x: x[1])

    for name, n, f0 in inits:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation: give more time to smaller n (faster convergence)
        base_time = 80.0 if n <= 384 else (50.0 if n <= 768 else 35.0)
        time_alloc = min(base_time, remaining * 0.6)
        if remaining < 40:
            time_alloc = max(10.0, remaining - 10)

        print(f"\n  [{name}] n={n}, time={time_alloc:.0f}s")
        t0 = time.monotonic()
        f_best, s_best = coord_descent(f0, max_passes=5000,
                                        t_limit=time_alloc, t_start=t0,
                                        verbose=False)
        elapsed = time.monotonic() - t0
        print(f"    -> score={s_best:.10f} (elapsed={elapsed:.1f}s, best={best_score:.10f})")

        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"    *** NEW BEST: {best_score:.10f} ***")

    print(f"\nAfter Phase 1: best_score = {best_score:.10f}")

    # Phase 2: LP refinement on best (only for smaller n where LP is feasible)
    if best_seq is not None and len(best_seq) <= 512:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_alloc = min(150.0, remaining * 0.35)
            print(f"\n{'='*60}")
            print(f"PHASE 2: LP Refinement ({t_alloc:.0f}s)")
            print(f"{'='*60}")
            f_lp, s_lp = lp_optimize(best_seq, t_alloc, time.monotonic(), verbose=True)
            if s_lp < best_score:
                best_score = s_lp
                best_seq = f_lp.copy()
                print(f"    LP improved: {best_score:.10f}")
            else:
                print(f"    LP did not improve: {s_lp:.10f} vs {best_score:.10f}")

    # Phase 3: CD after LP
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_alloc = min(120.0, remaining * 0.4)
            print(f"\n{'='*60}")
            print(f"PHASE 3: CD after LP ({t_alloc:.0f}s)")
            print(f"{'='*60}")
            f_cd2, s_cd2 = coord_descent(best_seq, max_passes=5000,
                                          t_limit=t_alloc, t_start=time.monotonic(),
                                          verbose=True)
            if s_cd2 < best_score:
                best_score = s_cd2
                best_seq = f_cd2.copy()
                print(f"    CD2: {best_score:.10f}")

    # Phase 4: Gradient polish
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 25:
            t_alloc = min(80.0, remaining * 0.4)
            print(f"\n{'='*60}")
            print(f"PHASE 4: Gradient Polish ({t_alloc:.0f}s)")
            print(f"{'='*60}")
            f_gp, s_gp = gradient_polish(best_seq, t_alloc, time.monotonic(), verbose=True)
            if s_gp < best_score:
                best_score = s_gp
                best_seq = f_gp.copy()
                print(f"    GP: {best_score:.10f}")

    # Phase 5: Final CD polish
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(60.0, remaining - 5.0)
            print(f"\n{'='*60}")
            print(f"PHASE 5: Final CD ({t_alloc:.0f}s)")
            print(f"{'='*60}")
            f_fin, s_fin = coord_descent(best_seq, max_passes=5000,
                                          t_limit=t_alloc, t_start=time.monotonic(),
                                          verbose=False)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()
                print(f"    Final CD: {best_score:.10f}")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL RESULT: score={final_score:.10f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    return [1.0] * 256
```
