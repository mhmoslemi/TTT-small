"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. LP-based first-order optimization (linearized auto-conv minimization)
2. Accelerated mirror descent (exponentiated gradient)
3. Coordinate descent on critical positions
4. Multiple starting points including structural variations
"""

import numpy as np
import time
import scipy.sparse as sparse
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def lp_first_order_step(a, n, time_limit=5.0):
    """Solve LP: min t s.t. 2*sum_j a[k-j]*a'_j <= t + conv_k, sum(a')=1, a'>=0.
    Returns optimal a'."""
    conv = fast_convolve(a)
    
    # Build constraint matrix
    # For each k: sum_j (2*a[k-j]) * a'_j - t <= conv_k
    # Variable order: [a'_0, ..., a'_{n-1}, t]
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        # 2*a[k-j] for j in valid range
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a[k - j])
        # -t term (variable index n)
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b = conv.copy()  # RHS: t + conv_k  =>  A @ x <= b
    
    # Objective: minimize t => c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'disp': False, 'time_limit': time_limit})
    
    if res.success:
        return np.maximum(res.x[:n], 0.0)
    return None

def lp_optimize(a0, time_limit):
    """LP-based first-order optimization with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        a_star = lp_first_order_step(a, n)
        if a_star is None:
            break
        
        s_star = a_star.sum()
        if s_star < 1e-12:
            break
        a_star /= s_star
        
        # Check convergence
        if np.max(np.abs(a_star - a)) < 1e-10:
            break
        
        # Line search
        best_alpha = 0
        best_trial_obj = best_obj
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_trial = (1 - alpha) * a + alpha * a_star
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-12:
                continue
            a_trial /= s_t
            obj_trial = 2.0 * n * fast_convolve(a_trial).max()
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_alpha = alpha
        
        if best_alpha == 0:
            patience += 1
            if patience > 30:
                break
        else:
            a = (1 - best_alpha) * a + best_alpha * a_star
            a = np.maximum(a, 0)
            a /= a.sum()
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
    
    return best_a, best_obj

def mirror_descend(a0, time_limit, t_param=50.0, eta0=0.3, nesterov=True):
    """Mirror descent with optional Nesterov acceleration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a /= a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = eta0
    patience = 0
    t0 = time.monotonic()
    
    # Nesterov
    a_prev = a.copy()
    m = 0.0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        if nesterov:
            a_x = a + m * (a - a_prev)
            a_x = np.maximum(a_x, 0)
            s_x = a_x.sum()
            if s_x < 1e-12:
                a_x = a
            else:
                a_x /= s_x
        else:
            a_x = a
        
        conv_x = fast_convolve(a_x)
        max_conv = conv_x.max()
        if max_conv < 1e-15:
            break
        
        # Compute smoothed subgradient
        scaled = t_param * conv_x / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(2*n-1)
            w[int(np.argmax(conv_x))] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient: sg[j] = 2 * sum_k w[k] * a_x[k-j] for valid k-j
        sg = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # sg[j] += 2 * w[k] * a_x[k-j] for j in [j_start, j_end]
            # a_x[k-j] for j in [j_start, j_end] = a_x[k-j_end : k-j_start+1][::-1]
            sg[j_start:j_end+1] += 2.0 * w[k] * a_x[k-j_end:k-j_start+1][::-1]
        
        # Mirror step
        lf = -eta * sg
        lf -= lf.max()  # for numerical stability
        a_new = a_x * np.exp(lf)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new /= s_new
        
        # Update momentum
        if nesterov:
            m_new = (m + 1) / (m + 2)
            a_prev = a.copy()
            a = a_new
            m = m_new
        
        # Evaluate
        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 20 == 0:
                eta *= 0.75
                if eta < 0.01:
                    t_param = min(t_param * 1.5, 500.0)
                    eta = 0.15
            if patience > 200:
                break
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, n_top=10):
    """Coordinate descent: repeatedly optimize small groups of coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a /= a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find the top contributing positions
        top_k = np.argsort(conv)[-n_top:]
        
        # For each top position, try to reduce it by adjusting the involved a's
        improved = False
        for k in top_k:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            indices = list(range(j_start, j_end + 1))
            
            # Try reducing the max contributor
            # The contribution to conv_k from a_j is a_j * a[k-j]
            # To reduce conv_k, we can reduce the largest a_j * a[k-j] product
            contributions = np.array([a[j] * a[k-j] for j in indices])
            max_idx = np.argmax(contributions)
            j = indices[max_idx]
            k_j = k - j
            
            # Try reducing a_j (transfer mass to a position that doesn't help conv_k)
            # Find a position p where reducing a_j and increasing a_p doesn't increase any top conv
            # Simple heuristic: reduce a_j slightly and increase the minimum a
            if a[j] > 1e-6:
                for frac in [0.5, 0.3, 0.1, 0.05]:
                    a_trial = a.copy()
                    delta = frac * a[j]
                    a_trial[j] -= delta
                    # Spread delta to positions far from k
                    # Just add to the minimum element
                    min_pos = np.argmin(a_trial)
                    a_trial[min_pos] += delta
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-12:
                        continue
                    a_trial /= s
                    obj = 2.0 * n * fast_convolve(a_trial).max()
                    if obj < best_obj - 1e-10:
                        a = a_trial
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
    
    return best_a, best_obj

Wait, I need to double-check my LP formulation. Let me re-derive:

We want to minimize max_k (a' * a')[k] subject to sum(a') = 1, a' >= 0.

The first-order approximation at point a:
(a'*a')[k] ≈ (a*a)[k] + (grad_k(a))^T (a' - a)

where grad_k(a)[j] = 2*a[k-j] (for 0 <= k-j < n).

So: (a'*a')[k] ≈ (a*a)[k] + 2 * sum_j a[k-j] * (a'_j - a_j)
= 2 * sum_j a[k-j] * a'_j - 2*(a*a)[k] + (a*a)[k]
= 2 * sum_j a[k-j] * a'_j - (a*a)[k]

We want this <= t for all k:
2 * sum_j a[k-j] * a'_j - (a*a)[k] <= t
=> 2 * sum_j a[k-j] * a'_j <= t + (a*a)[k]

So the LP is:
min t
s.t. 2 * sum_j a[k-j] * a'_j - t <= (a*a)[k]  for k = 0, ..., 2n-2
     sum_j a'_j = 1
     a'_j >= 0

In standard form for scipy's linprog (minimize c^T x s.t. A_ub @ x <= b_ub, bounds):
- Variables: x = [a'_0, ..., a'_{n-1}, t]
- Objective: c = [0, ..., 0, 1] (minimize t)
- Constraints: for each k: sum_j (2*a[k-j]) * a'_j + (-1) * t <= (a*a)[k]
- Equality: sum_j a'_j = 1 (this needs to be A_eq, b_eq)
- Bounds: a'_j >= 0 for j=0..n-1, t is unbounded (or t >= 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based first-order optimization (linearized auto-conv minimization with line search)
2. Accelerated mirror descent (exponentiated gradient with Nesterov momentum)
3. Coordinate descent on critical convolution positions
4. Random restarts to escape local minima
5. Multiple lengths (256, 512, 1024) and diverse initializations

import numpy as np
import time
import scipy.sparse as sparse
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def lp_first_order_step(a, n, time_limit=3.0):
    """Solve LP: min t s.t. 2*sum_j a[k-j]*a'_j - t <= conv_k for all k, sum(a')=1, a'>=0."""
    conv = fast_convolve(a)

    rows = []
    cols = []
    vals = []

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a[k - j])
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)

    A_ub = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = conv.copy()

    A_eq = sparse.csc_matrix((np.ones(n), (np.zeros(n, dtype=np.int32), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'disp': False, 'time_limit': time_limit})

    if res.success:
        return np.maximum(res.x[:n], 0.0)
    return None


def lp_optimize(a0, time_limit):
    """LP-based first-order optimization with line search and restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    patience = 0
    restarts = 0

    while time.monotonic() - t0 < time_limit:
        a_star = lp_first_order_step(a, n)
        if a_star is None:
            break

        s_star = a_star.sum()
        if s_star < 1e-12:
            break
        a_star /= s_star

        if np.max(np.abs(a_star - a)) < 1e-12:
            if restarts < 5:
                # Perturb and restart
                noise = np.random.default_rng(int(time.monotonic()*1000)).random(n) * 0.01
                a = a + noise
                a = np.maximum(a, 0)
                a /= a.sum()
                restarts += 1
                patience = 0
                continue
            else:
                break

        best_alpha = 0.0
        best_trial_obj = best_obj
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - alpha) * a + alpha * a_star
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-12:
                continue
            a_trial /= s_t
            obj_trial = 2.0 * n * fast_convolve(a_trial).max()
            if obj_trial < best_trial_obj - 1e-14:
                best_trial_obj = obj_trial
                best_alpha = alpha

        if best_alpha == 0.0:
            patience += 1
            if patience > 40:
                if restarts < 5:
                    rng_r = np.random.default_rng(int(time.monotonic() * 1000) + restarts)
                    noise = rng_r.random(n) * 0.02
                    a = a + noise
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    restarts += 1
                    patience = 0
                else:
                    break
        else:
            a = (1.0 - best_alpha) * a + best_alpha * a_star
            a = np.maximum(a, 0)
            a /= a.sum()
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0

    return best_a, best_obj


def mirror_descend(a0, time_limit, t_param=50.0, eta0=0.3, use_nesterov=True):
    """Mirror descent with smoothed subgradient and optional Nesterov acceleration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a /= a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()

    a_prev = a.copy()
    mom = 0.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if use_nesterov and it > 1:
            a_x = a + mom * (a - a_prev)
            a_x = np.maximum(a_x, 0)
            s_x = a_x.sum()
            if s_x < 1e-12:
                a_x = a.copy()
            else:
                a_x /= s_x
        else:
            a_x = a

        conv_x = fast_convolve(a_x)
        max_conv = conv_x.max()
        if max_conv < 1e-15:
            break

        # Smoothed weights
        scaled = t_param * conv_x / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(conv_x))] = 1.0
        else:
            w /= w_sum

        # Compute subgradient efficiently
        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end + 1] += 2.0 * w[k] * a_x[k - j_end:k - j_start + 1][::-1]

        # Mirror step
        lf = -eta * sg
        lf -= lf.max()
        a_new = a_x * np.exp(lf)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new /= s_new

        # Update momentum
        if use_nesterov:
            mom = (mom + 1.0) / (mom + 2.0)
            a_prev = a.copy()
            a = a_new

        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 30 == 0:
                eta *= 0.7
                if eta < 0.01:
                    t_param = min(t_param * 1.4, 600.0)
                    eta = 0.12
            if patience > 300:
                break

    return best_a, best_obj


def mirror_descend_exact(a0, time_limit, eta0=0.3, use_nesterov=True):
    """Mirror descent using exact subgradient at argmax positions only."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a /= a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()

    a_prev = a.copy()
    mom = 0.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if use_nesterov and it > 1:
            a_x = a + mom * (a - a_prev)
            a_x = np.maximum(a_x, 0)
            s_x = a_x.sum()
            if s_x < 1e-12:
                a_x = a.copy()
            else:
                a_x /= s_x
        else:
            a_x = a

        conv_x = fast_convolve(a_x)
        max_conv = conv_x.max()
        if max_conv < 1e-15:
            break

        # Exact subgradient: use top few positions
        n_top = max(3, n // 64)
        top_idx = np.argpartition(conv_x, -n_top)[-n_top:]

        # Average subgradient over top positions
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end + 1] += 2.0 * a_x[k - j_end:k - j_start + 1][::-1]
        sg /= len(top_idx)

        # Mirror step
        lf = -eta * sg
        lf -= lf.max()
        a_new = a_x * np.exp(lf)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new /= s_new

        if use_nesterov:
            mom = (mom + 1.0) / (mom + 2.0)
            a_prev = a.copy()
            a = a_new

        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 25 == 0:
                eta *= 0.75
                if eta < 0.008:
                    eta = 0.1
            if patience > 250:
                break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_top=8):
    """Coordinate descent: reduce mass from high-contributing positions."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a /= a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        top_k = np.argsort(conv)[-n_top:]

        improved = False
        for k in top_k:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            indices = range(j_start, j_end + 1)

            # Find the pair with max contribution
            max_contrib = -1
            max_j = -1
            for j in indices:
                c = a[j] * a[k - j]
                if c > max_contrib:
                    max_contrib = c
                    max_j = j

            if max_j < 0 or a[max_j] < 1e-7:
                continue

            k_j = k - max_j

            # Try reducing a[max_j] and adding to a less important position
            for frac in [0.3, 0.15, 0.05]:
                delta = frac * a[max_j]
                a_trial = a.copy()
                a_trial[max_j] -= delta

                # Distribute to positions that don't contribute to top conv positions
                # Simple: add to minimum element
                min_pos = np.argmin(a_trial)
                a_trial[min_pos] += delta
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-12:
                    continue
                a_trial /= s

                obj = 2.0 * n * fast_convolve(a_trial).max()
                if obj < best_obj - 1e-11:
                    a = a_trial
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 80:
                break
        else:
            no_improve = 0

    return best_a, best_obj


def gen_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
        if pp.sum() > 1e-10:
            inits['parent'] = pp / pp.sum()

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits['g_%d' % int(sf * 100)] = g / g.sum()

    w = max(1, n // 4)
    bi = np.ones(n) * 0.2
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.05)
    inits['tri'] = tri / tri.sum()

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    cb = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    cb2 = 0.5 * (1.0 + np.cos(4.0 * np.pi * x / n))
    inits['cos2'] = cb2 / cb2.sum()

    inits['rand'] = rng.dirichlet(np.ones(n) * 0.4)

    # Superposition: blend parent with structure
    if parent is not None and 'parent' in inits:
        p = inits['parent']
        for blend in [0.3, 0.5, 0.7]:
            v = (1.0 - blend) * p + blend * np.ones(n) / n
            v = np.maximum(v, 0)
            v /= v.sum()
            inits['blend_%d' % int(blend * 10)] = v

    return inits

            tr = min(tr_total / (len(phases) - i), 35.0)
            if tp > 0:
                f_new, _ = func(f, time_limit=tr, t_param=tp, eta0=eta)
            else:
                f_new, _ = func(f, time_limit=tr, eta0=eta)

            sc = float(evaluate_sequence(f_new.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = f_new.copy()
                f = f_new.copy()

            # Also try a final LP step
            if tr_total > 8:
                f_lp, _ = lp_optimize(f, time_limit=min(10.0, tr_total - 3))
                sc_lp = float(evaluate_sequence(f_lp.tolist()))
                if np.isfinite(sc_lp) and sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = f_lp.copy()
                    f = f_lp.copy()

            tr_total = deadline - time.monotonic() - 5
            print("  Final %d: score=%.6f" % (i, best_score))

    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print("FINAL: n=%d, score=%.6f" % (len(best_seq), best_score))
    return [float(x) for x in best_seq]
