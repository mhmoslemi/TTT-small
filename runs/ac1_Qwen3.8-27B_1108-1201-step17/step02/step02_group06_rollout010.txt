            stagnant += 1
        
        # Soft max gradient
        c_shifted = (conv - c_max) / tau
        exp_c = np.exp(c_shifted)
        w = exp_c / np.sum(exp_c)
        
        g_soft = 2.0 * np.correlate(w, a, mode='valid')
        dot_ga = np.dot(g_soft, a)
        grad_theta = (2.0 * n / S_target**2) * a * (g_soft - dot_ga / S_target)
        
        # Adam
        step += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad_theta
        v_adam = beta2 * v_adam + (1 - beta2) * (grad_theta ** 2)
        m_hat = m_adam / (1 - beta1**step)
        v_hat = v_adam / (1 - beta2**step)
        
        # Adaptive learning rate
        effective_lr = lr / (1 + 0.001 * stagnant)
        theta -= effective_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Anneal tau
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * max(0.05, (1 - elapsed_frac) ** 1.5)
        
        # Periodic perturbation
        if stagnant > 200 and rng.random() < 0.1:
            theta += rng.normal(0, 0.05, n)
            stagnant = 0
    
    return best_a, best_score
