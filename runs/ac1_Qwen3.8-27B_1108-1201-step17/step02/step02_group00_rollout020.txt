        a = a / s
    n = len(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based directional optimization
with adaptive blending, subgradient refinement, and multi-length search.
The LP approach finds the optimal direction to move in the simplex that
maintains the convolution peak bound, leading to faster convergence than
pure subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Build the constraint matrix for the LP: (f*g)_k for all k."""
    # A[k, j] = 1 if f_{k-j} is the coefficient, i.e., A[k,j] multiplies f_{k-j}*g_j
    # Actually, for a given f, the constraint is sum_j f_{k-j} * g_j <= M
    # So the matrix is A[k, j] = f_{k-j} (if valid)
    # But f changes each iteration, so we can't pre-build this.
    # Instead, we pre-build the index structure.
    pairs = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            pairs.append((k, j, k - j))
    return pairs


def solve_lp_direction(f, M, pairs):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix
    A_ub = np.zeros((num_constraints, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    
    b_ub = np.full(num_constraints, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                              'primal_feasibility_tolerance': 1e-9})
    
    if result.success:
        return np.maximum(result.x, 0.0)
    return None


def optimize_lp_directional(f_init, time_limit, blend_init=0.5, blend_min=0.01,
                            blend_halve_every=10):
    """LP-based directional optimization."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    pairs = build_lp_matrix(n)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        g = solve_lp_direction(a, M, pairs)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        g = g / g_sum
        
        # Blend
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive blend rate
        if stall_count >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall_count = 0
            if blend < blend_min:
                break
    
    return best_a, best_obj


def optimize_subgrad(a0, time_limit, lr_init=0.02, momentum=0.9, top_k=50,
                     temp_init=0.0003):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        
        # Softmax weighting
        scaled = peak_vals / max(temp_init, peak_vals.max() * 0.1)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]
        
        # Update
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        lr *= 0.9998
    
    return best_a, best_obj


def coordinate_descent(a_in, n_steps=100):
    """Simple coordinate descent: try small perturbations at each coordinate."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for step in range(n_steps):
        improved = False
        for i in range(n):
            # Try increasing a_i and decreasing others proportionally
            delta = a[i] * 0.1 + 0.01 / n
            if a[i] + delta > 1.0:
                continue
            
            # Create new vector
            a_test = a.copy()
            a_test[i] += delta
            # Reduce others proportionally
            others_mask = np.arange(n) != i
            others_sum = a_test[others_mask].sum()
            if others_sum > 1e-15:
                a_test[others_mask] *= (1.0 - a_test[i]) / others_sum
                a_test = np.maximum(a_test, 0.0)
                s_test = a_test.sum()
                if s_test > 1e-15:
                    a_test /= s_test
            
            conv_test = fast_convolve(a_test)
            obj_test = 2.0 * n * conv_test.max()
            
            if obj_test < best_obj - 1e-12:
                a = a_test
                best_obj = obj_test
                conv = conv_test
                improved = True
        
        if not improved:
            # Try decreasing some coordinates
            for i in range(n):
                if a[i] < 1e-6:
                    continue
                delta = a[i] * 0.1
                a_test = a.copy()
                a_test[i] -= delta
                others_mask = np.arange(n) != i
                others_sum = a_test[others_mask].sum()
                if others_sum > 1e-15:
                    a_test[others_mask] *= (1.0 - a_test[i]) / others_sum
                    a_test = np.maximum(a_test, 0.0)
                    s_test = a_test.sum()
                    if s_test > 1e-15:
                        a_test /= s_test
                
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                
                if obj_test < best_obj - 1e-12:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
        
        if not improved:
            break
    
    return a


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Triangular
    tri = np.minimum(x + 1, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Edge-concentrated
    edge = np.zeros(n)
    edge[:max(1, n//8)] = 1.0
    edge[-max(1, n//8):] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.25*n))**2)
    inits['gauss'] = g / g.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Ramp
    ramp = np.arange(1, n+1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        elif len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_up'] = p / ps
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based directional optimization
with adaptive blending, subgradient refinement, and multi-length search.
The LP finds the optimal direction in the simplex maintaining the convolution
peak bound, yielding faster convergence than pure subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_pairs(n):
    """Pre-build index pairs for LP constraint matrix."""
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs


def solve_lp_direction(f, M, pairs, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def optimize_lp(f_init, time_limit, blend_init=0.5, blend_min=0.005,
                blend_halve_every=12, sparsity_blend=0.3):
    """LP-based directional optimization with adaptive blending."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    pairs = build_lp_pairs(n)
    uniform = np.ones(n) / n

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    prev_obj = best_obj

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(a, M, pairs, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # If LP solution is too sparse, blend with uniform
        max_g = g.max()
        if max_g > 0.5:
            g = (1.0 - sparsity_blend) * g + sparsity_blend * uniform

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break

        # If no progress for many iterations, try a perturbation

            a = project_simplex(a + rng_local.standard_normal(n) * 0.01)

            v = np.zeros(n)
            stall = 0

    return best_a, best_obj


def optimize_subgrad(a0, time_limit, lr_init=0.015, momentum=0.9,
                     top_k=40, temp_init=0.0002):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        scaled = peak_vals / (temp_init + pmax * 0.05)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            a = project_simplex(a + np.random.default_rng().standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            v *= 0.5
            no_improve = 0
            lr = max(0.002, lr * 0.7)

        lr *= 0.9997

    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=80):
    """Coordinate descent: try swapping mass between pairs of coordinates."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()

    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            if a[i] < 1e-7:
                continue
            # Try moving a small fraction of mass from i to j
            for j in range(n):
                if j == i:
                    continue
                delta = a[i] * 0.05
                if delta < 1e-8:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-13:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
        if not improved:
            break

    return a
