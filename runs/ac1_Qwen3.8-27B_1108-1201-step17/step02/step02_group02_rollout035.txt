    best_score = current_score

    temperature = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.25:
            # Targeted peak reduction
            c = compute_c(a)
            peak_idx = int(np.argmax(c))
            # Find contributing pairs
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n, peak_idx + 1)
            for p in range(lo, hi):
                contribs[p] = a[p] * a[peak_idx - p]
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.012, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(10, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.012, size=n_inc))
        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.60:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 12))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.75:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.1, 0.1) * a[i]
                candidate[i] = max(0, a[i] - delta)
                candidate[j] = max(0, a[j] + delta)
        elif r < 0.90:
            # Trilinear perturbation
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if len(set([i, j, k])) == 3:
                delta = rng.uniform(-0.08, 0.08) * a[i]
                candidate[i] = max(0, a[i] - 2 * delta)
                candidate[j] = max(0, a[j] + delta)
                candidate[k] = max(0, a[k] + delta)
        else:
            # Small global
            candidate = a * np.exp(rng.normal(0.0, 0.005, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        s_cand = np.sum(candidate)
        if s_cand < 0.01:
            continue
        candidate = candidate * S / s_cand
        
        new_score = fast_score(candidate)
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    """Generate chirp-based initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (2.0 * n))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Generate multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 6):
        amp = 0.2 / k
        freq = rng.uniform(0.5, 2.0) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Generate random initial sequence."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main optimization: combines L-BFGS on autocorrelation energy,
    LSE refinement, subgradient descent, and simulated annealing.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 60.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    S_target = 100.0  # Fixed sum for optimization
    
    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0:
            s = fast_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()
    
    # ===== Phase 1: L-BFGS on ||c||² from provided sequence =====
    print("\n=== Phase 1: L-BFGS ||c||² (primary sequence) ===")
    if start_seq is not None and time.monotonic() < deadline - 400:
        n = len(start_seq)
        a0 = start_seq.copy()
        a0 = np.clip(a0, 0, None)
        s0 = np.sum(a0)
        if s0 > 0.01:
            a0 = a0 * S_target / s0
        b0 = to_log_params(a0, S_target)
        
        t1_limit = min(200.0, deadline - time.monotonic() - 400)
        b_opt = run_lbfgs(b0, S_target, t1_limit, obj_type='norm_c_sq')
        a_opt = softmax_a(b_opt, S_target)
        s_opt = fast_score(a_opt)
        print(f"  L-BFGS ||c||²: score={s_opt:.6f}")
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.6f} ***")
    
    # ===== Phase 2: L-BFGS from multiple starting points =====
    print(f"\n=== Phase 2: L-BFGS multi-start exploration ===")
    configs = []
    if start_seq is not None:
        for n_try in [len(start_seq), len(start_seq) + 128, len(start_seq) + 256]:
            configs.append((n_try, 'resample'))
    
    for n_try in [256, 384, 512, 640, 768, 1024]:
        configs.append((n_try, 'chirp'))
        configs.append((n_try, 'random'))
    
    t2_start = time.monotonic()
    t2_budget = min(500.0, deadline - t2_start - 300)
    
    for ci, (n, init_type) in enumerate(configs):
        if time.monotonic() > deadline - 300:
            break
        time_left = deadline - time.monotonic() - 250
        if time_left < 30:
            break
        t_per = max(15.0, min(60.0, time_left / max(1, len(configs) - ci)))
        
        if init_type == 'resample' and best_seq is not None:
            a0 = resample_sequence(best_seq, n)
        elif init_type == 'chirp':
            a0 = make_chirp_init(n, 0.3 + 0.3 * rng.random(), rng)
        else:
            a0 = make_random_init(n, rng, spread=0.4)
        
        a0 = np.clip(a0, 0.01, None)
        s0 = np.sum(a0)
        a0 = a0 * S_target / s0
        b0 = to_log_params(a0, S_target)
        
        try:
            b_opt = run_lbfgs(b0, S_target, t_per, obj_type='norm_c_sq')
            a_opt = softmax_a(b_opt, S_target)
            s_opt = fast_score(a_opt)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, init={init_type}: score={s_opt:.6f} *** NEW BEST ***")
            else:
                print(f"  n={n}, init={init_type}: score={s_opt:.6f}")
        except Exception as e:
            print(f"  n={n}, init={init_type}: error {e}")
    
    # ===== Phase 3: LSE refinement on best sequence =====
    print(f"\n=== Phase 3: LSE refinement ===")

        b0 = to_log_params(a0, S_target)

        epsilons = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

                a_opt = softmax_a(b_opt, S_target)

    # ===== Phase 4: Subgradient descent polish =====

    print(f"\n=== Phase 5: Final L-BFGS refinement ===")

        n = len(best_seq)

        s0 = np.sum(a0)
        a0 = a0 * S_target / s0

    print(f"\n=== Phase 6: Simulated annealing polish ===")
