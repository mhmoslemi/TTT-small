    best_score = compute_score(a)

    while time.monotonic() - t0 < time_limit:
        conv = fft_self_conv(a)
        peak_idx = int(np.argmax(conv))
        peak_val = float(conv[peak_idx])

        # Find elements contributing most to the peak lag
        # conv[peak_idx] = sum_i a[i]*a[peak_idx-i]
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
            else:
                # Check the other direction
                j2 = i - peak_idx
                if 0 <= j2 < n and j2 != i:
                    pass

        # Top contributors
        top_k = max(3, n // 15)
        top_idx = np.argsort(contribs)[-top_k:]

        # Strategy: reduce top contributors, redistribute to "safe" indices
        candidate = a.copy()
        total_reduce = 0.0
        for idx in top_idx:
            red = candidate[idx] * rng.uniform(0.05, 0.25)
            candidate[idx] -= red
            total_reduce += red

        # Redistribute to random non-contributing indices
        safe = np.setdiff1d(np.arange(n), top_idx)
        if len(safe) > 0:
            n_add = max(3, n // 25)
            add_idx = rng.choice(safe, size=min(n_add, len(safe)), replace=False)
            for idx in add_idx:
                candidate[idx] += total_reduce / len(add_idx)

        candidate = np.maximum(candidate, 0.0)
        candidate = normalize_unit(candidate)

        new_score = compute_score(candidate)
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
        elif new_score < best_score * 1.001 and rng.random() < 0.2:
            a = candidate  # Accept slight worsenings for exploration

        # Also try: swap mass between two random elements
        if rng.random() < 0.3:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                candidate2 = a.copy()
                transfer = candidate2[i] * rng.uniform(0.1, 0.5)
                candidate2[i] -= transfer
                candidate2[j] += transfer
                candidate2 = np.maximum(candidate2, 0.0)
                candidate2 = normalize_unit(candidate2)
                s2 = compute_score(candidate2)
                if s2 < best_score:
                    best_score = s2
                    best_a = candidate2.copy()
                    a = candidate2

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search for minimum peak autocorrelation sequence.
    Uses coordinate descent as the core optimizer with multiple
    initializations, sequence lengths, and a targeted local search polish.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def try_update(a, label=""):
        nonlocal best_score, best_seq
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        s = compute_score(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()
            print(f"    *** NEW BEST: {s:.8f} ({label}, n={len(a)})")
            return True
        return False

    # Get provided sequence
    provided = None
    try:
        provided = np.array(height_sequence_1, dtype=np.float64)
        provided = np.clip(provided, 0.0, 1000.0)
        ps = compute_score(provided)
        print(f"Provided sequence: n={len(provided)}, score={ps:.8f}")
        try_update(provided, "provided")
    except Exception:
        pass

    # Define initialization strategies
    def get_inits(n, n_inits=4):
        inits = []
        labels = []
        if provided is not None:
            inits.append(init_interpolate(provided, n))
            labels.append("interp")
        inits.append(init_sine(n, rng))
        labels.append("sine")
        inits.append(init_cos_window(n, rng))
        labels.append("coswin")
        inits.append(init_multi_sine(n, rng))
        labels.append("multisine")
        inits.append(init_random_smooth(n, rng))
        labels.append("randsmooth")
        return inits[:n_inits], labels[:n_inits]

    # Phase 1: CD from multiple initializations and lengths
    print("\n=== Phase 1: Coordinate Descent ===")
    lengths = [512, 640, 768, 1024]

    for n in lengths:
        if time.monotonic() > deadline - 90:
            break
        remaining = deadline - time.monotonic()
        inits, labels = get_inits(n)
        t_per = max(10.0, min(40.0, remaining / (len(inits) + 1)))

        for a_init, label in zip(inits, labels):
            if time.monotonic() > deadline - 30:
                break
            t_lim = min(t_per, deadline - time.monotonic() - 20)
            if t_lim < 5:
                break
            print(f"  n={n}, init={label}, t_limit={t_lim:.0f}s")
            result = cd_optimize(a_init, time_limit=t_lim, verbose=False)
            try_update(result, f"CD-{label}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # Phase 2: Extended CD on the best sequence
    print("\n=== Phase 2: Extended CD on best ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 20:
            t_lim = min(remaining, 180.0)
            print(f"  Extended CD, t_limit={t_lim:.0f}s")
            result = cd_optimize(best_seq, time_limit=t_lim, verbose=True)
            try_update(result, "ext_CD")

    # Phase 3: Targeted local search on best
    print(f"\n=== Phase 3: Targeted Local Search ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            t_lim = min(remaining, 90.0)
            print(f"  TLR, t_limit={t_lim:.0f}s")
            result = targeted_local_search(best_seq, time_limit=t_lim, rng=rng)
            try_update(result, "TLR")

    # Phase 4: Another CD round on best (different random seed for exploration)
    print(f"\n=== Phase 4: Final CD polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            t_lim = min(remaining, 60.0)
            # Perturb slightly for exploration
            a_perturb = best_seq * (1.0 + 0.01 * rng.standard_normal(len(best_seq)))
            a_perturb = np.maximum(a_perturb, 0.0)
            result1 = cd_optimize(a_perturb, time_limit=t_lim * 0.5, verbose=False)
            try_update(result1, "perturb_CD")

            remaining2 = deadline - time.monotonic() - 10
            if remaining2 > 5:
                result2 = cd_optimize(best_seq, time_limit=remaining2, verbose=False)
                try_update(result2, "final_CD")

    # Final verification
    if best_seq is not None:
        final = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        fs = evaluate_sequence(final)
        print(f"\n=== FINAL: evaluate={fs:.8f}, local={best_score:.8f}, n={len(final)} ===")
        return final

    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    return [1.0] * 640
