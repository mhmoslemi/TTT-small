    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    """Project onto simplex {x >= 0, sum(x) = 1}."""
    n = len(a)
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

            if 0 <= j1 < n:
                g[i] += a[j1]
            j2 = i - k

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Subgradient mirror descent (multiplicative weights) for efficient optimization
2. LP-based direction refinement for principled descent
3. Random perturbations and diverse initializations for exploration
4. Local search (pairwise mass transfer) for final polish

import numpy as np
import time
import scipy.optimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_norm(a):
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

def simplex_project(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a):
    """Subgradient of max_k conv(a,a)[k] w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    cmax = c.max()
    K_max = np.where(c >= cmax - 1e-10)[0]
    g = np.zeros(n)
    for k in K_max:
        # For each i: g[i] += a[k-i] (if valid) + a[i-k] (if valid)
        # Vectorized:
        # Part 1: a[k-i] for i where 0 <= k-i < n
        #   i ranges from max(0, k-n+1) to min(n-1, k)
        i1_lo = max(0, k - n + 1)
        i1_hi = min(n - 1, k)
        if i1_lo <= i1_hi:
            # a[k-i] for i in [i1_lo, i1_hi]
            # = a[k-i1_hi : k-i1_lo+1] reversed
            g[i1_lo:i1_hi+1] += a[k-i1_hi:k-i1_lo+1][::-1]
        # Part 2: a[i-k] for i where 0 <= i-k < n
        #   i ranges from max(0, k) to min(n-1, k+n-1) = min(n-1, k+n-1)
        i2_lo = max(0, k)
        i2_hi = min(n - 1, k + n - 1)
        if i2_lo <= i2_hi:
            g[i2_lo:i2_hi+1] += a[i2_lo-k:i2_hi-k+1]
    return g

def mirror_descent(a0, time_limit, eta0=1.0, tau=500.0, use_smoothed=False, temp=0.1):
    """Mirror descent with multiplicative weights for minimizing max conv(a,a)."""
    n = len(a0)
    a = np.maximum(a0, 1e-30).copy()
    a /= a.sum()
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        eta = eta0 / (1.0 + it / tau)
        
        c = fast_convolve(a)
        
        if use_smoothed:
            # Smoothed subgradient using top peaks
            cmax = c.max()
            cmin = c.min()
            if cmax - cmin < 1e-12:
                break
            weights = np.exp((c - cmax) / temp)
            weights /= weights.sum()
            g = np.zeros(n)
            # Only use peaks with significant weight
            for k in np.where(weights > 0.01)[0]:
                i1_lo = max(0, k - n + 1)
                i1_hi = min(n - 1, k)
                if i1_lo <= i1_hi:
                    g[i1_lo:i1_hi+1] += weights[k] * a[k-i1_hi:k-i1_lo+1][::-1]
                i2_lo = max(0, k)
                i2_hi = min(n - 1, k + n - 1)
                if i2_lo <= i2_hi:
                    g[i2_lo:i2_hi+1] += weights[k] * a[i2_lo-k:i2_hi-k+1]
        else:
            g = compute_subgradient(a)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= s
        
        obj_new = obj_norm(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Early stopping check
        if it % 10000 == 0 and obj_new >= best_obj:
            # Try smaller eta
            eta0 *= 0.8
    
    return best_a, best_obj

def solve_lp_direction(f, M, n_constraints=None):
    """Find direction g>=0 maximizing sum(g) s.t. conv(f,g) <= M."""
    n = len(f)
    # Build constraint matrix
    # conv(f,g)[k] = sum_j f[j]*g[k-j] <= M
    # This is a linear constraint in g
    size = 2 * n - 1
    if n_constraints is not None and n_constraints < size:
        # Only use the most binding constraints
        fg = fast_convolve(f)  # Wait, this is conv(f,f), not what we need
        # Actually we need conv(f, g) which depends on g
        # Let's just use all constraints for simplicity
        pass
    
    # For larger n, this LP might be too slow. Use a subset of constraints.
    # The most important constraints are near the peak.
    c_ff = fast_convolve(f)
    k_order = np.argsort(c_ff)[::-1]
    n_use = min(size, max(n, 200))  # Use at least n constraints
    
    A_rows = []
    b_rows = []
    for k in k_order[:n_use]:

def lp_optimize(f0, time_limit):
    """LP-based iterative optimization."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s
    
    best_f = f.copy()
    best_obj = obj_norm(f)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break
        
        g = solve_lp_direction(f, M)
        
        if np.allclose(g, f, atol=1e-10):
            no_improve += 3
        else:
            # Line search
            best_t = 0.0
            best_t_obj = obj_norm(f)
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                a_trial = (1.0 - t) * f + t * g
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-12:
                    continue
                a_trial /= st
                o = obj_norm(a_trial)
                if o < best_t_obj:
                    best_t_obj = o
                    best_t = t
            
            if best_t > 0 and best_t_obj < best_obj - 1e-10:
                f = (1.0 - best_t) * f + best_t * g
                f = np.maximum(f, 0.0)
                f /= f.sum()
                best_obj = best_t_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve >= 30:
            break
    
    return best_f, best_obj

def local_search(a, time_limit):
    """Pairwise mass transfer local search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    
    while time.monotonic() - t0 < time_limit and stuck < 500:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Find contributors to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        max_c = contrib.max()
        if max_c < 1e-15:
            break
        
        # Donors: high contribution to peak
        # Recipients: low contribution to peak
        sorted_idx = np.argsort(contrib)[::-1]
        donors = sorted_idx[:max(1, n // 10)]
        recipients = sorted_idx[-max(1, n // 10):]
        
        found = False
        for di in donors:
            if found:
                break
            for ri in recipients:
                if found:
                    break
                if di == ri:
                    continue
                for frac in [0.2, 0.5, 0.8]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[ri] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-10:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break
        
        if not found:
            stuck += 1
        else:
            stuck = 0
    
    return a, best_obj

def make_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    for sf in [0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g / g.sum()
    
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        hann = np.maximum(hann, 0.0)
        inits['hann'] = hann / hann.sum()
    
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Edge-concentrated
    edges = np.zeros(n)
    m = max(1, n // 3)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()
    
    # Alternating (high-low pattern)
    alt = np.ones(n)
    alt[1::2] = 0.3
    inits['alternating'] = alt / alt.sum()
    
    # Concentrated center
    center = np.zeros(n)
    cm = max(1, n // 5)
    center[n//2 - cm:n//2 + cm] = 1.0
    inits['center'] = center / center.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Subgradient mirror descent (multiplicative weights) for efficient per-iteration optimization
2. LP-based direction finding for principled descent when mirror descent stalls
3. Diverse initializations including parent sequence, structured patterns, and random
4. Random perturbations (kicks) to escape local minima
5. Pairwise mass-transfer local search for final polish
Multiple sequence lengths (256, 512, 1024) explored with adaptive time allocation.

import numpy as np
import time
import scipy.optimize


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective: 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def simplex_project(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a):
    """Vectorized subgradient of max_k conv(a,a)[k] w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    cmax = c.max()
    K_max = np.where(c >= cmax - 1e-10)[0]
    g = np.zeros(n)
    for k in K_max:
        i1_lo = max(0, k - n + 1)
        i1_hi = min(n - 1, k)
        if i1_lo <= i1_hi:
            g[i1_lo:i1_hi + 1] += a[k - i1_hi:k - i1_lo + 1][::-1]
        i2_lo = max(0, k)
        i2_hi = min(n - 1, k + n - 1)
        if i2_lo <= i2_hi:
            g[i2_lo:i2_hi + 1] += a[i2_lo - k:i2_hi - k + 1]
    return g


def compute_smoothed_gradient(a, temp=0.05):
    """Smoothed gradient using softmax-weighted peaks (top few)."""
    n = len(a)
    c = fast_convolve(a)
    cmax = c.max()
    cmin = c.min()
    if cmax - cmin < 1e-14:
        return np.zeros(n)
    weights = np.exp((c - cmax) / temp)
    total_w = weights.sum()
    if total_w < 1e-30:
        return np.zeros(n)
    weights /= total_w
    # Only use peaks with significant weight for efficiency
    significant = np.where(weights > 0.01 / len(c))[0]
    if len(significant) > 20:
        top_idx = np.argsort(weights)[::-1][:20]
        significant = top_idx
    g = np.zeros(n)
    for k in significant:
        w = weights[k]
        i1_lo = max(0, k - n + 1)
        i1_hi = min(n - 1, k)
        if i1_lo <= i1_hi:
            g[i1_lo:i1_hi + 1] += w * a[k - i1_hi:k - i1_lo + 1][::-1]
        i2_lo = max(0, k)
        i2_hi = min(n - 1, k + n - 1)
        if i2_lo <= i2_hi:
            g[i2_lo:i2_hi + 1] += w * a[i2_lo - k:i2_hi - k + 1]
    return g


def mirror_descent(a0, time_limit, eta0=1.0, tau=500.0, smoothed=False, temp=0.05,
                   stall_patience=5000, noise_level=0.05):
    """
    Mirror descent with multiplicative weights.
    Minimizes max_k conv(a,a)[k] subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-30).copy()
    a /= a.sum()
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    it = 0
    stall_count = 0
    eta_current = eta0

    while time.monotonic() - t0 < time_limit:
        it += 1
        eta = eta_current / (1.0 + it / tau)
        if eta < 1e-6:
            eta = 1e-6

        if smoothed:
            g = compute_smoothed_gradient(a, temp)
        else:
            g = compute_subgradient(a)

        # Multiplicative update
        log_update = -eta * g
        # Clip to prevent underflow/overflow
        log_update = np.clip(log_update, -50.0, 50.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= s

        obj_new = obj_norm(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        # Stall handling
        if stall_count >= stall_patience:
            stall_count = 0
            eta_current *= 0.7
            # Add noise to escape local optimum
            noise = noise_level * rng_global.standard_normal(n)
            a_noisy = a * (1.0 + noise)
            a_noisy = np.maximum(a_noisy, 1e-15)
            s = a_noisy.sum()
            if s > 1e-10:
                a = a_noisy / s

    return best_a, best_obj


def solve_lp_direction(f, M, max_constraints=None):
    """Find g>=0 maximizing sum(g) s.t. conv(f,g)[k] <= M for all k."""
    n = len(f)
    size = 2 * n - 1

    # Use subset of constraints (most binding ones)
    if max_constraints is None:
        max_constraints = min(size, max(n, 256))

    # Rank constraints by current binding-ness (use conv(f,f) as proxy)
    c_ff = fast_convolve(f)
    k_order = np.argsort(c_ff)[::-1]
    k_use = k_order[:max_constraints]

    A_rows = np.zeros((len(k_use), n))
    for idx, k in enumerate(k_use):
        i1_lo = max(0, k - n + 1)
        i1_hi = min(n - 1, k)
        if i1_lo <= i1_hi:
            # conv(f,g)[k] = sum_j f[j]*g[k-j]
            # For fixed k, the coefficient of g[j] is f[k-j]
            # j ranges where 0<=k-j<n, i.e., k-n+1<=j<=k
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            A_rows[idx, j_lo:j_hi + 1] = f[k - j_hi:k - j_lo + 1][::-1]

    b = np.full(len(k_use), M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = scipy.optimize.linprog(c_obj, A_ub=A_rows, b_ub=b, bounds=bounds,
                                     method='highs', options={'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize(f0, time_limit):
    """LP-based iterative optimization with line search."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s

    best_f = f.copy()
    best_obj = obj_norm(f)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break

        max_c = min(2 * n - 1, max(n, 256))
        g = solve_lp_direction(f, M, max_constraints=max_c)

        if np.allclose(g, f, atol=1e-10):
            no_improve += 3
        else:
            best_t = 0.0
            best_t_obj = obj_norm(f)
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                a_trial = (1.0 - t) * f + t * g
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-12:
                    continue
                a_trial /= st
                o = obj_norm(a_trial)
                if o < best_t_obj:
                    best_t_obj = o
                    best_t = t

            if best_t > 0 and best_t_obj < best_obj - 1e-10:
                f = (1.0 - best_t) * f + best_t * g
                f = np.maximum(f, 0.0)
                f /= f.sum()
                best_obj = best_t_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve >= 20:
            break

    return best_f, best_obj


def local_search(a, time_limit):
    """Pairwise mass-transfer local search targeting peak reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit and stuck < 300:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        # Compute contribution of each index to the peak
        contrib = np.zeros(n)
        i1_lo = max(0, k_star - n + 1)
        i1_hi = min(n - 1, k_star)
        if i1_lo <= i1_hi:
            contrib[i1_lo:i1_hi + 1] += a[k_star - i1_hi:k_star - i1_lo + 1][::-1]
        i2_lo = max(0, k_star)
        i2_hi = min(n - 1, k_star + n - 1)
        if i2_lo <= i2_hi:
            contrib[i2_lo:i2_hi + 1] += a[i2_lo - k_star:i2_hi - k_star + 1]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        # Sort by contribution: high contributors (donors) and low (recipients)
        sorted_idx = np.argsort(contrib)[::-1]
        n_donor = max(3, n // 15)
        n_recip = max(3, n // 15)
        donors = sorted_idx[:n_donor]
        recipients = sorted_idx[-n_recip:]

        found = False
        for di in donors:
            if found:
                break
            for ri in recipients:
                if found:
                    break
                if di == ri:
                    continue
                for frac in [0.15, 0.3, 0.5, 0.7]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[ri] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-10:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break

        if not found:
            stuck += 1
        else:
            stuck = 0

    return a, best_obj


def make_inits(n, rng):
    """Generate diverse initializations for sequence of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits['gauss_' + str(sf)] = g / g.sum()

    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        hann = np.maximum(hann, 0.0)
        hsum = hann.sum()
        if hsum > 0:
            inits['hann'] = hann / hsum

    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    edges = np.zeros(n)
    m = max(1, n // 4)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    alt = np.ones(n)
    alt[1::2] = 0.3
    inits['alternating'] = alt / alt.sum()

    center = np.zeros(n)
    cm = max(1, n // 6)
    center[n // 2 - cm:n // 2 + cm] = 1.0
    csum = center.sum()
    if csum > 0:
        inits['center'] = center / csum

    # Two-peak pattern
    twopeak = np.zeros(n)
    twopeak[n // 4] = 1.0
    twopeak[3 * n // 4] = 1.0
    tp = np.zeros(n)
    for i in range(n):
        tp[i] = np.exp(-0.5 * ((x[i] - n / 4) / (n / 10)) ** 2) + \
                np.exp(-0.5 * ((x[i] - 3 * n / 4) / (n / 10)) ** 2)
    tps = tp.sum()
    if tps > 0:
        inits['twopeak'] = tp / tps

    return inits

                    best_score = score_ls
                    best_sequence = a_ls.copy()
                    print("  Polish {} LS: score={:.6f}".format(polish_iter, best_score))

            # Fine mirror descent
            remaining = polish_deadline - time.monotonic()
            if remaining < 10:
                break
            md_time = min(remaining * 0.3, 45.0)
            if md_time > 3:
                a_md, obj_md = mirror_descent(best_sequence, time_limit=md_time,
                                              eta0=0.3, tau=500, smoothed=True, temp=0.03)
                score_md = float(evaluate_sequence(a_md.tolist()))
                if np.isfinite(score_md) and score_md < best_score - 1e-10:
                    best_score = score_md
                    best_sequence = a_md.copy()
                    print("  Polish {} MD: score={:.6f}".format(polish_iter, best_score))

            # LP (only for n <= 512)
            remaining = polish_deadline - time.monotonic()
            if remaining < 15 or len(best_sequence) > 512:
                break
            lp_time = min(remaining * 0.25, 40.0)
            if lp_time > 5:
                a_lp, obj_lp = lp_optimize(best_sequence, time_limit=lp_time)
                score_lp = float(evaluate_sequence(a_lp.tolist()))
                if np.isfinite(score_lp) and score_lp < best_score - 1e-10:
                    best_score = score_lp
                    best_sequence = a_lp.copy()
                    print("  Polish {} LP: score={:.6f}".format(polish_iter, best_score))

    # Final evaluation
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print("FINAL: score={:.6f}, n={}".format(best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
