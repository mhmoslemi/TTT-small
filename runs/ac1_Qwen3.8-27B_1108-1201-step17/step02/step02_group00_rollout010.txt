Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B on a smooth
log-sum-exp approximation with adaptive temperature, combined with
projected gradient refinement and multi-start search.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_grad(a, alpha):
    """Compute log-sum-exp objective value and gradient."""
    n = len(a)
    conv = fast_convolve(a)
    x = alpha * conv
    x_max = x.max()
    shifted = x - x_max
    w = np.exp(shifted)
    w_sum = w.sum()
    w /= w_sum
    lse_val = (x_max + np.log(w_sum)) / alpha
    
    # Gradient: 2 * sum_k w[k] * a[k-m] for m = 0..n-1
    fft_size = 1 << (len(conv) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(W) * A, fft_size)
    grad = 2.0 * corr[:n]
    
    return lse_val, grad

    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    
    return val, grad_z

    n = len(a)
    v = np.zeros(n)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_val = float('inf')
    
    while time.monotonic() - t0 < time_limit:
        val, grad = compute_lse_grad(a, alpha)
        
        if val < best_val:
            best_val = val
            best_a = a.copy()
        
        # Backtracking line search
        lr_cur = lr
        while lr_cur > 1e-10:
            a_trial = project_simplex(a - lr_cur * grad + v)
            val_trial, _ = compute_lse_grad(a_trial, alpha)
            if val_trial < val:
                break
            lr_cur *= 0.5
        
        v = momentum * v - lr_cur * grad
        a = project_simplex(a + v)
    
    return best_a

def compute_score(a):
    """Evaluate the actual objective (not smoothed)."""
    n = len(a)
    conv = fast_convolve(a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * conv.max() / (sum_a ** 2)

**Deterministic "peak-shaving" algorithm:**
1. Start with the parent solution.
2. At each step:
   a. Compute the autocorrelation c = (a*a).
   b. Find the set S of "significant" peaks (where c[k] > threshold * max(c)).
   c. For each peak k in S, compute the "contribution vector" v_k where v_k[j] = a[k-j] (the part of a that contributes to peak k).
   d. Compute the "average contribution" v_avg = (1/|S|) * sum_{k in S} v_k.
   e. Decrease a in the direction of v_avg: a_new = a - eta * v_avg.
   f. Project onto the simplex.
   g. If the objective improved, accept; otherwise, reduce eta and retry.

This is essentially a coordinate-wise "shrinkage" of the parts of a that contribute to the peaks. It's more targeted than a full gradient step.
