def softmax_smoothed_objective_and_grad(z, beta):
    """
    z: log-parameterization, a = exp(z)
    Returns: (objective_value, gradient_wrt_z)
    """
    a = np.exp(z)
    n = len(a)
    S = np.sum(a)
    C = fft_convolve(a, a)  # length 2n-1
    
    # Softmax-smoothed max
    C_max = np.max(C)
    exp_C = np.exp(beta * (C - C_max))
    sum_exp = np.sum(exp_C)
    w = exp_C / sum_exp
    soft_max = (C_max + np.log(sum_exp)) / beta  # This is just (1/beta)*log(sum(exp(beta*C)))

    # = (log(sum_exp) + beta*C_max)/beta = C_max + log(sum_exp)/beta

    soft_max = C_max + np.log(sum_exp) / beta
    
    objective = 2.0 * n * soft_max / (S * S)
    
    # Gradient of soft_max w.r.t. a
    # grad_a[i] = 2 * convolve(w, a, 'full')[i + (n-1)]

    # Chain rule: df/dz_i = df/da_i * da_i/dz_i = df/da_i * a_i
    grad_z = grad_f_da * a
    
    return objective, grad_z

"""
Smoothed gradient-based optimization for minimizing peak autocorrelation of
non-negative step functions. Uses softmax-smoothed max with Adam optimizer in
log-parameterization space, beta annealing, multiple restarts with varied
sequence lengths and initializations, and coordinate-descent polishing with
the exact (non-smoothed) objective. Differs from LP-based approach by using
continuous_gradient_descent on a smooth approximation of the non-differentiable
max operator.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast local evaluation of the objective."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def smoothed_obj_and_grad(z, beta):
    """
    Compute softmax-smoothed objective and gradient w.r.t. z where a = exp(z).
    Returns (objective, grad_z).
    """
    a = np.exp(z)
    n = len(a)
    S = np.sum(a)
    C = fft_convolve(a, a)

    # Numerically stable softmax
    C_max = np.max(C)
    exp_shifted = np.exp(beta * (C - C_max))
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    soft_max = C_max + np.log(sum_exp) / beta

    objective = 2.0 * n * soft_max / (S * S)

    # Gradient of soft_max w.r.t. a:
    # grad_a[i] = sum_k w[k] * (a[i+k-(n-1)] + a[i-k+(n-1)])
    #           = 2 * sum_k w[k] * a[i+k-(n-1)]  (by symmetry of w)
    # This is 2 * cross-correlation, computed as convolve(w, a) at positions n-1 .. 2n-2
    r_full = fft_convolve(w, a)
    r = r_full[n - 1:2 * n - 1]
    grad_soft_da = 2.0 * r

    # Chain rule for objective = 2n * soft_max / S^2
    grad_obj_da = (2.0 * n / (S * S)) * grad_soft_da - (2.0 * n * soft_max / (S ** 3))

    # Chain through a = exp(z): da/dz = a
    grad_z = grad_obj_da * a

    # Project out the scale-invariant direction (all-ones in z-space)
    grad_z = grad_z - np.mean(grad_z)

    return objective, grad_z


def adam_optimize_smoothed(z0, n_steps, lr, beta_start, beta_end, time_limit, rng, verbose=False):
    """Run Adam optimizer on the smoothed objective with beta annealing."""
    z = z0.copy().astype(np.float64)
    n = len(z)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8

    best_obj = np.inf
    best_z = z.copy()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal beta
        progress = step / max(1, n_steps - 1)
        beta = beta_start + (beta_end - beta_start) * progress

        obj, grad = smoothed_obj_and_grad(z, beta)

        if obj < best_obj:
            best_obj = obj
            best_z = z.copy()

        # ReLU-like fix: if numerical issues cause nan/inf
        if not np.isfinite(obj) or np.any(~np.isfinite(grad)):
            z = best_z.copy()
            continue

        step += 0  # just for counting (we're in the loop already)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Keep z in a reasonable range for numerical stability
        z_max = np.max(z)
        if z_max > 30:
            z = z - z_max + 10

        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      step={step}, obj={obj:.6f}, best={best_obj:.6f}, beta={beta:.2f}, elapsed={elapsed:.1f}s")

    return best_z, best_obj


def coord_descent_polish(a, time_limit, rng, max_passes=20):
    """
    Coordinate descent polish using the exact (non-smoothed) objective.
    For each coordinate, try several multipliers and keep the best.
    """
    a = a.copy().astype(np.float64)
    n = len(a)
    current_score = local_score(a)
    best_score = current_score
    best_a = a.copy()
    t0 = time.monotonic()

    order = np.arange(n)

    for _ in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        np.random.seed(int(rng.integers(0, 2**31)))
        perm = np.random.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-10:
                # Try turning on
                trial = a.copy()
                trial[i] = 0.1 * np.mean(a[a > 0]) if np.any(a > 0) else 1.0
                s = local_score(trial)
                if s < current_score:
                    a = trial
                    current_score = s
                continue

            # Try several multipliers
            best_local = current_score
            best_mult = 1.0
            for mult in [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0]:
                trial = a.copy()
                trial[i] = a[i] * mult
                s = local_score(trial)
                if s < best_local:
                    best_local = s
                    best_mult = mult

            if best_mult != 1.0:
                a[i] = a[i] * best_mult
                if best_local < current_score:
                    current_score = best_local
                    if current_score < best_score:
                        best_score = current_score
                        best_a = a.copy()

    return best_a, best_score


def targeted_peak_reduction(a, n_iters, rng):
    """
    Iteratively identify the peak lag and reduce contributions to it.
    """
    a = a.copy().astype(np.float64)
    n = len(a)
    current_score = local_score(a)
    best_score = current_score
    best_a = a.copy()

    for _ in range(n_iters):
        C = fft_convolve(a, a)
        peak_idx = int(np.argmax(C))
        lag = peak_idx - (n - 1)

        # Find pairs (i, j) with j - i = lag (or i - j = lag) that contribute to the peak
        if lag >= 0:
            pairs = [(i, i + lag) for i in range(n - lag)]
        else:
            ll = -lag
            pairs = [(i - ll, i) for i in range(ll, n)]

        if not pairs:
            break

        # Compute contributions
        contribs = np.array([a[i] * a[j] for i, j in pairs])
        total = np.sum(contribs)
        if total < 1e-10:
            break

        # Reduce the top contributors
        top_k = max(3, len(pairs) // 8)
        top_indices = np.argsort(contribs)[-top_k:]

        reduced = False
        for idx in top_indices:
            i, j = pairs[idx]
            # Reduce the larger of the two
            if a[i] >= a[j]:
                a[i] *= 0.85
            else:
                a[j] *= 0.85
            a = np.clip(a, 0.0, None)

            s = local_score(a)
            if s < current_score:
                current_score = s
                reduced = True
                if s < best_score:
                    best_score = s
                    best_a = a.copy()
            else:
                # Revert
                if a[i] < a[j] and lag >= 0:
                    pass  # already modified, just continue

        if not reduced:
            # Try small random perturbations
            idx = int(rng.integers(0, n))
            a[idx] *= np.exp(rng.normal(-0.03, 0.02))
            a = np.clip(a, 0.0, None)
            s = local_score(a)
            if s < current_score * 1.001:
                current_score = s
                if s < best_score:
                    best_score = s
                    best_a = a.copy()

    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.5):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_binary_like_init(n, rng, density=0.5):
    """Create a binary-like sequence with some noise."""
    a = (rng.random(n) < density).astype(np.float64)
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_anti_palindromic_init(n, rng):
    """Initialize to be anti-palindromic to reduce the center autocorrelation."""
    half = n // 2
    a = rng.random(half) + 0.1
    # Make the second half inversely correlated
    b = rng.random(half) + 0.1
    if n % 2 == 0:
        a_full = np.concatenate([a, b[::-1] * 0.3 + 0.2])
    else:
        a_full = np.concatenate([a, [0.5], b[::-1] * 0.3 + 0.2])
    return a_full[:n]


def resample_sequence(seq, new_n):
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)
