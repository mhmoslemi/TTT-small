import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

    n = len(f)
    # A has shape (2n-1, n), A[k,j] = f[j] if k-j is in [0, n-1], else 0
    # Actually, conv(f, b)[k] = sum_j f[j] * b[k-j]
    # So A[k, j] = f[j] if 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

        A[j:j + n, j] = f
    return A

            return res.x
    except Exception:
        pass
    return None

        f_best, s_best = lp_optimize(f0, t_alloc, verbose=(i == 0))
        print(f"  => LP score={s_best:.6f} (global best={best_s:.6f})")

        if s_best < best_s:
            best_s = s_best
            best_seq = f_best.copy()
            print(f"  *** NEW BEST: {best_s:.6f} ***")

    # Phase 2: Peak-directed search on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 60:
        t_sec = min(120.0, deadline - time.monotonic() - 40.0)
        if t_sec > 20:
            print(f"\n=== Phase 2: Peak-directed search ({t_sec:.0f}s) ===")
            f2, s2 = peak_directed_search(best_seq, t_sec, verbose=True)
            if s2 < best_s:
                best_s = s2
                best_seq = f2.copy()
                print(f"  After peak search: {best_s:.6f}")

    # Phase 3: Another LP round on the refined sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        t_third = min(150.0, deadline - time.monotonic() - 20.0)
        if t_third > 20:
            print(f"\n=== Phase 3: Final LP refinement ({t_third:.0f}s) ===")
            f3, s3 = lp_optimize(best_seq, t_third, verbose=True)
            if s3 < best_s:
                best_s = s3
                best_seq = f3.copy()
                print(f"  After final LP: {best_s:.6f}")

    # Phase 4: Quick peak-directed polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_fourth = min(60.0, deadline - time.monotonic() - 5.0)
        if t_fourth > 10:
            print(f"\n=== Phase 4: Final peak polish ({t_fourth:.0f}s) ===")
            f4, s4 = peak_directed_search(best_seq, t_fourth, verbose=True)
            if s4 < best_s:
                best_s = s4
                best_seq = f4.copy()
                print(f"  After final peak polish: {best_s:.6f}")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 50}")
        return final_list

    # Fallback
    return [1.0] * 256
