```python
import numpy as np
from scipy.optimize import minimize
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def compute_autocorr(f):
    """Compute autocorrelation of f (full, length 2n-1)."""
    return fft_convolve(f, f)

def local_score(f):
    """Fast evaluation of the objective."""
    f = np.clip(f, 0.0, 1000.0)
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return np.inf
    c = compute_autocorr(f)
    return 2.0 * n * np.max(c) / (s * s)

def softmax_autocorr_gradient(f, beta):
    """
    Compute gradient of the soft-max of autocorrelation.
    Returns (gradient, soft_max_value, c)
    """
    n = len(f)
    c = compute_autocorr(f)
    # Soft-max
    max_c = np.max(c)
    exp_bc = np.exp(beta * (c - max_c))  # numerical stability
    sum_exp = np.sum(exp_bc)
    p = exp_bc / sum_exp  # probability weights
    soft_max_val = (max_c + np.log(sum_exp)) / beta  # = log(sum(exp(beta*c)))/beta
    
    # Gradient: dL/df[i] = 2 * sum_k p[k] * f[k-i] (for valid k-i)
    # = 2 * (p cross-correlate f)[i]
    # = 2 * conv(f, p_rev)[2n-2-i]
    p_rev = p[::-1].copy()
    lp = len(p)  # = 2n-1
    
    size = 1
    while size < n + lp - 1:
        size <<= 1
    
    F = np.fft.rfft(f, size)
    Pr = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(F * Pr, size)
    
    # grad[i] = 2 * conv[2n-2-i] for i = 0, ..., n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    grad = 2.0 * conv[2*n-2 : n-1 : -1]
    
    return grad, soft_max_val, c, p

def soft_max_gradient_descent(f_init, time_limit, rng, verbose=True):
    """
    Minimize soft-max of autocorrelation using gradient descent.
    f is normalized to have sum = sqrt(2n).
    """
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f = f_init.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    sf = np.sum(f)
    if sf < 0.01:
        f = np.ones(n) * S / n
    else:
        f = f * S / sf
    
    current_score = local_score(f)
    best_f = f.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    beta = 5.0  # start with moderate smoothing
    lr = 0.1
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Increase beta gradually
        beta *= 1.02
        # Decrease learning rate
        lr = max(0.001, lr * 0.999)
        
        grad, sm_val, c, p = softmax_autocorr_gradient(f, beta)
        
        # Project gradient to maintain sum constraint
        # (subtract mean to make sum of gradient = 0)
        grad_centered = grad - np.mean(grad)
        
        # Update
        f_new = f - lr * grad_centered
        f_new = np.clip(f_new, 0, None)
        sf_new = np.sum(f_new)
        if sf_new < 0.01:
            f_new = f.copy()
        else:
            f_new = f_new * (S / sf_new)
        
        new_score = local_score(f_new)
        
        if new_score < current_score:
            f = f_new
            current_score = new_score
            stall_count = 0
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()
        else:
            stall_count += 1
            if stall_count > 50:
                # Perturb
                f = f * np.exp(rng.normal(0, 0.02, n))
                f = np.clip(f, 0, None)
                sf = np.sum(f)
                if sf > 0.01:
                    f = f * S / sf
                    current_score = local_score(f)
                stall_count = 0
                lr = 0.05
                beta = max(5.0, beta * 0.5)  # reduce beta to smooth more
        
        if verbose and int((time.monotonic() - t0) * 2) % 3 == 0:
            print(f"    smgd: beta={beta:.1f}, lr={lr:.4f}, score={current_score:.6f}, best={best_score:.6f}")
    
    return best_f, best_score
```
