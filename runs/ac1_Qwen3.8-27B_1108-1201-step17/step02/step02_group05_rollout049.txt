    return np.fft.irfft(A * A, fft_size)[:size]

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

def convolve_xy(a, b):
    """Linear convolution of a and b."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def line_search_convex(f, g, C, D, E):
    """Find t in [0,1] minimizing F(t) = max_k Q_k(t) where Q_k are quadratics.
    C_k = (f*f)[k], D_k = (f*g)[k], E_k = (g*g)[k]
    Q_k(t) = (1-t)^2 C_k + 2t(1-t) D_k + t^2 E_k
    """
    def Q_max(t):
        # Compute all Q_k(t) and return the max
        vals = (1-t)**2 * C + 2*t*(1-t) * D + t**2 * E
        return vals.max()
    
    def Q_grad_range(t):
        """Return (min, max) of Q_k'(t) over active k."""
        vals = (1-t)**2 * C + 2*t*(1-t) * D + t**2 * E
        M = vals.max()
        active = vals >= M - 1e-12
        # Q_k'(t) = -2(1-t)C_k + 2(1-2t)D_k + 2t E_k
        grads = -2*(1-t)*C + 2*(1-2*t)*D + 2*t*E
        return grads[active].min(), grads[active].max()
    
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        gmin, gmax = Q_grad_range(mid)
        if gmin > 1e-10:
            hi = mid
        elif gmax < -1e-10:
            lo = mid
        else:
            break
    return (lo + hi) / 2

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Multiplicative weight (mirror descent) updates guided by peak contributions
2. LP-based conditional gradient steps for principled descent
3. Convex line search along the search direction
4. Multi-peak flattening via active set expansion
Multiple sequence lengths and diverse initializations are explored.
"""

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Linear cross-convolution using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def obj_val(a):
    """Objective: 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

def compute_contributions(a, active_set):
    """For each index i, compute sum of a[k-i] over k in active_set."""
    n = len(a)
    contrib = np.zeros(n)
    for k in active_set:
        k = int(k)
        # For index i, j = k - i must be in [0, n-1]
        # So i must be in [k-(n-1), k] intersect [0, n-1]
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_min <= i_max:
            # contrib[i] += a[k - i] for i in [i_min, i_max]
            # a[k - i] for i in [i_min, i_max] is a[k - i_max : k - i_min + 1] reversed
            js = k - np.arange(i_min, i_max + 1)  # js = [k - i_min, ..., k - i_max]
            contrib[i_min:i_max+1] += a[js]
    return contrib

def mirror_descent_step(a, act_frac=0.95, eta=0.1):
    """One step of multiplicative weight / mirror descent."""
    conv = fast_convolve(a)
    M = conv.max()
    if M < 1e-15:
        return a, 0.0
    # Active set: positions within act_frac of max
    active = np.where(conv >= act_frac * M)[0]
    
    contrib = compute_contributions(a, active)
    
    # Normalized contribution for stable exponent
    c_max = contrib.max()
    if c_max < 1e-15:
        return a, M
    
    a_new = a * np.exp(-eta * contrib / c_max)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    
    return a_new, M

def mirror_descent_optimize(a0, time_limit, eta0=0.5, act_frac=0.95):
    """Iterative mirror descent optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        # Adaptive active fraction: start tight, loosen if stuck
        af = act_frac if no_improve < 100 else max(0.9, act_frac - 0.05 * (no_improve // 100))
        
        a_new, M = mirror_descent_step(a, act_frac=af, eta=eta)
        obj_new = obj_val(a_new)
        
        if obj_new < best_obj - 1e-14:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
            eta = max(eta * 0.995, 0.001)
        else:
            no_improve += 1
            eta = max(eta * 0.95, 0.001)
            if eta < 0.001:
                # Try with larger active set
                a_new, M = mirror_descent_step(a, act_frac=max(0.8, af - 0.1), eta=0.1)
                obj_new = obj_val(a_new)
                if obj_new < best_obj - 1e-14:
                    a = a_new
                    best_obj = obj_new
                    best_a = a.copy()
                    no_improve = 0
                    eta = 0.3
                else:

    return best_a, best_obj

def build_lp_matrix(n):
    """Build sparse LP constraint matrix for conv(f, g) <= M."""
    # conv(f, g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # Row k: A[k, j] = f[j] if 0 <= k-j < n
    # Use sparse representation
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return rows, cols

def solve_lp(f, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    
    A = sparse.csr_matrix((f[cols], (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()

def line_search_subgradient(C, D, E):
    """Find t in [0,1] minimizing max_k Q_k(t) where Q_k(t) = (1-t)^2 C_k + 2t(1-t) D_k + t^2 E_k.
    Uses bisection on the subgradient."""
    def F_and_subgrad(t):
        vals = (1-t)**2 * C + 2*t*(1-t) * D + t**2 * E
        M = vals.max()
        active = vals >= M - 1e-12
        # Q_k'(t) = -2(1-t)C_k + 2(1-2t)D_k + 2t E_k
        grads = -2*(1-t)*C + 2*(1-2*t)*D + 2*t*E
        return grads[active].min(), grads[active].max()
    
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        gmin, gmax = F_and_subgrad(mid)
        if gmin > 1e-10:
            hi = mid
        elif gmax < -1e-10:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

def lp_optimize(a0, time_limit):
    """LP-based optimization with convex line search."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    best_a = a.copy()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_f = fast_convolve(a)
        M = conv_f.max()
        
        remaining = time_limit - (time.monotonic() - t0)
        lp_time = min(5.0, remaining * 0.5)
        if lp_time < 1.0:
            break
        
        g = solve_lp(a, M, time_limit=lp_time)
        
        if np.allclose(g, a, atol=1e-10):
            no_improve += 1
        else:
            # Convex line search
            conv_fg = cross_convolve(a, g)
            conv_gg = fast_convolve(g)
            size = 2 * len(a) - 1
            if len(conv_fg) < size:
                conv_fg = np.pad(conv_fg, (0, size - len(conv_fg)))
            if len(conv_gg) < size:
                conv_gg = np.pad(conv_gg, (0, size - len(conv_gg)))
            
            t_opt = line_search_subgradient(conv_f, conv_fg, conv_gg)
            
            a_new = (1 - t_opt) * a + t_opt * g
            a_new = np.maximum(a_new, 0.0)
            ns = a_new.sum()
            if ns < 1e-12:
                break
            a_new /= ns
            
            obj_new = obj_val(a_new)
            if obj_new < best_obj - 1e-14:
                a = a_new
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve >= 20:
            break
    
    return best_a, best_obj

def get_parent(n):
    """Get parent sequence resized to length n."""
    try:

def make_inits(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n
    
    for sf in [0.2, 0.3, 0.5, 0.7]:
        g = np.exp(-0.5 * ((x - 0.5*n) / (sf*n))**2)
        inits[f'gauss_{sf}'] = g / g.sum()
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2*np.pi*x / (n-1))) if n > 1 else np.ones(n)
    cos_w = np.maximum(cos_w, 0)
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Edge-concentrated
    edges = np.zeros(n)
    m = max(1, n // 3)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.1) / 1.0
    
    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
        # Perturbed parent
        for scale in [0.01, 0.05, 0.1]:
            p_pert = parent + scale * rng.standard_normal(n)
            p_pert = np.maximum(p_pert, 0)
            if p_pert.sum() > 1e-10:
                inits[f'parent_pert_{scale}'] = p_pert / p_pert.sum()
    
    return inits

def compute_contributions(a, active_set):
    n = len(a)
    contrib = np.zeros(n)
    for k in active_set:
        k = int(k)
        i_min = max(0, k - (n - 1))

        if i_min <= i_max:
            indices = k - np.arange(i_min, i_max + 1)
            contrib[i_min:i_max+1] += a[indices]
    return contrib
