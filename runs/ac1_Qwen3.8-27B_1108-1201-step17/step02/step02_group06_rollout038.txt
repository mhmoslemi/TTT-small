def multiplicative_update(a, eta, tau=None):
    """One step of multiplicative peak reduction."""
    n = len(a)
    R = fft_convolve(a, a)
    M = np.max(R)
    
    if tau is None:
        # Hard max: only the peak lag
        active = R >= M * 0.99
    else:
        # Soft max: softmax weighting
        shifted = (R - M) / tau
        w = np.exp(shifted)
        w = w / np.sum(w)
        active = None  # use w directly
    
    # Compute pressure
    p = np.zeros(n)
    if active is not None:
        active_idx = np.where(active)[0]
        for k in active_idx:
            # a[k-i] for valid i: i in [max(0, k-n+1), min(k, n-1)]
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_min <= i_max:
                # a[k-i] for i in [i_min, i_max]
                # k-i ranges from k-i_max to k-i_min
                a_k_mi = a[k - i_max : k - i_min + 1][::-1]
                p[i_min:i_max+1] += a_k_mi
            
            # a[i-k] for valid i: i in [max(0, k), min(n-1, k+n-1)] = [k, min(n-1, k+n-1)]
            # Wait, i-k in [0, n-1] means i in [k, k+n-1], intersected with [0, n-1]
            # So i in [max(0,k), min(n-1, k+n-1)]
            i_min2 = max(0, k)
            i_max2 = min(n-1, k + n - 1)
            if i_min2 <= i_max2:
                # a[i-k] for i in [i_min2, i_max2]
                # i-k ranges from i_min2-k to i_max2-k
                p[i_min2:i_max2+1] += a[i_min2-k : i_max2-k+1]
    else:
        # Soft version
        for k_idx, w_val in enumerate(w):
            if w_val < 1e-10:
                continue
            i_min = max(0, k_idx - n + 1)
            i_max = min(k_idx, n - 1)
            if i_min <= i_max:
                a_k_mi = a[k_idx - i_max : k_idx - i_min + 1][::-1]
                p[i_min:i_max+1] += w_val * a_k_mi
            
            i_min2 = max(k_idx, 0)
            i_max2 = min(n-1, k_idx + n - 1)
            if i_min2 <= i_max2:
                p[i_min2:i_max2+1] += w_val * a[i_min2-k_idx : i_max2-k_idx+1]
    
    a_new = a * np.exp(-eta * p / M)
    a_new = np.clip(a_new, 1e-10, None)
    a_new = a_new / np.sum(a_new)
    return a_new

"""
Peak-directed multiplicative optimization with LP-assisted escaping
for minimizing normalized peak autocorrelation of non-negative sequences.

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_pressure_fast(a, w):
    """Compute pressure p[i] = sum_k w[k]*(a[k-i] + a[i-k]) efficiently.
    a has length n, w has length 2n-1.
    Returns p of length n."""
    n = len(a)
    # p2[i] = conv(w, a)[i] for i = 0..n-1
    ca = fft_convolve(w, a)  # length (2n-1) + n - 1 = 3n-2
    p2 = ca[:n]
    
    # p1[i] = conv(flip(w), a)[i + 2n-2] for i = 0..n-1
    ca2 = fft_convolve(w[::-1], a)  # same length
    p1 = ca2[2*n-2 : 3*n-2]
    
    return p1 + p2

def mul_update(a, eta, tau, top_k=None):
    """One multiplicative peak-reduction step.
    a: current sequence (normalized to sum=1)
    eta: step size
    tau: softmax temperature (0 means hard max)
    top_k: if set, only use top-k peaks
    Returns new a (normalized to sum=1)."""
    n = len(a)
    R = fft_convolve(a, a)  # length 2n-1
    M = np.max(R)
    if M < 1e-30:
        return a.copy()
    
    if tau == 0:
        # Hard max
        if top_k is not None:
            # Top-k peaks
            idx = np.argsort(R)[-top_k:]
            w = np.zeros(len(R))
            w[idx] = 1.0 / top_k
        else:
            active = R >= M * 0.999
            w = active.astype(float)
            w = w / np.sum(w)
    else:
        # Softmax
        shifted = (R - M) / tau
        w = np.exp(shifted)
        w = w / np.sum(w)
    
    p = compute_pressure_fast(a, w)
    a_new = a * np.exp(-eta * p / M)
    a_new = np.clip(a_new, 1e-15, None)
    a_new = a_new / np.sum(a_new)
    return a_new

def solve_lp_direction(f, M):
    """Solve LP to find best direction g: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    # Build constraint: conv(f, g)[k] <= M for all k
    # conv(f, g)[k] = sum_i f[i] * g[k-i]
    # Build A matrix where A[k, j] = coefficient of g[j] in conv[k]
    # conv[k] = sum_j f[k-j] * g[j] (when 0 <= k-j < n)
    # So A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] appears in conv[k] for k = j, j+1, ..., j+n-1
        # conv[k] includes f[k-j]*g[j]
        A[j:j+n, j] = f
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 10})
    if result.success:
        return result.x
    return None

Therefore: dM/dz_i = sum_k w_k * 2 * a_i * a_{k-i} * [0<=k-i<n]
= 2 * a_i * sum_k w_k * a_{k-i} * [0<=k-i<n]
= 2 * a_i * q_i

where q_i = sum_{k: 0<=k-i<n} w_k * a_{k-i}

And dS/dz_i = a_i.

So: dF/dz_i = 2*a_i*q_i/S^2 - 2*M*a_i/S^3 = 2*a_i/S^2 * (q_i - M/S)

With S=1: dF/dz_i = 2*a_i*(q_i - M)

So the gradient is proportional to a_i * (q_i - M).

The multiplicative update becomes:
z_i -= eta * 2*a_i*(q_i - M)
a_i *= exp(-2*eta*a_i*(q_i - M))

If q_i > M: a_i decreases (good, it's contributing too much to the peak)
If q_i < M: a_i increases (good, it's not contributing enough)

    fb = np.fft.rfft(b, size)

    a has length n, w has length 2n-1. Returns q of length n."""

    # This is cross-correlation of w and a
    # q = ifft(fft(w) * conj(fft(a))) ... need to be careful
    # Actually: q[i] = sum_j w[i+j] * a[j]
    # Let's compute via: conv(flip(w), a)

    ca = fft_convolve(w[::-1], a)  # length (2n-1)+n-1 = 3n-2

"""
Log-parameterized gradient optimization with adaptive step size, LP-assisted
escaping, and structured annealing for minimizing normalized peak autocorrelation
of non-negative step functions. Uses the subgradient structure of the max-convolution
objective in the log-domain for efficient peak-directed updates.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        return np.array([0.0])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_q(a, w):
    """Compute q[i] = sum_j w[i+j] * a[j] (one-sided cross-correlation).
    a has length n, w has length 2n-1. Returns q of length n.
    Uses FFT for efficiency."""
    n = len(a)
    # q[i] = conv(flip(w), a)[i + 2n-2]
    ca = fft_convolve(w[::-1], a)  # length 3n-2
    return ca[2 * n - 2: 3 * n - 2]


def compute_pressure_symmetric(a, w):
    """Compute symmetric pressure p[i] = sum_k w[k]*(a[k-i]+a[i-k]).
    Uses FFT for efficiency."""
    n = len(a)
    # p[i] = q[i] + q_rev[i] where q_rev[i] = sum_j w[i-j]*a[j]
    # q_rev[i] = conv(w, a)[i]
    ca = fft_convolve(w, a)  # length 3n-2
    q_rev = ca[:n]
    q_fwd = compute_q(a, w)
    return q_fwd + q_rev


def gradient_step(a, eta, tau, method='mean'):
    """One gradient step in log-space.
    a: normalized sequence (sum=1)
    eta: step size
    tau: softmax temperature for lag weighting
    method: 'mean' uses q - mean(q), 'peak' uses q - M
    Returns new normalized a."""
    n = len(a)
    R = fft_convolve(a, a)
    M = np.max(R)
    if M < 1e-30:
        return a.copy()

    # Compute lag weights
    if tau < 1e-10:
        # Hard max
        w = np.zeros(len(R))
        w[np.argmax(R)] = 1.0
    else:
        shifted = (R - M) / tau
        w = np.exp(shifted)
        w = w / np.sum(w)

    # Compute one-sided q
    q = compute_q(a, w)

    # Gradient of F = max_k R(k) / S^2 with S=1:
    # dF/dz_i = 2 * a_i * (q_i - M) (with S=1)
    # Use q_i - mean(q) for centering (removes the normalization coupling)
    if method == 'centered':
        grad = a * (q - np.mean(q))
    elif method == 'peak':
        grad = a * (q - M)
    else:  # 'mean'
        grad = a * (q - np.mean(q))

    # Multiplicative (log-space) update
    z_shift = -eta * grad
    a_new = a * np.exp(z_shift)
    a_new = np.clip(a_new, 1e-15, None)
    a_new = a_new / np.sum(a_new)
    return a_new


def solve_lp_direction(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns g or None."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})
    if result.success and result.x is not None:
        return result.x
    return None

def structured_sa(a_init, time_limit, seed=0, verbose=False):
    """Structured simulated annealing targeting peak lags."""
    rng = np.random.default_rng(seed)
    n = len(a_init)

    a = a / np.sum(a)

    best_score = 2.0 * n * np.max(fft_convolve(a, a))
    curr_score = best_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        R = fft_convolve(a, a)
        M = np.max(R)
        peak_lags = np.where(R >= M * 0.95)[0]

        candidate = a.copy()
        r = rng.random()

        if r < 0.4:
            # Peak-directed: reduce contributors to top peaks, boost others
            contribs = np.zeros(n)
            for k in peak_lags:
                q_k = compute_q(a, np.zeros(2*n-1) + 0.0)  # placeholder
                # Direct computation for specific lags
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        contribs[i] += a[i] * a[j]

            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))

            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_boost = min(4, len(other_idx))
                boost_idx = rng.choice(other_idx, size=n_boost, replace=False)
                candidate[boost_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_boost))

        elif r < 0.6:
            # Block oscillation
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))

        elif r < 0.75:
            # Mirror perturbation: add small amount of reversed sequence
            t_small = 0.01 + 0.03 * rng.random()
            candidate = (1 - t_small) * a + t_small * a[::-1]

        elif r < 0.85:
            # Frequency-domain perturbation
            A = np.fft.rfft(a, 2 * n)
            phase_shift = rng.normal(0, 0.02, size=len(A))
            A *= np.exp(1j * phase_shift)
            candidate = np.fft.irfft(A, 2 * n)[:n]
            candidate = np.clip(candidate, 1e-10, None)

        else:
            # Global smooth perturbation
            candidate = a * np.exp(rng.normal(0, 0.01, size=n))

        candidate = np.clip(candidate, 1e-10, None)
        candidate = candidate / np.sum(candidate)

        new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))
        iter_count += 1

        if new_score < curr_score:
            a = candidate
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
        elif rng.random() < np.exp(-(new_score - curr_score) / max(temperature, 1e-12)):
            a = candidate
            curr_score = new_score

        temperature *= cooling
        if temperature < 1e-6:
            temperature = 0.001  # reheat

    return best_a, best_score


def make_init(n, method, rng, provided_seq=None):
    """Generate initial sequence."""
    t = np.arange(n, dtype=np.float64)
    if method == 'provided' and provided_seq is not None:
        if len(provided_seq) == n:
            a = provided_seq.astype(np.float64).copy()
        else:
            x_old = np.linspace(0, 1, len(provided_seq))
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, provided_seq)
        a = np.clip(a, 0.01, None)
    elif method == 'chirp':
        alpha = 0.3 + 0.5 * rng.random()
        freq = 1.0 + rng.random()
        a = 1.0 + alpha * np.cos(np.pi * freq * t * t / n + rng.uniform(0, 2 * np.pi))
        a += 0.05 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    elif method == 'multitone':
        a = np.ones(n)
        for _ in range(8):
            k = int(rng.integers(1, max(2, n // 4)))
            amp = rng.uniform(0.05, 0.25)
            phase = rng.uniform(0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
        a += 0.03 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    elif method == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    elif method == 'block':
        # Block structure with varying amplitudes
        n_blocks = int(rng.integers(3, 10))
        block_size = n // n_blocks
        a = np.ones(n)
        for i in range(n_blocks):
            start = i * block_size
            end = start + block_size if i < n_blocks - 1 else n
            a[start:end] = rng.uniform(0.3, 2.0)
        a += 0.03 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    elif method == 'tapered':
        # Tapered (windowed) sequence
        window_type = rng.integers(0, 3)
        if window_type == 0:  # Hanning
            w = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
        elif window_type == 1:  # Hamming
            w = 0.54 + 0.46 * np.cos(2 * np.pi * t / (n - 1))
        else:  # Rectangular with noise
            w = np.ones(n)
        a = w + 0.1 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    else:
        a = np.ones(n) + 0.2 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)

    return a / np.sum(a)

        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    # Define search configurations
    # (n, init_method, time_fraction, seed_offset)
    n_prov = len(provided_seq) if provided_seq is not None else 256

    configs = [
        (n_prov, 'provided', 0.25, 100),
        (n_prov, 'chirp', 0.12, 200),
        (n_prov, 'multitone', 0.10, 300),
        (n_prov, 'block', 0.08, 400),
        (512, 'chirp', 0.10, 500),
        (512, 'provided', 0.08, 600),
        (n_prov, 'random', 0.08, 700),
        (n_prov, 'tapered', 0.07, 800),
        (384, 'multitone', 0.07, 900),
        (n_prov, 'provided', 0.05, 1000),
    ]

    available_time = deadline - time.monotonic()
    total_frac = sum(c[2] for c in configs)

    # Phase 1: Gradient optimization on multiple configs
    print("\n" + "=" * 60)
    print("Phase 1: Gradient optimization (log-space)")
    print("=" * 60)

    for n, init_name, frac, seed_off in configs:
        if time.monotonic() > deadline - 60:
            break

        time_alloc = max(15.0, frac / total_frac * available_time * 0.80)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 30)
        if time_alloc < 10:
            break

        a0 = make_init(n, init_name, rng, provided_seq)
        init_score = local_score(a0)
        print(f"  n={n}, init='{init_name}', init_score={init_score:.6f}, alloc={time_alloc:.0f}s")

        f_best, score = gradient_optimize(a0, time_alloc, seed=seed + seed_off)
        print(f"  -> score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}, init={init_name}) ***")

    # Phase 2: Continue optimizing the best with longer time
    print("\n" + "=" * 60)
    print("Phase 2: Extended gradient optimization on best")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        ext_time = min(300.0, remaining * 0.40)
        if ext_time > 20:
            print(f"  Extended gradient, n={len(best_seq)}, time={ext_time:.0f}s")
            a_ext, score_ext = gradient_optimize(best_seq, ext_time, seed=seed + 2000)
            print(f"  -> score={score_ext:.6f}")
            if score_ext < best_score:
                best_score = score_ext
                best_seq = a_ext.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 3: Structured SA polish
    print("\n" + "=" * 60)
    print("Phase 3: Structured simulated annealing")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(250.0, remaining * 0.45)
        if sa_time > 15:
            print(f"  SA polish, n={len(best_seq)}, time={sa_time:.0f}s")
            a_sa, score_sa = structured_sa(best_seq, sa_time, seed=seed + 3000)
            print(f"  -> score={score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Phase 4: Final gradient polish
    print("\n" + "=" * 60)
    print("Phase 4: Final gradient polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        final_time = max(20.0, remaining * 0.8)
        print(f"  Final gradient, n={len(best_seq)}, time={final_time:.0f}s")
        a_final, score_final = gradient_optimize(best_seq, final_time, seed=seed + 4000)
        print(f"  -> score={score_final:.6f}")
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Phase 5: Additional SA if time permits
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n  Final SA, time={remaining:.0f}s")
            a_fs, score_fs = structured_sa(best_seq, remaining * 0.9, seed=seed + 5000)
            if score_fs < best_score:
                best_score = score_fs
                best_seq = a_fs.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
