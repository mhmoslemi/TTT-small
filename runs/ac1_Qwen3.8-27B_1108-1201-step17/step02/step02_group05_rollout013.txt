```python
def smooth_max_grad(conv, beta):

    # Shift to avoid overflow
    conv_shifted = conv - conv.max()
```

```python
def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, s / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(a, beta):
    """Compute softmax-smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Softmax weights (numerically stable)
    conv_shifted = conv - conv.max()
    exp_vals = np.exp(beta * conv_shifted)
    weights = exp_vals / exp_vals.sum()
    
    # Smooth max value
    smooth_max = conv.max() + np.log(exp_vals.sum()) / beta
    
    # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j]
    # Using FFT cross-correlation
    N = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(W * np.conj(A), N)
    grad = 2.0 * cross[:n]
    
    return smooth_max, grad, conv


def smoothed_gradient_descent(a0, time_limit, lr_init=0.05, momentum=0.9, beta_init=50.0, beta_decay=0.999):
    """Smoothed gradient descent with momentum on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    n = len(a)
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0_start = time.monotonic()
    beta = beta_init
    lr = lr_init
    
    it = 0
    while time.monotonic() - t0_start < time_limit:

        batch = min(50, max(1, int((time_limit - (time.monotonic() - t0_start)) * 100)))

            a_trial = a + v

            a_new = project_simplex(a_trial, 1.0)
```

```python
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective: 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, s / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, beta):
    """Compute softmax-smoothed max of conv(a,a) and its gradient via FFT."""
    n = len(a)
    conv = fast_convolve(a)
    conv_shifted = conv - conv.max()
    exp_vals = np.exp(beta * conv_shifted)
    exp_sum = exp_vals.sum()
    weights = exp_vals / exp_sum
    smooth_val = conv.max() + np.log(exp_sum) / beta
    # grad[j] = 2 * sum_k weights[k] * a[k-j]
    N = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(W * np.conj(A), N)
    grad = 2.0 * cross[:n]
    return smooth_val, grad, conv


def smoothed_grad_desc(a0, time_limit, lr_init=0.02, momentum=0.9,
                       beta_init=100.0, beta_min=5.0, beta_decay=0.998,
                       lr_decay=0.997, perturb_every=200, perturb_mag=0.002):
    """Smoothed gradient descent with momentum and periodic perturbation."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    n = len(a)
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = obj_norm(a)
    t_start = time.monotonic()
    beta = beta_init
    lr = lr_init
    rng = np.random.default_rng(12345)
    it = 0

    while time.monotonic() - t_start < time_limit:
        # Batch processing
        batch_size = max(1, min(100, int((time_limit - (time.monotonic() - t_start)) * 50)))
        for _ in range(batch_size):
            if time.monotonic() - t_start >= time_limit:
                break

            smooth_val, grad, conv = compute_smooth_grad(a, beta)
            v = momentum * v - lr * grad
            a_trial = a + v
            a = project_simplex(a_trial, 1.0)
            it += 1

            # Check improvement periodically
            if it % 10 == 0:
                obj_cur = 2.0 * n * fast_convolve(a).max()
                if obj_cur < best_obj:
                    best_obj = obj_cur
                    best_a = a.copy()

        # Update hyperparameters
        lr = max(1e-6, lr * lr_decay)
        beta = max(beta_min, beta * beta_decay)

        # Periodic perturbation to escape local minima
        if it % perturb_every == 0:
            noise = rng.normal(0, perturb_mag, n)
            a = project_simplex(a + noise, 1.0)

    # Final check
    obj_final = obj_norm(a)
    if obj_final < best_obj:
        best_a = a.copy()
        best_obj = obj_final
    return best_a, best_obj


def solve_lp_full(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)[k] <= M for all k."""
    n = len(f)
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A_mat[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def solve_lp_topk(f, M, K=30):
    """Simplified LP with only top-K active constraints for large n."""
    n = len(f)
    conv_f = fast_convolve(f)
    top_k = np.argsort(conv_f)[::-1][:K]

    A_mat = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                A_mat[idx, j] = f[m]

    c = -np.ones(n)
    b = np.full(K, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_refine_single(f):
    """One LP refinement step with line search."""
    n = len(f)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-12:
        return f
    f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    if M < 1e-15:
        return f

    if n <= 1200:
        g = solve_lp_full(f, M)
    else:
        g = solve_lp_topk(f, M, K=50)

    if np.allclose(g, f, atol=1e-14):
        return f

    # Line search: try multiple t values
    best_t = 0.0
    best_obj = obj_norm(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_try = (1.0 - t) * f + t * g
        a_try = np.maximum(a_try, 0.0)
        st = a_try.sum()
        if st < 1e-12:
            continue
        a_try /= st
        o = obj_norm(a_try)
        if o < best_obj:
            best_obj = o
            best_t = t

    if best_t > 0:
        a_new = (1.0 - best_t) * f + best_t * g
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 1e-12:
            a_new /= sn
            return a_new
    return f


def lp_refine_iterative(f0, time_limit):
    """Multiple LP refinement steps."""
    f = f0.copy()
    t_start = time.monotonic()
    best_f = f.copy()
    best_obj = obj_norm(f)
    no_improve = 0

    while time.monotonic() - t_start < time_limit:
        f_new = lp_refine_single(f)
        obj_new = obj_norm(f_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

    return best_f, best_obj


def local_search(a0, time_limit):
    """Targeted local search: redistribute mass from peak-contributing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_obj = obj_norm(a)
    t_start = time.monotonic()
    stuck = 0
    max_stuck = 200
    rng = np.random.default_rng(777)

    while time.monotonic() - t_start < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        # Compute contribution of each index to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        # Identify dangerous and safe indices
        thr_high = 0.5 * max_c
        thr_low = 0.1 * max_c
        danger = np.where(contrib > thr_high)[0]
        safe = np.where(contrib < thr_low)[0]

        if len(danger) == 0 or len(safe) == 0:
            # Try random safe positions
            if len(safe) == 0:
                safe = np.where(contrib < 0.3 * max_c)[0]
            if len(danger) == 0:
                danger = np.argsort(contrib)[::-1][:10]
            if len(safe) == 0:
                stuck += 1
                continue

        found = False
        # Try moves from top danger to top safe
        n_danger = min(20, len(danger))
        n_safe = min(20, len(safe))
        for di in danger[:n_danger]:
            if found:
                break
            for si in safe[:n_safe]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-11:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break

        # Also try random perturbation moves
        if not found and rng.random() < 0.3:
            for _ in range(10):
                di = rng.integers(0, n)
                si = rng.integers(0, n)
                if di == si:
                    continue
                amount = a[di] * 0.1
                if amount < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[di] -= amount
                a_trial[si] += amount
                a_trial = np.maximum(a_trial, 0.0)
                ts = a_trial.sum()
                if ts < 1e-12:
                    continue
                a_trial /= ts
                o = obj_norm(a_trial)
                if o < best_obj - 1e-11:
                    best_obj = o
                    a = a_trial.copy()
                    found = True
                    break

        if not found:
            stuck += 1
        else:
            stuck = 0

    return a, best_obj


def get_parent_resized(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            x_old = np.linspace(0, 1, p.size)
            x_new = np.linspace(0, 1, n)
            pc = np.interp(x_new, x_old, p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_initializations(n, rng):
    """Generate diverse initializations for sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussian bumps
    for sf in [0.15, 0.25, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Two bumps
    b1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2)
    b2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2)
    two = b1 + b2
    two /= two.sum()
    inits['two_bump'] = two

    # Hann window
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    else:
        hann = np.ones(n)
    hann = np.maximum(hann, 0.0)
    inits['hann'] = hann / hann.sum()

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Edge concentration
    edges = np.zeros(n)
    m = max(1, n // 5)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Power law
    pw = (x + 1) ** (-0.5)
    inits['power'] = pw / pw.sum()

    # Alternating (broad)
    alt = np.ones(n) * 0.5 + 0.3 * np.cos(2 * np.pi * x / n)
    alt = np.maximum(alt, 0.0)
    inits['alt'] = alt / alt.sum()

    # Parent sequence
    parent = get_parent_resized(n)
    if parent is not None:
        inits['parent'] = parent

    # Parent with noise
    if parent is not None:
        noisy = parent + rng.normal(0, 0.01, n)
        noisy = np.maximum(noisy, 0.0)
        noisy /= noisy.sum()
        inits['parent_noisy'] = noisy

    return inits


def combined_optimize(a0, time_limit, use_lp=True):
    """Combined optimization: gradient descent + LP + local search in cycles."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    best_a = a.copy()
    best_obj = obj_norm(a)
    t_start = time.monotonic()
    n = len(a)

    # Phase A: Gradient descent
    grad_time = time_limit * 0.4
    a, obj = smoothed_grad_desc(a, time_limit=grad_time)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    # Phase B: LP refinement
    if use_lp:
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining > 5:
            lp_time = min(remaining * 0.3, 60.0)
            a_lp, obj_lp = lp_refine_iterative(best_a, time_limit=lp_time)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()

    # Phase C: Local search
    remaining = time_limit - (time.monotonic() - t_start)
    if remaining > 5:
        ls_time = min(remaining * 0.3, 45.0)
        a_ls, obj_ls = local_search(best_a, time_limit=ls_time)
        if obj_ls < best_obj:
            best_obj = obj_ls
            best_a = a_ls.copy()

    # Phase D: Another round of gradient + LP
    remaining = time_limit - (time.monotonic() - t_start)
    if remaining > 10:
        grad_time2 = min(remaining * 0.4, 30.0)
        a2, obj2 = smoothed_grad_desc(best_a, time_limit=grad_time2,
                                       lr_init=0.005, beta_init=50.0)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

        if use_lp:
            remaining2 = time_limit - (time.monotonic() - t_start)
            if remaining2 > 5:
                lp_time2 = min(remaining2 * 0.3, 30.0)
                a_lp2, obj_lp2 = lp_refine_iterative(best_a, time_limit=lp_time2)
                if obj_lp2 < best_obj:
                    best_obj = obj_lp2
                    best_a = a_lp2.copy()

    # Phase E: Final local search
    remaining = time_limit - (time.monotonic() - t_start)
    if remaining > 5:
        a_ls2, obj_ls2 = local_search(best_a, time_limit=remaining * 0.8)
        if obj_ls2 < best_obj:
            best_obj = obj_ls2
            best_a = a_ls2.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses smoothed gradient descent with momentum,
    periodic LP refinement, and targeted local search to minimize
    2n*max(conv(a,a))/(sum(a))^2. Tries multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(60.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 25.0

    best_sequence = None
    best_score = float('inf')

    print(f"Total budget: {total_budget}s, deadline in {total_budget - 25:.0f}s")

    # Phase 1: Quick scan (35% of time)
    phase1_time = (total_budget - 25.0) * 0.35
    print(f"Phase 1: Quick scan ({phase1_time:.0f}s)...")
    t_p1_start = time.monotonic()

    lengths = [512, 1024, 2048]
    results = []

    for n in lengths:
        if time.monotonic() > t_p1_start + phase1_time:
            break
        inits = make_initializations(n, rng)
        # Prioritize parent and diverse inits
        init_order = ['parent', 'parent_noisy', 'uniform', 'gauss_0.25',
                      'two_bump', 'hann', 'random']
        available_inits = [(name, inits[name]) for name in init_order if name in inits]
        # Add any remaining
        for name in inits:
            if name not in [x[0] for x in available_inits]:
                available_inits.append((name, inits[name]))

        n_inits = len(available_inits)
        time_per_init = max(5.0, (phase1_time - (time.monotonic() - t_p1_start)) / max(1, n_inits * 0.7))

        for name, a0 in available_inits:
            if time.monotonic() > t_p1_start + phase1_time:
                break
            remaining_p1 = t_p1_start + phase1_time - time.monotonic()
            if remaining_p1 < 5:
                break
            t_alloc = min(time_per_init, remaining_p1, 45.0)
            if t_alloc < 3:
                break

            a_opt, obj_opt = smoothed_grad_desc(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Phase1: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates (40% of time)
    phase2_time = (total_budget - 25.0) * 0.40
    print(f"Phase 2: Deep optimization ({phase2_time:.0f}s)...")
    t_p2_start = time.monotonic()

    results.sort(key=lambda x: x[0])
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in results:
        key = (n, name)
        if key not in seen:
            top_candidates.append((n, name, a_opt))
            seen.add(key)
        if len(top_candidates) >= 6:
            break

    # Also include best sequence
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), 'best', best_sequence.copy()))

    n_deep = len(top_candidates)
    time_per_deep = max(20.0, phase2_time / max(1, n_deep))

    for i, (n, name, a0) in enumerate(top_candidates):
        remaining_p2 = t_p2_start + phase2_time - time.monotonic()
        if remaining_p2 < 15:
            break
        t_alloc = min(time_per_deep, remaining_p2, 120.0)
        if t_alloc < 10:
            break

        a_opt, obj_opt = combined_optimize(a0, time_limit=t_alloc, use_lp=(n <= 1200))
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase2: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 2 done. Best: {best_score:.6f}")

    # Phase 3: Intensive polish on the best (remaining time)
    remaining = deadline - time.monotonic()
    print(f"Phase 3: Intensive polish ({remaining:.0f}s)...")

    if best_sequence is not None and remaining > 20:
        best_n = len(best_sequence)
        use_lp = best_n <= 1200
        polish_deadline = time.monotonic() + remaining * 0.85
        no_improve_count = 0

        while time.monotonic() < polish_deadline and no_improve_count < 6:
            iter_remaining = polish_deadline - time.monotonic()
            if iter_remaining < 5:
                break

            # Gradient descent
            gt = min(iter_remaining * 0.35, 40.0)
            a_g, obj_g = smoothed_grad_desc(best_sequence, time_limit=gt,
                                            lr_init=0.003, beta_init=30.0,
                                            beta_decay=0.999)
            score_g = float(evaluate_sequence(a_g.tolist()))
            improved = False
            if np.isfinite(score_g) and score_g < best_score - 1e-8:
                best_score = score_g
                best_sequence = a_g.copy()
                improved = True
                print(f"  Polish grad: score={best_score:.6f}")

            # LP
            if use_lp:
                iter_remaining = polish_deadline - time.monotonic()
                if iter_remaining > 10:
                    lt = min(iter_remaining * 0.25, 30.0)
                    a_l, obj_l = lp_refine_iterative(best_sequence, time_limit=lt)
                    score_l = float(evaluate_sequence(a_l.tolist()))
                    if np.isfinite(score_l) and score_l < best_score - 1e-8:
                        best_score = score_l
                        best_sequence = a_l.copy()
                        improved = True
                        print(f"  Polish LP: score={best_score:.6f}")

            # Local search
            iter_remaining = polish_deadline - time.monotonic()
            if iter_remaining > 5:
                lst = min(iter_remaining * 0.25, 25.0)
                a_ls, obj_ls = local_search(best_sequence, time_limit=lst)
                score_ls = float(evaluate_sequence(a_ls.tolist()))
                if np.isfinite(score_ls) and score_ls < best_score - 1e-8:
                    best_score = score_ls
                    best_sequence = a_ls.copy()
                    improved = True
                    print(f"  Polish LS: score={best_score:.6f}")

            if not improved:
                no_improve_count += 1
            else:
                no_improve_count = 0

    # Final validation
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Ensure valid output
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    if best_sequence.sum() < 0.01:
        best_sequence = np.ones(512)

    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score
        print(f"FINAL SCORE: {best_score:.6f}")
        print(f"Sequence length: {len(best_sequence)}")
    else:
        print("ERROR: Invalid final sequence!")
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    return [float(x) for x in best_sequence]
```
