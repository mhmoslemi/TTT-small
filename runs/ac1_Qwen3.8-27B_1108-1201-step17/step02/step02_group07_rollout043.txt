        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1

def gradient_descent_smooth(a_init, time_limit, beta=50.0):
    """Minimize smooth approximation of 2n*max(conv/(sum^2)) using FFT-based gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    a = a / S  # normalize to sum = 1
    
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    momentum = np.zeros(n)
    beta_m = 0.9  # momentum
    lr = 0.01  # learning rate
    
    size = 1
    while size < 3*n-2:
        size <<= 1
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        # Compute convolution
        c = fft_convolve(a)
        
        # Compute softmax weights
        c_shifted = c - np.max(c)
        w = np.exp(beta * c_shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w = w / w_sum
        
        # Compute gradient via FFT cross-correlation
        A = rfft(a, size)
        W = rfft(w, size)
        corr = irfft(np.conj(A) * W, size)[:n]
        g = 2.0 * corr
        
        # Normalize gradient (remove mean for scale invariance)
        g = g - np.mean(g)
        
        # Nesterov momentum
        momentum = beta_m * momentum + (1 - beta_m) * g
        a = a - lr * momentum
        a = np.clip(a, 0.0, None)
        S = np.sum(a)
        if S < 1e-10:
            a = a_init.copy()
            momentum = np.zeros(n)
            continue
        a = a / S  # re-normalize
        
        it += 1
        if it % 500 == 0:
            s = score(a)
            if s < best_score:

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FFT-based smooth subgradient (log-sum-exp), Nesterov momentum, adaptive beta schedule,
LP refinement via scipy HiGHS, spectral reinitialization, and local search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_conv_matrix(f):
    """Build matrix A where A @ b = conv(f, b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, timeout=5.0):
    """Solve: max sum(b) s.t. conv(f,b) <= M*1, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def gradient_logsumexp_fft(a, beta, size缓存=None):
    """Compute gradient of log-sum-exp(beta * conv[a,a]) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    if size_cache is None:
        size = 1
        while size < 3 * n - 2:
            size <<= 1
    else:
        size = size_cache
    c = fft_convolve(a, a)
    c_max = np.max(c)
    w = np.exp(beta * (c - c_max))
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum
    A = rfft(a, size)
    W = rfft(w, size)
    corr = irfft(np.conj(A) * W, size)[:n]
    g = 2.0 * corr
    return g, c, w, size


def smooth_gradient_descent(a_init, time_limit, beta_start=20.0, beta_end=200.0,
                            lr_start=0.05, momentum=0.9, verbose=False):
    """Nesterov-style gradient descent with log-sum-exp smooth approximation."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()

    size = 1
    while size < 3 * n - 2:
        size <<= 1

    v = np.zeros(n)  # momentum
    it = 0
    total_iters = 5000
    beta_curr = beta_start

    while time.monotonic() - t0 < time_limit - 1.0:
        progress = min(1.0, it / total_iters)
        beta_curr = beta_start + (beta_end - beta_start) * progress
        lr_curr = lr_start * (1.0 - 0.5 * progress)

        g, c, w, _ = gradient_logsumexp_fft(a, beta_curr, size)
        g = g - np.mean(g)

        v = momentum * v + (1.0 - momentum) * g
        a_new = a - lr_curr * v
        a_new = np.clip(a_new, 0.0, None)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / S_new

        a = a_new
        it += 1

        if it % 200 == 0:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    [SGD {elapsed:6.1f}s] it={it:5d} beta={beta_curr:.1f} score={s:.6f} best={best_score:.6f}")

    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    return best_a, best_score


def lp_refine(a_init, time_limit, t_step=0.08, verbose=False):
    """LP-based iterative refinement."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        return a, score_fast(a)
    a = a / S

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 3.0:
        c = fft_convolve(a, a)
        M = float(np.max(c))

        b = solve_lp_direction(a, M, timeout=4.0)
        if b is None:
            break

        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (1.0 / Sb)  # normalize to sum=1

        best_t = t_step
        best_s = np.inf
        for t_try in [t_step, 0.5 * t_step, 0.25 * t_step, 2.0 * t_step]:
            a_cand = (1.0 - t_try) * a + t_try * b
            a_cand = np.clip(a_cand, 0.0, None)
            Sc = np.sum(a_cand)
            if Sc < 1e-10:
                continue
            a_cand = a_cand / Sc
            s_cand = score_fast(a_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        a = (1.0 - best_t) * a + best_t * b
        a = np.clip(a, 0.0, None)
        S = np.sum(a)
        if S < 1e-10:
            break
        a = a / S
        it += 1

        if best_s < best_score - 1e-12:
            best_score = best_s
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [LP {elapsed:6.1f}s] it={it:3d} score={best_s:.6f} best={best_score:.6f} stall={stall}")

        if stall >= 40:
            rng = np.random.default_rng(42 + it * 31)
            a += 0.02 * rng.standard_normal(n)
            a = np.clip(a, 0.01, None)
            a = a / np.sum(a)
            stall = 0

    return best_a, best_score


def local_search(a_init, time_limit, verbose=False):
    """Perturbation-based local search."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    temp = 0.001

    while time.monotonic() - t0 < time_limit - 1.0:
        cand = a.copy()
        r = rng.random()

        if r < 0.5:
            c = fft_convolve(a, a)
            peak_k = int(np.argmax(c))
            contribs = np.zeros(n)
            lo = max(0, peak_k - (n - 1))
            hi = min(n - 1, peak_k)
            for j in range(lo, hi + 1):
                contribs[j] = a[j] * a[peak_k - j]
            top_frac = max(3, n // 10)
            threshold = np.sort(contribs)[-top_frac]
            top_mask = contribs >= threshold
            cand[top_mask] *= np.exp(rng.normal(-0.04, 0.03, size=top_mask.sum()))
            other = ~top_mask
            if other.sum() > 0:
                idxs = np.where(other)[0]
                chosen = rng.choice(idxs, size=min(8, len(idxs)), replace=False)
                cand[chosen] *= np.exp(rng.normal(0.04, 0.03, size=len(chosen)))
        elif r < 0.75:
            width = max(1, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, max(1, n - width)))
            cand[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))
        else:
            smoothness = rng.uniform(0.3, 1.5)
            kernel = np.exp(-np.arange(-5, 6) ** 2 / (2 * smoothness ** 2))
            kernel /= np.sum(kernel)
            pert = np.convolve(rng.standard_normal(n), kernel, mode='same')
            pert -= np.mean(pert)
            pert *= 0.03
            cand = cand * np.exp(pert)

        cand = np.clip(cand, 0.01, None)
        Sc = np.sum(cand)
        if Sc < 1e-10:
            continue
        cand = cand / Sc
        new_score = score_fast(cand)

        if new_score < best_score:
            best_score = new_score
            best_a = cand.copy()
            a = cand
            temp = max(temp * 0.99, 1e-6)
        else:
            cur_score = score_fast(a)
            accept_prob = np.exp(-(new_score - cur_score) / max(temp, 1e-15))
            if rng.random() < accept_prob:
                a = cand
            temp = max(temp * 0.998, 1e-6)

        if verbose and (time.monotonic() - t0) % 30 < 0.1:
            print(f"    [LS {time.monotonic()-t0:6.1f}s] score={new_score:.6f} best={best_score:.6f}")

    return best_a, best_score


def make_init(n, init_type, rng):
    """Generate various initializations."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a += 0.05 * rng.standard_normal(n)
    elif init_type == 'triangle':
        a = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
        a += 0.05 * rng.standard_normal(n)
    elif init_type == 'cosine':
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t / n)
        a += 0.03 * rng.standard_normal(n)
    elif init_type == 'bump':
        center = n / 2.0
        sigma = n / 4.0
        a = np.exp(-(t - center) ** 2 / (2 * sigma ** 2))
        a += 0.05 * rng.standard_normal(n)
    elif init_type == 'chebyshev':
        x = 2.0 * t / max(1, n - 1) - 1.0
        a = np.cos(np.arccos(np.clip(x, -1, 1)))
        a += 0.05 * rng.standard_normal(n)
    else:
        a = 0.5 + rng.random(n)
    return np.clip(a, 0.01, None)

        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    # Fallback
    fallback = [1.0] * 192
    print(f"Fallback: n=192, score={evaluate_sequence(fallback):.6f}")
    return fallback
