```python
def-project_simplex(v):
    """Project v onto the probability simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    # Binary search on tau
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
Combined optimization: multiplicative weights (smooth mirror descent) + LP-guided
alternating optimization. Uses diverse initializations (window functions, parent
sequence), multiple sequence lengths, Nesterov-style acceleration, and adaptive
parameter scheduling. Targets beating 1.511 by exploiting the smooth landscape
more effectively than pure LP approaches.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT. Result has length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_Wa(a, w, n):
    """Compute (W*a)[j] = sum_{m=0}^{n-1} w[m+j]*a[m] for j=0..n-1.
    w has length 2n-1, a has length n. Uses FFT via convolution identity:
    (W*a)[j] = (w * a_rev)[j+n-1] where a_rev = a[::-1]."""
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A, fft_size)[:conv_len]
    return conv[n - 1:2 * n - 1]


def project_simplex(v):
    """Project v onto the probability simplex {x : sum(x)=1, x>=0}. O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def make_lp_structure(n):
    """Precompute sparse structure for cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, max_lp_timeout=15):
    """LP-guided alternating optimization with backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f /= s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    if n <= 1024:
        rows, cols = make_lp_structure(n)
    else:
        rows, cols = None, None

    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        if rows is not None:
            g = solve_lp(f, M, rows, cols, n, timeout=min(max_lp_timeout, 8))
        else:
            g = None

        if g is None or g.sum() < 1e-15:
            if iter_count > 5:
                break
            # Fallback: use f itself
            g = f.copy()

        g /= g.sum()

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new /= sn
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.85
            if patience > 80:
                break

        if iter_count % 200 == 0:
            print(f"      LP iter {iter_count}, obj={best_obj:.8f}, t={t:.3f}")

    return best_f, best_obj


def multiplicative_weights_optimize(a0, time_limit, T0=50.0, eta0=0.15, T_final=5000.0):
    """Mirror descent (multiplicative weights) on the simplex for smooth peak minimization.
    Minimizes LSE_T((a*a)) which approximates max((a*a)) for large T."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a /= a.sum()

    T = T0
    eta = eta0
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    patience = 0
    iter_count = 0
    momentum = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    m_v = np.zeros(n)
    v_v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        q = fast_convolve(a)
        max_q = q.max()
        if max_q < 1e-15:
            break

        # Softmax weights (numerically stable)
        Tq = (T / max_q) * q
        Tq -= Tq.max()
        p = np.exp(Tq)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(q))] = 1.0
        else:
            p /= p_sum

        # Gradient of smooth max w.r.t. a: grad_j = 2 * sum_m p[m+j] * a[m]
        grad = 2.0 * compute_Wa(a, p, n)

        # Center gradient for simplex (subtract mean weighted by a)
        grad_centered = grad - np.dot(grad, a)

        # Adam-like update
        m_v = beta1 * m_v + (1 - beta1) * grad_centered
        v_v = beta2 * v_v + (1 - beta2) * grad_centered ** 2
        m_hat = m_v / (1 - beta1 ** iter_count)
        v_hat = v_v / (1 - beta2 ** iter_count)
        update = m_hat / (np.sqrt(v_hat) + 1e-8)

        # Multiplicative update with Adam-scaled step
        log_factor = -eta * update
        log_factor -= log_factor.max()
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a_new / s

        q = fast_convolve(a)
        obj = 2.0 * n * q.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience % 200 == 0:
                eta *= 0.9
                T = min(T * 1.08, T_final)
            if patience > 2000:
                break

        # Gradually increase T
        if iter_count % 500 == 0:
            T = min(T * 1.02, T_final)

        if iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MW iter {iter_count}, obj={best_obj:.8f}, T={T:.1f}, eta={eta:.4f}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    half = 0.5 * n

    # Uniform
    inits['uni'] = np.ones(n) / n

    # Gaussian windows with different widths
    for sf in [0.05, 0.1, 0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - half) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    # Hamming window
    ham = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    ham = np.maximum(ham, 1e-6)
    inits['hamm'] = ham / ham.sum()

    # Hanning window
    han = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    han = np.maximum(han, 1e-6)
    inits['han'] = han / han.sum()

    # Blackman window
    bla = (0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1)))
    bla = np.maximum(bla, 1e-6)
    inits['blak'] = bla / bla.sum()

    # Taylor-like window (approximate)
    Nr = 4  # number of sidelobes
    SL = -30.0  # sidelobe level in dB
    zeta = np.sqrt(np.exp(np.log(10) * SL / 10.0) / (np.pi * Nr) - 1)
    taylor = np.zeros(n)
    for i in range(n):
        val = 1.0
        for m in range(Nr):
            term = 1.0
            for k in range(m + 1):
                num = 1.0
                den = 1.0
                for j in range(k):
                    num *= (zeta ** 2 - (Nr - 1 - 2 * j) ** 2)
                    den *= (k - j) ** 2 - (Nr - 1 - 2 * j) ** 2
                if den != 0:
                    term += (num / den) * ((2 * i / n - 1) ** (m + 1))
            taylor[i] = val
        # Simpler Taylor approximation
        taylor[i] = 0.4 + 0.6 * np.cos(np.pi * (i / (n - 1) - 0.5)) ** 2
    taylor = np.maximum(taylor, 1e-6)
    inits['taylor'] = taylor / taylor.sum()

    # Triangle
    tri = np.maximum(1.0 - np.abs(x - half) / half, 0.01)
    inits['tri'] = tri / tri.sum()

    # Bimodal (two peaks)
    w1 = max(1, n // 6)
    bi = np.zeros(n)
    bi[:w1] = 1.0
    bi[-w1:] = 1.0
    inits['bi'] = bi / bi.sum()

    # Random (Dirichlet with low concentration)
    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.2)

    # Flat-top
    ft = np.ones(n)
    ft[:n // 8] = 0.5
    ft[-n // 8:] = 0.5
    inits['ftop'] = ft / ft.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry: combined MW + LP optimization with diverse inits and multiple lengths."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    def try_candidate(f, source=""):
        nonlocal best_seq, best_score
        if f is None:
            return
        f = np.maximum(np.asarray(f, dtype=np.float64), 0.0)
        if f.size == 0 or f.sum() < 1e-10:
            return
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"  NEW BEST: {source} n={f.size} score={score:.6f}")

    # ===== Phase 1: Multiplicative weights scan (500s) =====
    print("Phase 1: MW scan across lengths and initializations")
    phase1_deadline = time.monotonic() + min(500.0, (deadline - time.monotonic()) * 0.35)

    lengths = [256, 512, 1024, 2048]
    phase1_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        inits = gen_inits(n, rng)

        # Add parent if available
        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp / pp.sum()

        # Pick top initializations to try (limit to 8 per length)
        init_names = list(inits.keys())
        # Prioritize known-good inits
        priority = ['parent', 'hamm', 'han', 'g15', 'g25', 'blak', 'taylor', 'uni',
                    'tri', 'bi', 'g10', 'g5', 'g40', 'rnd', 'ftop']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)
        init_names = init_names[:8]

        n_inits = len(init_names)
        time_per_init = max(10.0, min(60.0, (phase1_deadline - time.monotonic()) / max(1, n_inits * len(lengths))))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 20:
                break
            ta = min(time_per_init, phase1_deadline - time.monotonic() - 15)
            if ta < 5:
                break

            a0 = inits[name]
            f_opt, obj_opt = multiplicative_weights_optimize(a0, time_limit=ta, T0=30.0, eta0=0.2)
            try_candidate(f_opt, f"P1 MW n={n} {name}")
            phase1_results.append((obj_opt, n, name, f_opt))

    print(f"Phase 1 done. Best so far: {best_score:.6f}")

    # Sort phase 1 results
    phase1_results.sort(key=lambda x: x[0])

    # ===== Phase 2: LP deep optimization (remaining - 200s) =====
    remaining = deadline - time.monotonic() - 200.0
    if remaining > 100 and phase1_results:
        print(f"Phase 2: LP deep optimization, {remaining:.0f}s available")

        # Also add parent as a candidate for LP
        lp_candidates = [(r[0], r[1], r[2], r[3]) for r in phase1_results[:6]]
        if parent is not None:
            pn = min(parent.size, 1024)
            pp = np.maximum(parent[:pn], 0.0) / np.maximum(parent[:pn].sum(), 1e-10)
            lp_candidates.append((2.0, pn, 'parent_lp', pp))

        # Deduplicate by length
        seen_lengths = set()
        unique_candidates = []
        for item in lp_candidates:
            if item[1] not in seen_lengths:
                seen_lengths.add(item[1])
                unique_candidates.append(item)
        # Allow up to 5 candidates
        unique_candidates = unique_candidates[:5]

        t_each = remaining / len(unique_candidates)
        for idx, (sc, n, name, f) in enumerate(unique_candidates):
            if time.monotonic() > deadline - 210:
                break
            ta = min(t_each, deadline - time.monotonic() - 200)
            if ta < 15:
                break

            print(f"    LP deep [{idx}] n={n} {name} t={ta:.0f}s")
            f2, obj2 = lp_optimize(f, time_limit=ta * 0.7, max_lp_timeout=min(15, ta // 10))
            try_candidate(f2, f"P2 LP n={n} {name}")

            # MW polish after LP
            if time.monotonic() < deadline - 205:
                tr = min(ta * 0.3, deadline - time.monotonic() - 200)
                if tr > 5:
                    f3, obj3 = multiplicative_weights_optimize(f2, time_limit=tr, T0=100.0, eta0=0.1)
                    try_candidate(f3, f"P2 MW-polish n={n} {name}")

    # ===== Phase 3: Final intensive polish (200s) =====
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"Phase 3: Final polish on best (n={len(best_seq)}), {remaining:.0f}s")

            # Multiple MW passes with different parameters
            for T0, eta0, label in [(50, 0.15, 'mw1'), (200, 0.08, 'mw2'),
                                     (500, 0.04, 'mw3'), (1000, 0.02, 'mw4')]:
                if time.monotonic() > deadline - 10:
                    break
                tr = min(40.0, (deadline - time.monotonic() - 10) / 3)
                if tr < 3:
                    break
                f5, _ = multiplicative_weights_optimize(best_seq, time_limit=tr, T0=T0, eta0=eta0)
                try_candidate(f5, f"P3 {label}")

            # LP polish if n <= 1024
            if len(best_seq) <= 1024 and time.monotonic() < deadline - 15:
                tr = min(50.0, deadline - time.monotonic() - 10)
                if tr > 10:
                    f6, _ = lp_optimize(best_seq, time_limit=tr, max_lp_timeout=10)
                    try_candidate(f6, "P3 LP")

            # Final MW with high T
            if time.monotonic() < deadline - 5:
                tr = deadline - time.monotonic() - 5
                f7, _ = multiplicative_weights_optimize(best_seq, time_limit=tr, T0=2000.0, eta0=0.02, T_final=10000.0)
                try_candidate(f7, "P3 final-MW")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
```
