def compute_obj(a):
    """Compute the objective."""
    n = len(a)
    c = np.convolve(a, a)

    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

    """Normalize to target sum."""

    if S < 1e-12:

    """Build the constraint matrix for the LP."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # f[j] contributes to rows j through j+n-1
        A[j:j+n, j] = f
    return A

def solve_lp(f, M, time_limit=20):
    """
    Solve: max sum(g) s.t. A@g <= M*ones, g >= 0.
    Returns optimal g vector or None.
    """
    n = len(f)
    if n > 1024:
        return None
    
    c_obj = -np.ones(n)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses aggressive LP-based alternating direction with progressive constraint tightening,
adaptive blending via line search, FFT-accelerated autocorrelation, and multi-start
strategy with structured perturbations to escape local minima.
Key improvement: progressive LP tightening (geometric M reduction) combined with
proper line search along LP-defined directions, plus sub-sampling for large n.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def build_lp_matrix(f):
    """Build the (2n-1) x n convolution constraint matrix."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, time_limit=20, max_n=640):
    """
    Solve LP: max sum(g) s.t. convolve(f,g) <= M (elementwise), g >= 0.
    Returns optimal g vector (non-negative) or None.
    """
    n = len(f)
    if n > max_n:
        return None

    c_obj = -np.ones(n)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-8:
                return g
    except Exception:
        pass
    return None


def solve_lp_relaxed(f, M, relax_factor, time_limit=20, max_n=640):
    """Solve LP with relaxed constraint: conv(f,g) <= M * relax_factor."""
    return solve_lp(f, M * relax_factor, time_limit=time_limit, max_n=max_n)


def line_search_blend(a, g, S_t, max_iters=30):
    """
    Line search: find best t in [0, 1] such that normalize((1-t)*a + t*g) minimizes obj.
    Uses golden section search.
    """
    g_norm = g * (S_t / g.sum())

    def obj_at_t(t):
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-6:
            return float('inf')
        a_new = a_new * (S_t / S)
        return compute_obj(a_new)

    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = obj_at_t(c_pt)
    fd = obj_at_t(d_pt)

    for _ in range(max_iters):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = obj_at_t(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj_at_t(d_pt)

    t_best = (lo + hi) / 2.0
    a_best = (1.0 - t_best) * a + t_best * g_norm
    a_best = np.maximum(a_best, 0.0)
    S = a_best.sum()
    if S < 1e-6:
        return a.copy(), 0.0
    a_best = a_best * (S_t / S)
    return a_best, t_best


def gradient_descent_polish(a, deadline, max_steps=2000):
    """Gradient descent on peak autocorrelation with momentum."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    momentum = np.zeros(n)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        k_star = int(np.argmax(c))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        grad = grad - grad.mean()
        lr = 0.1 * (0.999 ** step)
        lr = max(lr, 0.01)

        momentum = 0.9 * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-6:
            continue
        a_new = a_new * (S_t / S)

        s = compute_obj(a_new)
        if s < best_score - 1e-10:
            best = a_new.copy()
            best_score = s
            a = a_new
        else:
            a = a_new  # still accept for exploration

    return best, best_score
