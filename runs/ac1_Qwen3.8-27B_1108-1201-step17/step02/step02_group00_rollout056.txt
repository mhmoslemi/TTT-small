def project_simplex(v, target_sum=1.0):

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed max (log-sum-exp) 
approximation with projected gradient descent and adaptive temperature.
Uses Nesterov-style momentum, backtracking line search, and multi-phase

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([target_sum])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_b(a):
    return fast_convolve(a)

def smooth_max_and_grad(b, tau):
    """Compute log-sum-exp(b/tau)*tau and gradient weights."""
    scaled = b / tau
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    obj = b.max() + tau * log_sum  # = tau * log(sum(exp(b/tau)))
    w = exp_scaled / exp_scaled.sum()
    return obj, w

def gradient_from_weights(w, a):
    """Compute gradient of smoothed max given softmax weights w and sequence a."""
    n = len(a)
    corr = np.correlate(w, a, mode='full')
    return 2.0 * corr[:n]

    Optimize using smoothed max with projected gradient.
    tau is gradually decreased from tau_init to tau_final.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = compute_b(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    # Create tau schedule (geometric)
    tau_schedule = np.geomspace(tau_init, tau_final, n_tau_steps)
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break

        # L ≈ 2 * max(w) * 2 * ||a||_2^2 (rough estimate)
        # For safety, start with a reasonable lr and use backtracking
        lr = 1.0 / (4.0 * np.sum(a**2) + 1e-8)
        lr = min(lr, 50.0)  # cap
        
        momentum = 0.9
        v = np.zeros(n)

        max_iters = max(200, int(500000 / n))
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit - 5:
                break
            
            b = compute_b(a)
            _, w = smooth_max_and_grad(b, tau)
            grad = gradient_from_weights(w, a)
            
            # Momentum update
            v = momentum * v - lr * grad
            a_new = project_simplex(a + v)
            
            # Check improvement
            b_new = compute_b(a_new)
            obj_new = 2.0 * n * b_new.max()
            
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
            
            a = a_new
            
            # Periodic lr adjustment
            if it % 100 == 0 and it > 0:
                lr *= 0.95
                momentum = min(0.95, momentum + 0.002)

        # (already handled by the loop above)
    
    return best_a, best_obj / (2.0 * n)  # Return max_b (normalized)

def optimize_nesterov_smoothed(a_init, time_limit, tau_init=0.005, tau_final=0.00005):

    Nesterov accelerated projected gradient on smoothed max.

    """

        a = a / s

    best_obj = 2.0 * n * b.max()

    for tau in taus:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov parameters
        t_curr = 1.0
        a_prev = a.copy()
        a_curr = a.copy()

            if time.monotonic() - t0 > time_limit - 2:
                break

            t_prev = t_curr
            t_curr = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0

            a_y = np.maximum(a_y, 0)

            step = min(step, 10.0)

                if s_new < 1e-10:

    log_sum = np.log(exp_scaled.sum())
    obj = b_max + tau * log_sum  # = tau * log(sum(exp(b/tau)))
    
    # Weights for gradient
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient: dF/da[j] = 2 * sum_k w[k] * a[k-j] for valid k,j
    # = 2 * (w cross-correlate a)[j]
    corr = np.correlate(w, a, mode='full')
    grad = 2.0 * corr[:n]
    
    return obj, grad

def optimize_smoothed_max(a_init, time_limit, tau_list=None):
    """
    Optimize using smoothed max with Nesterov accelerated projected gradient.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if tau_list is None:
        tau_list = np.geomspace(0.005, 0.00005, 12)
    
    b = fast_convolve(a)
    best_obj = b.max()  # We track max_b (since 2n is constant for fixed n)
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov state
        t_val = 1.0
        x = a.copy()  # current point
        x_prev = a.copy()  # previous point
        
        max_iters = max(300, int(200000 / n))
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            t_prev = t_val
            t_val = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            
            # Nesterov extrapolation
            y = x + (t_prev - 1.0) / t_val * (x - x_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            else:
                y = x.copy()
            
            # Gradient at y
            _, grad = smooth_objective_and_grad(y, tau)
            
            # Step size: 2 / (t_val * L) where L is Lipschitz constant
            # L = 2 * max_j ||db[k]/da[j]|| * max_k w[k]
            # ||db[k]/da||_2 <= 2 * ||a||_2
            # So L <= 4 * ||a||_2^2 * max(w)
            # For safety, use a conservative estimate
            L_est = 4.0 * np.sum(y ** 2) + 1e-8
            step = 2.0 / (t_val * L_est)
            step = min(step, 5.0)
            
            # Gradient step + projection
            x_new = project_simplex(y - step * grad)
            
            # Track best using TRUE max (not smoothed)
            b_new = fast_convolve(x_new)
            obj_new = b_new.max()
            
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = x_new.copy()
            
            x_prev = x
            x = x_new
        
        a = x
    
    return best_a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed max (log-sum-exp) to obtain a differentiable surrogate
2. Nesterov accelerated projected gradient (FISTA) with adaptive step size
3. Multi-scale temperature schedule (coarse-to-fine)
4. Coordinate descent polish for final refinement

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([target_sum])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_grad(a, tau):
    """
    Compute smoothed max of autocorrelation and its gradient.
    F(a) = tau * log(sum_k exp(b[k]/tau))
    grad = 2 * (w cross-correlate a)[:n] where w = softmax(b/tau)
    """
    b = fast_convolve(a)
    b_max = b.max()
    scaled = (b - b_max) / tau
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    obj = b_max + tau * log_sum
    w = exp_s / exp_s.sum()
    corr = np.correlate(w, a, mode='full')
    grad = 2.0 * corr[:len(a)]
    return obj, grad, b


def fista_smoothed(a_init, time_limit, tau_schedule, step_scale=2.0):

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        x = a.copy()
        x_prev = a.copy()
        t_val = 1.0
        
        max_iters = max(200, int(300000 / n))
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            t_prev = t_val
            t_val = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0

            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            else:
                y = x.copy()
            
            # Gradient
            _, grad, _ = smooth_max_grad(y, tau)

                best_a = x_new.copy()

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    r = np.arange(1, n + 1, dtype=float)
    inits['ramp'] = r / r.sum()
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    for sf in [0.1, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g / g.sum()
    
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random2'] = rng.dirichlet(np.ones(n) * 2.0)
    
    if parent is not None:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        pp = np.interp(np.linspace(0, 1, n),
                       np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent_interp'] = pp / ps
    
    return inits

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([target_sum])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
