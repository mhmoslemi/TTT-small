def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def get_score(conv, n, S):
    return 2.0 * n * np.max(conv) / (S * S)

```python
"""
Coordinate descent with incremental convolution updates for minimizing
the peak autocorrelation of non-negative step functions.

Key ideas:
- For each coordinate, the 1D subproblem is solved via numerical search
- Convolution is updated incrementally in O(n) per coordinate
- Multiple starting points and perturbation restarts for global search
"""

import numpy as np
import time


def compute_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def get_score(conv, n, S):
    """Compute 2*n*max(conv)/S^2."""
    return 2.0 * n * np.max(conv) / (S * S)


def cd_pass(a, conv, S, n, n_eval=35, rng=None):
    """
    One pass of coordinate descent.
    Returns (a, conv, S, improved).
    """
    improved = False
    order = np.arange(n)
    if rng is not None:
        rng.shuffle(order)
    
    for i in order:
        ai = a[i]
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        
        # Precompute for this coordinate
        offsets = np.arange(k_start, k_end + 1) - i
        base_vals = conv[k_start:k_end + 1]
        slopes = 2.0 * a[offsets]
        has_center = (k_start <= 2 * i <= k_end)
        center_idx = (2 * i - k_start) if has_center else -1
        
        # Unaffected peak
        if k_start > 0 and k_end < 2 * n - 2:
            unaff = max(np.max(conv[:k_start]), np.max(conv[k_end + 1:]))
        elif k_start > 0:
            unaff = np.max(conv[:k_start])
        elif k_end < 2 * n - 2:
            unaff = np.max(conv[k_end + 1:])
        else:
            unaff = 0.0
        
        cur_score = get_score(conv, n, S)
        
        if ai < 1e-10:
            # Try small positive values
            target = S / n
            candidates = [target * 0.1, target * 0.5, target]
            best_d = 0.0
            best_sc = cur_score
            for d in candidates:
                new_S = S + d
                new_vals = base_vals + 2.0 * d * a[offsets]
                if has_center:
                    new_vals[center_idx] = base_vals[center_idx] + d * d
                new_max = max(np.max(new_vals), unaff)
                sc = 2.0 * n * new_max / (new_S * new_S)
                if sc < best_sc:
                    best_sc = sc
                    best_d = d
            if best_d > 0:
                conv[k_start:k_end + 1] = base_vals + 2.0 * best_d * a[offsets]
                if has_center:
                    conv[2 * i] = base_vals[center_idx] + best_d * best_d
                a[i] = best_d
                S += best_d
                improved = True
            continue
        
        # Normal case: search for best delta
        d_min = -ai * 0.999
        d_max = ai * 0.5
        
        # Coarse grid
        deltas = np.linspace(d_min, d_max, n_eval)
        best_d = 0.0
        best_sc = cur_score
        
        for d in deltas:
            new_S = S + d
            if new_S <= 0.01:
                continue
            new_vals = base_vals + d * slopes
            if has_center:
                new_vals[center_idx] = base_vals[center_idx] + 2 * d * ai + d * d
            new_max = max(np.max(new_vals), unaff)
            sc = 2.0 * n * new_max / (new_S * new_S)
            if sc < best_sc:
                best_sc = sc
                best_d = d
        
        # Fine search
        if best_d != 0.0:
            step = (d_max - d_min) / n_eval
            lo = max(d_min, best_d - 2 * step)
            hi = min(d_max, best_d + 2 * step)
            fine_deltas = np.linspace(lo, hi, 20)
            for d in fine_deltas:
                new_S = S + d
                if new_S <= 0.01:
                    continue
                new_vals = base_vals + d * slopes
                if has_center:
                    new_vals[center_idx] = base_vals[center_idx] + 2 * d * ai + d * d
                new_max = max(np.max(new_vals), unaff)
                sc = 2.0 * n * new_max / (new_S * new_S)
                if sc < best_sc:
                    best_sc = sc
                    best_d = d
        
        # Analytical candidates (top few)
        n_analytic = min(8, k_end - k_start + 1)
        if rng is not None:
            sample_k = rng.choice(k_end - k_start + 1, size=n_analytic, replace=False)
        else:
            sample_k = np.linspace(0, k_end - k_start, n_analytic, dtype=int)
        
        for idx_k in sample_k:
            k = k_start + int(idx_k)
            if k == 2 * i:
                # Quadratic case
                # Minimize (conv[2i] + 2*d*ai + d^2) / (S+d)^2
                # Let u = S+d: minimize (conv[2i] + 2(u-S)*ai + (u-S)^2) / u^2
                # = (u^2 + u*(2ai-2S) + conv[2i]-2S*ai+S^2) / u^2
                # = 1 + (2ai-2S)/u + (conv[2i]-2S*ai+S^2)/u^2
                # Derivative: -(2ai-2S)/u^2 - 2(conv[2i]-2S*ai+S^2)/u^3 = 0
                # => -(2ai-2S)*u = 2(conv[2i]-2S*ai+S^2)
                # => u = -2(conv[2i]-2S*ai+S^2) / (2ai-2S) = (conv[2i]-2S*ai+S^2) / (S-ai)
                denom = S - ai
                if abs(denom) < 1e-10:
                    continue
                C = conv[2 * i] - 2 * S * ai + S * S
                u_opt = C / denom
                d_cand = u_opt - S
            else:
                ak_i = a[k - i]
                if ak_i < 1e-10:
                    continue
                d_cand = S - conv[k] / ak_i
            
            if d_cand < d_min or d_cand > d_max or abs(d_cand) < 1e-10:
                continue
            new_S = S + d_cand
            if new_S <= 0.01:
                continue
            new_vals = base_vals + d_cand * slopes
            if has_center:
                new_vals[center_idx] = base_vals[center_idx] + 2 * d_cand * ai + d_cand * d_cand
            new_max = max(np.max(new_vals), unaff)
            sc = 2.0 * n * new_max / (new_S * new_S)
            if sc < best_sc:
                best_sc = sc
                best_d = d_cand
        
        # Apply
        if best_d != 0.0 and best_sc < cur_score:
            conv[k_start:k_end + 1] = base_vals + best_d * slopes
            if has_center:
                conv[2 * i] = base_vals[center_idx] + 2 * best_d * ai + best_d * best_d
            a[i] += best_d
            if a[i] < 0:
                a[i] = 0
                S -= (a[i] - best_d)  # correction
                # Actually this shouldn't happen if d_min = -0.999*ai
            S += best_d
            improved = True
    
    return a, conv, S, improved


def coordinate_descent(a_init, time_limit, rng, verbose=False):
    """Run coordinate descent until time limit or convergence."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0)
    S = np.sum(a)
    if S < 0.01:
        a = np.full(n, 0.1)
        S = np.sum(a)
    
    conv = compute_conv(a)
    conv = np.clip(conv, 0.0, None)  # Remove tiny negatives from FFT
    current_score = get_score(conv, n, S)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    pass_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        a, conv, S, improved = cd_pass(a, conv, S, n, n_eval=35, rng=rng)
        current_score = get_score(conv, n, S)
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodically recompute convolution to avoid numerical drift
        if pass_count % 100 == 0:
            conv = compute_conv(a)
            conv = np.clip(conv, 0.0, None)
            current_score = get_score(conv, n, S)
        
        if verbose and pass_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Pass {pass_count}: score={current_score:.8f}, best={best_score:.8f}, t={elapsed:.1f}s")
        
        if no_improve > 15:
            break
    
    return best_a, best_score


def resample_to_length(seq, new_n):
    """Resample a sequence to a new length via linear interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)
