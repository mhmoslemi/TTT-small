"""
Soft-max gradient descent with spectral initialization for minimizing
normalized peak autocorrelation of non-negative sequences.
Uses annealed soft-max surrogate, Adam optimizer, alternating projection
initialization, LP refinement, and targeted simulated annealing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_A(f):
    """Build constraint matrix A for the LP."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_constraint_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 20})
    if result.success:
        return result.x
    return None


def spectral_init(n, rng, n_iters=200):
    """Alternating projection initialization for flat power spectrum."""
    a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    a = a / np.sum(a)

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for _ in range(n_iters):
        # FFT
        A = np.fft.rfft(a, N)
        # Set flat magnitude (preserve energy)
        target_mag = np.sqrt(np.sum(a ** 2) / N)
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        # Inverse FFT
        a_new = np.fft.irfft(A_new, N)[:n]
        # Project to non-negative
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new

    return a


def softmax_gradient_descent(a_init, time_limit, tau_start=None, tau_end=None,
                             lr=0.5, top_k=15, verbose=False):
    """Annealed soft-max gradient descent with Adam."""
    n = len(a_init)
    a = np.clip(a_init, 1e-6, None).astype(np.float64)
    a = a / np.sum(a)  # normalize sum to 1

    if tau_start is None:
        tau_start = 0.05
    if tau_end is None:
        tau_end = 0.003

    # Parameterize: a_i = exp(theta_i) / sum(exp(theta_j))
    theta = np.log(a + 1e-15)
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step_count = 0

    best_score = local_score(a_init)
    best_a = a_init.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Cosine annealing of tau
        tau = tau_end + 0.5 * (tau_start - tau_end) * (1 + np.cos(np.pi * elapsed_frac))

        # Compute a from theta
        theta_shifted = theta - np.max(theta)
        a = np.exp(theta_shifted)
        a = a / np.sum(a)

        # Compute convolution
        c = fft_convolve(a, a)

        # Soft-max weights
        c_scaled = c / tau
        c_scaled = c_scaled - np.max(c_scaled)
        w = np.exp(c_scaled)
        w_sum = np.sum(w)
        w = w / w_sum

        # Gradient: grad_a[j] = 2 * sum_m a[m] * w[m+j]
        # Use top-k lags for efficiency
        top_m = min(top_k, 2 * n - 1)
        top_idx = np.argsort(w)[-top_m:]

        grad_a = np.zeros(n)
        for k in top_idx:
            wk = w[k]
            if wk < 1e-10:
                continue
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            indices = np.arange(j_min, j_max + 1)
            values = a[k - indices]
            grad_a[indices] += 2.0 * wk * values

        # Chain rule: d/dtheta_m = a_m * (grad_a[m] - dot(grad_a, a))
        ga_dot_a = np.dot(grad_a, a)
        grad_theta = a * (grad_a - ga_dot_a)

        # Adam update
        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * grad_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (grad_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        lr_effective = lr / (1.0 + 0.001 * step_count)
        theta = theta - lr_effective * m1_hat / (np.sqrt(m2_hat) + eps)

        # Evaluate
        theta_shifted2 = theta - np.max(theta)
        a_new = np.exp(theta_shifted2)
        a_new = a_new / np.sum(a_new)
        score = local_score(a_new)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()

        if verbose and step_count % 500 == 0:
            print(f"      SM-GD step={step_count}, tau={tau:.5f}, "
                  f"score={score:.6f}, best={best_score:.6f}")

    return best_a, best_score

def make_interpolated_init(provided_seq, n, rng):
    """Interpolate provided sequence to target length."""
    if len(provided_seq) == n:
        return provided_seq.astype(np.float64).copy()
    x_old = np.linspace(0, 1, len(provided_seq))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, provided_seq)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chirp_init(n, rng):

    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

    for _ in range(8):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.03, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_sinusoidal_init(n, rng):
    """Sinusoidal modulation initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_harmonics = int(rng.integers(2, 6))
    for _ in range(n_harmonics):
        freq = rng.uniform(0.5, n / 4)
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.sin(2 * np.pi * freq * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Spectral initialization + quick soft-max GD on multiple configs.
    Phase 2: Extended soft-max GD on best candidates.
    Phase 3: LP refinement on top candidate.
    Phase 4: Simulated annealing polish.
    Phase 5: Final LP if time permits.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    # === Phase 1: Multiple initializations with quick optimization ===
    print("\n" + "=" * 60)
    print("Phase 1: Multi-config soft-max gradient descent")
    print("=" * 60)

    n_prov = len(provided_seq) if provided_seq is not None else 256
    time_per_config_1 = 80  # seconds per config

    configs = []
    # Provided sequence at native length
    if provided_seq is not None:
        configs.append((n_prov, 'provided'))
    # Different lengths and inits
    configs.append((256, 'spectral'))
    configs.append((256, 'chirp'))
    configs.append((384, 'spectral'))
    configs.append((384, 'multitone'))
    configs.append((512, 'spectral'))
    configs.append((512, 'chirp'))
    configs.append((n_prov, 'multitone'))
    configs.append((n_prov, 'sinusoidal'))

    # Adapt number of configs based on time
    max_configs = max(3, int((deadline - time.monotonic() - 400) / time_per_config_1))
    configs = configs[:min(len(configs), max_configs)]

    candidates = []  # list of (score, sequence)

    for idx, (n, init_name) in enumerate(configs):
        if time.monotonic() > deadline - 350:
            break

        time_alloc = min(time_per_config_1, deadline - time.monotonic() - 300)
        if time_alloc < 20:
            break

        # Generate initialization
        if init_name == 'provided' and provided_seq is not None:
            a0 = make_interpolated_init(provided_seq, n, rng)
        elif init_name == 'spectral':
            a0 = spectral_init(n, rng, n_iters=150)
        elif init_name == 'chirp':
            a0 = make_chirp_init(n, rng)
        elif init_name == 'multitone':
            a0 = make_multitone_init(n, rng)
        else:
            a0 = make_sinusoidal_init(n, rng)

        s0 = local_score(a0)
        print(f"  [{idx+1}/{len(configs)}] n={n}, init='{init_name}', initial score={s0:.6f}, time={time_alloc:.0f}s")

        # Run soft-max GD
        a_best, score = softmax_gradient_descent(a0, time_alloc, tau_start=0.04, tau_end=0.003, lr=0.4)
        print(f"         -> score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = a_best.copy()
            print(f"         *** New best: {best_score:.6f} (n={n}) ***")

        candidates.append((score, a_best.copy()))

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"\n  Top candidates: {[(f'{s:.5f}', len(a)) for s, a in candidates[:5]]}")

    # === Phase 2: Extended optimization on top candidates ===
    print("\n" + "=" * 60)
    print("Phase 2: Extended soft-max GD + LP on top candidates")
    print("=" * 60)

    time_for_phase2 = min(500.0, deadline - time.monotonic() - 200)
    n_top = min(3, len(candidates))
    if n_top > 0 and time_for_phase2 > 50:
        time_per_cand2 = time_for_phase2 / n_top

        for idx in range(n_top):
            if time.monotonic() > deadline - 180:
                break

            cand_score, cand_seq = candidates[idx]
            time_alloc = min(time_per_cand2, deadline - time.monotonic() - 150)
            if time_alloc < 30:
                break

            n = len(cand_seq)
            print(f"  Candidate {idx+1} (n={n}, score={cand_score:.6f}), time={time_alloc:.0f}s")

            # Extended soft-max GD with finer tau
            sm_time = time_alloc * 0.5
            a_sm, score_sm = softmax_gradient_descent(cand_seq, sm_time,
                                                       tau_start=0.003, tau_end=0.001, lr=0.2)
            print(f"    SM-GD: {score_sm:.6f}")

            current_best = cand_seq if cand_score < score_sm else a_sm
            current_score = min(cand_score, score_sm)

            # LP refinement
            lp_time = time_alloc * 0.5
            a_lp, score_lp = lp_refine(current_best, lp_time, t_init=0.25)
            print(f"    LP:    {score_lp:.6f}")

            if score_lp < current_score:
                current_best = a_lp
                current_score = score_lp

            if current_score < best_score:
                best_score = current_score
                best_seq = current_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 3: Simulated annealing on best ===
    print("\n" + "=" * 60)
    print("Phase 3: Simulated annealing")
    print("=" * 60)

    if best_seq is not None:
        sa_time = min(300.0, deadline - time.monotonic() - 100)
        if sa_time > 30:
            print(f"  SA on n={len(best_seq)}, score={best_score:.6f}, time={sa_time:.0f}s")
            a_sa, score_sa = sa_peak_reduction(best_seq, sa_time, rng)
            print(f"  SA result: {score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Final LP polish ===
    print("\n" + "=" * 60)
    print("Phase 4: Final LP polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 30:
        final_time = min(120.0, deadline - time.monotonic() - 10)
        if final_time > 20:
            print(f"  Final LP, n={len(best_seq)}, time={final_time:.0f}s")
            a_final, score_final = lp_refine(best_seq, final_time, t_init=0.15)
            print(f"  Final LP: {score_final:.6f}")
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 5: Quick SM-GD polish ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        polish_time = min(60.0, deadline - time.monotonic() - 5)
        if polish_time > 10:
            a_polish, score_polish = softmax_gradient_descent(
                best_seq, polish_time, tau_start=0.001, tau_end=0.0005, lr=0.1)
            if score_polish < best_score:
                best_score = score_polish
                best_seq = a_polish.copy()
                print(f"  Polish: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
