Gradient-based optimization for minimizing peak linear convolution of non-negative sequences.
Uses log-sum-exp smoothed gradient descent with projection onto the probability simplex,
Nesterov momentum, adaptive temperature, and coordinate descent polishing.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute the evaluation score (fast local version)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_conv(a):
    """Compute linear convolution of a with itself."""
    return fft_convolve(a, a)

def logsumexp_gradient(c, T):
    """Compute softmax weights for log-sum-exp smoothing."""
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / np.sum(exp_c)
    return w

def convolve_with_weights(w, a):

    # = sum_j w[j+i] * a[j] where j = k-i, 0<=j<len(a)

def project_simplex(v):

    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    m = len(b)

"""
Optimizes non-negative step function to minimize peak linear convolution relative to sum squared.
Uses smoothed subgradient descent (log-sum-exp) with simplex projection,
coordinate descent polishing, and multi-resolution search.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence (fast local version)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s ** 2))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. (2011)
    if np.sum(v) <= 1 and np.all(v >= 0):
        # Already in simplex (after normalization)
        s = np.sum(v)
        if s > 0:
            return v / s
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
