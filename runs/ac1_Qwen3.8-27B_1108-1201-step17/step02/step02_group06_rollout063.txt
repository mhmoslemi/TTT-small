"""
Gradient-based optimization with adaptive softmax temperature for minimizing
normalized peak self-convolution of non-negative sequences. Uses Adam in
log-space with softmax parameterization, multiple initializations, and
simulated annealing for final polishing.
"""

import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score_and_conv(a):
    """Compute score and convolution for a non-negative sequence."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf, None
    c = fft_convolve(a, a)
    g = np.max(c)
    score = 2.0 * n * g / (s * s)
    return score, c

def gradient_descent(a0, time_limit, tau_start=1.0, tau_end=0.01, 
                     base_lr=0.5, seed=42, verbose=False):
    """
    Adam gradient descent in log-space with softmax parameterization.
    Optimizes: minimize 2*n*max(conv(a,a)) / sum(a)^2
    Using: a = softmax(theta) (auto-normalizes to sum=1)
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a0_normed = a0 / np.sum(a0)
    theta = np.log(a0_normed + 1e-15)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    
    best_score = compute_score(a0)
    best_a = a0.copy()
    t0 = time.monotonic()
    
    # Adaptive temperature
    tau = tau_start
    
    while time.monotonic() - t0 < time_limit:
        a = a0_normed.copy()  # Will be updated from theta
        # Compute a from theta (softmax)
        theta_shifted = theta - np.max(theta)
        exp_theta = np.exp(theta_shifted)
        a = exp_theta / np.sum(exp_theta)
        
        # Compute convolution
        c = fft_convolve(a, a)
        g = np.max(c)
        score = 2.0 * n * g  # Since sum(a) = 1
        
        # Update best
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Compute soft-max weights
        c_shifted = c / tau
        c_shifted -= np.max(c_shifted)
        exp_c = np.exp(c_shifted)
        w = exp_c / np.sum(exp_c)
        
        # Compute gradient: h = 2 * correlate(w, a)[:n]
        h = 2.0 * np.correlate(w, a, mode='full')[:n]
        
        # grad_theta = 4*n*a*(h - dot(h,a))
        h_mean = np.dot(h, a)
        grad_theta = 4.0 * n * a * (h - h_mean)
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * (grad_theta ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        lr = base_lr / np.sqrt(1 + 0.01 * step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Anneal temperature
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        if verbose and step % 500 == 0:
            print(f"  GD step={step}, score={compute_score(a):.6f}, "
                  f"best={best_score:.6f}, tau={tau:.4f}")
    
    return best_a, best_score

"""
Gradient-based optimization with adaptive softmax temperature for minimizing
normalized peak self-convolution of non-negative sequences.
Uses direct gradient descent with projection, Adam optimizer in log-space,
multiple diverse initializations, iterated local search with restarts,
and simulated annealing for final polishing.

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 for non-negative a."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    g = np.max(c)
    return 2.0 * n * g / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gradient_descent_direct(a0, time_limit, tau_init=None, base_lr=0.1,

    """

    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())

        tau_init = max(0.01 * (np.max(c0) - np.min(c0)), 1e-6 / n)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-10
    step = 0

    best_a = a.copy()
    t0 = time.monotonic()
    last_improve = t0
    stagnant = 0

        c = fft_convolve(a, a)
        c_max = np.max(c)

        spread = max(c_max - c_min, 1e-12)

        frac = min(1.0, elapsed / time_limit)

        h = 2.0 * np.correlate(w, a, mode='full')[:n]
        h_mean = np.dot(h, a)
        grad = 4.0 * n * (h - h_mean)

def sa_polish(a0, time_limit, seed=42, verbose=False):
    """Simulated annealing for final polishing."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    
    t0 = time.monotonic()
    # Adaptive temperature
    c0 = fft_convolve(a, a)
    score_scale = 2.0 * n * (np.max(c0) - np.min(c0))
    temperature = 0.002 * score_scale
    min_temp = 1e-8 * score_scale
    cooling = 0.9998
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Batch of iterations (check time less frequently)
        for _ in range(50):
            if time.monotonic() - t0 >= time_limit:
                break
            a_new = a.copy()
            r = rng.random()
            
            if r < 0.3:
                # Peak-targeted perturbation
                c = fft_convolve(a, a)
                peak_idx = int(np.argmax(c))
                # Find contributors to peak
                contribs = np.zeros(n)
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = a[i] * a[j]
                top_k = max(2, n // 8)
                top_idx = np.argsort(contribs)[-top_k:]
                a_new[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))
                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 0:
                    n_inc = min(4, len(other))
                    inc_idx = rng.choice(other, size=n_inc, replace=False)
                    a_new[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
            elif r < 0.5:
                # Window perturbation
                width = max(1, int(rng.integers(1, max(2, n // 4))))
                start = int(rng.integers(0, n - width))
                a_new[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
            elif r < 0.7:
                # Multiplicative noise
                a_new *= np.exp(rng.normal(0.0, 0.01, size=n))
            elif r < 0.85:
                # Swap-based perturbation
                i, j = rng.integers(0, n, size=2)
                delta = a[i] * 0.1
                a_new[i] = max(0, a[i] - delta)
                a_new[j] = a[j] + delta
            else:
                # Gaussian in log-space
                log_a = np.log(a + 1e-15)
                log_a += rng.normal(0.0, 0.008, size=n)
                a_new = np.exp(log_a - np.max(log_a))
            
            a_new = np.clip(a_new, 0.0, None)
            a_new = normalize(a_new)
            new_score = compute_score(a_new)
            iter_count += 1
            
            if new_score < current_score:
                a = a_new
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    if verbose and iter_count % 5000 == 0:
                        print(f"    SA new best: {best_score:.6f} at iter {iter_count}")
            elif temperature > min_temp:
                if rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
                    a = a_new
                    current_score = new_score
        
        temperature *= cooling
    
    return best_a, best_score


def make_chirp_init(n, rng, alpha=None):
    """Chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    if alpha is None:
        alpha = 0.3 + 0.5 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_decay_init(n, rng, rate=None):
    """Exponential decay initialization (asymmetric)."""
    t = np.arange(n, dtype=np.float64)
    if rate is None:
        rate = rng.uniform(0.5, 2.0) / np.sqrt(n)
    direction = rng.random() < 0.5
    if direction:
        a = np.exp(-rate * t)
    else:
        a = np.exp(-rate * (n - 1 - t))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng):
    """Random initialization with various distributions."""
    r = rng.random()
    if r < 0.3:
        a = rng.exponential(1.0, size=n)
    elif r < 0.6:
        a = 1.0 + 0.5 * rng.standard_normal(n)
    else:
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * rng.uniform(1, 5) * t / n)
    return np.clip(a, 0.01, None)


def make_triangular_init(n, rng, peak_pos=None):
    """Triangular (tent) initialization with random peak position."""
    t = np.arange(n, dtype=np.float64)
    if peak_pos is None:
        peak_pos = int(rng.integers(0, n))
    a = 1.0 - np.abs(t - peak_pos) / max(peak_pos, n - 1 - peak_pos, 1)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
