Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient descent
on the probability simplex with adaptive log-sum-exp temperature,
backtracking line search, and multi-restart strategy.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_obj_and_grad(a, t, c=None):
    """Compute the log-sum-exp smoothed objective and gradient.
    
    F(a) = (1/t) * log(sum_k exp(t * c[k])) where c = conv(a,a)
    dF/da[i] = sum_k p[k] * 2*a[k-i] where p[k] = exp(t*c[k]) / sum_j exp(t*c[j])
    
    Returns (F, grad, c, p)
    """
    n = len(a)
    if c is None:
        c = fast_convolve(a)
    
    c_max = c.max()
    if c_max < 1e-15:
        return 0.0, np.zeros(n), c, np.ones(len(c)) / len(c)

    w = np.exp(scaled)
    w_sum = w.sum()
    p = w / w_sum
    
    F = c_max + np.log(w_sum) / t
    
    # Gradient: dF/da[i] = sum_k p[k] * 2*a[k-i] (for valid k-i in [0, n-1])
    grad = np.zeros(n)

        pk = p[k]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)

def lbfgs_optimize(x0, obj_grad_fn, time_limit, m=10, maxiter=10000):
    """Simple L-BFGS implementation.
    
    x0: initial point
    obj_grad_fn: function that returns (f, g) where f is objective and g is gradient
    time_limit: seconds
    m: number of correction pairs to store
    """
    x = x0.copy()
    f, g = obj_grad_fn(x)
    
    # L-BFGS storage
    s_list = []  # step directions
    y_list = []  # gradient differences
    rho_list = []  # 1/(y^T s)
    
    # Two-loop recursion to compute search direction
    def two_loop_recursion(g, m):
        q = g.copy()
        alphas = []
        for i in range(len(s_list) - 1, -1, -1):
            alpha = rho_list[i] * np.dot(s_list[i], q)
            alphas.append(alpha)
            q -= alpha * y_list[i]
        alphas.reverse()
        
        # Initial Hessian approximation: H0 = 1 (identity)
        r = q.copy()
        
        for i in range(len(s_list)):
            beta = rho_list[i] * np.dot(y_list[i], r)
            r += (alphas[i] - beta) * s_list[i]
        return r
    
    t0 = time.monotonic()
    d = -g  # initial direction
    step = 1.0
    
    for it in range(maxiter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Line search (backtracking)
        f_new, g_new = obj_grad_fn(x + step * d)
        Wolfe_c1 = 1e-4
        Wolfe_c2 = 0.9
        ls_iter = 0
        while f_new > f + Wolfe_c1 * step * np.dot(g, d) or \
              np.dot(g_new, d) < Wolfe_c2 * np.dot(g, d):
            step *= 0.5
            f_new, g_new = obj_grad_fn(x + step * d)
            ls_iter += 1
            if ls_iter > 30 or step < 1e-10:
                break
        
        # Update L-BFGS memory
        s = step * d
        y = g_new - g
        sy = np.dot(s, y)
        if sy > 1e-10:
            s_list.append(s.copy())
            y_list.append(y.copy())
            rho_list.append(1.0 / sy)
            if len(s_list) > m:
                s_list.pop(0)
                y_list.pop(0)
                rho_list.pop(0)
        
        x = x + s
        f = f_new
        g = g_new
        
        # Compute new search direction
        if len(s_list) > 0:
            d = -two_loop_recursion(g, len(s_list))
        else:
            d = -g
        
        # Reset step if line search failed
        if step < 1e-10:
            step = 1.0
    
    return x, f

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def smooth_obj_grad(x, t, n):
    a = softmax(x)
    c = fast_convolve(a)
    c_max = c.max()
    if c_max < 1e-15:
        return 0.0, np.zeros(n)
