def subgradient(f, active_lags):
    """Compute subgradient of max_{k in active_lags} (f*f)[k] w.r.t. f."""
    n = len(f)
    sg = np.zeros(n)
    for k in active_lags:
        if k >= n:
            k = 2*n - 2 - k  # symmetry
        if k == 0:
            # (f*f)[0] = sum f[i]^2, d/df[i] = 2*f[i]
            sg += 2 * f
        else:
            sg[:n-k] += f[k:]
            sg[k:] += f[:n-k]
    return sg

    if S is None:
        S = np.sqrt(2.0 * n)
    f = np.clip(f, 0, None).copy()
    s = f.sum()
    if s < 0.01:
        f = np.ones(n) * S / n
    else:
        f = f * S / s

    best_f = f.copy()
    best_score = curr_score
    t0 = time.monotonic()
    
    f_prev = f.copy()
    momentum = 0.0
    lr = 0.1
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        y = f + momentum * (f - f_prev)
        y = np.clip(y, 0, None)
        
        # Compute autocorrelation at y
        c = autocorr(y)
        max_c = c.max()
        
        # Find active lags (within 1% of max)
        thresh = max_c * 0.999
        active = np.where(c >= thresh)[0].tolist()
        
        # Compute subgradient
        sg = subgradient(y, active)
        sg_mean = sg.mean()
        sg_centered = sg - sg_mean  # center to preserve sum
        
        # Update
        f_new = y - lr * sg_centered
        f_new = np.clip(f_new, 0, None)
        s_new = f_new.sum()
        if s_new < 0.01:
            f_new = f.copy()
        else:
            f_new = f_new * S / s_new
        
        new_score = score(f_new)
        
        if new_score < curr_score:
            momentum = min(0.9, momentum * 1.01 + 0.001)
            lr = min(0.5, lr * 1.01)
            stall = 0
        else:
            momentum *= 0.95
            lr *= 0.9
            stall += 1
        
        f_prev = f
        f = f_new
        curr_score = new_score
        
        if curr_score < best_score:
            best_score = curr_score
            best_f = f.copy()
        
        # Escape mechanism
        if stall > 100:
            f = f * np.exp(rng.normal(0, 0.05, n))
            f = np.clip(f, 0, None)
            s = f.sum()
            if s > 0.01:
                f = f * S / s
                curr_score = score(f)
            lr = 0.05
            momentum = 0
            stall = 0
        
        if verbose and (int((time.monotonic() - t0) * 10) % 50 == 0):
            print(f"  NS: t={time.monotonic()-t0:.1f}s, score={curr_score:.6f}, best={best_score:.6f}, lr={lr:.4f}")
    
    return best_f, best_score
