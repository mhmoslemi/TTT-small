import numpy as np

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_and_conv(a):
    """Compute score and full autocorrelation."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return None, np.inf
    c = fft_convolve(a, a)
    score = 2.0 * n * np.max(c) / (S * S)
    return c, score


def soft_gradient_step(a, alpha, tau):
    """
    Gradient step on smoothed objective.
    Uses top-k lags approximation for efficiency.
    """
    n = len(a)
    S = np.sum(a)
    c = fft_convolve(a, a)
    
    c_shifted = (c - np.max(c)) / tau
    exp_c = np.exp(c_shifted)
    w = exp_c / np.sum(exp_c)
    
    # Only use lags with significant weight
    sorted_idx = np.argsort(w)[::-1]
    top_k = sorted_idx[:min(20, n)]  # top 20 lags
    
    grad_C = np.zeros(n)
    sm_val = np.max(c) + tau * np.log(np.sum(exp_c) + 1e-30)
    
    for idx in top_k:
        k = int(idx)
        wk = w[k]
        if wk < 1e-8:
            continue
        if k == 0:
            grad_C += wk * 2.0 * a
        else:
            valid = n - k
            grad_C[:valid] += wk * a[k:]
            grad_C[k:] += wk * a[:valid]
    
    # Gradient of f = 2n * sm_val / S^2
    # df/da_i = (2n/S^2)(d(sm_val)/da_i - 2*sm_val/S)
    step_vec = grad_C - 2.0 * sm_val / S
    a_new = a - alpha * step_vec / S
    a_new = np.clip(a_new, 0.0, None)
    S_new = np.sum(a_new)
    if S_new < 1e-10:
        return a
    a_new = a_new * (S / S_new)
    return a_new


def peak_targeted_mutation(a, rng, n, strength=0.08):
    """
    Mutate elements that contribute most to peak autocorrelation.
    Reduce peak contributors, increase non-contributors.
    """
    c = fft_convolve(a, a)
    peak_lag = int(np.argmax(c))
    
    if peak_lag == 0:
        # Peak at DC: high elements contribute to a_i^2
        contribs = a * a
    else:
        contribs = np.zeros(n)
        valid = n - peak_lag
        contribs[:valid] = a[:valid] * a[peak_lag:]
        # Also the "other side"
        contribs[peak_lag:] += a[:valid] * a[peak_lag:]  # Wait, this is the same
    
    # Actually for C_k = sum_{i=k}^{n-1} a_i * a_{i-k}:
    # a_i contributes a_i * a_{i-k} (from the "right" side)
    # a_{i-k} contributes a_{i-k} * a_i (from the "left" side)
    # So a_j contributes: a_j * a_{j+k} (if j+k < n) + a_j * a_{j-k} (if j >= k)
    if peak_lag == 0:
        contribs = 2.0 * a * a
    else:
        k = peak_lag
        contribs = np.zeros(n)
        contribs[:n-k] += a[:n-k] * a[k:]
        contribs[k:] += a[k:] * a[:n-k]
    
    total_contrib = contribs.sum()
    if total_contrib < 1e-20:
        return a.copy()
    
    # Rank by contribution
    sorted_idx = np.argsort(contribs)[::-1]
    
    n_reduce = max(5, n // 16)
    n_increase = max(3, n // 20)
    
    reduce_idx = sorted_idx[:n_reduce]
    candidates = np.setdiff1d(np.arange(n), reduce_idx)
    increase_idx = rng.choice(candidates, size=min(n_increase, len(candidates)), replace=False)
    
    a_new = a.copy()
    for idx in reduce_idx:
        a_new[idx] *= np.exp(rng.normal(-strength, strength * 0.3))
    for idx in increase_idx:
        a_new[idx] *= np.exp(rng.normal(strength * 0.7, strength * 0.3))
    
    a_new = np.clip(a_new, 0.0, None)
    S = np.sum(a)
    S_new = np.sum(a_new)
    if S_new > 1e-10:
        a_new = a_new * (S / S_new)
    return a_new

def crossover(p1, p2, rng):
    """Two-point crossover with reverse of middle segment."""
    n = len(p1)
    p1, p2 = p1.copy(), p2.copy()
    if n < 4:
        return p1
    pt1 = int(rng.integers(1, n - 1))
    pt2 = int(rng.integers(pt1 + 1, n))

    child = np.concatenate([
        p1[:pt1],
        p2[pt1:pt2][::-1],  # reverse for diversity
        p1[pt2:]
    ])
    return child

    """
    Gradient step on smoothed objective using top-k lag approximation.
    Minimizes 2n * logsumexp(C_k/tau) * tau / S^2
    """

    S = np.sum(a)
    if S < 1e-10:
        return a

    threshold = 1e-6

        return a

    a = a / S * np.sqrt(2.0 * n)

    best_a = a.copy()
    _, best_score = score_and_conv(a)
    current_score = best_score
    iter_count = 0

    tau = 15.0 / np.sqrt(n)
    alpha = 0.8
    sa_temp = 0.006
    sa_cooling = 0.9998
    alpha_decay = 0.9996
    tau_decay = 0.995

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        if iter_count % 300 == 0 and iter_count > 0:
            tau = max(1.5 / n, tau * tau_decay)
            alpha = max(0.05, alpha * alpha_decay)
            sa_temp = max(0.0003, sa_temp * (sa_cooling ** 3))

        r = rng.random()
        if r < 0.55:
            a_new = soft_gradient_step(a, alpha, tau)
        elif r < 0.78:
            strength = 0.04 + 0.06 * (sa_temp / 0.006)
            strength = min(strength, 0.15)
            a_new = peak_targeted_mutation(a, rng, n, strength)
        elif r < 0.90:
            strength = 0.02 + 0.04 * (sa_temp / 0.006)
            strength = min(strength, 0.10)
            a_new = window_mutation(a, rng, n, strength)
        else:
            strength = 0.015 + 0.025 * (sa_temp / 0.006)
            strength = min(strength, 0.08)
            a_new = global_perturbation(a, rng, n, strength)

        _, new_score = score_and_conv(a_new)

        if new_score < current_score:
            a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            delta = new_score - current_score
            accept_prob = np.exp(-delta / max(sa_temp, 1e-10))
            if accept_prob > rng.random():
                a = a_new
                current_score = new_score

        iter_count += 1
        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      iter={iter_count}, cur={current_score:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.4f}, alpha={alpha:.3f}, T={sa_temp:.5f}, {elapsed:.0f}s/{time_budget:.0f}s")

    return best_a, best_score


def make_chirp(n, freq, rng):
    """Generate chirp initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t * t / n)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_multitone(n, rng):
    """Generate multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(6):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.08, 0.25)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_random_smooth(n, rng, n_basis=8):
    """Generate smooth random initialization via random Fourier basis."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_basis):
        k = int(rng.integers(1, max(2, n // 5)))
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_periodic(n, rng, period):
    """Generate periodic initialization with noise."""
    n_eff = min(period, n)
    base = 1.0 + 0.3 * rng.standard_normal(n_eff)
    base = np.clip(base, 0.1, None)
    a = np.tile(base, n // n_eff + 1)[:n]
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase evolutionary optimization:
    Phase 1: Diversified initialization + individual optimization
    Phase 2: C进化rossover + re-optimization  
    Phase 3: Intensive SA on top candidates
    Phase 4: Final fine-tuning
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Get provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.array(height_sequence_1, dtype=np.float64)
        provided = np.clip(provided, 0.01, 1000.0)
        _, prov_score = score_and_conv(provided)
        print(f"Provided sequence: n={len(provided)}, score={prov_score:.6f}")

    n_base = len(provided) if provided is not None else 256

    best_score = np.inf
    best_seq = None
    population = []

    def record_result(a, score, label=""):
        nonlocal best_score, best_seq
        population.append((a.copy(), score))
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  *** New best: {score:.6f} ({label}) ***")

    # === Phase 1: Diversified initialization + optimization ===
    print("\n" + "=" * 60)
    print("Phase 1: Diversified initialization + optimization")
    print("=" * 60)

    initializations = []

    if provided is not None:
        initializations.append(("provided", provided.copy()))
        sym = (provided + provided[::-1]) / 2.0
        initializations.append(("sym", sym))
        a_p1 = provided * np.exp(rng.normal(0, 0.08, size=n_base))
        initializations.append(("pert08", np.clip(a_p1, 0.01, None)))
        a_p2 = provided * np.exp(rng.normal(0, 0.15, size=n_base))
        initializations.append(("pert15", np.clip(a_p2, 0.01, None)))
        a_p3 = provided * np.exp(rng.normal(0, 0.25, size=n_base))
        initializations.append(("pert25", np.clip(a_p3, 0.01, None)))
        for shift in [n_base // 5, n_base // 3, n_base // 2]:
            a_sh = np.roll(provided, shift)
            initializations.append((f"shift{shift}", a_sh))

    for i in range(3):
        a_r = 1.0 + 0.5 * rng.standard_normal(n_base)
        initializations.append((f"rand{i}", np.clip(a_r, 0.05, None)))

    for freq in [0.3, 0.7, 1.3]:
        initializations.append((f"chirp{freq}", make_chirp(n_base, freq, rng)))

    for i in range(2):
        initializations.append((f"multit{i}", make_multitone(n_base, rng)))

    for i in range(2):
        initializations.append((f"smooth{i}", make_random_smooth(n_base, rng)))

    for period in [4, 8, 16]:
        if period < n_base:
            initializations.append((f"per{period}", make_periodic(n_base, rng, period)))

    print(f"  Total initializations: {len(initializations)}")

    phase1_fraction = 0.32
    available = deadline - time.monotonic()
    phase1_time = available * phase1_fraction
    per_init_time = phase1_time / len(initializations)

    for i, (name, a0) in enumerate(initializations):
        if time.monotonic() > deadline - 250:
            break
        alloc = min(per_init_time * 1.2, deadline - time.monotonic() - 250)
        if alloc < 5:
            alloc = 5
        if time.monotonic() + alloc > deadline - 250:
            break

        a_opt, score = optimize_single(a0, alloc, rng, verbose=False)
        print(f"  [{name}] score={score:.6f} (alloc={alloc:.0f}s)")
        record_result(a_opt, score, name)

    population.sort(key=lambda x: x[1])
    print(f"  Phase 1 done. Best so far: {best_score:.6f}, population={len(population)}")

    # === Phase 2: Crossover + re-optimization ===
    print("\n" + "=" * 60)
    print("Phase 2: Crossover + re-optimization")
    print("=" * 60)

    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.28
    if phase2_time > 40:
        top_k = min(6, len(population))
        top_parents = [p[0] for p in population[:top_k]]

        n_offspring = 8
        per_off_time = phase2_time / n_offspring

        for i in range(n_offspring):
            if time.monotonic() > deadline - 180:
                break
            alloc = min(per_off_time * 1.3, deadline - time.monotonic() - 180)
            if alloc < 10:
                break

            p1 = top_parents[rng.integers(0, top_k)]
            p2 = top_parents[rng.integers(0, top_k)]

            cross_type = rng.random()
            if cross_type < 0.3:
                child = crossover_two_point(p1, p2, rng)
            elif cross_type < 0.6:
                child = uniform_crossover(p1, p2, rng)
            else:
                child = arith_crossover(p1, p2, rng)

            child = child * np.exp(rng.normal(0, 0.03, size=n_base))
            child = np.clip(child, 0.01, None)

            a_opt, score = optimize_single(child, alloc, rng, verbose=False)
            print(f"  [cross{i}] score={score:.6f}")
            record_result(a_opt, score, f"cross{i}")

    population.sort(key=lambda x: x[1])
    print(f"  Phase 2 done. Best so far: {best_score:.6f}")

    # === Phase 3: Intensive optimization on top candidates ===
    print("\n" + "=" * 60)
    print("Phase 3: Intensive optimization on top candidates")
    print("=" * 60)

    remaining = deadline - time.monotonic()
    phase3_time = remaining * 0.28
    if phase3_time > 40:
        top_n = min(4, len(population))
        top_cands = population[:top_n]
        per_cand_time = phase3_time / top_n

        for i, (a_cand, s_cand) in enumerate(top_cands):
            if time.monotonic() > deadline - 100:
                break
            alloc = min(per_cand_time * 1.2, deadline - time.monotonic() - 100)
            if alloc < 20:
                break

            print(f"  Intensive on candidate {i} (score={s_cand:.6f}), time={alloc:.0f}s")
            a_opt, score = optimize_single(a_cand, alloc, rng, verbose=True)
            print(f"  [intensive{i}] score={score:.6f}")
            record_result(a_opt, score, f"intensive{i}")

    population.sort(key=lambda x: x[1])
    print(f"  Phase 3 done. Best so far: {best_score:.6f}")

    # === Phase 4: Final fine-tuning ===
    print("\n" + "=" * 60)
    print("Phase 4: Final fine-tuning")
    print("=" * 60)

    remaining = deadline - time.monotonic() - 20
    if remaining > 30 and best_seq is not None:
        print(f"  Final tuning with {remaining:.0f}s remaining")
        a_final, score_final = optimize_single(best_seq, remaining, rng, verbose=True)
        print(f"  [final] score={score_final:.6f}")
        record_result(a_final, score_final, "final")

    # Final verification
    print("\n" + "=" * 60)
    print(f"FINAL RESULT: score={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")
    print("=" * 60)

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        return final_list

    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]

    return [1.0] * 256
