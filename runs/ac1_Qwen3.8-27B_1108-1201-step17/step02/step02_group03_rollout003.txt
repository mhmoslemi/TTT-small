"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses improved LP-based refinement with adaptive blending, multi-scale exploration,
and gradient-based escape from local minima.

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def gradient_soft(a, tau):
    """Soft-max gradient using cross-correlation via convolution."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(-tau * c_shifted)  # Note: negative tau to down-weight peaks
    # Actually, for minimizing max, we want to up-weight peaks
    w = np.exp(tau * (c - c.max()))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad

def gradient_hard(a):
    """Hard-max gradient."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad

def lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    # Total constraints: 2n-1
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j)
        hi = min(2 * n - 2, j + n - 1)
        for m in range(lo, hi + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0:
            return res.x
    except Exception as e:
        pass
    return None

"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses LP-based refinement with active-set constraint selection, adaptive blending,
gradient-based escape, and multi-start exploration. Builds on known-good starting
point with targeted improvements to the LP iteration strategy.

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def gradient_soft(a, tau):
    """Soft-max gradient using cross-correlation."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def gradient_hard(a):
    """Hard-max gradient."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def lp_solve_active(f, M, n_active=None, time_limit=30):
    """
    Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Uses active-set: only constrains lags near the peak for speed.
    """
    n = len(f)
    c = np.convolve(f, f)
    
    # Select active constraints: lags where conv(f,f) > 0.8*M
    if n_active is None:
        n_active = min(2 * n - 1, max(n, int(0.9 * (2 * n - 1))))
    
    # Sort lags by value, take top ones
    active_lags = np.argsort(c)[::-1][:n_active]
    
    c_obj = -np.ones(n)
    A = np.zeros((len(active_lags), n))
    
    for row, m in enumerate(active_lags):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[row, j] = f[m - j]
    
    b = np.full(len(active_lags), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_solve_full(f, M, time_limit=60):
    """Solve the full LP with all constraints."""
    n = len(f)
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_iteration(a, M, S_t, use_full=False, time_limit=30):
    """
    One LP iteration: solve LP, normalize solution, return it.

    g_norm = g * (S_t / g_sum)
