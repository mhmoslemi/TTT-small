import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (S * S)


def softmax_normalize(theta, S):
    """Convert theta to normalized a with sum = S."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e) * S


def compute_grad_a(a, p):
    """Compute gradient of LSE w.r.t. a given softmax weights p.
    g_a[i] = sum_j p[i+j]*a[j] + a[i]
    = (a_rev * p)[n-1+i] + a[i]
    """
    n = len(a)
    a_rev = a[::-1]
    ap = fft_convolve(a_rev, p)
    return ap[n - 1:2 * n - 1] + a


def gradient_step(theta, tau, S):
    """One gradient computation. Returns (a, g_theta, lse_val, conv_max)."""
    n = len(theta)
    a = softmax_normalize(theta, S)
    conv = fft_convolve(a, a)
    conv_shifted = conv / tau
    conv_max_shifted = np.max(conv_shifted)
    exp_shifted = np.exp(conv_shifted - conv_max_shifted)
    p = exp_shifted / np.sum(exp_shifted)
    g_a = compute_grad_a(a, p)
    inner = np.dot(g_a, a) / S
    g_theta = a * (g_a - inner)
    lse_val = tau * (conv_max_shifted + np.log(np.sum(exp_shifted)))
    return a, g_theta, lse_val, np.max(conv)

            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(
                rng.normal(0.0, 0.015, size=width))
        elif r < 0.75:
            # Pair swap (exchange values at two positions)
            i, j = rng.choice(n, size=2, replace=False)
            tmp = candidate[i]
            candidate[i] = candidate[j]
            candidate[j] = tmp
        elif r < 0.9:
            # Small global perturbation
            candidate *= np.exp(rng.normal(0.0, 0.004, size=n))
        else:
            # Larger perturbation (rare)
            width = max(2, n // 10)
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(
                rng.normal(0.0, 0.03, size=width))

        candidate = np.clip(candidate, 0.0, 1000.0)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate *= (S / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            f = candidate
            current_score = new_score

        temperature = max(temperature * cooling, 1e-6)

    return best_f, best_score


def make_random_init(n, rng, style='uniform'):
    """Generate a random initialization."""
    if style == 'uniform':
        a = rng.uniform(0.5, 1.5, size=n)
    elif style == 'exponential':
        a = np.exp(-np.arange(n) / (n * rng.uniform(0.5, 2.0)))
        a += 0.1 * rng.standard_normal(n)
    elif style == 'bell':
        t = np.arange(n) / n
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
        a += 0.05 * rng.standard_normal(n)
    elif style == 'chirp':
        t = np.arange(n)
        freq = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
        a += 0.05 * rng.standard_normal(n)
    elif style == 'two_peak':
        t = np.arange(n) / n
        c1, c2 = sorted(rng.uniform(0.1, 0.9, size=2))
        a = np.exp(-0.5 * ((t - c1) / 0.15) ** 2) + np.exp(-0.5 * ((t - c2) / 0.15) ** 2)
        a += 0.05 * rng.standard_normal(n)
    elif style == 'linear':
        slope = rng.uniform(-0.5, 0.5)
        a = 1.0 + slope * np.arange(n) / n
        a += 0.1 * rng.standard_normal(n)
    else:
        a = rng.uniform(0.5, 1.5, size=n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for minimum normalized peak self-convolution.
    
    Phase 1: Multi-start exploration with different n and initializations.
    Phase 2: Extended Adam optimization on best candidate with basin hopping.
    Phase 3: SA polish.
    Phase 4: Final Adam refinement if time permits.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided)
        print(f"Provided: n={len(provided)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided.copy()

    n_prov = len(provided) if provided is not None else 256

    # === Phase 1: Multi-start exploration ===
    print("\n" + "=" * 60)
    print("Phase 1: Multi-start exploration")
    print("=" * 60)

    # Define search configurations: (n, init_style, time_alloc_fraction)
    # Try multiple lengths and initialization styles
    search_configs = []
    if provided is not None:
        search_configs.append((n_prov, 'provided', 0.20))
    search_configs.extend([
        (256, 'chirp', 0.12),
        (256, 'uniform', 0.10),
        (256, 'two_peak', 0.08),
        (384, 'chirp', 0.10),
        (384, 'uniform', 0.08),
        (192, 'chirp', 0.07),
        (512, 'chirp', 0.08),
        (512, 'uniform', 0.07),
        (256, 'bell', 0.05),
        (320, 'exponential', 0.05),
    ])

    total_frac = sum(c[2] for c in search_configs)
    phase1_time = (deadline - time.monotonic()) * 0.55

    for n, init_style, frac in search_configs:
        if time.monotonic() > deadline - phase1_time * 0.3:
            break
        time_alloc = frac / total_frac * phase1_time
        time_alloc = max(10.0, min(time_alloc, deadline - time.monotonic() - 10))
        if time_alloc < 10:
            break

        # Generate initialization
        if init_style == 'provided' and provided is not None and n == n_prov:
            a0 = provided.copy()
        else:
            a0 = make_random_init(n, rng, style=init_style)
            # Normalize for consistent comparison
            a0 = a0 / np.sum(a0) * np.sqrt(2.0 * n)

        print(f"  n={n}, init='{init_style}', time={time_alloc:.0f}s", end="")
        a_opt, score = adam_optimize(a0, time_alloc, seed=int(rng.integers(0, 10000)))
        elapsed = time.monotonic() - t_global
        print(f" -> score={score:.6f} (elapsed={elapsed:.0f}s)")

        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}) ***")

    # === Phase 2: Extended optimization on best ===
    print("\n" + "=" * 60)
    print("Phase 2: Extended optimization")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        phase2_time = min(remaining * 0.3, 300.0)
        if phase2_time > 20:
            print(f"  n={len(best_seq)}, time={phase2_time:.0f}s")
            # Try multiple restarts from best with different seeds
            for restart in range(3):
                if time.monotonic() > deadline - 60:
                    break
                alloc = phase2_time / (3 - restart)
                s = int(seed * 1000 + restart * 7)
                a_opt, score = adam_optimize(best_seq, alloc, seed=s)
                print(f"  Restart {restart}: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_seq = a_opt.copy()
                    print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 3: SA polish ===
    print("\n" + "=" * 60)
    print("Phase 3: SA polish")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(remaining * 0.3, 200.0)
        if sa_time > 15:
            print(f"  n={len(best_seq)}, time={sa_time:.0f}s")
            a_sa, score_sa = sa_polish(best_seq, sa_time, seed=seed + 999)
            print(f"  SA result: {score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Final refinement ===
    print("\n" + "=" * 60)
    print("Phase 4: Final refinement")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        final_time = min(remaining * 0.5, 150.0)
        if final_time > 15:
            print(f"  Final Adam, n={len(best_seq)}, time={final_time:.0f}s")
            a_fin, score_fin = adam_optimize(best_seq, final_time, seed=seed + 4242)
            print(f"  Final Adam: {score_fin:.6f}")
            if score_fin < best_score:
                best_score = score_fin
                best_seq = a_fin.copy()
                print(f"  *** New best: {best_score:.6f} ***")

            # Quick SA after final Adam
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                a_sa2, score_sa2 = sa_polish(best_seq, min(remaining2 * 0.6, 60.0), seed=seed + 7777)
                if score_sa2 < best_score:
                    best_score = score_sa2
                    best_seq = a_sa2.copy()
                    print(f"  *** Final SA best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    return [1.0] * 256
