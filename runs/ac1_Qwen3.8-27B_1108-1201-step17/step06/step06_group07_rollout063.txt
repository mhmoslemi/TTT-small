from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_constraint_matrix(a, n, m):
    """
    Build the LP constraint matrix for the symmetric case.
    a: full symmetric sequence of length n = 2m
    Returns A (n x m) where A[k, i] = coefficient of x_i in (a*g)_k
    (for k = 0, ..., n-1, exploiting symmetry)
    """
    A = np.zeros((n, m))
    for i in range(m):
        # x_i appears in g at positions i and n-1-i
        for k in range(n):
            coeff = 0.0
            # Contribution from g_i (position i): a_{k-i}
            idx1 = k - i
            if 0 <= idx1 < n:
                coeff += a[idx1]
            # Contribution from g_{n-1-i} (position n-1-i): a_{k-(n-1-i)}
            idx2 = k - (n - 1 - i)
            if 0 <= idx2 < n:
                coeff += a[idx2]
            A[k, i] = coeff
    return A

    return np.concatenate([x, x[::-1]])

"""
LP-based mirror descent with symmetric constraint for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact LP direction finding (faster convergence than subgradient), symmetry exploitation,
multi-start with diverse initializations, and subgradient polishing.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def sym_to_full(x, m):
    return np.concatenate([x, x[::-1]])


def full_to_sym(a, n):
    m = n // 2
    return a[:m].copy()


def build_A(a, n, m):
    """Build LP constraint matrix A[k,i] = coeff of x_i in (a*g)_k for k=0..n-1."""
    k = np.arange(n)[:, None]
    i = np.arange(m)[None, :]
    idx1 = k - i
    idx2 = k - (n - 1 - i)
    A = np.zeros((n, m))
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    # Need to be careful with fancy indexing
    rows1, cols1 = np.where(valid1)
    A[rows1, cols1] += a[idx1[rows1, cols1]]
    rows2, cols2 = np.where(valid2)
    A[rows2, cols2] += a[idx2[rows2, cols2]]
    return A


def lp_solve(a, n, m):
    """Solve the LP: min M s.t. (a*g)_k <= M, sum(g)=1, g>=0, g symmetric. Returns (g_full, M_opt)."""
    A = build_A(a, n, m)
    
    # Variables: x_0,...,x_{m-1}, M (total m+1)
    c = np.zeros(m + 1)
    c[m] = 1.0
    
    A_ub = np.zeros((n, m + 1))
    A_ub[:, :m] = A
    A_ub[:, m] = -1.0
    b_ub = np.zeros(n)
    
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 2.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (m + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None, None
    
    x_opt = result.x[:m]
    M_opt = result.x[m]
    g_full = sym_to_full(x_opt, m)
    return g_full, M_opt

    n = len(a_init)
    m = n // 2

    a = 0.5 * (a + a[::-1])
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)

        g, M_opt = lp_solve(a, n, m)

        if M_opt >= M_current - 1e-10:

            if no_improve > 30:

                a = sym_to_full(x, m)
                no_improve = 0

            if no_improve > 50:
                break
            continue
        
        a = (1 - best_t) * a + best_t * g

        a = 0.5 * (a + a[::-1])

            no_improve = 0
            if verbose and it % 20 == 0:
                print(f"    {label} LP iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
    
    return best_a, best_obj


def subgradient_polish(a_init, time_limit, verbose=True, label=""):
    """Subgradient method for fine polishing."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        peak_idx = np.argmax(conv)
        
        # Subgradient of max(conv) at peak
        k = int(peak_idx)
        g = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        g[js:je+1] = 2.0 * a[k - np.arange(js, je+1)]
        
        # Update
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
        
        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.5
                no_improve = 0
            if lr < 1e-8:
                break
    
    return best_a, best_obj

    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = a / a.sum()
    return a

    a = (1.0 - t ** 4)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

"""
LP-based optimization with symmetry exploitation for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact LP direction finding (faster than subgradient), symmetry constraint to halve
problem size, multi-start with diverse initializations, and subgradient polishing.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def sym_to_full(x, m):
    """Convert first-half x (length m) to full symmetric sequence (length 2m)."""
    return np.concatenate([x, x[::-1]])


def full_to_sym(a, n):
    """Convert full symmetric sequence (length n=2m) to first-half (length m)."""
    m = n // 2
    return a[:m].copy()


def build_A(a, n, m):
    """Build LP constraint matrix: A[k,i] = coeff of x_i in (a*g)_k for k=0..n-1."""
    k = np.arange(n)[:, None]
    i = np.arange(m)[None, :]
    idx1 = k - i
    idx2 = k - (n - 1 - i)
    A = np.zeros((n, m))
    rows1, cols1 = np.where((idx1 >= 0) & (idx1 < n))
    A[rows1, cols1] += a[idx1[rows1, cols1]]
    rows2, cols2 = np.where((idx2 >= 0) & (idx2 < n))
    A[rows2, cols2] += a[idx2[rows2, cols2]]
    return A


def lp_solve(a, n, m):
    """
    Solve LP: min M s.t. (a*g)_k <= M for k=0..n-1, sum(g)=1, g>=0, g symmetric.
    Returns (g_full, M_opt) or (None, None) on failure.
    """
    A = build_A(a, n, m)

    c = np.zeros(m + 1)
    c[m] = 1.0

    A_ub = np.zeros((n, m + 1))
    A_ub[:, :m] = A
    A_ub[:, m] = -1.0
    b_ub = np.zeros(n)

    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 2.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (m + 1)

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')

    if not result.success:
        return None, None

    x_opt = result.x[:m]
    M_opt = result.x[m]
    g_full = sym_to_full(x_opt, m)
    return g_full, M_opt


def lp_optimize(a_init, time_limit, verbose=True, label=""):
    """LP-based optimization with line search on symmetric sequences."""
    n = len(a_init)
    m = n // 2
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n * fast_convolve(a).max()
    a = a / s
    a = 0.5 * (a + a[::-1])
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(a)
        M_current = conv.max()

        g, M_opt = lp_solve(a, n, m)

        if g is None:
            break

        if M_opt >= M_current - 1e-10:
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng(it * 7919 + 13)
                x = full_to_sym(a, n)
                noise = rng.standard_normal(m) * (0.002 + 0.003 * min(no_improve / 100, 1.0))
                x = np.maximum(x + noise, 0)
                ss = x.sum()
                if ss > 1e-15:
                    x = x / ss
                    a = sym_to_full(x, m)
                no_improve = 0
            elif no_improve > 10:
                # Try slightly larger perturbation
                rng = np.random.default_rng(it * 104729)
                x = full_to_sym(a, n)
                noise = rng.standard_normal(m) * 0.008
                x = np.maximum(x + noise, 0)
                ss = x.sum()
                if ss > 1e-15:
                    x = x / ss
                    a = sym_to_full(x, m)
            continue

        # Line search
        best_t = 0.0
        best_new_obj = 2.0 * n * M_current
        a_best_trial = a.copy()
        for t in [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            a_trial = 0.5 * (a_trial + a_trial[::-1])
            a_trial = a_trial / a_trial.sum()
            obj = 2.0 * n * fast_convolve(a_trial).max()
            if obj < best_new_obj:
                best_new_obj = obj
                best_t = t
                a_best_trial = a_trial.copy()

        if best_t == 0.0:
            no_improve += 1
            if no_improve > 60:
                break
            continue

        a = a_best_trial

        if best_new_obj < best_obj - 1e-12:
            best_obj = best_new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    {label} LP iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

    return best_a, best_obj


def subgradient_polish(a_init, time_limit, verbose=True, label=""):
    """Subgradient method for fine polishing (uses top-k peaks)."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n * fast_convolve(a).max()
    a = a / s
    a = 0.5 * (a + a[::-1])
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.004
    top_k = 5
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Use top-k peaks for smoother gradient
        K = min(top_k, len(conv))
        peak_idx = np.argpartition(conv, -K)[-K:]
        pv = conv[peak_idx]
        w = pv ** 4  # Emphasize larger peaks
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(K)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = np.zeros(n)
        for ki in range(K):
            k = int(peak_idx[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                g[js:je + 1] += 2.0 * w[ki] * a[k - np.arange(js, je + 1)]

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        a_new = 0.5 * (a_new + a_new[::-1])
        a_new = a_new / a_new.sum()
        a = a_new

        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                lr *= 0.7
                no_improve = 0
            if lr < 1e-9:
                break
        lr = max(lr * 0.9995, 1e-9)

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = a / a.sum()
    return a


def make_bell(n, width_frac=0.7):
    x = np.arange(n) - n / 2.0
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = np.maximum(1.0 - t ** 4, 0)
    a = a / a.sum()
    return a


def make_cosine(n, power=2.0):
    x = np.arange(n) / (n - 1) * 2.0 - 1.0
    a = np.cos(np.pi * x / 2.0) ** power
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def make_flat_top(n, frac=0.5):
    x = np.arange(n) - n / 2.0
    half = n * frac / 2.0
    t = np.clip(np.abs(x - half) / half, 0, 1.0)
    a = np.maximum(1.0 - t ** 2, 0)
    a = a / a.sum()
    return a


def symmetrize(a):
    a_s = 0.5 * (a + a[::-1])
    a_s = np.maximum(a_s, 0)
    s = a_s.sum()
    if s > 1e-15:
        a_s = a_s / s
    return a_s


def try_sequence(a, state, label=""):

    return score
