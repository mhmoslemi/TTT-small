```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a Majorization-Minimization (MM)
approach with LP linearization, combined with subgradient descent, L-BFGS
polishing, and diverse initializations.

The key innovation is the MM-LP step: at each iteration, we linearize the
quadratic auto-convolution around the current point and solve the resulting
LP to find a new candidate. This provides a global (over the simplex) step
rather than a local gradient step, enabling faster convergence.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):

    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    One MM-LP iteration.
    Linearize b[k] = a^T Q_k a around a_current, solve LP.
    Returns (a_new, t_new) or None if LP fails.
    """

    b_current = fast_convolve(a_current)

    cols = []
    vals = []

        hi = min(n - 1, k)

                cols.append(j)
                vals.append(val)
        # t variable (index n)
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)

    b_ub = b_current
    
    # Objective: minimize t (last variable)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

            a_new = res.x[:n]
            t_new = res.x[n]
            return a_new, t_new, b_current
        else:
            return None
    except Exception as e:

        return None
```

Actually, the matrix-vector product `M @ x` where M is a Hankel matrix can be done in O(n log n) using FFT! Because a Hankel matrix is related to a Toeplitz matrix, and Toeplitz matrix-vector products can be done via FFT.

Specifically, if `M[i,j] = w[i+j]`, then `(Mx)[i] = sum_j w[i+j] x[j]`. This is a cross-correlation of `w` and `x` (reversed). We can compute this via FFT in O(n log n).

```python
def hankel_mv(w, x):
    """Compute M @ x where M[i,j] = w[i+j] for i,j in range(n)."""
    n = len(x)
    # (Mx)[i] = sum_j w[i+j] x[j]
    # This is the cross-correlation of w and x (flipped)
    # Let x_flipped = x[::-1]
    # Then (Mx)[i] = sum_j w[i+j] x[j] = sum_k w[k] x[k-i] (substituting k = i+j)
    # = cross-correlation at lag -i
    # Actually, let me just use direct FFT approach
    # (Mx)[i] = sum_j w[i+j] x[j] for j = 0..n-1, i = 0..n-1
    # Let's define y = x reversed: y[j] = x[n-1-j]
    # Then (Mx)[i] = sum_j w[i+j] x[j] = sum_k w[i + (n-1-k)] y[k]
    # Hmm, this is getting complicated. Let me just use the direct approach with FFT.
    
    # Alternative: M = H(w) where H is the Hankel matrix
    # M @ x can be computed as: flip M to get a Toeplitz matrix, then use FFT
    # M[i,j] = w[i+j]. If we flip M (reverse columns), we get M'[i,j] = w[i + (n-1-j)]
    # This is a Toeplitz matrix with first column w[0], w[1], ..., w[n-1]
    # and first row w[n-1], w[n-2], ..., w[0]
    
    # Actually, let me just do it directly with FFT for the convolution form:
    # (Mx)[i] = sum_{j=0}^{n-1} w[i+j] * x[j]
    # = <w[i:i+n], x> (dot product)
    
    # This is like a sliding dot product. We can compute all n values via FFT:
    # Define W = w (length 2n-1), X = x (length n)
    # (Mx)[i] = sum_j W[i+j] * X[j]
    # = cross-correlation of W and X at lag i
    # = (W * X_flipped)[n-1+i] where * is linear convolution and X_flipped = X[::-1]
    
    X_flipped = x[::-1]
    # Linear convolution of W (length 2n-1) and X_flipped (length n)
    # Result has length 2n-1 + n - 1 = 3n-2
    fft_size = 1 << (3*n - 1).bit_length()
    W_fft = rfft(w, fft_size)
    Xf_fft = rfft(X_flipped, fft_size)
    conv = irfft(W_fft * Xf_fft, fft_size)
    # (Mx)[i] = conv[i + (n-1)] for i = 0..n-1
    return conv[n-1 : 2*n-1]
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. MM-LP: Majorization-Minimization with LP linearization of the quadratic
   auto-convolution, providing global steps over the simplex.
2. IRLS-QP: Iteratively Reweighted Least Squares with efficient Hankel
   matrix-vector products via FFT.
3. Subgradient descent with Nesterov momentum for fine-tuning.
4. Coordinate descent for final micro-optimization.
5. Diverse structural initializations and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mm_lp_step(a_current):
    """One MM-LP iteration: linearize and solve LP."""
    n = len(a_current)
    b_current = fast_convolve(a_current)
    n_constr = 2 * n - 1

    # Build sparse constraint matrix
    # Estimate max nonzeros: each row has at most n nonzeros + 1 for t
    max_nnz = n_constr * (n + 1)
    rows = np.zeros(max_nnz, dtype=np.int32)
    cols = np.zeros(max_nnz, dtype=np.int32)
    vals = np.zeros(max_nnz, dtype=np.float64)
    ptr = 0

    for k in range(n_constr):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            v = 2.0 * a_current[k - j]
            if abs(v) > 1e-15:
                if ptr >= max_nnz:
                    break
                rows[ptr] = k
                cols[ptr] = j
                vals[ptr] = v
                ptr += 1
        if ptr >= max_nnz:
            break
        rows[ptr] = k
        cols[ptr] = n
        vals[ptr] = -1.0
        ptr += 1

    A_ub = csr_matrix((vals[:ptr], (rows[:ptr], cols[:ptr])), shape=(n_constr, n + 1))
    b_ub = b_current.astype(np.float64)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            a_new = np.maximum(res.x[:n], 0.0)
            t_new = float(res.x[n])
            return a_new, t_new, b_current
        return None
    except Exception:
        return None


def mm_lp_optimize(a0, time_limit, max_iters=5000, use_line_search=True):
    """MM-LP optimization with optional line search."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_obj = best_obj
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit - 3:
            break

        result = mm_lp_step(a)
        if result is None:
            stagnant += 1
            if stagnant > 30:
                break
            continue
        a_new, t_new, b_cur = result
        stagnant = 0

        if use_line_search:
            improved = False
            for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
                a_trial = (1.0 - alpha) * a + alpha * a_new
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-10:
                    continue
                a_trial = a_trial / st
                obj = fast_convolve(a_trial).max()
                if obj < cur_obj - 1e-14:
                    a = a_trial
                    cur_obj = obj
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                    improved = True
                    break
            if not improved:
                # Small perturbation to escape
                a = project_simplex(a + np.random.standard_normal(n) * 0.001)
                cur_obj = fast_convolve(a).max()
        else:
            a = a_new
            s = a.sum()
            if s > 1e-10:
                a = a / s
            cur_obj = fast_convolve(a).max()
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()

    return best_a, best_obj


def hankel_mv(w, x):
    """Compute M @ x where M[i,j] = w[i+j] using FFT."""
    n = len(x)
    Xf = x[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W_fft = rfft(w, fft_size)
    Xf_fft = rfft(Xf, fft_size)
    conv = irfft(W_fft * Xf_fft, fft_size)
    return conv[n - 1: 2 * n - 1]


def irls_qp_optimize(a0, time_limit, max_iters=200, weight_power=2.0):
    """IRLS with QP subproblems using FFT-accelerated Hankel products."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit - 3:
            break

        b = fast_convolve(a)
        b_clipped = np.maximum(b, 1e-15)
        w = 1.0 / (b_clipped ** weight_power)
        w = w / w.sum()

        # Solve QP: minimize x^T M x s.t. sum(x)=1, x>=0
        # where M[i,j] = w[i+j] (Hankel matrix)

        qp_obj_count = [0]

        def qp_obj(x):
            qp_obj_count[0] += 1
            return float(x @ hankel_mv(w, x))

        def qp_grad(x):
            qp_obj_count[0] += 1
            return 2.0 * hankel_mv(w, x)

        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                 'jac': lambda x: np.ones_like(x)}]
        bnds = [(0, None)] * n

        try:
            res = minimize(qp_obj, a.copy(), jac=qp_grad, method='L-BFGS-B',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10})
            a_new = np.maximum(res.x, 0.0)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new = a_new / sn
            obj = fast_convolve(a_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
        except Exception:
            continue

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.9,
                         temp_init=0.0003, temp_growth=1.01,
                         kick_interval=2000, kick_size=0.003,
                         n_top=None, seed=42):
    """Projected subgradient with Nesterov momentum."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        for _ in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            it_count += 1
            b = fast_convolve(a)
            b_max = b.max()

            scaled = b / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum

            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w[k] * a[k - idx]

            v = momentum * v - cur_lr * g
            a_new = project_simplex(a + v)
            b_new = fast_convolve(a_new)
            obj_new = b_new.max()

            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new

        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95

        cur_temp = min(cur_temp * temp_growth, 0.5)

    return best_a, best_obj


def coordinate_polish(a0, time_limit, n_trials=6):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.8, 2.5])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    return best_a, best_obj


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    a = np.maximum(a, 1e-10)

def make_tukey_init(n, r=0.5):

def make_uniform_init(n):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lbgfs_refine(a0, T, time_limit, maxiter=3000):
    """L-BFGS in log-space with soft-max objective."""
    t0 = time.monotonic()
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))

    def func(z):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T) * log_sum_exp

    def jac(z):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # grad_a = 2n * cross_corr(w, a)
        # cross_corr[w, a][i] = sum_k w[k+i] * a[i]
        size = n + len(b) - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * n * irfft(np.conj(W) * A, fft_size)[:n]
        S = np.dot(grad_a, a)
        return a * (grad_a - a * S)

    best_obj = func(z0)
    best_z = z0.copy()
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxfun': maxiter * 10})
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x
            best_obj = final_obj
    except Exception:
        pass
    return softmax_from_z(best_z), best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining MM-LP, IRLS-QP, subgradient,
    and coordinate descent to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1152) / 1152
        print("No parent, using uniform n=1152")

    n_parent = len(parent)
    print(f"Starting optimization, parent n={n_parent}")

    # === Phase 1: MM-LP from parent ===
    print(f"\n=== Phase 1: MM-LP from parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 120:
        t_alloc = min(rem, 500.0)
        print(f"  Allocating {t_alloc:.0f}s for MM-LP")
        a_mm, obj_mm = mm_lp_optimize(parent, time_limit=t_alloc, max_iters=5000)
        try_seq(a_mm, f"mm_lp_n{n_parent}")

    # === Phase 2: IRLS-QP refinement ===
    print(f"\n=== Phase 2: IRLS-QP (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_alloc = min(rem, 300.0)
            print(f"  IRLS-QP t={t_alloc:.0f}s")
            a_irls, _ = irls_qp_optimize(best_seq, time_limit=t_alloc,
                                          max_iters=100, weight_power=2.0)
            try_seq(a_irls, "irls_q
p")
            # Try different weight powers
            rem = deadline - time.monotonic() - 300
            if rem > 30:
                t_alloc2 = min(rem, 120.0)
                a_irls2, _ = irls_qp_optimize(best_seq, time_limit=t_alloc2,
                                               max_iters=50, weight_power=4.0)
                try_seq(a_irls2, "irls_qp_p4")

    # === Phase 3: Subgradient fine-tuning ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_alloc = min(rem, 200.0)
            print(f"  Subgradient n={n_best}, t={t_alloc:.0f}s")
            a_sg, _ = subgradient_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=0.002, momentum=0.92,
                temp_init=0.0002, temp_growth=1.01,
                kick_interval=1000, kick_size=0.002,
                n_top=max(20, n_best // 10), seed=seed)
            try_seq(a_sg, f"sg_n{n_best}")

    # === Phase 4: Diverse initializations ===
    print(f"\n=== Phase 4: Diverse inits (best={best_score:.6f}) ===")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits = {
        "uniform": make_uniform_init(n_base),
        "gauss_0.2": make_gaussian_init(n_base, 0.2),
        "gauss_0.4": make_gaussian_init(n_base, 0.4),
        "cosine": make_cosine_init(n_base),
        "tukey": make_tukey_init(n_base, 0.4),
    }

    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 300
        if rem < 20:
            break
        t_alloc = min(rem * 0.2, 60.0)
        if t_alloc < 10:
            continue
        print(f"  {name}: MM-LP t={t_alloc:.0f}s")
        a_opt, _ = mm_lp_optimize(init_a, time_limit=t_alloc, max_iters=2000)
        try_seq(a_opt, f"div_{name}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.75, 1.25, 1.5]:
            tl = int(n_parent * factor)
            tl = max(512, min(3072, tl))
            if tl not in target_lengths and tl != n_parent:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:3]:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 80.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = mm_lp_optimize(f_interp, time_limit=t_alloc, max_iters=2000)
            try_seq(a_opt, f"len_n{target_n}")

    # === Phase 6: L-BFGS soft-max polish ===
    print(f"\n=== Phase 6: L-BFGS polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for T in [50, 100]:
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem * 0.4, 40.0)
            if t_alloc < 10:
                continue
            print(f"  L-BFGS T={T}, t={t_alloc:.0f}s")
            a_lb, _ = lbgfs_refine(best_seq, T=T, time_limit=t_alloc, maxiter=2000)
            try_seq(a_lb, f"lbgfs_T{T}")

    # === Phase 7: Coordinate descent ===
    print(f"\n=== Phase 7: CD polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_cd = min(rem, 60.0)
            print(f"  CD t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd")

    # === Phase 8: Final MM-LP deep run ===
    print(f"\n=== Phase 8: Final MM-LP (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 30:
            t_alloc = min(rem, 120.0)
            print(f"  Final MM-LP t={t_alloc:.0f}s")
            a_fin, _ = mm_lp_optimize(best_seq, time_limit=t_alloc, max_iters=3000)
            try_seq(a_fin, "final_mm")

            # One more coordinate polish
            rem = deadline - time.monotonic() - 5
            if rem > 10 and best_seq is not None:
                a_cd2, _ = coordinate_polish(best_seq, time_limit=min(rem, 20.0))
                try_seq(a_cd2, "cd2")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
```
