    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play_optimize(a0, time_limit, solver=None, verbose=False):
    """
    Fictitious play with harmonic averaging for faster convergence.
    Uses the LP to compute best responses alternately.
    """
    n = len(a0)
    t0 = time.monotonic()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s

    b = a.copy()
    # Use exponential moving average with rate that decreases as ~1/t
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Best response for b given a
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        # Best response for a given b
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        # Harmonic averaging (faster convergence than arithmetic mean)
        alpha = 2.0 / (iters + 2.0)
        a_avg = (1 - alpha) * a_avg + alpha * a

        # Evaluate candidates
        candidates = [a, b_avg if iters > 1 else a, a_avg, 0.5 * (a + b)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            sc_val = 2.0 * n * fast_convolve(cand_norm).max()
            if sc_val < best_score:
                best_score = sc_val
                best_a = cand_norm.copy()

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            ma_str = f"{M_a:.6f}" if M_a is not None else "N/A"
            print(f"    FP iter={iters} ({elapsed:.0f}s): M_a={ma_str} best={best_score:.6f}")

    return best_a, best_score, iters


def mirror_descent_refine(a0, time_limit, eta_init=0.5, verbose=False):
    """
    Mirror descent (exponentiated gradient) on the simplex.
    Uses subgradients of the max auto-correlation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Compute auto-correlation
        conv = fast_convolve(a)
        # Find top peaks for subgradient
        # Use all peaks above average to get richer gradient
        threshold = conv.mean() + 0.5 * (conv.max() - conv.mean())
        peak_indices = np.where(conv > threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])

        # Compute weighted subgradient
        weights = conv[peak_indices] - conv.mean()
        weights = np.maximum(weights, 0)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.ones(len(peak_indices))
        else:
            weights /= w_sum

        grad = np.zeros(n)
        for i, k in enumerate(peak_indices):
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            grad[idx] += weights[i] * a[k_int - idx]

        # Scale gradient
        grad = 2.0 * grad

        # Multiplicative (mirror descent) update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        sc_val = 2.0 * n * fast_convolve(a_new).max()
        if sc_val < best_score - 1e-14:
            best_score = sc_val
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Increase eta slightly on improvement
            eta = min(eta * 1.01, 2.0)
        else:
            no_improve += 1
            # Decrease eta when no improvement
            eta = max(eta * 0.98, 0.01)

        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
            eta = eta_init
            no_improve = 0

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.6f} eta={eta:.4f}")

    return best_a, best_score


def subgradient_momentum(a0, time_limit, lr_init=0.005, top_k=30, verbose=False):
    """Subgradient method with Nesterov momentum for fine-tuning."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        # Temperature for weighting
        temp = 0.01 * max(pv.max() - pv.min(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - lr * sg

        # Nesterov step
        a_candidate = a + v + (mom - 1) * (a - (a + v))
        # Actually just a + v for simplicity
        a_candidate = a + v

        # Project onto simplex {a >= 0, sum(a) = 1}
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        sc_val = 2.0 * n * fast_convolve(a_new).max()
        if sc_val < best_score - 1e-14:
            best_score = sc_val
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            a = a_new  # still move

        lr *= 0.9997
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 5, 0.003)
            no_improve = 0

        if verbose and no_improve % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradMom {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, verbose=False):
    """
    Cycle through coordinates, trying to reduce each one while maintaining
    the sum constraint. Focus on coordinates that contribute most to peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Find the peak
        peak_k = int(np.argmax(conv))
        # Find which coordinates contribute most to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Try reducing the top contributors
        top_contrib = np.argsort(contribs)[::-1][:5]
        improved = False

        for ci in top_contrib:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-12:
                continue
            # Try reducing i and redistributing
            for factor in [0.9, 0.8, 0.7, 0.95]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                osum = others.sum()
                if osum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / osum
                sc_val = 2.0 * n * fast_convolve(a_trial).max()
                if sc_val < best_score - 1e-14:
                    a = a_trial
                    best_score = sc_val
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break

        if verbose and no_improve % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.6f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T=200, maxiter=5000, verbose=False):
    """L-BFGS refinement in log-space with soft-max smoothing of the max."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-10)
    a_safe = a_safe / a_safe.sum()
    z0 = np.log(a_safe)
    t0 = time.monotonic()
    stopped = [False]

    def soft_max_obj(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15, np.zeros(n)
        a = np.exp(z - z.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        Tb = conv * T
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        obj = (2.0 * n / T) * log_sum_exp

        # Gradient
        w = exp_Tb / exp_Tb.sum()
        # grad of (a*a)[k] w.r.t. a_i = a_{k-i} + a_{k-i} (with bounds)
        grad_a = np.zeros(n)
        for k in range(len(conv)):
            if w[k] < 1e-12:
                continue
            k_int = k
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k_int - idx]

        # Chain rule: d/dz = a * (grad_a - a * sum(grad_a * a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return obj, grad_z

    if stopped[0]:
        return a0.copy()

    obj0, grad0 = soft_max_obj(z0)
    if stopped[0]:
        return a0.copy()

    def func(z):
        obj, _ = soft_max_obj(z)
        return obj

    def jac(z):
        _, g = soft_max_obj(z)
        return g

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a0.copy()

    a_out = np.maximum(a_out, 0)
    a_out = a_out / a_out.sum()
    return a_out


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Cosine
    t = np.linspace(0, 2 * np.pi, n)
    for freq in [1, 2, 3]:
        for amp in [0.1, 0.3, 0.5]:
            c = 1.0 + amp * np.cos(freq * t)
            c = np.maximum(c, 0)
            c = c / c.sum()
            inits.append(c)
    # Gaussian
    for sigma in [0.2, 0.4, 0.6, 0.8]:
        g = np.exp(-0.5 * ((np.arange(n) - n / 2) / (sigma * n)) ** 2)
        g = g / g.sum()
        inits.append(g)
    # Sinc-like
    for width in [0.3, 0.5, 0.7]:
        x = (np.arange(n) - n / 2) / (width * n)
        sinc = np.sinc(x)
        sinc = np.maximum(sinc, 0)
        ss = sinc.sum()
        if ss > 1e-10:
            inits.append(sinc / ss)
    # Random
    for _ in range(5):
        r = rng.standard_normal(n)
        r = np.maximum(r, 0)
        rs = r.sum()
        if rs > 1e-10:
            inits.append(r / rs)
    # Power law
    for alpha in [0.5, 1.0, 2.0]:
        p = np.power(np.linspace(0.01, 1, n), alpha)
        p = p / p.sum()
        inits.append(p)
    return inits


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate a candidate and return updated best."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        sc = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        sc = np.inf
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        if label:
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining fictitious play, mirror descent,
    L-BFGS, and coordinate descent for minimizing auto-correlation peak.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Find LP solver
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Determine lengths to try
    if parent is not None:
        n_parent = len(parent)
        n_values = [n_parent]
        # Add nearby lengths
        for factor in [0.75, 0.9, 1.1, 1.25, 1.5]:
            alt_n = int(n_parent * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_parent:
                n_values.append(alt_n)
        # Add some standard lengths
        for std_n in [512, 768, 1024, 1280, 1536, 2048]:
            if std_n not in n_values and 256 <= std_n <= 4096:
                n_values.append(std_n)
    else:
        n_values = [512, 768, 1024, 1280]

    n_values = list(set(n_values))
    n_values.sort(key=lambda x: abs(x - (n_parent if parent is not None else 768)))
    n_values = n_values[:8]

    print(f"Trying lengths: {n_values}")

    # === Phase 1: Fictitious Play (40% of budget) ===
    print("\n=== Phase 1: Fictitious Play ===")
    total_fp_budget = (deadline - time.monotonic()) * 0.45
    time_per_n = total_fp_budget / len(n_values)
    fp_results = {}

    for n_try in n_values:
        rem = deadline - time.monotonic() - 400
        if rem < 60:
            break
        t_alloc = min(time_per_n, rem * 0.85)
        if t_alloc < 20:
            continue

        print(f"  FP n={n_try}, budget={t_alloc:.0f}s")

        # Choose initialization
        if parent is not None and len(parent) == n_try:
            initial = parent
        elif parent is not None:
            initial = interpolate_sequence(parent, n_try)
        elif best_seq is not None:
            initial = interpolate_sequence(best_seq, n_try)
        else:
            initial = np.ones(n_try) / n_try

        a_fp, score_fp, iters = fictitious_play_optimize(
            initial, t_alloc, solver=solver, verbose=False
        )
        best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"FP_n{n_try}")
        fp_results[n_try] = a_fp

    # === Phase 2: Try multiple initializations at best length ===
    print(f"\n=== Phase 2: Multi-init FP (best={best_score:.6f}) ===")
    n_best = len(best_seq) if best_seq is not None else 768

    rem = deadline - time.monotonic() - 350
    if rem > 60:
        inits = generate_initializations(n_best, rng)
        # Also add interpolated parent
        if parent is not None and len(parent) != n_best:
            inits.append(interpolate_sequence(parent, n_best))
        if best_seq is not None:
            inits.append(best_seq)

        time_per_init = rem / len(inits)
        for i, init in enumerate(inits[:min(len(inits), 8)]):
            rem = deadline - time.monotonic() - 250
            if rem < 15:
                break
            t_alloc = min(time_per_init, rem * 0.8)
            if t_alloc < 10:
                continue
            a_i, sc_i, it_i = fictitious_play_optimize(
                init, t_alloc, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_i, best_seq, best_score, f"init_{i}_n{n_best}")

    # === Phase 3: Mirror Descent Refinement ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_alloc = min(rem * 0.5, 120.0)
            print(f"  MirrorDescent n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_md, _ = mirror_descent_refine(best_seq, t_alloc, eta_init=0.3)
            best_seq, best_score = check_and_update(a_md, best_seq, best_score, "mirror_desc")

    # === Phase 4: L-BFGS with multiple temperatures ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        for T in [30, 100, 300, 800]:
            rem = deadline - time.monotonic() - 150
            if rem < 10:
                break
            t_alloc = min(rem * 0.3, 60.0)
            print(f"  L-BFGS T={T}, n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lb = lbgfs_refine(best_seq, t_alloc, T=T, maxiter=3000)
            best_seq, best_score = check_and_update(a_lb, best_seq, best_score, f"lbgfs_T{T}")

    # === Phase 5: Subgradient with momentum ===
    print(f"\n=== Phase 5: Subgradient Momentum (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem * 0.5, 80.0)
            top_k = max(20, len(best_seq) // 8)
            a_sg, _ = subgradient_momentum(best_seq, t_alloc, lr_init=0.004, top_k=top_k)
            best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "subgrad_mom")

    # === Phase 6: Coordinate Descent / Peak Shaving ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_alloc = min(rem * 0.6, 50.0)
            a_cd, _ = coordinate_descent_polish(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "coord_desc")

    # === Phase 7: Second round of FP + refinement ===
    print(f"\n=== Phase 7: FP Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 50
        if rem > 30:
            t_alloc = min(rem * 0.5, 150.0)
            print(f"  FP round 2: n={n_best}, budget={t_alloc:.0f}s")
            a_fp2, sc_fp2, it2 = fictitious_play_optimize(
                best_seq, t_alloc, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_fp2, best_seq, best_score, f"FP2_n{n_best}")

            # Follow up with L-BFGS
            if best_seq is not None:
                rem = deadline - time.monotonic() - 30
                if rem > 10:
                    t_alloc = min(rem * 0.6, 40.0)
                    a_lb2 = lbgfs_refine(best_seq, t_alloc, T=200, maxiter=2000)
                    best_seq, best_score = check_and_update(a_lb2, best_seq, best_score, "lbgfs_2")

                    # Final mirror descent
                    rem = deadline - time.monotonic() - 15
                    if rem > 8:
                        t_alloc = min(rem * 0.7, 20.0)
                        a_md2, _ = mirror_descent_refine(best_seq, t_alloc, eta_init=0.2)
                        best_seq, best_score = check_and_update(a_md2, best_seq, best_score, "mirror_2")

    # === Phase 8: Try a different length for the final best ===
    print(f"\n=== Phase 8: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        alt_lengths = [int(n_cur * f) for f in [0.8, 1.2, 0.6, 1.5]
                       if 256 <= int(n_cur * f) <= 4096 and int(n_cur * f) != n_cur]
        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            t_alloc = min(rem * 0.5, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, sc_alt, _ = fictitious_play_optimize(
                init_alt, t_alloc, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # === Final: micro-polish with coordinate descent ===
    print(f"\n=== Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_alloc = min(rem * 0.8, 15.0)
            a_final, _ = coordinate_descent_polish(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_final, best_seq, best_score, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
