```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def objective(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (S * S)

def project_nonneg_sum1(v):
    """Project onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    # Water-filling algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_star = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_star] - 1.0) / (rho_star + 1)
    return np.maximum(v - theta, 0.0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max L-BFGS with aggressive temperature annealing (main workhorse)
2. Direct subgradient with top-K peak averaging and diminishing step size
3. Proper coordinate descent with 1D bounded optimization
4. Multi-length exploration and symmetric constraints

Key differences from prior approach:
- Systematic T-annealing in L-BFGS (high T -> low T) for near-exact max approximation
- FFT-based cross-correlation for efficient gradient in log-space
- scipy minimize_scalar for coordinate descent (not fixed multipliers)
- Symmetric sequence exploitation (a_i = a_{n-1-i})
- Better time allocation focusing on deep optimization of top candidates
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_wa(w, a):
    """Compute g[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for i=0,...,n-1.
    w has length 2n-1, a has length n. Uses FFT for efficiency."""
    n = len(a)
    fw = len(w)
    size = n + fw - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    R = irfft(W * np.conj(A), fft_size)
    return R[:n]


def softmax_obj_grad(z, T):
    """Soft-max smoothed objective and gradient in z-space (unconstrained).
    a = softmax(z), so sum(a)=1, a>=0 automatically."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    c = fast_convolve(a)
    
    # Numerically stable log-sum-exp
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    log_sum_exp = Tc_max + np.log(exp_Tc.sum())
    m_T = log_sum_exp / T
    
    # Objective (S=1 since a is on simplex)
    obj = 2.0 * n * m_T
    
    # Weights for soft-max
    w = exp_Tc / exp_Tc.sum()
    
    # Gradient w.r.t. a: dF/da_i = 4n * g_i where g_i = sum_j w_{i+j} * a_j
    g = cross_corr_wa(w, a)
    da_grad = 4.0 * n * g
    
    # Transform to z-space: dz_k = a_k * (da_k - sum_i a_i * da_i)
    mean_grad = np.dot(a, da_grad)
    dz_grad = a * (da_grad - mean_grad)
    
    return obj, dz_grad


def lbgfs_softmax_anneal(a0, T_list, time_limit, maxiter_per_T=2000):
    """L-BFGS on soft-max objective with temperature annealing.
    Starts at high T (smooth) and anneals to low T (sharp)."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-300))
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    for T in T_list:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        func_count = [0]
        
        def func(z):
            func_count[0] += 1
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = softmax_obj_grad(z, T)
            return obj
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            _, g = softmax_obj_grad(z, T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_T, 'ftol': 1e-15,
                         'gtol': 1e-12, 'maxfun': maxiter_per_T * 30,
                         'disp': False}
            )
            z = res.x
            a = np.exp(z - z.max())
            a = a / a.sum()
            
            # Evaluate true objective
            true_obj = 2.0 * n * fast_convolve(a).max()
            if true_obj < best_obj:
                best_obj = true_obj
                best_a = a.copy()
            print(f"    T={T}: truth_obj={true_obj:.8f}, lbfgs_obj={func(res.x):.8f}, iters={func_count[0]}")
        except Exception as e:
            print(f"    T={T}: L-BFGS error: {e}")
            break
    
    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.002, top_k=50,
                         diminishing=True, kick_interval=2000, kick_size=0.003):
    """Direct subgradient with top-K peak averaging and diminishing step size."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy()
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    it = 0
    v = np.zeros(n)
    momentum = 0.9
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        c_max = c.max()
        
        if c_max < best_obj - 1e-15:
            best_obj = c_max
            best_a = a.copy()
        
        # Get top-K peaks
        if top_k < len(c):
            top_idx = np.argpartition(c, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(c))
        
        # Compute subgradient: average gradient over top-K peaks
        # d c_k / d a_i = 2 * a[k-i] if 0 <= k-i < n
        g = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi+1] += a[k - np.arange(lo, hi + 1)]
        g = g / len(top_idx) * 2.0
        
        # Step size
        if diminishing:
            lr = lr_init / np.sqrt(1.0 + it * 0.01)
        else:
            lr = lr_init
        
        # Nesterov momentum step
        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / ss
        a = a_new
        
        # Periodic kick
        if it % kick_interval == 0:
            noise = np.random.standard_normal(n) * kick_size
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_passes_max=20):
    """Coordinate descent with proper 1D bounded optimization."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    def obj_1d(t, i, a_rest, s_rest):
        """Objective when a[i]=t, others scaled to maintain sum=1."""
        if t < 0:
            return 1e10
        if t >= 1.0:
            return 1e10
        scale = (1.0 - t) / s_rest if s_rest > 1e-10 else 0
        a_trial = a_rest * scale
        a_trial[i] = t
        c = fast_convolve(a_trial)
        return c.max()
    
    for pass_num in range(n_passes_max):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[i]
            if old_val < 1e-12:
                continue
            s_rest = 1.0 - old_val
            if s_rest < 1e-10:
                continue
            a_rest = a.copy()
            a_rest[i] = 0.0
            
            # Bounded 1D minimization
            res = minimize_scalar(
                lambda t: obj_1d(t, i, a_rest, s_rest),
                bounds=(0.0, min(0.99, old_val * 3.0)),
                method='bounded',
                options={'xatol': 1e-10}
            )
            if res.fun < best_obj - 1e-14:
                new_val = res.x
                scale = (1.0 - new_val) / s_rest
                a = a_rest * scale
                a[i] = new_val
                a = a / a.sum()
                best_obj = res.fun
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_star = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_star] - 1.0) / (rho_star + 1)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_random_init(n, seed=42):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
