import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def sca_lp_iteration(f, n):
    """
    One SCA iteration: solve the LP
    min M
    s.t. sum(a) = 1, a >= 0, M >= 0
         (f*a)[k] - M/2 <= (f*f)[k]/2 for all k
    """
    conv_f = fast_convolve(f)
    d = conv_f / 2.0  # d_k = (f*f)[k] / 2
    
    # Variables: a_0, ..., a_{n-1}, M -> total n+1 variables
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: (f*a)[k] - M/2 <= d_k for k = 0, ..., 2n-2
    # In standard form: A_ub @ x <= b_ub
    num_constraints = 2 * n - 1
    A_ub = np.zeros((num_constraints, n + 1))
    b_ub = np.zeros(num_constraints)

        j_start = max(0, k - n + 1)

        for j in range(j_start, j_end + 1):

    A_ub[:, -1] = -0.5
    return A_ub

q_k(a) = a^T C_k a

So: q_k(a) = q_k(a^(t)) + 2(C_k a^(t))^T (a - a^(t)) + (a-a^(t))^T C_k (a-a^(t))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Successive Convex Approximation (SCA) with exact LP solves
2. Diverse multi-peak initializations (structured low-autocorrelation starts)
3. Adam smooth-surrogate polishing
4. Multi-length exploration
Key innovation: SCA linearizes quadratic autocorrelation constraints into LPs,
enabling exact subproblem solves. Multi-peak initializations exploit the fact
that sequences with separated mass reduce the central autocorrelation peak.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):
    """Compute gradient of log-sum-exp smooth approximation of max-conv."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv


def build_sca_lp(f, n):
    """
    Build and solve the SCA LP:
    min M
    s.t. sum(a) = 1, a >= 0, M >= 0
         (f*a)[k] - M/2 <= (f*f)[k]/2 for all k
    Returns (a_new, M_new) or (None, None) on failure.
    """
    conv_f = fast_convolve(f)
    d = conv_f / 2.0

    num_constraints = 2 * n - 1
    # Build A_ub efficiently: row k has f[j] at positions j where valid
    A_ub = np.zeros((num_constraints, n + 1))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            A_ub[k, j_start:j_end + 1] = f[j_start:j_end + 1]
        A_ub[k, -1] = -0.5

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c, A_ub=A_ub, b_ub=d, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            a_new = np.maximum(result.x[:n], 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new /= s
            M_new = result.x[-1]
            return a_new, M_new
    except Exception:
        pass
    return None, None


def sca_optimize(f0, time_limit, max_iters=5000, verbose=False):
    """
    SCA optimization: iterate LP linearization until convergence.
    Returns (best_f, best_score) where best_score = 2*n*max(conv).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        a_new, M_lp = sca_lp_iteration(f, n)
        if a_new is None:
            no_improve += 5
            if no_improve > 50:
                break
            continue

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Check convergence: if LP solution is close to current, we're at a fixed point
        if np.max(np.abs(a_new - f)) < 1e-10:
            if no_improve > 5:
                break
        else:
            f = a_new

        if verbose and it % 50 == 0:
            print(f"    SCA iter {it}: true_obj={2*n*obj_new:.8f}, lp_M={M_lp:.6e}, best={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def sca_lp_iteration(f, n):
    """Single SCA LP iteration."""
    conv_f = fast_convolve(f)
    d = conv_f / 2.0
    num_constraints = 2 * n - 1

    A_ub = np.zeros((num_constraints, n + 1))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            A_ub[k, j_start:j_end + 1] = f[j_start:j_end + 1]
        A_ub[k, -1] = -0.5

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c, A_ub=A_ub, b_ub=d, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            a_new = np.maximum(result.x[:n], 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new /= s
            M_new = result.x[-1]
            return a_new, M_new
    except Exception:
        pass
    return None, None


def adam_smooth_optimize(f0, time_limit, tau_init=None, lr_init=0.003, verbose=False):
    """Adam optimization of smooth surrogate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.05 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    t = 0
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = compute_smooth_grad(f, tau)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        tau *= 0.9995
        if tau < 1e-8:
            tau = 1e-8

        if no_improve > 80:
            lr *= 0.8
        if no_improve > 300:
            rng = np.random.default_rng(t * 13 + 7)
            noise_scale = 0.001 + 0.002 * min(no_improve / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.5
            tau = 0.03 * fast_convolve(f).max()

        if verbose and t % 200 == 0:
            print(f"    Adam iter {t}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def make_multi_peak(n, num_peaks, centers=None, width=0.06, seed=42):
    """Create a multi-peak Gaussian sequence."""
    rng = np.random.default_rng(seed)
    if centers is None:
        # Spread peaks across the sequence with some randomness
        base_centers = np.linspace(0.1, 0.9, num_peaks)
        jitter = rng.uniform(-0.03, 0.03, num_peaks)
        centers = np.clip(base_centers + jitter, 0.02, 0.98)

    a = np.zeros(n)
    for c in centers:
        mu = c * n
        sigma = width * n
        a += np.exp(-0.5 * (np.arange(n) - mu) ** 2 / sigma ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_modulated(n, freq=1.0, phase=0.0, width=0.15):
    """Sequence modulated by cosine to create alternating structure."""
    x = np.arange(n) / n
    envelope = np.exp(-0.5 * ((x - 0.5) / width) ** 2)
    a = envelope * (1.0 + 0.8 * np.cos(2 * np.pi * freq * x + phase))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gabor(n, center=0.5, sigma=0.2, freq=2.0):
    """Gabor-like function: Gaussian envelope * cosine."""
    x = np.arange(n) / n
    envelope = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = envelope * (1.0 + np.cos(2 * np.pi * freq * (x - center)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def peak_shaving(f, n_shaves=3, verbose=False):
    """
    Identify peak-contributing pairs and redistribute mass.
    A refinement step after SCA.
    """
    f = np.maximum(f.copy(), 0.0)
    f /= f.sum()
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    for iteration in range(50):
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Find top peaks
        top_k = min(n_shaves, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max

            if j_max == j_other:
                # Diagonal: reduce and redistribute
                old_val = f[j_max]
                if old_val < 1e-10:
                    continue
                for factor in [0.5, 0.65, 0.8]:
                    trial = f.copy()
                    trial[j_max] *= factor
                    diff = old_val - trial[j_max]
                    # Redistribute to positions far from this peak
                    mask = np.ones(n, dtype=bool)
                    for pk2 in peak_indices[:3]:
                        js2 = max(0, int(pk2) - n + 1)
                        je2 = min(n - 1, int(pk2))
                        mask[js2:je2 + 1] = False
                    if mask.sum() > 0:
                        trial[mask] += diff * f[mask] / f[mask].sum()
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-12:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
            else:
                # Off-diagonal: reduce the pair product
                old_jm = f[j_max]
                old_jo = f[j_other]
                for scale in [0.5, 0.65, 0.8]:
                    trial = f.copy()
                    trial[j_max] = old_jm * scale
                    diff = old_jm - trial[j_max]
                    mask = np.ones(n, dtype=bool)
                    mask[j_max] = False
                    mask[j_other] = False
                    if mask.sum() > 0:
                        trial[mask] += diff * f[mask] / f[mask].sum()
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-12:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            break

    return best_f, 2 * n * best_obj


def constrained_sca_with_line_search(f0, time_limit, verbose=False):
    """
    SCA with line search safeguard: if the SCA step increases the true objective,
    do a binary search between old and new points.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < 3000:
        it += 1
        a_new, _ = sca_lp_iteration(f, n)
        if a_new is None:
            no_improve += 5
            if no_improve > 40:
                break
            continue

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new <= best_obj + 1e-12:
            # Accept (possibly with line search if obj increased slightly)
            if obj_new > fast_convolve(f).max() * (1 + 1e-10):
                # Line search
                best_alpha = 0.0
                best_val = fast_convolve(f).max()
                for alpha in [0.5, 0.25, 0.75, 1.0]:
                    trial = f + alpha * (a_new - f)
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial /= ts
                    val = fast_convolve(trial).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = alpha
                if best_alpha > 0:
                    f = f + best_alpha * (a_new - f)
                    f = np.maximum(f, 0)
                    f /= f.sum()
                else:
                    f = a_new
            else:
                f = a_new

            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Rejected step: try partial step
            best_alpha = 0.0
            best_val = fast_convolve(f).max()
            for alpha in [0.3, 0.15, 0.5, 0.1]:
                trial = f + alpha * (a_new - f)
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                val = fast_convolve(trial).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

            if best_alpha > 0:
                f = f + best_alpha * (a_new - f)
                f = np.maximum(f, 0)
                f /= f.sum()
                no_improve = 0
            else:
                no_improve += 1

        if np.max(np.abs(a_new - f)) < 1e-11 and no_improve > 3:
            break

        if verbose and it % 100 == 0:
            print(f"    C-SCA iter {it}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Diverse structured initialization (30s)
    Phase 2: SCA optimization from best starts (1100s)
    Phase 3: Adam smooth polish (300s)
    Phase 4: Peak shaving + multi-n exploration (250s)
    Phase 5: Final polish (100s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Diverse Initialization ===
    print("=== Phase 1: Diverse Initialization ===")
    t_phase1_start = time.monotonic()

    # Try parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = try_seq(parent, "parent")
            print(f"  Parent score: {p_score:.6f}")
    except Exception:
        pass

    # Generate diverse structured initializations
    candidates = []
    n_base = 768 if parent is None else len(parent)

    # Multi-peak structures
    for n_try in [n_base, 1024]:
        for num_peaks in [2, 3, 4, 5, 6]:
            for width in [0.04, 0.06, 0.08, 0.1]:
                for sp in range(3):
                    a = make_multi_peak(n_try, num_peaks, width=width, seed=seed + num_peaks * 10 + sp)
                    candidates.append((a, f"multi_{num_peaks}p_w{width}_s{sp}_n{n_try}"))

    # Cosine-modulated
    for n_try in [n_base, 1024]:
        for freq in [1.0, 2.0, 3.0, 5.0]:
            for width in [0.1, 0.15, 0.2]:
                a = make_cosine_modulated(n_try, freq=freq, width=width)
                candidates.append((a, f"cos_f{freq}_w{width}_n{n_try}"))

    # Gabor
    for n_try in [n_base, 1024]:
        for freq in [1.0, 2.0, 3.0, 4.0]:
            for sigma in [0.15, 0.2, 0.25]:
                a = make_gabor(n_try, sigma=sigma, freq=freq)
                candidates.append((a, f"gabor_f{freq}_s{sigma}_n{n_try}"))

    # Random smooth
    for n_try in [n_base]:
        for sp in range(5):
            a = np.abs(rng.standard_normal(n_try)) ** 1.5
            a = np.convolve(a, np.ones(5) / 5, mode='same')
            candidates.append((a, f"rand_s{sp}_n{n_try}"))

    # Evaluate all candidates quickly
    for a, label in candidates:
        if time.monotonic() > deadline - 1500:
            break
        sc = try_seq(a, label)

    if parent is None and best_seq is None:
        best_seq = make_multi_peak(768, 4, width=0.06, seed=42)
        best_score = try_seq(best_seq, "fallback")

    print(f"  Phase 1 done in {time.monotonic()-t_phase1_start:.1f}s, best={best_score:.6f}")

    # === Phase 2: SCA Optimization ===
    print("\n=== Phase 2: SCA Optimization ===")
    t_phase2_start = time.monotonic()

    # Sort candidates by score, take top few for SCA
    # We'll run SCA from the current best and a few diverse others
    sca_starts = [best_seq]

    # Add a few diverse starts
    for a, label in candidates[:30]:
        sc = evaluate_sequence(a.tolist())
        if np.isfinite(sc) and (len(sca_starts) < 5 or sc < best_score + 0.05):
            if len(sca_starts) < 5:
                sca_starts.append(a / a.sum())

    for i, start in enumerate(sca_starts):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        n_sca = len(start)
        t_alloc = min(rem * 0.5, 300)
        if i == 0:
            t_alloc = min(rem * 0.6, 500)  # Give more time to the best start

        print(f"  SCA from start {i} (n={n_sca}): {t_alloc:.0f}s")
        a_opt, obj = sca_optimize(start, time_limit=t_alloc, verbose=(i == 0))
        score = try_seq(a_opt, f"sca_{i}")

        # Also try constrained SCA from the result
        if time.monotonic() < deadline - 450:
            rem2 = deadline - time.monotonic() - 450
            t2 = min(rem2 * 0.3, 150)
            a_opt2, obj2 = constrained_sca_with_line_search(
                a_opt, time_limit=t2, verbose=False)
            try_seq(a_opt2, f"csca_{i}")

    print(f"  Phase 2 done in {time.monotonic()-t_phase2_start:.1f}s, best={best_score:.6f}")

    # === Phase 3: Adam Polish ===
    print("\n=== Phase 3: Adam Polish ===")
    t_phase3_start = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t3 = min(rem, 350)
            print(f"  Time: {t3:.0f}s, n={len(best_seq)}")
            a_opt, obj = adam_smooth_optimize(
                best_seq, time_limit=t3 * 0.6,
                tau_init=0.03 * fast_convolve(best_seq).max(),
                lr_init=0.002, verbose=True)
            try_seq(a_opt, "adam_polish")

            # Second Adam pass with lower temp
            if time.monotonic() < deadline - 280:
                rem2 = deadline - time.monotonic() - 280
                t3b = min(rem2, 150)
                a_opt2, obj2 = adam_smooth_optimize(
                    best_seq, time_limit=t3b,
                    tau_init=0.01 * fast_convolve(best_seq).max(),
                    lr_init=0.001, verbose=False)
                try_seq(a_opt2, "adam_fine")

    print(f"  Phase 3 done in {time.monotonic()-t_phase3_start:.1f}s, best={best_score:.6f}")

    # === Phase 4: Peak Shaving + Multi-n ===
    print("\n=== Phase 4: Peak Shaving + Multi-n ===")
    t_phase4_start = time.monotonic()
    if best_seq is not None:
        # Peak shaving
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t4 = min(rem * 0.4, 100)
            a_opt, obj = peak_shaving(best_seq, n_shaves=5)
            try_seq(a_opt, "peak_shave")

        # Multi-n exploration
        for target_n in [1024, 1536, 640]:
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            try_seq(f_interp, f"interp_n{target_n}")

            # Brief SCA on interpolated
            t4b = min(rem * 0.4, 80)
            if t4b > 15:
                a_opt, obj = sca_optimize(f_interp, time_limit=t4b, verbose=False)
                try_seq(a_opt, f"sca_n{target_n}")

    print(f"  Phase 4 done in {time.monotonic()-t_phase4_start:.1f}s, best={best_score:.6f}")

    # === Phase 5: Final SCA + Polish ===
    print("\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            # Final SCA squeeze
            t5 = min(rem * 0.5, 80)
            a_opt, obj = sca_optimize(best_seq, time_limit=t5, verbose=False)
            try_seq(a_opt, "final_sca")

        # Final Adam
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t5b = min(rem, 40)
            a_opt, obj = adam_smooth_optimize(
                best_seq, time_limit=t5b,
                tau_init=0.005 * fast_convolve(best_seq).max(),
                lr_init=0.0005, verbose=False)
            try_seq(a_opt, "final_adam")

        # Final peak shave
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = peak_shaving(best_seq, n_shaves=3)
            try_seq(a_opt, "final_shave")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
