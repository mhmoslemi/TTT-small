Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent (exponentiated gradient)
with adaptive temperature on the softmax-smoothed objective, combined with
multi-start initialization, length exploration, and LP-FW polish.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, softmax_w, fft_size):
    """
    Compute grad[j] = sum_m a[m] * softmax_w[j+m] for j = 0, ..., n-1.
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    A = np.fft.rfft(a, fft_size)
    S = np.fft.rfft(softmax_w, fft_size)
    # Circular cross-correlation at lags 0 to n-1
    corr = np.fft.irfft(np.conj(A) * S, fft_size)
    return corr[:n]


def mirror_descent_optimize(f0, time_limit=300, lr_init=0.1, tau_init=0.01, 
                            tau_decay=0.999, verbose=False):
    """
    Mirror descent (exponentiated gradient) on the simplex, optimizing
    the softmax-smoothed max of auto-convolution.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # FFT size for gradient computation
    # Need fft_size >= 3n-2 for correct cross-correlation
    fft_size = 1 << (3 * n - 2).bit_length()
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    tau = tau_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute softmax weights
        conv_shifted = conv - conv.max()  # numerical stability
        exp_conv = np.exp(conv_shifted / tau)
        softmax_w = exp_conv / exp_conv.sum()
        
        # Compute gradient
        grad = compute_gradient(f, softmax_w, fft_size)
        
        # Center the gradient (for more stable updates)
        grad_centered = grad - grad.mean()
        
        # Exponentiated gradient update
        f = f * np.exp(-lr * grad_centered)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        # Evaluate
        conv = fast_convolve(f)
        M = conv.max()
        obj = 2.0 * n * M
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate and temperature
        if no_improve > 20:
            lr *= 0.95
            tau *= tau_decay
            if no_improve > 100:
                # Re-heat: add small noise
                rng = np.random.default_rng(it * 7 + 3)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
                f = f / f.sum()
                lr = lr_init * 0.5
                tau = tau_init * 2.0
                no_improve = 0
        
        if lr < 1e-8:
            break
        
        if verbose and it % 50 == 0:
            print(f"    MD iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, tau={tau:.6f}")
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent (exponentiated gradient)
with adaptive temperature annealing on the softmax-smoothed objective, combined with
multi-start initialization, FFT-based gradient computation, length exploration,
and LP-FW/coordinate-descent polish. Key innovations: provably-correct FFT gradient,
adaptive KL-mirror geometry for simplex, multi-scale temperature schedule, and
structured restarts for escaping local minima.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(a, softmax_w, fft_size):
    """
    Compute grad[j] = sum_{m=0}^{n-1} a[m] * softmax_w[j+m] for j=0,...,n-1.
    This is the cross-correlation of a and softmax_w at lags 0 to n-1.
    """
    n = len(a)
    A = np.fft.rfft(a, fft_size)
    S = np.fft.rfft(softmax_w, fft_size)
    corr = np.fft.irfft(np.conj(A) * S, fft_size)
    return corr[:n]


def obj_value(f):
    """Compute objective 2*n*max(conv) for sum-normalized f."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max(), conv


def mirror_descent_core(f0, time_limit=300.0, lr_init=0.05, tau_init=0.005,
                        tau_min=0.0005, tau_decay=0.9995, verbose=False):
    """
    Mirror descent (exponentiated gradient) on simplex for smoothed max-conv objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    fft_size = 1 << (2 * n - 1).bit_length()

    _, conv = obj_value(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    tau = tau_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute softmax weights (numerically stable)
        conv_shifted = conv - conv.max()
        exp_vals = np.exp(conv_shifted / tau)
        softmax_w = exp_vals / exp_vals.sum()

        # Compute gradient via FFT
        grad = compute_gradient_fft(f, softmax_w, fft_size)

        # Center gradient for stability
        grad_centered = grad - grad.mean()

        # Exponentiated gradient update
        update = np.exp(-lr * grad_centered)
        f = f * update
        f = np.maximum(f, 1e-18)
        f = f / f.sum()

        # Evaluate
        _, conv = obj_value(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve == 30:
            lr *= 0.9
            tau *= tau_decay
        elif no_improve == 80:
            lr *= 0.8
            tau = max(tau * 0.9, tau_min)
        elif no_improve == 150:
            # Re-heat with perturbation
            rng = np.random.default_rng(it * 13 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.6
            tau = max(tau_init, tau * 3.0)
            no_improve = 0
        elif no_improve > 400:
            break

        if lr < 1e-10:
            break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD it={it}: obj={best_obj:.10f}, lr={lr:.6f}, tau={tau:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj, conv

    """
    f = np.maximum(f0.astype(np.float64), 0.0)

    if s < 1e-15:

        f = f / s

    fft_size = 1 << (2 * n - 1).bit_length()

    t0 = time.monotonic()
    best_obj = float('inf')
    best_f = f.copy()
    _, conv = obj_value(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    # Temperature schedule: geometric from hot to cold
    tau_hot = 0.02
    tau_warm = 0.005
    tau_cold = 0.001

    # Phase 1: Hot (30% of time)
    t1 = time_limit * 0.3
    f, obj1, conv = mirror_descent_core(f, time_limit=t1, lr_init=0.08,
                                         tau_init=tau_hot, tau_min=tau_warm,
                                         tau_decay=0.998, verbose=False)
    if obj1 < best_obj:
        best_obj = obj1
        best_f = f.copy()
    if verbose:
        print(f"      Phase 1 (hot): obj={obj1:.10f}")

    if time.monotonic() - t0 >= time_limit * 0.85:
        return best_f, best_obj, fast_convolve(best_f)

    # Phase 2: Warm (40% of time)
    t2 = time_limit * 0.4
    elapsed = time.monotonic() - t0
    t2 = min(t2, time_limit - elapsed - time_limit * 0.2)
    if t2 > 10:
        f, obj2, conv = mirror_descent_core(f, time_limit=t2, lr_init=0.04,
                                             tau_init=tau_warm, tau_min=tau_cold,

        if obj2 < best_obj:
            best_obj = obj2
            best_f = f.copy()
        if verbose:

    if time.monotonic() - t0 >= time_limit * 0.95:
        return best_f, best_obj, fast_convolve(best_f)

    # Phase 3: Cold (25% of time)

    t3 = min(t3, time_limit - elapsed - 5)

def subgradient_polish(f0, time_limit=60.0, top_m=5, verbose=False):
    """
    Direct subgradient method: use top-m convolution lags for gradient estimation.
    More aggressive than mirror descent for final polish.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    _, conv = obj_value(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Find top-m lags
        if top_m < len(conv):
            peak_lags = np.argpartition(conv, -top_m)[-top_m:]
            # Sort by value descending
            order = np.argsort(conv[peak_lags])[::-1]
            peak_lags = peak_lags[order]
        else:
            peak_lags = np.arange(len(conv))

        # Compute weighted gradient from top lags
        peak_vals = conv[peak_lags]
        weights = peak_vals ** 4  # sharpen: emphasize largest peaks
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for i, k in enumerate(peak_lags):
            if weights[i] < 1e-15:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += weights[i] * f[k - idx]

        # Projected gradient step
        grad_centered = grad - grad.mean()
        f_new = project_simplex(f - lr * grad_centered)
        f = f_new

        _, conv = obj_value(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            lr *= 0.85
        if no_improve > 200:
            # Perturb and restart
            rng = np.random.default_rng(it * 23 + 1)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 0.0)
            f = f / f.sum()
            lr = 0.01
            no_improve = 0
        if no_improve > 500:
            break
        if lr < 1e-10:
            break

        if verbose and it % 50 == 0:
            print(f"      SubGrad it={it}: obj={best_obj:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit=60.0, verbose=False):
    """
    Coordinate descent: try adjusting pairs of elements that contribute to peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    _, conv = obj_value(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    sweep = 0

    while time.monotonic() - t0 < time_limit and no_improve < 50:
        sweep += 1
        _, conv = obj_value(f)
        improved = False

        # Focus on top peaks
        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Try reducing top contributors and redistributing
            order = np.argsort(products)[::-1]
            for oi in range(min(3, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-15:
                    continue

                for frac in [0.7, 0.8, 0.9]:
                    new_val = old_val * frac
                    d = old_val - new_val
                    if d < 1e-15:
                        continue

                    trial = f.copy()
                    trial[j_max] = new_val

                    # Redistribute to non-contributing elements
                    non_peak = np.ones(n, dtype=bool)
                    non_peak[js:je + 1] = False
                    cnt = non_peak.sum()
                    if cnt > 0:
                        w = f[non_peak]
                        wsum = w.sum()
                        if wsum > 1e-15:
                            trial[non_peak] += d * w / wsum

                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Aggressive: scale down all peak contributors
                k = int(np.argmax(fast_convolve(f)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                trial = f.copy()
                trial[js:je + 1] *= 0.95
                d = f.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc_w = f[non_contrib]
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_contrib] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

    return best_f, best_obj


def lp_fw_step(f, M, n):
    """Single LP-FW step: find direction g via LP."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs > 1e-15:
            return g / gs
    return None


def lp_fw_optimize(f0, time_limit=120.0, verbose=False):
    """LP-Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    _, conv = obj_value(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    m_exp = 1.0
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = lp_fw_step(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8:
                no_improve += 2
            if no_improve > 30:
                break
            continue

        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)
            if no_improve > 20:
                # Perturb
                rng = np.random.default_rng(it * 37 + 1)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                _, conv = obj_value(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_alpha_val = M
        for a in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = f + a * (g - f)
            ft = np.maximum(ft, 0)
            sn = ft.sum()
            if sn < 1e-15:
                continue
            ft = ft / sn
            v = fast_convolve(ft).max()
            if v < best_alpha_val:
                best_alpha_val = v
                best_alpha = a

        if best_alpha < 1e-8:
            no_improve += 1
            continue

        f = f + best_alpha * (g - f)
        f = np.maximum(f, 0)
        f = f / f.sum()
        _, conv = obj_value(f)
        M = conv.max()
        m_exp = 1.0

        obj = 2.0 * n * M
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            break
        if verbose and it % 20 == 0:
            print(f"      LP-FW it={it}: obj={best_obj:.10f}, m_exp={m_exp:.2f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_ramp(n, power=1.0):
    """Monotone ramp: a[i] proportional to (i+1)^power."""
    x = np.arange(n) + 1
    a = x ** power
    a = a / a.sum()
    return a


def make_bimodal(n, center_frac=0.5, width=0.15):
    """Two bumps."""
    x = np.arange(n) / n
    c = center_frac
    w = width
    a = np.exp(-((x - c + 0.25) / w) ** 2) + np.exp(-((x - c - 0.25) / w) ** 2)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate and track best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score
