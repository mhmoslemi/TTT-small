import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_val(a):
    """Objective for normalized sequence (sum=1): 2n*max_conv"""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def make_gaussian(n, sigma_frac):
    x = np.arange(n) - (n-1)/2.0
    a = np.exp(-0.5 * (x / (n * sigma_frac))**2)
    return a / a.sum()

def make_hann(n):
    a = 0.5 * (1 + np.cos(2*np.pi*np.arange(n)/n))
    return a / a.sum()

def make_blackman(n):
    n1 = n - 1
    a = 0.42 - 0.5*np.cos(2*np.pi*np.arange(n)/n1) + 0.08*np.cos(4*np.pi*np.arange(n)/n1)
    a = np.maximum(a, 0)
    return a / a.sum()

def make_flat_taper(n, taper_frac=0.3):
    """Flat middle with tapered edges"""
    a = np.ones(n)
    t = int(n * taper_frac / 2)
    if t > 0:
        ramp = np.linspace(0, 1, t)
        a[:t] = ramp
        a[-t:] = ramp[::-1]
    return a / a.sum()

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result

def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0"""
    A_ub = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        A_ub[k_start:k_end+1, j] = f[:k_end-k_start+1]
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

def multiplicative_step(f, peaks, weights, lr):
    """One multiplicative descent step"""
    n = len(f)
    sg = np.zeros(n)
    for k, w in zip(peaks, weights):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w * f[k - idx]
    f_new = f * np.exp(-lr * sg)
    f_new = np.maximum(f_new, 1e-15)
    f_new /= f_new.sum()
    return f_new

def coordinate_descent(f, n_peaks=5, n_trials=5):

            other = peak_idx - j

                trial = f.copy()
                old = trial[j]
                trial[j] = old * frac

                diff = trial[j] - old  # negative if frac < 1
                # Add to elements that don't contribute to this peak

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive multiplicative descent

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_val(a):
    """Objective for sum-normalized sequence: 2n*max_conv."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def make_gaussian(n, sigma_frac):
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / (n * sigma_frac)) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hann(n):
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
    s = a.sum()
    if s > 1e-15:

    return a

def make_brickwall(n, duty=0.5):
    """Periodic on-off pattern."""
    a = np.zeros(n)
    period = max(2, int(0.5 * n))
    on = max(1, int(period * duty))
    for i in range(n):
        if i % period < on:
            a[i] = 1.0
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

    result = np.interp(new_x, old_x, a)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result


def solve_lp_dir(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        x = np.maximum(res.x, 0.0)
        s = x.sum()
        if s > 1e-15:
            return x / s
    return None


def subgradient_at_peaks(f, peak_indices, peak_weights):
    """Compute the subgradient at given weighted peaks."""
    n = len(f)
    sg = np.zeros(n)
    for ki, wi in zip(peak_indices, peak_weights):
        k = int(ki)
        w = float(wi)
        if w < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w * f[k - idx]
    return sg


def multiplicative_update(f, sg, lr):
    """Apply multiplicative (exponential) update."""
    f_new = f * np.exp(-lr * sg)
    f_new = np.maximum(f_new, 1e-16)
    s = f_new.sum()
    if s < 1e-15:
        return f
    return f_new / s


def do_line_search(f, sg, base_lr):
    """Try multiple step sizes and return the best."""
    best_f = f
    best_obj = np.inf
    lrs = [base_lr * 0.2, base_lr * 0.4, base_lr * 0.7, base_lr, base_lr * 1.5, base_lr * 2.0, base_lr * 3.0]
    for lr in lrs:
        trial = multiplicative_update(f, sg, lr)
        o, _ = obj_val(trial)
        if o < best_obj:
            best_obj = o
            best_f = trial
    return best_f, best_obj


def coordinate_refine(f, n_passes=5, n_top=3):
    """Coordinate descent focusing on the top peak contributors."""
    n = len(f)
    conv = fast_convolve(f)
    obj_cur = 2.0 * n * conv.max()

    for _ in range(n_passes):
        improved = False
        peak_idx = int(np.argmax(conv))
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        if je <= js:
            break

        # Get contributions to peak
        idx_range = np.arange(js, je + 1)
        products = f[idx_range] * f[peak_idx - idx_range]
        total = products.sum()
        if total < 1e-15:
            break

        # Try adjusting top contributors
        order = np.argsort(products)[::-1]
        for oi in range(min(6, len(order))):
            j = int(idx_range[order[oi]])
            old_val = f[j]
            if old_val < 1e-12:
                continue
            for frac in [0.6, 0.75, 0.85, 1.15, 1.3]:
                trial = f.copy()
                delta = trial[j] * (1.0 - frac)
                trial[j] = old_val * frac
                # Redistribute to non-contributing positions
                mask = np.ones(n, dtype=bool)
                mask[js:je + 1] = False
                if mask.any() and abs(delta) > 1e-15:
                    nc = trial[mask]
                    nc_sum = nc.sum()
                    if nc_sum > 1e-15:
                        trial[mask] += (-delta) * nc / nc_sum
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                tconv = fast_convolve(trial)
                tobj = 2.0 * n * tconv.max()
                if tobj < obj_cur - 1e-14:
                    f = trial
                    obj_cur = tobj
                    conv = tconv
                    improved = True

        if not improved:
            # Try reducing the peak region uniformly
            trial = f.copy()
            trial[js:je + 1] *= 0.9
            delta = f.sum() - trial.sum()
            if delta > 0:
                mask = np.ones(n, dtype=bool)
                mask[js:je + 1] = False
                if mask.any():
                    nc = trial[mask]
                    nc_sum = nc.sum()
                    if nc_sum > 1e-15:
                        trial[mask] += delta * nc / nc_sum
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts > 1e-15:
                trial /= ts
                tconv = fast_convolve(trial)
                tobj = 2.0 * n * tconv.max()
                if tobj < obj_cur - 1e-14:
                    f = trial
                    obj_cur = tobj
                    conv = tconv
                else:
                    break
            else:
                break

    return f, obj_cur


def adaptive_multiplicative_optimize(f0, time_limit, verbose=False):
    """Main optimization: adaptive multiplicative descent with LP refresh."""
    f = np.maximum(f0.astype(np.float64), 1e-16)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    temperature = 0.03
    no_improve = 0
    it = 0
    lp_interval = 30
    last_lp_time = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()
        current_obj = 2.0 * n * M

        # Periodic LP direction
        if it % lp_interval == 1 and elapsed > last_lp_time + 2.0 and (time_limit - elapsed) > 60:
            g = solve_lp_dir(f, M * 1.005, n)
            last_lp_time = time.monotonic()
            if g is not None:
                # Frank-Wolfe step with diminishing alpha
                alpha = 2.0 / (it + 2.0)
                alpha = min(alpha, 0.5)
                f_fw = (1.0 - alpha) * f + alpha * g
                f_fw = np.maximum(f_fw, 1e-16)
                f_fw /= f_fw.sum()
                fw_obj, _ = obj_val(f_fw)
                if fw_obj < current_obj:
                    f = f_fw
                    current_obj = fw_obj
                    if current_obj < best_obj - 1e-14:
                        best_obj = current_obj
                        best_f = f.copy()
                    no_improve = 0

        # Find top peaks
        k_max = min(8, len(conv))
        top_vals, top_idx = np.partition(conv, -k_max)[-k_max:], np.argsort(conv)[-k_max:]
        top_vals = conv[top_idx]
        top_idx = top_idx[np.argsort(top_vals)][::-1]

        # Adaptive temperature weighting
        temp = max(0.005, temperature)
        weights = np.exp(-(top_vals - top_vals[0]) / max(temp, 1e-10))
        weights /= weights.sum()

        # Compute subgradient
        sg = subgradient_at_peaks(f, top_idx, weights)

        # Adaptive learning rate based on gradient scale
        sg_max = sg.max()
        if sg_max > 1e-15:
            base_lr = 1.0 / sg_max
        else:
            base_lr = 1.0
        base_lr = min(base_lr, 10.0)
        base_lr = max(base_lr, 0.01)

        # Line search
        f_new, new_obj = do_line_search(f, sg, base_lr)

        if new_obj < current_obj - 1e-15:
            f = f_new
            current_obj = new_obj
            if current_obj < best_obj - 1e-14:
                best_obj = current_obj
                best_f = f.copy()
                no_improve = 0
                temperature = max(0.005, temperature * 0.97)
            else:
                no_improve += 1
                temperature = min(1.0, temperature * 1.01)
        else:
            no_improve += 1
            temperature = min(1.0, temperature * 1.03)

        # Periodic coordinate descent on stagnation
        if no_improve > 40 and verbose and (it % 100 == 0):
            print(f"  iter {it}: obj={best_obj:.8f}, no_improve={no_improve}, temp={temperature:.3f}")

        if no_improve > 80:
            f_co, obj_co = coordinate_refine(f, n_passes=3, n_top=3)
            if obj_co < best_obj - 1e-14:
                best_obj = obj_co
                best_f = f_co.copy()
                f = f_co
            no_improve = 0
            temperature = 0.03

        # Random restart perturbation on long stagnation
        if no_improve > 200:
            rng = np.random.default_rng(42 + it)
            scale = 0.005
            f_pert = f + rng.standard_normal(n) * scale
            f_pert = np.maximum(f_pert, 1e-16)
            f_pert /= f_pert.sum()
            p_obj, _ = obj_val(f_pert)
            if p_obj < best_obj:
                f = f_pert
                best_obj = p_obj
                best_f = f.copy()
            no_improve = 0
            temperature = 0.05

        if it % 500 == 0 and verbose:
            print(f"  iter {it}: best_obj={best_obj:.8f}, temp={temperature:.3f}, elapsed={elapsed:.0f}s")

        # Increase LP interval as we progress (LPs get more expensive relatively)
        if it > 500:
            lp_interval = 40
        if it > 2000:
            lp_interval = 50

        if no_improve > 400:
            break

    return best_f, best_obj


def short_optimize(f0, time_limit):
    """Quick optimization for multi-length exploration."""
    f = np.maximum(f0.astype(np.float64), 1e-16)
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        current_obj = 2.0 * n * M

        peak_idx = int(np.argmax(conv))
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        sg = np.zeros(n)
        sg[idx] = 2.0 * f[peak_idx - idx]

        f_new, new_obj = do_line_search(f, sg, 1.0 / max(sg.max(), 1e-10))
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f_new
        f = f_new

    return best_f


def propose_candidate(budget_s=1800):
    """
    Multi-strategy optimization: adaptive multiplicative descent with LP refresh,
    coordinate descent, multi-length exploration, and diverse initialization.
    """
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score) and p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    # === Phase 1: Diverse initialization ===
    print("=== Phase 1: Diverse Initializations ===")
    inits = []
    n_parent = len(parent)

    # Parent and interpolations
    inits.append(('parent', parent.copy()))
    for n_try in [512, 640, 768, 832, 896]:
        if n_try == n_parent:
            inits.append((f'p_{n_try}', parent.copy()))
        else:
            inits.append((f'p_int_{n_try}', interpolate_seq(parent, n_try)))

    # Various structured initializations
    for n in [512, 640, 768, 896]:
        for sf in [0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
            inits.append((f'g_{n}_{sf}', make_gaussian(n, sf)))
        inits.append((f'hann_{n}', make_hann(n)))
        inits.append((f'black_{n}', make_blackman(n)))
        inits.append((f'ft_{n}_03', make_flat_taper(n, 0.3)))
        inits.append((f'ft_{n}_05', make_flat_taper(n, 0.5)))
        for r in [0.95, 0.97, 0.98, 0.99]:
            inits.append((f'geo_{n}_{r}', make_geometric(n, r)))
        inits.append((f'tri_{n}', make_ramp_symmetric(n)))

    # Evaluate all initializations
    for name, a in inits:
        if time.monotonic() > deadline - 1500:
            break
        s = a.sum()
        if s < 1e-10:
            continue
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()

    print(f"  Phase 1 best: score={best_score:.6f}, n={len(best_seq)}")

    # === Phase 2: Main adaptive multiplicative optimization ===
    print(f"\n=== Phase 2: Main Optimization (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic() - 700
    if rem > 120:
        t_alloc = min(rem, 600)
        print(f"  Allocating {t_alloc:.0f}s for main optimization")
        a_opt, obj_opt = adaptive_multiplicative_optimize(best_seq, time_limit=t_alloc, verbose=True)
        a_score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(a_score) and a_score < best_score:
            best_score = a_score
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {a_score:.6f}")
        else:
            print(f"  No improvement from main opt (got {a_score:.6f})")

    # === Phase 3: Coordinate descent refinement ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Coordinate Descent ===")
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  Allocating {t_alloc:.0f}s")
            f_cd = best_seq.copy()
            t0 = time.monotonic()
            best_cd_obj = 2.0 * len(f_cd) * fast_convolve(f_cd).max()
            while time.monotonic() - t0 < t_alloc:
                f_cd, obj_cd = coordinate_refine(f_cd, n_passes=10, n_top=5)
                if obj_cd < best_cd_obj - 1e-14:
                    best_cd_obj = obj_cd
                else:
                    break
            cd_score = float(evaluate_sequence(f_cd.tolist()))
            if np.isfinite(cd_score) and cd_score < best_score:
                best_score = cd_score
                best_seq = f_cd.copy()
                print(f"  NEW BEST: {cd_score:.6f}")

    # === Phase 4: Multi-length exploration ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Multi-length Exploration ===")
        base = best_seq
        tried_lengths = {len(base)}
        for target_n in [512, 640, 700, 768, 832, 896, 960]:
            if time.monotonic() > deadline - 150:
                break
            if target_n in tried_lengths or abs(target_n - len(base)) < 20:
                continue
            tried_lengths.add(target_n)

            f_interp = interpolate_seq(base, target_n)
            s_score = float(evaluate_sequence(f_interp.tolist()))
            if np.isfinite(s_score) and s_score < best_score:
                best_score = s_score
                best_seq = f_interp.copy()
                print(f"  NEW BEST (interp n={target_n}): {s_score:.6f}")

            # Short optimization
            rem = deadline - time.monotonic() - 150
            if rem > 20:
                t_short = min(rem * 0.3, 40)
                if t_short > 10:
                    f_opt = short_optimize(f_interp, time_limit=t_short)
                    o_score = float(evaluate_sequence(f_opt.tolist()))
                    if np.isfinite(o_score) and o_score < best_score:
                        best_score = o_score
                        best_seq = f_opt.copy()
                        print(f"  NEW BEST (opt n={target_n}): {o_score:.6f}")

    # === Phase 5: Final deep optimization on best ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Final Deep Optimization ===")
        rem = deadline - time.monotonic() - 50
        if rem > 60:
            t_alloc = min(rem, 200)
            print(f"  Allocating {t_alloc:.0f}s")
            a_final, _ = adaptive_multiplicative_optimize(best_seq, time_limit=t_alloc, verbose=False)
            f_score = float(evaluate_sequence(a_final.tolist()))
            if np.isfinite(f_score) and f_score < best_score:
                best_score = f_score
                best_seq = a_final.copy()
                print(f"  NEW BEST: {f_score:.6f}")

            # Additional coordinate descent
            rem2 = deadline - time.monotonic() - 30
            if rem2 > 20:
                t_cd = min(rem2, 80)
                f_cd2 = best_seq.copy()
                t0 = time.monotonic()
                while time.monotonic() - t0 < t_cd:
                    f_cd2, obj_cd2 = coordinate_refine(f_cd2, n_passes=5, n_top=4)
                    if obj_cd2 >= 2.0 * len(f_cd2) * fast_convolve(f_cd2).max():
                        break
                cd2_score = float(evaluate_sequence(f_cd2.tolist()))
                if np.isfinite(cd2_score) and cd2_score < best_score:
                    best_score = cd2_score
                    best_seq = f_cd2.copy()
                    print(f"  NEW BEST (final CD): {cd2_score:.6f}")

    # === Phase 6: Perturbation restart ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Perturbation Restart ===")
        rem = deadline - time.monotonic() - 30
        if rem > 40:
            n_best = len(best_seq)
            rng = np.random.default_rng(123)
            for pert in range(3):
                if time.monotonic() > deadline - 30:
                    break
                t_pert = min(rem / (3 - pert), 50)
                if t_pert < 20:
                    break

                if pert == 0:
                    noise = rng.standard_normal(n_best) * 0.003
                    f_init = np.maximum(best_seq + noise, 1e-16)
                    f_init /= f_init.sum()
                elif pert == 1:
                    # Blend with uniform
                    f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                    f_init /= f_init.sum()
                else:
                    # Reverse-symmetric blend
                    f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                    f_init /= f_init.sum()

                a_p, _ = adaptive_multiplicative_optimize(f_init, time_limit=t_pert, verbose=False)
                p_score = float(evaluate_sequence(a_p.tolist()))
                if np.isfinite(p_score) and p_score < best_score:
                    best_score = p_score
                    best_seq = a_p.copy()
                    print(f"  NEW BEST (pert {pert}): {p_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, elapsed={elapsed:.0f}s)")
    return [float(x) for x in best_seq]
