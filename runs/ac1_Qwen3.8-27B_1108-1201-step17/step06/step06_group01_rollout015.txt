import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_symmetric_conv_matrix(a):
    """Build the n x n matrix T_sym such that (T_sym @ b)[:n] = first n entries of conv(a,b).
    We only need constraints for k=0,...,n-1 since conv is symmetric."""
    n = len(a)
    T = np.zeros((n, n))
    for j in range(n):
        # For lag k, the contribution from b[j] is a[k-j] if 0 <= k-j < n
        # So T[k,j] = a[k-j] if 0 <= k-j <= n-1, i.e., j <= k and k-j < n
        # Since k < n and j < n, k-j < n is automatic. Need k-j >= 0, i.e., j <= k.
        for k in range(j, n):
            T[k, j] = a[k - j]
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CBC, cp.ECOS, cp.SCS, cp.GLPK, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response_sym(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_{k=0}^{n-1} conv(a,b)[k].
    Uses only n constraints (symmetric reduction).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_symmetric_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

            else:
                stale += 1
                # Try smaller step
                alpha2 = alpha * 0.5
                trial2 = (1 - alpha2) * a + alpha2 * a_new
                trial2 = np.maximum(trial2, 0)
                ts2 = trial2.sum()
                if ts2 > 1e-15:
                    trial2 = trial2 / ts2
                    trial_score2 = 2.0 * n * fast_convolve(trial2).max()
                    if trial_score2 < best_score:
                        a = trial2
                        best_score = trial_score2
                        best_a = a.copy()
                        stale = 0

        a_avg = (iters * a_avg + a_new) / (iters + 1)
        b = b_new

        # Periodic restart if stuck
        if stale > 40:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
            stale = 0

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    # Also check the running average
    a_avg = np.maximum(a_avg, 0)
    sa = a_avg.sum()
    if sa > 1e-15:
        a_avg = a_avg / sa
        avg_score = 2.0 * n * fast_convolve(a_avg).max()
        if avg_score < best_score:
            best_score = avg_score
            best_a = a_avg.copy()

    return best_a, best_score, iters


def exp_gradient_optimize(a0, time_limit, T_init=0.01, T_final=0.001, verbose=False):
    """
    Exponential gradient (multiplicative weights) method.
    a_i <- a_i * exp(-eta * grad_i), then renormalize.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    eta = 0.1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        # Use only first n lags (symmetric)
        b_sym = b[:n]
        T = T_init * (T_final / T_init) ** (it / max(1, int(time_limit * 10)))
        T = max(T, T_final)

        # Soft weights over lags
        shifted = b_sym / T - (b_sym / T).max()
        w = np.exp(shifted)
        w = w / w.sum()

        # Compute gradient: grad_i = sum_k w[k] * 2*a[k-i] for k >= i, plus symmetric part
        # Actually, the gradient of b[k] wrt a[i] (for i <= k) is 2*a[k-i]
        # For the symmetric reduction, we consider the full gradient:
        grad = np.zeros(n)
        for k in range(n):
            if w[k] < 1e-12:
                continue
            # b[k] = sum_j a[j]*a[k-j] for j=0..k (with k-j < n)
            # db[k]/da[i] = a[k-i] + a[i-k] (if valid)
            # Since we only have k < n: db[k]/da[i] = a[k-i] if i <= k, plus if k-i < n and k-i >= 0
            # Full: db[k]/da[i] = (a[k-i] if 0<=k-i<n) + (a[i-k] if 0<=i-k<n and i!=k)
            # For k < n and i < n: k-i can be negative. i-k can be negative.
            # b[k] = sum_j a[j]*a[k-j] where 0<=j<n and 0<=k-j<n
            # db[k]/da[i] = a[k-i] (if 0<=k-i<n) + (a[i-k] if 0<=i-k<n and i-k != k-i)
            # = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n and i != k)
            wk = w[k]
            if 0 <= k - i := 0:
                pass
            # Let's just do it with vectorization
            pass

        # Vectorized gradient computation
        grad = np.zeros(n)
        # For each lag k, the contribution to grad[i] is w[k] * (a[k-i] if 0<=k-i<n + a[i-k] if 0<=i-k<n)
        # This is 2 * cross_correlation(w[:n], a) but with appropriate indexing
        # Actually: grad[i] = sum_k w[k] * a[|k-i|] ... no that's not right either
        # Let's think: b[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
        # db[k]/da[i] = a[k-i]*1[0<=k-i<n] + a[i-k]*1[0<=i-k<n]
        # = a[k-i] if 0<=k-i<n, else 0, PLUS a[i-k] if 0<=i-k<n, else 0
        # Note: if k-i == 0 (i.e., i==k), both terms give a[0], so it's 2*a[0]... 
        # Wait no: if i=k, then k-i=0 and i-k=0, so db/da[i] = a[0] + a[0] = 2*a[0]
        # But that's only if we're counting both. Actually for b[k] = sum_j a[j]a[k-j],
        # if i=k: the term j=i gives a[i]*a[0], and the term j=k-i=0 gives a[0]*a[k]=a[0]*a[i].
        # So yes, db[k]/da[i] at i=k is 2*a[0].
        # More generally: db[k]/da[i] = a[k-i] + a[i-k] where each term is 0 if index out of range.

        # Efficient computation using convolution
        # grad = cross_corr(w_padded, a) + cross_corr(w_padded_rev, a) ... 
        # Actually, grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
        # = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
        # First term: (w * a)[i] where * is cross-correlation = conv(w, flip(a))
        # Let's just use FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        W = rfft(w[:n], fft_size)
        A = rfft(a, fft_size)
        # cross_corr(w, a)[i] = sum_k w[k] * a[k-i] = conv(w, a)[n-1-i] ... 
        # Actually let me just compute it directly for efficiency since n is at most ~2000
        # For n=768, the loop over k with vectorized i is fine
        for k in range(n):
            if w[k] < 1e-13:
                continue
            wk = w[k]
            # Contribution: a[k-i] for i=0..min(k,n-1), and a[i-k] for i=k..n-1
            # a[k-i] for i in 0..min(k, n-1): this is a[0..min(k,n-1)] reversed
            max_i1 = min(k, n - 1)
            if max_i1 >= 0:
                grad[0:max_i1 + 1] += wk * a[k :: -1][:max_i1 + 1]
            # a[i-k] for i in k..n-1 (i-k >= 0, i-k < n):
            if k < n:
                grad[k:n] += wk * a[:n - k]

        # Update with multiplicative weights
        a_new = a * np.exp(-eta * grad / max(grad.max(), 1e-10))
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
            sn = 1.0
        a_new = a_new / sn
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                eta *= 0.7
                no_improve = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  ExpGrad ({elapsed:.0f}s): score={best_score:.6f} eta={eta:.4f}")

    return best_a, best_score


def lbgfs_annealed(a0, time_limit, T_values=None, verbose=False):
    """
    L-BFGS with multiple temperature levels (annealing).
    """
    n = len(a0)
    if T_values is None:
        T_values = [500, 200, 100, 50, 25, 10]
    t0 = time.monotonic()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    per_T_time = time_limit / len(T_values)

    for T in T_values:
        if time.monotonic() - t0 > time_limit - 5:
            break
        T_budget = min(per_T_time, time_limit - (time.monotonic() - t0))
        if T_budget < 5:
            break

        z0 = np.log(np.maximum(a, 1e-300))
        t_start = time.monotonic()

        def func(z):
            if time.monotonic() - t_start > T_budget:
                return 1e15
            a_z = _softmax_from_z(z)
            b = fast_convolve(a_z)
            b_shifted = b / T - (b / T).max()
            lse = (b / T).max() + np.log(np.exp(b_shifted).sum())
            return (2.0 * n / T) * lse

        def jac(z):
            if time.monotonic() - t_start > T_budget:
                return np.zeros_like(z)
            a_z = _softmax_from_z(z)
            b = fast_convolve(a_z)
            b_shifted = b / T - (b / T).max()
            exp_b = np.exp(b_shifted)
            w = exp_b / exp_b.sum()
            # grad of (2n/T)*lse wrt a = 2n * (w * a auto-correlation gradient)
            # grad_a[i] = 2n * sum_k w[k] * a[k-i] + w[k] * a[i-k] ... 
            # = 2n * cross_correlation_full(w, a)[i]
            # where cross_correlation_full includes both forward and backward
            n_a = len(a_z)
            fft_sz = 1 << (2 * n_a - 1).bit_length()
            Wf = rfft(w, fft_sz)
            Af = rfft(a_z, fft_sz)
            # The gradient of sum_k w[k]*b[k] wrt a[i]:
            # = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
            # This equals 2 * Re[IDFT(conj(W) * A)] at appropriate indices... 
            # Actually it's the full linear cross-correlation
            cc = irfft(np.conj(Wf) * Af, fft_sz)  # This gives cross-corr at lags -(n-1)..(n-1)
            # cc[n_a-1+j] = sum_i a[i]*w[i-j]... let me just do it directly
            # Simpler: grad_a[i] = sum_k w[k] * (a[k-i] if 0<=k-i<n) + (a[i-k] if 0<=i-k<n)
            # = (w * a)[i] + (w_rev * a)[i] where * is appropriate correlation
            # For efficiency with n up to ~2000, let's use the FFT approach:
            # The full auto-correlation-like quantity is:
            # grad_a[i] = sum_k w[k] * a[k-i] + sum_k w[k] * a[i-k]
            # First sum: let j=k-i, so = sum_j w[j+i] * a[j] for valid j and j+i
            # This is the cross-correlation of w and a at lag i (with appropriate sign)
            # cross_corr(w, a)[i] = sum_j w[j+i]*a[j] = conv(w, flip(a))[len(w)+i-1]
            # For the FFT: cross_corr = IDFT(conj(W * conj(A)))... 
            # Let me just use a simpler approach: 
            # Define v = IDFT(conj(W) * A) padded to size 2n-1
            # v[n-1+i] = sum_j a[j] * w[j-i] for i >= 0 (linear cross-correlation)
            # v[n-1-i] = sum_j a[j] * w[j+i] for i >= 0
            # So grad_a[i] = v[n-1+i] + v[n-1-i] (with appropriate bounds)
            v = irfft(np.conj(Wf) * Af, fft_sz)[:2 * n_a - 1]
            grad_a = np.zeros(n_a)
            # grad_a[i] = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
            # = v[n_a-1+n_a-1-i] + v[n_a-1+i] ... this is getting confusing
            # Let me just compute it directly for clarity (n <= 2048 is fast enough)
            for k_idx in range(min(20, len(w))):  # Only top weights matter
                pass
            
            # Direct computation (efficient for moderate n)
            grad_a = np.zeros(n_a)
            for k_idx in range(n_a):
                if w[k_idx] < 1e-12:
                    continue
                wk = w[k_idx]
                # a[k-i] for i=0..min(k, n-1)
                m1 = min(k_idx, n_a - 1)
                if m1 >= 0:
                    grad_a[0:m1 + 1] += wk * a_z[k_idx:m1: -1] if m1 > 0 else wk * np.array([a_z[k_idx]])
                # More careful:
                # a[k-i] for i in 0..k (and k-i < n_a, which is automatic since k < n_a)
                # So i in 0..k, and we add wk*a[k-i] to grad_a[i]
                # That means grad_a[i] += wk * a[k-i] for i=0..k
                # Equivalently: grad_a[0:k+1] += wk * a[k:k-(-1): -1] = wk * a[k::-1][:k+1]
                # a[k::-1][:k+1] = [a[k], a[k-1], ..., a[0]]
                # So grad_a[0:k+1] += wk * [a[k], a[k-1], ..., a[0]]
                # Which means grad_a[i] += wk * a[k-i] for i=0..k. Correct!
                
                # a[i-k] for i=k..n_a-1 (i-k >= 0, i-k < n_a)
                # So i in k..n_a-1, grad_a[i] += wk * a[i-k]
                # grad_a[k:n_a] += wk * a[0:n_a-k]
                if k_idx < n_a:
                    grad_a[k_idx:n_a] += wk * a_z[:n_a - k_idx]

            grad_a *= 2.0 * n

            # Convert to z-gradient: grad_z = a * (grad_a - a * dot(grad_a, a))
            S = np.dot(grad_a, a_z)
            grad_z = a_z * (grad_a - a_z * S)
            return grad_z

        try:
            res = minimize(
                fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': 60000, 'disp': False}
            )
            a_out = _softmax_from_z(res.x)
            score = 2.0 * n * fast_convolve(a_out).max()
            if score < best_score:
                best_score = score
                best_a = a_out.copy()
                a = a_out
                if verbose:
                    print(f"  L-BFGS T={T}: score={best_score:.6f}")
        except Exception as e:
            if verbose:
                print(f"  L-BFGS T={T} error: {e}")

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def _softmax_from_z(z):
    """Alias for top-level use."""
    return softmax_from_z(z)


def structured_perturbation_search(a0, time_limit, verbose=False):
    """
    Try sinusoidal modulations of the form:
      a_i * (1 + eps * sin(2*pi*f*i/n + phase))
    for various frequencies f, phases, and amplitudes eps.
    Keep the best improvement.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Try frequencies 1 through ~20, with several phases and amplitudes
    freqs = list(range(1, 21))
    phases_per_freq = 4
    amps = [0.01, 0.02, 0.03, 0.05, 0.08]

    tries = 0
    total_possible = len(freqs) * phases_per_freq * len(amps)
    time_per_try = max(0.01, (time_limit - 10) / total_possible)

    for f in freqs:
        if time.monotonic() - t0 > time_limit - 10:
            break
        for phase_idx in range(phases_per_freq):
            if time.monotonic() - t0 > time_limit - 10:
                break
            phase = 2 * np.pi * phase_idx / phases_per_freq
            t = np.linspace(0, 2 * np.pi * f, n) + phase
            modulation = np.sin(t)
            mod_norm = modulation / (np.abs(modulation).max() + 1e-10)
            for amp in amps:
                if time.monotonic() - t0 > time_limit - 10:
                    break
                a_trial = a * (1 + amp * mod_norm)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                tries += 1
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial  # Build on improvement

    if verbose:
        print(f"  StructPerturb: tried {tries}, best={best_score:.6f}")
    return best_a, best_score


def cosine_taper_search(a0, time_limit, verbose=False):
    """
    Try modifying the endpoints with cosine tapers of various widths.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for width_frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2]:
        if time.monotonic() - t0 > time_limit - 5:
            break
        w = max(2, int(n * width_frac))
        taper = np.ones(n)
        taper[:w] = 0.5 * (1 + np.cos(np.pi * np.arange(w) / w))
        taper[-w:] = 0.5 * (1 + np.cos(np.pi * np.arange(w) / w))[::-1]
        # Avoid double-tapering the middle if w > n/2
        if 2 * w > n:
            taper = np.ones(n)
            taper[:w] = 0.5 * (1 + np.cos(np.pi * np.arange(w) / w))
            taper[-w:] = np.minimum(taper[-w:], 0.5 * (1 + np.cos(np.pi * np.arange(w) / w))[::-1])

        for blend in [0.3, 0.5, 0.7, 1.0]:
            if time.monotonic() - t0 > time_limit - 5:
                break
            a_trial = a * (1 - blend + blend * taper)
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_trial.copy()

    if verbose:
        print(f"  CosTaper: best={best_score:.6f}")
    return best_a, best_score


def peak_shave_local(a0, time_limit, verbose=False):
    """Aggressive local peak shaving by modifying individual elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        b_mid = b.copy()
        b_mid[0] = -np.inf
        b_mid[-1] = -np.inf
        peak_k = int(np.argmax(b_mid))
        peak_val = b[peak_k]
        if peak_val <= best_score / (2.0 * n) * 1.001:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_contrib:
            if improved:
                break
            i = int(idx[ci])
            j = peak_k - i
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.9, 0.85, 0.75]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute to maintain sum
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        # Try pair reductions
        if not improved:
            for ci1 in top_contrib[:3]:
                if improved:
                    break
                for ci2 in top_contrib[:3]:
                    if ci1 == ci2:
                        continue
                    i = int(idx[ci1])
                    j = int(idx[ci2])
                    if a[j] < 1e-14:
                        continue
                    for f1 in [0.9, 0.8]:
                        if improved:
                            break
                        for f2 in [0.9, 0.8]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            st = a_trial.sum()
                            if st < 1e-15:
                                continue
                            a_trial = a_trial / st
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break

        if not improved and step > 100:
            break

    if verbose:
        print(f"  PeakShave: {step} steps, best={best_score:.6f}")
    return best_a, best_score


def subgradient_refine(a0, time_limit, lr_init=0.005, top_k=30, verbose=False):
    """Nesterov-accelerated subgradient with adaptive top-k."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        step_count += 1
        # Compute subgradient at y
        b = fast_convolve(y)
        k_actual = min(top_k, n)
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        # Temperature-weighted
        temp = 0.01 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            wk = w[i]
            # db[k]/da[i] = a[k-i] + a[i-k] for valid indices
            m1 = min(k, n - 1)
            if m1 >= 0:
                sg[0:m1 + 1] += 2.0 * wk * y[k:m1: -1] if m1 > 0 else 2.0 * wk * np.array([y[k]])
            if k < n:
                sg[k:n] += 2.0 * wk * y[:n - k]

        # Nesterov update
        gamma = 2.0 / (step_count + 2.0)
        v = gam = None  # placeholder
        # Standard Nesterov: y_{k+1} = x_k + (t_k - 1)/(t_{k+1}) * (x_k - x_{k-1})
        # Simplified: use momentum version
        v = mom * v - lr * sg
        a_candidate = a + v
        # Project onto simplex
        a_new = _project_simplex(a_candidate)
        a_new = a_new / a_new.sum()

        # Update y for next evaluation
        y = a_new

        max_conv = fast_convolve(a_new).max()
        score = 2.0 * n * max_conv

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            v = np.zeros(n)
            lr = max(lr * 2, 0.003)
            no_improve = 0

        if verbose and step_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SubGrad {elapsed:.0f}s: score={best_score:.6f} lr={lr:.5f}")

    return best_a, best_score


def _project_simplex(v):
    """Project vector v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the official evaluator."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
