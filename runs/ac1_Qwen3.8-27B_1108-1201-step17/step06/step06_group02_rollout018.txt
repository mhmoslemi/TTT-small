    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        peaks = np.argpartition(conv, -K_actual)[-K_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = compute_subgradient(f, peaks, w)

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        # Evaluate
        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt parameters
        lr *= lr_decay
        temperature *= 1.002
        K = min(2 * n - 1, K + (1 if it % 200 == 0 else 0))

        # Restart logic
        if no_improve > restart_threshold:
            rng = np.random.default_rng(it * 31 + 7)
            noise_scale = 0.003 * (1 + 0.1 * (no_improve // restart_threshold))
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = 0.015
            temperature = 0.0005
            no_improve = 0
            restart_threshold = min(300, restart_threshold + 20)

        if lr < 1e-9:
            lr = 0.02
            temperature = 0.0003
            no_improve = 0

    return best_f, best_obj


def solve_lp_with_constraints(f, M, n, active_set=None, max_constraints=None):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for k in active_set, g >= 0."""
    if active_set is None:
        active_set = range(2 * n - 1)
    if max_constraints is not None:
        active_set = list(active_set)[:max_constraints]

    n_cons = len(active_set)
    if n_cons == 0:
        return np.ones(n) / n

    c = -np.ones(n)
    A_ub = np.zeros((n_cons, n))
    for idx, k in enumerate(active_set):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[idx, js:je + 1] = f[js:je + 1][::-1]
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """
    LP-Frank-Wolfe with constraint generation and adaptive M relaxation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    active_set = list(range(2 * n - 1))
    m_margin = 1.0
    prev_lp_sol = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3.0:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Sort active constraints by tightness (most violated first)
        npk = 2 * n - 1
        slack = M * m_margin - conv
        # Use top constraints (least slack)
        n_use = min(npk, max(100, n))
        tightest = np.argsort(slack)[:n_use]
        active = list(tightest)

        # Try solving LP with current M
        g = solve_lp_with_constraints(f, M * m_margin, n, active)

        if g is None:
            # Escalate M
            m_margin = min(m_margin * 1.3, 3.0)
            if m_margin > 3.0:
                # Reset with perturbation
                rng = np.random.default_rng(it * 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                m_margin = 1.0
                no_improve += 1
                if no_improve > 40:
                    break
                continue
            continue

        if g.sum() < 1e-12:
            m_margin = min(m_margin * 1.2, 3.0)
            no_improve += 1
            if no_improve > 30:
                break
            continue

        g = g / g.sum()

        # Check if direction is meaningful
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            m_margin = min(m_margin * 1.1, 3.0)
            if no_improve > 50:
                break
            continue
        no_improve = 0
        m_margin = max(1.0, m_margin * 0.95)

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-10:
            no_improve += 1
            m_margin = min(m_margin * 1.15, 3.0)
            continue

        # Fine line search around best_alpha
        if best_alpha == alphas[0]:
            lo, hi = 0.0, alphas[1]
        elif best_alpha == alphas[-1]:
            lo, hi = alphas[-2], 1.0
        else:
            idx = alphas.index(best_alpha)
            lo = alphas[max(0, idx - 1)]
            hi = alphas[min(len(alphas) - 1, idx + 1)]

        for _ in range(10):
            a1 = hi - 0.618 * (hi - lo)
            a2 = lo + 0.618 * (hi - lo)
            ft1 = np.maximum(f + a1 * (g - f), 0.0)
            ft2 = np.maximum(f + a2 * (g - f), 0.0)
            ft1 = ft1 / ft1.sum()
            ft2 = ft2 / ft2.sum()
            v1 = fast_convolve(ft1).max()
            v2 = fast_convolve(ft2).max()
            if v1 < v2:
                hi = a2
            else:
                lo = a1
            mid = (lo + hi) / 2.0
            ftm = np.maximum(f + mid * (g - f), 0.0)
            ftm = ftm / ftm.sum()
            vm = fast_convolve(ftm).max()
            if vm < best_val:
                best_val = vm
                best_alpha = mid

        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        f = f_new
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_polish(f0, time_limit, verbose=False):
    """
    Peak-specific coordinate descent: redistribute mass from peak contributors
    to non-contributors to reduce the maximum convolution value.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 80:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Get top 5 peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]

        found_improvement = False
        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            contrib = np.arange(js, je + 1)
            if len(contrib) == 0:
                continue

            products = f[contrib] * f[pk - contrib]
            total = products.sum()
            if total < 1e-15:
                continue

            # Find top contributor
            top_idx = contrib[np.argmax(products)]
            partner = pk - top_idx

            # Determine source (reduce the larger)
            if f[top_idx] >= f[partner]:
                source = top_idx
            else:
                source = partner

            # Non-contributing indices
            non_contrib_mask = np.ones(n, dtype=bool)
            non_contrib_mask[contrib] = False
            non_contrib_mask[source] = False
            nc_indices = np.where(non_contrib_mask)[0]

            if len(nc_indices) < 2:
                continue

            # Try different transfer amounts
            best_trial_obj = M + 1e-15
            best_trial = None

            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                diff = f[source] * frac
                if diff < 1e-15:
                    continue
                trial = f.copy()
                trial[source] -= diff

                # Distribute proportionally to non-contributing
                nc_vals = trial[nc_indices]
                nc_sum = nc_vals.sum()
                if nc_sum < 1e-15:
                    # Distribute uniformly
                    trial[nc_indices] += diff / len(nc_indices)
                else:
                    trial[nc_indices] += diff * nc_vals / nc_sum

                trial = np.maximum(trial, 0.0)
                trial = trial / trial.sum()

                obj = fast_convolve(trial).max()
                if obj < best_trial_obj:
                    best_trial_obj = obj
                    best_trial = trial

            if best_trial is not None and best_trial_obj < M - 1e-14:
                f = best_trial
                if best_trial_obj < best_obj:
                    best_obj = best_trial_obj
                    best_f = f.copy()
                    no_improve = 0
                    if verbose and no_improve == 0:
                        print(f"    CoordPolish: peak={pk}, obj={best_obj:.10f}")
                found_improvement = True
                break

        if not found_improvement:
            no_improve += 1
            # Try random perturbation
            rng = np.random.default_rng(no_improve * 7 + 3)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 0.0)
            f = f / f.sum()

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hann(n):
    """Hann window."""
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat(n):
    """Uniform sequence."""
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Linearly interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization:
    1. Diverse spectral initializations
    2. Fast subgradient optimization (primary)
    3. LP-Frank-Wolfe polishing (secondary)
    4. Coordinate descent polish (tertiary)
    5. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.3)
        print(f"No parent found, using Gaussian init n=768")

    # === Phase 1: Try diverse initializations (quick eval) ===
    print("\n=== Phase 1: Diverse Initializations ===")
    t0 = time.monotonic()

    init_candidates = []
    n_values = [768, 896, 1024]

    for n_try in n_values:
        init_candidates.append(('gauss', make_gaussian(n_try, 0.25)))
        init_candidates.append(('gauss_w', make_gaussian(n_try, 0.35)))
        init_candidates.append(('bell', make_bell(n_try, 0.5)))
        init_candidates.append(('bell_w', make_bell(n_try, 0.7)))
        init_candidates.append(('hann', make_hann(n_try)))
        init_candidates.append(('cos1', make_cosine_mod(n_try, 1, 0.3)))
        init_candidates.append(('cos2', make_cosine_mod(n_try, 2, 0.2)))
        init_candidates.append(('cos3', make_cosine_mod(n_try, 3, 0.15)))
        init_candidates.append(('flat', make_flat(n_try)))

    if parent is not None:
        init_candidates.append(('parent', parent))

    # Quick evaluate all
    scored = []
    for label, a in init_candidates:
        sc = objective_normalized(a)
        scored.append((sc, label, a))
        if sc < 1.8:
            try_seq(a, f"init_{label}_{len(a)}")

    scored.sort(key=lambda x: x[0])
    print(f"  Best init: {scored[0][1]} n={len(scored[0][2])}, obj={scored[0][0]:.6f}")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Subgradient optimization on top initializations ===
    print("\n=== Phase 2: Subgradient Optimization ===")
    rem = total_deadline - time.monotonic() - 300
    if rem > 60:
        # Allocate time among top initializations
        n_tries = min(4, len(scored))
        t_per = min(rem / n_tries, 200)

        for trial_idx in range(n_tries):
            rem = total_deadline - time.monotonic() - 250
            if rem < 30:
                break
            sc_init, label, a_init = scored[trial_idx]
            t_alloc = min(t_per, rem)
            print(f"  SG [{label}] n={len(a_init)}: alloc={t_alloc:.0f}s (init_obj={sc_init:.6f})")
            a_opt, obj = subgradient_optimize(a_init, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"sg_{label}")

    # === Phase 3: LP-FW polishing on best ===
    print("\n=== Phase 3: LP-FW Polishing ===")
    if best_seq is not None:
        rem = total_deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.7, 250)
            print(f"  LP-FW: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 4: Coordinate polish ===
    print("\n=== Phase 4: Coordinate Polish ===")
    if best_seq is not None:
        rem = total_deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  CoordPolish: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = coordinate_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "coord")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_lengths = [768, 896, 1024, 1280]
        target_lengths = [tn for tn in target_lengths if tn != base_n]

        for target_n in target_lengths:
            rem = total_deadline - time.monotonic() - 60
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = objective_normalized(f_interp)
            print(f"  Interp n={target_n}: obj={sc:.6f}")
            try_seq(f_interp, f"interp_{target_n}")

            if sc < best_score + 0.1:
                # Optimize at this length
                t_alloc = min(rem * 0.4, 80)
                if t_alloc > 15:
                    a_opt, obj = subgradient_optimize(f_interp, time_limit=t_alloc)
                    try_seq(a_opt, f"sg_interp_{target_n}")

    # === Phase 6: Second-pass subgradient on best ===
    print("\n=== Phase 6: Second-Pass Subgradient ===")
    if best_seq is not None:
        rem = total_deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(rem * 0.6, 120)
            print(f"  SG pass 2: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = subgradient_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "sg_pass2")

    # === Phase 7: LP-FW second pass ===
    print("\n=== Phase 7: LP-FW Second Pass ===")
    if best_seq is not None:
        rem = total_deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.7, 100)
            print(f"  LP-FW pass 2: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_pass2")

    # === Phase 8: Final coordinate polish ===
    print("\n=== Phase 8: Final Coordinate Polish ===")
    if best_seq is not None:
        rem = total_deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 30)
            print(f"  Final coord: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_final")

    # === Phase 9: Final quick subgradient squeeze ===
    print("\n=== Phase 9: Final Squeeze ===")
    if best_seq is not None:
        rem = total_deadline - time.monotonic() - 3
        if rem > 5:
            t_alloc = min(rem, 15)
            a_opt, obj = subgradient_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final")

    # Fallback
    if best_seq is None:
        best_seq = make_flat(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback score: {best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - (total_deadline - max(1.0, float(budget_s) - 10.0)):.1f}s")
    return [float(x) for x in best_seq]
