import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_obj_and_grad_simplex(a, tau):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. a (on simplex)."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / tau
    shifted = scaled - scaled.max()
    log_sum_exp = scaled.max() + np.log(np.sum(np.exp(shifted)))
    val = tau * log_sum_exp  # This is the smoothed objective
    
    # For the smoothed objective F = tau * log(sum(exp(conv_k/tau)))
    # dF/da[j] = (2/tau) * sum_k conv_k/tau * exp(conv_k/tau) / sum_l exp(conv_l/tau) * a[k-j]
    #         = 2 * sum_k w_k * a[k-j] where w_k = softmax(conv/tau)
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # h[j] = sum_k w[k] * a[k-j] (with valid index range)
    # This is the cross-correlation of a with w
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    cr = np.fft.irfft(A * W, fft_size)
    h = cr[n-1:2*n-1][::-1]
    
    grad = 2.0 * h
    return val, grad

def smooth_obj_grad_param(z, tau):
    """Objective and gradient w.r.t. parameter z (softmax parameterization)."""
    a = softmax(z)
    val, grad_a = smooth_obj_and_grad_simplex(a, tau)
    # Center the gradient for softmax
    grad_z = a * (grad_a - np.dot(grad_a, a))
    return val, grad_z

        a_new = project_simplex(a + v)

        best_lr = lr
        best_obj = M
        for factor in [1.0, 0.5, 0.25, 0.1]:

            obj = fast_convolve(trial).max()

def peak_equalize(a0, time_limit, verbose=False):
    """
    Peak equalization: identify active peaks and adjust to equalize them.
    Uses iterative water-filling on the active set.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

        it += 1
        conv = fast_convolve(a)

            best_val = M
            best_a = a.copy()
            if verbose and it % 100 == 0:
                print(f"    Peak-eq iter {it}: {2*n*best_val:.8f}")
        
        # Find active peaks (within 1% of max)
        threshold = M * 0.99
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            active = [int(np.argmax(conv))]
        
        # For each active peak, compute the "excess" and redistribute
        # Strategy: reduce contributions to the highest peaks
        peak_vals = conv[active]
        peak_mean = peak_vals.mean()
        
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            products = a[idx] * a[k - idx]
            total = products.sum()
            
            if total < 1e-15:
                continue
            
            # Find the largest contributor
            j_max = idx[np.argmax(products)]
            
            # Reduce a[j_max] slightly and redistribute
            excess = (conv[k] - peak_mean) * 0.1  # Fraction of excess to remove
            if excess <= 0:
                continue
            
            # Reduce a[j_max]
            delta = excess / (a[j_max] + 1e-15)  # How much to reduce
            delta = min(delta, a[j_max] * 0.1)  # Cap at 10% of current value
            a[j_max] -= delta
            
            # Redistribute to positions that don't contribute to active peaks
            mask = np.ones(n, dtype=bool)
            for k2 in active:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                mask[js2:je2+1] = False
            
            if mask.sum() > 0 and a[mask].sum() > 1e-15:
                a[mask] += delta * a[mask] / a[mask].sum()
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a /= s
    
    return best_a, best_val
