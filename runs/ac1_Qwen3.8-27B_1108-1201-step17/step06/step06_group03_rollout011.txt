"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with adaptive
log-sum-exp smoothing, peak equalization heuristics, and multi-length exploration.
Key innovations over previous approach:
1. Adam optimizer with per-coordinate adaptive learning rates
2. "Peak pressure" heuristic that pushes toward equalizing top peaks
3. L-BFGS refinement on smoothed objective
4. Diverse structured restarts (shift, compress, morph)

import numpy as np
from scipy.optimize import fmin_l_bfgs_b, linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smoothed_grad(f, tau, top_frac=0.1):
    """Compute gradient of smoothed max(conv) w.r.t. f."""
    n = len(f)
    conv = fast_convolve(f)
    len_c = len(conv)
    k_top = max(8, int(len_c * top_frac))
    top_idx = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[top_idx]
    
    scaled = tau * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        return conv.max(), np.zeros(n), conv
    
    w /= ws
    smoothed_max = (pv.max() + np.log(ws)) / tau
    
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        if w[i] < 1e-15:
            continue
        k_pos = int(top_idx[i])
        js = max(0, k_pos - n + 1)
        je = min(n - 1, k_pos)
        idx = np.arange(js, je + 1)
        contrib = f[k_pos - idx] + f[idx]
        grad[js:je + 1] += w[i] * contrib
    
    return smoothed_max, grad, conv


def adam_optimize(f0, time_limit, verbose=False,
                  lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                  tau_init=5.0, tau_end=200.0,
                  tau_decay=1.005, restart_every=150,
                  momentum_decay=0.9995):
    """
    Adam optimizer on the nonnegative simplex for minimizing max(conv).
    Uses log-sum-exp smoothing with adaptive temperature.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    t0 = time.monotonic()
    tau = tau_init
    no_improve = 0
    current_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        smoothed_max, grad, conv = compute_smoothed_grad(f, tau)
        M = conv.max()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Multiplicative update (keep nonnegativity)
        update = current_lr * m_hat / np.sqrt(v_hat + eps)
        f_new = f * np.exp(-update)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        f_new = f_new / s
        
        # Check improvement
        new_conv = fast_convolve(f_new)
        new_M = new_conv.max()
        
        if new_M < M * 1.02:  # Accept if not too bad
            f = f_new
            if new_M < best_obj - 1e-12:
                best_obj = new_M
                best_f = f.copy()
                no_improve = 0
                if verbose and t % 50 == 0:
                    print(f"      Adam iter {t}: obj={best_obj:.10f}, tau={tau:.1f}, lr={current_lr:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            current_lr *= 0.8
        
        # Adaptive parameters
        tau *= tau_decay
        current_lr *= momentum_decay
        
        # Restart
        if no_improve > restart_every:
            rng = np.random.default_rng(t * 13 + 7)
            noise = rng.standard_normal(n) * 0.003
            f = np.maximum(best_f + noise, 1e-15)
            f = f / f.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            current_lr = lr * 0.5
            tau = min(tau, tau_init * 5)
            no_improve = 0
        
        if current_lr < 1e-8:
            break
    
    return best_f, best_obj


def peak_equalization(f0, time_limit, verbose=False):
    """
    Iteratively equalize the top peaks of the convolution.
    At each step, identify the top peaks and compute a direction
    that reduces the spread between them.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        k_top = max(5, min(20, len(conv) // 20))
        top_peaks = np.argpartition(conv, -k_top)[-k_top:]
        peak_vals = conv[top_peaks]
        
        # Weight peaks by how far above their "fair share" they are
        # Fair share = mean of all conv values
        fair = conv.mean()
        excess = np.maximum(peak_vals - fair, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break
        w = excess / total_excess
        
        # Compute combined gradient
        grad = np.zeros(n)
        for i in range(len(top_peaks)):
            if w[i] < 1e-12:
                continue
            k_pos = int(top_peaks[i])
            js = max(0, k_pos - n + 1)
            je = min(n - 1, k_pos)
            idx = np.arange(js, je + 1)
            grad[js:je + 1] += w[i] * (f[k_pos - idx] + f[idx])
        
        # Multiplicative descent
        f_new = f * np.exp(-lr * grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        new_conv = fast_convolve(f_new)
        new_M = new_conv.max()
        
        if new_M < M:
            f = f_new
            if new_M < best_obj - 1e-12:
                best_obj = new_M
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.01, 0.08)
                if verbose and it % 50 == 0:
                    print(f"      PE iter {it}: obj={best_obj:.10f}, lr={lr:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            lr *= 0.85
        
        if no_improve > 80:
            rng = np.random.default_rng(it * 23 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 1e-15)
            f = f / f.sum()
            lr = 0.02
            no_improve = 0
        
        if lr < 1e-8:
            break
    
    return best_f, best_obj


def lbfgs_refine(f0, time_limit, verbose=False):
    """Use L-BFGS-B on a smoothed objective for fine refinement."""
    f = np.maximum(f0.astype(np.float64), 1e-10)
    s = f.sum()
    f = f / s
    n = len(f)
    
    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    tau = 20.0
    
    def objective_and_grad(x):
        x = np.maximum(x, 1e-15)
        sx = x.sum()
        if sx < 1e-10:
            return 1e10, np.zeros(n)
        x = x / sx
        conv = fast_convolve(x)
        # Softmax-smoothed max
        scaled = tau * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            return conv.max(), np.zeros(n)
        w /= ws
        obj = (conv.max() + np.log(ws)) / tau
        
        # Gradient
        grad = np.zeros(n)
        for k_pos in range(len(conv)):
            if w[k_pos] < 1e-12:
                continue
            js = max(0, k_pos - n + 1)
            je = min(n - 1, k_pos)
            idx = np.arange(js, je + 1)
            grad[js:je + 1] += w[k_pos] * (x[k_pos - idx] + x[idx])
        
        # Adjust gradient for the normalization (x = x/sx)
        # Actually, since we re-normalize inside, the gradient is approximate
        # But for L-BFGS purposes, this should work
        return obj, grad
    
    # Multiple L-BFGS calls with decreasing tau
    for tau in [10.0, 30.0, 80.0, 200.0]:
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj_grad(x, tau=tau):
            return objective_and_grad(x)
        
        # Constrain to simplex via bound [0, 1]
        res = fmin_l_bfgs_b(obj_grad, f, fprime=None, pgtol=1e-10,
                            maxiter=200, maxfun=500,
                            bounds=[(1e-15, 1.0)] * n,
                            iprint=0)
        
        f_new = np.maximum(res.x, 1e-15)
        f_new = f_new / f_new.sum()
        obj = fast_convolve(f_new).max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            if verbose:
                print(f"      L-BFGS tau={tau}: obj={best_obj:.10f}")
        else:
            f = f_new  # Still use it as starting point
    
    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP for FW direction: max sum(g) s.t. (f*g) <= M, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def make_init(n, variant, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed + variant * 1000)
    x = np.arange(n)
    
    if variant == 0:
        a = np.exp(-x / (n * 0.12))
    elif variant == 1:
        a = np.exp(-x[::-1] / (n * 0.12))
    elif variant == 2:
        a = np.exp(-x / (n * 0.25)) * (1 + 0.4 * np.sin(2 * np.pi * 3 * x / n))
        a = np.maximum(a, 0)
    elif variant == 3:
        a = np.linspace(0.05, 1.0, n)
    elif variant == 4:
        t = x / n
        a = np.exp(-80 * (t - 0.25)**2) + 0.4 * np.exp(-80 * (t - 0.65)**2)
    elif variant == 5:
        a = (1 - x / n) ** 2 + 0.1
    elif variant == 6:
        a = np.exp(-(x - n * rng.uniform(0.2, 0.5))**2 / (2 * (n * 0.15)**2))
        a += 0.05
    elif variant == 7:
        # Power law
        a = (x + 1) ** (-rng.uniform(0.3, 0.8))
    elif variant == 8:
        # Random with smooth structure
        a = np.ones(n)
        a[:n//4] *= rng.uniform(1.0, 2.0)
        a[3*n//4:] *= rng.uniform(0.3, 0.7)
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
    elif variant == 9:
        # S-curve (sigmoid)
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.3)
        a = 1 / (1 + np.exp(-(x - center) / width))
        a = a - a.min() + 0.05
    elif variant == 10:
        # Double exponential
        a = np.exp(-x / (n * 0.08)) + 0.3 * np.exp(-x / (n * 0.4))
    elif variant == 11:
        # Quadratic ramp
        a = (x / n) ** rng.uniform(0.5, 2.0) + 0.05
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage approach:
1. Adam optimizer with log-sum-exp smoothing for fast initial convergence
2. Frank-Wolfe with LP oracle for structured descent
3. Gradient-matching heuristic to flatten the KKT landscape
4. Vectorized coordinate descent for final polish
5. Multi-length exploration with interpolation
Key novelty: "gradient flatness" correction that directly enforces KKT

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(f):
    """Compute the normalized objective (assuming sum=1)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max(), conv


def grad_at_peak(f, k, n):
    """Compute gradient of conv_k w.r.t. f for a single peak position k."""
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if je < js:
        return np.zeros(n), js, je
    idx = np.arange(js, je + 1)
    g = np.zeros(n)
    g[js:je + 1] = f[k - idx] + f[idx]
    return g, js, je


def adam_optimize(f0, time_limit, verbose=False,
                  lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                  tau_init=5.0, tau_decay=1.005,
                  restart_every=120, top_frac=0.1):
    """Adam optimizer with multiplicative update on nonnegative simplex."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    tau = tau_init
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        M = conv.max()
        len_c = len(conv)
        k_top = max(8, int(len_c * top_frac))
        top_idx = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[top_idx]

        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            continue
        w /= ws

        grad = np.zeros(n)
        for i in range(len(top_idx)):
            if w[i] < 1e-14:
                continue
            k_pos = int(top_idx[i])
            js = max(0, k_pos - n + 1)

            idx = np.arange(js, je + 1)
            grad[js:je + 1] += w[i] * (f[k_pos - idx] + f[idx])

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        update = cur_lr * m_hat / np.sqrt(v_hat + eps)
        f_new = f * np.exp(-update)

            else:

            v = np.zeros(n)
            t = 0
            cur_lr = lr * 0.4
            no_improve = 0

        if cur_lr < 1e-9:
            break

    return best_f, best_obj


def lp_fw_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def gradient_flatness_correct(f0, time_limit, verbose=False):
    """
    Directly correct the gradient to be flat on the support.
    At the KKT optimum, the weighted gradient over active peaks should
    be constant where a_j > 0. We enforce this iteratively.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find active peaks (within 1% of max)
        threshold = M * 0.99
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        # Compute uniform-weighted gradient over active peaks
        grad = np.zeros(n)
        for k_pos in active:
            k_pos = int(k_pos)
            js = max(0, k_pos - n + 1)
            je = min(n - 1, k_pos)
            idx = np.arange(js, je + 1)
            grad[js:je + 1] += (f[k_pos - idx] + f[idx])
        grad /= len(active)

        # On the support, the gradient should be constant.
        # Compute the "excess" gradient on the support and push towards flatness.
        support = f > 1e-8
        if support.sum() < 2:
            break
        grad_support = grad[support]
        mean_grad = grad_support.mean()
        # Gradient correction: reduce where grad > mean, increase where grad < mean
        correction = np.zeros(n)
        correction[support] = grad_support - mean_grad

        # Apply multiplicative correction
        f_new = f * np.exp(-lr * correction)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        new_conv = fast_convolve(f_new)
        new_M = new_conv.max()

        if new_M < M:
            f = f_new
            if new_M < best_obj - 1e-13:
                best_obj = new_M
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.03)
                if verbose and it % 20 == 0:
                    print(f"      Flat t={it}: obj={best_obj:.10f}, lr={lr:.5f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            lr *= 0.85

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 19)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.005
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False, n_trials=8):
    """
    Vectorized coordinate descent: for each coordinate, try multiple
    perturbation levels and keep the best. Uses the fact that changing
    one coordinate affects the convolution quadratically.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    total_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        total_sweeps += 1
        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000 + total_sweeps)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Try relative perturbations
            deltas = [old_val * d for d in [-0.4, -0.25, -0.15, -0.08, 0.05, 0.1, 0.2, 0.4]]
            if old_val < 0.01:
                deltas = [0.005, 0.01, 0.02, -0.001, -0.005]

            for new_val in deltas:
                if new_val < 0 or new_val > 0.8:
                    continue
                # Scale others to maintain sum=1
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            # Try adding mass to zero coordinates
            zeros = np.where(f < 1e-10)[0]
            if len(zeros) == 0:
                break
            found = False
            for j in zeros[:min(20, len(zeros))]:
                small_vals = [0.001, 0.002, 0.005, 0.01]
                for new_val in small_vals:
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= (1.0 - new_val)
                    f_trial = np.maximum(f_trial, 0)
                    ss = f_trial.sum()
                    if ss < 1e-15:
                        continue
                    f_trial = f_trial / ss
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
            if not improved:
                break

        if verbose and total_sweeps % 3 == 0:
            print(f"      CD sweep {total_sweeps}: obj={best_obj:.10f}")

    return best_f, best_obj


def make_init(n, variant, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed + variant * 1000)
    x = np.arange(n, dtype=np.float64)

    if variant == 0:
        a = np.exp(-x / (n * 0.1))
    elif variant == 1:
        a = np.exp(-x[::-1] / (n * 0.1))
    elif variant == 2:
        a = np.exp(-x / (n * 0.2)) * (1 + 0.3 * np.sin(2 * np.pi * 2 * x / n))
        a = np.maximum(a, 0)
    elif variant == 3:
        a = np.linspace(0.03, 1.0, n)
    elif variant == 4:
        t = x / n
        a = np.exp(-100 * (t - 0.3)**2) + 0.3 * np.exp(-100 * (t - 0.65)**2)
    elif variant == 5:
        a = (1 - x / n) ** 1.5 + 0.05
    elif variant == 6:
        center = n * rng.uniform(0.2, 0.5)
        width = n * rng.uniform(0.1, 0.2)
        a = np.exp(-(x - center)**2 / (2 * width**2)) + 0.03
    elif variant == 7:
        a = (x + 1) ** (-rng.uniform(0.4, 1.0))
    elif variant == 8:
        a = np.ones(n)
        a[:n // 3] *= rng.uniform(1.2, 2.0)
        a[2 * n // 3:] *= rng.uniform(0.3, 0.6)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
    elif variant == 9:
        center = n * rng.uniform(0.3, 0.6)
        width = n * rng.uniform(0.08, 0.25)
        a = 1.0 / (1.0 + np.exp(-(x - center) / width))
        a = a - a.min() + 0.03
    elif variant == 10:
        a = np.exp(-x / (n * 0.06)) + 0.2 * np.exp(-x / (n * 0.3))
    elif variant == 11:
        a = (x / n) ** rng.uniform(0.3, 1.5) + 0.03
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization: Adam -> FW -> gradient flatness -> coordinate descent.
    Explores multiple lengths and uses structured restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Initialization ===
    print("\n=== Phase 1: Initialization ===")
    lengths = [512, 640, 768, 896]
    all_inits = []

    for n_try in lengths:
        for v in range(12):
            a_init = make_init(n_try, variant=v, seed=seed)
            all_inits.append((a_init, f"v{v}_n{n_try}"))

    if parent is not None:
        all_inits.append((parent, "parent"))

    scored = []
    for a_init, label in all_inits:
        conv = fast_convolve(a_init)
        obj = 2 * len(a_init) * conv.max()
        scored.append((obj, a_init, label))

    scored.sort(key=lambda x: x[0])
    top_inits = scored[:5]
    print(f"  Top: " + ", ".join([f"{l}:{o:.5f}" for o, _, l in top_inits[:3]]))

    for obj, a_init, label in top_inits[:3]:
        try_seq(a_init, f"init_{label}")

    # === Phase 2: Adam on top initializations ===
    print("\n=== Phase 2: Adam ===")
    for idx in range(min(3, len(top_inits))):
        rem = deadline - time.monotonic() - 500
        if rem < 20:
            break
        t_alloc = min(rem / max(1, 3 - idx), 120)
        if t_alloc < 10:
            t_alloc = 10
        _, a_init, label = top_inits[idx]
        print(f"  Adam {label} (n={len(a_init)}): {t_alloc:.0f}s")
        a_opt, _ = adam_optimize(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"adam_{label}")

    # === Phase 3: Frank-Wolfe on best ===
    print("\n=== Phase 3: Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.7, 200)
            if t_alloc < 15:
                t_alloc = 15
            print(f"  FW (n={len(best_seq)}): {t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "fw")

    # === Phase 4: Gradient Flatness ===
    print("\n=== Phase 4: Gradient Flatness ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_alloc = min(rem * 0.5, 100)
            if t_alloc < 10:
                t_alloc = 10
            print(f"  Flat (n={len(best_seq)}): {t_alloc:.0f}s")
            a_opt, _ = gradient_flatness_correct(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "flat")

    # === Phase 5: Second FW pass ===
    print("\n=== Phase 5: FW Pass 2 ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.6, 150)
            if t_alloc < 15:
                t_alloc = 15
            print(f"  FW2 (n={len(best_seq)}): {t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True,
                                   m_exp_init=1.0, max_m_exp=3.0)
            try_seq(a_opt, "fw2")

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length ===")
    if best_seq is not None:
        for target_n in [512, 640, 896, 1024]:
            rem = deadline - time.monotonic() - 120
            if rem < 30:
                break
            if len(best_seq) == target_n:
                continue
            f_interp = interpolate_seq(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.015:
                t_alloc = min(rem * 0.4, 50)
                if t_alloc < 10:
                    t_alloc = 10
                print(f"  FW n={target_n}: {t_alloc:.0f}s")
                a_opt, _ = fw_optimize(f_interp, time_limit=t_alloc)
                try_seq(a_opt, f"fw_n{target_n}")

    # === Phase 7: Coordinate Descent ===
    print("\n=== Phase 7: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem * 0.7, 120)
            if t_alloc < 10:
                t_alloc = 10
            print(f"  CD (n={len(best_seq)}): {t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 8: Final Adam squeeze ===
    print("\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 30)
            print(f"  Final Adam: {t_alloc:.0f}s")
            a_opt, _ = adam_optimize(best_seq, time_limit=t_alloc,
                                     lr=0.003, tau_init=30.0, tau_decay=1.002,
                                     restart_every=40, top_frac=0.05)
            try_seq(a_opt, "final_adam")

    # === Phase 9: Final CD polish ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            t_alloc = min(rem, 15)
            print(f"  Final CD: {t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
