def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_grad(f, tau):
    """Gradient of (1/tau)*log(sum exp(conv/tau))."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1: 2 * n - 1][::-1]
    
    grad = (2.0 / tau) * h
    return grad, conv, w

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage approach:
1. Multi-length diverse initialization
2. Adam smooth optimization with entropy-adaptive temperature
3. Focused subgradient descent targeting top peaks
4. Direct peak-shaving via directional search
5. Final fine-tuning polish
Key innovation: adaptive temperature based on effective peak count,
combined with a "focused gradient" that only considers the binding
constraints, and direct peak-reduction directions.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):
    """Gradient of (1/tau)*log(sum exp(conv/tau)) via FFT."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1: 2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv, w


def focused_gradient(f, top_indices):
    """Compute gradient focused only on top convolution indices."""
    n = len(f)
    w = np.zeros(2 * n - 1)
    for k in top_indices:
        w[k] = 1.0
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_r = np.fft.irfft(A * W, fft_size)
    h = conv_r[n - 1: 2 * n - 1][::-1]
    return 2.0 * h


def peak_direction(f, k_peak):
    """Compute the direction that reduces conv[k_peak]: -2*(P_k f)."""
    n = len(f)
    Pk_f = np.zeros(n)
    # (P_k f)[i] = f[k-i] for 0 <= k-i < n
    i_range = np.arange(n)
    j = k_peak - i_range
    valid = (j >= 0) & (j < n)
    Pk_f[valid] = f[j[valid]]
    return -2.0 * Pk_f


def adam_optimize(f0, time_limit, tau_ratio=0.05, lr_init=0.004, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    tau = tau_ratio * conv.max()

    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    lr = lr_init

    while time.monotonic() - t0 < time_limit:

        M = conv.max()
        eff_peaks = 1.0 / np.sum(w ** 2)

        # Adaptive temperature: target effective peaks in [3, 15]
        if eff_peaks > 15:
            tau *= 0.85
        elif eff_peaks < 3:
            tau *= 1.15
        tau = max(tau, conv.max() * 1e-7)

        # Adam update
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * grad * grad
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new

            no_improve += 1

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, center=0.5, sigma=0.25):
    x = np.arange(n) / n - center
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n, center=0.5, width=0.55):
    x = np.arange(n) / n
    t = np.abs(x - center) / (width / 2)
    a = np.where(t < 1.0, 0.5 * (1.0 + np.cos(np.pi * t)), 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, flat_frac=0.4, edge_frac=0.3):
    x = np.abs(np.arange(n) / n - 0.5)
    half_flat = flat_frac / 2
    a = np.zeros(n)
    mask1 = x <= half_flat
    mask2 = (x > half_flat) & (x <= half_flat + edge_frac)
    a[mask1] = 1.0
    t = (x[mask2] - half_flat) / edge_frac
    a[mask2] = 0.5 * (1.0 + np.cos(np.pi * t))
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

            print(f"  NEW BEST [{label}]: score={score:.6f} (n={len(a_norm)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.5, 0.25)
        try_seq(parent, "default")

    # === Phase 1: Diverse initialization ===
    print("=== Phase 1: Diverse Init ===")
    t1_start = time.monotonic()

    lengths_to_try = [640, 704, 768, 832, 896, 960, 1024, 1152]
    inits = []

    for n_try in lengths_to_try:
        inits.append((make_gaussian(n_try, 0.5, 0.22), f"g_{n_try}"))
        inits.append((make_gaussian(n_try, 0.45, 0.28), f"ga_{n_try}"))
        inits.append((make_raised_cosine(n_try, 0.5, 0.5), f"rc_{n_try}"))
        inits.append((make_flat_top(n_try, 0.35, 0.35), f"ft_{n_try}"))
        a_exp = np.exp(-np.arange(n_try) / (n_try * 0.35))
        a_exp /= a_exp.sum()
        inits.append((a_exp, f"ex_{n_try}"))

    if parent is not None:
        for n_try in lengths_to_try:
            if n_try != len(parent):
                inits.append((interpolate_sequence(parent, n_try), f"ip_{n_try}"))

    for a_init, label in inits:
        if time.monotonic() > deadline - 1400:
            break
        try_seq(a_init, label)

    print(f"  Phase 1 done ({time.monotonic() - t1_start:.1f}s), best={best_score:.6f}")

    work_seq = best_seq.copy()
    work_n = len(work_seq)
    print(f"  Working: n={work_n}, score={best_score:.6f}")

    # === Phase 2: Adam smooth optimization (main) ===
    print(f"\n=== Phase 2: Adam (n={work_n}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 480)
    if t2 > 40:
        a_opt, obj = adam_optimize(work_seq, time_limit=t2 * 0.55,
                                   tau_ratio=0.05, lr_init=0.004, verbose=True)
        try_seq(a_opt, "adam_main")

        if time.monotonic() < deadline - 550:
            rem2 = min(deadline - time.monotonic() - 550, 200)
            a_opt2, obj2 = adam_optimize(best_seq, time_limit=rem2,
                                         tau_ratio=0.03, lr_init=0.002, verbose=True)
            try_seq(a_opt2, "adam_fine2")

    # === Phase 3: Focused subgradient ===
    print(f"\n=== Phase 3: Focused Subgradient ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t3 = min(rem, 200)
            a_opt, obj = focused_subgrad_optimize(best_seq, time_limit=t3,
                                                  top_m=5, verbose=True)
            try_seq(a_opt, "fsubgrad")

            if time.monotonic() < deadline - 200:
                rem2 = min(deadline - time.monotonic() - 200, 100)
                a_opt2, obj2 = focused_subgrad_optimize(best_seq, time_limit=rem2,
                                                        top_m=3, verbose=False)
                try_seq(a_opt2, "fsubgrad3")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 180
        if rem > 20:
            t4 = min(rem, 120)
            a_opt, obj = peak_shave_optimize(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "peakshave")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length ===")
    if best_seq is not None:
        cur_n = len(best_seq)
        nearby_lengths = [cur_n - 64, cur_n + 64, cur_n - 128, cur_n + 128, 896, 1024]
        seen = set()
        for target_n in nearby_lengths:
            if target_n < 256 or target_n > 2048 or target_n in seen:
                continue
            seen.add(target_n)
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"ml_{target_n}")
            t5 = min(rem * 0.25, 60)
            if t5 > 15:
                a_opt, obj = adam_optimize(f_interp, time_limit=t5,
                                           tau_ratio=0.04, lr_init=0.003, verbose=False)
                try_seq(a_opt, f"ml_a_{target_n}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t6 = min(rem, 80)
            a_opt, obj = adam_optimize(best_seq, time_limit=t6 * 0.5,
                                       tau_ratio=0.02, lr_init=0.001, verbose=False)
            try_seq(a_opt, "polish_adam")

        if time.monotonic() < deadline - 10:
            rem2 = min(deadline - time.monotonic() - 10, 40)
            if rem2 > 10:
                a_opt2, obj2 = focused_subgrad_optimize(best_seq, time_limit=rem2,
                                                        top_m=3, verbose=False)
                try_seq(a_opt2, "polish_sub")

        if time.monotonic() < deadline - 5:
            rem3 = min(deadline - time.monotonic() - 5, 20)
            if rem3 > 5:
                a_opt3, obj3 = peak_shave_optimize(best_seq, time_limit=rem3, verbose=False)
                try_seq(a_opt3, "polish_peak")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
