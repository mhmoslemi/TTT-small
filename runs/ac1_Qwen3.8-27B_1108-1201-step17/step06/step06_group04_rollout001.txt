"""

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_super_gaussian(n, p=4.0, width=0.35):

    half_w = n * width / 2.0
    t = np.abs(x) / half_w
    a = np.exp(-t ** p)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:

    return a

    s = a.sum()

    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def multiplicative_descent_adaptive(f0, time_limit, verbose=False, lr_init=0.006, decay=0.9995):
    """
    Adaptive multiplicative descent with backtracking line search and
    multi-peak weighted responsibility.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temperature = 0.003
    it = 0
    k_peaks = min(20, len(conv))

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        # Try multiple step sizes and pick the best (backtracking-like)
        best_trial = None
        best_trial_obj = best_obj
        for lr_try in [lr, lr * 0.5, lr * 2.0, lr * 0.25]:
            trial = f * np.exp(-lr_try * sg)
            trial = np.maximum(trial, 1e-15)
            trial = trial / trial.sum()
            to = 2.0 * n * fast_convolve(trial).max()
            if to < best_trial_obj - 1e-14:
                best_trial_obj = to
                best_trial = trial
                lr = lr_try

        if best_trial is not None:
            f = best_trial
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        lr *= decay
        temperature *= 1.01

        if no_improve > 100:
            rng = np.random.default_rng(it * 37 + 13)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = max(lr_init * 0.7, 1e-6)
            no_improve = 0
            temperature = 0.003

        if no_improve > 300:
            break

        if lr < 1e-9:
            break

    return best_f, best_obj


def build_lp_matrix(f, n):
    """Build the constraint matrix for LP: (f*g)[k] for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n, A_ub=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    if A_ub is None:
        A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M expansion and perturbation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP direction
        A_ub = build_lp_matrix(f, n)
        g = solve_lp_direction(f, M * m_exp, n, A_ub=A_ub)
        if g is None:
            m_exp *= 1.2
            if m_exp > 8.0:
                # Perturbation
                conv_cur = fast_convolve(best_f)
                k_peak = int(np.argmax(conv_cur))
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                f = best_f.copy()
                f[js:je + 1] *= 0.85
                diff = 1.0 - f.sum()
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je + 1] = False
                if non_peak.sum() > 0:
                    w = f[non_peak]
                    wsum = w.sum()
                    if wsum > 1e-15:
                        f[non_peak] += diff * w / wsum
                f = np.maximum(f, 0)
                f = f / f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            if no_improve > 80:
                rng = np.random.default_rng(it * 23 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            if no_improve > 200:
                break
            continue
        else:
            if diff > 1e-5:
                no_improve = 0
            elif diff > 1e-7:
                no_improve += 1
            else:
                no_improve += 3

        # Line search along direction
        improved = False
        best_alpha_val = fast_convolve(f).max()
        for alpha in [0.3, 0.5, 0.7, 1.0, 0.2, 0.8, 0.4, 0.6]:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            cv = fast_convolve(f_new).max()
            if cv < best_alpha_val - 1e-14:
                best_alpha_val = cv
                f = f_new
                M = cv
                m_exp = 1.0
                improved = True

        if improved:
            current_obj = 2.0 * n * M
            if current_obj < best_obj - 1e-15:
                best_obj = current_obj
                best_f = f.copy()
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 6.0)

        if verbose and it % 20 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}, m_exp={m_exp:.2f}")

    return best_f, best_obj


def peak_cd_refine(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent with smarter redistribution.
    Uses top peaks and tries multiple reduction strategies.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 50:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Strategy 1: Reduce top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(3, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.7, 0.8, 0.9]:
                    trial = f.copy()
                    trial[j_max] = old_val * mult
                    d = old_val * (1.0 - mult)

                    # Redistribute to positions NOT contributing to this peak
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[js:je + 1] = False
                    cnt = non_contrib.sum()
                    if cnt > 0:
                        # Distribute proportionally to current values, with boost for low values
                        w = np.maximum(f[non_contrib], 0.01 * f.max())
                        wsum = w.sum()
                        if wsum > 1e-15:
                            trial[non_contrib] += d * w / wsum

                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True

            if improved:
                break  # Restart peak selection

        if not improved:
            no_improve += 1
            # Aggressive: scale down all peak contributors uniformly
            k = int(np.argmax(fast_convolve(f)))
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            for factor in [0.95, 0.9, 0.85]:
                trial = f.copy()
                trial[js:je + 1] *= factor
                d = f.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc_w = np.maximum(f[non_contrib], 0.01)
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_contrib] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    improved = True
                    break

        else:
            no_improve = 0

        if verbose and it % 10 == 0:
            print(f"    Peak CD: obj={best_obj:.10f}, no_improve={no_improve}")

    return best_f, best_obj


def gradient_descent_spectral(f0, time_limit, verbose=False):
    """
    Gradient descent using subgradients of the max convex function.
    At each step, reduces elements that contribute to the top peaks.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Get top peaks
        k_peaks = min(10, len(conv))
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]

        # Compute subgradient: for peak k, grad_j = a_{k-j}
        grad = np.zeros(n)
        for kk in peaks:
            kk = int(kk)
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            # Only add if this peak is near the max
            if conv[kk] > 0.8 * M:
                grad[idx] += f[kk - idx]

        grad /= (k_peaks / 3.0)  # Scale down

        # Projected gradient step
        f_new = f - lr * grad
        f_new = project_simplex(f_new)

        if np.max(np.abs(f_new - f)) < 1e-12:
            lr *= 0.5
            no_improve += 1
        else:
            f = f_new
            obj = 2.0 * n * fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.05)
            else:
                f = best_f + 0.5 * (f_new - best_f)
                f = project_simplex(f)
                no_improve += 1
                lr *= 0.7

        if no_improve > 200:
            rng = np.random.default_rng(it * 41 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            lr = 0.01
            no_improve = 0

        if no_improve > 500 or lr < 1e-8:
            break

    return best_f, best_obj


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence, return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    a = np.minimum(a, 1000.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining parametric initialization,
    adaptive multiplicative descent, LP-Frank-Wolfe, gradient descent,
    peak CD, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "default")

    # === Phase 0: Parametric shape search ===
    print("=== Phase 0: Parametric Shape Search ===")
    n_parent = len(parent)
    candidates = []

    for n_try in [700, 736, 768, 800, 832, 896]:
        if n_try == n_parent and parent is not None:
            candidates.append(parent.copy())
        else:
            if parent is not None:
                candidates.append(interpolate_sequence(parent, n_try))
            candidates.append(make_gaussian(n_try, 0.22))
            candidates.append(make_gaussian(n_try, 0.28))
            candidates.append(make_gaussian(n_try, 0.32))
            candidates.append(make_super_gaussian(n_try, p=3, width=0.4))
            candidates.append(make_super_gaussian(n_try, p=5, width=0.35))
            candidates.append(make_cosine_bell(n_try, 0.75))
            candidates.append(make_cosine_bell(n_try, 0.85))
            candidates.append(make_lanczos(n_try, lobes=2, width=0.85))
            candidates.append(make_piecewise_poly(n_try, 0.8))
            candidates.append(make_bimodal(n_try, separation=0.25, sigma=0.15))
            candidates.append(make_bimodal(n_try, separation=0.35, sigma=0.2))

    best_init = None
    best_init_obj = float('inf')
    for ic in candidates:
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        obj = obj_normalized(a_norm)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        best_seq, best_score = try_sequence(a_norm, best_seq, best_score,
                                             f"init_n{len(a_norm)}")

    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")

    # Also try to optimize the parametric family
    print("  Optimizing parametric family...")
    for n_try in [768, 800]:
        best_param_obj = float('inf')
        best_param_a = None
        for s1 in [0.18, 0.20, 0.22, 0.25, 0.28, 0.30, 0.33, 0.36]:
            a = make_gaussian(n_try, s1)
            obj = obj_normalized(a)
            if obj < best_param_obj:
                best_param_obj = obj
                best_param_a = a
        for p in [2.5, 3, 4, 5, 6]:
            for w in [0.3, 0.35, 0.4, 0.45]:
                a = make_super_gaussian(n_try, p=p, width=w)
                obj = obj_normalized(a)
                if obj < best_param_obj:
                    best_param_obj = obj
                    best_param_a = a
        if best_param_a is not None:
            if best_param_obj < best_init_obj:
                best_init_obj = best_param_obj
                best_init = best_param_a
            best_seq, best_score = try_sequence(best_param_a, best_seq, best_score,
                                                 f"param_n{n_try}")

    # === Phase 1: Fast multiplicative descent ===
    print(f"\n=== Phase 1: Multiplicative Descent (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 180)
        print(f"  MD: t={t_alloc:.0f}s")
        a_opt, obj = multiplicative_descent_adaptive(best_init, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "md_phase1")

    # === Phase 2: Gradient descent ===
    if best_seq is not None:
        print(f"\n=== Phase 2: Gradient Descent (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  GD: t={t_alloc:.0f}s")
            a_opt, obj = gradient_descent_spectral(best_seq, time_limit=t_alloc)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "gd_phase2")

    # === Phase 3: LP-Frank-Wolfe ===
    if best_seq is not None:
        print(f"\n=== Phase 3: LP-FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_alloc = min(rem, 200)
            print(f"  LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw_phase3")

    # === Phase 4: Peak CD ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Peak CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  Peak CD: t={t_alloc:.0f}s")
            a_opt, obj = peak_cd_refine(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "peak_phase4")

    # === Phase 5: Multi-length exploration ===
    base = best_seq if best_seq is not None else best_init
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    for target_n in [720, 784, 832, 896]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break
        f_interp = interpolate_sequence(base, target_n)
        best_seq, best_score = try_sequence(f_interp, best_seq, best_score,
                                             f"interp_n{target_n}")
        t_alloc = min(rem * 0.35, 70)
        if t_alloc < 20:
            continue
        print(f"  MD n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = multiplicative_descent_adaptive(f_interp, time_limit=t_alloc)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, f"md_n{target_n}")

    # === Phase 6: Deep refinement on best ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Deep Refinement (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 150
        if rem > 50:
            t_alloc = min(rem, 180)
            # Run MD again on the current best
            t1 = t_alloc * 0.4
            t2 = t_alloc * 0.35
            t3 = t_alloc * 0.25

            print(f"  Deep MD: t={t1:.0f}s")
            a_opt, obj = multiplicative_descent_adaptive(best_seq, time_limit=max(t1, 5),
                                                          lr_init=0.003)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "deep_md")

            if best_seq is not None:
                rem2 = deadline - time.monotonic() - 80
                if rem2 > 40:
                    t2a = min(t2, rem2 - 40)
                    if t2a > 20:
                        print(f"  Deep LP-FW: t={t2a:.0f}s")
                        a_opt, obj = lp_fw_optimize(best_seq, time_limit=t2a)
                        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "deep_lpfw")

            if best_seq is not None:
                rem3 = deadline - time.monotonic() - 30
                if rem3 > 20:
                    t3a = min(t3, rem3)
                    if t3a > 10:
                        print(f"  Deep Peak CD: t={t3a:.0f}s")
                        a_opt, obj = peak_cd_refine(best_seq, time_limit=t3a)
                        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "deep_peak")

    # === Phase 7: Perturbed restarts ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Perturbed Restarts ===")
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 25:
                break
            t_alloc = min(rem / (3 - p), 60)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                # Symmetrize
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Blend with broader shape
                broad = make_gaussian(n_best, 0.35)
                f_init = 0.7 * best_seq + 0.3 * broad
                f_init = f_init / f_init.sum()

            print(f"  Perturb {p}: t={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent_adaptive(f_init, time_limit=max(t_alloc, 5),
                                                          lr_init=0.005)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, f"perturb_{p}")

    # === Phase 8: Final fast polish ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Final Polish ===")
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 40)
            print(f"  Final MD: t={t_alloc:.0f}s")
            t1 = t_alloc * 0.5
            t2 = t_alloc * 0.3
            t3 = t_alloc * 0.2

            a_opt, obj = multiplicative_descent_adaptive(best_seq, time_limit=max(t1, 3),
                                                          lr_init=0.002)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_md")

            if best_seq is not None:
                rem2 = deadline - time.monotonic() - 5
                if rem2 > 8:
                    a_opt, obj = peak_cd_refine(best_seq, time_limit=max(t2, 3))
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_peak")

            if best_seq is not None:
                rem3 = deadline - time.monotonic() - 2
                if rem3 > 5:
                    a_opt, obj = lp_fw_optimize(best_seq, time_limit=max(t3, 3))
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
