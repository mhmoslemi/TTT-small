```python
import numpy as np
from scipy.optimize import linprog

def solve_lp(a, M):
    n = len(a)
    # Constraint matrix: A[k, j] = a[j] if 0 <= k-j < n, else 0
    # This means: (a * g)_k = sum_j A[k,j] * g[j] <= M
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k-n+1)
        je = min(n-1, k)
        if je >= js:
            A[k, js:je+1] = a[js:je+1]
    
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return result.x
```

```python
    n = len(a)
    A_mat = np.zeros((2*n-1, n))
    # For each j, a[j] * g[k-j] contributes to row k
```

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based iterative optimization.
The main workhorse is a linear programming approach where at each iteration we solve
max sum(g) s.t. (a*g)_k <= M, g >= 0, and blend the solution with the current iterate.
This is complemented by mirror descent and coordinate descent for polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_sparse_constraint_info(a, M, n_active=None):
    """
    Build the list of active constraint indices for the LP.
    Returns indices k where conv[k] is near the peak.
    """
    conv = fast_convolve(a)
    if n_active is None or n_active >= len(conv):
        return np.arange(len(conv))
    # Get top n_active peaks
    top_idx = np.argpartition(conv, -n_active)[-n_active:]
    return top_idx


def solve_lp_for_direction(a, M, n_active=None):
    """
    Solve: max sum(g) s.t. (a*g)_k <= M for active k, g >= 0.
    Returns the optimal g vector (or None if failed).
    """
    n = len(a)
    conv = fast_convolve(a)
    
    if n_active is None or n_active >= 2*n-1:
        active_idx = np.arange(2*n-1)
    else:
        n_active = min(n_active, 2*n-1)
        active_idx = np.argpartition(conv, -n_active)[-n_active:]
    
    m = len(active_idx)
    
    # Build constraint matrix A where A[i, j] gives coefficient of g[j] in constraint i
    # (a*g)_k = sum_j a[j] * g[k-j] for valid j
    # So A[i, j] = a[k-j] if 0 <= k-j < n, where k = active_idx[i]
    A_mat = np.zeros((m, n))
    for i, k in enumerate(active_idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            # g index = k - j, ranges from k-je to k-js
            g_start = k - je
            g_end = k - js
            A_mat[i, g_start:g_end+1] = a[js:je+1][::-1]
    
    b = np.full(m, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x, result.fun
    return None, None

def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent: adjust one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def symmetrize(a):
    """Make sequence palindromic."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return a


def make_fejer(n):
    """Fejér kernel (squared triangular)."""
    center = (n - 1) / 2.0
    x = np.arange(n)
    t = np.abs(x - center) / (center + 0.5)
    a = np.maximum(1.0 - t, 0.0) ** 2
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_optimal_shape(n, alpha=2.0):
    """Generalized bell shape: (1 - |x|^alpha)^beta."""
    center = (n - 1) / 2.0
    x = np.abs(np.arange(n) - center) / center
    a = np.maximum(1.0 - x**alpha, 0.0)**2
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
```

```python
        return result.x
    return None
```

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based iterative optimization
as the main workhorse, complemented by mirror descent and coordinate descent for
polishing. The LP at each iteration finds the best non-negative direction g
maximizing sum(g) subject to (a*g)_k <= M for active peaks, then blends with
the current iterate via line search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(a, active_idx):
    """Build LP constraint matrix for given active constraint indices."""
    n = len(a)
    m = len(active_idx)
    A_mat = np.zeros((m, n))
    for i in range(m):
        k = int(active_idx[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            A_mat[i, k - je:k - js + 1] = a[js:je + 1][::-1]
    return A_mat


def solve_lp_direction(a, M, n_active=None):
    """
    Solve: max sum(g) s.t. (a*g)_k <= M for top n_active peaks, g >= 0.
    Returns optimal g or None on failure.
    """
    n = len(a)
    conv = fast_convolve(a)

    if n_active is None or n_active >= 2 * n - 1:
        active_idx = np.arange(2 * n - 1)
    else:
        n_active = min(n_active, 2 * n - 1)
        active_idx = np.argpartition(conv, -n_active)[-n_active:]

    A_mat = build_constraint_matrix(a, active_idx)
    b = np.full(len(active_idx), M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_iteration(a0, time_limit, n_active_init=80, verbose=False, label=""):
    """
    LP-based iterative optimization: at each step solve LP for best direction,
    blend with current iterate via line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    n_active = n_active_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        g_star = solve_lp_direction(a, M, n_active=n_active)

        if g_star is None:
            n_active = max(20, n_active // 2)
            no_improve += 1
            if no_improve > 100:
                break
            continue

        g_star = np.maximum(g_star, 0.0)
        g_sum = g_star.sum()
        if g_sum < 1e-10:
            break

        g_norm = g_star / g_sum

        # Line search
        best_t = 0.0
        best_trial_obj = M
        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.1]:
            a_trial = (1.0 - t) * a + t * g_norm
            a_trial = np.maximum(a_trial, 0.0)

            a = a / a.sum()
            no_improve = 0

                best_a = a.copy()
                if verbose and it % 50 == 0:

def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def mirror_descent(a0, time_limit, top_k=30, temp_init=0.0005, lr_init=0.01,
                   lr_decay=0.9999, verbose=False, label=""):
    """Multiplicative weight mirror descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = compute_subgradient(a, peak_indices, w)

        best_lr = lr * 0.1
        best_trial_obj = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            if trial_lr < 1e-10:
                continue
            log_update = np.clip(-trial_lr * g, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        log_update = np.clip(-best_lr * g, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        new_obj = fast_convolve(a).max()

        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} MD iter {it}: obj={best_obj:.10f}, lr={best_lr:.6f}")
        else:
            no_improve += 1

        temp = min(temp * 1.002, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * lr_decay, 1e-8)

        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.001 + 0.002 * min(no_improve / 5000, 1.0)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0

        if lr < 1e-9:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent: adjust one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def symmetrize(a):
    """Make sequence palindromic."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return a


def make_fejer(n):
    """Fejér kernel (squared triangular)."""
    center = (n - 1) / 2.0
    x = np.arange(n)
    t = np.abs(x - center) / (center + 0.5)
    a = np.maximum(1.0 - t, 0.0) ** 2
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_shape(n, alpha, beta=2.0):
    """Generalized shape: (1 - |x|^alpha)^beta."""
    center = (n - 1) / 2.0
    x = np.abs(np.arange(n) - center) / center
    a = np.maximum(1.0 - x**alpha, 0.0)**beta
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine(n):
    """Cosine window."""
    a = np.cos(np.pi * np.arange(n) / (n - 1) - np.pi / 2) ** 2
    a[0] = 0
    a[-1] = 0
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def update_best(best_seq, best_score, a, label=""):
    """Evaluate sequence and update best if improved. Returns (best_seq, best_score)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: LP-based iteration (main workhorse),
    mirror descent, and coordinate descent with diverse initializations
    and symmetrization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse Initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    init_candidates = []

    n_values = [768, 896, 1024, 640, 960]
    if parent is not None:
        n_values.insert(0, len(parent))

    for n_try in n_values:
        init_candidates.append((make_gaussian(n_try, 0.25), f"g25_{n_try}"))
        init_candidates.append((make_gaussian(n_try, 0.35), f"g35_{n_try}"))
        init_candidates.append((make_fejer(n_try), f"fj_{n_try}"))
        init_candidates.append((make_bell(n_try, 0.5), f"bl_{n_try}"))
        init_candidates.append((make_shape(n_try, 2.0, 2.0), f"s2_{n_try}"))
        init_candidates.append((make_shape(n_try, 3.0, 2.0), f"s3_{n_try}"))
        init_candidates.append((make_shape(n_try, 4.0, 2.0), f"s4_{n_try}"))
        init_candidates.append((make_cosine(n_try), f"cs_{n_try}"))

    if parent is not None:
        init_candidates.append((parent, "parent"))
        init_candidates.append((symmetrize(parent), "parent_sym"))

    best_init = None
    best_init_obj = float('inf')
    for a_init, lbl in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        best_seq, best_score = update_best(best_seq, best_score, a_norm, f"init_{lbl}")
        conv = fast_convolve(a_norm)
        c_obj = 2 * len(a_norm) * conv.max()
        if c_obj < best_init_obj:
            best_init_obj = c_obj
            best_init = a_norm

    if best_init is None:
        best_init = make_gaussian(768, 0.3)

    print(f"  Best init: 2n*maxconv={best_init_obj:.6f}, n={len(best_init)}")

    # === Phase 2: Quick MD to refine ===
    print(f"\n=== Phase 2: Quick MD ===")
    remaining = deadline - time.monotonic()
    t_md_q = min(80.0, remaining * 0.04)
    if t_md_q > 10:
        a_md, _ = mirror_descent(best_init, time_limit=t_md_q, top_k=30,
                                 temp_init=0.0003, lr_init=0.01,
                                 verbose=False, label="MD_q")
        best_seq, best_score = update_best(best_seq, best_score, a_md, "MD_quick")
        if best_seq is not None and len(best_seq) == len(a_md):
            best_init = a_md

    # === Phase 3: Main LP Iteration ===
    print(f"\n=== Phase 3: LP Main ===")
    remaining = deadline - time.monotonic()
    t_lp_main = remaining * 0.45
    print(f"  Time: {t_lp_main:.0f}s")
    a_lp, _ = lp_iteration(best_init, time_limit=t_lp_main, n_active_init=80,
                           verbose=True, label="LP_main")
    best_seq, best_score = update_best(best_seq, best_score, a_lp, "LP_main")

    # === Phase 4: MD Polish ===
    print(f"\n=== Phase 4: MD Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_md_p = remaining * 0.18
        a_opt, _ = mirror_descent(best_seq, time_limit=t_md_p, top_k=30,
                                  temp_init=0.0002, lr_init=0.008,
                                  verbose=True, label="MD_pol")
        best_seq, best_score = update_best(best_seq, best_score, a_opt, "MD_polish")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coord Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = remaining * 0.08
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd,
                                      verbose=False, label="CD")
        best_seq, best_score = update_best(best_seq, best_score, a_opt, "CD")

    # === Phase 6: LP Pass 2 (symmetrized start, more constraints) ===
    print(f"\n=== Phase 6: LP Pass 2 ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_lp2 = remaining * 0.30
        if t_lp2 > 30:
            a_sym = symmetrize(best_seq)
            best_seq, best_score = update_best(best_seq, best_score, a_sym, "sym_lp2")
            a_opt, _ = lp_iteration(a_sym, time_limit=min(t_lp2 * 0.5, remaining * 0.15),
                                    n_active_init=50, verbose=True, label="LP_sym")
            best_seq, best_score = update_best(best_seq, best_score, a_opt, "LP_sym")

            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_opt, _ = lp_iteration(best_seq, time_limit=min(t_lp2 * 0.4, remaining - 20),
                                        n_active_init=150, verbose=True, label="LP_fin
e")
                best_seq, best_score = update_best(best_seq, best_score, a_opt, "LP_fine")

    # === Phase 7: Multi-restart MD ===
    print(f"\n=== Phase 7: Multi-restart ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_restarts = 3
        t_per = min(remaining * 0.12 / n_restarts, 40)
        if t_per > 10:
            for r in range(n_restarts):
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                if r == 0:
                    f_init = best_seq.copy()
                elif r == 1:
                    noise = rng.standard_normal(len(best_seq)) * 0.001
                    f_init = np.maximum(best_seq + noise, 1e-15)
                    f_init = f_init / f_init.sum()
                else:
                    f_init = 0.95 * best_seq + 0.05 * (np.ones(len(best_seq)) / len(best_seq))
                    f_init = f_init / f_init.sum()

                t_alloc = min(t_per, remaining - 15)
                if t_alloc < 5:
                    break
                a_opt, _ = mirror_descent(f_init, time_limit=t_alloc, top_k=20,
                                          temp_init=0.0001, lr_init=0.005,
                                          verbose=False, label=f"r{r}")
                best_seq, best_score = update_best(best_seq, best_score, a_opt, f"restart_{r}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_pol = min(remaining - 10, 50)
            a_opt, _ = mirror_descent(best_seq, time_limit=t_pol, top_k=10,
                                      temp_init=0.00005, lr_init=0.003,
                                      verbose=True, label="final_pol")
            best_seq, best_score = update_best(best_seq, best_score, a_opt, "final_pol")

            a_s = symmetrize(best_seq)
            best_seq, best_score = update_best(best_seq, best_score, a_s, "final_sym")

            # Try LP one more time on the final result
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                a_opt, _ = lp_iteration(best_seq, time_limit=min(remaining2 - 10, 30),
                                        n_active_init=40, verbose=False, label="final_lp")
                best_seq, best_score = update_best(best_seq, best_score, a_opt, "final_lp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
