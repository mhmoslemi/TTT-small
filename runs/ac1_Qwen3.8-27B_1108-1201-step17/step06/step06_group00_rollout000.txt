import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(a, M, solver='SCIP'):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)(k) <= M for all k.
    Returns the optimal b vector.
    """
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    
    # Build convolution constraints
    # (a*b)(k) = sum_j a(j) * b(k-j) for valid j
    # This is a * b (convolution), which can be written as C @ b where C is the convolution matrix
    
    # Build the convolution matrix (sparse)
    # Row k has: a(j) in column (k-j) for j such that 0 <= k-j <= n-1
    # i.e., j ranges from max(0, k-n+1) to min(k, n-1)
    # Column index is k-j
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(k - j)
            vals.append(a[j])
    
    C = cp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [C @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if b.value is not None:
            return np.asarray(b.value).flatten()
    except Exception as e:
        print(f"  LP solve failed: {e}")

        try:
            prob.solve(solver=alt_solver)
            if b.value is not None:

        except Exception:

    return None

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(a):
    """Build the sparse matrix C such that C @ b = a * b (linear convolution)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(k - j)
            vals.append(a[j])
    import scipy.sparse as sp
    return sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(a, M, solver='SCIP'):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)(k) <= M for all k.
    Returns optimal b or None.
    """
    n = len(a)
    C = build_conv_matrix_sparse(a)
    b = cp.Variable(n, nonneg=True)
    constraints = [C @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    
    for s in [solver] + [s2 for s2 in ['ECOS', 'SCIP', 'GLPK'] if s2 != solver]:
        try:
            prob.solve(solver=s, verbose=False)
            if b.value is not None and np.isfinite(b.value).all():
                return np.asarray(b.value).flatten()
        except Exception:
            continue
    return None


def lp_refine(a0, time_limit, solver='SCIP', blend_rate=0.3):

    LP-based iterative refinement.
    """

            break
        
        b_conv = fast_convolve(a)
        M = b_conv.max()
        
        # Solve LP
        b_opt = lp_direction(a, M, solver=solver)
        if b_opt is None:
            break
        
        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-8:
            # Can't improve
            break
        
        # Normalize to probability vector
        g = b_opt / sum_b
        
        # Line search for best blend
        best_t = 0
        best_M_trial = M
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, blend_rate]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_convolve(a_trial).max()
            if M_trial < best_M_trial:

                best_t = t

            break

        a = np.maximum(a, 0)
        a = a / a.sum()
        
        M_new = fast_convolve(a).max()
        if M_new < best_M - 1e-15:

            best_a = a.copy()

    return best_a, best_M

            best_j_val = old_val

                val = fast_convolve(a_trial).max()

                a[j] = best_j_val

                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
