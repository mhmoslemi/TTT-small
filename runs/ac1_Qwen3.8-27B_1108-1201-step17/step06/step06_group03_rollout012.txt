import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(f):
    """Compute 2n * max(conv(f,f)) for normalized f (sum=1)."""
    n = len(f)
    c = fast_convolve(f)
    return 2.0 * n * c.max()


def parametric_shifted_gaussian(n, a, b, c):
    """
    f[i] = exp(-a*(t-c)^2 - b*(t-c)) where t = i/(n-1)
    a > 0 controls width, b controls asymmetry, c controls center.
    """
    t = np.linspace(0.0, 1.0, n)
    x = t - c
    vals = np.exp(-a * x * x - b * x)
    s = vals.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return vals / s


def parametric_two_sided(n, a_left, a_right, c):
    """
    f[i] = exp(-a_left * (c - t)) for t <= c
    f[i] = exp(-a_right * (t - c)) for t > c
    """
    t = np.linspace(0.0, 1.0, n)
    vals = np.zeros(n)
    mask_left = t <= c
    mask_right = ~mask_left
    vals[mask_left] = np.exp(-a_left * (c - t[mask_left]))
    vals[mask_right] = np.exp(-a_right * (t[mask_right] - c))
    s = vals.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return vals / s


def parametric_power_decay(n, alpha, c):
    """f[i] = |t - c|^(-alpha) smoothed, or (1+|t-c|)^(-alpha)"""
    t = np.linspace(0.0, 1.0, n)
    x = np.abs(t - c) + 0.01
    vals = x ** (-alpha)
    s = vals.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return vals / s


def nesterov_mirror_descent(f0, time_limit, verbose=False,
                            tau_init=10.0, tau_rate=1.01,
                            eta_init=0.02, eta_decay=0.99995,
                            momentum=0.9, restart_every=150):
    """
    Nesterov-accelerated mirror descent (exponential family) for minimizing
    max_k conv(f,f) subject to f >= 0, sum(f) = 1.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # Nesterov momentum: y is the lookahead point
    f_prev = f.copy()
    y = f.copy()

    tau = tau_init
    eta = eta_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    momentum_current = momentum

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute convolution at y (lookahead point)
        conv_y = fast_convolve(y)
        M_y = conv_y.max()

        # Weighted subgradient using top peaks with softmax
        len_c = len(conv_y)
        k_top = max(20, int(len_c * 0.25))
        peaks = np.argpartition(conv_y, -k_top)[-k_top:]
        pv = conv_y[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient at y
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            contrib = y[k - idx] + y[idx]
            # Double count diagonal
            diag_mask = (idx == k - idx)
            contrib[diag_mask] *= 2.0
            sg[js:je + 1] += w[i] * contrib

        # Normalize subgradient
        sg_mean = sg.mean()
        if sg_mean > 1e-15:
            sg = sg / sg_mean

        # Mirror descent step (exponential family)
        y_new = y * np.exp(-eta * sg)
        y_new = np.maximum(y_new, 1e-15)
        y_new = y_new / y_new.sum()

        # Nesterov update
        f_new = (1.0 + momentum_current) * y_new - momentum_current * f_prev

        # Project to simplex (non-negative, sum=1)
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            f_new = y_new.copy()
        else:
            f_new = f_new / sn

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        # Backtracking if objective increased too much
        if obj_new > M_y * 1.1:
            # Dampen
            for damp in [0.5, 0.25, 0.1]:
                f_trial = (1.0 - damp) * f + damp * f_new
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M_y:
                    f_new = f_trial
                    obj_new = obj_trial
                    break
            else:
                eta *= 0.5
                f_prev = f.copy()
                y = f.copy()
                continue

        f_prev = f.copy()
        f = f_new
        y = f_new  # Reset y for next iteration (simplified Nesterov)

        # Track best
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"        NMD iter {it}: obj={best_obj * 2 * n:.10f}, tau={tau:.1f}, eta={eta:.6f}")
        else:
            no_improve += 1

        # Adaptive parameters
        tau *= tau_rate
        eta *= eta_decay

        # Restart with perturbation
        if no_improve > restart_every:
            rng = np.random.default_rng(it * 13 + 7)
            noise_scale = 0.002 + 0.003 * min(no_improve / restart_every, 2.0)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            f_prev = f.copy()
            y = f.copy()
            eta = eta_init * 0.5
            tau = max(tau_init, tau * 0.5)
            no_improve = 0

        if eta < 1e-8:
            break

    return best_f, best_obj


def active_set_refine(f0, time_limit, verbose=False, n_peaks=20):
    """
    Refine by focusing exclusively on the top n_peaks peaks.
    Uses exact subgradient at the argmax with line search.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    eta = 0.005
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Get top peaks
        k_top = max(5, min(n_peaks, 2 * n - 1))
        peaks = np.argsort(conv)[-k_top:]

        # Compute pressure: weighted sum of contributions to top peaks
        pressure = np.zeros(n)
        for k in peaks:
            k = int(k)
            # Weight by how close to max
            weight = (conv[k] / M) ** 4  # Sharp weight on top peaks
            if weight < 0.01:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] + f[idx]
            diag_mask = (idx == k - idx)
            contrib[diag_mask] *= 2.0
            pressure[js:je + 1] += weight * contrib

        # Normalize
        p_mean = pressure.mean()
        if p_mean < 1e-15:
            break
        pressure = pressure / p_mean

        # Multiplicative update
        f_new = f * np.power(pressure, -eta)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        # Line search
        conv_new = fast_convolve(f_new)
        if conv_new.max() < M:
            f = f_new
            if conv_new.max() < best_obj - 1e-14:
                best_obj = conv_new.max()
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"        ASR iter {it}: obj={best_obj * 2 * n:.10f}, eta={eta:.6f}")
            eta = min(eta * 1.01, 0.05)
        else:
            eta *= 0.7
            no_improve += 1

        if no_improve > 200:
            break

    return best_f, best_obj


def multi_scale_optimize(f_init_small, target_n, time_limit):
    """
    Coarse-to-fine: optimize at small scale, upsample, optimize at larger scale.
    """
    f = f_init_small.copy()
    current_n = len(f)
    t0 = time.monotonic()

    scales = []
    n = current_n
    while n < target_n:
        n = min(n * 2, target_n)
        scales.append(n)

    for target_n_s in scales:
        if target_n_s == len(f):
            t_remaining = time_limit - (time.monotonic() - t0)
            if t_remaining < 5:
                break
            a_opt, _ = nesterov_mirror_descent(f, time_limit=t_remaining * 0.7)
            f = a_opt
            continue

        # Upsample by interpolation
        old_x = np.linspace(0, 1, len(f))
        new_x = np.linspace(0, 1, target_n_s)
        f = np.interp(new_x, old_x, f)
        f = np.maximum(f, 0)
        f = f / f.sum()

        t_remaining = time_limit - (time.monotonic() - t0)
        if t_remaining < 5:
            break
        t_alloc = t_remaining * 0.6
        a_opt, _ = nesterov_mirror_descent(f, time_limit=t_alloc)
        f = a_opt

    return f


def coordinate_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find optimal value via ternary search
    on the 1D function (adjusting f[j] while keeping others proportional).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)
        improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-10:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Try a range of new values
            candidates = [old_val * 0.5, old_val * 0.7, old_val * 0.85,
                          old_val * 1.15, old_val * 1.3, old_val * 1.5,
                          old_val + 0.02, old_val - 0.02, 0.0, old_val * 2.0]
            for new_val in candidates:
                if new_val < 0 or new_val > 1.0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-12:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-13:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                    if best_j_obj < best_obj - 1e-13:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True

        if not improved:
            break

    return best_f, best_obj

            t_alloc = min(t_remaining, 120)
            print(f"  ASR on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = active_set_refine(best_seq, time_limit=t_alloc, verbose=True, n_peaks=15)
            try_seq(a_opt, "asr")

            # Second pass with more peaks
            if time.monotonic() < t_phase3_end - 30:
                t_remaining2 = t_phase3_end - time.monotonic()
                t_alloc2 = min(t_remaining2, 80)
                a_opt, obj = active_set_refine(best_seq, time_limit=t_alloc2, verbose=True, n_peaks=30)
                try_seq(a_opt, "asr30")

    # === Phase 4: Multi-scale refinement ===
    print("=== Phase 4: Multi-scale Refinement ===")
    if best_seq is not None:
        t_phase4_end = time.monotonic() + 900
        if time.monotonic() < t_phase4_end:
            # Try different target lengths
            base_n = len(best_seq)
            for target_n in [640, 896, 1024]:
                if time.monotonic() > t_phase4_end - 30:
                    break
                if target_n == base_n:
                    continue
                t_remaining = t_phase4_end - time.monotonic()
                t_alloc = min(t_remaining * 0.4, 80)
                if t_alloc < 15:
                    continue
                print(f"  Multi-scale to n={target_n}: t={t_alloc:.0f}s")
                # Start from best_seq (interpolated to small)
                small_n = min(64, target_n)
                x_old = np.linspace(0, 1, base_n)
                x_small = np.linspace(0, 1, small_n)
                f_small = np.interp(x_small, x_old, best_seq)
                f_small = np.maximum(f_small, 0)
                f_small = f_small / f_small.sum()

                f_large = multi_scale_optimize(f_small, target_n, time_limit=t_alloc)
                try_seq(f_large, f"multiscale_n{target_n}")

    # === Phase 5: More NMD with different parameters ===
    print("=== Phase 5: NMD Parameter Sweep ===")
    if best_seq is not None:
        t_phase5_end = time.monotonic() + 1100
        n_best = len(best_seq)

        param_sets = [
            {"tau_init": 3.0, "tau_rate": 1.005, "eta_init": 0.01, "eta_decay": 0.9999, "momentum": 0.85, "restart_every": 80},
            {"tau_init": 20.0, "tau_rate": 1.015, "eta_init": 0.008, "eta_decay": 0.99993, "momentum": 0.95, "restart_every": 200},
            {"tau_init": 8.0, "tau_rate": 1.01, "eta_init": 0.02, "eta_decay": 0.99996, "momentum": 0.92, "restart_every": 100},
        ]

        for ps_idx, ps in enumerate(param_sets):
            if time.monotonic() > t_phase5_end:
                break
            t_remaining = t_phase5_end - time.monotonic()
            t_alloc = min(t_remaining * 0.4, 60)
            if t_alloc < 10:
                continue

            # Perturb best for diversity
            rng = np.random.default_rng(seed + 100 + ps_idx)
            f_init = np.maximum(best_seq + rng.standard_normal(n_best) * 0.003, 1e-12)
            f_init = f_init / f_init.sum()

            print(f"  NMD sweep {ps_idx} (n={n_best}): t={t_alloc:.0f}s")
            a_opt, obj = nesterov_mirror_descent(f_init, time_limit=t_alloc, verbose=False, **ps)
            try_seq(a_opt, f"nmd_sweep_{ps_idx}")

    # === Phase 6: Coordinate Descent Polish ===
    print("=== Phase 6: Coordinate Polish ===")
    if best_seq is not None:
        t_phase6_end = time.monotonic() + 1350
        if time.monotonic() < t_phase6_end:
            t_remaining = t_phase6_end - time.monotonic()
            t_alloc = min(t_remaining, 120)
            print(f"  CD polish (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = coordinate_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 7: Final Active Set Squeeze ===
    print("=== Phase 7: Final ASR ===")
    if best_seq is not None:
        t_phase7_end = time.monotonic() + 1500
        if time.monotonic() < t_phase7_end:
            t_remaining = t_phase7_end - time.monotonic()
            t_alloc = min(t_remaining, 80)
            print(f"  Final ASR: t={t_alloc:.0f}s")
            a_opt, obj = active_set_refine(best_seq, time_limit=t_alloc, verbose=True, n_peaks=10)
            try_seq(a_opt, "final_asr")

    # === Phase 8: One last NMD with very low learning rate ===
    if best_seq is not None:
        t_phase8_end = time.monotonic() + 1650
        if time.monotonic() < t_phase8_end:
            t_remaining = t_phase8_end - time.monotonic()
            t_alloc = min(t_remaining, 60)
            print(f"  Final NMD (fine): t={t_alloc:.0f}s")
            a_opt, obj = nesterov_mirror_descent(best_seq, time_limit=t_alloc, verbose=False,
                                                 tau_init=50.0, tau_rate=1.002,
                                                 eta_init=0.005, eta_decay=0.9998,
                                                 momentum=0.98, restart_every=60)
            try_seq(a_opt, "final_nmd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
