```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()

def compute_subgradient(f, k_peak):
    """Compute subgradient of (f*f)[k_peak] w.r.t. f."""
    n = len(f)
    sg = np.zeros(n)
    for i in range(n):
        j = k_peak - i
        if 0 <= j < n:
            sg[i] = 2.0 * f[j]
    return sg

def weighted_subgradient(f, peaks, weights):
    """Compute weighted average of subgradients at given peaks."""
    n = len(f)
    sg = np.zeros(n)
    for p, w in zip(peaks, weights):
        pp = int(p)
        for i in range(n):
            j = pp - i
            if 0 <= j < n:
                sg[i] += w * 2.0 * f[j]
    return sg

def mirror_descent_step(f, sg, eta):
    """Multiplicative update: f_new[i] = f[i] * exp(-eta * sg[i]) / Z"""
    n = len(f)
    log_f = np.log(np.maximum(f, 1e-30)) - eta * sg
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s

def peak_shaving(f, k_peak, strength=0.5):
    """Directly reduce the peak at k_peak by redistributing mass."""
    n = len(f)
    f_new = f.copy()
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    idx = np.arange(js, je + 1)
    products = f_new[idx] * f_new[k_peak - idx]
    total = products.sum()
    if total < 1e-15:
        return f_new
    
    # Reduce top contributors
    order = np.argsort(products)[::-1]
    reduced_mass = 0
    for oi in range(min(6, len(order))):
        j = int(idx[order[oi]])
        # Reduce proportionally to its contribution
        frac = strength * (products[order[oi]] / total)
        f_new[j] *= (1 - frac)
        reduced_mass += f[j] * frac
    
    # Redistribute to non-contributing indices
    contrib_mask = np.zeros(n, dtype=bool)
    contrib_mask[js:je+1] = True
    non_contrib = ~contrib_mask
    if non_contrib.sum() > 0:
        nc_w = f_new[non_contrib].copy()
        nc_sum = nc_w.sum()
        if nc_sum > 1e-15:
            f_new[non_contrib] += reduced_mass * nc_w / nc_sum
    
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent with
momentum, periodic LP-Frank-Wolfe refinement, peak-targeted coordinate descent,
and multi-length exploration. Key innovations: adaptive step-size line search,
heavy-ball momentum for accelerated convergence, spectral reweighting of
subgradients, and diverse multi-start strategy.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_step(f, sg, eta):
    """Multiplicative (mirror) descent step on the simplex."""
    log_f = np.log(np.maximum(f, 1e-30)) - eta * sg
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s


def compute_weighted_sg(f, conv, top_k, temperature):
    """Compute weighted subgradient using top peaks."""
    n = len(f)
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    # Softmax weighting
    scaled = (pv - pv.max()) / temperature
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(top_k)
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # sg[i] += w * 2*f[k-i] for i in [js, je]
        f_slice = f[je - (je - js): js + (je - js) + 1][::-1]
        # Simpler: for i in [js, je], f[k-i]
        # k-i ranges from k-js to k-je as i goes from js to je
        f_idx = np.arange(k - js, k - je - 1, -1)
        sg[js:je+1] += w[idx] * 2.0 * f[f_idx]
    return sg


def lp_direction(f, M, n):
    """Solve LP for Frank-Wolfe direction: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        ks = j
        ke = min(j + n - 1, 2 * n - 2)
        A_ub[ks:ke+1, j] = f[:ke - ks + 1]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence centered."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_lorentzian(n, gamma=0.3):
    """Lorentzian (Cauchy) initial sequence."""
    x = np.arange(n) - (n - 1) / 2.0
    t = x / n
    a = 1.0 / (1.0 + (t / gamma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=3.0):
    """Asymmetric exponential decay."""
    x = np.arange(n)
    a = np.exp(-decay * x / n)
    a = a / a.sum()
    return a


def make_cosine_mod(n, beta=0.3):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + beta * np.cos(2 * np.pi * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    top_k = 8

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe for refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        g = lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                # Perturbation
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.05, 5.0)
            if no_improve > 80:
                break
            continue
        
        if diff > 1e-5:
            no_improve = 0
        
        # Line search along (g - f)
        best_alpha = 0.0
        best_val = 2.0 * n * fast_convolve(f).max()
        for alpha in [0.1, 0.25, 0.5, 0.7, 0.85, 1.0]:
            f_try = f + alpha * (g - f)
            f_try = np.maximum(f_try, 0)
            ss = f_try.sum()
            if ss < 1e-15:
                continue
            f_try = f_try / ss
            obj_try = 2.0 * n * fast_convolve(f_try).max()
            if obj_try < best_val:
                best_val = obj_try
                best_alpha = alpha
        
        if best_alpha > 0 and best_val < best_obj - 1e-14:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            f = f_new
            M = fast_convolve(f).max()
            m_exp = 1.0
            best_obj = min(best_obj, 2.0 * n * M)
            if 2.0 * n * M < best_obj:
                best_obj = 2.0 * n * M
                best_f = f.copy()
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 4.0)
        
        if verbose and it % 5 == 0:

def coord_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: try adjusting pairs of coordinates to reduce the peak.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 20:
        conv = fast_convolve(f)
        M = conv.max()
        k_peak = int(np.argmax(conv))
        
        improved = False
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        products = f[idx] * f[k_peak - idx]
        
        # Try reducing top contributors and redistributing
        order = np.argsort(products)[::-1]
        
        for oi in range(min(5, len(order))):
            j = int(idx[order[oi]])
            if f[j] < 1e-12:
                continue
            
            # Try different reduction factors
            for mult in [0.5, 0.7, 0.85, 1.2, 1.5]:
                trial = f.copy()
                old_val = trial[j]
                trial[j] = old_val * mult
                d = old_val - trial[j]
                
                # Redistribute to non-peak indices
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je+1] = False
                if non_peak.sum() > 0 and abs(d) > 1e-15:
                    w = trial[non_peak]
                    ws = w.sum()
                    if ws > 1e-15:
                        trial[non_peak] += d * w / ws
                
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = trial.copy()
                    f = trial
                    improved = True
                    # Reconv
                    conv = fast_convolve(f)
                    M = conv.max()
                    break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
            if verbose and no_improve == 0:
                print(f"    CD: obj={best_obj:.10f}")
    
    return best_f, best_obj
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent with
heavy-ball momentum, multi-scale peak weighting, periodic LP-Frank-Wolfe
refinement, coordinate descent polish, and multi-length/multi-start exploration.
Key innovations: 5-point adaptive line search on mirror steps, gradual
momentum injection, temperature annealing for peak selection, and structured
perturbations (shift, reverse-blend, noise) to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_step(f, sg, eta):
    """Multiplicative (mirror) descent step on the simplex."""
    log_f = np.log(np.maximum(f, 1e-30)) - eta * sg
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s


def compute_weighted_sg(f, conv, top_k, temperature):
    """Compute weighted subgradient using top peaks with softmax weighting."""
    n = len(f)
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / temperature
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(top_k)
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        f_idx = np.arange(k - js, k - je - 1, -1)
        sg[js:je + 1] += w[idx] * 2.0 * f[f_idx]
    return sg


def compute_single_peak_sg(f, k_peak):
    """Subgradient at a single peak location."""
    n = len(f)
    sg = np.zeros(n)
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    f_idx = np.arange(k_peak - js, k_peak - je - 1, -1)
    sg[js:je + 1] = 2.0 * f[f_idx]
    return sg


def lp_direction(f, M, n):
    """Solve LP for Frank-Wolfe direction: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        ks = j
        ke = min(j + n - 1, 2 * n - 2)
        A_ub[ks:ke + 1, j] = f[:ke - ks + 1]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 20, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence centered."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_lorentzian(n, gamma=0.3):
    """Lorentzian (Cauchy) initial sequence."""
    x = np.arange(n) - (n - 1) / 2.0
    t = x / n
    a = 1.0 / (1.0 + (t / gamma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=3.0):
    """Asymmetric exponential decay."""
    x = np.arange(n)
    a = np.exp(-decay * x / n)
    a = a / a.sum()
    return a


def make_cosine_mod(n, beta=0.3):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + beta * np.cos(2 * np.pi * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tapered(n, power=2.0):
    """Tapered (raised cosine power) sequence."""
    x = np.arange(n) / (n - 1)
    a = np.sin(np.pi * x) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def mirror_descent_optimize(f0, time_limit, verbose=False):
    """
    Adaptive mirror descent with heavy-ball momentum and 5-point line search.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    eta = 0.06
    momentum = 0.0
    sg_buf = np.zeros(n)
    no_improve = 0
    it = 0
    temperature = 0.004
    top_k = 6

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        M = conv.max()
        current_obj = 2.0 * n * M

        # Compute weighted subgradient
        sg = compute_weighted_sg(f, conv, top_k, temperature)

        # Momentum blend
        if momentum > 0.01:
            sg_buf = momentum * sg_buf + (1.0 - momentum) * sg
            sg_use = sg_buf
        else:
            sg_use = sg

        # 5-point line search
        best_trial = None
        best_trial_obj = current_obj
        best_eta = eta * 0.5

        for eta_try in [eta * 0.3, eta * 0.6, eta, eta * 1.8, eta * 3.0]:
            if eta_try > 2.5:
                break
            f_try = mirror_step(f, sg_use, eta_try)
            obj_try = 2.0 * n * fast_convolve(f_try).max()
            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_trial = f_try
                best_eta = eta_try

        if best_trial is not None and best_trial_obj < best_obj - 1e-14:
            f = best_trial
            eta = best_eta
            best_obj = best_trial_obj
            best_f = f.copy()
            no_improve = 0
            # Gradually ramp momentum
            if it > 30:
                momentum = min(0.80, momentum + 0.003)
            # Anneal temperature (wider peak selection)
            temperature *= 1.004
            # Increase top_k over time
            top_k = min(16, 4 + it // 80)
        else:
            no_improve += 1
            eta *= 0.75
            momentum *= 0.9

            if no_improve > 40:
                # Structured perturbation to escape local minimum
                rng = np.random.default_rng(it * 17 + 3)
                strategy = it % 4
                if strategy == 0:
                    # Gaussian noise
                    scale = 0.004 + 0.003 * min(no_improve / 60.0, 1.0)
                    f_pert = project_simplex(best_f + rng.standard_normal(n) * scale)
                elif strategy == 1:
                    # Random shift
                    shift = int(rng.integers(1, max(2, n // 6)))
                    f_pert = np.roll(best_f, shift)
                    f_pert = np.maximum(f_pert, 1e-15)
                    f_pert = f_pert / f_pert.sum()
                elif strategy == 2:
                    # Blend with reverse
                    blend = float(rng.uniform(0.1, 0.4))
                    f_pert = (1 - blend) * best_f + blend * best_f[::-1]
                    f_pert = np.maximum(f_pert, 1e-15)
                    f_pert = f_pert / f_pert.sum()
                else:
                    # Blend with uniform
                    blend = float(rng.uniform(0.05, 0.2))
                    f_pert = (1 - blend) * best_f + blend * (np.ones(n) / n)
                    f_pert = f_pert / f_pert.sum()

                f = f_pert
                eta = max(0.02, eta * 0.8)
                momentum = 0.2
                no_improve = 0
                temperature = max(0.002, temperature * 0.6)
                top_k = 6

            if no_improve > 200:
                break

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it}: obj={best_obj:.10f}, eta={eta:.4f}, "
                  f"mom={momentum:.2f}, no_imp={no_improve}, t={elapsed:.0f}s")

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe for refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        g = lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.05, 5.0)
            if no_improve > 80:
                break
            continue

        if diff > 1e-5:
            no_improve = 0

        # Line search along (g - f)
        best_alpha = 0.0
        best_val = 2.0 * n * fast_convolve(f).max()
        for alpha in [0.1, 0.25, 0.5, 0.75, 0.9, 1.0]:
            f_try = f + alpha * (g - f)
            f_try = np.maximum(f_try, 0)
            ss = f_try.sum()
            if ss < 1e-15:
                continue
            f_try = f_try / ss
            obj_try = 2.0 * n * fast_convolve(f_try).max()
            if obj_try < best_val:
                best_val = obj_try
                best_alpha = alpha

        if best_alpha > 0 and best_val < best_obj - 1e-14:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            f = f_new
            new_obj = 2.0 * n * fast_convolve(f).max()
            if new_obj < best_obj:
                best_obj = new_obj
                best_f = f.copy()
            M = fast_convolve(f).max()
            m_exp = 1.0
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 4.0)

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}, t={elapsed:.0f}s")

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: adjust individual coordinates to reduce the peak.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 25:
        conv = fast_convolve(f)
        M = conv.max()
        k_peak = int(np.argmax(conv))

        improved = False
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        products = f[idx] * f[k_peak - idx]

        order = np.argsort(products)[::-1]

        for oi in range(min(6, len(order))):
            j = int(idx[order[oi]])
            if f[j] < 1e-12:
                continue

            for mult in [0.5, 0.65, 0.8, 0.9, 1.1, 1.3, 1.5]:
                trial = f.copy()
                old_val = trial[j]
                trial[j] = old_val * mult
                d = old_val - trial[j]

                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je + 1] = False
                if non_peak.sum() > 0 and abs(d) > 1e-15:
                    w = trial[non_peak]
                    ws = w.sum()
                    if ws > 1e-15:
                        trial[non_peak] += d * w / ws

                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = trial.copy()
                    f = trial
                    improved = True
                    conv = fast_convolve(f)
                    M = conv.max()
                    break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0
            if verbose and no_improve == 0:
                print(f"    CD: obj={best_obj:.10f}")

    return best_f, best_obj


def peak_targeted_md(f0, time_limit, verbose=False):
    """
    Mirror descent that targets ONLY the single worst peak.
    More aggressive than multi-peak version.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    eta = 0.1
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        M = conv.max()
        k_peak = int(np.argmax(conv))
        current_obj = 2.0 * n * M

        # Single peak subgradient
        sg = compute_single_peak_sg(f, k_peak)

        # Try several step sizes
        best_trial = None
        best_trial_obj = current_obj
        for eta_try in [eta * 0.5, eta, eta * 2.0, eta * 4.0]:
            if eta_try > 3.0:
                break
            f_try = mirror_step(f, sg, eta_try)
            obj_try = 2.0 * n * fast_convolve(f_try).max()
            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_trial = f_try
                eta = eta_try

        if best_trial is not None and best_trial_obj < best_obj - 1e-14:
            f = best_trial
            best_obj = best_trial_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.7
            if no_improve > 30:
                rng = np.random.default_rng(it * 23 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                eta = 0.08
                no_improve = 0
            if no_improve > 150:
                break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakMD iter {it}: obj={best_obj:.10f}, t={elapsed:.0f}s")

    return best_f, best_obj


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate and update best if improved."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
