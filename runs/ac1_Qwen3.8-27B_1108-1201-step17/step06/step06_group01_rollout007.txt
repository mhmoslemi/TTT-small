"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Optimistic Mirror Descent (OMD) with LP best responses for fast convergence
2. Active set KKT refinement for exact local optimization
3. Adaptive L-BFGS polishing
4. Multi-start with diverse initializations at multiple sequence lengths
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def solve_lp_best_response(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_k conv(a, b)[k]. Returns (b, M, dual_y).
    dual_y is the dual variable (distribution over lags).
    """
    if solver is None:
        solver = get_working_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix: T @ b <= M
    # T[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # Instead of building the full matrix, use the structure
    # T @ b = conv(a, b)
    
    # For cvxpy, we need to express conv(a, b) <= M
    # conv(a, b)[k] = sum_i a[k-i] * b[i] for k = 0, ..., 2n-2
    # This is a linear function of b.
    
    # Build the convolution matrix
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        T[i:i+n, i] = a_fixed
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    
    M_val = float(M_var.value)
    
    # Get dual variables
    # The constraint T @ b <= M has dual variables y (length 2n-1)
    # We need to extract them from the problem
    try:
        # In cvxpy, dual variables for constraints are accessible via .dual_value
        # The first constraint is T @ b <= M
        y_dual = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
        y_dual = np.maximum(y_dual, 0)
        y_sum = y_dual.sum()
        if y_sum > 1e-15:
            y_dual = y_dual / y_sum
    except:
        y_dual = None
    
    return b_val, M_val, y_dual


def get_working_solver():
    """Find a working LP solver."""
    global _working_solver
    if '_working_solver' in globals() and _working_solver is not None:
        return _working_solver
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.GLPK, cp.PDLP, cp.MOSEK, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x >= 0.1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _working_solver = solver_name
                return solver_name
        except Exception:
            continue
    _working_solver = cp.SCS
    return cp.SCS


def compute_subgradient(b, a, n):
    """
    Compute subgradient of max_k (a*b)[k] with respect to a.
    Returns the subgradient at the argmax lag.
    """
    conv = fast_convolve_cross(a, b)  # cross-correlation a * b
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = b[j]
    return grad, k_star


def fast_convolve_cross(a, b):
    """Linear cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def omd_optimize(n, time_limit, initial=None, solver=None, epsilon=1e-6):
    """
    Optimistic Mirror Descent for the auto-correlation minimization game.
    Uses KL divergence (negative entropy) as the Bregman divergence.
    """
    if solver is None:
        solver = get_working_solver()
    t0 = time.monotonic()
    
    if initial is None:
        a = np.ones(n) / n
        b = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0).copy()
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        b = a.copy()
    
    a_prev = a.copy()
    b_prev = b.copy()
    
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    eta = 0.5  # initial learning rate
    iters = 0
    patience = 0
    max_patience = 200
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Compute best response for b given a
        b_new, M_b, y_b = solve_lp_best_response(a, solver)
        if b_new is None:
            break
        
        # Compute best response for a given b
        a_new, M_a, y_a = solve_lp_best_response(b, solver)
        if a_new is None:
            break
        
        # OMD update for a (minimizer):
        # Use extrapolated opponent: b_extrap = 2*b - b_prev
        # Subgradient of max_k (a * b_extrap)[k] w.r.t. a
        b_extrap = 2.0 * b - b_prev
        b_extrap = np.maximum(b_extrap, 0)
        se = b_extrap.sum()
        if se > 1e-15:
            b_extrap = b_extrap / se
        
        # Subgradient: grad[i] = b_extrap[k*-i]
        conv_check = fast_convolve_cross(a, b_extrap)
        k_star = int(np.argmax(conv_check))
        grad_a = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad_a[i] = b_extrap[j]
        
        # OMD update: a_new_omd[i] = a[i] * exp(-eta * (grad_a[i] + (1/eta_omd)(a[i] - a_prev[i])))
        # Simplified: a_new_omd[i] = a[i] * exp(-eta * grad_a[i] + eta * (a[i] - a_prev[i]))
        # More standard OMD: 
        # W_t = grad_t + (1/eta)(x_t - x_{t-1})
        # x_{t+1} = argmin <W_t, x> + (1/eta) KL(x || x_t)
        # Solution: x_{t+1}[i] = x_t[i] * exp(-eta * W_t[i]) / Z
        #          = x_t[i] * exp(-eta * grad_t[i] - (x_t[i] - x_{t-1}[i])) / Z
        
        W_a = grad_a + (a - a_prev)
        exponent = -eta * W_a
        exponent -= exponent.max()  # for numerical stability
        a_omd = a * np.exp(exponent)
        sa = a_omd.sum()
        if sa < 1e-15:
            a_omd = a.copy()
        else:
            a_omd = a_omd / sa
        
        # OMD update for b (maximizer - use negative gradient):
        a_extrap = 2.0 * a - a_prev
        a_extrap = np.maximum(a_extrap, 0)
        se_a = a_extrap.sum()
        if se_a > 1e-15:
            a_extrap = a_extrap / se_a
        else:
            a_extrap = a.copy()
        
        conv_check_b = fast_convolve_cross(a_extrap, b)
        k_star_b = int(np.argmax(conv_check_b))
        grad_b = np.zeros(n)
        for i in range(n):
            j = k_star_b - i
            if 0 <= j < n:
                grad_b[i] = a_extrap[j]
        
        # For maximizer, we "maximize" so use negative in the exponent
        W_b = -grad_b + (b - b_prev)
        exponent_b = -eta * W_b
        exponent_b -= exponent_b.max()
        b_omd = b * np.exp(exponent_b)
        sb = b_omd.sum()
        if sb < 1e-15:
            b_omd = b.copy()
        else:
            b_omd = b_omd / sb
        
        # Update
        a_prev = a.copy()
        b_prev = b.copy()
        a = a_omd
        b = b_omd
        
        # Evaluate
        max_conv = fast_convolve(a).max()
        score = 2.0 * n * max_conv
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Adaptive learning rate
        if patience > 50:
            eta *= 0.9
        if patience > max_patience:
            # Reset with perturbation
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = a.copy()
            a_prev = a.copy()
            b_prev = b.copy()
            eta = 0.3
            patience = 0
        
        if iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    OMD n={n} iter={iters} ({elapsed:.0f}s): score={best_score:.6f} eta={eta:.4f}")
    
    return best_a, best_score


def active_set_refine(a0, solver=None, max_iters=50):
    """
    Active set KKT refinement.
    At the optimum, for the active set S of lags:
    sum_{k in S} lambda_k * a[k-i] = M for all i (KKT condition)
    sum(a) = 1, a >= 0, lambda >= 0, sum(lambda) = 1
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    for iteration in range(max_iters):
        # Compute auto-correlation
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M
        
        # Identify active set: lags where conv[k] is close to M
        tol = 1e-8 * M + 1e-12
        active_mask = conv >= (M - tol)
        active_lags = np.where(active_mask)[0]
        
        if len(active_lags) == 0:
            break
        
        # Solve LP: find lambda (distribution over active lags) and a
        # such that T_a^T @ lambda = M * 1 (componentwise for active set)
        # Actually: sum_k lambda[k] * a[k-i] = M for all i
        
        # This means: for the active lags, the cross-correlation of lambda and a gives M
        # Let's set up: A_mat @ lambda = M * ones(n) where A_mat[i, k] = a[k-i] (with bounds)
        # But this is an overdetermined system (n equations, len(active_lags) unknowns)
        # Plus sum(lambda) = 1, lambda >= 0
        
        # Alternative: solve for a given lambda
        # If we fix lambda, then: sum_k lambda[k] * a[k-i] = M for all i
        # This is n linear equations in n unknowns (a[0], ..., a[n-1]) plus the constraint sum(a) = 1
        # (which replaces one of the n equations if the system is consistent)
        
        # Better approach: use the LP dual
        # Solve: min M s.t. T(a) @ b <= M, sum(b) = 1, b >= 0
        # Get dual y* (distribution over lags)
        b_star, M_lp, y_star = solve_lp_best_response(a, solver)
        if y_star is None:
            break
        
        # The dual says: min_i sum_k y_star[k] * a[k-i] = M_lp
        # At the optimum, this minimum is achieved at all i in the "active" set (where a[i] > 0)
        
        # Let's try to solve: for all i, sum_k y_star[k] * a[k-i] = M_lp
        # This is a linear system. Let's set it up.
        
        # Define matrix B where B[i, j] = coefficient of a[j] in the i-th equation
        # Equation i: sum_k y_star[k] * a[k-i] = M_lp
        # = sum_j a[j] * y_star[j+i] = M_lp
        # So B[i, j] = y_star[j+i] (if j+i < 2n-1, else 0)
        
        B = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                ki = j + i
                if ki < 2 * n - 1:
                    B[i, j] = y_star[ki]
        
        # Solve B @ a = M_lp * ones(n)
        # But we also need sum(a) = 1, so let's add that as a constraint
        # Replace the last equation with sum(a) = 1
        
        B_mod = B.copy()
        rhs = M_lp * np.ones(n)
        B_mod[-1, :] = 1.0
        rhs[-1] = 1.0
        
        try:
            a_candidate = np.linalg.solve(B_mod, rhs)
        except np.linalg.LinAlgError:
            try:
                a_candidate = np.linalg.lstsq(B_mod, rhs, rcond=None)[0]
            except:
                break
        
        a_candidate = np.maximum(a_candidate, 0)
        sa = a_candidate.sum()
        if sa < 1e-15:
            break
        a_candidate = a_candidate / sa
        
        # Check if this is actually better
        conv_new = fast_convolve(a_candidate)
        M_new = conv_new.max()
        score_new = 2.0 * n * M_new
        
        # Also check KKT: is the min of sum_k y_star[k]*a_candidate[k-i] close to M_new?
        # B_a = B @ a_candidate (with original B, not B_mod)
        Ba = B @ a_candidate
        min_Ba = Ba[a_candidate > 1e-10].min() if (a_candidate > 1e-10).any() else Ba.min()
        kkt_gap = abs(min_Ba - M_new)
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_candidate.copy()
            a = a_candidate
        elif kkt_gap < 1e-10:
            # At the KKT point
            break
        else:
            # Blend
            a = 0.5 * a + 0.5 * a_candidate
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score


def lbgfs_adaptive(a0, time_limit, T_start=1000, T_end=50, maxiter=5000):
    """L-BFGS with adaptive temperature (starts smooth, gets sharper)."""
    n = len(a0)
    from scipy.optimize import minimize
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    z = np.log(np.maximum(a, 1e-300))
    best_z = z.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    def softmax_from_z(z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        return exp_z / exp_z.sum()
    
    def compute_obj_grad(z, T):
        a_local = softmax_from_z(z)
        b = fast_convolve(a_local)
        max_b = b.max()
        exp_Tb = np.exp(T * b - T * max_b)
        log_sum_exp = max_b + np.log(exp_Tb.sum()) / T
        obj = 2.0 * n * log_sum_exp
        
        w = exp_Tb / exp_Tb.sum()
        # grad with respect to a: 2n * cross_corr(w, a)
        # cross_corr(w, a)[i] = sum_k w[k] * a[k-i]
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_local, fft_size)
        cross = irfft(np.conj(W) * A, fft_size)[:n]
        
        grad_a = 2.0 * n * cross
        S = np.dot(grad_a, a_local)
        grad_z = a_local * (grad_a - a_local * S)
        
        return obj, grad_z
    
    # Run L-BFGS with decreasing T
    for T in [T_start, T_start // 2, T_start // 4, T_start // 8, T_end]:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        stop_flag = [False]
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                stop_flag[0] = True
                return 1e15
            obj, _ = compute_obj_grad(z, T)
            return obj
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                stop_flag[0] = True
                return np.zeros_like(z)
            _, g = compute_obj_grad(z, T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': min(maxiter, int(remaining * 50)),
                         'ftol': 1e-14, 'gtol': 1e-10, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        a_current = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a_current).max()
        if score < best_score - 1e-14:
            best_score = score
            best_z = z.copy()
    
    best_a = softmax_from_z(best_z)
    return best_a, best_score


def coordinate_descent_shave(a0, time_limit, max_rounds=10000):
    """
    Coordinate descent that specifically targets the peak.
    For each coordinate in the peak contribution, try reducing it and redistributing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    for _ in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        # Find the peak (prefer interior peaks)
        conv_arr = conv.copy()
        conv_arr[0] = -np.inf
        conv_arr[-1] = -np.inf
        peak_k = int(np.argmax(conv_arr))
        peak_val = conv[peak_k]
        
        if peak_val <= best_score / (2.0 * n) + 1e-12:
            break
        
        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributors
        top = np.argsort(contribs)[::-1][:5]
        improved = False
        
        for ci in top:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            
            # Try different reduction factors
            for factor in [0.98, 0.95, 0.9, 0.85, 0.8]:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * factor
                diff = old_val - new_val
                a_trial[i] = new_val
                
                # Redistribute to other elements proportional to their current values
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if
