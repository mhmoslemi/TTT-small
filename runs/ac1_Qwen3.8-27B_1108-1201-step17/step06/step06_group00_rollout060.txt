"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Nesterov-accelerated subgradient descent with adaptive peak weighting
2. Novel "peak balancing" Newton-like refinement (solves local linear system
   to equalize active peaks)
3. Multi-length exploration with interpolation
4. Coordinate descent micro-polish
5. Perturbation restarts to escape local minima

Key new idea vs previous: The peak balancing step solves a minimum-norm
linear system to directly equalize the active peaks of the auto-convolution,
acting as a second-order correction that converges much faster than pure
subgradient near the optimum.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_value(a):
    """Compute the true objective 2n*max(conv)/(sum)^2 for normalized a."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def subgradient_step(a, v, lr, momentum, b, temp, n_top, rng):
    """One subgradient step with Nesterov momentum and temperature-weighted peaks."""
    n = len(a)
    max_b = b.max()
    
    # Temperature-weighted peak selection
    if temp > 1e-10:
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    
    # Restrict to top peaks for efficiency
    if n_top is not None and n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        else:
            w_sel = np.zeros_like(w_sel)
            w_sel[top_idx[b[top_idx].argmax()]] = 1.0
        w = np.zeros(len(b))
        w[top_idx] = w_sel
    
    # Compute subgradient
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * a[k - idx]
    
    # Nesterov step
    v_new = momentum * v - lr * g
    a_new = project_simplex(a + v_new)
    return a_new, v_new


def peak_balance_refine(a, max_iters=200, tol=1e-10, peak_frac=0.9995):
    """
    Peak balancing refinement: iteratively solve a minimum-norm linear system
    to equalize the active peaks of the auto-convolution.
    
    At each step:
    - Identify active peaks (within peak_frac of max)
    - Solve min ||delta|| s.t. grad(q_k)(a)^T delta = (max_b - b[k]) for active k,
      and sum(delta) = 0
    - Take a damped step and re-normalize
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    
    for it in range(max_iters):
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * peak_frac
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Check convergence: are all active peaks already equal?
        active_vals = b[active]
        spread = active_vals.max() - active_vals.min()
        if spread < tol * max_b:
            break
        
        # Build the linear system:
        # For each active peak k: sum_t 2*a[k-t]*delta[t] = max_b - b[k]
        # Plus: sum_t delta[t] = 0
        m = len(active)
        # Build matrix efficiently using the structure
        A_mat = np.zeros((m + 1, n))
        rhs = np.zeros(m + 1)
        
        for jj, k in enumerate(active):
            # Row jj: A_mat[jj, t] = 2 * a[k-t] if 0 <= k-t < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_mat[jj, lo:hi+1] = 2.0 * a[k - np.arange(lo, hi+1)]
            rhs[jj] = max_b - b[k]
        
        # Sum constraint
        A_mat[-1, :] = 1.0
        rhs[-1] = 0.0
        
        # Minimum norm least squares solution
        try:
            delta, residuals, rank, sv = np.linalg.lstsq(A_mat, rhs, rcond=1e-12)
        except np.linalg.LinAlgError:
            break
        
        delta_norm = np.linalg.norm(delta)
        if delta_norm < 1e-15:
            break
        
        # Damped step: try different step sizes
        best_local = b.max()
        best_delta = delta
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            a_trial = project_simplex(a + alpha * delta)
            b_trial = fast_convolve(a_trial)
            if b_trial.max() < best_local:
                best_local = b_trial.max()
                best_delta = alpha * delta
        
        a = a + best_delta
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    return a, fast_convolve(a).max()


def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.92,
                         temp_init=0.0003, temp_growth=1.015,
                         kick_interval=1500, kick_size=0.002,
                         n_top=None, seed=42, balance_every=300):
    """
    Subgradient optimization with periodic peak-balancing refinements.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    last_balance = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        a_new, v = subgradient_step(a, v, cur_lr, momentum, b, cur_temp, n_top, rng)
        a = a_new
        it_count += 1
        
        b_new = fast_convolve(a)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic adjustments
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.97
        
        cur_temp *= temp_growth
        if cur_temp > 0.3:
            cur_temp = 0.3
        
        # Periodic peak balancing
        if balance_every > 0 and (it_count - last_balance) >= balance_every:
            rem = time_limit - (time.monotonic() - t0)
            if rem > 5:
                a_balanced, obj_balanced = peak_balance_refine(
                    a, max_iters=50, tol=1e-11, peak_frac=0.9998
                )
                if obj_balanced < best_obj:
                    best_obj = obj_balanced
                    best_a = a_balanced.copy()
                a = a_balanced
                v = np.zeros(n)
                no_improve = 0
            last_balance = it_count
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit, n_sweeps=50):
    """Coordinate descent: try scaling each element by various multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_skewed_exp_init(n, rate=1.0, floor=0.01):
    """Exponentially skewed initialization: a[t] ~ exp(rate * t/n) + floor."""
    x = np.arange(n) / n
    a = np.exp(rate * (x - 0.5)) + floor
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_ramp_init(n, sharpen=1.0):
    """Ramp-shaped initialization with adjustable sharpness."""
    x = np.arange(n) / max(n - 1, 1)
    a = x ** sharpen + 0.01
    a = np.maximum(a, 1e-10)
    # Also try reversed
    return a / a.sum()


def make_bimodal_init(n, sep=0.3):
    """Two-bump initialization (asymmetric)."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.35) / 0.15) ** 2) + sep * np.exp(-((x - 0.7) / 0.12) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_sinusoidal_init(n, freq=1.0, phase=0.0):
    """Sinusoidal modulation of a base envelope."""
    x = np.linspace(0, 1, n)
    base = 0.5 + 0.5 * x  # ramp envelope
    mod = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x + phase)
    a = base * mod
    a = np.maximum(a, 1e-10)
    return a / a.sum()

            a_pb, _ = peak_balance_refine(a_opt, max_iters=50, tol=1e-12, peak_frac=0.9999)
            try_seq(a_pb, f"perturb_{attempt}")
    
    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final micro-polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_cd = min(rem * 0.5, 40.0)
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "final_cd")
            
            rem2 = deadline - time.monotonic() - 10
            if rem2 > 5:
                a_pb, _ = peak_balance_refine(a_cd, max_iters=100, tol=1e-14, peak_frac=0.99999)
                try_seq(a_pb, "final_pb")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        fs = best_score
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
