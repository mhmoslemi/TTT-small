Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L2-flattening gradient descent (quartic objective) with FFT-based gradients
2. L-infinity mirror descent for final polishing
3. Multi-scale and multi-start strategies

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def conv_correlation(c, a):
    """Compute r_i = sum_k c_k * a_{k-i} for i=0,...,n-1 using FFT."""
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    C = np.fft.rfft(c, fft_size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(C * np.conj(A), fft_size)[:n]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

    Gauss-Newton step for the problem: find delta_a minimizing ||A @ delta_a + e||^2
    where A is the convolution matrix.
    Solution: delta_a = -(A^T A)^{-1} A^T e

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Gauss-Newton iterations (L2 flattening of auto-convolution)
2. Adaptive mirror descent (L∞ peak reduction)

import numpy as np
import time
import scipy.linalg

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym

def gauss_newton_flatten(a0, time_limit, verbose=False, label=""):

        a = a / s
    n = len(a)
    target = 1.0 / (2 * n - 1)

    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:

        conv = fast_convolve(a)

        auto = np.fft.irfft(A_fft * np.conj(A_fft), fft_size)
        
        idx = np.abs(np.arange(n)[:, None] - np.arange(n)[None, :])
