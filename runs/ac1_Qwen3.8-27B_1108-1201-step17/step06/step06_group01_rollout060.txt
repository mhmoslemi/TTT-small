Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play with warm starts for fast game equilibrium finding.
2. Smooth max (log-sum-exp) gradient polishing via L-BFGS in softmax parameterization.
3. FFT-structured Newton steps for final convergence.
4. Multi-length exploration and interpolation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.XPRESS, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return _lp_solver

def solve_lb_best_response(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_k conv(a, b)[k].
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:

    """
    Solve the KKT system using Newton's method.
    a0: initial guess for a (non-negative, sum=1)
    active_lags: array of active lag indices
    Returns: (a, lambda, nu, M, success)
    """

    m = len(active_lags)

    a = a0.copy()
    R = fast_convolve(a)
    M = R[active_lags].max()

    # For m < n, the gradient condition overdetermines lambda.
    # Let's just start with uniform lambda.
    lam = np.ones(m) / m

        R = fast_convolve(a)
        F1 = R[active_lags] - M  # m residuals
        L = np.zeros(n)

        F4 = lam.sum() - 1.0  # 1 residual

            J[m + valid_i, n + j] = 2.0 * a[k - valid_i]

        J[m:m+n, n+m+1] = -1.0  # Wait, column index for dnu is n+m, not n+m+1
        # Let me fix the column indexing:
        # Columns: da[0..n-1], dlam[0..m-1], dnu, dM
        # So dnu is at column n+m, dM is at column n+m+1
        
        # Actually I made an error. Let me redo:

        # Column indices:
        #   da: 0 to n-1

            valid_i = np.arange(max(0, k-n+1), min(n, k+1))
            J[j, valid_i] = 2.0 * a[k - valid_i]
        
        # Block (F1, dM): rows 0..m-1, col n+m+1
        J[:m, n+m+1] = -1.0
        
        # Block (F2, da): rows m..m+n-1, cols 0..n-1
        for j, k in enumerate(active_lags):
            if abs(lam[j]) < 1e-15:
                continue
            for i in range(n):
                l = k - i
                if 0 <= l < n:
                    J[m + i, l] += 2.0 * lam[j]
        
        # Block (F2, dlam): rows m..m+n-1, cols n..n+m-1
        for j, k in enumerate(active_lags):
            valid_i = np.arange(max(0, k-n+1), min(n, k+1))
            J[m + valid_i, n + j] = 2.0 * a[k - valid_i]
        
        # Block (F2, dnu): rows m
