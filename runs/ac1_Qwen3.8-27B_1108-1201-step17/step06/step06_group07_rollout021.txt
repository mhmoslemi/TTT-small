def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

gradient descent with adaptive peak weighting, structural perturbation restarts,
multi-length search, and thorough coordinate descent polishing.
Key differences from previous approach: uses additive Nesterov method with

perturbations for restarts, and frequency-domain perturbations.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def nesterov_smoothed_descent(a0, time_limit, top_k=60, tau_init=0.005,
                               lr_init=0.02, beta=0.9, verbose=False, label=""):

    Nesterov-accelerated gradient descent with log-sum-exp smoothed objective.
    Uses adaptive temperature and backtracking.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    lr = lr_init
    tau = tau_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute objective and gradient at y (lookahead point)
        conv_y = fast_convolve(y)
        M_y = conv_y.max()

        # Log-sum-exp weights
        K = min(top_k, len(conv_y))
        if K < len(conv_y):
            peak_indices = np.argpartition(conv_y, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv_y))

        pv = conv_y[peak_indices]
        scaled = pv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = compute_subgradient(y, peak_indices, w)

        # Nesterov update: a_new = project(a - lr * g)
        # y_new = a_new + beta * (a_new - a)
        a_new = a - lr * g
        a_new = project_simplex(a_new)
        y_new = a_new + beta * (a_new - a)
        y_new = np.maximum(y_new, 1e-15)
        y_new = y_new / y_new.sum()

        # Evaluate at a_new
        conv_new = fast_convolve(a_new)
        new_obj = conv_new.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            no_improve = 0
            a, y = a_new, y_new
            if verbose and it % 500 == 0:
                print(f"    {label} iter {it}: obj={best_obj:.10f}, tau={tau:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1
            # Backtracking: try smaller step
            improved = False
            for scale in [0.5, 0.25, 0.1, 0.05]:

                a_trial = project_simplex(a_trial)
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < M_y * 0.999:
                    a, y = a_trial, a_trial + beta * (a_trial - a)
                    y = np.maximum(y, 1e-15)
                    y = y / y.sum()
                    improved = True
                    if obj_trial < best_obj - 1e-14:
                        best_obj = obj_trial
                        best_a = a.copy()
                        no_improve = 0
                    break
            if not improved:
                a, y = a, y  # Stay put

        # Adaptive temperature: decrease slowly
        tau = max(tau * 0.9995, 0.0005)

        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.7
            no_improve = 0
        lr = max(lr * 0.9999, 1e-8)

        # Momentum decay when stuck
        if no_improve > 200:
            beta = max(beta * 0.95, 0.5)

        # Restart if stuck too long
        if no_improve > 500:
            rng = np.random.default_rng(it * 104729 + 7)
            noise_scale = 0.002 + 0.003 * min(no_improve / 2000, 1.0)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 1e-15)
            a = a / a.sum()
            y = a.copy()
            v = np.zeros(n)
            lr = lr_init * 0.4
            beta = 0.9
            tau = min(tau * 3, 0.01)
            no_improve = 0

def exact_max_descent(a0, time_limit, top_k=40, temp_init=0.002, lr_init=0.008,
                       verbose=False, label=""):

    Uses a "leveling" strategy: when peaks are close, weight them equally
    to encourage leveling.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    momentum = 0.8
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Adaptive peak selection: use peaks above M - threshold
        threshold = M * (1.0 - 0.05 * min(it / 1000, 1.0))
        active_mask = conv >= threshold
        active_count = active_mask.sum()

        # If too many active peaks, use top_k
        if active_count > top_k:
            peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        elif active_count > 0:
            peak_indices = np.where(active_mask)[0]
        else:
            peak_indices = np.array([np.argmax(conv)])

        # Weighting: emphasize peaks that are close to max (levelling)
        pv = conv[peak_indices]
        gap = M - pv  # 0 for the max peak, positive for others
        # Exponential weighting based on gap: peaks close to max get more weight
        w = np.exp(-gap / (temp + 1e-10))
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = compute_subgradient(a, peak_indices, w)

        # Heavy-ball momentum update
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        conv_new = fast_convolve(a_new)
        new_obj = conv_new.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and it % 300 == 0:
                print(f"    {label} EMA iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            # Backtrack
            improved = False
            for scale in [0.5, 0.25, 0.1]:
                a_trial = a - lr * scale * v
                a_trial = project_simplex(a_trial)
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < M * 0.999:
                    a = a_trial
                    if obj_trial < best_obj - 1e-14:
                        best_obj = obj_trial
                        best_a = a.copy()
                        no_improve = 0
                    improved = True
                    break
            if not improved:
                v = np.zeros(n)
                # Pure gradient step
                a_trial = a - lr * 0.1 * g
                a_trial = project_simplex(a_trial)
                a = a_trial

        # Adapt parameters
        temp = min(temp * 1.002, 0.05)
        if no_improve > 80:
            lr *= 0.8
            no_improve = 0
        lr = max(lr * 0.99995, 1e-8)

        # Restart
        if no_improve > 400:
            rng = np.random.default_rng(it * 31337 + 11)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 1e-15)

            v = np.zeros(n)

def frequency_perturbation(a, rng, magnitude=0.1):
    """Perturb in frequency domain: multiply DFT by random phase, preserve magnitude."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    # Random phase perturbation
    phase = np.exp(1j * rng.uniform(-magnitude, magnitude, size=A.shape))
    A_pert = A * phase

    # Take IFFT and project to non-negative
    a_pert = np.fft.irfft(A_pert, fft_size)[:size]
    if len(a_pert) >= n:
        a_pert = a_pert[:n]

    # Project to simplex
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s > 1e-10:
        a_pert = a_pert / s
    else:
        a_pert = a.copy()

    return a_pert


def structural_perturbation(a, rng, pert_type=0):
    """Apply a structural perturbation to the sequence."""
    n = len(a)
    a_new = a.copy()

    if pert_type == 0:
        # Split: take mass from a peak and spread to neighbors
        j = int(np.argmax(a_new))
        mass = a_new[j] * 0.3
        a_new[j] -= mass
        # Spread to neighbors
        spread_range = min(3, n // 2)
        for offset in range(1, spread_range + 1):
            for direction in [-1, 1]:
                idx = j + offset * direction
                if 0 <= idx < n:
                    a_new[idx] += mass / (2 * spread_range)
    elif pert_type == 1:
        # Local flattening: average a random window
        win = min(8, n // 4)
        start = rng.integers(0, max(1, n - win))
        a_new[start:start + win] = np.mean(a_new[start:start + win])
    elif pert_type == 2:
        # Frequency perturbation
        a_new = frequency_perturbation(a, rng, magnitude=0.15)
    elif pert_type == 3:
        # Asymmetric shift of mass
        half = n // 2
        mass = 0.02
        src_start, src_end = rng.integers(0, half - 1), rng.integers(half, n - 1)
        mass_to_move = a_new[src_start] * 0.5
        if mass_to_move > 1e-10:
            a_new[src_start] -= mass_to_move
            a_new[src_end] += mass_to_move
    else:
        # Random Gaussian
        a_new = a + rng.standard_normal(n) * 0.005

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def coordinate_descent_polish(a0, time_limit, verbose=False, label=""):
    """Thorough coordinate descent with golden-section 1D search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0
    golden_ratio = (np.sqrt(5) - 1) / 2  # ~0.618

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        round_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-14:
                continue

            # Golden section search for optimal a[j]
            # Parameterize: a[j] = t, others scaled to maintain sum=1
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            def obj_1d(t):
                if t < 0 or t > 0.999:
                    return float('inf')
                trial = a.copy()
                trial[j] = t
                trial[np.arange(n) != j] *= (1.0 - t) / denom
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-14:
                    return float('inf')
                trial = trial / ss
                return fast_convolve(trial).max()

            # Search in [0, min(2*old_val, 0.3)]
            lo = 0.0
            hi = min(2.0 * old_val, 0.3)
            if hi < 1e-14:
                continue

            # Golden section: 5 evaluations
            c = hi - golden_ratio * (hi - lo)
            d = lo + golden_ratio * (hi - lo)
            fc = obj_1d(c)
            fd = obj_1d(d)

            for _ in range(4):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - golden_ratio * (hi - lo)
                    fc = obj_1d(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + golden_ratio * (hi - lo)
                    fd = obj_1d(d)

            best_t = (lo + hi) / 2
            best_f = obj_1d(best_t)

            # Also try the current value
            f_current = obj_1d(old_val)
            if best_f < min(f_current, best_obj) - 1e-14:
                a[j] = best_t
                a[np.arange(n) != j] *= (1.0 - best_t) / denom
                a = np.maximum(a, 0.0)
                ss = a.sum()
                if ss > 1e-14:
                    a = a / ss
                if best_f < best_obj - 1e-14:
                    best_obj = best_f
                    best_a = a.copy()
                    round_improved = True

        if not round_improved:
            break

        if verbose and rounds % 5 == 0:
            print(f"    {label} CD round {rounds}: obj={best_obj:.10f}")

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_sinc_like(n, bandwidth=0.4):
    """Sinc-like sequence (bandlimited)."""
    x = np.arange(n) - n / 2.0
    a = np.sinc(x / (n * bandwidth))
    a = np.abs(a)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym
