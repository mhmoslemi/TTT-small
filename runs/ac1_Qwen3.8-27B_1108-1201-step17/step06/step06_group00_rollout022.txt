LP-based alternating optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses the fact that for fixed a, the problem of finding a' that minimizes
max_k conv(a',a)_k is an LP. Alternates between a and a' with line search.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def build_conv_matrix(a, n):
    """Build matrix C where (C @ x)[k] = conv(x, a)[k]."""
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def solve_lp_for_direction(a, solver=None):
    """Solve LP: min t s.t. C@a' <= t*1, a'>=0, sum(a')=1. Returns a'."""
    n = len(a)
    C = build_conv_matrix(a, n)
    
    a_prime = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    
    constraints = [
        cp.sum(a_prime) == 1,
        cp.multiply(C, a_prime) <= t,
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    if solver is None:
        # Try solvers in order of preference
        for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]:
            try:
                prob.solve(solver=s, verbose=False)
                if a_prime.value is not None:
                    break
            except:
                continue
    else:
        prob.solve(solver=solver, verbose=False)
    
    if a_prime.value is None:
        return None
    
    val = np.clip(a_prime.value, 0, None)
    s = val.sum()
    if s < 1e-10:
        return None
    return val / s


def lp_alternating(a_init, time_limit, t_candidates=None, max_iter=1000):
    """LP-based alternating optimization."""
    if t_candidates is None:
        t_candidates = [0.5, 0.3, 0.2, 0.1]
    
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = eval_score(a)
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_prime = solve_lp_for_direction(a)
        if a_prime is None:
            break
        
        # Line search
        best_t_score = float('inf')
        best_t_a = None
        for t in t_candidates:
            a_new = (1 - t) * a + t * a_prime
            a_new = np.clip(a_new, 0, None)
            s = a_new.sum()
            if s < 1e-10:

            a_new = a_new / s
            score = eval_score(a_new)
            if score < best_t_score:
                best_t_score = score
                best_t_a = a_new
        
        if best_t_a is None:
            break
        
        a = best_t_a
        if best_t_score < best_score - 1e-12:
            best_score = best_t_score
            best_a = a.copy()
            if it % 20 == 0:
                print(f"    LP alt iter {it}: score={best_score:.6f}")
    
    return best_a, best_score

def solve_lp_scipy(a, n):
    """Solve LP using scipy linprog."""
    C = build_conv_matrix(a, n)
    
    # Variables: [a'_0, ..., a'_{n-1}, t]
    # Minimize: [0, ..., 0, 1] (just minimize t)

    c[-1] = 1

    # i.e., [C, -1] @ x <= 0

        s = a_prime.sum()

"""
LP-based alternating optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.

Key idea: For fixed a, finding a' that minimizes max_k conv(a',a)_k is an LP.
We alternate between a and a' with line search, then refine with subgradient
descent and coordinate descent.

Phases:
1. LP alternating from parent (main optimization)
2. Subgradient refinement
3. Multi-length exploration
4. Final polish (coordinate descent + fine subgradient)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def build_conv_matrix(a, n):
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def solve_lp_direction(a, n=None, top_k=None):
    """Solve LP: min t s.t. conv(a', a)_k <= t for all k (or top_k), a'>=0, sum(a')=1."""
    if n is None:
        n = len(a)
    
    C_full = build_conv_matrix(a, n)
    
    if top_k is not None and top_k < 2 * n - 1:
        # Only use top_k constraints
        b_current = C_full @ a  # This is conv(a, a)
        top_idx = np.argpartition(b_current, -top_k)[-top_k:]
        C = C_full[top_idx, :]
    else:
        C = C_full
    
    m = C.shape[0]
    
    # Variables: [a'_0, ..., a'_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: C @ a' - t <= 0 → [C | -1] @ [a'; t] <= 0
    A_ub = np.hstack([C, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(a') = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'disp': False})
        if res.success:
            a_prime = np.clip(res.x[:n], 0, None)
            s = a_prime.sum()
            if s > 1e-10:
                return a_prime / s
    except Exception as e:
        print(f"    LP error: {e}")
    
    return None


def lp_alternating(a_init, time_limit, max_iter=500, t_candidates=None, top_k=None):
    if t_candidates is None:
        t_candidates = [0.5, 0.3, 0.2, 0.1]
    
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = 1.0
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = eval_score(a)
    
    stagnation = 0
    adaptive_t = 0.3
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_prime = solve_lp_direction(a, n, top_k)
        if a_prime is None:
            break
        
        # Line search
        best_t_score = float('inf')
        best_t_a = None
        for t in t_candidates:
            a_new = (1 - t) * a + t * a_prime
            a_new = np.clip(a_new, 0, None)
            ss = a_new.sum()
            if ss < 1e-10:
                continue
            a_new = a_new / ss
            score = eval_score(a_new)
            if score < best_t_score:
                best_t_score = score
                best_t_a = a_new
        
        if best_t_a is None:
            break
        
        a = best_t_a
        if best_t_score < best_score - 1e-12:
            best_score = best_t_score
            best_a = a.copy()
            stagnation = 0
            if it % 50 == 0:
                print(f"    LP iter {it}: score={best_score:.6f}")
        else:
            stagnation += 1
            if stagnation > 50:
                # Reduce mixing and restart
                adaptive_t = max(0.05, adaptive_t * 0.7)
                t_candidates = [adaptive_t, adaptive_t * 0.5, adaptive_t * 2]
                stagnation = 0
                if stagnation > 100:
                    break
    
    return best_a, best_score


def subgradient_optimize(a_init, time_limit, lr_init=0.003, momentum=0.9,
                         n_top=50, kick_size=0.002, kick_interval=500):
    rng = np.random.default_rng(12345)
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = eval_score(a)
    cur_lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        
        # Find top peaks and compute weights
        n_top_eff = min(n_top, len(b))
        top_idx = np.argpartition(b, -n_top_eff)[-n_top_eff:]
        w = b[top_idx].copy()
        # Temperature weighting
        w_scale = w / (w.max() + 1e-30)
        w = w_scale ** 5
        w_sum = w.sum()
        if w_sum < 1e-30:
            w[top_idx == top_idx[w.argmax()]] = 1.0  # fallback
        else:
            w /= w_sum
        
        # Compute subgradient via cross-correlation
        # g[i] ≈ sum_k w[k] * a[k-i]
        # This is np.correlate(w_full, a, 'valid') where w_full is the full convolution-weighted vector
        w_full = np.zeros(len(b))
        w_full[top_idx] = w
        g = np.correlate(w_full, a, mode='valid')
        # The valid mode gives length (len(w_full) - len(a) + 1) = (2n-1) - n + 1 = n. 
        
        # Add diagonal correction (approximately)
        # For each k in top_idx, if k is even, the term i=k/2 gets only 1x instead of 2x
        # This is a small correction, skip for now
        
        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = a + v
        a_new = np.clip(a_new, 0, None)
        ss = a_new.sum()
        if ss < 1e-10:
            a_new = a
        else:
            a_new = a_new / ss
        
        score = eval_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > kick_interval:
            a = best_a + rng.standard_normal(n) * kick_size
            a = np.clip(a, 0, None)
            ss = a.sum()
            if ss > 1e-10:
                a = a / ss
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.8, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr = max(cur_lr * 0.95, lr_init * 0.05)
    
    return best_a, best_score


def coordinate_descent(a_init, time_limit, multipliers=None):
    if multipliers is None:
        multipliers = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]
    
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_score = eval_score(a)
    best_a = a.copy()
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j = old_val
            best_j_score = best_score
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.arange(n) != j
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                score = eval_score(a_trial)
                if score < best_j_score - 1e-13:
                    best_j_score = score
                    best_j = new_val
            
            if best_j != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j) / denom
                a[j] = best_j
                mask = np.arange(n) != j
                a[mask] *= scale
                a = a / a.sum()
                if best_j_score < best_score - 1e-13:
                    best_score = best_j_score
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
