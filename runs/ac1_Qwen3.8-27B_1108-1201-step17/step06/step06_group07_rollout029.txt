"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated mirror descent with
adaptive constraint sets, Nesterov momentum, peak equalization, and multi-scale
optimization. Improves on previous approach with better convergence properties.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_conv_grad(a, peak_indices, weights):
    """Compute weighted gradient of sum w_k * (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += w_k * a[k - np.arange(js, je + 1)]
    return 2.0 * g

    best_a = a.copy()

    no_improve = 0

        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        improved = False

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def symmetrize(a):

            a = a / a.sum()
            a_prev = a.copy()
            momentum = 0.0
            lr = 0.006
            temp = 0.0008
            no_improve = 0

        # Symmetrize periodically
        if it % 400 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj - 1e-14:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()
                a_prev = a.copy()

        if lr < 1e-9:
            break

    return best_a, best_obj


def linear_descent_phase(a0, time_limit, verbose=False, label=""):
    """
    Rectilinear (additive) descent on the simplex.
    Different geometry from multiplicative, can find different optima.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    dt = 0.005
    temp = 0.0005
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(20 + it // 100, 60, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = compute_conv_grad(a, peak_indices, w)
        g_centered = g - g.mean()

        # Backtracking
        best_dt = dt * 0.01
        best_trial_obj = M * 1.01
        for factor in [1.0, 0.5, 0.25, 0.1]:
            trial_dt = dt * factor
            if trial_dt < 1e-12:
                continue
            a_trial = a - trial_dt * g_centered
            a_trial = project_simplex(a_trial)
            trial_conv = fast_convolve(a_trial)
            trial_obj = trial_conv.max()
            if trial_obj <= best_trial_obj:
                best_dt = trial_dt
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        a = a - best_dt * g_centered
        a = project_simplex(a)

        conv = fast_convolve(a)
        new_obj = conv.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 300 == 0:
                print(f"    {label} LD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        temp = max(temp * 0.9995, 0.0001)
        dt = max(dt * 0.99995, 1e-9)

        if no_improve > 300:
            rng = np.random.default_rng(it * 104729 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            dt = 0.003
            no_improve = 0

        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = project_simplex(a_sym)
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj - 1e-14:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        if dt < 1e-10:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """
    Fine coordinate descent: adjust one coordinate at a time.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    ss = a.sum()
    if ss < 1e-15:
        return a, float('inf')
    a = a / ss
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for frac in [0.6, 0.8, 0.9, 0.97, 1.03, 1.1, 1.25, 1.5]:
                new_val = old_val * frac
                if new_val < 0 or new_val > 0.9:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                s = trial.sum()
                if s < 1e-15:
                    continue
                trial = trial / s
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Random perturbation to escape
            rng = np.random.default_rng(rounds * 31 + 7)
            for _ in range(10):
                j = int(rng.integers(n))
                old_val = a[j]
                if old_val < 1e-14:
                    continue
                for frac in [0.5, 1.5, 0.7, 1.3, 2.0]:
                    new_val = old_val * frac
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    trial = a.copy()
                    trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    trial[mask] *= scale
                    trial = np.maximum(trial, 0.0)
                    s = trial.sum()
                    if s < 1e-15:
                        continue
                    trial = trial / s
                    val = fast_convolve(trial).max()
                    if val < best_obj - 1e-14:
                        a = trial
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
            if not improved:
                break

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence normalized to sum=1."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 5
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n, c1=0.3, c2=0.7, sigma=0.08):
    """Bimodal Gaussian."""
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make a sequence palindromic."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def update_best(a, best_seq, best_score, label=""):
    """Try a sequence and update best if improved. Returns (best_seq, best_score, score)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score, float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score, score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: accelerated mirror descent with Nesterov momentum,
    linear descent phase, coordinate descent, multi-length exploration, and diverse
    restarts. All phases share time from a 1800s budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # Determine lengths to try
    if parent is not None:
        lengths_to_try = [len(parent), 512, 896, 1024, 640]
    else:
        lengths_to_try = [512, 768, 896, 1024]
    lengths_to_try = sorted(set(lengths_to_try))
    print(f"Lengths to try: {lengths_to_try}")

    # === Phase 1: Quick evaluation of initializations ===
    print("\n=== Phase 1: Initializations ===")
    best_init = None
    best_init_score = float('inf')

    init_list = []
    for n_try in lengths_to_try:
        init_list.append((make_gaussian(n_try, 0.25), f"gauss_{n_try}"))
        init_list.append((make_gaussian(n_try, 0.35), f"gaussW_{n_try}"))
        init_list.append((make_bell(n_try, 0.5), f"bell_{n_try}"))

    if parent is not None:
        init_list.append((parent, "parent"))
        init_list.append((symmetrize(parent), "parent_sym"))
        for n_try in lengths_to_try:
            if n_try != len(parent):
                init_list.append((interpolate_sequence(parent, n_try), f"parent_int_{n_try}"))

    for a_init, lbl in init_list:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        best_seq, best_score, sc = update_best(a_norm, best_seq, best_score, f"init_{lbl}")
        if sc < best_init_score:
            best_init_score = sc
            best_init = a_norm

    if best_init is None:
        best_init = make_gaussian(768, 0.3)

    print(f"  Best init score: {best_init_score:.6f}")

    # === Phase 2: Main Accelerated Mirror Descent ===
    print(f"\n=== Phase 2: AMd main ===")
    remaining = deadline - time.monotonic()
    t_md = remaining * 0.35
    print(f"  Time: {t_md:.0f}s")
    a_opt, obj = accelerated_mirror_descent(best_init, time_limit=t_md,
                                            verbose=True, label="AMd_main")
    best_seq, best_score, _ = update_best(a_opt, best_seq, best_score, "AMd_main")

    # === Phase 3: Linear Descent (different geometry) ===
    print(f"\n=== Phase 3: Linear Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_ld = remaining * 0.15
        print(f"  Time: {t_ld:.0f}s")
        a_opt, obj = linear_descent_phase(best_seq, time_limit=t_ld,
                                          verbose=True, label="LD")
        best_seq, best_score, _ = update_best(a_opt, best_seq, best_score, "LD")

    # === Phase 4: AMd with symmetrized start ===
    print(f"\n=== Phase 4: AMd sym ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_md2 = remaining * 0.20
        print(f"  Time: {t_md2:.0f}s")
        a_sym = symmetrize(best_seq)
        a_opt, obj = accelerated_mirror_descent(a_sym, time_limit=t_md2 * 0.55,
                                                verbose=True, label="AMd_sym")
        best_seq, best_score, _ = update_best(a_opt, best_seq, best_score, "AMd_sym")

        if time.monotonic() < deadline - 60:
            a_noisy = np.maximum(best_seq + rng.standard_normal(len(best_seq)) * 0.002, 1e-15)
            a_noisy = a_noisy / a_noisy.sum()
            a_opt, obj = accelerated_mirror_descent(a_noisy, time_limit=t_md2 * 0.45,
                                                    verbose=True, label="AMd_noise")
            best_seq, best_score, _ = update_best(a_opt, best_seq, best_score, "AMd_noise")

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Other lengths ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        other_lengths = [lt for lt in lengths_to_try if lt != len(best_seq)]
        if other_lengths and remaining > 60:
            # Try at most 2 other lengths
            for n_try in other_lengths[:2]:
                remaining = deadline - time.monotonic()
                if remaining < 40:
                    break
                a_init = interpolate_sequence(best_seq, n_try)
                t_alloc = min(remaining * 0.08, 80)
                if t_alloc < 20:
                    break
                a_opt, obj = accelerated_mirror_descent(a_init, time_limit=t_alloc,
                                                        verbose=True, label=f"AMd_n{n_try}")
                best_seq, best_score, _ = update_best(a_opt, best_seq, best_score, f"AMd_n{n_try}")

    # === Phase 6: Coordinate Descent Polish ===
    print(f"\n=== Phase 6: CD Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = min(remaining * 0.12, 150)
        print(f"  Time: {t_cd:.0f}s")
        a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd,
                                        verbose=True, label="CD")
        best_seq, best_score, _ = update_best(a_opt, best_seq, best_score, "CD")

    # === Phase 7: Final fine-grained AMd ===
    print(f"\n=== Phase 7: Final AMd ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_final = min(remaining - 20, 120)
            print(f"  Time: {t_final:.0f}s")
            # Use very small temperature for final convergence
            a_opt, obj = accelerated_mirror_descent(best_seq, time_limit=t_final,
                                                    verbose=True, label="Final")
            best_seq, best_score, _ = update_best(a_opt, best_seq, best_score, "Final")

            # Try symmetrized version
            a_s = symmetrize(best_seq)
            best_seq, best_score, _ = update_best(a_s, best_seq, best_score, "final_sym")

            # Final CD
            if time.monotonic() < deadline - 20:
                t_cd2 = min(deadline - time.monotonic() - 10, 80)
                a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd2,
                                                verbose=True, label="CD_final")
                best_seq, best_score, _ = update_best(a_opt, best_seq, best_score, "CD_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
