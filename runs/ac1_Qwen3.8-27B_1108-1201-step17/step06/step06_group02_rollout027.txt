import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def evaluate_with_normalize(a):
    """Evaluate the normalized objective (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

    """Gerchberg-Saxton style spectral flattening."""

    if s < 1e-15:
        return f

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()

    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_with_normalize(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""

def active_set_balancing(f, n_iters, eta, verbose=False):

    """

                if conv[ki] <= conv[kj]:

                je_i = min(n - 1, ki)
                js_j = max(0, kj - n + 1)
                je_j = min(n - 1, kj)
                js = max(js_i, js_j)

    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    temp = 0.002
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.3:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Compute softmax weights over top lags
        k_top = min(30, len(conv))
        top_idx = np.argpartition(conv, -k_top)[-k_top:]
        top_vals = conv[top_idx]

        scaled = top_vals / temp
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        weights = exp_vals / exp_vals.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(top_idx)):
            if weights[i] < 1e-12:
                continue
            k = int(top_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * 2.0 * f[k - idx]

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            temp *= 1.5
            continue
        sg = sg / sg_norm

        # Momentum
        v = momentum * v - sg
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            continue
        v = v / v_norm

        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]:
            trial = f - alpha * v
            trial = project_simplex(trial)
            val = fast_convolve(trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-12:
            no_improve += 1
            temp *= 1.4
            if no_improve > 80:
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                temp = 0.002
                no_improve = 0
            continue

        f = f - best_alpha * v
        f = project_simplex(f)

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_f = f.copy()
            no_improve = 0
            temp = max(0.0005, temp * 0.98)
            if verbose and it % 100 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}, temp={temp:.6f}")
        else:
            no_improve += 1
            temp *= 1.03

        if it % 300 == 0 and no_improve > 50:
            temp = min(0.1, temp * 2.5)
            v = np.zeros(n)

    return best_f, best_obj


def peak_targeted_cd(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: identifies peaks and moves mass
    from contributing coordinates to non-contributing ones.
    """
    f = np.maximum(f0, 0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        if no_improve > 300:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Find top peaks
        n_peaks = min(5, len(conv))
        peak_idxs = np.argsort(conv)[-n_peaks:]

        found = False
        for pk in peak_idxs:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            contrib_idx = np.arange(js, je + 1)
            if len(contrib_idx) == 0:
                continue

            products = f[contrib_idx] * f[pk - contrib_idx]
            max_prod_pos = np.argmax(products)
            source = contrib_idx[max_prod_pos]

            # Find non-contributing targets
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[contrib_idx] = False
            if non_contrib.sum() < 2:
                continue

            # Try moving mass from source to non-contributing coords
            non_contrib_idx = np.where(non_contrib)[0]
            # Pick best target (where it won't create a new peak)
            best_target = non_contrib_idx[np.argmax(non_contrib_idx * 0 + 1)]  # just pick first

            for frac in [0.05, 0.1, 0.2, 0.3]:
                diff = f[source] * frac
                if diff < 1e-10:
                    continue
                trial = f.copy()
                trial[source] -= diff
                nc_vals = trial[non_contrib]
                nc_sum = nc_vals.sum()
                if nc_sum < 1e-15:
                    continue
                trial[non_contrib] += diff * nc_vals / nc_sum
                trial = np.maximum(trial, 0.0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-15:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    found = True
                    if verbose:
                        print(f"    PCD: peak={pk}, obj={best_obj:.10f}")
                    break
            if found:
                break

        if not found:
            # Random pair swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                no_improve += 1
                continue
            for direction in [1, -1]:
                delta = min(f[i], f[j]) * 0.02 * direction
                if abs(delta) < 1e-12:
                    continue
                trial = f.copy()
                trial[i] -= delta
                trial[j] += delta
                if trial[i] < 0 or trial[j] < 0:
                    continue
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-15:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    found = True
                    break
            if not found:
                no_improve += 1

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_bell(n, width_frac):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def combined_optimize(f0, time_limit, verbose=False):
    """
    Combined optimization: interleave spectral flattening,
    active-set balancing, and adaptive subgradient.
    """
    f = np.maximum(f0, 0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Phase A: Spectral flattening (first 15%)
    t_spec = time_limit * 0.15
    f = spectral_flatten(f, n_iters=40, alpha=0.75, verbose=verbose)
    obj = evaluate_with_normalize(f)
    if obj < best_obj:
        best_obj = obj
        best_f = f.copy()

    # Phase B: Active-set balancing (next 20%)
    t0b = time.monotonic()
    t_asb = time_limit * 0.20
    n_asb_iters = max(20, int(t_asb * 50))
    f = active_set_balancing(f, n_iters=n_asb_iters, eta=0.005, verbose=verbose)
    obj = evaluate_with_normalize(f)
    if obj < best_obj:
        best_obj = obj
        best_f = f.copy()
    if verbose:
        print(f"    After ASB: obj={best_obj:.10f}")

    # Phase C: Adaptive subgradient (remaining time)
    rem = time_limit - (time.monotonic() - t0) - 2.0
    if rem > 1.0:
        f_sg, obj_sg = adaptive_subgradient(best_f, time_limit=rem, verbose=verbose)
        if obj_sg < best_obj - 1e-15:
            best_obj = obj_sg
            best_f = f_sg.copy()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining spectral flattening,
    active-set balancing, adaptive subgradient, and coordinate descent.
    Explores multiple sequence lengths for robustness.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = try_sequence(parent, state, "parent")
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = make_bell(768, 0.6)
        try_sequence(parent, state, "default")

    n_parent = len(parent)
    print(f"Starting: n={n_parent}, best={state['best_score']:.8f}")

    # === Phase 1: Combined optimization on parent ===
    print("\n=== Phase 1: Combined Optimization (Parent) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.4, 600)
    print(f"  Allocating {t1:.0f}s")
    f_opt, obj_opt = combined_optimize(parent, time_limit=t1, verbose=True)
    try_sequence(f_opt, state, "combined_parent")

    # === Phase 2: Spectral variants ===
    print("\n=== Phase 2: Spectral Variants ===")
    if state['best_seq'] is not None:
        for alpha in [0.5, 0.65, 0.85, 0.95]:
            rem = deadline - time.monotonic()

                break
            t = min(rem * 0.08, 40)
            f_sp = spectral_flatten(state['best_seq'], n_iters=max(10, int(t * 8)), alpha=alpha)
            try_sequence(f_sp, state, f"spec_a{alpha}")

    # === Phase 3: Adaptive subgradient on best ===
    print("\n=== Phase 3: Adaptive Subgradient ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic()
        t3 = min(rem * 0.25, 350)
        print(f"  Allocating {t3:.0f}s")
        f_sg, obj_sg = adaptive_subgradient(state['best_seq'], time_limit=t3, verbose=True)
        try_sequence(f_sg, state, "subgrad")

    # === Phase 4: Multi-length exploration ===
    print("\n=== Phase 4: Multi-Length Exploration ===")
    if state['best_seq'] is not None:
        base_n = len(state['best_seq'])
        candidate_ns = [base_n - 64, base_n + 64, base_n - 128, base_n + 128]
        # Add some fixed lengths
        for cnd in [512, 640, 896, 1024, 1280]:
            if cnd not in candidate_ns and 128 <= cnd <= 2048:
                candidate_ns.append(cnd)
        candidate_ns = sorted(set(candidate_ns))
        candidate_ns = [n for n in candidate_ns if n != base_n and 128 <= n <= 2048]

        for target_n in candidate_ns:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_interp = min(rem * 0.15, 80)
            if t_interp < 10:
                break

            f_interp = interpolate_sequence(state['best_seq'], target_n)
            # Quick spectral flatten
            f_opt_len = spectral_flatten(f_interp, n_iters=20, alpha=0.7)
            try_sequence(f_opt_len, state, f"len_{target_n}")

            # Short combined optimization
            rem = deadline - time.monotonic()
            if rem > 20:
                t_short = min(rem * 0.12, 50)
                if t_short > 8:
                    f_opt2, _ = combined_optimize(f_opt_len, time_limit=t_short)
                    try_sequence(f_opt2, state, f"len_opt_{target_n}")

    # === Phase 5: Peak-targeted coordinate descent ===
    print("\n=== Phase 5: Peak-Targeted CD ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.2, 150)
        print(f"  Allocating {t5:.0f}s")
        f_cd, _ = peak_targeted_cd(state['best_seq'], time_limit=t5, verbose=True)
        try_sequence(f_cd, state, "pcd")

    print("\n=== Phase 6: Final Squeeze ===")
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t6 = min(rem * 0.7, 120)
            print(f"  Allocating {t6:.0f}s")
            f_final, _ = adaptive_subgradient(state['best_seq'], time_limit=t6, verbose=True)
            try_sequence(f_final, state, "final_sg")

        # One more CD pass
        rem = deadline - time.monotonic()
        if rem > 10:
            t_cd2 = min(rem * 0.6, 40)
            f_cd2, _ = peak_targeted_cd(state['best_seq'], time_limit=t_cd2)
            try_sequence(f_cd2, state, "final_cd")

    # === Phase 7: Active-set balancing on final best ===
    if state['best_seq'] is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            n_iters = max(10, int(rem * 40))
            f_asb, _ = active_set_balancing(state['best_seq'], n_iters=n_iters, eta=0.003)
            try_sequence(f_asb, state, "final_asb")

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(768) / 768
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
