import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_topk_gradient(f, c, K):
    """
    Compute gradient of top-K average of convolution.
    F_K = (1/K) * sum_{k in topK} c_k
    grad_i = (2/K) * sum_{k in topK} f_{k-i} [if 0<=k-i<n]
    """
    n = len(f)
    topK = np.argpartition(c, -K)[-K:]
    w = np.zeros(len(c))
    w[topK] = 1.0 / K

    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]

    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g


def compute_lse_gradient(f, c, beta):
    """Compute gradient of log-sum-exp smoothed max of convolution."""
    n = len(f)
    scaled = beta * c
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()

    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]

    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g


def solve_lp_direction_active(f, M, n, K_active=20):
    """
    Solve LP with only top-K_active convolution constraints.
    max sum(g) s.t. (f*g)_k <= M for k in active_set, g >= 0
    """
    c = fast_convolve(f)
    # Get active set: top K constraint positions
    if K_active < len(c):
        active_idx = np.argpartition(c, -K_active)[-K_active:]
    else:
        active_idx = np.arange(len(c))

    # Set up LP
    # Variables: g_0, ..., g_{n-1}
    # Constraints: for each active k, sum_j f_{k-j} * g_j <= M (where 0<=k-j<n)
    c_obj = -np.ones(n)

    A_rows = []
    b_rows = []
    for k in active_idx:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = f[k - j]
        A_rows.append(row)
        b_rows.append(M)

    if not A_rows:
        return np.ones(n) / n

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


def solve_lp_direction_full(f, M, n):
    """Solve full LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


def fw_vertex_step(f, c):
    """
    Standard Frank-Wolfe vertex step.
    Compute subgradient, pick vertex with minimum gradient.
    Returns the direction (vertex - f).
    """
    n = len(f)
    # Find all positions achieving (near) max
    max_c = c.max()
    tol = max_c * 1e-8
    active = np.where(c >= max_c - tol)[0]

    # Average subgradient over active set
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += f[k - j]
    g /= len(active)

    # Vertex with minimum gradient
    j_min = np.argmin(g)
    s = np.zeros(n)
    s[j_min] = 1.0
    return s - f, g


def topk_subgradient_descent(f0, time_limit, K_schedule=None, verbose=False):
    """
    Gradient descent using top-K smoothed objective.
    Progressively increases K (from small to large) for annealing.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()

    t0 = time.monotonic()
    if K_schedule is None:
        K_schedule = [3, 5, 8, 12, 20, 30, 50, len(c)]
    else:
        K_schedule = [min(k, len(c)) for k in K_schedule]

    momentum = np.zeros(n)
    mu = 0.85

    for K_cur in K_schedule:
        if time.monotonic() - t0 > time_limit * 0.9:
            break

        stage_start = time.monotonic()
        stage_time = min(time_limit * 0.1, (time_limit - (time.monotonic() - t0)) * 0.15)
        lr = 0.5 / np.sqrt(n)
        no_improve = 0
        it = 0

        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.9:
            it += 1
            c = fast_convolve(f)
            g = compute_topk_gradient(f, c, K_cur)

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break

            # Normalize gradient to unit norm for consistent step size
            g_normed = g / (g_norm + 1e-15)

            # Momentum
            momentum = mu * momentum - lr * g_normed
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)

            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()

            if obj_trial <= c.max() + 1e-12:
                f = f_trial
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
                momentum = np.zeros(n)
                lr *= 0.7

            if no_improve > 100:
                rng = np.random.default_rng(it * 13 + K_cur)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                momentum = np.zeros(n)
                lr = 0.5 / np.sqrt(n)
                no_improve = 0

            if verbose and it % 200 == 0:
                print(f"    TopK GD K={K_cur} iter {it}: obj={c.max():.10f}, best={best_obj:.10f}")

    return best_f, best_obj

def peak_leveling_restart(f, K=3):
    """
    Perturb f to reduce the gap between the top-K peaks of conv(f,f).
    """
    n = len(f)
    c = fast_convolve(f)
    topK = np.argpartition(c, -K)[-K:]
    peak_vals = c[topK]
    min_peak = peak_vals.min()
    max_peak = peak_vals.max()
    gap = max_peak - min_peak

    if gap < 1e-10:
        return f.copy()

    # Perturb in direction that reduces the largest peak
    cc = fast_convolve(f)
    k_max = np.argmax(cc)
    # Gradient of conv(k_max)
    g = np.zeros(n)
    js = max(0, k_max - n + 1)
    je = min(n - 1, k_max)
    for j in range(js, je + 1):
        g[j] += f[k_max - j]
    g = g / (np.linalg.norm(g) + 1e-15)

    # Move slightly in -gradient direction projected to simplex
    scale = gap / (np.linalg.norm(f) * np.sqrt(n))
    f_pert = f - scale * g
    f_pert = project_simplex(f_pert)
    return f_pert

def interpolate_sequence(f, new_n):

    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Light smoothing
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    """Raised cosine (Hann window)."""
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_optimal_guess(n, width=0.55):
    """
    Create a sequence based on the expected optimal shape.
    Uses a modified Gaussian that's flatter in the center.
    """
    x = np.arange(n) - n / 2.0
    t = np.abs(x) / (n * width / 2.0)
    t = np.minimum(t, 1.0)
    # Super-Gaussian (power=3) for flatter top
    a = np.cos(np.pi * t / 2.0) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization with improved LP-FW and systematic exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.3)
        p_score = float(evaluate_sequence(parent.tolist()))
        best_score = p_score
        best_seq = parent.copy()
        print(f"No parent found. Using Gaussian n=768: score={p_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Quick screening (60s) ===
    print("\n=== Phase 1: Initialization Screening (60s) ===")
    t1_start = time.monotonic()
    t1_limit = 60

    init_candidates = []
    init_candidates.append(("parent", parent))
    init_candidates.append(("gauss_tight", make_gaussian(n_parent, 0.22)))
    init_candidates.append(("gauss_wide", make_gaussian(n_parent, 0.38)))
    init_candidates.append(("raised_cos", make_raised_cosine(n_parent)))
    init_candidates.append(("opt_guess", make_optimal_guess(n_parent, 0.55)))
    init_candidates.append(("opt_guess_wide", make_optimal_guess(n_parent, 0.65)))
    init_candidates.append(("opt_guess_tight", make_optimal_guess(n_parent, 0.45)))

    # Random perturbations of parent
    for t in range(8):
        noise_scale = [0.005, 0.01, 0.015, 0.02, 0.005, 0.01, 0.015, 0.02][t]
        noise = rng.standard_normal(n_parent) * noise_scale
        a_pert = np.maximum(parent + noise, 0)
        a_pert = a_pert / a_pert.sum()
        init_candidates.append((f"pert{t}", a_pert))

    best_init = None
    best_init_obj = float('inf')
    for name, a in init_candidates:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        c = fast_convolve(a)
        obj = 2 * len(a) * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a.copy()
        try_seq(a, f"init_{name}")

        if time.monotonic() - t1_start > t1_limit:
            break

    print(f"  Best init obj={best_init_obj:.8f}, t={time.monotonic()-t1_start:.1f}s")

    # === Phase 2: Top-K gradient descent (120s) ===
    print("\n=== Phase 2: Top-K Gradient Descent (120s) ===")
    t2_start = time.monotonic()
    rem = deadline - time.monotonic() - 900
    t2_alloc = max(30, min(120, rem))
    print(f"  Allocation: {t2_alloc:.0f}s")

    a_opt, obj = topk_subgradient_descent(best_init, time_limit=t2_alloc, verbose=True)
    try_seq(a_opt, "topk_gd")

    # Also try LSE from parent if different from best_init
    if not np.allclose(best_init, parent) and time.monotonic() < deadline - 850:
        rem = deadline - time.monotonic() - 850
        t2b = min(60, rem * 0.5)
        if t2b > 20:
            a_opt2, obj2 = lse_gradient_descent(parent, time_limit=t2b)
            try_seq(a_opt2, "lse_parent")

    print(f"  Phase 2 done, t={time.monotonic()-t2_start:.1f}s")

    # === Phase 3: Main LP-FW (300s) ===
    print("\n=== Phase 3: Main LP-FW (300s) ===")
    t3_start = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        t3_alloc = max(60, min(300, rem))
        print(f"  Allocation: {t3_alloc:.0f}s")

        a_opt, obj = lp_fw_optimize_improved(best_seq, time_limit=t3_alloc,
                                             verbose=True, use_active=True, K_active=15)
        try_seq(a_opt, "lpfw_main")

        # Second LP-FW run with different parameters
        if time.monotonic() < deadline - 400:
            rem = deadline - time.monotonic() - 400
            t3b = min(120, rem * 0.5)
            if t3b > 30:
                # Restart with peak-leveling
                f_pert = peak_leveling_restart(best_seq, K=3)
                a_opt2, obj2 = lp_fw_optimize_improved(f_pert, time_limit=t3b,
                                                       use_active=True, K_active=10)
                try_seq(a_opt2, "lpfw_pert")

    print(f"  Phase 3 done, t={time.monotonic()-t3_start:.1f}s")

    # === Phase 4: Multi-length exploration (240s) ===
    print("\n=== Phase 4: Multi-Length Exploration (240s) ===")
    t4_start = time.monotonic()
    base = best_seq if best_seq is not None else best_init

    target_lengths = []
    n_base = len(base)
    # Try lengths around the current best
    for frac in [0.75, 0.85, 1.0, 1.1, 1.25, 1.5, 2.0]:
        target_n = max(64, int(n_base * frac))
        if target_n not in target_lengths:
            target_lengths.append(target_n)
    # Add some specific lengths
    for nl in [512, 640, 768, 896, 1024, 1280, 1536]:
        if nl not in target_lengths and 64 <= nl <= 2048:
            target_lengths.append(nl)

    # Sort by proximity to current best (most promising first)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:6]:
        if time.monotonic() > deadline - 150:
            break

        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")

        # Quick gradient descent
        rem = deadline - time.monotonic() - 120
        if rem < 40:
            break
        t_gd = min(rem * 0.25, 60)
        if t_gd > 20:
            a_opt, obj = topk_subgradient_descent(f_interp, time_limit=t_gd)
            try_seq(a_opt, f"gd_n{target_n}")

        # LP-FW
        rem2 = deadline - time.monotonic() - 100
        if rem2 > 50:
            t_lpfw = min(rem2 * 0.4, 80)
            if t_lpfw > 20:
                a_opt, obj = lp_fw_optimize_improved(a_opt, time_limit=t_lpfw,
                                                     use_active=True, K_active=12)
                try_seq(a_opt, f"lpfw_n{target_n}")

    print(f"  Phase 4 done, t={time.monotonic()-t4_start:.1f}s")

    # === Phase 5: Deep LP-FW with diverse restarts (180s) ===
    print("\n=== Phase 5: Deep LP-FW (180s) ===")
    t5_start = time.monotonic()
    if best_seq is not None:
        n_best = len(best_seq)
        restart_strategies = [
            ("direct", best_seq.copy()),
            ("small_noise", project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)),
            ("peak_level", peak_leveling_restart(best_seq, K=5)),
            ("reverse_blend", None),  # computed below
            ("shape_mix", None),      # computed below
        ]

        # Compute special restarts
        restart_strategies[3] = ("reverse_blend",
                                  (best_seq + best_seq[::-1]) / ((best_seq + best_seq[::-1]).sum())
                                  if (best_seq + best_seq[::-1]).sum() > 1e-10
                                  else best_seq.copy())
        rc = make_optimal_guess(n_best, 0.55)
        restart_strategies[4] = ("shape_mix",
                                  (0.8 * best_seq + 0.2 * rc) / (0.8 * best_seq + 0.2 * rc).sum())

        for name, f_init in restart_strategies:
            if time.monotonic() > deadline - 60:
                break
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break
            t_alloc = min(rem / max(1, len(restart_strategies) - 1), 60)

            a_opt, obj = lp_fw_optimize_improved(f_init, time_limit=t_alloc,
                                                 use_active=True, K_active=12)
            try_seq(a_opt, f"deep_{name}")

    print(f"  Phase 5 done, t={time.monotonic()-t5_start:.1f}s")

    # === Phase 6: Final polish (90s) ===
    print("\n=== Phase 6: Final Polish (90s) ===")
    t6_start = time.monotonic()
    if best_seq is not None:
        # Coordinate descent
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_coord = min(rem * 0.5, 45)
            a_opt, obj = coordinate_polish(best_seq, time_limit=t_coord)
            try_seq(a_opt, "coord_final")

        # One more LP-FW squeeze
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t_final = min(rem, 40)
            a_opt, obj = lp_fw_optimize_improved(a_opt, time_limit=t_final,
                                                 use_active=True, K_active=8)
            try_seq(a_opt, "final_lpfw")

    print(f"  Phase 6 done, t={time.monotonic()-t6_start:.1f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
