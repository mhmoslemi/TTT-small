def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad_avg(a, conv, threshold_frac=0.99):
    """Average subgradient over peaks near the max."""
    n = len(a)
    M = conv.max()
    # Find peaks above threshold
    active = np.nonzero(conv >= M * threshold_frac)[0]
    g = np.zeros(n)
    for k in active:
        src = k - np.arange(n)
        mask = (src >= 0) & (src < n)
        g[mask] += a[src[mask]]
    g *= 2.0 / len(active)
    return g

    it = 0

        w = np.exp(scaled)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
