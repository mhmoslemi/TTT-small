```
def solve_lp(b, time_limit, prior=None, eps=1e-7, row_indices=None):
    n=len(b); m=2*n-1
    if row_indices is None:
       rows=np.arange(m)
    else: rows=np.asarray(row_indices,dtype=int); rows=rows[(rows>=0)&(rows<m)]
    if len(rows)==0: return None, None
    A = np.zeros((len(rows), n+1), dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    for loc, k in enumerate(rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            # i from js to je, b index k - i from k-je down to k-js
            A[loc, js:je+1] = b[k-je:k-js+1][::-1]
    A[:, n] = -1.0
    c = np.zeros(n+1); c[n]=1.0
    if prior is not None and eps != 0:
        p=np.asarray(prior,dtype=float); if len(p)==n: c[:n] += eps*p
    A_eq = np.zeros((1,n+1)); A_eq[0,:n]=1.0
    b_eq = np.array([1.0])
    bounds = [(0.0,1.0)]*(n+1)
    # maybe for t upper 1.0; already

    except Exception as e:

    if res.x is None: return None,None
```

```
def build_convolution_matrix(b, rows, include_t=False):
    b = np.asarray(b, dtype=np.float64)
    n = len(b)
    rows = np.asarray(rows, dtype=np.int64)
    rows = rows[(rows >= 0) & (rows < 2 * n - 1)]
    width = n + (1 if include_t else 0)
    A = np.zeros((len(rows), width), dtype=np.float64)
    for loc in range(len(rows)):
        k = int(rows[loc])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            A[loc, js:je+1] = b[k-je:k-js+1][::-1]
    if include_t:
        A[:, -1] = -1.0
    return A, rows
```

```
def build_convolution_matrix(b, rows, include_t=False):
    b = np.asarray(b, dtype=np.float64)
    n = len(b)
    rows = np.asarray(rows, dtype=np.int64)
    m = 2 * n - 1
    rows = rows[(rows >= 0) & (rows < m)]
    width = n + (1 if include_t else 0)

        if je >= js:
            A[loc, js:je + 1] = b[k - je:k - js + 1][::-1]
    if include_t:
        A[:, -1] = -1.0
    return A, rows
```

```python
"""
Searches for non-negative step lengths minimizing

    2*n*max(np.convolve(a,a)) / (sum(a))**2.

The method combines:
  - initial resampling / Gaussian sweeps,
  - LP-feasible random/gradient vertices against a kernel b
      max q^T a  s.t.  a*b <= T, sum a = 1, a >= 0,
  - LP best-response min_t max(a*b),
  - smooth mirror descent,
  - fast one-dimensional coordinate descent on max convolution coefficient.
"""

import numpy as np
import time

try:
    from scipy import optimize as sco
except Exception:
    sco = None


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_crossconv(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n, m = a.size, b.size
    if n == 0 or m == 0:
        return np.array([])
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def local_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    if not np.isfinite(a).all():
        a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S <= 1e-15:
        return float("inf")
    c = fast_convolve(a)
    if c.size == 0:
        return float("inf")
    M = float(np.max(c))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * a.size * M / (S * S))


def actual_score(a):
    try:
        a = normalize_prob(a)
        if a.size == 0:
            return float("inf")
        val = evaluate_sequence([float(x) for x in a])
        val = float(val)
        return val if np.isfinite(val) else float("inf")
    except Exception:
        return float("inf")


def normalize_prob(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def symmetrize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    b = 0.5 * (a + a[::-1])
    return normalize_prob(b)


def blend_seqs(a, b, t):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    if a.size == 0:
        return b
    if b.size == 0:
        return a
    return (1.0 - float(t)) * a + float(t) * b


def resample_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([])
    if a.size == 0:
        return np.ones(new_n, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(new_n, dtype=np.float64)
    a = a / s
    if a.size == 1:
        return np.ones(new_n, dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, a)
    y = np.maximum(y, 0.0)
    ys = float(y.sum())
    if ys <= 1e-15:
        return np.ones(new_n, dtype=np.float64)
    return y / ys


def make_gaussian(n, sigma_frac=0.3):
    n = int(n)
    if n <= 0:
        return np.array([])
    x = np.arange(n) - (n - 1) / 2.0
    sigma = max(1.0, n * float(sigma_frac))
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return normalize_prob(a)


def normalize_prior(q):
    q = np.asarray(q, dtype=np.float64).ravel()
    if q.size == 0:
        return np.array([])
    q = np.where(np.isfinite(q), q, 0.0)
    q = q - q.mean()
    nrm = float(np.linalg.norm(q))
    if nrm < 1e-15:
        q = np.arange(q.size, dtype=np.float64)
        q = q - q.mean()
        nrm = float(np.linalg.norm(q))
        if nrm < 1e-15:
            return np.zeros(q.size, dtype=np.float64)
    return q / nrm


def make_prior(n, mode, rng):
    n = int(n)
    if n <= 0:
        return np.array([])
    i = np.arange(n, dtype=np.float64)
    r = int(mode) % 6
    if r == 0:
        q = rng.standard_normal(n)
    elif r == 1:
        k = 1 + int(rng.integers(0, 6))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        q = np.sin(2.0 * np.pi * k * i / max(1, n) + phase)
    elif r == 2:
        k = 1 + int(rng.integers(0, 6))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        q = np.cos(2.0 * np.pi * k * i / max(1, n) + phase)
    elif r == 3:
        center = 0.5 * (n - 1)
        sigma = max(1.0, 0.15 * n)
        q = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    elif r == 4:
        if n > 1:
            q = np.sin(np.pi * i / max(1, n - 1))
        else:
            q = np.ones(n)
    else:
        q = rng.uniform(-1.0, 1.0, n)
    return normalize_prior(q)


def compute_subgradient(a, peak_indices, weights):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w = float(weights[idx])
        if not np.isfinite(w) or w <= 0.0:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w * a[k - je:k - js + 1][::-1]
    return g


def make_gradient_prior(a, conv, beta=80.0, K=100):
    a = np.asarray(a, dtype=np.float64).ravel()
    conv = np.where(np.isfinite(np.asarray(conv, dtype=np.float64)), np.asarray(conv, dtype=np.float64), 0.0)
    m = conv.size
    if m == 0 or a.size == 0:
        return np.zeros(a.size, dtype=np.float64)
    K = min(int(K), m)
    if K <= 0:
        K = 1
    if K < m:
        peaks = np.argpartition(conv, -K)[-K:]
    else:
        peaks = np.arange(m)
    pv = conv[peaks]
    scaled = pv * float(beta)
    if scaled.size:
        scaled = scaled - float(np.max(scaled))
        w = np.exp(scaled)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws <= 1e-300:
            w = np.zeros_like(w)
            w[int(np.argmax(pv))] = 1.0
        else:
            w = w / ws
    else:
        w = np.zeros(0)
    g = compute_subgradient(a, peaks, w)
    return normalize_prior(-g)


def build_convolution_matrix(b, rows, include_t=False):
    b = np.asarray(b, dtype=np.float64).ravel()
    n = b.size
    rows = np.atleast_1d(np.asarray(rows, dtype=np.int64))
    m = 2 * n - 1
    rows = rows[(rows >= 0) & (rows < m)]
    width = n + (1 if include_t else 0)
    A = np.zeros((len(rows), width), dtype=np.float64)
    for loc in range(len(rows)):
        k = int(rows[loc])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            A[loc, js:je + 1] = b[k - je:k - js + 1][::-1]
    if include_t:
        A[:, -1] = -1.0
    return A, rows


def run_linprog(c, A_ub, b_ub, A_eq, b_eq, bounds, time_limit):
    if sco is None:
        return None
    try:
        return sco.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(time_limit), "output_flag": False},
        )
    except Exception:
        try:
            return sco.linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="interior-point",
                options={"maxiter": 200},
            )
        except Exception:
            return None


def solve_lp_bound(b, prior, time_limit, target_t, row_indices=None):
    """Maximize prior^T a subject to a*b <= target_t on selected rows."""
    if sco is None:
        return None
    b = np.asarray(b, dtype=np.float64).ravel()
    n = b.size
    if n == 0 or time_limit <= 0:
        return None
    target_t = float(target_t)
    if not np.isfinite(target_t) or target_t < 0.0:
        return None

    if row_indices is None:
        rows = np.arange(2 * n - 1)
    else:
        rows = row_indices
    A, rows = build_convolution_matrix(b, rows, include_t=False)
    if len(rows) == 0:
        return None

    if prior is None:
        c = np.zeros(n, dtype=np.float64)
    else:
        c = -np.asarray(prior, dtype=np.float64).ravel()
        if c.size != n or not np.isfinite(c).all():
            c = np.zeros(n, dtype=np.float64)

    b_ub = np.full(len(rows), target_t, dtype=np.float64)
    A_eq = np.ones((1, n), dtype=np.float64)
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, 1.0)] * n

    res = run_linprog(c, A, b_ub, A_eq, b_eq, bounds, time_limit)
    if res is None or getattr(res, "x", None) is None:
        return None
    status = getattr(res, "status", 0)
    if status not in (0, 1):
        return None

    x = np.asarray(res.x, dtype=np.float64).ravel()[:n]
    if not np.isfinite(x).all():
        return None
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-15:
        return None
    return x / s


def solve_lp_min_t(b, prior, time_limit, eps=1e-7, row_indices=None):
    """Minimize max(a*b) on selected rows, with optional tiny linear prior."""
    if sco is None:
        return None, None
    b = np.asarray(b, dtype=np.float64).ravel()
    n = b.size
    if n == 0 or time_limit <= 0:
        return None, None

    if row_indices is None:
        rows = np.arange(2 * n - 1)
    else:
        rows = row_indices
    A, rows = build_convolution_matrix(b, rows, include_t=True)
    if len(rows) == 0:
        return None, None

    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    if prior is not None and eps != 0.0:
        p = np.asarray(prior, dtype=np.float64).ravel()
        if p.size == n and np.isfinite(p).all():
            c[:n] += float(eps) * p

    b_ub = np.zeros(len(rows), dtype=np.float64)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    bounds = [(0.0, 1.0)] * (n + 1)

    res = run_linprog(c, A, b_ub, A_eq, b_eq, bounds, time_limit)
    if res is None or getattr(res, "x", None) is None:
        return None, None
    status = getattr(res, "status", 0)
    if status not in (0, 1):
        return None, None

    x = np.asarray(res.x, dtype=np.float64).ravel()[:n]
    if not np.isfinite(x).all():
        return None, None
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-15:
        return None, None
    x = x / s
    t = float(res.x[n]) if res.x.size > n else 0.0
    if not np.isfinite(t):
        t = 0.0
    return x, t


def select_active_rows(values, top_k, extra, rng):
    values = np.asarray(values, dtype=np.float64).ravel()
    m = values.size
    if m == 0:
        return np.array([])
    K = min(int(top_k), m)
    if K <= 0:
        rows = np.arange(m)
    elif K < m:
        rows = np.argpartition(values, -K)[-K:]
    else:
        rows = np.arange(m)

    e = min(int(extra), m)
    if e > 0 and e < m:
        px = rng.choice(m, size=e, replace=False)
        rows = np.unique(np.concatenate((rows, px)))
    return rows


def alternating_lp(start, time_limit, rng, verbose=False, top_k=140):
    start = np.asarray(start, dtype=np.float64).ravel()
    if start.size == 0 or time_limit <= 0:
        return start, float("inf")

    best = normalize_prob(start)
    n = best.size
    best_conv = fast_convolve(best)
    best_M = float(np.max(best_conv))
    best_score = float(2.0 * n * best_M)

    b_explore = normalize_prob(symmetrize(best) + rng.standard_normal(n) * 1e-3)
    prev_x = None

    t0 = time.monotonic()
    it = 0
    no_imp = 0
    lp_time = 2.0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5.0:
            break

        if it % 4 < 2:
            use_direct = True
            b = best.copy()
        else:
            use_direct = False
            b = b_explore.copy()

        if use_direct:
            cross = best_conv
            Mcross = best_M
        else:
            cross = fast_crossconv(best, b)
            cross = np.maximum(cross, 0.0)
            Mcross = float(np.max(cross))

        if use_direct:
            prior = make_gradient_prior(best, best_conv, beta=80.0, K=top_k)
            if no_imp <= 5:
                target = best_M * (1.0 + 1e-12)
            else:
                target = best_M * (1.0 - min(0.04, 0.004 * no_imp))
            if n <= 900 or it % 3 == 0:
                rows = None
            else:
                rows = select_active_rows(best_conv, top_k, 20, rng)
        else:
            prior = make_prior(n, it, rng)
            if no_imp <= 5:
                target = Mcross * (1.0 + 1e-12)
            else:
                target = Mcross * (1.0 - min(0.04, 0.004 * no_imp))
            if n <= 768 and it % 5 == 0:
                rows = None
            else:
                rows = select_active_rows(cross, top_k, 30, rng)

        t_per = min(lp_time, rem * 0.35, 12.0)
        x = solve_lp_bound(b, prior, t_per, target, row_indices=rows)

        if x is None and target < Mcross * (1.0 + 1e-12):
            x = solve_lp_bound(b, prior, t_per, Mcross * (1.0 + 1e-9) + 1e-15, row_indices=rows)

        if x is None:
            rows_mt = None if (n <= 800 and it % 6 == 0) else rows
            xt, tt = solve_lp_min_t(b, prior, t_per, eps=1e-7, row_indices=rows_mt)
            if xt is not None:
                x = normalize_prob(xt)

        if x is None:
            no_imp += 1
        else:
            prev_x = x
            old_best = best.copy()
            improved = False

            candidates = [x, symmetrize(x)]
            for alpha in (0.1, 0.2, 0.35, 0.5, 0.7):
                candidates.append(blend_seqs(old_best, x, alpha))

            for cand in candidates:
                cand = normalize_prob(cand)
                sc = local_score(cand)
                if np.isfinite(sc) and sc < best_score - 1e-13:
                    best_score = sc
                    best = cand.copy()
                    improved = True

            if improved:
                best_conv = fast_convolve(best)
                best_M = float(np.max(best_conv))
                best_score = float(2.0 * n * best_M)
                no_imp = 0
                lp_time = max(1.0, min(lp_time * 0.9 + 0.1, 6.0))
                b_explore = normalize_prob(blend_seqs(best, x, 0.25) + rng.standard_normal(n) * 1e-5)
            else:
                no_imp += 1
                if no_imp >= 3:
                    lp_time = min(lp_time * 1.25, 12.0)
                if it % 3 == 0:
                    b_explore = normalize_prob(blend_seqs(b_explore, x, 0.18) + rng.standard_normal(n) * 1e-5)

        if no_imp > 22:
            pert = normalize_prob(best + rng.standard_normal(n) * 0.002)
            scp = local_score(pert)
            if np.isfinite(scp) and scp < best_score - 1e-12:
                best = pert.copy()
                best_score = scp
                best_conv = fast_convolve(best)
                best_M = float(np.max(best_conv))
            b_explore = normalize_prob(symmetrize(best) + rng.standard_normal(n) * 0.002)
            prev_x = None
            no_imp = 0
            lp_time = 2.0

        if verbose and it % 12 == 0:
            print(f"    ALP it={it} local={best_score:.8f} no_imp={no_imp}")

        it += 1

    return best, best_score


def smooth_mirror_descent(a0, time_limit, top_k=90, beta_init=60.0, lr_init=0.02, rng=None, verbose=False):
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    if a0.size == 0 or time_limit <= 0:
        return a0, local_score(a0)
    if rng is None:
        rng = np.random.default_rng(20240501)

    a = normalize_prob(a0)
    n = a.size
    c = fast_convolve(a)
    M = float(np.max(c))
    best_M = M
    best = a.copy()

    t0 = time.monotonic()
    beta = float(beta_init)
    lr = float(lr_init)
    no_imp = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        M = float(np.max(c))
        m = c.size
        K = min(int(top_k), m)
        if K < m:
            peaks = np.argpartition(c, -K)[-K:]
        else:
            peaks = np.arange(m)

        pv = c[peaks]
        scaled = pv * beta
        if scaled.size:
            scaled -= float(np.max(scaled))
            w = np.exp(scaled)
            ws = float(w.sum())
            if not np.isfinite(ws) or ws <= 1e-300:
                w = np.zeros_like(w)
                w[int(np.argmax(pv))] = 1.0
            else:
                w /= ws
        else:
            w = np.zeros(0)

        g = compute_subgradient(a, peaks, w)
        gn = float(np.linalg.norm(g))
        if not np.isfinite(gn):
            g = np.zeros(n)
            gn = 0.0

        best_lr = max(1e-9, lr * 0.05)
        best_trial = M * 1.01
        for trial in (lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05):
            if trial <= 0:
                continue
            logu = -trial * g
            logu = np.clip(logu, -50.0, 50.0)
            at = a * np.exp(logu)
            at = np.maximum(at, 1e-15)
            ss = float(at.sum())
            if ss <= 1e-12:
                continue
            at = at / ss
            tm = float(np.max(fast_convolve(at)))
            if tm <= best_trial:
                best_lr = trial
                best_trial = tm
                if tm < M * 0.999:
                    break

        logu = -best_lr * g
        logu = np.clip(logu, -50.0, 50.0)
        a = a * np.exp(logu)
        a = np.maximum(a, 1e-15)
        a = a / float(a.sum())

        c = fast_convolve(a)
        newM = float(np.max(c))
        if not np.isfinite(newM):
            a = normalize_prob(best + rng.standard_normal(n) * 0.001)
            newM = float(np.max(fast_convolve(a)))

        if newM < best_M - 1e-14:
            best_M = newM
            best = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        beta = min(beta * 1.002, 3000.0)
        if no_imp > 40:
            lr *= 0.7
            no_imp = 0
        lr = max(lr * 0.9995, 1e-9)

        if it % 250 == 0:
            sym = symmetrize(a)
            symM = float(np.max(fast_convolve(sym)))
            if symM < best_M - 1e-14:
                best_M = symM
                best = sym.copy()
                a = sym.copy()
                no_imp = 0

        if no_imp > 350 or (lr < 1e-9 and no_imp > 80):
            a = normalize_prob(best + rng.standard_normal(n) * (0.0008 + 0.002 * min(no_imp / 500, 1.0)))
            lr = max(1e-4, lr_init * 0.35)
            no_imp = 0

        if verbose and it % 200 == 0:
            print(f"    MD iter={it} M={best_M:.10f} score={2.0*n*best_M:.8f}")

        it += 1

    return best, float(2.0 * n * best_M)


def make_x_candidates(v):
    v = float(v)
    mults = np.array([0.05, 0.15, 0.3, 0.5, 0.7, 0.85, 1.0, 1.12, 1.28, 1.5, 2.0, 4.0])
    vals = np.concatenate((np.array([0.0, v]), v * mults))
    vals = np.clip(vals, 0.0, 1.0)
    vals = np.unique(np.round(vals, 12))
    if vals.size == 0:
        vals = np.array([0.0])
    return vals


def compute_c_for_x(C, a, j, v, x):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    den = 1.0 - float(v)
    if abs(den) < 1e-14:
        s = 0.0
    else:
        s = (1.0 - float(x)) / den
    c = (s * s) * C
    ks = j + np.arange(n)
    contrib = 2.0 * float(x) * s * a
    if 2 * j < m:
        contrib[j] = float(x) * float(x)
    c[ks] += contrib
    return c


def compute_x_maxs(C, a, j, v, X):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    X = np.asarray(X, dtype=np.float64).ravel()
    if X.size == 0:
        return np.array([])
    den = 1.0 - float(v)
    if abs(den) < 1e-14:
        return np.full(X.size, np.inf)
    s = (1.0 - X) / den
    base = (s * s).reshape(-1, 1) * C.reshape(1, -1)
    ks = j + np.arange(n)
    contrib = (2.0 * X).reshape(-1, 1) * s.reshape(-1, 1) * a.reshape(1, -1)
    if 2 * j < m:
        contrib[:, j] = X * X
    base[:, ks] += contrib
    return np.max(base, axis=1)


def fast_coordinate_descent(start, time_limit, refine=True, verbose=False):
    a = normalize_prob(start)
    n = a.size
    if n == 0 or time_limit <= 0:
        return a, local_score(a)

    t0 = time.monotonic()
    c = fast_convolve(a)
    c = np.maximum(c, 0.0)
    current_M = float(np.max(c))
    best_M = current_M
    best = a.copy()
    best_score = float(2.0 * n * best_M)
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        if rounds % 2 == 0:
            order = range(n)
        else:
            order = range(n - 1, -1, -1)

        improved_round = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            v = float(a[j])
            if v > 1.0 - 1e-8:
                continue

            C = c.copy()
            ks = j + np.arange(n)
            base = (2.0 * v) * a
            if 2 * j < 2 * n - 1:
                base[j] = v * v
            C[ks] -= base
            C = np.maximum(C, 0.0)

            X = make_x_candidates(v)
            if X.size == 0:
                continue
            maxs = compute_x_maxs(C, a, j, v, X)
            maxs = np.where(np.isfinite(maxs), maxs, np.inf)
            if maxs.size == 0:
                continue

            idx = int(np.argmin(maxs))
            if maxs[idx] >= current_M - 1e-14:
                continue

            best_x = float(X[idx])
            best_max = float(maxs[idx])

            if refine:
                step = max(1e-4, min(0.05, 0.25 * abs(best_x - v) + 0.005))
                offsets = np.array([-3.0, -2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0, 3.0]) * step
                Xf = np.concatenate(([best_x], best_x + offsets))
                Xf = np.clip(Xf, 0.0, 1.0)
                Xf = np.unique(np.round(Xf, 12))
                maxf = compute_x_maxs(C, a, j, v, Xf)
                maxf = np.where(np.isfinite(maxf), maxf, np.inf)
                if maxf.size:
                    idxf = int(np.argmin(maxf))
                    if maxf[idxf] < best_max - 1e-15:
                        best_x = float(Xf[idxf])
                        best_max = float(maxf[idxf])

            if best_max < current_M - 1e-14:
                c_new = compute_c_for_x(C, a, j, v, best_x)
                c_new = np.maximum(c_new, 0.0)
                ss_c = float(c_new.sum())
                if ss_c > 1e-15:
                    c_new = c_new / ss_c

                s = (1.0 - best_x) / (1.0 - v) if abs(1.0 - v) > 1e-14 else 0.0
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= s
                a[j] = best_x
                a = np.maximum(a, 0.0)

                ss_a = float(a.sum())
                if ss_a <= 1e-15:
                    a = normalize_prob(a)
                    c = fast_convolve(a)
                    current_M = float(np.max(c))
                else:
                    scale = 1.0 / ss_a
                    a = a * scale
                    c_new = c_new * (scale * scale)
                    c = c_new
                    current_M = float(np.max(c))

                if current_M < best_M - 1e-14:
                    best_M = current_M
                    best = a.copy()
                    best_score = float(2.0 * n * best_M)
                improved_round = True

        if not improved_round:
            c = fast_convolve(a)
            c = np.maximum(c, 0.0)
            current_M = float(np.max(c))
            if current_M < best_M - 1e-14:
                best_M = current_M
                best = a.copy()
                best_score = float(2.0 * n * best_M)
                improved_round = True
            else:
                sym = symmetrize(a)
                sc = local_score(sym)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best_M = sc / (2.0 * n)
                    best = sym.copy()
                    a = sym.copy()
                    c = fast_convolve(a)
                    current_M = float(np.max(c))
                    improved_round = True
                else:
                    if not np.allclose(a, best, rtol=1e-12, atol=1e-14):
                        blend = normalize_prob(0.5 * a + 0.5 * best)
                        sc = local_score(blend)
                        if np.isfinite(sc) and sc < best_score - 1e-12:
                            best_score = sc
                            best_M = sc / (2.0 * n)
                            best = blend.copy()
                            a = blend.copy()
                            c = fast_convolve(a)
                            current_M = float(np.max(c))
                            improved_round = True

            if not improved_round:
                break

        if verbose and rounds % 8 == 0:
            print(f"    CD round={rounds} local={best_score:.8f}")

    return best, best_score


def consider_candidate(a, best_seq, best_score, label=""):
    if a is None:
        return best_seq, best_score
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return best_seq, best_score
    if not np.isfinite(arr).all():
        arr = np.where(np.isfinite(arr), arr, 0.0)
    sc = local_score(arr)
    if np.isfinite(sc) and sc < best_score - 1e-15:
        best_seq = normalize_prob(arr)
        best_score = sc
        print(f"local [{label}]: {best_score:.8f} n={best_seq.size}")
    return best_seq, best_score


def consider_actual(a, best_seq, best_score, label=""):
    if a is None:
        return best_seq, best_score
    arr = normalize_prob(a)
    sc = actual_score(arr)
    if np.isfinite(sc) and sc < best_score - 1e-12:
        best_seq = arr.copy()
        best_score = sc
        print(f"ACTUAL [{label}]: {best_score:.8f} n={best_seq.size}")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(int(seed))
    budget = max(60.0, float(budget_s))
    horizon = time.monotonic() + max(10.0, budget - 60.0)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 0:
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.maximum(p, 0.0)
            if p.sum() > 1e-12:
                parent = normalize_prob(p)
                print(f"parent: n={parent.size}")
    except Exception:
        parent = None

    local_seq = None
    local_score_val = float("inf")
    actual_seq = None
    actual_score_val = float("inf")

    if parent is not None:
        local_seq, local_score_val = consider_candidate(parent, local_seq, local_score_val, "parent")
        actual_seq, actual_score_val = consider_actual(parent, actual_seq, actual_score_val, "parent")

    base_start = parent if parent is not None else make_gaussian(768, 0.3)
    local_seq, local_score_val = consider_candidate(base_start, local_seq, local_score_val, "base")

    sweep_n = [256, 320, 384, 512, 640, 768, 896, 1024, 1152, 1280, 1536, 1792, 2048]
    for nn in sweep_n:
        if time.monotonic() > horizon - 180.0:
            break
        cand = resample_sequence(base_start, nn)
        local_seq, local_score_val = consider_candidate(cand, local_seq, local_score_val, f"res_{nn}")
        cand = symmetrize(cand)
        local_seq, local_score_val = consider_candidate(cand, local_seq, local_score_val, f"res_sym_{nn}")

    for nn in [512, 768, 1024, 1536]:
        for sig in [0.15, 0.25, 0.35]:
            if time.monotonic() > horizon - 120.0:
                break
            cand = make_gaussian(nn, sig)
            local_seq, local_score_val = consider_candidate(cand, local_seq, local_score_val, f"gauss_{nn}_{sig}")

    if local_seq is None:
        local_seq = make_gaussian(768, 0.3)
        local_score_val = local_score(local_seq)

    actual_seq, actual_score_val = consider_actual(local_seq, actual_seq, actual_score_val, "initial")

    rem = horizon - time.monotonic()
    if rem > 80.0:
        t_md = min(120.0, rem * 0.10)
        a, sc = smooth_mirror_descent(local_seq, t_md, top_k=100, beta_init=60.0, lr_init=0.015, rng=rng, verbose=False)
        local_seq, local_score_val = consider_candidate(a, local_seq, local_score_val, "md0")
        actual_seq, actual_score_val = consider_actual(local_seq, actual_seq, actual_score_val, "md0")

    rem = horizon - time.monotonic()
    if rem > 120.0:
        t_lp = min(rem * 0.42, 780.0)
        a, sc = alternating_lp(local_seq, t_lp, rng, verbose=True)
        local_seq, local_score_val = consider_candidate(a, local_seq, local_score_val, "alp1")
        actual_seq, actual_score_val = consider_actual(local_seq, actual_seq, actual_score_val, "alp1")

    rem = horizon - time.monotonic()
    if rem > 60.0:
        t_cd = min(rem * 0.22, 320.0)
        a, sc = fast_coordinate_descent(local_seq, t_cd, refine=True, verbose=True)
        local_seq, local_score_val = consider_candidate(a, local_seq, local_score_val, "cd1")
        actual_seq, actual_score_val = consider_actual(local_seq, actual_seq, actual_score_val, "cd1")

    # Larger-length attempts.
    rem = horizon - time.monotonic()
    if rem > 500.0 and local_seq is not None:
        current_n = int(local_seq.size)
        cand_ns = []
        for factor in (1.25, 1.5, 1.75, 2.0):
            nn = int((current_n * factor) // 32) * 32
            if nn > current_n + 64 and nn <= 2048:
                cand_ns.append(nn)
        if current_n < 1024:
            cand_ns.append(1024)
        if current_n < 1536 and rem > 800.0:
            cand_ns.append(1536)

        seen = set()
        unique_ns = []
        for nn in cand_ns:
            nn = int(nn)
            if nn not in seen:
                seen.add(nn)
                unique_ns.append(nn)
        unique_ns.sort()

        for nn in unique_ns:
            rem = horizon - time.monotonic()
            if rem < 220.0:
                break

            tmp_seq = None
            tmp_score = float("inf")
            base_big = resample_sequence(local_seq, nn)
            tmp_seq, tmp_score = consider_candidate(base_big, tmp_seq, tmp_score, f"big_{nn}")
            tmp_seq, tmp_score = consider_candidate(symmetrize(base_big), tmp_seq, tmp_score, f"big_sym_{nn}")
            tmp_seq, tmp_score = consider_candidate(base_big + rng.standard_normal(nn) * 0.002, tmp_seq, tmp_score, f"big_rand_{nn}")
            if parent is not None:
                pp = resample_sequence(parent, nn)
                tmp_seq, tmp_score = consider_candidate(symmetrize(pp), tmp_seq, tmp_score, f"big_parent_{nn}")
            if tmp_seq is None:
                tmp_seq = make_gaussian(nn, 0.25)

            t_opt = min(rem * 0.35, 360.0)
            a, sc = alternating_lp(tmp_seq, t_opt, rng, verbose=True)
            local_seq, local_score_val = consider_candidate(a, local_seq, local_score_val, f"big_alp_{nn}")
            actual_seq, actual_score_val = consider_actual(local_seq, actual_seq, actual_score_val, f"big_alp_{nn}")

            rem = horizon - time.monotonic()
            if rem > 60.0:
                t_cd = min(rem * 0.16, 120.0)
                a, sc = fast_coordinate_descent(a, t_cd, refine=True, verbose=False)
                local_seq, local_score_val = consider_candidate(a, local_seq, local_score_val, f"big_cd_{nn}")
                actual_seq, actual_score_val = consider_actual(local_seq, actual_seq, actual_score_val, f"big_cd_{nn}")

    rem = horizon - time.monotonic()
    if rem > 60.0 and local_seq is not None:
        t_lp = min(rem * 0.25, 220.0)
        a, sc = alternating_lp(local_seq, t_lp, rng, verbose=False)
        local_seq, local_score_val = consider_candidate(a, local_seq, local_score_val, "alp2")
        actual_seq, actual_score_val = consider_actual(local_seq, actual_seq, actual_score_val, "alp2")

    rem = horizon - time.monotonic()
    if rem > 40.0 and local_seq is not None:
        t_cd = min(rem * 0.35, 240.0)
        a, sc = fast_coordinate_descent(local_seq, t_cd, refine=True, verbose=False)
        local_seq, local_score_val = consider_candidate(a, local_seq, local_score_val, "cd2")
        actual_seq, actual_score_val = consider_actual(local_seq, actual_seq, actual_score_val, "cd2")

    rem = horizon - time.monotonic()
    if rem > 30.0 and local_seq is not None:
        t_md = min(rem * 0.35, 150.0)
        a, sc = smooth_mirror_descent(local_seq, t_md, top_k=60, beta_init=500.0, lr_init=0.004, rng=rng, verbose=False)
        local_seq, local_score_val = consider_candidate(a, local_seq, local_score_val, "md2")
        actual_seq, actual_score_val = consider_actual(local_seq, actual_seq, actual_score_val, "md2")

    rem = horizon - time.monotonic()
    if rem > 20.0 and local_seq is not None:
        t_cd = min(rem * 0.5, 150.0)
        a, sc = fast_coordinate_descent(local_seq, t_cd, refine=False, verbose=False)
        local_seq, local_score_val = consider_candidate(a, local_seq, local_score_val, "cd_final")
        actual_seq, actual_score_val = consider_actual(local_seq, actual_seq, actual_score_val, "cd_final")

    if local_seq is not None:
        sym = symmetrize(local_seq)
        local_seq, local_score_val = consider_candidate(sym, local_seq, local_score_val, "final_sym")
        actual_seq, actual_score_val = consider_actual(sym, actual_seq, actual_score_val, "final_sym")

    if local_seq is not None:
        actual_seq, actual_score_val = consider_actual(local_seq, actual_seq, actual_score_val, "final_local")

    out = actual_seq if actual_seq is not None else local_seq
    if out is None:
        out = np.ones(768, dtype=np.float64) / 768.0
    out = normalize_prob(out)

    final_val = actual_score(out)
    if not np.isfinite(final_val):
        final_val = float("inf")

    print(f"FINAL local={local_score_val:.8f} actual={final_val:.8f} n={out.size}")
    return [float(x) for x in out]
```
