def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:

2. FISTA (Nesterov accelerated gradient) with backtracking on smooth approximation

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_h(a, w):
    """Compute h[i] = sum_k w[k] * a_contribution(i, k) via FFT."""
    n = len(a)
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    return conv_result[n - 1:2 * n - 1][::-1]


def smooth_obj_grad(a, tau):
    """Compute smooth objective value and gradient."""
    conv = fast_convolve(a)
    cmax = conv.max()
    scaled = (conv - cmax) / tau
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum
    F = cmax + tau * np.log(w_sum)
    h = compute_h(a, w)
    grad = (2.0 / tau) * h
    return F, grad, conv


def make_gaussian(n, center_frac, sigma_frac):
    """Create a Gaussian-like sequence."""
    x = np.arange(n)
    mu = center_frac * n
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_double_gaussian(n, c1, s1, c2, s2, w1, w2):
    """Sum of two Gaussians for more flexible shapes."""
    x = np.arange(n)
    a = w1 * np.exp(-0.5 * ((x - c1 * n) / (s1 * n)) ** 2) + \
        w2 * np.exp(-0.5 * ((x - c2 * n) / (s2 * n)) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_exponential(n, center_frac, decay_frac):
    """Exponential decay from a center point."""
    x = np.arange(n)
    mu = center_frac * n
    a = np.exp(-np.abs(x - mu) / (decay_frac * n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def fista_optimize(a0, time_limit, tau_init=None, verbose=False):
    """
    FISTA (Nesterov accelerated gradient) with backtracking line search
    on the log-sum-exp smooth approximation of max-convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    if tau_init is None:
        tau = 0.06 * conv.max()
    else:
        tau = tau_init

    a_prev = a.copy()
    t_k = 1.0
    step = 1.0 / (tau * 2.0 * np.max(conv))
    best_obj = conv.max()
    best_a = a.copy()
    no_improve = 0
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + (t_k - 1.0) / t_new * (a - a_prev)
        y = project_simplex(y)

        # Gradient at y
        conv_y = fast_convolve(y)
        cmax_y = conv_y.max()
        scaled = (conv_y - cmax_y) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        w /= w_sum
        h = compute_h(y, w)
        grad = (2.0 / tau) * h

        # Backtracking line search
        alpha = step
        f_curr = cmax_y + tau * np.log(w_sum)
        lippo = f_curr - np.dot(grad, y)
        while alpha > 1e-8:
            trial = y - alpha * grad
            trial = project_simplex(trial)
            trial_conv = fast_convolve(trial)
            tcmax = trial_conv.max()
            tscaled = (trial_conv - tcmax) / tau
            tw = np.exp(tscaled)
            tw_sum = tw.sum()
            trial_f = tcmax + tau * np.log(tw_sum)
            if trial_f <= lippo - 0.5 * alpha * alpha * np.dot(grad, grad):
                break
            alpha *= 0.5

        a_new = y - alpha * grad
        a_new = project_simplex(a_new)

        # Check if step was too small; if so, use larger step
        if alpha < step * 0.01:
            alpha = step
            a_new = y - alpha * grad
            a_new = project_simplex(a_new)

        # Update
        conv_a = fast_convolve(a_new)
        M = conv_a.max()
        if M < best_obj - 1e-13:
            best_obj = M
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a_prev = a
        a = a_new
        t_k = t_new

        # Cooling schedule: fast initially, then slow
        if it < 500:
            tau *= 0.9995
        else:
            tau *= 0.9999
        if tau < 5e-7:
            tau = 5e-7

        # Update step size
        conv_a = fast_convolve(a)
        step = 1.0 / (tau * 2.0 * max(conv_a.max(), 1e-10))

        # Reheat strategy
        if no_improve > 300:
            tau = 0.04 * fast_convolve(best_a).max()
            rng = np.random.default_rng(it * 13 + 5)
            noise = 0.002 + 0.004 * min(no_improve / 1000.0, 2.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            a_prev = a.copy()
            t_k = 1.0
            no_improve = 0
            step = 1.0 / (tau * 2.0 * max(fast_convolve(a).max(), 1e-10))
            if verbose:
                print(f"    FISTA reheat at iter {it}, tau={tau:.2e}")

        if verbose and it % 500 == 0:
            print(f"    FISTA iter {it}: obj={2*n*best_obj:.8f}, tau={tau:.2e}")

    return best_a, 2.0 * n * best_obj


def mirror_descent_optimize(a0, time_limit, verbose=False):

    Mirror descent (multiplicative/exponential updates) on the smooth approximation.
    Uses relative entropy Bregman divergence, naturally maintaining non-negativity.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    tau = 0.10 * conv.max()
    lr = 0.3
    best_obj = conv.max()
    best_a = a.copy()
    no_improve = 0
    t0 = time.monotonic()
    it = 0
    momentum_a = a.copy()
    beta = 0.7

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv_a = fast_convolve(a)
        M = conv_a.max()
        scaled = (conv_a - M) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        w /= w_sum
        h = compute_h(a, w)
        grad = (2.0 / tau) * h

        momentum_a = beta * momentum_a + (1.0 - beta) * a

        a_new = momentum_a * update
        a_new_sum = a_new.sum()

        if M < best_obj - 1e-13:

            best_a = a.copy()

        else:
            no_improve += 1

            a = best_a.copy()
            momentum_a = a.copy()
            no_improve = 0
            if verbose:
                print(f"    Mirror reheat at iter {it}")

        if verbose and it % 1000 == 0:
            print(f"    Mirror iter {it}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, lr={lr:.4f}")

    return best_a, 2.0 * n * best_obj


def peak_directed_cd(a0, time_limit, verbose=False):
    """
    Aggressive peak-directed coordinate descent.
    Identifies top peaks, finds contributing pairs, and makes
    targeted moves to reduce the maximum convolution value.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        improved = False

        # Get top peaks
        top_k = min(3, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = a[idx] * a[pk - idx]

            # Find the dominant contributing pair
            top_pairs = np.argsort(products)[-min(5, len(idx)):]
            for pi in top_pairs:
                j = idx[pi]
                k_other = pk - j
                if j < 0 or j >= n or k_other < 0 or k_other >= n:
                    continue

                # Try reducing a[j] and redistributing
                for factor in [0.5, 0.7, 0.85]:
                    diff = a[j] * (1.0 - factor)
                    if diff < 1e-12:
                        continue
                    trial = a.copy()
                    trial[j] -= diff
                    # Redistribute proportional to a[i] for non-contributing i
                    mask = np.ones(n, dtype=bool)
                    # Mark elements that contribute to this peak
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    mask[js2:je2 + 1] = False
                    mask[j] = False
                    if mask.sum() > 0 and a[mask].sum() > 1e-15:
                        trial[mask] += diff * a[mask] / a[mask].sum()
                    trial = np.maximum(trial, 0.0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial /= tsum
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        a = trial
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10:
                # Larger perturbation
                rng = np.random.default_rng(it * 7 + 3)
                noise_scale = 0.003 + 0.004 * min(no_improve / 50.0, 3.0)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    PeakCD iter {it}: obj={2*n*best_obj:.8f}")

    return best_a, 2.0 * n * best_obj


def lp_direction_search(a0, time_limit, verbose=False):
    """
    LP-based Frank-Wolfe with relaxed constraints.
    Solves LP to find direction g that maximizes sum(g) subject to
    (a*g)[k] <= M * relaxation for all k, then does line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    relax = 1.0

    from scipy.optimize import linprog

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0
        # (a*g)[k] = sum_j g[j] * a[k-j] (cross-correlation)
        # Build constraint matrix: for each k, sum_j A[k,j] * g[j] <= M*relax
        # A[k,j] = a[k-j] if 0 <= k-j < n else 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = a
        b_ub = np.full(2 * n - 1, M * relax)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': 10})
            g = res.x if res.success else None
        except Exception:
            g = None

        if g is None or g.sum() < 1e-15:
            relax *= 1.15
            if relax > 5.0:
                rng = np.random.default_rng(it * 17)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(a)
                M = conv.max()
                relax = 1.0
            continue

        g = g / g.sum()

        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            relax = min(relax * 1.1, 4.0)
            if no_improve > 50:
                rng = np.random.default_rng(it * 23 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(a)
                M = conv.max()
                relax = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = np.maximum(a + alpha * (g - a), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Refine with bisection
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = np.maximum(a + mid * (g - a), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            a_new = np.maximum(a + best_alpha * (g - a), 0.0)
            a_new /= a_new.sum()
            conv = fast_convolve(a_new)
            M = conv.max()
            a = a_new
            relax = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                if verbose and it % 5 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            relax = min(relax * 1.1, 4.0)

        if no_improve > 80:
            break

    return best_a, 2.0 * n * best_obj


def parameter_sweep(remaining_time, best_seq_init=None):
    """
    Quick sweep over shape parameters to find diverse good initializations.
    Returns list of (sequence, score) sorted by score.
    """
    candidates = []
    t0 = time.monotonic()

    # Parent sequence if available
    if best_seq_init is not None:
        p = np.asarray(best_seq_init, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            pn = p / p.sum()
            sc = float(evaluate_sequence(pn.tolist()))
            if np.isfinite(sc):
                candidates.append((pn, sc))

    n_values = [512, 640, 768, 896, 1024]

    for n in n_values:
        if time.monotonic() - t0 > remaining_time * 0.7:
            break
        # Vary center and width
        for center in [0.20, 0.25, 0.30, 0.35, 0.40]:
            for width in [0.12, 0.15, 0.18, 0.22, 0.28, 0.35]:
                if time.monotonic() - t0 > remaining_time * 0.8:
                    break
                a = make_gaussian(n, center, width)
                sc = float(evaluate_sequence(a.tolist()))
                if np.isfinite(sc):
                    candidates.append((a, sc))

    # Some double Gaussians
    for n in [640, 768]:
        if time.monotonic() - t0 > remaining_time * 0.85:
            break
        for c1, s1, c2, s2, w1, w2 in [
            (0.25, 0.12, 0.6, 0.08, 0.7, 0.3),
            (0.2, 0.15, 0.5, 0.1, 0.8, 0.2),
            (0.3, 0.1, 0.55, 0.12, 0.6, 0.4),
        ]:
            a = make_double_gaussian(n, c1, s1, c2, s2, w1, w2)
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc):
                candidates.append((a, sc))

    # Exponential shapes
    for n in [512, 640, 768]:
        if time.monotonic() - t0 > remaining_time:
            break
        for center in [0.3, 0.4]:
            for decay in [0.15, 0.25, 0.35]:
                a = make_exponential(n, center, decay)
                sc = float(evaluate_sequence(a.tolist()))
                if np.isfinite(sc):
                    candidates.append((a, sc))

    candidates.sort(key=lambda x: x[1])
    return candidates


def minimax_game(a0, time_limit, verbose=False):
    """
    Minimax game approach: alternate between
    - Inner: minimize f^T Q(p) f over simplex (via projected gradient)
    - Outer: multiplicative weight update on p (distribution over lags)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv_len = 2 * n - 1
    p = np.ones(conv_len) / conv_len
    lr_p = 0.1
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Inner: minimize a^T Q(p) a via a few projected gradient steps
        inner_iters = max(5, 30 - it // 50)
        inner_lr = 2.0 / (max(fast_convolve(a).max(), 1e-10) / max(p.min(), 1e-15) + 1.0)
        inner_lr = min(inner_lr, 1.0 / (n * 0.5))

        for _ in range(inner_iters):
            # Gradient of a^T Q(p) a: 2 * (Q(p) a)
            # (Q(p) a)[j] = sum_k p[k] * a_contribution for lag k
            p_rev = p[::-1]
            total_len = n + conv_len - 1
            fft_size = 1 << (total_len - 1).bit_length()
            A = np.fft.rfft(a, fft_size)
            P = np.fft.rfft(p_rev, fft_size)
            conv_r = np.fft.irfft(A * P, fft_size)
            h = conv_r[n - 1:2 * n - 1][::-1]
            grad = 2.0 * h
            a = a - inner_lr * grad
            a = project_simplex(a)

        conv_a = fast_convolve(a)
        M = conv_a.max()
        if M < best_obj - 1e-13:
            best_obj = M
            best_a = a.copy()
            if verbose and it % 100 == 0:
                print(f"    Minimax iter {it}: obj={2*n*best_obj:.8f}")

        # Outer: multiplicative weight update
        scaled = conv_a / (0.03 * M + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        p = (1.0 - lr_p) * p + lr_p * w
        p /= p.sum()

        # Adapt lr_p
        if it % 200 == 0:
            lr_p = max(0.02, lr_p * 0.93)

    return best_a, 2.0 * n * best_obj


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phases:
    1. Parameter sweep for diverse initializations (~100s)
    2. FISTA main optimization (~500s)
    3. Mirror descent complementary (~250s)
    4. Peak-directed CD for escaping local optima (~250s)
    5. LP-FW refinement (~200s)
    6. Minimax game (~150s)
    7. Final FISTA polish (~100s)
    8. Multi-length exploration (~remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Parameter sweep ===
    print("=== Phase 1: Parameter Sweep ===")
    t1 = min(100.0, (deadline - time.monotonic()) * 0.06)
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score):
                try_seq(parent, "parent")
    except Exception:
        pass

    candidates = parameter_sweep(t1, parent)
    print(f"  Found {len(candidates)} candidates, best={candidates[0][1]:.6f}" if candidates else "  No candidates")
    for a_cand, sc in candidates[:5]:
        if sc < best_score:
            try_seq(a_cand, f"sweep_n{len(a_cand)}")

    if best_seq is None:
        best_seq = make_gaussian(640, 0.30, 0.20)
        try_seq(best_seq, "fallback")

    print(f"  After sweep: best={best_score:.6f}, n={len(best_seq)}")

    # === Phase 2: FISTA main optimization ===
    print(f"\n=== Phase 2: FISTA Main (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic() - 800.0
    t2 = max(60.0, min(rem, 500.0))
    if t2 > 60:
        print(f"  Time: {t2:.0f}s")
        tau_init = 0.05 * fast_convolve(best_seq).max()
        a_opt, obj = fista_optimize(best_seq, time_limit=t2, tau_init=tau_init, verbose=True)
        try_seq(a_opt, "fista_main")

        # Second FISTA from perturbed best
        if time.monotonic() < deadline - 700:
            rem2 = min(deadline - time.monotonic() - 700, 200)
            if rem2 > 50:
                pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.008)
                tau_init2 = 0.04 * fast_convolve(pert).max()
                a_opt2, obj2 = fista_optimize(pert, time_limit=rem2, tau_init=tau_init2)
                try_seq(a_opt2, "fista_pert")

    # === Phase 3: Mirror descent ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    rem = deadline - time.monotonic() - 600.0
    t3 = max(50.0, min(rem, 250.0))
    if t3 > 50:
        print(f"  Time: {t3:.0f}s")
        a_opt, obj = mirror_descent_optimize(best_seq, time_limit=t3, verbose=True)
        try_seq(a_opt, "mirror")

    # === Phase 4: Peak-directed CD ===
    print(f"\n=== Phase 4: Peak-Directed CD ===")
    rem = deadline - time.monotonic() - 450.0
    t4 = max(30.0, min(rem, 250.0))
    if t4 > 30:
        print(f"  Time: {t4:.0f}s")
        a_opt, obj = peak_directed_cd(best_seq, time_limit=t4, verbose=True)
        try_seq(a_opt, "peak_cd")

        # Run from perturbed version too
        if time.monotonic() < deadline - 350:
            rem2 = min(deadline - time.monotonic() - 350, 100)
            if rem2 > 30:
                pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.005)
                a_opt2, obj2 = peak_directed_cd(pert, time_limit=rem2)
                try_seq(a_opt2, "peak_cd_pert")

    # === Phase 5: LP-FW refinement ===
    print(f"\n=== Phase 5: LP-FW ===")
    rem = deadline - time.monotonic() - 250.0
    t5 = max(30.0, min(rem, 200.0))
    if t5 > 30:
        print(f"  Time: {t5:.0f}s")
        a_opt, obj = lp_direction_search(best_seq, time_limit=t5, verbose=True)
        try_seq(a_opt, "lpfw")

    # === Phase 6: Minimax game ===
    print(f"\n=== Phase 6: Minimax Game ===")
    rem = deadline - time.monotonic() - 150.0
    t6 = max(30.0, min(rem, 150.0))
    if t6 > 30:
        print(f"  Time: {t6:.0f}s")
        a_opt, obj = minimax_game(best_seq, time_limit=t6, verbose=True)
        try_seq(a_opt, "minimax")

    # === Phase 7: Final FISTA polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    rem = deadline - time.monotonic() - 80.0
    t7 = max(20.0, min(rem, 100.0))
    if t7 > 20:
        tau_init = 0.015 * fast_convolve(best_seq).max()
        a_opt, obj = fista_optimize(best_seq, time_limit=t7, tau_init=tau_init, verbose=False)
        try_seq(a_opt, "final_fista")

    # === Phase 8: Multi-length exploration ===
    print(f"\n=== Phase 8: Multi-Length ===")
    rem = deadline - time.monotonic() - 40.0
    if rem > 40:
        base = best_seq
        for target_n in [1024, 896, 640, 512]:
            if time.monotonic() > deadline - 30:
                break
            if len(base) == target_n:
                continue
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score:
                # Quick FISTA on the new length
                rem2 = min(deadline - time.monotonic() - 20, 60)
                if rem2 > 30:
                    tau_init = 0.03 * fast_convolve(f_interp).max()
                    a_opt, obj = fista_optimize(f_interp, time_limit=rem2, tau_init=tau_init)
                    try_seq(a_opt, f"fista_n{target_n}")

    # === Phase 9: Final squeeze with very low temperature ===
    rem = deadline - time.monotonic() - 10.0
    if rem > 15:
        tau_init = 0.008 * fast_convolve(best_seq).max()
        a_opt, obj = fista_optimize(best_seq, time_limit=rem, tau_init=tau_init)
        try_seq(a_opt, "final_squeeze")

    # Ensure we have a valid result
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
