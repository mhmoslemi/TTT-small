"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using asymmetric initializations
and adaptive multiplicative descent with momentum and coordinate refinement.
Key insight: asymmetric sequences can break the Cauchy-Schwarz barrier at
the center peak, allowing lower maximum convolution values.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def make_linear_ramp(n, reverse=False):
    a = np.linspace(0.1, 1.0, n)
    if reverse:
        a = a[::-1]
    a = a / a.sum()
    return a

def make_step(n, frac=0.5):
    a = np.ones(n)
    split = int(n * frac)
    a[:split] = 1.0
    a[split:] = 0.3
    a = a / a.sum()
    return a

def make_exp_decay(n, rate=0.5):
    a = np.exp(-rate * np.arange(n) / n)
    a = a / a.sum()
    return a

def make_quadratic_ramp(n):
    x = np.arange(n) / (n - 1)
    a = 0.2 + 0.8 * x ** 2
    a = a / a.sum()
    return a

def make_cos_asym(n, freq=2, amp=0.3):
    x = np.arange(n) / n
    a = 1.0 + amp * np.cos(2 * np.pi * freq * x + np.pi/4)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

def multiplicative_descent_v2(a0, time_limit, seed=0, verbose=False):
    """
    Adaptive multiplicative descent with:
    - Momentum (EMA of subgradients)
    - Adaptive learning rate
    - Top-K peak selection with annealing
    - Periodic perturbation
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    t0 = time.monotonic()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    lr = 0.003
    momentum = 0.7
    sg_ema = np.zeros(n)
    top_k_start = 10
    top_k_end = 50
    total_iters = 5000
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break
        
        # Progress fraction for annealing
        progress = min(1.0, it / total_iters)
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        top_k = min(top_k, len(a) * 2 - 1)
        
        # Compute convolution and find top peaks
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        # Select top peaks
        if top_k >= len(conv):
            peak_idx = np.arange(len(conv))
            peak_vals = conv
        else:
            peak_idx = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peak_idx]
        
        # Weight peaks by softmax (temperature anneals down)
        temp = 0.001 + 0.01 * (1 - progress)
        scaled = (peak_vals - peak_vals.max()) / temp
        weights = np.exp(scaled)
        weights /= weights.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if weights[i] < 1e-10:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * weights[i] * a[k - idx]
        
        # Update EMA
        sg_ema = momentum * sg_ema + (1 - momentum) * sg
        
        # Multiplicative update
        a = a * np.exp(-lr * sg_ema)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Evaluate
        new_obj = fast_convolve(a).max()
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            # Increase LR slightly
            lr = min(lr * 1.01, 0.01)
            if verbose and it % 100 == 0:
                print(f"      MD2 iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1
            lr *= 0.999
            if no_improve % 100 == 0 and no_improve > 200:
                # Perturbation
                noise = rng.standard_normal(n) * 0.005
                a = np.maximum(best_a + noise, 1e-15)
                a = a / a.sum()
                sg_ema *= 0.5
                no_improve = 0
        
        # Decay learning rate
        lr *= 0.99995
        
        if lr < 1e-7:
            break
    
    return best_a, best_obj


def coordinate_descent_refine(a0, time_limit, verbose=False):
    """
    Coordinate descent that exploits the monotonicity of the objective
    in each coordinate. Uses a simple strategy: identify the worst peak,
    reduce the most contributing coordinate, redistribute mass.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    t0 = time.monotonic()
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 30:
            break
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        peak_k = int(np.argmax(conv))
        
        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib = np.arange(js, je + 1)
        if len(contrib) == 0:
            break
        
        products = a[contrib] * a[peak_k - contrib]
        max_idx = contrib[np.argmax(products)]
        
        # Try reducing max_idx and redistributing
        best_trial = None
        best_trial_obj = cur_obj
        
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            diff = a[max_idx] * frac
            if diff < 1e-15:
                continue
            trial = a.copy()
            trial[max_idx] -= diff
            if trial[max_idx] < 0:
                continue
            
            # Redistribute to non-contributing indices (weighted by current value)
            mask = np.ones(n, dtype=bool)
            mask[contrib] = False
            if mask.sum() == 0:
                mask[:] = True
                mask[max_idx] = False
            
            # Also try redistributing to the "other side" of the convolution
            # (indices that contribute to a different, smaller peak)
            nc_vals = trial[mask]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                # Distribute uniformly
                trial[mask] += diff / mask.sum()
            else:
                # Weighted distribution favoring smaller values
                weights_redist = 1.0 / (nc_vals + 0.01 * nc_sum)
                weights_redist /= weights_redist.sum()
                trial[mask] += diff * weights_redist
            
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-15:
                continue
            trial = trial / s
            
            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-14:
                best_trial_obj = obj
                best_trial = trial
        
        if best_trial is not None:
            a = best_trial
            if best_trial_obj < best_obj - 1e-12:
                best_obj = best_trial_obj
                best_a = a.copy()
                no_improve = 0
                if verbose:
                    print(f"      CD: peak={peak_k}, obj={best_obj:.10f}")
        else:
            no_improve += 1
            # Try a random coordinate perturbation
            i = np.random.randint(n)
            for frac in [0.05, 0.1, 0.2]:
                diff = a[i] * frac
                if diff < 1e-15:
                    continue
                trial = a.copy()
                trial[i] -= diff
                # Distribute to a random set of coordinates
                targets = np.random.choice(n, size=min(10, n), replace=False)
                targets = targets[targets != i]
                if len(targets) == 0:
                    continue
                trial[targets] += diff / len(targets)
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-12:
                    a = trial
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                    break
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_obj


def subgradient_descent_v2(a0, time_limit, seed=0, verbose=False):
    """
    Improved subgradient descent with:
    - Nesterov acceleration
    - Adaptive step size (Polyak-like)
    - Focus on top peaks with exponential weighting
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    t0 = time.monotonic()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    # Nesterov momentum
    y = a.copy()
    beta = 0.9
    lr = 0.005
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 150:
            # Restart with perturbation
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 1e-15)
            a = a / a.sum()
            y = a.copy()
            lr = 0.005
            no_improve = 0
            continue
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        peak_k = int(np.argmax(conv_y))
        
        # Subgradient: partner values at the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        sg = np.zeros(n)
        sg[idx] = 2.0 * y[peak_k - idx]
        
        # Also consider second and third peaks
        top3 = np.argsort(conv_y)[-3:]
        for k2 in top3:
            if k2 == peak_k:
                continue
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            idx2 = np.arange(js2, je2 + 1)
            sg[idx2] += y[k2 - idx2]
        
        # Projected gradient step
        a_new = y - lr * sg
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Nesterov update
        beta = 0.5 * (1 + np.sqrt(1 + 4 * beta * beta))
        y = a_new + (beta - 1) / beta * (a_new - a)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        a = a_new
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      SGD2 iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        lr *= 0.9999
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using asymmetric initializations,
adaptive multiplicative descent with momentum, and coordinate descent refinement.
Key insight: breaking symmetry allows the center convolution peak to drop below
the Cauchy-Schwarz barrier (1/n) that constrains symmetric sequences.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_linear_ramp(n, reverse=False):
    a = np.linspace(0.1, 1.0, n)
    if reverse:
        a = a[::-1]
    a = a / a.sum()
    return a


def make_step(n, frac=0.5, high=1.0, low=0.3):
    a = np.ones(n)
    split = int(n * frac)
    a[:split] = high
    a[split:] = low
    a = a / a.sum()
    return a


def make_exp_decay(n, rate=0.5):
    a = np.exp(-rate * np.arange(n) / n)
    a = a / a.sum()
    return a


def make_quadratic_ramp(n, power=2):
    x = np.arange(n) / max(n - 1, 1)
    a = 0.2 + 0.8 * x ** power
    a = a / a.sum()
    return a


def make_cos_asym(n, freq=2, amp=0.3, phase=np.pi / 4):
    x = np.arange(n) / n
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def make_bilinear(n, alpha=0.3):
    """Bilinear mix of ramp and uniform."""
    ramp = make_linear_ramp(n)
    uniform = np.ones(n) / n
    a = (1 - alpha) * uniform + alpha * ramp
    a = a / a.sum()
    return a


def multiplicative_descent_v2(a0, time_limit, seed=0, verbose=False):
    """
    Adaptive multiplicative descent with momentum, annealed top-K peaks,
    and periodic perturbation to escape local minima.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    t0 = time.monotonic()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    lr = 0.004
    momentum = 0.75
    sg_ema = np.zeros(n)
    top_k_min = 5
    top_k_max = 80
    anneal_iter = 3000
    it = 0
    no_improve = 0
    best_lr = 0.004

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        progress = min(1.0, it / anneal_iter)
        top_k = int(top_k_min + (top_k_max - top_k_min) * progress)
        top_k = min(top_k, 2 * n - 1)

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if top_k >= len(conv):
            peak_idx = np.arange(len(conv))
            peak_vals = conv
        else:
            peak_idx = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peak_idx]

        temp = 0.0005 + 0.02 * (1.0 - progress) ** 2
        scaled = (peak_vals - peak_vals.max()) / temp
        weights = np.exp(scaled)
        weights /= weights.sum()

        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if weights[i] < 1e-10:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * weights[i] * a[k - idx]

        sg_ema = momentum * sg_ema + (1.0 - momentum) * sg

        a = a * np.exp(-lr * sg_ema)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        new_obj = fast_convolve(a).max()
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            best_lr = lr
            lr = min(lr * 1.005, 0.015)
            if verbose and it % 200 == 0:
                print(f"      MD2 iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, topk={top_k}")
        else:
            no_improve += 1
            lr *= 0.998
            if no_improve > 300:
                noise_scale = 0.003 + 0.005 * min(1.0, no_improve / 1000.0)
                noise = rng.standard_normal(n) * noise_scale
                a = np.maximum(best_a + noise, 1e-15)
                a = a / a.sum()
                sg_ema *= 0.3
                lr = best_lr * 0.7
                no_improve = 0

        lr *= 0.99998

        if lr < 5e-7:
            break

    return best_a, best_obj


def multiplicative_descent_v3(a0, time_limit, seed=0, verbose=False):
    """
    Multiplicative descent with dual tracking: maintains a "dual certificate"
    that provides an upper bound and guides the primal updates.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    conv_len = 2 * n - 1

    t0 = time.monotonic()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    # Dual variables (weights on peak constraints)
    dual = np.ones(conv_len) / conv_len
    lr = 0.003
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        conv = fast_convolve(a)
        cur_obj = conv.max()

        # Update dual: concentrate on peaks above threshold
        threshold = cur_obj * 0.7
        active = conv > threshold
        if not active.any():
            active[np.argmax(conv)] = True
        dual_new = np.zeros(conv_len)
        dual_new[active] = conv[active] ** 2
        dual_new /= dual_new.sum()
        dual = 0.9 * dual + 0.1 * dual_new

        # Compute subgradient from dual-weighted peaks
        sg = np.zeros(n)
        active_idx = np.where(dual > 1e-12)[0]
        for k in active_idx:
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * dual[int(k)] * a[int(k) - idx]

        # Multiplicative update
        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        new_obj = fast_convolve(a).max()
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 300 == 0:
                print(f"      MD3 iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve > 200:
                a = np.maximum(best_a + rng.standard_normal(n) * 0.004, 1e-15)
                a = a / a.sum()
                dual = np.ones(conv_len) / conv_len
                lr = 0.005
                no_improve = 0

        lr *= 0.99995
        if lr < 5e-7:
            break

    return best_a, best_obj


def coordinate_descent_refine(a0, time_limit, verbose=False, seed=0):
    """
    Coordinate descent exploiting monotonicity: since obj increases in each
    coordinate, we reduce high-contributing coords and redistribute to low ones.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    t0 = time.monotonic()
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 50:
        it += 1
        conv = fast_convolve(a)
        cur_obj = conv.max()
        peak_k = int(np.argmax(conv))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib = np.arange(js, je + 1)
        if len(contrib) == 0:
            break

        products = a[contrib] * a[peak_k - contrib]
        max_pos = np.argmax(products)
        max_idx = contrib[max_pos]
        partner_idx = peak_k - max_idx

        best_trial = None
        best_trial_obj = cur_obj

        # Try reducing the source and redistributing to various targets
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            diff = a[max_idx] * frac
            if diff < 1e-15:
                continue
            trial = a.copy()
            trial[max_idx] -= diff
            if trial[max_idx] < -1e-15:
                continue
            trial[max_idx] = max(trial[max_idx], 0.0)

            # Strategy 1: redistribute to non-contributing indices
            mask = np.ones(n, dtype=bool)
            mask[contrib] = False
            if mask.any():
                nc_vals = trial[mask]
                nc_sum = nc_vals.sum()
                if nc_sum > 1e-15:
                    # Weighted toward smaller values (flatten)
                    w_redist = 1.0 / (nc_vals + 0.05 * nc_sum + 1e-10)
                    w_redist /= w_redist.sum()
                    trial[mask] += diff * w_redist
                else:
                    trial[mask] += diff / mask.sum()

                trial = np.maximum(trial, 0.0)
                s = trial.sum()
                if s > 1e-15:
                    trial = trial / s
                    obj = fast_convolve(trial).max()
                    if obj < best_trial_obj - 1e-14:
                        best_trial_obj = obj
                        best_trial = trial.copy()

            # Strategy 2: redistribute to partner (reduce the pair product)
            if partner_idx != max_idx and 0 <= partner_idx < n:
                trial2 = a.copy()
                trial2[max_idx] -= diff
                trial2[partner_idx] = max(trial2[partner_idx], 0.0)
                # Put mass at a random non-peak location
                non_peak = np.ones(n, dtype=bool)
                # Find indices that don't contribute to top 3 peaks
                top3 = np.argsort(conv)[-3:]
                for pk in top3:
                    js2 = max(0, int(pk) - n + 1)
                    je2 = min(n - 1, int(pk))
                    non_peak[js2:je2 + 1] = False
                targets = np.where(non_peak)[0]
                if len(targets) > 0:
                    t = rng.choice(targets, size=min(5, len(targets)), replace=False)
                    trial2[t] += diff / len(t)
                    trial2 = np.maximum(trial2, 0.0)
                    s2 = trial2.sum()
                    if s2 > 1e-15:
                        trial2 = trial2 / s2
                        obj2 = fast_convolve(trial2).max()
                        if obj2 < best_trial_obj - 1e-14:
                            best_trial_obj = obj2
                            best_trial = trial2.copy()

        if best_trial is not None:
            a = best_trial
            if best_trial_obj < best_obj - 1e-12:
                best_obj = best_trial_obj
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"      CD iter {it}: peak={peak_k}, obj={best_obj:.10f}")
        else:
            no_improve += 1
            # Try random coordinate swaps
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                for swap_frac in [0.1, 0.3]:
                    amount = min(a[i], a[j]) * swap_frac
                    trial = a.copy()
                    trial[i] -= amount
                    trial[j] += amount
                    trial = np.maximum(trial, 0.0)
                    s = trial.sum()
                    if s > 1e-15:
                        trial = trial / s
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-12:
                            a = trial
                            best_obj = obj
                            best_a = a.copy()
                            no_improve = 0
                            break

    return best_a, best_obj


def lp_direction_optimize(f0, time_limit, seed=0, verbose=False):
    """LP-based Frank-Wolfe with asymmetric emphasis."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_factors = [1.0, 1.02, 1.05, 1.1, 1.2, 1.5, 2.0, 3.0]
    m_idx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        g = None
        for level_try in range(m_idx, len(m_factors)):
            M_try = conv.max() * m_factors[level_try]
            c_vec = -np.ones(n)
            A_ub = np.zeros((2 * n - 1, n))
            for j in range(n):
                A_ub[j:j + n, j] = f
            b_ub = np.full(2 * n - 1, M_try)
            bounds = [(0, None)] * n
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
            if res.success and res.x is not None and res.x.sum() > 1e-12:
                g = res.x
                m_idx = level_try
                break
            m_idx = level_try + 1

        if g is None or g.sum() < 1e-12:
            # Perturb and retry
            noise = rng.standard_normal(n) * 0.003
            f = np.maximum(best_f + noise, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_idx = 0
            no_improve += 1
            if no_improve > 50:
                break
            continue

        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_factors) - 1)
            if no_improve > 40:
                break
            continue
        no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_factors) - 1)
            continue

        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        M = conv.max()
        f = f_new
        m_idx = 0

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 10 == 0:
                print(f"      LPFW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def annealed_multistart(a0, time_limit, seed=0, verbose=False):
    """
    Multi-start with simulated annealing: run MD from multiple perturbed
    starting points, keeping the best.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    n_starts = 3

    per_start_time = time_limit / n_starts

    for start_idx in range(n_starts):
        if start_idx > 0:
            # Perturb from best
            scale = 0.01 * (start_idx + 1)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 1e-15)
            a = a / a.sum()

        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        t_alloc = min(per_start_time, rem - 5)
        a_opt, obj = multiplicative_descent_v2(a, time_limit=t_alloc, seed=seed + start_idx * 100)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"      AS start {start_idx}: obj={best_obj:.10f}")

    return best_a, best_obj

        print(f"  MD2 [{label}] n={n}: alloc={t_alloc:.0f}s (init={obj_v:.6f})")
        a_opt, opt_obj = multiplicative_descent_v2(a_init, time_limit=t_alloc, seed=seed + idx, verbose=True)
        _try_seq(a_opt, f"md2_{label}")

    # === Phase 3: MD3 (dual-guided) on best ===
    print("\n=== Phase 3: Dual-Guided MD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 60:
            t_alloc = min(rem * 0.4, 120)
            print(f"  MD3: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, _ = multiplicative_descent_v3(best_seq, time_limit=t_alloc, seed=seed + 50, verbose=True)
            _try_seq(a_opt, "md3")

    # === Phase 4: LP-Frank-Wolfe ===
    print("\n=== Phase 4: LP-Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 60:
            t_alloc = min(rem * 0.4, 100)
            print(f"  LPFW: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, _ = lp_direction_optimize(best_seq, time_limit=t_alloc, seed=seed + 10, verbose=True)
            _try_seq(a_opt, "lpfw")

    # === Phase 5: Coordinate descent ===
    print("\n=== Phase 5: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem * 0.5, 100)
            print(f"  CD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_refine(best_seq, time_limit=t_alloc, verbose=True, seed=seed + 20)
            _try_seq(a_opt, "cd")

    # === Phase 6: Multi-start annealing ===
    print("\n=== Phase 6: Annealed Multi-Start ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 50:
            t_alloc = min(rem * 0.5, 100)
            print(f"  AS: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, _ = annealed_multistart(best_seq, time_limit=t_alloc, seed=seed + 30, verbose=True)
            _try_seq(a_opt, "annealed")

    # === Phase 7: Asymmetric perturbations of best ===
    print("\n=== Phase 7: Asymmetric Perturbations ===")
    if best_seq is not None:
        n_best = len(best_seq)
        perturbations = []

        for shift in [1, 2, 3, 5, 10, 20]:
            perturbations.append((f'shift_{shift}', np.roll(best_seq, shift)))

        half = n_best // 2
        for scale in [0.7, 0.8, 0.9, 1.1, 1.3]:
            p = best_seq.copy()
            p[:half] *= scale
            p = np.maximum(p, 0.0)
            p = p / p.sum()
            perturbations.append((f'scaleL_{scale}', p))
            p2 = best_seq.copy()
            p2[half:] *= scale
            p2 = np.maximum(p2, 0.0)
            p2 = p2 / p2.sum()
            perturbations.append((f'scaleR_{scale}', p2))

        for trend in [0.05, 0.1, 0.15, 0.2, 0.3]:
            x = np.linspace(-0.5, 0.5, n_best)
            p = best_seq * (1.0 + trend * x)
            p = np.maximum(p, 0.0)
            p = p / p.sum()
            perturbations.append((f'trend_{trend}', p))

        for label, p in perturbations:
            rem = deadline - time.monotonic() - 50
            if rem < 10:
                break
            sc = _try_seq(p, f"pert_{label}")
            if sc < best_score + 0.01:
                t_alloc = min(rem * 0.25, 25)
                if t_alloc > 8:
                    a_opt, _ = multiplicative_descent_v2(p, time_limit=t_alloc, seed=seed + 500)
                    _try_seq(a_opt, f"md2_pert_{label}")

    # === Phase 8: Final polish ===
    print("\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            t_alloc = min(rem * 0.5, 40)
            print(f"  CD polish: alloc={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_refine(best_seq, time_limit=t_alloc, verbose=False, seed=seed + 99)
            _try_seq(a_opt, "cd_polish")

        rem = deadline - time.monotonic() - 5
        if rem > 15:
            t_alloc = min(rem * 0.6, 25)
            print(f"  MD2 polish: alloc={t_alloc:.0f}s")
            a_opt, _ = multiplicative_descent_v2(best_seq, time_limit=t_alloc, seed=seed + 999)
            _try_seq(a_opt, "md2_polish")

    # Fallback
    if best_seq is None:
        best_seq = make_linear_ramp(768)

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    total_time = time.monotonic() - (deadline - max(1.0, budget_s - 15.0))
    print(f"Total time: {total_time:.1f}s")
    return [float(x) for x in best_seq]
