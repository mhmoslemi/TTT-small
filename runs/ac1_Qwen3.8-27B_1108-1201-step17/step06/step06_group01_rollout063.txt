Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using gradient-based optimization
in log-space (softmax parameterization) with adaptive temperature scheduling,
multi-length exploration, and peak-equalization local search.
Key innovations vs. previous approach:
1. Efficient FFT-based gradient computation for the smoothed max objective.
2. Adaptive temperature schedule (low to high T) for progressively sharper optimization.
3. Structured initialization using exponential shapes to exploit asymmetry.
4. Peak equalization via projected subgradient descent.
5. Multi-length exploration to find the optimal sequence length.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax(z):
    """Compute softmax in a numerically stable way."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def smoothed_objective_and_grad(z, T, n):
    """
    Compute the smoothed objective and gradient in z-space.
    a = softmax(z), objective = (2n/T) * log(sum_k exp(T * conv(a,a)[k]))
    """
    a = softmax(z)
    b = fast_convolve(a)
    
    # Softmax weights on b
    T_b = T * b
    max_Tb = max_Tb = T_b.max()
    exp_Tb = np.exp(T_b - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Weights
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient in a-space: grad_a[i] = 2n * [2*(w*a_cross)[i] - w[2i]*a[i]*I(2i<2n-1)]
    # where (w*a_cross)[i] = sum_k w[k]*a[k-i] (cross-correlation)
    
    # Compute cross-correlation of w and a via FFT
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross_corr[i] = sum_k w[k] * a[k-i]
    # This is the cross-correlation, which is: IFFT(conj(W) * A)
    cross_corr = irfft(np.conj(W) * A, fft_size)[:n]
    
    grad_a = 2.0 * n * 2.0 * cross_corr
    # Subtract diagonal correction
    for i in range(n):
        k = 2 * i
        if k < 2 * n - 1:
            grad_a[i] -= 2.0 * n * w[k] * a[i]
    
    # Gradient in z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lbgfs_optimize(a0, T, time_budget, maxiter=5000):
    """Run L-BFGS-B optimization in log-space."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-15))
    t0 = time.monotonic()
    stopped = [False]
    
    def func(z):
        if time.monotonic() - t0 > time_budget:
            stopped[0] = True
            return 1e18
        obj, _ = smoothed_objective_and_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_budget:
            stopped[0] = True
            return np.zeros_like(z)
        _, g = smoothed_objective_and_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8,
                     'maxfun': maxiter * 50}
        )
        a_out = softmax(res.x)
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        a_out = a0
    
    return a_out


def subgradient_descent(a0, time_budget, lr_init=0.01, n_iter=50000):
    """
    Projected subgradient descent on the simplex.
    Uses the exact max (not smoothed) with adaptive step size.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    momentum = 0.9
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_budget:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_pos = np.argmax(b)
        
        # Subgradient of b[peak_pos]
        k = peak_pos
        sg = np.zeros(n)
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                sg[i] = 2.0 * a[ki]
                if k == 2 * i:
                    sg[i] = a[i]
        
        # Project subgradient onto tangent space (sum = 0)
        sg = sg - sg.mean()
        
        # Momentum update
        v = momentum * v - lr * sg
        
        # Project onto simplex: a + v should be >= 0 and sum to 1
        a_candidate = a + v
        a_new = project_simplex(a_candidate)
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            a = best_a.copy()  # revert
        
        lr *= 0.9999
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            v = np.zeros(n)
            lr = max(lr * 2, 0.005)
            no_improve = 0
        
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_equalize(a0, time_budget, n_iter=100000):
    """
    Peak equalization: iteratively find the peak and reduce it by
    redistributing mass to low-sensitivity positions.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_budget:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find all near-peak positions
        threshold = 0.98 * max_b
        peak_indices = np.where(b >= threshold)[0]
        
        if len(peak_indices) == 0:
            break
        
        # Compute combined gradient from all peaks
        sg = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    sg[i] += 2.0 * a[ki]
                    if k == 2 * i:
                        sg[i] -= a[ki]  # correct the double count
        
        # Normalize
        sg = sg / max(sg.max(), 1e-15)
        
        # Find positions with low sensitivity (where we can add mass)
        # and high sensitivity (where we should remove mass)
        mean_sg = sg.mean()
        high_mask = sg > mean_sg + 0.5 * (sg.max() - mean_sg)
        low_mask = sg < mean_sg - 0.1 * (sg.max() - mean_sg)
        
        if not high_mask.any() or not low_mask.any():
            break
        
        # Move mass from high to low
        total_high = a[high_mask].sum()
        total_low = a[low_mask].sum()
        if total_high < 1e-15 or total_low < 1e-15:
            break
        
        # Small transfer
        transfer = min(total_high * 0.01, 0.001)
        
        a_new = a.copy()
        # Proportionally reduce high positions
        a_new[high_mask] -= transfer * a[high_mask] / total_high
        # Proportionally increase low positions
        a_new[low_mask] += transfer * a[low_mask] / total_low
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller transfer
            transfer *= 0.5
            if transfer < 1e-12:
                break
    
    return best_a, best_score


def exponential_init(n, decay_rate, seed=42):
    """Create an exponentially decaying initial sequence."""
    rng = np.random.default_rng(seed)
    a = np.exp(-decay_rate * np.arange(n) / n)
    # Add small random perturbation
    a += rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using gradient-based optimization in
log-space (softmax parameterization) with correct FFT-based gradients, adaptive
temperature scheduling, projected subgradient descent, and multi-length exploration.
Key innovations: (1) correct gradient via cross-correlation (no diagonal correction),
(2) progressive temperature schedule for L-BFGS, (3) exact-max subgradient as complement,
(4) structured exploration across sequence lengths.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Global state for passing parameters to top-level objective/gradient functions
_opt_state = {}


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_cross_corr(w, a, n):
    """Compute cross-correlation c[i] = sum_k w[k]*a[k-i] for i=0..n-1."""
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    c = irfft(np.conj(W) * A, fft_size)[:n]
    return c


def _obj_func(z):
    """Objective function using global state."""
    T = _opt_state['T']
    n = _opt_state['n']
    deadline = _opt_state['deadline']
    if time.monotonic() > deadline:
        return 1e18
    a = softmax_fn(z)
    b = fast_convolve(a)
    T_b = T * b
    max_Tb = T_b.max()
    exp_Tb = np.exp(T_b - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * lse


def _grad_func(z):
    """Gradient function using global state."""
    T = _opt_state['T']
    n = _opt_state['n']
    deadline = _opt_state['deadline']
    if time.monotonic() > deadline:
        return np.zeros(n)
    a = softmax_fn(z)
    b = fast_convolve(a)
    T_b = T * b
    max_Tb = T_b.max()
    exp_Tb = np.exp(T_b - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    # grad_a[i] = 4n * cross_corr(w, a)[i]
    cross = compute_cross_corr(w, a, n)
    grad_a = 4.0 * n * cross
    # Convert to z-space gradient
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z


def lbgfs_optimize(a0, T, time_budget, maxiter=5000):
    """Run L-BFGS-B in log-space with given temperature T."""
    n = len(a0)
    _opt_state['T'] = T
    _opt_state['n'] = n
    _opt_state['deadline'] = time.monotonic() + time_budget
    z0 = np.log(np.maximum(a0, 1e-15))
    # Fix one coordinate to remove gauge freedom
    z0 = z0 - z0[0]
    z0[0] = 0.0
    try:
        res = minimize(
            fun=_obj_func,
            x0=z0,
            jac=_grad_func,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10,
                     'maxfun': maxiter * 80, 'maxls': 50}
        )
        a_out = softmax_fn(res.x)
        status = res.status
        message = res.message if hasattr(res, 'message') else ''
    except Exception as e:
        print(f"    L-BFGS error (T={T}): {e}")
        a_out = a0
        status = -1
        message = str(e)
    return a_out, status


def compute_score(a):
    """Compute the objective value directly (fast, no evaluate_sequence overhead)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return 2.0 * n * b.max()


def projected_subgradient(a0, time_budget, lr_init=0.01, max_iter=200000):
    """
    Projected subgradient descent on the simplex using the exact max.
    Uses momentum and adaptive step size.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros(n)
    momentum = 0.85
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    stalled = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Find active peaks (within 1% of max)
        threshold = 0.995 * max_b
        peak_idx = np.where(b >= threshold)[0]

        # Compute subgradient: average of db[k]/da for active peaks
        sg = np.zeros(n)
        for k in peak_idx:
            # db[k]/da[i] = 2*a[k-i] if 0 <= k-i < n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * a[k - idx]
        sg /= len(peak_idx)

        # Project to tangent space (sum = 0)
        sg = sg - sg.mean()

        # Nesterov-like momentum
        v = momentum * v - lr * sg

        # Candidate update
        a_cand = a + v
        # Project onto simplex
        u = np.sort(a_cand)[::-1]
        cssv = np.cumsum(u)
        rho_vals = np.arange(1, n + 1)
        cond = u * rho_vals > (cssv - 1.0)
        if cond.any():
            rho = int(np.nonzero(cond)[0][-1])
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_cand - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        a_new = a_new / a_new.sum()

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            stalled = 0
        else:
            no_improve += 1
            stalled += 1

        # Adaptive step size
        lr *= 0.9998
        if no_improve > 200:
            lr = max(lr * 0.5, 1e-6)
            no_improve = 0
        if stalled > 2000:
            # Random restart
            rng = np.random.default_rng(hash((it, n)) % 2**31)
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                a = best_a.copy()
            else:
                a = a / s
            v = np.zeros(n)
            lr = max(lr * 3.0, 0.003)
            stalled = 0
            no_improve = 0

        if it % 10000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"      PG it={it} ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.6e}")

    return best_a, best_score


def peak_shave_local(a0, time_budget, max_iter=50000):
    """
    Direct peak shaving: identify peak, reduce contributing elements,
    redistribute mass to low-impact positions.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find contributing elements to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.max() < 1e-15:
            break

        # Find the top contributor
        top_i = idx[np.argmax(contribs)]
        top_val = a[top_i]
        if top_val < 1e-12:
            break

        # Find positions where adding mass has minimal impact on the peak
        # Impact of adding to position j: 2*a[peak_k - j] (if valid)
        impacts = np.zeros(n)
        for j in range(n):
            kj = peak_k - j
            if 0 <= kj < n:
                impacts[j] = 2.0 * a[kj]
            else:
                impacts[j] = 0.0

        # Exclude the peak contributor
        impacts[top_i] = np.inf

        # Find the least-impact position with substantial mass to borrow from
        # or an empty position to deposit to
        low_impact = np.argsort(impacts)[:10]
        low_impact = [j for j in low_impact if j != top_i and impacts[j] < np.inf]

        if not low_impact:
            break

        improved = False

            new_val = top_val * factor
            diff = top_val - new_val
