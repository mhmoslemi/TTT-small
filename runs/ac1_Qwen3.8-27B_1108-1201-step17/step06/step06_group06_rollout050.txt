```python
def greedy_fw_step(f, grad):
    """Find the vertex of the simplex minimizing <grad, e_i> = grad[i]."""
    i_min = np.argmin(grad)
    return i_min

def away_step_fw(f, grad, S):
    """
    Away-step Frank-Wolfe.
    f is current point on simplex.
    grad is the gradient.
    S is the set of "support" vertices (where f > 0).
    Returns (step_alpha, target) where target is either a coordinate or -i for away-step.
    """
    # Standard FW step
    i_min = np.argmin(grad)
    fw_gain = grad[i_min] - grad @ f  # = <grad, e_i - f>
    
    # Away step: for each j in S, check <grad, f - (n_j/|S|)*(e_j)>
    # Actually, the away step direction is f - e_j for j in S
    # Gain for away step in direction -e_j: <grad, -e_j/|S| + f/|S|>... 
    # More precisely, the away step is: f_t = (1-tappa)*f + tappa*(f - e_j) * (|S|-1)/|S|
    # Simplified: move from f toward f - e_j (removing mass from j)
    
    best_gain = fw_gain
    best_target = (1, i_min)  # (type, index) where type=1 means toward, type=-1 means away
    
    if len(S) > 1:
        j_away = np.argmax([grad[j] * f[j] / f[j] for j in S])  # maximize grad[j]*f[j]
        # Actually, the away step toward removing coordinate j:
        # direction = (f - f[j]*e_j) / (1 - f[j]) - f = -f[j]*(e_j - f)/(1-f[j])
        # Linear minimization over this: the gain is proportional to -grad[j]*f[j]/(1-f[j])
        # So we want to maximize grad[j]*f[j]/(1-f[j])
        j_away = max(S, key=lambda j: grad[j] * f[j] / max(1 - f[j], 1e-15))
        away_gain = -grad[j_away] * f[j_away] / max(1 - f[j_away], 1e-15)
        if away_gain > best_gain:
            best_gain = away_gain
            best_target = (-1, j_away)  # away from j_away
    
    return best_target, best_gain
```

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Adam smooth optimization with adaptive temperature,
(2) Exact subgradient with Nesterov momentum,
(3) Away-step Frank-Wolfe,
(4) Coordinate descent polish with multi-length exploration.
Key innovation: exploits the simple structure of the subgradient
(grad of conv(a,a)[k] is 2*a shifted/reversed) for O(n) gradient eval,
and uses away-step FW for linear convergence in the final phase.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(f, tau):
    """
    Compute gradient of log(sum_k exp(conv(f,f)[k]/tau)).
    Returns (gradient vector, convolution).
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # grad[i] = (2/tau) * sum_k w[k] * f[k-i]
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv


def exact_subgradient(f, k_star):
    """
    Compute the (sub)gradient of conv(f,f)[k_star] w.r.t. f.
    grad[i] = 2*f[k_star - i] if 0 <= k_star - i < n, else 0.
    This is O(n).
    """
    n = len(f)
    grad = np.zeros(n)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    for i in range(i_lo, i_hi + 1):
        grad[i] = 2.0 * f[k_star - i]
    return grad


def adam_smooth_optimize(f0, time_limit, tau_init=None, lr_init=0.003, verbose=False):
    """Adam optimization on the log-sum-exp smooth surrogate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv0 = fast_convolve(f)
    if tau_init is None:
        tau = 0.06 * conv0.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv0.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:

        grad, conv = smooth_gradient(f, tau)
        M = conv.max()

        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        if tau < 1e-8:
            tau = 1e-8

            lr *= 0.9
        if no_improve > 600:

            v = np.zeros(n)

def nesterov_exact_subgrad(f0, time_limit, verbose=False):
    """
    Nesterov momentum on the exact max of convolution.
    Uses the simple O(n) subgradient (shifted/reversed a).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    x = f.copy()
    y = f.copy()
    v = np.zeros(n)
    beta = 0.9  # momentum
    lr = 0.5 / n  # initial step size
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    consecutive_bt = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Nesterov extrapolation
        momentum = beta * (1.0 - 0.5 * (t % 1000) / 1000.0)
        y = x + momentum * (x - y + lr * v)
        y = project_simplex(y)

        # Compute convolution and find max
        conv_y = fast_convolve(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient at k_star
        grad = exact_subgradient(y, k_star)

        # Update velocity
        v = beta * v + grad

        # Step with backtracking
        step_lr = lr
        for _ in range(8):
            x_new = project_simplex(y - step_lr * v)
            conv_new = fast_convolve(x_new)
            if conv_new.max() <= conv_y[k_star] + 1e-14:
                break
            step_lr *= 0.5
            consecutive_bt += 1
        else:
            x_new = x  # accept no move
            consecutive_bt += 1

        if consecutive_bt > 5:
            lr *= 0.7
            consecutive_bt = 0

        if fast_convolve(x_new).max() < best_obj - 1e-14:
            best_obj = fast_convolve(x_new).max()
            best_f = x_new.copy()
            no_improve = 0

            if verbose and t % 500 == 0:
                print(f"    Nesterov-Exact t={t}: obj={2*n*best_obj:.8f}, lr={lr:.6f}")
        else:
            no_improve += 1

        x = x_new

        # Periodic restart
        if no_improve > 1000:
            rng = np.random.default_rng(t * 7 + 3)
            x = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            y = x.copy()
            v = np.zeros(n)
            lr = 0.3 / n
            no_improve = 0

    return best_f, 2 * n * best_obj


def away_step_fw(f0, time_limit, verbose=False):
    """
    Away-step Frank-Wolfe on the exact objective.
    Linear oracle: argmin of gradient (the shifted/reversed a at the peak).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # Track support set (indices where f[i] > threshold)
    support_threshold = 1e-6
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))
        M = conv[k_star]

        # Gradient (subgradient at peak)
        grad = exact_subgradient(f, k_star)

        # Support set
        S = np.where(f > support_threshold)[0]
        if len(S) == 0:
            S = np.array([np.argmax(f)])

        # === Standard FW step ===
        i_min = int(np.argmin(grad))
        fw_gain = grad[i_min] - np.dot(grad, f)

        # === Away step ===
        best_gain = fw_gain
        best_type = 1  # 1 = toward, -1 = away
        best_idx = i_min

        if len(S) > 1:
            # Away step: maximize grad[j]*f[j] / (1 - f[j])
            away_scores = np.zeros(len(S))
            for idx, j in enumerate(S):
                denom = max(1.0 - f[j], 1e-12)
                away_scores[idx] = grad[j] * f[j] / denom
            j_away = S[np.argmax(away_scores)]
            away_score = away_scores[np.argmax(away_scores)]
            if away_score > best_gain:
                best_gain = away_score
                best_type = -1
                best_idx = int(j_away)

        if best_gain < -1e-12:
            # Take the step
            if best_type == 1:
                # Move toward e_best_idx
                direction = np.zeros(n)
                direction[best_idx] = 1.0
                direction -= f
            else:
                # Move away from e_best_idx: direction = (f - f[idx]*e_idx)/(1-f[idx]) - f
                idx = best_idx
                if f[idx] >= 1.0 - 1e-12:
                    continue
                direction = np.zeros(n)
                direction = f / (1.0 - f[idx])
                direction[idx] = 0.0
                direction -= f

            # Line search
            best_alpha = 0.0
            best_val = M
            # Try several alphas
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                f_trial = f + alpha * direction
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial = f_trial / st
                val = fast_convolve(f_trial).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

            if best_alpha > 1e-8:
                f = np.maximum(f + best_alpha * direction, 0.0)
                f = f / f.sum()
                conv_new = fast_convolve(f)
                if conv_new.max() < best_obj - 1e-14:
                    best_obj = conv_new.max()
                    best_f = f.copy()
                    no_improve = 0
                    if verbose and it % 100 == 0:
                        print(f"    AFW iter {it}: obj={2*n*best_obj:.8f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > 50:
            rng = np.random.default_rng(it * 11 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0

        if no_improve > 300:
            break

    return best_f, 2 * n * best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: try adjusting pairs of coordinates to reduce the peak.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        k_star = int(np.argmax(conv))

        # Find the pair (i, j) that contributes most to the peak
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        idx = np.arange(i_lo, i_hi + 1)
        products = f[idx] * f[k_star - idx]
        max_pos = idx[np.argmax(products)]
        other_pos = k_star - max_pos

        improved = False

        if max_pos == other_pos:
            # Diagonal: reduce f[max_pos]
            for frac in [0.7, 0.8, 0.9, 0.5]:
                trial = f.copy()
                old_val = trial[max_pos]
                trial[max_pos] *= frac
                diff = old_val - trial[max_pos]
                # Redistribute to a random non-peak coordinate
                # Find coordinates that don't contribute to top peaks
                top_peaks = np.argsort(conv)[-3:]
                dangerous = set()
                for pk in top_peaks:
                    lo = max(0, int(pk) - n + 1)
                    hi = min(n - 1, int(pk))
                    for ii in range(lo, hi + 1):
                        dangerous.add(ii)
                        dangerous.add(int(pk - ii))
                safe = [i for i in range(n) if i not in dangerous and f[i] < 0.05]
                if not safe:
                    safe = [i for i in range(n) if i != max_pos]
                if safe:
                    target = rng.choice(safe)
                    trial[target] += diff
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
        else:
            # Off-diagonal: reduce the product by adjusting one coordinate
            for target_coord, frac in [(max_pos, 0.7), (max_pos, 0.85),
                                        (other_pos, 0.7), (other_pos, 0.85)]:
                if f[target_coord] < 1e-8:
                    continue
                trial = f.copy()
                old_val = trial[target_coord]
                trial[target_coord] *= frac
                diff = old_val - trial[target_coord]
                # Redistribute
                top_peaks = np.argsort(conv)[-3:]
                dangerous = set()
                for pk in top_peaks:
                    lo = max(0, int(pk) - n + 1)
                    hi = min(n - 1, int(pk))
                    for ii in range(lo, hi + 1):
                        dangerous.add(ii)
                        dangerous.add(int(pk - ii))
                safe = [i for i in range(n) if i not in dangerous and i != target_coord]
                if safe:
                    target = rng.choice(safe)
                    trial[target] += diff
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Try a small random perturbation
                trial = project_simplex(f + rng.standard_normal(n) * 0.0005)
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                if not improved:
                    f = best_f.copy()
                no_improve = 0
        else:
            no_improve = 0

        if no_improve > 100:
            break

    return best_f, 2 * n * best_obj


def make_raised_cosine(n):
    """Raised cosine (Hann window) initialization."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.3):
    """Asymmetric Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, flat_frac=0.5):
    """Flat-top window initialization."""
    x = np.arange(n) / (n - 1)
    a = 0.2156 + 0.4160 * np.cos(2 * np.pi * x) + \
        0.2781 * np.cos(4 * np.pi * x) + 0.0840 * np.cos(6 * np.pi * x) + \
        0.0089 * np.cos(8 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Smooth bell with flat top."""
    x = np.arange(n) - n / 2.0
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = (1.0 - t ** 2) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_exponential(n, decay_frac=0.3):
    """Exponential decay initialization."""
    a = np.exp(-np.arange(n) / (n * decay_frac))
    a /= a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization (30s)
    Phase 2: Adam smooth optimization (600s)
    Phase 3: Nesterov exact subgradient (400s)
    Phase 4: Away-step Frank-Wolfe (300s)
    Phase 5: Multi-length + CD polish + final squeeze (370s + buffer)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # Generate diverse initializations
    inits = []
    for n_try in [512, 640, 768, 1024]:
        inits.append(make_raised_cosine(n_try))
        inits.append(make_asymmetric_gaussian(n_try, 0.35, 0.25))
        inits.append(make_asymmetric_gaussian(n_try, 0.5, 0.2))
        inits.append(make_flat_top(n_try))
        inits.append(make_bell(n_try, 0.55))
        inits.append(make_bell(n_try, 0.7))
        inits.append(make_exponential(n_try, 0.25))
        inits.append(np.ones(n_try) / n_try)

    # Also try symmetric versions
    for n_try in [640, 768]:
        a = make_asymmetric_gaussian(n_try, 0.4, 0.3)
        a_sym = (a + a[::-1]) / 2
        a_sym /= a_sym.sum()
        inits.append(a_sym)

    # Weighted sum of Gaussians (multi-peak)
    for n_try in [640, 768]:
        a1 = make_asymmetric_gaussian(n_try, 0.3, 0.15)
        a2 = make_asymmetric_gaussian(n_try, 0.7, 0.15)
        a_mix = 0.5 * a1 + 0.5 * a2
        a_mix /= a_mix.sum()
        inits.append(a_mix)

    for i, a_init in enumerate(inits):
        if time.monotonic() > deadline - 1400:
            break
        sc = try_seq(a_init, f"init_{i}_n{len(a_init)}")

    # Select best for main optimization
    work_seq = best_seq if best_seq is not None else make_raised_cosine(768)
    work_n = len(work_seq)
    print(f"  Phase 1 done. Working with n={work_n}, score={best_score:.6f}")
    print(f"  Phase 1 time: {time.monotonic() - t_start:.1f}s")

    # === Phase 2: Adam smooth optimization (main workhorse) ===
    print(f"\n=== Phase 2: Adam Smooth Optimization (n={work_n}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 550)
    if t2 > 60:
        print(f"  Time allocated: {t2:.0f}s")
        a_opt, obj = adam_smooth_optimize(work_seq, time_limit=t2 * 0.65,
                                          lr_init=0.004, verbose=True)
        try_seq(a_opt, "adam_main")

        # Second Adam pass with different params
        if time.monotonic() < deadline - 500:
            rem2 = deadline - time.monotonic() - 500
            t2b = min(rem2, 200)
            tau_fine = 0.02 * fast_convolve(best_seq).max()
            a_opt2, obj2 = adam_smooth_optimize(best_seq, time_limit=t2b,
                                                tau_init=tau_fine,
                                                lr_init=0.0015, verbose=True)
            try_seq(a_opt2, "adam_fine")

    # === Phase 3: Nesterov exact subgradient ===
    print(f"\n=== Phase 3: Nesterov Exact Subgradient ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 50:
            t3 = min(rem, 400)
            print(f"  Time allocated: {t3:.0f}s")
            a_opt, obj = nesterov_exact_subgrad(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "nesterov_exact")

    # === Phase 4: Away-step Frank-Wolfe ===
    print(f"\n=== Phase 4: Away-step Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 40:
            t4 = min(rem, 300)
            print(f"  Time allocated: {t4:.0f}s")
            a_opt, obj = away_step_fw(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "afw")

            # Try AFW from perturbed start
            if time.monotonic() < deadline - 150:
                rem2 = deadline - time.monotonic() - 150
                t4b = min(rem2, 100)
                perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.003)
                a_opt2, obj2 = away_step_fw(perturbed, time_limit=t4b, verbose=False)
                try_seq(a_opt2, "afw_pert")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 896, 640, 512]:
            if target_n == len(base):
                continue
            rem = deadline - time.monotonic() - 120
            if rem < 40:
                break
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            # Quick Adam on interpolated
            t5 = min(rem * 0.35, 80)
            if t5 > 20:
                a_opt, obj = adam_smooth_optimize(f_interp, time_limit=t5, lr_init=0.003)
                try_seq(a_opt, f"adam_n{target_n}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coordinate Descent Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 30:
            t6 = min(rem, 120)
            print(f"  Time allocated: {t6:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t6, verbose=True)
            try_seq(a_opt, "cd_polish")

    # === Phase 7: Final Adam polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t7 = min(rem, 50)
            tau_final = 0.01 * fast_convolve(best_seq).max()
            a_opt, obj = adam_smooth_optimize(
                best_seq, time_limit=t7,
                tau_init=tau_final,
                lr_init=0.0008, verbose=False)
            try_seq(a_opt, "final_polish")

    # === Phase 8: Final AFW squeeze ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t8 = min(rem, 30)
            a_opt, obj = away_step_fw(best_seq, time_limit=t8, verbose=False)
            try_seq(a_opt, "final_afw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    return [float(x) for x in best_seq]
