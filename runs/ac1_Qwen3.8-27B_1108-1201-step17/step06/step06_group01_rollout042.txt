def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

    return f_interp

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

    cross = irfft(W * np.conj(A), L)[:n]

    # Gradient w.r.t. z: a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

def lbgfs_continuation(a0, time_limit, T_list=None, verbose=True):
    """L-BFGS optimization with continuation on temperature T."""
    n = len(a0)
    if T_list is None:
        T_list = [500.0, 200.0, 100.0, 50.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.2, 1.05, 1.02, 1.01, 1.0]
    
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    z = np.log(a)
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        
        def func(z_in):
            obj, _ = compute_smoothed_obj_and_grad(z_in, T, n)
            return obj
        
        def jac(z_in):
            _, g = compute_smoothed_obj_and_grad(z_in, T, n)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxfun': 20000, 'disp': False, 'iprint': -1}
            )
            z = res.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS T={T} failed: {e}")
            break
        
        a_new = softmax(z)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        s = score_of(a_new)
        if s < best_score:
            best_score = s
            best_a = a_new.copy()
        
        z = np.log(np.maximum(a_new, 1e-30))
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LBFGS T={T:.4f} ({elapsed:.0f}s): score={s:.8f}")
    
    return best_a, best_score

def subgradient_optimize(a0, time_limit, lr_init=0.01, mom=0.9, verbose=True):
    """Subgradient method with momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        M = b.max()
        
        # Find active peaks (within 1% of max)
        threshold = M * 0.99
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Weighted subgradient (uniform weights over active set)
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)
        
        # Momentum update
        v = mom * v - lr * grad
        
        # Step and project
        a_candidate = a + v
        a_new = simplex_project(a_candidate)
        
        # Check improvement
        s_new = score_of(a_new)
        if s_new < best_score - 1e-14:
            best_score = s_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Try without momentum
            a_trial = a - lr * grad
            a_trial = simplex_project(a_trial)
            s_trial = score_of(a_trial)
            if s_trial < best_score - 1e-14:
                best_score = s_trial
                best_a = a_trial.copy()
                a = a_trial
                no_improve = 0
                v = np.zeros(n)
        
        # Adaptive learning rate
        lr *= 0.999
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e9) % (2**31))
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2, 0.003)
            no_improve = 0
        
        if verbose and (int((time.monotonic() - t0) * 2)) % 2 == 0:
            elapsed = time.monotonic() - t0
            if int(elapsed) > 0 and int(elapsed * 10) % 10 == 0:
                print(f"    SubGrad ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.6f} active={len(active)}")
    
    return best_a, best_score

def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def peak_shave(a0, time_limit, verbose=True):
    """Direct peak shaving: reduce contributions to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Find peaks (excluding edges where they're always small)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]
        
        if peak_val <= best_score / (2.0 * n) * 1.000001:
            break  # At the global peak
        
        # Find top contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_indices = idx[np.argsort(contribs)[::-1][:5]]
        
        improved = False
        for i in top_indices:
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7, 0.6]:
                a_trial = a.copy()
                diff = a[i] * (1 - factor)
                a_trial[i] *= factor
                # Redistribute
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                s = score_of(a_trial)
                if s < best_score - 1e-14:
                    a = a_trial
                    best_score = s
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing pairs
            for i in top_indices[:3]:
                j = peak_k - i
                if i == j or j < 0 or j >= n or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8, 0.7]:
                    for f2 in [0.9, 0.8, 0.7]:
                        a_trial = a.copy()
                        d1 = a[i] * (1 - f1)
                        d2 = a[j] * (1 - f2)
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        diff = d1 + d2
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        mask[j] = False
                        others_sum = a_trial[mask].sum()
                        if others_sum < 1e-15:
                            continue
                        a_trial[mask] += diff * a_trial[mask] / others_sum
                        a_trial = a_trial / a_trial.sum()
                        s = score_of(a_trial)
                        if s < best_score - 1e-14:
                            a = a_trial
                            best_score = s
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
    
    return best_a, best_score

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with log-sum-exp continuation in log-space (softmax parameterization)
2. Subgradient method with momentum and adaptive step size
3. Peak shaving local search
4. Multi-length exploration with interpolation
5. Active-set subgradient polishing for final refinement
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv(a,a)) for a normalized sequence (sum=1)."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def softmax_fn(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def compute_smoothed_obj_and_grad(z, T, n):
    """
    Compute log-sum-exp smoothed objective and gradient w.r.t. z.
    f(z) = (2n/T) * log(sum_k exp(T * b_k)) where b = conv(softmax(z), softmax(z))
    Returns (obj, grad_z).
    """
    a = softmax_fn(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = T * (b - max_b)
    exp_scaled = np.exp(scaled)
    log_sum_exp = max_b + np.log(exp_scaled.sum()) / T
    obj = 2.0 * n * log_sum_exp

    # Weights
    w = exp_scaled / exp_scaled.sum()

    # grad_a[m] = (4n/T) * sum_k w[k] * a[k-m] = (4n/T) * cross_corr(w, a)[m]
    # Using FFT: cross_corr(w,a) = IFFT(FFT(w,L) * conj(FFT(a,L)))[:n]
    size_w = len(w)  # 2n-1
    size_a = n
    needed = size_w + size_a - 1  # 3n-2
    L = 1 << (needed - 1).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    cross = irfft(W * np.conj(A), L)[:n]
    grad_a = (4.0 * n / T) * cross

    # Gradient w.r.t. z (log-parameterization): a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lbgfs_continuation(a0, time_limit, T_list=None, verbose=True):
    """L-BFGS optimization with continuation on temperature T."""
    n = len(a0)
    if T_list is None:
        T_list = [500.0, 200.0, 100.0, 50.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.2, 1.05, 1.02, 1.0]

    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    z = np.log(a)

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break

        def func(z_in):
            obj, _ = compute_smoothed_obj_and_grad(z_in, T, n)
            return obj

        def jac(z_in):
            _, g = compute_smoothed_obj_and_grad(z_in, T, n)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxfun': 15000, 'disp': False, 'iprint': -1}
            )
            z = res.x
        except Exception:
            if verbose:
                print(f"    L-BFGS T={T} failed, breaking.")
            break

        a_new = softmax_fn(z)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        z = np.log(np.maximum(a_new, 1e-30))

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LBFGS T={T:.4f} ({elapsed:.0f}s): score={sc:.8f}")

    return best_a, best_score


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_optimize(a0, time_limit, lr_init=0.008, mom=0.9, verbose=True):
    """Subgradient method with momentum and adaptive step size in direct space."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        M = b.max()

        # Find active peaks (within 1% of max)
        threshold = M * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient: average of active peak gradients
        # grad_k[m] = 2*a[k-m] if 0 <= k-m < n
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)

        # Momentum update
        v = mom * v - lr * grad
        a_candidate = a + v
        a_new = simplex_project(a_candidate)

        step += 1
        s_new = score_of(a_new)

        if s_new < best_score - 1e-14:
            best_score = s_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            # Try step without momentum
            no_improve += 1
            if no_improve % 10 == 0:
                a_trial = a - lr * grad
                a_trial = simplex_project(a_trial)
                s_trial = score_of(a_trial)
                if s_trial < best_score - 1e-14:
                    best_score = s_trial
                    best_a = a_trial.copy()
                    a = a_trial
                    no_improve = 0
                    v = np.zeros(n)

        lr *= 0.9995

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1e9) % (2**31))
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.003)
            no_improve = 0

        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step} ({elapsed:.0f}s): score={best_score:.8f} lr={lr:.6f} |A|={len(active)}")

    return best_a, best_score


def peak_shave(a0, time_limit, verbose=True):
    """Direct peak shaving: reduce top contributors to the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]

        if peak_val <= best_score / (2.0 * n) * 1.0000001:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        order = np.argsort(contribs)[::-1][:5]
        top_indices = idx[order]

        improved = False
        # Try reducing single elements
        for i in top_indices:
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7, 0.55]:
                a_trial = a.copy()
                diff = a[i] * (1.0 - factor)
                a_trial[i] *= factor
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                s = score_of(a_trial)
                if s < best_score - 1e-14:
                    a = a_trial

                    improved = True

            if improved:
                break
