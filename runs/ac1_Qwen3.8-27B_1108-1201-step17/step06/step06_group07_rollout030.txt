    n = len(a)
    old = a[j]
    # Affected k range
    k_lo = max(0, j)
    k_hi = min(2*n-2, j+n-1)

    #   if k == 2*j: x^2 - old^2
    #   else: 2*(x - old) * a[k-j]

    # A reasonable upper bound is 2*max(a) or the current peak itself
    x_hi = max(2 * np.max(a), np.max(conv) * 2 + 1)
    x_lo = 0.0

    best_val = np.max(conv)

                best_x = m2

def eval_coord(conv, a, j, x, k_lo, k_hi):
    """Evaluate max(conv_new) for a given x = a[j]."""
    n = len(a)
    k_vals = np.arange(k_lo, k_hi + 1)
    # Linear part: conv[k] + 2*(x - a[j]) * a[k-j]
