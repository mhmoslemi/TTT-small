def mirror_descent_subgrad(f0, time_limit, lr_schedule='adaptive'):
    f = f0.copy()
    n = len(f)
    t0 = time.monotonic()
    
    # Accumulated subgradients for Polyak averaging
    avg_f = f.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj:
            best_obj = M
            best_f = f.copy()
        
        # Compute subgradient (averaged over near-max positions)
        # For smoothness, use top-k positions
        k = min(5, 2*n-1)
        top_ks = np.argsort(conv)[-k:]
        # Weight by exp((conv - M)/small)
        weights = np.exp((conv - M) / (0.01 * M + 1e-10))
        weights[top_ks] += 1  # Ensure top positions get weight
        weights /= weights.sum()
        
        # Subgradient: g[j] = 2 * sum_k weights[k] * a[k-j] (with bounds)
        # Using FFT
        w = weights.copy()
        w_rev = w[::-1]
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        W = np.fft.rfft(w_rev, fft_size)
        conv_r = np.fft.irfft(A * W, fft_size)[:total_len]
        g = 2.0 * conv_r[2*n-2 : 2*n-2-n : -1]
        
        # Mirror descent update
        if lr_schedule == 'adaptive':
            lr = 1.0 / np.sqrt(t + 1)  # O(1/sqrt(T)) rate for convex
            # Scale by problem size
            lr = lr / np.sqrt(n) * 2.0
        else:
            lr = 0.1 / np.sqrt(t + 1)
        
        f = f * np.exp(-lr * g)
        f = np.maximum(f, 0)
        ss = f.sum()
        if ss < 1e-30:
            f = np.ones(n) / n
        else:
            f = f / ss
        
        # Polyak averaging
        avg_f = (t * avg_f + f) / (t + 1)
    
    return avg_f, best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage Lp-norm FISTA optimization,
log-sum-exp annealing, mirror descent (exponentiated gradient), and cutting-plane refinement.
Key innovations: FFT-based Lp gradient with FISTA acceleration, exponentiated gradient
on the simplex for the non-smooth max, and adaptive multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_grad(f, p):
    """Compute gradient of sum(conv(f,f)^p) using FFT."""
    n = len(f)
    conv = fast_convolve(f)
    conv_clipped = np.maximum(conv, 1e-30)
    w = conv_clipped ** (p - 1)
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_au = np.fft.irfft(A * W, fft_size)[:total_len]
    grad = 2.0 * p * conv_au[n - 1:2 * n - 1][::-1]
    return grad, conv


def lse_grad(f, tau):
    """Compute gradient of log(sum(exp(conv/tau))) using FFT."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_au = np.fft.irfft(A * W, fft_size)[:total_len]
    grad = (2.0 / tau) * conv_au[n - 1:2 * n - 1][::-1]
    return grad, conv


def fista_lp(f0, time_limit, p, lr=None, verbose=False):
    """FISTA optimization of Lp norm of convolution on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    if lr is None:
        # Estimate Lipschitz constant: L ~ 2*p*max(conv^(p-1))*2 (rough)
        L = 4.0 * p * np.max(conv ** max(p - 1, 1))
        lr = 1.0 / (L + 1e-10)
        # Scale down for stability
        lr = lr * 0.8

    y = f.copy()
    f_prev = f.copy()
    t_k = 1.0

    best_obj = M
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        grad_y, conv_y = lp_grad(y, p)
        obj_y = fast_convolve(y).max()

        # FISTA update
        f_new = project_simplex(y - lr * grad_y)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = f_new + ((t_k - 1.0) / t_new) * (f_new - f_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = f_new.copy()

        f_prev = f_new
        t_k = t_new

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-15:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
        if no_improve > 1000:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7 + 3)
            f_new = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f_new.copy()
            f_prev = f_new.copy()
            t_k = 1.0
            no_improve = 0
            lr = 4.0 * p * np.max(fast_convolve(f_new) ** max(p - 1, 1))
            lr = (1.0 / (lr + 1e-10)) * 0.8

        if verbose and it % 500 == 0:
            print(f"      FISTA-p{p} iter {it}: obj={2*n*best_obj:.8f}, lr={lr:.4e}")

    return best_f, 2 * n * best_obj


def adam_lp(f0, time_limit, p, lr_init=0.003, verbose=False):
    """Adam optimization of Lp norm of convolution on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = lp_grad(f, p)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-15:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if no_improve > 200:
            lr *= 0.9
        if no_improve > 800:
            rng = np.random.default_rng(t * 11 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.6

        if verbose and t % 500 == 0:
            print(f"      Adam-p{p} iter {t}: obj={2*n*best_obj:.8f}, lr={lr:.5f}")

    return best_f, 2 * n * best_obj


def lse_anneal(f0, time_limit, tau_init=None, lr_init=0.003, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)

        grad, conv = lse_grad(f, tau)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-15:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        tau *= 0.9998
        if tau < 1e-8:
            tau = 1e-8

        if no_improve > 150:
            lr *= 0.85
        if no_improve > 500:
            rng = np.random.default_rng(t * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.6
            tau = 0.03 * fast_convolve(f).max()

        if verbose and t % 500 == 0:
            print(f"      LSE iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}")

    return best_f, 2 * n * best_obj


def mirror_descent(f0, time_limit, verbose=False):
    """Mirror descent (exponentiated gradient) for the non-smooth max-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()

        # Subgradient with smoothing over near-max positions
        # Use top-k with softmax weighting
        k = min(10, 2 * n - 1)
        top_ks = np.argsort(conv)[-k:]
        # Softmax weights with temperature
        temp = 0.02 * M
        w = np.zeros(2 * n - 1)
        conv_top = conv[top_ks]
        w_top = np.exp((conv_top - conv_top.max()) / temp)
        w_top /= w_top.sum()
        w[top_ks] = w_top
        w /= w.sum()

        # Compute subgradient: g[j] = 2 * sum_k w[k] * f[k-j]
        w_rev = w[::-1]
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        W = np.fft.rfft(w_rev, fft_size)
        conv_au = np.fft.irfft(A * W, fft_size)[:total_len]
        g = 2.0 * conv_au[n - 1:2 * n - 1][::-1]

        # Exponentiated gradient update
        lr = 2.0 / np.sqrt(t + 1) / np.sqrt(n)
        f = f * np.exp(-lr * g)
        f = np.maximum(f, 1e-30)
        ss = f.sum()
        f = f / ss

        # Polyak averaging
        if verbose and t % 1000 == 0:
            print(f"      Mirror iter {t}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def subgradient_averaged(f0, time_limit, verbose=False):
    """Averaged subgradient method for max-convolution on simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    avg_f = f.copy()
    t0 = time.monotonic()
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()

        # Polyak step size: need a lower bound on optimal value
        # Lower bound: S^2/(2n-1) where S=sum(a)=1, so LB = 1/(2n-1)
        LB = 1.0 / (2 * n - 1)
        lr = (M - LB) / (n * M + 1e-10)
        lr = min(lr, 0.5)

        # Subgradient: 2 * e_k * f where k = argmax
        # For robustness, use top-3 average
        top_k = min(3, 2 * n - 1)
        top_indices = np.argsort(conv)[-top_k:]
        # Weight: 1 for max, 0.5 for others
        weights = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_indices):
            weights[idx] = 1.0 if i == top_k - 1 else 0.5
        weights /= weights.sum()

        w_rev = weights[::-1]
        total_len = n + len(weights) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        W = np.fft.rfft(w_rev, fft_size)
        conv_au = np.fft.irfft(A * W, fft_size)[:total_len]
        g = 2.0 * conv_au[n - 1:2 * n - 1][::-1]

        f_new = f - lr * g
        f_new = project_simplex(f_new)

        # Check if new point is better
        conv_new = fast_convolve(f_new)
        if conv_new.max() < conv.max():
            f = f_new
        else:
            # Don't update f, but still accumulate
            pass

        # Polyak averaging
        avg_f = (t * avg_f + f) / (t + 1)

        # Check averaged point
        if t % 100 == 0:
            avg_obj = fast_convolve(avg_f).max()
            if avg_obj < best_obj:
                best_obj = avg_obj
                best_f = avg_f.copy()

        if verbose and t % 2000 == 0:
            print(f"      Subgrad iter {t}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, center_frac=0.5, sigma_frac=0.3):
    """Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multimodal(n, num_peaks=2, center_fracs=None, sigma_frac=0.2):
    """Multi-modal initialization."""
    if center_fracs is None:
        center_fracs = np.linspace(0.2, 0.8, num_peaks)
    a = np.zeros(n)
    for c in center_fracs:
        x = np.arange(n) - n * c
        sigma = n * sigma_frac
        a += np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_power_law(n, exponent=2.0):
    """Power-law shaped initialization."""
    x = np.arange(n) / (n - 1) * 2 - 1  # [-1, 1]
    a = (1 - x ** 2) ** exponent
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def lp_fw_fast(f0, time_limit, verbose=False):
    """
    Fast LP-Frank-Wolfe using a heuristic direction finding
    (avoids full LP solve, uses greedy coordinate selection).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 50 == 0:
            conv = fast_convolve(f)
            M = conv.max()

        # Heuristic direction: find position j that reduces max convolution most
        # Simple approach: move mass from high-contributing to low-contributing positions
        conv = fast_convolve(f)
        M = conv.max()
        peak_idx = np.argmax(conv)

        # Find which j contributes most to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        products = f[idx] * f[peak_idx - idx]
        j_max = idx[np.argmax(products)]

        # Find a "safe" position to move mass to (low contribution to peaks)
        top_peaks = np.argsort(conv)[-5:]
        contrib_score = np.zeros(n)
        for pk in top_peaks:
            js2 = max(0, int(pk) - n + 1)
            je2 = min(n - 1, int(pk))
            for j2 in range(js2, je2 + 1):
                contrib_score[j2] += f[peak_idx if j2 == j_max else 0] * conv[pk]

        # Move mass from j_max to the position with lowest contribution score
        candidates = np.argsort(contrib_score)[:n // 4]
        j_min = candidates[0]

        if j_max == j_min:
            no_improve += 1
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5]:
            trial = f.copy()
            trial[j_max] -= alpha * f[j_max]
            trial[j_min] += alpha * f[j_max]
            trial = np.maximum(trial, 0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial /= st
            val = fast_convolve(trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-6:
            f[j_max] -= best_alpha * f[j_max]
            f[j_min] += best_alpha * f[j_max]
            f = np.maximum(f, 0)
            f /= f.sum()
            no_improve = 0
            if best_val < best_obj - 1e-15:
                best_obj = best_val
                best_f = f.copy()
                if verbose and it % 100 == 0:
                    print(f"      LP-FW-fast iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1

        if no_improve > 50:
            # Try a different perturbation strategy
            rng = np.random.default_rng(it * 23 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0

    return best_f, 2 * n * best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining Lp-FISTA, LSE annealing,
    mirror descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.4, 0.28)
        try_seq(parent, "default_init")

    # Try diverse initializations quickly
    inits = []
    for n_try in [512, 768, 1024]:
        inits.append(make_gaussian(n_try, 0.35, 0.25))
        inits.append(make_gaussian(n_try, 0.5, 0.3))
        inits.append(make_multimodal(n_try, 2, [0.3, 0.7], 0.15))
        inits.append(make_power_law(n_try, 1.5))
        inits.append(make_power_law(n_try, 3.0))
        inits.append(np.ones(n_try) / n_try)

    for i, a_init in enumerate(inits):
        if time.monotonic() > deadline - 1500:
            break
        try_seq(a_init, f"init_{i}")

    work_seq = best_seq if best_seq is not None else parent
    print(f"  Starting optimization with n={len(work_seq)}, score={best_score:.6f}")

    # === Phase 2: Lp FISTA (multi-stage p) ===
    print(f"\n=== Phase 2: Lp FISTA (n={len(work_seq)}) ===")
    f_work = work_seq.copy()

    # Stage p=4
    rem = deadline - time.monotonic() - 600
    t_alloc = min(rem, 300)
    if t_alloc > 30:
        print(f"  Lp p=4, time={t_alloc:.0f}s")
        f_work, obj = fista_lp(f_work, time_limit=t_alloc, p=4, verbose=True)
        try_seq(f_work, "fista_p4")

    # Stage p=8
    rem = deadline - time.monotonic() - 400
    t_alloc = min(rem, 250)
    if t_alloc > 20:
        print(f"  Lp p=8, time={t_alloc:.0f}s")
        f_work, obj = fista_lp(best_seq, time_limit=t_alloc, p=8, verbose=True)
        try_seq(f_work, "fista_p8")

    # Stage p=16
    rem = deadline - time.monotonic() - 250
    t_alloc = min(rem, 200)
    if t_alloc > 20:
        print(f"  Lp p=16, time={t_alloc:.0f}s")
        f_work, obj = adam_lp(best_seq, time_limit=t_alloc, p=16, lr_init=0.002, verbose=True)
        try_seq(f_work, "adam_p16")

    # Stage p=32
    rem = deadline - time.monotonic() - 150
    t_alloc = min(rem, 120)
    if t_alloc > 15:
        print(f"  Lp p=32, time={t_alloc:.0f}s")
        f_work, obj = adam_lp(best_seq, time_limit=t_alloc, p=32, lr_init=0.001, verbose=True)
        try_seq(f_work, "adam_p32")

    # === Phase 3: LSE Annealing ===
    print(f"\n=== Phase 3: LSE Annealing (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic() - 200
    t_alloc = min(rem, 250)
    if t_alloc > 20:
        print(f"  LSE, time={t_alloc:.0f}s")
        f_work, obj = lse_anneal(best_seq, time_limit=t_alloc, lr_init=0.003, verbose=True)
        try_seq(f_work, "lse_anneal")

    # Second LSE run with lower initial temperature
    rem = deadline - time.monotonic() - 100
    t_alloc = min(rem, 100)
    if t_alloc > 15:
        print(f"  LSE fine, time={t_alloc:.0f}s")
        tau_init = 0.02 * fast_convolve(best_seq).max()
        f_work, obj = lse_anneal(best_seq, time_limit=t_alloc, tau_init=tau_init, lr_init=0.001, verbose=True)
        try_seq(f_work, "lse_fine")

    # === Phase 4: Mirror Descent ===
    print(f"\n=== Phase 4: Mirror Descent (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic() - 100
    t_alloc = min(rem, 150)
    if t_alloc > 15:
        print(f"  Mirror descent, time={t_alloc:.0f}s")
        f_work, obj = mirror_descent(best_seq, time_limit=t_alloc, verbose=True)
        try_seq(f_work, "mirror")

    # === Phase 5: Averaged Subgradient ===
    print(f"\n=== Phase 5: Averaged Subgradient ===")
    rem = deadline - time.monotonic() - 80
    t_alloc = min(rem, 100)
    if t_alloc > 10:
        print(f"  Subgradient, time={t_alloc:.0f}s")
        f_work, obj = subgradient_averaged(best_seq, time_limit=t_alloc, verbose=True)
        try_seq(f_work, "subgrad_avg")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length Exploration ===")
    base = best_seq
    for target_n in [1024, 640, 512, 1280]:
        rem = deadline - time.monotonic() - 60
        if rem < 30 or target_n == len(base):
            continue
        f_interp = interpolate_sequence(base, target_n)
        sc = try_seq(f_interp, f"interp_n{target_n}")

        # Quick Lp optimization on interpolated
        t5 = min(rem * 0.35, 80)
        if t5 > 10:
            f_work, obj = fista_lp(f_interp, time_limit=t5, p=8, verbose=False)
            try_seq(f_work, f"fista_n{target_n}")

    # === Phase 7: Final quick refinements on best ===
    print(f"\n=== Phase 7: Final Refinements ===")
    if best_seq is not None:
        # Quick Adam Lp
        rem = deadline - time.monotonic() - 30
        t7 = min(rem, 60)
        if t7 > 10:
            f_work, obj = adam_lp(best_seq, time_limit=t7, p=16, lr_init=0.001)
            try_seq(f_work, "final_adam")

        # Quick LSE
        rem = deadline - time.monotonic() - 15
        t8 = min(rem, 30)
        if t8 > 5:
            tau_init = 0.01 * fast_convolve(best_seq).max()
            f_work, obj = lse_anneal(best_seq, time_limit=t8, tau_init=tau_init, lr_init=0.0005)
            try_seq(f_work, "final_lse")

        #LKPFW fast
        rem = deadline - time.monotonic() - 5
        t9 = min(rem, 20)
        if t9 > 5:
            f_work, obj = lp_fw_fast(best_seq, time_limit=t9, verbose=False)
            try_seq(f_work, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
