import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():

    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute gradient of mu*logsumexp(conv/mu) using FFT."""
    n = len(a)
    conv = fast_convolve(a)

    conv_s = conv / mu
    conv_s -= conv_s.max()
    exp_s = np.exp(conv_s)
    w = exp_s / exp_s.sum()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
