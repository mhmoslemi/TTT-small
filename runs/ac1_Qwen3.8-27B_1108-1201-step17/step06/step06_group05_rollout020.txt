def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]

    rho = np.nonzero(cond)[0][-1]

def build_constraint_matrix(f):

    """Compute gradient of (1/p) * sum(c^p) where c = f*f."""
    n = len(f)
    if p <= 1.0:
        w = np.ones(len(c))
    else:
        max_c = np.max(c)
        if max_c < 1e-15:
            return np.zeros(n)
        w = (c / max_c) ** (p - 1)
    w = w / w.sum()  # normalize for step size
    
    # Gradient: g_j = 2 * sum_i f_i * w_{j+i}
    # Using cross-correlation via FFT
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)

    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    bounds = [(0, None)] * n
    return c, A_ub, bounds


def solve_lp(f, M):
    """Solve the LP direction. Returns normalized g or None."""
    n = len(f)
    c, A_ub, bounds = build_lp_matrices(f)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """
    LP-based Frank-Wolfe optimization.
    Minimizes max_k (f*f)[k] subject to f>=0, sum(f)=1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        
        # Solve LP for direction
        t_lp = time.monotonic()
        g = solve_lp(f, M)
        lp_time = time.monotonic() - t_lp
        
        if g is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break
        
        # Check fixed point
        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            if no_improve >= 5:
                # Perturb to escape
                scale = 0.0005 * np.sqrt(no_improve / 5.0 + 1.0)
                noise = rng.standard_normal(n) * scale
                f = project_simplex(f + noise)
                c = fast_convolve(f)
                M = c.max()
                if M > best_obj * 1.001:
                    # Revert if perturbation was too destructive
                    f = best_f.copy()
                    c = fast_convolve(f)
                    M = c.max()
                no_improve = 0
                if verbose:
                    print(f"    iter {it}: perturbed, M={M:.12f}")
            continue
        
        no_improve = 0
        
        # Line search: try full step, then half, then bisect
        h_full = g  # f + 1*(g-f)
        obj_full = fast_convolve(h_full).max()
        
        if obj_full < M * (1 - 1e-14):
            f = g
            M = obj_full
        else:
            h_half = 0.5 * (f + g)
            obj_half = fast_convolve(h_half).max()
            
            if obj_half < M * (1 - 1e-14):
                # Bisect between 0.5 and 1.0
                lo, hi = 0.5, 1.0
                best_a = 0.5
                best_v = obj_half
                for _ in range(12):
                    mid = (lo + hi) / 2.0
                    h = f + mid * (g - f)
                    v = fast_convolve(h).max()
                    if v < best_v:
                        best_v = v
                        best_a = mid
                        lo = mid
                    else:
                        hi = mid
                f = f + best_a * (g - f)
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                M = best_v
            else:
                # Try smaller steps
                for a in [0.1, 0.2, 0.3]:
                    h = f + a * (g - f)
                    v = fast_convolve(h).max()
                    if v < M * (1 - 1e-14):
                        f = h
                        M = v
                        break
                else:
                    # Can't improve, count as no improvement
                    no_improve += 1
        
        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 100 == 0:
                print(f"    iter {it}: obj={best_obj:.14f}, t={elapsed:.1f}s")
    
    if verbose:
        print(f"  LP-FW done: {it} iters, obj={best_obj:.14f}, time={time.monotonic()-t0:.1f}s")
    
    return best_f, best_obj


def compute_pnorm_weights(c, p):
    """Compute weights for p-norm smoothing."""
    max_c = np.max(c)
    if max_c < 1e-15:
        return np.ones(len(c)) / len(c)
    if p <= 1.0:
        return np.ones(len(c))
    w = (c / max_c) ** (p - 1)
    s = w.sum()
    if s < 1e-15:
        return np.ones(len(c)) / len(c)
    return w / s


def compute_pnorm_gradient(f, c, p):
    """
    Gradient of (1/p)*sum(c^p) where c = f*f.
    g_j = 2 * sum_i f_i * w_{j+i}
    """
    n = len(f)
    w = compute_pnorm_weights(c, p)
    
    # Cross-correlation: g_j = 2 * sum_i f_i * w_{j+i}
    # = 2 * (f cross-correlate w)[j]
    # Using FFT: correlate(f, w)_j = sum_i f_i * w_{j+i}
    # = sum_i f_i * w_rev[2(n-1)-(j+i)]... 
    # Let's use: convolve(f, w_rev)[j + 2n - 2] = sum_i f[i] * w_rev[j+2n-2-i]
    # w_rev[k] = w[2n-2-k]
    # So conv[j+2n-2] = sum_i f[i] * w[2n-2-j-2n+2+i] = sum_i f[i] * w[i-j]
    # Hmm, let me just use a different formulation.
    # 
    # We want: g_j = 2 * sum_i f[i] * w[j+i] (where w is defined on [0, 2n-2])
    # = 2 * sum_i f[i] * w[j+i]
    #
    # Let w_shifted[k] = w[k] for k in [0, 2n-2]
    # Then g_j = 2 * (f * w_shifted)[j + (2n-1)] ... no
    #
    # Actually: (f * w)_k = sum_i f[i] * w[k-i]
    # We want sum_i f[i] * w[j+i] = (f * w)[j] with w replaced by w_rev
    # (f * w_rev)_k = sum_i f[i] * w_rev[k-i] = sum_i f[i] * w[2n-2-(k-i)] = sum_i f[i] * w[2n-2-k+i]
    # Set 2n-2-k = j => k = 2n-2-j
    # (f * w_rev)[2n-2-j] = sum_i f[i] * w[j+i] ✓
    
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1  # = n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    
    indices = 2 * n - 2 - np.arange(n)  # [2n-2, 2n-3, ..., n-1]
    g = 2.0 * conv[indices]
    return g


def pnorm_gradient_descent(f0, time_limit, verbose=False):
    """
    p-norm smoothed gradient descent with momentum and adaptive step size.
    Progressively increases p to sharpen the approximation to max.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    
    p_schedule = [4, 8, 16, 32, 64, 128, 256]
    stage_time = time_limit / len(p_schedule)
    
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit * 0.92:
            break
        stage_start = time.monotonic()
        lr = 1.0 / n
        momentum = np.zeros(n)
        mu = 0.7
        no_improve = 0
        it = 0
        
        while time.monotonic() - stage_start < stage_time:
            it += 1
            c = fast_convolve(f)
            g = compute_pnorm_gradient(f, c, float(p))
            
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            
            lr = min(lr, 0.8 / (gn * gn + 1e-10))
            
            momentum = mu * momentum - lr * g
            f_trial = project_simplex(f + momentum)
            
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()
            
            if obj_trial <= c.max() + 1e-12:
                f = f_trial
                if obj_trial < best_obj - 1e-14:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    lr *= 1.05
                else:
                    no_improve += 1
            else:
                no_improve += 1
                lr *= 0.6
                momentum *= 0.5
            
            if no_improve > 40:
                noise = rng.standard_normal(n) * 0.002
                f = project_simplex(best_f + noise)
                momentum = np.zeros(n)
                lr = 1.0 / n
                no_improve = 0
    
    return best_f, best_obj


def make_raised_cosine(n):
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1.0 + np.cos(np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Pad by 1 on each side for smoother interpolation
    a_padded = np.concatenate([[a[0]], a, [a[-1]]])
    x_padded = np.linspace(-1.0/(old_n-1), 1.0+1.0/(old_n-1), old_n+2)
    a_new = np.interp(new_x, x_padded, a_padded)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a sequence and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
    return best_seq, best_score
