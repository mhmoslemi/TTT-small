"""

in log-space with LSE smoothing, combined with L-BFGS polishing and multi-start

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_weighted(w, a):
    """
    Compute g_i = sum_k w_k * a_{k-i} for i=0,...,n-1
    where a_m = 0 for m outside [0, n-1].
    This is the cross-correlation of w (length 2n-1) and a (length n).
    """
    n = len(a)
    m = len(w)  # should be 2n-1
    size = 1 << (n + m - 1)
    # g_i = sum_j a_j * w_{i+j}
    # = conv(a[::-1], w)[n-1 : n-1+n]
    a_rev = a[::-1]
    conv_full = np.convolve(a_rev, w, mode='full')
    return conv_full[n-1 : n-1+n]


def cross_corr_weighted_fft(w, a):
    """FFT-based cross-correlation for speed."""
    n = len(a)
    m = len(w)
    size = 1 << (n + m - 2).bit_length()
    A = rfft(a, size)
    W = rfft(w, size)
    # g_i = sum_j a_j * w_{i+j}
    # This is the cross-correlation: (a ⊛ w)[i] = sum_j a_j * w_{j+i}
    # In FFT: ifft(conj(fft(a)) * fft(w)) gives sum_j a_{j-i} * w_j (not quite what we want)
    # Let's use: g = ifft(fft(a) * conj(fft(w)))... 
    # Actually, let me think again.
    # g_i = sum_{j=0}^{n-1} a_j * w_{i+j}
    # Define x = a (length n), y = w (length 2n-1)
    # g_i = sum_j x_j * y_{j+i}
    # This is (x_rev ⋆ y)[i] where ⋆ denotes cross-correlation... 
    # Or: g = ifft(fft(x_rev) * conj(fft(y))) ... 
    # Let me just use the time-domain approach for clarity, or use scipy.signal
    # Actually, for n up to ~4096, np.convolve is fast enough.
    a_rev = a[::-1]
    conv_full = np.convolve(a_rev, w, mode='full')
    return conv_full[n-1 : n-1+n]


def lse_objective_and_grad(z, T, n):
    """Compute LSE-smoothed objective and gradient in log-space."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    B = fast_convolve(a)
    TB = T * B
    TB_shifted = TB - TB.max()
    exp_TB = np.exp(TB_shifted)
    log_sum_exp = TB.max() + np.log(exp_TB.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    w = exp_TB / exp_TB.sum()
    g = cross_corr_weighted(w, a)
    grad_a = 4.0 * n * g
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def exponentiated_gradient_optimize(n, time_limit, T=50, eta_initial=0.1, verbose=True):
    """
    Exponentiated gradient (mirror descent) for minimizing the smoothed peak autocorrelation.
    Works in log-space: z_{t+1} = z_t - eta * grad_z(z_t)
    """
    t0 = time.monotonic()
    
    # Initialize with a monotonically increasing sequence
    z = np.arange(n, dtype=np.float64) * 0.1  # geometric in log-space
    z = z - z.mean()  # center it
    
    best_score = float('inf')
    best_z = z.copy()
    best_a = None
    
    eta = eta_initial
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        obj, grad_z = lse_objective_and_grad(z, T, n)
        
        # Update: z <- z - eta * grad_z
        z = z - eta * grad_z
        
        # Periodically adjust temperature and step size
        if iters % 10 == 0:
            a_check = np.exp(z - z.max())
            a_check = a_check / a_check.sum()
            B_check = fast_convolve(a_check)
            score_check = 2.0 * n * B_check.max()
            
            if score_check < best_score:
                best_score = score_check
                best_z = z.copy()
                best_a = a_check.copy()
            
            if verbose and iters % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    EG iter={iters} ({elapsed:.0f}s): LSE_obj={obj:.6f} exact_score={score_check:.6f} best={best_score:.6f}")
        
        # Decrease step size
        eta *= 0.999
        
        # Occasionally increase T for sharper peak identification
        if iters % 500 == 0 and T < 1000:
            T = min(T * 1.5, 1000.0)
    
    # Final evaluation
    a = np.exp(best_z - best_z.max())
    a = a / a.sum()
    score = 2.0 * n * fast_convolve(a).max()
    
    return a, score, iters


def lbgfs_refine(a0, time_limit, T=200, maxiter=2000, verbose=True):
    """L-BFGS refinement in log-space with LSE smoothing."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    z0 = z0 - z0.mean()  # center
    t0 = time.monotonic()
    stopped = [False]
    best_obj = lse_objective_and_grad(z0, T, n)[0]
    best_z = z0.copy()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        obj, _ = lse_objective_and_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        _, g = lse_objective_and_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        if not stopped[0]:
            a_out = np.exp(res.x - res.x.max())
            a_out = a_out / a_out.sum()
            return a_out
    except Exception as e:
        print(f"    L-BFGS error: {e}")
    
    return np.exp(best_z - best_z.max()) / np.exp(best_z - best_z.max()).sum()


def pow_smooth_refine(a0, time_limit, p=8, verbose=True):
    """
    Alternative smoothing: use (sum B[k]^p)^(1/p) instead of LSE.
    This gives a different landscape and might escape different local minima.
    """
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    z0 = z0 - z0.mean()
    t0 = time.monotonic()
    stopped = [False]
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_s = z - z.max()
        a = np.exp(z_s)
        a = a / a.sum()
        B = fast_convolve(a)
        return 2.0 * n * (np.sum(B**p))**(1.0/p)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        z_s = z - z.max()
        a = np.exp(z_s)
        s = a.sum()
        a = a / s
        B = fast_convolve(a)
        Bp_sum = np.sum(B**p)
        # d/dB_k (sum B^p)^(1/p) = B_k^(p-1) / Bp_sum^(1/p)
        dB_obj = (2.0 * n / Bp_sum**(1.0/p)) * B**((p-1))
        # dB_k / da_i = 2 * a_{k-i}
        g = cross_corr_weighted(dB_obj, a) * 2.0
        # Chain rule for softmax
        grad_a = g
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': 150000, 'disp': False}
        )
        if not stopped[0]:
            a_out = np.exp(res.x - res.x.max())
            a_out = a_out / a_out.sum()
            return a_out
    except Exception as e:
        print(f"    PowSmooth error: {e}")
    
    return a0


def soft_threshold_refine(a0, time_limit, n_steps=5000):
    """
    Direct subgradient method with adaptive step size and restarts.
    Focuses on the top few peaks of the autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.01
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        B = fast_convolve(a)
        # Weighted subgradient from top peaks
        top_k = min(20, len(B))
        peaks = np.argpartition(B, -top_k)[-top_k:]
        # Weight by B values (sharper = more weight)
        w = B[peaks] ** 4
        w = w / w.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-15:
                continue
            k = int(peaks[i])
            sg += w[i] * 2.0 * np.pad(a, (0, k))[:n] * np.pad(a, (k, 0))[:n]
            # Actually, dB[k]/da_i = 2*a_{k-i} for 0<=k-i<=n-1
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                sg[j] += 2.0 * w[i] * a[k - j]
        
        # Projected gradient step
        a_new = a - eta * sg
        # Project onto simplex
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum > 1e-10:
            a_new = a_new / a_new_sum
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            eta *= 0.8
            if eta < 1e-6:
                # Restart with perturbation
                a = best_a + np.random.default_rng(step).standard_normal(n) * 0.01
                a = np.maximum(a, 0)
                a = a / a.sum()
                eta = 0.005
                no_improve = 0
        
        if step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradSoft {elapsed:.0f}s: score={best_score:.6f} eta={eta:.6f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def monotonically_increasing_init(n, sharpness=1.0):
    """Create a monotonically increasing initialization."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(x * sharpness / n)
    a = a / a.sum()
    return a
