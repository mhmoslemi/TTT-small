       n = len(v)
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u)
       cond = u * np.arange(1, n+1) > (cssv - 1)
       rho = np.nonzero(cond)[0][-1]
       theta = (cssv[rho] - 1.0) / (rho + 1.0)
       return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

- Systematic coordinate descent with pair redistribution for fine polishing

import numpy as np
import time


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_normalized(a):
    """Objective for normalized a (sum=1): 2*n*max(conv)."""
    n = len(a)
    b = fast_conv(a)
    return 2.0 * n * b.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s


def make_ramp(n):
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def make_cosine(n, freq=1, amp=0.3):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def make_multi_cosine(n, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, 4):
        amp = rng.uniform(0.1, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def make_bell(n, width=0.5):
    x = np.arange(n) - n / 2.0
    hw = n * width / 2.0
    t = np.clip(np.abs(x) / hw, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f = np.interp(new_x, old_x, a)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def mirror_descent(a0, time_limit, verbose=False, lr_init=0.005, decay=0.99985,
                   num_peaks=40, temp_init=0.0008, temp_growth=1.001,
                   restart_every=800):
    """Mirror descent with multi-peak softmax subgradients."""
    f = np.maximum(a0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    b = fast_conv(f)
    best_obj = b.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    last_print = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(f)

        # Select peaks with softmax weighting
        k_act = min(num_peaks, len(b))
        peaks = np.argpartition(b, -k_act)[-k_act:]
        pv = b[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            sg[idx] += w[i] * (f[k - idx] + f[idx])

        # Mirror descent (multiplicative weights) update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        cur = fast_conv(f).max()
        if cur < best_obj - 1e-14:
            best_obj = cur
            best_f = f.copy()
            no_improve = 0
            if verbose and it - last_print > 500:
                print(f"    MD it={it}: obj={best_obj:.10f}, lr={lr:.6f}")
                last_print = it
        else:
            no_improve += 1

        it += 1
        lr *= decay
        temp *= temp_growth

        # Periodic restart with noise
        if no_improve >= restart_every:
            rng = np.random.default_rng(it * 7 + 3)
            noise_scale = 0.002 + 0.003 * (no_improve // restart_every)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = lr_init * max(0.3, 0.7 ** (no_improve // restart_every))
            no_improve = 0

        if lr < 1e-10:
            lr = lr_init * 0.1
            no_improve = 0

    return best_f, best_obj


def euclidean_fw(a0, time_limit, verbose=False, num_peaks=50, temp=0.001):
    """Euclidean Frank-Wolfe with away steps."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    b = fast_conv(f)
    best_obj = b.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(f)

        # Find active peaks
        k_act = min(num_peaks, len(b))
        peaks = np.argpartition(b, -k_act)[-k_act:]
        pv = b[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            sg[idx] += w[i] * (f[k - idx] + f[idx])

        # FW direction: minimize <sg, v> over simplex = vertex with min sg
        v_idx = np.argmin(sg)

        # Away step: maximize <sg, v> over support of f
        support = np.where(f > 1e-10)[0]
        if len(support) > 1:
            away_idx = support[np.argmax(sg[support])]
        else:
            away_idx = v_idx

        # Try forward and away steps with line search
        best_alpha = 0.0
        best_val = b.max()

        # Forward step
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            ft = f.copy()
            ft[v_idx] += alpha
            ft = np.maximum(ft, 0.0)
            ft = ft / ft.sum()
            val = fast_conv(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Away step (if different from forward)
        if away_idx != v_idx:
            for alpha in [0.05, 0.1, 0.15, 0.2, 0.3]:
                ft = f.copy()
                ft[away_idx] -= alpha * f[away_idx]
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-12:
                    continue
                ft = ft / st
                val = fast_conv(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = -alpha  # negative indicates away step

        if best_alpha > 1e-10:
            f = f.copy()
            f[v_idx] += best_alpha
            f = np.maximum(f, 0.0)
            f = f / f.sum()
        elif best_alpha < -1e-10:
            a_amt = -best_alpha
            f = f.copy()
            f[away_idx] -= a_amt * f[away_idx]
            f = np.maximum(f, 0.0)
            f = f / f.sum()

        if b.max() < best_obj - 1e-14 or best_val < best_obj - 1e-14:
            b = fast_conv(f)
            if b.max() < best_obj:
                best_obj = b.max()
                best_f = f.copy()
                if verbose and it % 50 == 0:
                    print(f"    EFW it={it}: obj={best_obj:.10f}")

        it += 1
        if it % 200 == 0 and best_obj - b.max() > 0.005:
            # Reset if stuck far from best
            f = best_f.copy()

    return best_f, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Targeted coordinate descent: reduce peak contributors, redistribute."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    b = fast_conv(f)
    best_obj = b.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(f)

        # Find top peaks to target
        num_peaks = min(8, len(b))
        peak_indices = np.argpartition(b, -num_peaks)[-num_peaks:]
        peak_indices = peak_indices[np.argsort(b[peak_indices])[::-1]]

        improved = False
        for k in peak_indices:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            if len(idx) == 0:
                continue
            products = f[idx] * f[k - idx]

            # Try reducing top contributors
            top_contrib = idx[np.argsort(products)[::-1][:5]]

            for src in top_contrib:
                partner = k - int(src)
                # Find target: non-contributing coordinates
                nc_mask = np.ones(n, dtype=bool)
                nc_mask[idx] = False

                for frac in [0.05, 0.1, 0.15, 0.25, 0.4]:
                    diff = f[int(src)] * frac
                    if diff < 1e-12:
                        continue
                    trial = f.copy()
                    trial[int(src)] -= diff
                    nc_vals = trial[nc_mask]
                    nc_sum = nc_vals.sum()
                    if nc_sum < 1e-12:
                        continue
                    trial[nc_mask] += diff * nc_vals / nc_sum
                    trial = np.maximum(trial, 0.0)
                    trial = trial / trial.sum()
                    obj = fast_conv(trial).max()
                    if obj < best_obj - 1e-13:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random pair moves
            rng = np.random.default_rng(it)
            for _ in range(30):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] < 1e-10:
                    continue
                for frac in [0.1, 0.3, 0.5, 0.7]:
                    amt = f[i] * frac
                    trial = f.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    trial = np.maximum(trial, 0.0)
                    trial = trial / trial.sum()
                    obj = fast_conv(trial).max()
                    if obj < best_obj - 1e-13:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    CD it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 80:
                rng = np.random.default_rng(it * 13 + 5)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 0.0)
                f = f / f.sum()
                no_improve = 0

        it += 1

    return best_f, best_obj

            try_seq(a_opt, "md_parent")

    # === Phase 2: Euclidean FW ===
    print("\n=== Phase 2: Euclidean FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t_alloc = min(rem * 0.5, 150)
            print(f"  EFW: n={len(best_seq)}, {t_alloc:.0f}s")
            a_opt, obj = euclidean_fw(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "efw")

    # === Phase 3: Coordinate Descent ===
    print("\n=== Phase 3: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(rem * 0.6, 200)
            print(f"  CD: n={len(best_seq)}, {t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 4: Multi-length exploration ===
    print("\n=== Phase 4: Multi-Length ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_lengths = [1024, 1280, 896, 640, 512, 1536, 2048]
        target_lengths = [tl for tl in target_lengths if tl != base_n]

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            f_interp = interpolate_seq(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            if sc < best_score + 0.03:
                # Optimize at this length
                t_md = min(rem * 0.35, 120)
                if t_md > 20:
                    print(f"  MD n={target_n}: {t_md:.0f}s")
                    a_opt, obj = mirror_descent(f_interp, time_limit=t_md, verbose=False)
                    try_seq(a_opt, f"md_n{target_n}")
                    if best_seq is not None and len(best_seq) == target_n:
                        t_cd = min(rem * 0.15, 40)
                        if t_cd > 10:
                            a_opt2, _ = coordinate_descent(best_seq, time_limit=t_cd)
                            try_seq(a_opt2, f"cd_n{target_n}")

    # === Phase 5: Second mirror descent round with refined params ===
    print("\n=== Phase 5: Refined MD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem * 0.7, 200)
            print(f"  Refined MD: n={len(best_seq)}, {t_alloc:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc,
                                        lr_init=0.003, decay=0.99992,
                                        num_peaks=60, temp_init=0.002,
                                        temp_growth=1.0008, verbose=True)
            try_seq(a_opt, "refined_md")

    # === Phase 6: Fine polish ===
    print("\n=== Phase 6: Fine Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.7, 100)
            print(f"  Fine polish: n={len(best_seq)}, {t_alloc:.0f}s")
            a_opt, obj = fine_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "fine_polish")

    # === Phase 7: Final squeeze ===
    print("\n=== Phase 7: Final ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_alloc = min(rem, 50)
            print(f"  Final MD: {t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc,
                                      lr_init=0.001, decay=0.99995,
                                      num_peaks=80, temp_init=0.005)
            try_seq(a_opt, "final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
