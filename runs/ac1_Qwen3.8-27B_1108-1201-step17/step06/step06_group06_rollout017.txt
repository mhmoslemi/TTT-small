"""

Uses: (1) mirror descent with entropy regularization on the simplex,

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent_optimize(f0, time_limit, tau_init=None, verbose=False):
    """
    Mirror descent with negative entropy on the simplex.
    Update: a_new[j] ∝ a[j] * exp(-eta * grad[j])
    This is equivalent to multiplicative weights.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
        s = 1.0
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.05 * conv.max()
    else:
        tau = tau_init

    eta = 0.5  # Step size for mirror descent
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv, w = compute_subgrad_smoothed(f, tau)
        M = conv.max()

        # Mirror descent update: a_new ∝ a * exp(-eta * grad)
        # Need to be careful with scaling
        log_update = -eta * grad
        log_update -= log_update.max()  # For numerical stability
        f_new = f * np.exp(log_update)
        f_new = np.maximum(f_new, 1e-30)
        s_new = f_new.sum()
        f_new /= s_new

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive tau
        tau *= 0.9995
        if tau < 1e-8:
            tau = 1e-8

        # Adaptive eta
        if no_improve > 50:
            eta *= 0.9
        elif no_improve == 0:
            eta = min(eta * 1.02, 2.0)

        if no_improve > 300:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7 + 3)
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            f = np.maximum(f, 1e-10)
            f /= f.sum()
            m = 0
            eta = 0.5
            tau = 0.04 * fast_convolve(f).max()
            no_improve = 0
            if verbose:

        if verbose and it % 200 == 0:
            print(f"    MD iter {it}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, eta={eta:.4f}")

    return best_f, 2 * n * best_obj


def adam_optimize(f0, time_limit, tau_init=None, lr_init=0.003, verbose=False):
    """Adam optimization with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.06 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = compute_subgrad_smoothed(f, tau)[:2]
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        tau *= 0.9997
        if tau < 1e-8:
            tau = 1e-8

        if no_improve > 80:
            lr *= 0.8
        if no_improve > 350:
            rng = np.random.default_rng(t * 11 + 5)
            noise_scale = 0.002 + 0.002 * min(no_improve / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.6
            tau = 0.04 * fast_convolve(f).max()
            if verbose:
                print(f"    Adam restart at iter {t}: obj={2*n*best_obj:.8f}")

        if verbose and t % 500 == 0:
            print(f"    Adam iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}")

    return best_f, 2 * n * best_obj


def pairwise_cd(f0, time_limit, verbose=False):
    """
    Pairwise coordinate descent: for each peak, find the best pair of
    coordinates to transfer mass between to reduce that peak.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Focus on top peaks
        n_peaks = min(3, 2 * n - 1)
        peak_indices = np.argsort(conv)[-n_peaks:]

        for pk in peak_indices:
            pk = int(pk)
            # Contributing pairs: (j, pk-j) for valid j
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            
            # Find the pair with maximum product
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max
            
            # Try transferring mass from j_max to some other position
            # that doesn't contribute to this peak
            old_val = f[j_max]
            if old_val < 1e-10:
                continue
            
            # Candidate positions to transfer to (not contributing to top peaks)
            contrib_mask = np.zeros(n, dtype=bool)
            for pk2 in peak_indices[:2]:
                js2 = max(0, int(pk2) - n + 1)
                je2 = min(n - 1, int(pk2))
                contrib_mask[js2:je2 + 1] = True
            
            # Also mark j_other as contributing (it's part of the peak pair)
            contrib_mask[j_other] = True
            
            free_mask = ~contrib_mask
            if free_mask.sum() < 2:
                continue
            
            # Try a few transfer fractions
            for frac in [0.1, 0.2, 0.35, 0.5]:
                diff = old_val * frac
                trial = f.copy()
                trial[j_max] = old_val - diff
                # Distribute to free positions proportionally
                free_idx = np.where(free_mask)[0]
                weights = trial[free_idx] + 1e-10
                weights /= weights.sum()
                trial[free_idx] += diff * weights
                
                trial = np.maximum(trial, 0)
                trial /= trial.sum()
                obj = fast_convolve(trial).max()
                
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it % 50 == 0:

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_spectral_init(n, center=0.4):
    """
    Create initial sequence based on spectral design.
    We want |A(ω)|² to be as flat as possible (to minimize peak of inverse transform).
    A flat |A(ω)|² corresponds to a delta autocorrelation.
    We approximate this with a window that has good spectral properties.
    """
    # Start with a raised cosine window (Hann-like) but asymmetric
    x = np.arange(n)
    t = (x - n * center) / (n * 0.35)
    a = np.cos(np.pi * t / 2) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tophat_init(n, frac=0.6):
    """Top-hat with smooth edges."""
    x = np.arange(n) / (n - 1)
    a = np.ones(n)
    edge = n * (1 - frac) / 2
    for i in range(n):
        if i < edge:
            a[i] = 0.5 * (1 + np.cos(np.pi * i / edge))
        elif i > n - 1 - edge:
            a[i] = 0.5 * (1 + np.cos(np.pi * (n - 1 - i) / edge))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def multi_peak_mirror_descent(f0, time_limit, verbose=False):
    """
    Mirror descent that tracks multiple peaks simultaneously.
    Uses a weighted combination of peak subgradients.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    tau = 0.03 * conv.max()
    eta = 0.3

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    it = 0

    # Momentum for the multiplicative update
    log_momentum = np.zeros(n)
    beta_m = 0.7

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Compute weights with multiple temperature scales
        # This gives more weight to peaks but also considers near-peaks
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute gradient
        w_rev = w[::-1]
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        W = np.fft.rfft(w_rev, fft_size)
        conv_result = np.fft.irfft(A * W, fft_size)
        h = conv_result[n - 1:2 * n - 1][::-1]
        grad = 2.0 * h

        # Mirror descent with momentum
        log_momentum = beta_m * log_momentum + (1 - beta_m) * (-eta * grad)
        log_update = log_momentum
        log_update -= log_update.max()
        f_new = f * np.exp(log_update)
        f_new = np.maximum(f_new, 1e-30)
        f_new /= f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        tau *= 0.9998
        if tau < 1e-8:
            tau = 1e-8

        if no_improve > 60:
            eta *= 0.92
        elif no_improve == 0:
            eta = min(eta * 1.01, 1.5)

        if no_improve > 400:
            rng = np.random.default_rng(it * 17 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            f = np.maximum(f, 1e-10)
            f /= f.sum()
            log_momentum = np.zeros(n)
            eta = 0.3
            tau = 0.03 * fast_convolve(f).max()
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    MMD iter {it}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def lp_direction_fast(f, M, n, n_constraints=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for selected constraints, g >= 0.
    Uses a subset of constraints for speed.
    """
    if n_constraints is None:
        n_constraints = min(2 * n - 1, 200)
    
    # Select the most important constraints (those closest to being binding)
    conv_f = fast_convolve(f)
    # Constraints where (f*g) is most likely binding are those with large f-conv values
    # Actually, the constraint is (f*g)[k] <= M for all k
    # The "most important" ones are where the current (f*f)[k] is largest
    # (since g is close to f at the start)
    important = np.argsort(conv_f)[-n_constraints:]
    
    c = -np.ones(n)
    A_ub = np.zeros((len(important), n))
    for i, k in enumerate(important):
        js = max(0, int(k) - n + 1)
        je = min(n - 1, int(k))
        A_ub[i, js:je+1] = f[js:je+1]
    
    b_ub = np.full(len(important), M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_fast(f0, time_limit, verbose=False):
    """
    Fast LP-Frank-Wolfe with reduced constraint set.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_direction_fast(f, M * m_exp, n, n_constraints=150)
        if g is None:
            m_exp *= 1.3
            if m_exp > 8.0:
                rng = np.random.default_rng(it * 19 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.2
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.15, 6.0)
            if no_improve > 100:
                break
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new /= f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 6.0)

    return best_f, 2 * n * best_obj

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
(1) Mirror descent with negative entropy (multiplicative updates) on the simplex,
    which is more natural than Euclidean projection for non-negative sequences.
(2) Multi-peak tracking with adaptive softmax weighting to address multiple
    near-optimal convolution peaks simultaneously.
(3) Aggressive simulated-annealing-style restarts with large perturbations.
(4) Fast LP-Frank-Wolfe with reduced constraint sets.
(5) Peak-targeted random coordinate perturbation as an exploration mechanism.
(6) Multi-length exploration with interpolation-based transfer.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_conv(f, tau):
    """
    Compute gradient of the smoothed max-convolution objective.
    Uses softmax weighting: w[k] = exp(conv[k]/tau) / sum(exp(conv/tau))
    Returns (gradient, convolution).
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # Gradient: g[j] = 2 * sum_k w[k] * f[k-j] (valid range)
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]

    grad = (2.0 / tau) * h
    return grad, conv


def mirror_descent_anneal(f0, time_limit, tau_init=None, verbose=False):
    """
    Mirror descent with negative entropy on the simplex + simulated annealing.
    Uses multiplicative updates: a_new[j] ∝ a[j] * exp(-eta * grad[j])
    With periodic large restarts to escape local optima.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.08 * conv.max()
    else:
        tau = tau_init

    eta = 0.4
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)

    # Momentum on the log-update
    log_mom = np.zeros(n)
    beta_m = 0.6

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv = compute_grad_conv(f, tau)
        M = conv.max()

        # Mirror descent with momentum
        log_mom = beta_m * log_mom + (1 - beta_m) * (-eta * grad)
        lu = log_mom - log_mom.max()
        f_new = f * np.exp(lu)
        f_new = np.maximum(f_new, 1e-30)
        f_new /= f_new.sum()

        # Simulated annealing acceptance
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        temp = 0.002 * M * max(0.1, 1.0 - it / 2000.0)

        if obj_new <= best_obj or rng.random() < np.exp(-(obj_new - best_obj) / (temp + 1e-15)):
            f = f_new
        else:
            f = f_new  # Still move, but don't update best

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature (annealing)
        tau *= 0.9996
        if tau < 1e-8 * M:
            tau = 1e-8 * M

        # Adaptive step size
        if no_improve > 40:
            eta *= 0.93
        elif no_improve == 0:
            eta = min(eta * 1.015, 1.8)
        eta = max(eta, 0.01)

        # Large restart
        if no_improve > 350:
            scale = 0.005 + 0.01 * min(no_improve / 1000.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            f = np.maximum(f, 1e-10)
            f /= f.sum()
            log_mom = np.zeros(n)
            eta = 0.4
            tau = 0.05 * fast_convolve(f).max()
            no_improve = 0
            if verbose:
                print(f"    MD-A restart iter {it}: obj={2*n*best_obj:.8f}")

        if verbose and it % 300 == 0:
            print(f"    MD-A iter {it}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, eta={eta:.4f}")

    return best_f, 2 * n * best_obj


def adam_refine(f0, time_limit, tau_init=None, lr_init=0.003, verbose=False):
    """Adam with adaptive restarts for refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.05 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = compute_grad_conv(f, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        tau *= 0.9997
        if tau < 1e-8 * conv.max():
            tau = 1e-8 * conv.max()

        if no_improve > 60:
            lr *= 0.85
        if no_improve > 300:
            scale = 0.003 + 0.003 * min(no_improve / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.7
            tau = 0.04 * fast_convolve(f).max()
            if verbose:
                print(f"    Adam restart iter {t}: obj={2*n*best_obj:.8f}")

        if verbose and t % 500 == 0:
            print(f"    Adam iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}")

    return best_f, 2 * n * best_obj


def peak_random_perturb(f0, time_limit, verbose=False):
    """
    Peak-targeted random perturbation: specifically attacks the peak by
    reducing high-contribution coordinates and redistributing to safe positions.
    Uses simulated annealing acceptance.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find top 2 peaks
        peak_indices = np.argsort(conv)[-min(2, len(conv)):]
        improved = False

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]

            # Pick a random high-contribution coordinate
            top_contrib = idx[np.argsort(products)[-min(5, len(idx)):]]
            j_pick = top_contrib[rng.integers(len(top_contrib))]
            j_other = pk - j_pick

            old_val = f[j_pick]
            if old_val < 1e-8:
                continue

            # Try several transfer fractions
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                diff = old_val * frac
                trial = f.copy()
                trial[j_pick] = old_val - diff

                # Redistribute to random safe positions
                # Safe = not contributing to top peaks
                safe_mask = np.ones(n, dtype=bool)
                for pk2 in peak_indices:
                    js2 = max(0, int(pk2) - n + 1)
                    je2 = min(n - 1, int(pk2))
                    safe_mask[js2:je2 + 1] = False
                safe_mask[j_other] = False

                if not safe_mask.any():
                    continue
                safe_idx = np.where(safe_mask)[0]
                n_transfer = min(len(safe_idx), rng.integers(1, min(5, len(safe_idx)) + 1))
                targets = safe_idx[rng.choice(len(safe_idx), n_transfer, replace=False)]
                per_target = diff / n_transfer
                trial[targets] += per_target

                trial = np.maximum(trial, 0)
                trial /= trial.sum()
                obj = fast_convolve(trial).max()

                # Accept with annealing
                temp = 0.001 * M * max(0.1, 1.0 - it / 500.0)
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
                elif rng.random() < np.exp(-(obj - best_obj) / (temp + 1e-15)):
                    f = trial
                    break

        if not improved and it % 50 == 0:
            # Periodic random restart
            if it % 200 == 0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                f = np.maximum(f, 1e-10)
                f /= f.sum()

        if verbose and it % 100 == 0:
            print(f"    PRP iter {it}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def lp_fw_reduced(f0, time_limit, verbose=False):
    """
    LP-Frank-Wolfe with reduced constraint set for speed.
    Only enforces the top-K most binding constraints.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 5:
            break

        # Solve LP with reduced constraints
        g = _solve_lp_reduced(f, M * m_exp, n, n_top=120)
        if g is None:
            m_exp *= 1.3
            if m_exp > 10.0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.2
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.1, 8.0)
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new /= f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 5 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.15, 8.0)

    return best_f, 2 * n * best_obj


def _solve_lp_reduced(f, M, n, n_top=100):
    """Solve LP with top-n_top most binding constraints only."""
    conv_f = fast_convolve(f)
    important = np.argsort(conv_f)[-n_top:]

    c = -np.ones(n)
    A_ub = np.zeros((len(important), n))
    for i, k in enumerate(important):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[i, js:je + 1] = f[js:je + 1]

    b_ub = np.full(len(important), M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_diverse_init(n, center_frac=0.42, width_frac=0.35):
    """Create a diverse smooth initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * width_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    # Add a secondary bump for asymmetry
    x2 = np.arange(n) - n * (1 - center_frac) * 0.7
    sigma2 = n * 0.15
    a += 0.3 * np.exp(-0.5 * (x2 / sigma2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining mirror descent, Adam, LP-FW,
    peak-targeted perturbation, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 0: Load parent ===
    print("=== Phase 0: Setup ===")
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_score = p_score
            best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_diverse_init(512)
        try_seq(parent, "default")

    n_parent = len(parent)

    # === Phase 1: Quick diverse initialization ===
    print("\n=== Phase 1: Diverse Initialization ===")
    inits = [parent.copy()]
    for n_try in [n_parent, n_parent + 128, 1024]:
        inits.append(make_diverse_init(n_try, 0.35, 0.3))
        inits.append(make_diverse_init(n_try, 0.5, 0.25))
        # Top-hat
        x = np.arange(n_try) / n_try
        a_th = np.ones(n_try)
        edge = 0.15
        mask = (x < edge) | (x > 1 - edge)
        a_th[mask] = 0.5 * (1 + np.cos(np.pi * np.where(x[mask] < 0.5, x[mask] / edge, (1 - x[mask]) / edge)))
        a_th = np.maximum(a_th, 0)
        a_th /= a_th.sum()
        inits.append(a_th)

    for i, a_init in enumerate(inits):
        if time.monotonic() > deadline - 1500:
            break
        try_seq(a_init, f"init_{i}")

    print(f"  After init: best={best_score:.6f} (n={len(best_seq)})")

    # === Phase 2: Mirror descent with annealing (main innovation) ===
    work = best_seq.copy()
    print(f"\n=== Phase 2: Mirror Descent Annealing (n={len(work)}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 550)
    if t2 > 60:
        print(f"  Time: {t2:.0f}s")
        a_opt, obj = mirror_descent_anneal(work, time_limit=t2 * 0.55, verbose=True)
        try_seq(a_opt, "MDA_1")

        if time.monotonic() < deadline - 550:
            rem2 = deadline - time.monotonic() - 550
            t2b = min(rem2, 250)
            a_opt2, obj2 = mirror_descent_anneal(best_seq, time_limit=t2b,
                                                  tau_init=0.03 * fast_convolve(best_seq).max(), verbose=True)
            try_seq(a_opt2, "MDA_2")

    # === Phase 3: Adam refinement ===
    print(f"\n=== Phase 3: Adam Refinement ===")
    if time.monotonic() < deadline - 450:
        rem = deadline - time.monotonic() - 450
        t3 = min(rem, 300)
        print(f"  Time: {t3:.0f}s")
        a_opt, obj = adam_refine(best_seq, time_limit=t3, verbose=True)
        try_seq(a_opt, "adam")

    # === Phase 4: LP-FW with reduced constraints ===
    print(f"\n=== Phase 4: LP-FW (reduced) ===")
    if time.monotonic() < deadline - 250:
        rem = deadline - time.monotonic() - 250
        t4 = min(rem, 250)
        print(f"  Time: {t4:.0f}s")
        a_opt, obj = lp_fw_reduced(best_seq, time_limit=t4, verbose=True)
        try_seq(a_opt, "lpfw")

    # === Phase 5: Peak-targeted random perturbation ===
    print(f"\n=== Phase 5: Peak Random Perturbation ===")
    if time.monotonic() < deadline - 150:
        rem = deadline - time.monotonic() - 150
        t5 = min(rem, 120)
        print(f"  Time: {t5:.0f}s")
        a_opt, obj = peak_random_perturb(best_seq, time_limit=t5, verbose=True)
        try_seq(a_opt, "prp")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length ===")
    if time.monotonic() < deadline - 120:
        for target_n in [len(best_seq) + 256, 1536, 2048]:
            if time.monotonic() > deadline - 60:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            try_seq(f_interp, f"interp_{target_n}")

            rem = deadline - time.monotonic() - 50
            if rem > 30 and target_n <= 1536:
                a_opt, obj = adam_refine(f_interp, time_limit=min(rem, 60), verbose=False)
                try_seq(a_opt, f"adam_{target_n}")

    # === Phase 7: Second round of mirror descent on best ===
    print(f"\n=== Phase 7: Second MD Round ===")
    if time.monotonic() < deadline - 60:
        rem = deadline - time.monotonic() - 50
        t7 = min(rem, 100)
        a_opt, obj = mirror_descent_anneal(best_seq, time_limit=t7,
                                           tau_init=0.02 * fast_convolve(best_seq).max(), verbose=False)
        try_seq(a_opt, "MDA_3")

    # === Phase 8: Final Adam polish ===
    if time.monotonic() < deadline - 20:
        rem = deadline - time.monotonic() - 10
        t8 = min(rem, 40)
        a_opt, obj = adam_refine(best_seq, time_limit=t8,
                                 tau_init=0.015 * fast_convolve(best_seq).max(),
                                 lr_init=0.001, verbose=False)
        try_seq(a_opt, "polish_adam")

    # === Phase 9: Final LP-FW squeeze ===
    if time.monotonic() < deadline - 10:
        rem = deadline - time.monotonic() - 5
        t9 = min(rem, 20)
        a_opt, obj = lp_fw_reduced(best_seq, time_limit=t9, verbose=False)
        try_seq(a_opt, "polish_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
