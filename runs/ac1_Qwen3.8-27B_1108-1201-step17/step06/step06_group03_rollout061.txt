def fast_convolve(a):

            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] + f[idx]
            sg[js:je + 1] += w[i] * contrib
        
        # Multiplicative update
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s
        
        # Evaluate and backtrack if needed
        obj_new = objective_fast(f_new)
        if obj_new > M * 1.01 + 1e-12:
            # Dampen
            best_damp = 0
            best_trial_obj = obj_new
            for damp in [0.5, 0.3, 0.1, 0.05]:
                f_trial = (1.0 - damp) * f + damp * f_new
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = objective_fast(f_trial)
                if obj_trial < M + 1e-12:
                    best_damp = damp
                    best_trial_obj = obj_trial
                    break
            if best_damp > 0:
                f_new = (1.0 - best_damp) * f + best_damp * f_new
                f_new = np.maximum(f_new, 1e-15)
                f_new = f_new / f_new.sum()
                obj_new = best_trial_obj
            else:
                lr *= 0.5
                if lr < 1e-8:
                    break
                continue
        
        f = f_new
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"        SMD it{it}: obj={best_obj:.8f} tau={tau:.1f} lr={lr:.5f}")
        else:
            no_improve += 1
        
        # Adaptive parameters
        tau *= tau_decay
        lr *= lr_decay
        
        # Restart
        if no_improve > restart_every:
            rng = np.random.default_rng(it * 7 + int(tau))
            noise = rng.standard_normal(n) * 0.004
            f = np.maximum(best_f + noise, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.5
            tau = tau_init * 2
            no_improve = 0
        
        if lr < 1e-8:
            break
    
    return best_f, best_obj


def peak_pair_search(f0, time_limit, verbose=False):
    """
    Local search that directly attacks the worst peak by adjusting pairs.
    For the worst peak at position k, finds pairs (j, k-j) with large product
    and tries to make them more unequal.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    best_obj = objective_fast(f)
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        k_peak = int(np.argmax(conv))
        
        # Find contributing pairs
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        products = f[idx] * f[k_peak - idx]
        
        # Try the top-5 contributing pairs
        top_pairs = idx[np.argsort(products)[-5:]]
        
        improved = False
        for j in top_pairs:
            other = k_peak - j
            if other < 0 or other >= n or other == j:
                continue
            
            # Try transferring mass to make them more unequal
            a_j = f[j]
            a_other = f[other]
            if a_j < 1e-10 or a_other < 1e-10:
                continue
            
            # Try several transfer amounts
            for frac in [0.1, 0.2, 0.3, 0.1, 0.2, 0.3]:
                # Transfer from smaller to larger (make more unequal)
                if a_j > a_other:
                    delta = a_other * frac
                    f_trial = f.copy()
                    f_trial[j] = a_j + delta
                    f_trial[other] = a_other - delta
                else:
                    delta = a_j * frac
                    f_trial = f.copy()
                    f_trial[j] = a_j - delta
                    f_trial[other] = a_other + delta
                
                f_trial = np.maximum(f_trial, 0)
                s = f_trial.sum()
                if s < 1e-15:
                    continue
                f_trial = f_trial / s
                
                obj_trial = objective_fast(f_trial)
                if obj_trial < best_obj - 1e-12:
                    f = f_trial
                    best_obj = obj_trial
                    best_f = f.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Try random perturbations
                rng = np.random.default_rng(it * 13)
                j1 = rng.integers(0, n)
                j2 = rng.integers(0, n)
                if j1 != j2:
                    delta = f[j1] * 0.05 * (1 if rng.random() < 0.5 else -1)
                    f_trial = f.copy()
                    f_trial[j1] += delta
                    f_trial[j2] -= delta
                    f_trial = np.maximum(f_trial, 0)
                    s = f_trial.sum()
                    if s > 1e-15:
                        f_trial = f_trial / s
                        obj_trial = objective_fast(f_trial)
                        if obj_trial < best_obj - 1e-12:
                            f = f_trial
                            best_obj = obj_trial
                            best_f = f.copy()
                            no_improve = 0
                            continue
                no_improve = 1
                if no_improve > 200:
                    break
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find the optimal value
    (given others fixed) via 1D golden section or grid search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = objective_fast(f)
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        improved_this_pass = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try several candidate values
            candidates = [old_val * mult for mult in [0.5, 0.7, 0.8, 1.2, 1.3, 1.5, 2.0, 0.3, 0.9, 1.1]]
            candidates.append(0.0)  # try zeroing out
            
            for new_val in candidates:
                if new_val < -1e-10 or new_val > 0.5:
                    continue
                new_val = max(0.0, new_val)
                
                # Rescale others to maintain sum = 1
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                
                val = objective_fast(f_trial)
                if val < best_j_obj - 1e-13:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved_this_pass = True
        
        if not improved_this_pass:
            break
    
    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f[:n]  # f has length n, and for each j, the nonzero entries are at positions j..j+n-1
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = objective_fast(f)
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_factor = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        g = lp_direction(f, M * m_factor, n)
        if g is None:
            m_factor *= 1.15
            if m_factor > 10.0:
                rng = np.random.default_rng(it * 17)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 0)
                f = f / f.sum()
                m_factor = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_factor *= 1.1
            continue
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                m_factor = min(m_factor * 1.05, 5.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 31)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 0)
                f = f / f.sum()
                m_factor = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = objective_fast(ft)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            # Fine search
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(5):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = objective_fast(ft)
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
            
            f_new = (1.0 - best_alpha) * f + best_alpha * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            conv_new = fast_convolve(f_new)
            M = conv_new.max()
            f = f_new
            m_factor = 1.0
            
            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"          LP-FW it{it}: obj={best_obj:.8f}")
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.05, 5.0)
        
        if no_improve > 100:
            break
    
    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization strategy:
1. Multi-length exploration with diverse asymmetric initializations
2. Smoothed Mirror Descent (SMD) with adaptive temperature and learning rate
3. Peak-Pair local search (directly attacks worst convolution peak)
4. LP-Frank-Wolfe refinement
5. Coordinate descent polish
6. Coarse-to-fine interpolation
Key insight: asymmetric sequences break the symmetric lower bound of 2,
and the optimal shape is typically a steeply decaying front-loaded profile.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_fast(a):
    """Fast objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes valid non-negative input."""
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


def make_init(n, variant, seed=42):

    if variant == 0:

        a = np.exp(3.0 * x)
    elif variant == 2:

        a = np.maximum(a, 0)
    elif variant == 3:
        a = np.linspace(0.3, 1.0, n)
    elif variant == 4:

def smoothed_mirror_descent(f0, time_limit, tau_init=15.0, tau_decay=1.004,
                            lr_init=0.015, lr_decay=0.9997,
                            top_frac=0.08, restart_every=120, verbose=False):
    """
    Smoothed Mirror Descent with adaptive temperature.
    Uses top-k peaks with softmax weighting for smooth gradient approximation.
    Multiplicative updates maintain non-negativity naturally.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_obj = objective_fast(f)
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Top-k peak selection
        len_c = len(conv)
        k_top = max(3, int(len_c * top_frac))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]

        # Softmax weighting (temperature-controlled)
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[js:je + 1] += w[i] * (f[k - idx] + f[idx])

        # Multiplicative update
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        # Evaluate with backtracking
        obj_new = objective_fast(f_new)
        if obj_new > M + 1e-10:
            accepted = False
            for damp in [0.5, 0.3, 0.1, 0.05, 0.02]:
                f_trial = (1.0 - damp) * f + damp * f_new
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = objective_fast(f_trial)
                if obj_trial <= M + 1e-12:
                    f_new = f_trial
                    obj_new = obj_trial
                    accepted = True
                    break
            if not accepted:
                lr *= 0.5
                if lr < 1e-9:
                    break
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"        SMD it{it}: obj={best_obj:.8f} tau={tau:.1f} lr={lr:.5f}")
        else:
            no_improve += 1

        # Adaptive parameter update
        tau *= tau_decay
        lr *= lr_decay

        # Periodic restart with perturbation
        if no_improve > restart_every:
            rng = np.random.default_rng(it * 7 + int(tau) % 1000)
            noise = rng.standard_normal(n) * 0.003
            f = np.maximum(best_f + noise, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.4
            tau = tau_init * 1.5
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def peak_pair_search(f0, time_limit, verbose=False):
    """
    Local search that directly attacks the worst convolution peak.
    For peak at position k, finds pairs (j, k-j) with large product and
    tries making them more unequal (which reduces the product for fixed sum).
    Also tries random coordinate swaps as escape mechanism.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_obj = objective_fast(f)
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Consider top-3 peaks
        top3 = np.argsort(conv)[-3:]
        improved = False

        for k_peak in top3:
            k_peak = int(k_peak)
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k_peak - idx]

            # Top-3 contributing pairs
            top_pairs = idx[np.argsort(products)[-3:]]

            for j in top_pairs:
                other = k_peak - j
                if other < 0 or other >= n or other == j:
                    continue

                a_j = f[j]
                a_other = f[other]
                if a_j < 1e-10 or a_other < 1e-10:
                    continue

                # Try making them more unequal (reduces product)
                for frac in [0.15, 0.25, 0.35, 0.15, 0.25, 0.35]:
                    if a_j > a_other:
                        delta = a_other * frac
                    else:
                        delta = a_j * frac

                    f_trial = f.copy()
                    if a_j > a_other:
                        f_trial[j] = a_j + delta
                        f_trial[other] = a_other - delta
                    else:
                        f_trial[j] = a_j - delta
                        f_trial[other] = a_other + delta

                    f_trial = np.maximum(f_trial, 0)
                    s = f_trial.sum()
                    if s < 1e-15:
                        continue
                    f_trial = f_trial / s

                    obj_trial = objective_fast(f_trial)
                    if obj_trial < best_obj - 1e-12:
                        f = f_trial
                        best_obj = obj_trial
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Random swap escape
                j1 = rng.integers(0, n)
                j2 = rng.integers(0, n)
                if j1 != j2:
                    for frac in [0.05, 0.1, 0.2, -0.05, -0.1, -0.2]:
                        delta = f[j1] * frac
                        f_trial = f.copy()
                        f_trial[j1] += delta
                        f_trial[j2] -= delta
                        f_trial = np.maximum(f_trial, 0)
                        s = f_trial.sum()
                        if s < 1e-15:
                            continue
                        f_trial = f_trial / s
                        obj_trial = objective_fast(f_trial)
                        if obj_trial < best_obj - 1e-12:
                            f = f_trial
                            best_obj = obj_trial
                            best_f = f.copy()
                            no_improve = 0
                            improved = True
                            break
                if not improved:
                    no_improve += 1

        if no_improve > 300:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """
    Coordinate descent with golden-ratio-inspired candidate values.
    For each coordinate, tries multiple candidate values and keeps the best.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = objective_fast(f)
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved_this_pass = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-13:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Candidate multipliers
            mults = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.5, 0.3, 2.0]
            for mult in mults:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue

                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss

                val = objective_fast(f_trial)
                if val < best_j_obj - 1e-13:
                    best_j_obj = val
                    best_j_val = new_val

            # Also try zeroing
            if old_val > 1e-6:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    f_trial = f.copy()
                    f_trial[j] = 0.0
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] /= denom
                    f_trial = np.maximum(f_trial, 0)
                    ss = f_trial.sum()
                    if ss > 1e-15:
                        f_trial = f_trial / ss
                        val = objective_fast(f_trial)
                        if val < best_j_obj - 1e-13:
                            best_j_obj = val
                            best_j_val = 0.0

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved_this_pass = True

        if not improved_this_pass:
            break

    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10.0})
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation factor."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def smooth_and_check(f, window):
    """Apply moving average smoothing and return if it improves."""
    if window <= 1:
        return f, objective_fast(f)
    kernel = np.ones(window) / window
    f_smooth = np.convolve(f, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    s = f_smooth.sum()
    if s < 1e-15:
        return f, objective_fast(f)
    f_smooth = f_smooth / s
    obj_smooth = objective_fast(f_smooth)
    obj_orig = objective_fast(f)
    if obj_smooth < obj_orig:
        return f_smooth, obj_smooth
    return f, obj_orig

        print(f"    {label}: {sc:.6f}")

    # === Phase 2: SMD on top candidates ===
    print("\n=== Phase 2: SMD Refinement ===")
    top_k = 3
    for idx in range(min(top_k, len(scored_candidates))):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        _, a_init, label = scored_candidates[idx]
        t_alloc = min(rem * 0.3, 350)
        if t_alloc < 30:
            t_alloc = 30
        print(f"  SMD on {label} (n={len(a_init)}): {t_alloc:.0f}s")
        a_opt, obj = smoothed_mirror_descent(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"smd_{label}")

    # === Phase 3: SMD with focused parameters ===
    print("\n=== Phase 3: Focused SMD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_alloc = min(rem * 0.25, 250)
            if t_alloc < 25:
                t_alloc = 25
            print(f"  SMD (high tau, low lr) on best: {t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                                  tau_init=80.0, tau_decay=1.001,
                                                  lr_init=0.004, lr_decay=0.9999,
                                                  top_frac=0.04, restart_every=80)
            try_seq(a_opt, "smd_focused")

    # === Phase 4: Peak-Pair Search ===
    print("\n=== Phase 4: Peak-Pair Search ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.12, 150)
            if t_alloc < 15:
                t_alloc = 15
            print(f"  PPS: {t_alloc:.0f}s")
            a_opt, obj = peak_pair_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "pps")

    # === Phase 5: LP-Frank-Wolfe ===
    print("\n=== Phase 5: LP-Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 50:
            t_alloc = min(rem * 0.25, 300)
            if t_alloc < 30:
                t_alloc = 30
            print(f"  LP-FW (n={len(best_seq)}): {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        for target_n in [640, 896, 1024, 1200]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            if len(best_seq) == target_n:
                continue
            f_interp = interpolate_seq(best_seq, target_n)
            try_seq(f_interp, f"interp_n{target_n}")
            t_alloc = min(rem * 0.25, 100)
            if t_alloc < 15:
                t_alloc = 15
            print(f"  SMD n={target_n}: {t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent(f_interp, time_limit=t_alloc,
                                                  tau_init=20.0, tau_decay=1.004,
                                                  lr_init=0.012, restart_every=80)
            try_seq(a_opt, f"smd_n{target_n}")

    # === Phase 7: Coordinate Descent ===
    print("\n=== Phase 7: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.15, 120)
            if t_alloc < 10:
                t_alloc = 10
            print(f"  CD: {t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 8: Final Squeeze ===
    print("\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.35, 150)
            if talloc_check := t_alloc >= 15:
                pass
            if t_alloc < 15:
                t_alloc = 15
            print(f"  Final SMD (very focused): {t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                                  tau_init=200.0, tau_decay=1.0005,
                                                  lr_init=0.002, lr_decay=0.99995,
                                                  top_frac=0.02, restart_every=40)
            try_seq(a_opt, "final_smd")

            rem = deadline - time.monotonic()
            if rem > 15:
                t_alloc = min(rem * 0.4, 100)
                if t_alloc < 10:
                    t_alloc = 10
                print(f"  Final PPS: {t_alloc:.0f}s")
                a_opt, obj = peak_pair_search(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "final_pps")

            rem = deadline - time.monotonic()
            if rem > 10:
                t2 = min(rem * 0.5, 60)
                if t2 >= 5:
                    print(f"  Final CD: {t2:.0f}s")
                    a_opt, obj = coordinate_descent(best_seq, time_limit=t2)
                    try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.exp(-3.0 * np.linspace(0, 1, 512))
        best_seq /= best_seq.sum()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
