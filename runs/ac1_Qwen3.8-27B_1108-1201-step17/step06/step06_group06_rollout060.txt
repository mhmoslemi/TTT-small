def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_cross(f, w):
    """
    Compute cross[j] = sum_m f[m] * w[m+j] for j = 0,...,n-1
    where f has length n and w has length 2n-1.
    Uses FFT for efficiency.
    """
    n = len(f)
    a_flipped = f[::-1]
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AF = np.fft.rfft(a_flipped, fft_size)
    conv_result = np.fft.irfft(W * AF, fft_size)
    return conv_result[n - 1:2 * n - 1]


def compute_smooth_grad(f, tau):

    Compute gradient of F(a) = log(sum_k exp((a*a)[k]/tau)).

    w = np.exp(scaled)
    w /= w.sum()
    cross = compute_cross(f, w)
    grad = (2.0 / tau) * cross
    return grad, conv

        it += 1

        # Inner: minimize a^T P_p a via projected gradient
        inner_iters = max(5, min(40, 800 // (it + 1)))
        inner_lr = 2.0 / (n * 1.3)

        for _ in range(inner_iters):
            cross = compute_cross(f, p)
            grad = 2.0 * cross
            f = f - inner_lr * grad
            f = project_simplex(f)

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()

        # Outer: multiplicative weight update
        temp = 0.02 * M + 1e-15
        scaled = conv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        p = (1 - lr_p) * p + lr_p * w

        if it % 80 == 0:
            lr_p = max(0.03, lr_p * 0.95)

        # Occasionally perturb to escape
        if it % 200 == 0:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)

    return best_f, 2 * n * best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix: A_ub[k, j] = f[j] if 0 <= k-j < n, else 0
    # For efficiency, use the tridiagonal-like structure
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, seed=0):
    """
    LP-based Frank-Wolfe with adaptive relaxation.
    """
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp = min(m_exp * 1.2, 8.0)
            if m_exp > 4.0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp = min(m_exp * 1.1, 8.0)
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 60:
                m_exp = min(m_exp * 1.15, 8.0)
            if no_improve > 200:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 8.0)

        if no_improve > 150:
            break

    return best_f, 2 * n * best_obj


def peak_shave(f0, time_limit, seed=0):
    """
    Targeted peak shaving: identifies peak-contributing pairs and
    redistributes mass to reduce the peak.
    """
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get top peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max

            if np.max(products) < 1e-20:
                continue

            old_val = f[j_max]
            if old_val < 1e-12:
                continue

            for fraction in [0.3, 0.5, 0.7, 0.85]:
                diff = old_val * (1 - fraction)
                if diff < 1e-15:
                    continue

                trial = f.copy()
                trial[j_max] -= diff

                # Redistribute away from peak-contributing positions
                # Use inverse of mirror values as weights
                mirrors = pk - np.arange(n)
                valid_mask = (mirrors >= 0) & (mirrors < n) & (np.arange(n) != j_max)
                if valid_mask.sum() < 5:
                    # Fallback: uniform redistribution to non-contributing
                    valid_mask = (np.arange(n) != j_max)
                    if valid_mask.sum() == 0:
                        continue
                    trial[valid_mask] += diff / valid_mask.sum()
                else:
                    mirror_vals = np.zeros(n)
                    mirror_vals[valid_mask] = f[mirrors[valid_mask]]
                    weights = np.zeros(n)
                    weights[valid_mask] = 1.0 / (mirror_vals[valid_mask] + 1e-10)
                    total_w = weights.sum()
                    if total_w < 1e-15:
                        continue
                    trial += diff * weights / total_w

                trial = np.maximum(trial, 0)
                trial /= trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 25:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
        else:
            no_improve = 0

    return best_f, 2 * n * best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Track best
    def update_best(a_norm, label=""):
        nonlocal best_seq, best_score
        a_norm = np.asarray(a_norm, dtype=np.float64)
        a_norm = np.maximum(a_norm, 0.0)
        s = a_norm.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a_norm / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Diverse Initialization ===
    print("=== Phase 1: Diverse Initialization ===")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"  Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # Generate diverse initializations
    init_list = []
    for n_try in [512, 640, 768, 896]:
        init_list.append((f"exp_{n_try}", init_exponential(n_try)))
        init_list.append((f"ramp_{n_try}", init_ramp(n_try, 1.0)))
        init_list.append((f"ramp2_{n_try}", init_ramp(n_try, 2.0)))
        init_list.append((f"gauss_{n_try}", init_gaussian_asym(n_try, 0.3, 0.22)))
        init_list.append((f"gauss_b_{n_try}", init_gaussian_asym(n_try, 0.35, 0.18)))
        init_list.append((f"step_{n_try}", init_step(n_try, 0.25)))
        init_list.append((f"bell_{n_try}", init_bell(n_try, 0.6)))
        init_list.append((f"lininc_{n_try}", init_linear_inc(n_try)))

    # Parent interpolated
    if parent is not None:
        for tn in [512, 640, 896, 1024]:
            init_list.append((f"par_n{tn}", interpolate_sequence(parent, tn)))

    for label, a_init in init_list:
        if time.monotonic() > deadline - 1400:
            break
        try:
            update_best(a_init, label)
        except Exception:
            pass

    if best_seq is None:
        best_seq = init_exponential(768)
        best_score = update_best(best_seq, "fallback")

    print(f"  Best after init: n={len(best_seq)}, score={best_score:.6f}")

    # === Phase 2: Smooth Adam Optimization (main workhorse) ===
    print("\n=== Phase 2: Smooth Adam Optimization ===")

    # Primary optimization on the best
    rem = deadline - time.monotonic()
    if rem > 200:
        t2 = min(rem * 0.4, 500)
        print(f"  Primary Adam: n={len(best_seq)}, time={t2:.0f}s")
        try:
            a_opt, _ = smooth_adam_optimize(best_seq, time_limit=t2, lr=0.004, seed=1)
            update_best(a_opt, "adam_primary")
        except Exception as e:
            print(f"  Adam error: {e}")

    # Second pass with lower temperature
    if time.monotonic() < deadline - 500:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t2b = min(rem, 200)
            conv_b = fast_convolve(best_seq)
            print(f"  Fine Adam: time={t2b:.0f}s")
            try:
                a_opt, _ = smooth_adam_optimize(best_seq, time_limit=t2b,
                                                 tau_init=0.02 * conv_b.max(),
                                                 lr=0.002, seed=2)
                update_best(a_opt, "adam_fine")
            except Exception:
                pass

    # Try from a different initialization
    if time.monotonic() < deadline - 400:
        rem = deadline - time.monotonic() - 350
        if rem > 50:
            t2c = min(rem, 200)
            alt = init_exponential(len(best_seq)) if len(best_seq) in [512, 640, 768, 896] else init_exponential(768)
            print(f"  Alt Adam: n={len(alt)}, time={t2c:.0f}s")
            try:
                a_opt, _ = smooth_adam_optimize(alt, time_limit=t2c, lr=0.004, seed=3)
                update_best(a_opt, "adam_alt")
            except Exception:
                pass

    # === Phase 3: Minimax MW ===
    print("\n=== Phase 3: Minimax MW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 150:
            t3 = min(rem * 0.25, 350)
            print(f"  Minimax: time={t3:.0f}s")
            try:
                a_opt, _ = minimax_mw_optimize(best_seq, time_limit=t3, seed=4)
                update_best(a_opt, "minimax")
            except Exception:
                pass

    # === Phase 4: LP-FW Refinement ===
    print("\n=== Phase 4: LP-FW Refinement ===")
    if best_seq is not None and len(best_seq) <= 896:
        rem = deadline - time.monotonic()
        if rem > 100:
            t4 = min(rem * 0.2, 300)
            print(f"  LP-FW: n={len(best_seq)}, time={t4:.0f}s")
            try:
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t4, seed=5)
                update_best(a_opt, "lpfw")
            except Exception:
                pass

            # Perturbed LP-FW
            if time.monotonic() < deadline - 200:
                rem2 = deadline - time.monotonic() - 150
                if rem2 > 50:
                    t4b = min(rem2, 150)
                    rng4 = np.random.default_rng(123)
                    perturbed = project_simplex(best_seq + rng4.standard_normal(len(best_seq)) * 0.005)
                    try:
                        a_opt, _ = lp_fw_optimize(perturbed, time_limit=t4b, seed=6)
                        update_best(a_opt, "lpfw_pert")
                    except Exception:
                        pass

    # === Phase 5: Peak Shaving ===
    print("\n=== Phase 5: Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 80:
            t5 = min(rem, 250)
            print(f"  Peak Shave: time={t5:.0f}s")
            try:
                a_opt, _ = peak_shave(best_seq, time_limit=t5, seed=7)
                update_best(a_opt, "peak_shave")
            except Exception:
                pass

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        for target_n in [512, 640, 896, 1024, 1280]:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            update_best(f_interp, f"interp_n{target_n}")

            # Quick optimization
            t6 = min(rem * 0.2, 100)
            if t6 > 15:
                try:
                    a_opt, _ = smooth_adam_optimize(f_interp, time_limit=t6, lr=0.003, seed=10 + target_n)
                    update_best(a_opt, f"adam_n{target_n}")
                except Exception:
                    pass

    # === Phase 7: Final fine-tuning ===
    print("\n=== Phase 7: Final Fine-tuning ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t7 = min(rem, 150)
            conv_b = fast_convolve(best_seq)
            print(f"  Final Adam: time={t7:.0f}s")
            try:
                a_opt, _ = smooth_adam_optimize(best_seq, time_limit=t7,
                                                 tau_init=0.008 * conv_b.max(),
                                                 lr=0.001, seed=20)
                update_best(a_opt, "final_adam")
            except Exception:
                pass

        rem = deadline - time.monotonic()
        if rem > 30:
            t7b = min(rem, 80)
            try:
                a_opt, _ = peak_shave(best_seq, time_limit=t7b, seed=21)
                update_best(a_opt, "final_peak_shave")
            except Exception:
                pass

    # === Phase 8: Ultra-fine polish ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t8 = min(rem, 50)
            conv_b = fast_convolve(best_seq)
            try:
                a_opt, _ = smooth_adam_optimize(best_seq, time_limit=t8,
                                                 tau_init=0.003 * conv_b.max(),
                                                 lr=0.0005, seed=30)
                update_best(a_opt, "ultra_polish")
            except Exception:
                pass

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
