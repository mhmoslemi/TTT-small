def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def compute_weighted_grad(a, c, top_frac=0.08, temperature=0.1):
    """Compute weighted subgradient of max(c) using top peaks with softmax."""
    n = len(a)
    len_c = len(c)
    k_top = max(5, int(len_c * top_frac))
    peaks = np.argpartition(c, -k_top)[-k_top:]
    pv = c[peaks]
    temp = temperature * (pv.max() - pv.min() + 1e-15)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = a[k - idx] + a[idx]
        grad[js:je + 1] += (w[i] / k_top) * contrib
    return grad


def peak_surgery_step(a, c, lr_surg=0.02):
    """
    Directly reduce the worst peak by surgically adjusting contributors.
    Returns modified a.
    """
    n = len(a)
    k_star = int(np.argmax(c))
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    
    # Find the most contributing pairs
    contributions = np.zeros(n)
    for i in range(js, je + 1):
        contributions[i] = a[i] * a[k_star - i]
    
    # Get top contributors
    n_top = max(3, n // 20)
    top_idx = np.argsort(contributions)[::-1][:n_top]
    
    # Reduce top contributors
    total_reduction = 0.0
    for i in top_idx:
        i = int(i)
        reduce_amt = lr_surg * a[i] * 0.5
        a[i] -= reduce_amt
        total_reduction += reduce_amt
    
    a = np.maximum(a, 0.0)
    
    # Redistribute mass to positions that contribute least to the top peaks
    # Find "safe" positions (low gradient contribution)
    grad_proxy = np.zeros(n)
    top3 = np.argsort(c)[-3:]
    for k2 in top3:

        js2 = max(0, k2 - n + 1)

    # Add mass to positions with lowest gradient proxy

    per_pos = total_reduction / max(1, len(low_positions))
    for i in low_positions:
        a[int(i)] += per_pos
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    # Build constraint matrix efficiently
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    elif variant == 2:
        # Two bumps asymmetric
        a = np.exp(-30 * (x - 0.15) ** 2) + 0.6 * np.exp(-30 * (x - 0.7) ** 2)
    elif variant == 3:
        # Power law
        a = (x + 0.01) ** (-0.7)
    elif variant == 4:
        # Inverted power
        a = (1 - x + 0.01) ** (-0.7)
    elif variant == 5:
        # Sine weighted
        a = np.sin(np.pi * x) ** 2 * np.exp(-x * 2)
        a = np.maximum(a, 1e-6)
    elif variant == 6:
        # Random with smoothing
        a = rng.standard_normal(n)
        a = np.abs(a)
        # Smooth
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
    elif variant == 7:
        # Cosine ramp
        a = 0.3 + 0.7 * np.cos(np.pi * x * 1.5) ** 2
    elif variant == 8:
        # Log-sine
        a = np.exp(-np.arange(n) / (n * 0.25)) * (1 + 0.4 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 1e-6)
    elif variant == 9:
        # Triangular asymmetric
        a = np.minimum(x / 0.3, (1 - x) / 0.7)
        a = np.maximum(a, 0.01)
    elif variant == 10:
        # Gaussian mixture
        a = np.exp(-20*(x-0.2)**2) + 0.4*np.exp(-20*(x-0.6)**2) + 0.1*np.exp(-20*(x-0.9)**2)
    elif variant == 11:
        # Heavy-tail front
        a = 1.0 / (np.arange(n) + 1.0) ** 0.8
    elif variant == 12:
        # Step-like
        a = np.ones(n)
        a[:n//4] = 2.0
        a[n//4:n//2] = 1.0
        a[n//2:3*n//4] = 0.5
        a[3*n//4:] = 0.2
        a += rng.standard_normal(n) * 0.05
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse inits -> adaptive subgradient ->
    peak surgery -> LP-FW refinement -> multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse initializations and quick screening ===
    print("=== Phase 1: Diverse Initializations ===")
    test_lengths = [640, 768, 896, 1024]
    init_candidates = []

    for n_try in test_lengths:
        for v in range(13):
            a_init = make_diverse_init(n_try, variant=v, seed=seed)
            init_candidates.append((a_init, f"init_v{v}_n{n_try}"))

    if parent is not None:
        init_candidates.append((parent, "parent"))

    # Score all initializations
    scored = []
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        scored.append((obj, a_norm, label))

    scored.sort(key=lambda x: x[0])
    print(f"  Top 5 inits: {[f'{l}:{o:.4f}' for o, _, l in scored[:5]]}")

    # Evaluate top inits
    for obj, a_norm, label in scored[:8]:
        try_seq(a_norm, label)

    # === Phase 2: Adaptive subgradient on top candidates ===
    print("\n=== Phase 2: Adaptive Subgradient ===")
    top_to_optimize = scored[:5]
    # Always include parent if available
    if parent is not None:
        top_to_optimize.append((2.0, parent, "parent"))

    for idx, (obj_init, a_init, label) in enumerate(top_to_optimize):
        rem = deadline - time.monotonic() - 600
        if rem < 30:
            break
        t_alloc = min(rem / max(1, len(top_to_optimize) - idx), 180)
        if t_alloc < 20:
            t_alloc = 20
        print(f"  SMD+ Surgery on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
        a_opt, obj = adaptive_subgradient(
            a_init, time_limit=t_alloc, 
            top_frac=0.06, temperature=0.08,
            lr_init=0.004, momentum_beta=0.88,
            restart_every=120, verbose=True
        )
        try_seq(a_opt, f"surg_{label}")

    # === Phase 3: LP-Frank-Wolfe on best ===
    print("\n=== Phase 3: LP-Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_alloc = min(rem, 200)
            n_best = len(best_seq)
            print(f"  LP-FW on best (n={n_best}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_best")

    # === Phase 4: Second round of subgradient with different params ===
    print("\n=== Phase 4: Fine-tuning with different params ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 120)
            # More focused: fewer peaks, lower temperature
            print(f"  Fine SMD on best: t={t_alloc:.0f}s")
            a_opt, obj = adaptive_subgradient(
                best_seq, time_limit=t_alloc,
                top_frac=0.03, temperature=0.04,
                lr_init=0.002, momentum_beta=0.92,
                restart_every=200, verbose=True
            )
            try_seq(a_opt, "fine_smd")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [896, 1024, 1152, 1280]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            if len(base) == target_n:
                continue
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.005:
                t_alloc = min(rem * 0.4, 80)
                if t_alloc < 20:
                    t_alloc = 20
                print(f"  Subgradient n={target_n}: t={t_alloc:.0f}s")
                a_opt, obj = adaptive_subgradient(
                    f_interp, time_limit=t_alloc,
                    top_frac=0.06, temperature=0.08,
                    lr_init=0.004, restart_every=100
                )
                try_seq(a_opt, f"surg_n{target_n}")

    # === Phase 6: Random restarts ===
    print("\n=== Phase 6: Random Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(5):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem / max(1, 5 - r), 60)
            if t_alloc < 15:
                t_alloc = 15

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = np.maximum(best_seq + noise, 1e-10)
                f_init = f_init / f_init.sum()
            elif r == 1:
                asym = make_diverse_init(n_best, variant=int(rng.integers(0, 13)), seed=seed + 100 + r)
                f_init = 0.75 * best_seq + 0.25 * asym
                f_init = f_init / f_init.sum()
            elif r == 2:
                # Shift and blend
                shifted = np.roll(best_seq, int(n_best * 0.1))
                f_init = 0.6 * best_seq + 0.4 * shifted
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            elif r == 3:
                # Local perturbation on top contributors
                c = fast_convolve(best_seq)
                k_peak = int(np.argmax(c))
                f_init = best_seq.copy()
                js = max(0, k_peak - n_best + 1)
                je = min(n_best - 1, k_peak)
                for i in range(js, je + 1):
                    f_init[i] *= (1 - 0.05 * rng.random())
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            else:
                noise = rng.standard_normal(n_best) * 0.008
                f_init = np.maximum(best_seq + noise, 1e-10)
                f_init = f_init / f_init.sum()

            print(f"  Restart {r}: t={t_alloc:.0f}s")
            a_opt, obj = adaptive_subgradient(
                f_init, time_limit=t_alloc,
                top_frac=0.06, temperature=0.06,
                lr_init=0.003, momentum_beta=0.88,
                restart_every=100
            )
            try_seq(a_opt, f"restart_{r}")

    # === Phase 7: Final LP-FW squeeze ===
    print("\n=== Phase 7: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            t_alloc = min(rem, 60)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "final_lpfw")

    # === Phase 8: Ultra-fine subgradient ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem, 30)
            print(f"  Ultra-fine SMD: t={t_alloc:.0f}s")
            a_opt, obj = adaptive_subgradient(
                best_seq, time_limit=t_alloc,
                top_frac=0.02, temperature=0.02,
                lr_init=0.001, momentum_beta=0.95,
                restart_every=300
            )
            try_seq(a_opt, "ultra_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
