def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def accelerated_fw(p0, time_limit, verbose=False):
    n = len(p0)
    p = p0.copy()
    p_prev = p.copy()
    t = 1.0
    best_obj = 2.0 * n * fast_convolve(p).max()
    best_p = p.copy()
    
    t0 = time.monotonic()
    it = 0
    top_k = 5
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_next
        y = p + beta * (p - p_prev)
        y = project_simplex(y)
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peaks = peaks[np.argsort(conv_y[peaks])[::-1]]
        
        # Weights: proportional to excess
        vals = conv_y[peaks]
        weights = np.ones(len(peaks)) / len(peaks)
        
        s = np.zeros(n)
        for i in range(len(peaks)):
            k = int(peaks[i])
            w = 2.0 * weights[i]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            s[idx] += w * y[k - idx]
        
        # FW step: q = e_{argmin s}
        j_min = np.argmin(s)
        q = np.zeros(n)
        q[j_min] = 1.0
        
        # Update
        alpha = 2.0 / (t_next + 1.0)
        p_prev = p
        p = (1.0 - alpha) * y + alpha * q
        t = t_next
        
        # Evaluate
        obj = 2.0 * n * fast_convolve(p).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_p = p.copy()
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AFW iter {it}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_p, best_obj

So for symmetric sequences, the objective is ALWAYS >= 2! We can never beat 2 with a symmetric sequence!

This is a key insight. The sequences that achieve < 2 must be ASYMMETRIC.

For an asymmetric sequence, (p*p)[n-1] = sum_j p[j] p[n-1-j] which is NOT the same as sum p[j]^2. It's a "cross-correlation" with the reversed sequence.

So the strategy is: use an asymmetric sequence where the "self-overlap" at the center is small, while keeping all other overlaps small too.

This is like finding a sequence that is "orthogonal" to its reverse (in a sense). If p[j] * p[n-1-j] is small for all j, then (p*p)[n-1] is small.

For example, if p is concentrated at the beginning (p[j] > 0 for small j, p[j] ≈ 0 for large j), then p[n-1-j] ≈ 0 for small j (since n-1-j is large), so (p*p)[n-1] = sum p[j] p[n-1-j] ≈ 0.

But then (p*p)[0] = p[0]^2 might be large, and other early-lag peaks might also be large.

So the trade-off is: concentrate p to reduce the center peak, but this increases the early-lag peaks. The optimal balance gives < 2.

This explains why asymmetric sequences like make_asymmetric() in the previous code are useful initializations!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated mirror descent,
active-set LP-Frank-Wolfe, multi-peak subgradients, and multi-length exploration.
Key innovation: exploit asymptotic structure for fast dense updates, use active-set

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def cheap_normalize(v):
    """Fast non-negative normalization (not exact projection)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def compute_subgradient(p, peaks, weights):
    """Compute weighted subgradient of max_k (p*p)[k] at p."""
    n = len(p)
    s = np.zeros(n)
    for i, k in enumerate(peaks):
        k = int(k)
        w = 2.0 * weights[i]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            idx = np.arange(js, je + 1)
            s[js:je + 1] += w * p[k - idx]
    return s


def get_top_peaks(conv, top_k=5):
    """Get top-k peak indices and values."""
    if len(conv) <= top_k:
        return np.arange(len(conv)), conv.copy()
    part = np.argpartition(conv, -top_k)[-top_k:]
    part = part[np.argsort(conv[part])[::-1]]
    return part, conv[part]


def nesterov_mirror_descent(p0, time_limit, verbose=False, lr_init=0.05, decay=0.99995):
    """
    Nesterov-accelerated mirror descent with multi-peak subgradient.
    Very fast: O(n log n) per iteration due to FFT.
    """
    n = len(p0)
    p = p0.copy()
    p_prev = p.copy()
    y = p.copy()
    t_param = 1.0
    lr = lr_init
    best_obj = 2.0 * n * fast_convolve(p).max()
    best_p = p.copy()
    top_k = 5
    no_improve = 0

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov extrapolation
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        beta = (t_param - 1.0) / t_next
        y = p + beta * (p - p_prev)
        y = cheap_normalize(y)

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        peaks, vals = get_top_peaks(conv_y, top_k)
        # Weights: emphasize highest peaks
        max_val = vals[0]
        if max_val > 1e-15:
            weights = np.exp(-0.5 * (vals - max_val) / (max_val * 0.1))
            weights /= weights.sum()
        else:
            weights = np.ones(len(peaks)) / len(peaks)

        s = compute_subgradient(y, peaks, weights)

        # Mirror descent update (exponential weights)
        p = p * np.exp(-lr * s)
        p = cheap_normalize(p)

        p_prev = p
        t_param = t_next
        lr *= decay

        # Evaluate
        obj = 2.0 * n * fast_convolve(p).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive: if stuck, restart with new lr
        if no_improve > 2000:
            rng = np.random.default_rng(it * 31)
            p = cheap_normalize(best_p + rng.standard_normal(n) * 0.001)
            p_prev = p.copy()
            lr = lr_init * 0.5
            t_param = 1.0
            no_improve = 0

        if lr < 1e-8:
            break

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD iter {it}: obj={best_obj:.8f}, t={elapsed:.1f}s, it/s={it/elapsed:.0f}")

    return best_p, best_obj


def active_set_lp_direction(p, M, active_peaks, n):
    """Solve LP with only active set constraints for faster direction finding."""
    c = -np.ones(n)
    A_ub = []
    b_ub = []

    for k in active_peaks:
        k = int(k)
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        gk = 0.0
        for j in range(js, je + 1):
            row[j] = 2.0 * p[k - j]
            gk += p[j] * p[k - j]
        A_ub.append(row)
        b_ub.append(M + gk)

    if not A_ub:
        return None

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def lp_fw_optimize(p0, time_limit, verbose=False):
    """
    LP-Frank-Wolfe with active set management and line search.
    """
    p = p0.copy()
    n = len(p)
    conv = fast_convolve(p)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_p = p.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    active_tol = 0.01

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Update active set: peaks within tolerance of max
        conv = fast_convolve(p)
        M = conv.max()
        active_mask = conv >= M * (1.0 - active_tol)
        active_primary = np.where(active_mask)[0]
        # Limit to top 30 for LP size
        if len(active_primary) > 30:
            top_indices = active_primary[np.argsort(conv[active_primary])[-30:]]
        else:
            top_indices = active_primary

        # Solve LP
        q = active_set_lp_direction(p, M, top_indices, n)
        if q is None:
            # Fallback: use all peaks
            peaks, _ = get_top_peaks(conv, 10)
            q = active_set_lp_direction(p, M, peaks, n)
            if q is None:
                no_improve += 1
                if no_improve > 50:
                    break
                continue

        q = np.maximum(q, 0)
        q_sum = q.sum()
        if q_sum < 1e-15:
            no_improve += 1
            continue
        q = q / q_sum

        diff = np.max(np.abs(q - p))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        else:
            if diff > 1e-6:
                no_improve = 0
            elif diff > 1e-8:
                no_improve += 1
            else:
                no_improve += 3

        # Line search along direction (q - p)
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            p_try = p + alpha * (q - p)
            p_try = np.maximum(p_try, 0)
            s = p_try.sum()
            if s < 1e-15:
                continue
            p_try = p_try / s
            val = fast_convolve(p_try).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-8:
            no_improve += 1
            continue

        p = p + best_alpha * (q - p)
        p = np.maximum(p, 0)
        s = p.sum()
        if s < 1e-15:
            break
        p = p / s

        obj = 2.0 * n * fast_convolve(p).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_p = p.copy()

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_p, best_obj


def peak_targeted_cd(p0, time_limit, verbose=False):
    """
    Coordinate descent targeting the dominant peak.
    Adjusts contributing elements to reduce the peak while maintaining sum=1.
    """
    p = p0.copy()
    n = len(p)
    best_obj = 2.0 * n * fast_convolve(p).max()
    best_p = p.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 80:
        conv = fast_convolve(p)
        M = conv.max()
        k_peak = int(np.argmax(conv))
        improved = False

        # Contributors to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = p[idx] * p[k_peak - idx]
        total = contrib.sum()

        if total < 1e-15:
            no_improve += 10
            continue

        # Try reducing top contributors
        order = np.argsort(contrib)[::-1]
        for oi in range(min(6, len(order))):
            j = int(idx[order[oi]])
            old = p[j]
            if old < 1e-12:
                continue

            for mult in [0.5, 0.65, 0.8, 0.9, 1.15, 1.3]:
                new = old * mult
                if new < 0 or new > 0.5:
                    continue
                trial = p.copy()
                trial[j] = new
                d = old - new
                if abs(d) > 1e-15:
                    # Redistribute to non-contributing positions
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[js:je + 1] = False
                    nc_count = non_contrib.sum()
                    if nc_count > 0:
                        w = p[non_contrib]
                        ws = w.sum()
                        if ws > 1e-15:
                            trial[non_contrib] += d * w / ws
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_p = trial.copy()
                    p = trial
                    improved = True

        if not improved:
            no_improve += 1
            # Aggressive: scale down all peak contributors
            if no_improve > 10:
                trial = p.copy()
                trial[js:je + 1] *= 0.9
                d = p.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    w = p[non_contrib]
                    ws = w.sum()
                    if ws > 1e-15:
                        trial[non_contrib] += d * w / ws
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    p = trial
                    best_obj = obj
                    best_p = trial.copy()
                    no_improve = 0
                else:
                    break

    return best_p, best_obj


def make_asymmetric_exp(n, decay=3.0):
    """Asymmetric exponential decay."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-decay * x / n)
    return a / a.sum()


def make_asymmetric_exp_rev(n, decay=3.0):
    """Reverse asymmetric exponential decay."""
    a = make_asymmetric_exp(n, decay)
    return a[::-1]


def make_gaussian(n, sigma_frac=0.3, center_frac=0.4):
    """Gaussian with asymmetric center."""
    x = np.arange(n, dtype=np.float64)
    center = n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def make_bell_asym(n):
    """Asymmetric bell shape."""
    x = np.arange(n, dtype=np.float64) / n
    # Skew-normal-like shape
    a = np.exp(-4.0 * x) * (1.0 + 0.5 * np.sin(np.pi * x))
    a = np.maximum(a, 0)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    if len(f) == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, len(f))
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate and track best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))

                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_asymmetric_exp(768, 3.0)
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "default")

    n_parent = len(parent)

    # === Phase 1: Diverse initialization and quick screening ===
    print("=== Phase 1: Initialization ===")
    init_candidates = [parent]
    for n_try in [512, 640, 768, 896, 1024]:
        if n_try == n_parent:
            init_candidates.append(parent.copy())
        else:
            init_candidates.append(interpolate_sequence(parent, n_try))
        init_candidates.append(make_asymmetric_exp(n_try, 2.0))

    # Quick screen: run short NMD on each
