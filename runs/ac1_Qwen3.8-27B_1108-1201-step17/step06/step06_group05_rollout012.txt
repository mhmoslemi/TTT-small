- The subgradient of the max function at these active peaks must be in the normal cone of the constraint set
- The subgradient of (a*a)[k] with respect to a[j] is 2*a[k-j] (for valid indices)
- So the KKT condition is: sum_{k in S} λ_k * 2 * a[k-j] = μ (constant) for all j with a[j] > 0, where λ_k >= 0, sum λ_k = 1

This is a useful characterization! It means we're looking for a sequence a and a set of weights on its autocorrelation peaks such that a specific linear combination of shifted a's is constant.

This suggests an iterative algorithm:
1. Start with a candidate a
2. Find the active peaks (where conv is maximal or near-maximal)
3. Solve for the weights λ that make the weighted sum constant
4. Update a to better satisfy the KKT conditions

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def make_gaussian(n, sigma_frac=0.25):

    a = np.exp(-0.5 * (x / sigma) ** 2)

def compute_lse_gradient(f, c, beta):
    """Gradient of LSE_beta(c) w.r.t. f."""
    n = len(f)
    scaled = beta * c
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # g_j = 2 * sum_i f[i] * w[j+i] (convolution structure)
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

def coordinate_polish(f0, time_limit):
    """Coordinate descent for final refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.8, 0.9, 1.1, 1.25]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                f[j] = best_j_val
                f = np.maximum(f, 0)
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj

In matrix form: A[1,j] * g[j] summed over j should give a*g[1] + b*g[0].
So A[1,0] = b (coefficient of g[0]) and A[1,1] = a (coefficient of g[1]).

So A[k,j] = f[k-j] (not f[j]!).

Let me re-verify: A[k,j] = f[k-j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k.

For k=1, j=0: A[1,0] = f[1-0] = f[1] = b. ✓
For k=1, j=1: A[1,1] = f[1-1] = f[0] = a. ✓

So the correct formula is: A[k,j] = f[k-j] for max(0,k-n+1) <= j <= min(n-1,k).

In code:
```python
For row k, columns js to je:
A[i, js:je+1] = f[k-je : k-js+1]
```

Because:
- For column j=js: A[i, js] = f[k-js]
- For column j=js+1: A[i, js+1] = f[k-js-1]
- ...
- For column j=je: A[i, je] = f[k-je]

So the values are f[k-js], f[k-js-1], ..., f[k-je], which is f[k-je : k-js+1][::-1] or equivalently f[k-je:k-js+1] read in reverse.

Wait: f[k-je : k-js+1] gives [f[k-je], f[k-je+1], ..., f[k-js]]
But we want [f[k-js], f[k-js-1], ..., f[k-je]]
Which is f[k-je : k-js+1][::-1]

Let me verify: k=1, js=0, je=1.
f[k-je : k-js+1] = f[0:2] = [f[0], f[1]] = [a, b]
f[0:2][::-1] = [b, a]
A[1, 0:2] = [b, a]
A[1,0]*g[0] + A[1,1]*g[1] = b*g[0] + a*g[1] ✓

Great! So the code is:
```python
def build_constraint_matrix(f, active_indices):
    n = len(f)
    K = len(active_indices)
    A = np.zeros((K, n))
    for i, k in enumerate(active_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[i, js:je+1] = f[k-je:k-js+1][::-1]
    return A
```

This is O(K*n) in the worst case but in practice much less since each row has at most n non-zeros.

Now, can I build this matrix even faster? For the full LP (all 2n-1 rows), the matrix is a (2n-1) × n Toeplitz-like matrix. I can build it using a loop over columns:

```python
def build_full_constraint_matrix(f):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A
```

Wait, let me check. Column j has non-zeros in rows k where 0 <= k-j < n, i.e., j <= k <= j+n-1. And A[k,j] = f[k-j].

So column j is: [0,...,0, f[0], f[1], ..., f[n-1], 0,...,0] with f[0] at row j.

In code: A[j:j+n, j] = f. Yes! That's correct and vectorized.

For a subset of rows (active set), I can build the full matrix and then select the needed rows. But for large n, building the full (2n-1)×n matrix is O(n^2) which might be too slow.

Actually for n=768, the full matrix is 1535×768 = 1.18M elements, which is fine. For n=2048, it's 4095×2048 = 8.4M elements, still manageable.

So the fastest approach is:
1. Build the full constraint matrix once: A_full[j:j+n, j] = f for j=0..n-1
2. For each LP solve, select the active rows: A = A_full[active_indices, :]
3. Solve the LP

This is O(n^2) to build + O(K*n) to select, which is very fast.

Let me now write the complete, optimized code:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Analytical initialization (Gaussian, raised-cosine, flat-top shapes)
2. Nesterov-accelerated log-sum-exp smoothed gradient descent
3. LP-based Frank-Wolfe with active-set constraint selection for high precision
4. Multi-length exploration (768 to 3072)
5. Coordinate descent final polish

Key improvements over previous version:
- Vectorized constraint matrix construction for faster LP solves
- Active-set LP (top-K peaks only) for dramatically faster iterations
- Nesterov acceleration for faster gradient descent convergence
- Better time allocation with more emphasis on LP-FW
- Multi-length search with interpolation
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian-shaped initial sequence."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    """Raised cosine (Hann window) shape."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a[0] = max(a[0], 0)
    a[-1] = max(a[-1], 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, edge_frac=0.3):
    """Flat-top with smooth cosine edges."""
    a = np.ones(n)
    edge = max(2, int(n * edge_frac))
    for i in range(edge):
        t = i / edge
        val = 0.5 * (1 - np.cos(np.pi * t))
        a[i] = val
        a[n - 1 - i] = val
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def build_full_constraint_matrix(f):
    """Build the full (2n-1) x n constraint matrix for conv(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def compute_lse_gradient(f, c, beta):
    """Gradient of log-sum-exp smoothed max of convolution."""
    n = len(f)
    scaled = beta * c
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient: g_j = 2 * sum_i f[i] * w[j+i]
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    return 2.0 * conv[indices]


def nesterov_gd(f0, time_limit, verbose=False):
    """Nesterov-accelerated gradient descent with progressive LSE sharpening."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    beta_schedule = [4, 8, 16, 32, 64, 128, 256]
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit * 0.92:
            break
        
        y = f.copy()  # Nesterov lookahead
        t_nest = 1.0
        it = 0
        no_improve = 0
        
        stage_start = time.monotonic()
        remaining_stages = len(beta_schedule) - beta_schedule.index(beta)
        stage_time = max(5.0, (time_limit * 0.92 - (time.monotonic() - t0)) / remaining_stages)
        
        while time.monotonic() - stage_start < stage_time:
            it += 1
            t_nest = (1.0 + np.sqrt(1.0 + 4.0 * t_nest ** 2)) / 2.0
            momentum = (t_nest - 1.0) / t_nest
            
            y = f + momentum * (y - f)
            c_y = fast_convolve(y)
            g = compute_lse_gradient(y, c_y, beta)
            
            lr = 1.0 / (beta * 2.0 * (np.max(f) + 1e-10) * np.sqrt(n))
            f_new = project_simplex(y - lr * g)
            
            c_new = fast_convolve(f_new)
            obj_new = c_new.max()
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            f = f_new
            
            if no_improve > 40:
                rng = np.random.default_rng(it * 11 + int(beta) * 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                y = f.copy()
                no_improve = 0
            
            if verbose and it % 300 == 0:
                print(f"    NGD beta={beta} it={it}: obj={obj_new:.10f} best={best_obj:.10f}")
    
    return best_f, best_obj
