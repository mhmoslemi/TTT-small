def adam_soft_max_optimize(p0, time_limit, verbose=False):
    """Adam optimizer on soft-max approximation of max(conv)."""
    p = p0.astype(np.float64).copy()
    p = project_simplex(p)
    n = len(p)
    
    best_obj = fast_convolve(p).max()
    best_p = p.copy()
    
    t0 = time.monotonic()
    total_time = time_limit
    
    # Hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Start with moderate temperature, increase over time
    gamma_start = 2.0
    gamma_end = 200.0
    
    lr_start = 5e-4
    lr_end = 1e-6
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / total_time
        
        # Adaptive hyperparameters
        gamma = gamma_start * (gamma_end / gamma_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Handle stagnation: perturb if no improvement for a while
        if no_improve > 50:
            if no_improve > 200:
                # Larger perturbation
                rng = np.random.default_rng(t * 7919)
                p = project_simplex(best_p + rng.standard_normal(n) * 0.01)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0
                lr = lr_start * 0.1
            else:
                # Small perturbation
                rng = np.random.default_rng(t * 104729)
                p = project_simplex(p + rng.standard_normal(n) * 0.001)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                continue
        
        # Compute soft-max weights
        conv = fast_convolve(p)
        shifted = conv / gamma
        shifted -= shifted.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        grad = compute_soft_max_grad(w, p)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        p_new = p - lr * m_hat / (np.sqrt(v_hat) + eps)
        p_new = project_simplex(p_new)
        
        # Evaluate
        obj = fast_convolve(p_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_p = p_new.copy()
            no_improve = 0
            if verbose and t % 500 == 0:
                print(f"      Adam iter {t}: obj={best_obj:.10f}, gamma={gamma:.1f}")
        else:
            no_improve += 1
        
        p = p_new
    
    return best_p, best_obj

def spectral_flattening(p0, time_limit, alpha=0.1, verbose=False):
    """Iterative spectral flattening: modify |P|^2 to be flatter."""
    p = p0.astype(np.float64).copy()
    p = project_simplex(p)
    n = len(p)
    
    N = 1 << (2 * n - 1).bit_length()
    
    best_obj = fast_convolve(p).max()
    best_p = p.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # FFT
        P = np.fft.rfft(p, N)
        amp = np.abs(P)
        phase = np.angle(P)
        
        # Flatten the amplitude
        amp_mean = amp.mean()
        amp_new = amp - alpha * (amp - amp_mean)
        amp_new = np.maximum(amp_new, 0)
        
        # Reconstruct
        P_new = amp_new * np.exp(1j * phase)
        p_new = np.fft.irfft(P_new, N)[:n]
        p_new = project_simplex(p_new)
        
        obj = fast_convolve(p_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_p = p_new.copy()
            no_improve = 0
            if verbose:
                print(f"      Spectral iter: obj={best_obj:.10f}")
        else:
            no_improve += 1
            # Decrease alpha
            alpha = max(alpha * 0.99, 0.001)
        
        p = p_new
        
        if no_improve > 100:
            break
    
    return best_p, best_obj

def peak_targeted_descent(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 80:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        
        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        if len(contrib_idx) == 0:
            break
        
        products = f[contrib_idx] * f[peak_k - contrib_idx]
        if products.sum() < 1e-15:
            break
        
        max_prod_idx = contrib_idx[np.argmax(products)]
        partner_idx = peak_k - max_prod_idx
        
        if f[max_prod_idx] >= f[partner_idx]:
            source = max_prod_idx
        else:
            source = partner_idx
        
        # Find non-contributing coordinates
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[contrib_idx] = False
        if non_contrib.sum() < 2:
            non_contrib[:] = True
            non_contrib[source] = False
        
        best_trial = None
        best_trial_obj = M
        
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            diff = f[source] * frac
            if diff < 1e-15:
                continue
            trial = f.copy()
            trial[source] -= diff
            nc_vals = trial[non_contrib]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                continue
            trial[non_contrib] += diff * nc_vals / nc_sum
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()

                best_trial = trial.copy()
        
        if best_trial is not None:
            f = best_trial
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_f = f.copy()
                no_improve = 0
                if verbose:
                    print(f"      PTD: peak_k={peak_k}, obj={best_obj:.10f}")
        else:
            no_improve += 1
            # Try next top peak
            top_peaks = np.argsort(conv)[-10:]
            found = False
            for pk in top_peaks:
                pk = int(pk)
                if pk == peak_k:
                    continue
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                cidx2 = np.arange(js2, je2 + 1)
                if len(cidx2) == 0:
                    continue
                prods2 = f[cidx2] * f[pk - cidx2]
                if prods2.sum() < 1e-15:
                    continue
                src2 = cidx2[np.argmax(prods2)]
                for frac2 in [0.1, 0.2, 0.3]:
                    diff2 = f[src2] * frac2
                    trial = f.copy()
                    trial[src2] -= diff2
                    nc2 = np.ones(n, dtype=bool)
                    nc2[cidx2] = False
                    nc2_vals = trial[nc2]
                    nc2_sum = nc2_vals.sum()
                    if nc2_sum < 1e-15:
                        continue
                    trial[nc2] += diff2 * nc2_vals / nc2_sum
                    trial = np.maximum(trial, 0.0)
                    trial = trial / trial.sum()
                    obj2 = fast_convolve(trial).max()
                    if obj2 < best_obj - 1e-14:
                        f = trial
                        best_obj = obj2
                        best_f = f.copy()
                        no_improve = 0
                        found = True
                        break
                if found:
                    break
            if not found:
                no_improve += 1
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    n = len(p)

    t_a = time_limit * 0.6
    t_b = time_limit * 0.4
    
    if verbose:
        print(f"    MixedTF: Phase A (Adam) {t_a:.0f}s, Phase B (Spectral) {t_b:.0f}s")
    
    p, obj_a = adam_soft_max_optimize(p, t_a, verbose=verbose)
    if obj_a < best_obj:
        best_obj = obj_a
        best_p = p.copy()
    
    # Phase B: Spectral flattening
    p2, obj_b = spectral_flattening(p, t_b, alpha=0.05, verbose=verbose)
    if obj_b < best_obj:
        best_obj = obj_b
        best_p = p2.copy()
    
    # Phase C: Short Adam on the best
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 5:
        p3, obj_c = adam_soft_max_optimize(best_p, remaining * 0.7, verbose=verbose)
        if obj_c < best_obj:
            best_obj = obj_c
            best_p = p3.copy()
    
    return best_p, best_obj
