import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def symmetrize(a):
    """Make sequence palindromic and normalize to sum=1."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_symmetric_simplex(v):
    """Project onto symmetric simplex: x>=0, sum=1, x[i]=x[n-1-i]."""
    # Symmetrize first
    v_sym = 0.5 * (v + v[::-1])
    # Project onto simplex
    v_proj = project_simplex(v_sym)
    # Symmetrize again and renormalize
    v_final = 0.5 * (v_proj + v_proj[::-1])
    v_final = np.maximum(v_final, 0.0)
    s = v_final.sum()
    if s > 1e-15:
        v_final /= s
    return v_final

    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w


def compute_gradient_fft(a, w):

    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * cross[:n]

def subspace_optimize(a0, time_limit, M_modes=30, verbose=False, label=""):
    """
    Optimize in a cosine subspace. Represent a as linear combination of first M cosine
    modes, optimize coefficients using L-BFGS-B. This can escape local minima.
    """
    n = len(a0)
    a0 = symmetrize(a0)

    # Build symmetric cosine basis
    k = np.arange(n)
    basis = np.zeros((M_modes, n))
    basis[0] = 1.0 / np.sqrt(n)
    for m in range(1, M_modes):
        basis[m] = np.sqrt(2.0 / n) * np.cos(np.pi * m * (k + 0.5) / n)

    # Project a0 onto subspace
    coeffs = basis @ a0

    def objective(c):
        a_candidate = basis @ c
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = symmetrize(a_candidate)
        conv = fast_convolve(a_candidate)
        return conv.max()

    def gradient(c):
        a_candidate = basis @ c
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = symmetrize(a_candidate)
        # Numerical gradient (central differences)
        eps = 1e-7
        grad = np.zeros(M_modes)
        base = objective(c)
        for i in range(M_modes):
            c_plus = c.copy()
            c_plus[i] += eps
            c_minus = c.copy()
            c_minus[i] -= eps
            grad[i] = (objective(c_plus) - objective(c_minus)) / (2.0 * eps)
        return grad

    t0 = time.monotonic()
    best_coeff = coeffs.copy()
    best_val = objective(coeffs)

    # Run L-BFGS-B with time limit via callback
    stop_flag = [False]

    def callback(_ck):
        if time.monotonic() - t0 > time_limit:
            stop_flag[0] = True
            return True
        return False

    result = minimize(objective, coeffs, jac=gradient, method='L-BFGS-B',
                      options={'maxiter': 20000, 'maxfun': 100000, 'maxls': 50},
                      callback=callback)

    if result.fun < best_val:
        best_coeff = result.x
        best_val = result.fun

    a_result = symmetrize(np.maximum(basis @ best_coeff, 0.0))
    if verbose:
        print(f"    {label} Subspace done: obj={best_val:.10f} (M={M_modes})")
    return a_result, best_val

    basis = np.zeros((M_modes, n))
    basis[0] = 1.0 / np.sqrt(n)
    for m in range(1, M_modes):
        basis[m] = np.sqrt(2.0 / n) * np.cos(np.pi * m * (k + 0.5) / n)

    coeffs = basis @ a0

    def objective_smooth(c):
        a_candidate = basis @ c
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = symmetrize(a_candidate)
        conv = fast_convolve(a_candidate)
        scaled = conv / T
        scaled -= scaled.max()
        return T * (np.log(np.sum(np.exp(scaled))) + scaled.max())

    def gradient_smooth(c):
        a_candidate = basis @ c
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = symmetrize(a_candidate)
        conv = fast_convolve(a_candidate)
        w = softmax_weights(conv, T)
        g_a = compute_gradient_fft(a_candidate, w)
        # Gradient with respect to coefficients
        return basis @ g_a

    t0 = time.monotonic()
    best_coeff = coeffs.copy()
    best_val = objective_smooth(coeffs)

    stop_flag = [False]

    def callback(_ck):
        if time.monotonic() - t0 > time_limit:
            stop_flag[0] = True
            return True
        return False

    result = minimize(objective_smooth, coeffs, jac=gradient_smooth, method='L-BFGS-B',
                      options={'maxiter': 30000, 'maxfun': 100000, 'maxls': 50,
                               'gtol': 1e-10},
                      callback=callback)

    if result.fun < best_val:
        best_coeff = result.x
        best_val = result.fun

    a_result = symmetrize(np.maximum(basis @ best_coeff, 0.0))
    final_obj = fast_convolve(a_result).max()
    if verbose:
        print(f"    {label} SubspaceSmoothed done: obj={final_obj:.10f} (M={M_modes})")
    return a_result, final_obj


def peak_shave(a0, time_limit, verbose=False, label=""):
    """
    Targeted peak reduction: for each significant peak, identify the most
    contributing index pairs and try redistributing mass.
    """
    a = symmetrize(a0.astype(np.float64))
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        improved = False

        # Get top peaks (exclude edges which are naturally small)
        K = min(12, len(conv))
        peak_indices = np.argsort(conv)[-K:]
        # Sort by value descending
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k in peak_indices:
            k = int(k)
            if conv[k] < M * 0.95:
                break

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            a_idx = a[idx]
            a_k_minus = a[k - idx]
            products = a_idx * a_k_minus

            # Find top contributing pairs
            n_top = min(5, len(idx))
            top_pairs = np.argsort(products)[-n_top:]

            for pi in top_pairs:
                j = idx[pi]
                l = k - j
                if j == l:
                    continue  # Skip self-pairs for now
                if j >= n or l >= n:
                    continue

                # Try reducing a[j] (and by symmetry a[n-1-j]) and a[l] (and a[n-1-l])
                old_j = a[j]
                old_l = a[l]
                if old_j < 1e-12 or old_l < 1e-12:
                    continue

                for red_frac in [0.1, 0.2, 0.3, 0.5]:
                    trial = a.copy()
                    # Reduce by symmetry
                    trial[j] -= old_j * red_frac
                    trial[n - 1 - j] -= a[n - 1 - j] * red_frac
                    trial[l] -= old_l * red_frac
                    trial[n - 1 - l] -= a[n - 1 - l] * red_frac
                    trial = np.maximum(trial, 0.0)

                    # Redistribute to low-contribution positions
                    diff = old_j * red_frac + a[n - 1 - j] * red_frac + \
                           old_l * red_frac + a[n - 1 - l] * red_frac
                    if diff < 1e-15:
                        continue

                    # Find positions not contributing to this peak
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[js:je + 1] = False
                    # Also exclude symmetric positions
                    non_contrib[js:je + 1] = False
                    if non_contrib.sum() < 5:
                        # Use all positions with low values
                        threshold = np.percentile(a, 25)
                        non_contrib = a < threshold
                        if non_contrib.sum() < 5:
                            continue

                    nc_mask = non_contrib
                    nc_vals = trial[nc_mask]
                    nc_sum = nc_vals.sum()
                    if nc_sum < 1e-12:
                        continue
                    trial[nc_mask] += diff * nc_vals / nc_sum

                    trial = symmetrize(trial)
                    obj = fast_convolve(trial).max()

                    if obj < best_obj - 1e-14:
                        a = trial
                        best_obj = obj
                        best_a = trial.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Try broader reduction
                k_max = int(np.argmax(conv))
                js = max(0, k_max - n + 1)
                je = min(n - 1, k_max)
                for scale in [0.95, 0.9, 0.85, 0.8]:
                    trial = a.copy()
                    trial[js:je + 1] *= scale
                    # Compensate
                    deficit = a.sum() - trial.sum()
                    if deficit > 0:
                        low_mask = a < np.percentile(a, 50)
                        if low_mask.sum() > 0:
                            trial[low_mask] += deficit * a[low_mask] / a[low_mask].sum()
                    trial = symmetrize(trial)
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        a = trial
                        best_obj = obj
                        best_a = trial.copy()
                        no_improve = 0
                        improved = True
                        break
                if not improved and no_improve > 60:
                    break
        else:
            no_improve = 0

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian window normalized to sum=1."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = a / a.sum()
    return a


def make_hann(n):
    """Hann window normalized to sum=1."""
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_blackman(n):
    """Blackman window normalized to sum=1."""
    a = 0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / n) + \
        0.08 * np.cos(4 * np.pi * np.arange(n) / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, edge_frac=0.2):
    """Flat-topped window with cosine edges."""
    a = np.ones(n)
    edge = int(n * edge_frac)
    for i in range(edge):
        a[i] = 0.5 * (1 + np.cos(np.pi * i / edge))
        a[n - 1 - i] = a[i]
    a = a / a.sum()
    return a


def interpolate_seq(a, new_n):
    """Linearly interpolate a sequence to new length, maintaining sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s > 1e-15:
        a_interp /= s
    return a_interp


def multiplicative_weights(a0, time_limit, top_k=50, T_init=0.001, T_min=1e-5,
                           lr_init=0.02, verbose=False, label=""):
    """
    Multiplicative weights (mirror descent) on symmetric simplex.
    Different from FISTA - uses exponential updates which preserve positivity naturally.
    """
    a = symmetrize(a0.astype(np.float64))
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Weights: focus on peaks
        K = min(top_k, len(conv))
        if K < len(conv):
            threshold = np.partition(conv, len(conv) - K)[len(conv) - K]
            active = conv >= threshold * 0.9
        else:
            active = np.ones(len(conv), dtype=bool)

        w = softmax_weights(conv, T)
        g = compute_gradient_fft(a, w)

        # Multiplicative update with backtracking
        best_alpha = lr * 0.01
        best_trial = a.copy()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            if trial_lr < 1e-12:
                continue
            log_update = -trial_lr * g / M  # Normalize by M for scale invariance
            log_update = np.clip(log_update, -50, 50)
            trial = a * np.exp(log_update)
            trial = np.maximum(trial, 1e-15)
            trial = symmetrize(trial)
            trial_obj = fast_convolve(trial).max()
            if trial_obj <= M * 1.005:
                best_alpha = trial_lr
                best_trial = trial
            if trial_obj < M * 0.999:
                best_alpha = trial_lr
                best_trial = trial
                break

        a = best_trial
        conv = fast_convolve(a)
        new_obj = conv.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    {label} MW iter {it}: obj={best_obj:.10f}, lr={best_alpha:.6f}")
        else:
            no_improve += 1

        T = max(T * 0.9997, T_min)
        if no_improve > 100:
            lr *= 0.7
            no_improve = 0
        if no_improve > 500:
            rng = np.random.default_rng(it * 27181 + 11)
            a = symmetrize(np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-15))
            lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining FISTA, multiplicative weights,
    subspace optimization, and peak shaving across diverse initializations
    and sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_eval(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        # Scale to reasonable range for evaluation
        a_eval = a * (1.0 / s)  # sum = 1
        score = float(evaluate_sequence(a_eval.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_eval.copy()
            print(f"  NEW BEST [{label}]: score={score:.6f} (n={len(a_eval)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = try_eval(parent, "parent")
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Generate diverse initializations ===
    print("\n=== Phase 1: Initializations ===")
    n_base = 768
    if parent is not None:
        n_base = len(parent)

    inits = []
    if parent is not None:
        inits.append((symmetrize(parent), "parent_sym"))
        # Also try parent at slightly different lengths
        for n_try in [n_base - 32, n_base + 32]:
            if n_try > 100:
                inits.append((symmetrize(interpolate_seq(parent, n_try)), f"parent_n{n_try}"))

    for n_try in [n_base, 640, 896]:
        if n_try == n_base and parent is not None:
            continue
        inits.append((symmetrize(make_gaussian(n_try, 0.25)), f"gauss25_n{n_try}"))
        inits.append((symmetrize(make_gaussian(n_try, 0.35)), f"gauss35_n{n_try}"))
        inits.append((symmetrize(make_hann(n_try)), f"hann_n{n_try}"))
        inits.append((symmetrize(make_blackman(n_try)), f"blackman_n{n_try}"))
        inits.append((symmetrize(make_flat_top(n_try, 0.15)), f"flat_n{n_try}"))

    # Evaluate all initializations
    best_init = None
    best_init_obj = float('inf')
    for a_init, label in inits:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        obj = fast_convolve(a_norm).max() * 2 * len(a_norm)
        try_eval(a_norm, f"init_{label}")
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm

    if best_init is None:
        best_init = symmetrize(make_gaussian(n_base, 0.3))

    print(f"  Best init: obj*2n = {best_init_obj:.6f}")

    # === Phase 2: FISTA (main optimizer) ===
    print(f"\n=== Phase 2: FISTA ===")
    remaining = deadline - time.monotonic()
    t_fista = remaining * 0.35
    print(f"  Time allocated: {t_fista:.0f}s")
    a_opt, obj = fista_symmetric(best_init, time_limit=t_fista, T_init=0.003,
                                 T_decay=0.9997, L_init=5.0, verbose=True, label="FISTA1")
    try_eval(a_opt, "FISTA1")

    # === Phase 3: Multiplicative Weights ===
    print(f"\n=== Phase 3: Multiplicative Weights ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_mw = remaining * 0.20
        print(f"  Time allocated: {t_mw:.0f}s")
        a_opt, obj = multiplicative_weights(best_seq, time_limit=t_mw, top_k=50,
                                            T_init=0.002, lr_init=0.015,
                                            verbose=True, label="MW1")
        try_eval(a_opt, "MW1")

    # === Phase 4: Subspace Optimization ===
    print(f"\n=== Phase 4: Subspace Optimization ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_sub = remaining * 0.20
        print(f"  Time allocated: {t_sub:.0f}s")
        # Try multiple mode counts
        for M in [20, 35]:
            t_sub_i = t_sub * 0.5
            if time.monotonic() > deadline - 60:
                break
            a_opt, obj = subspace_optimize_smoothed(best_seq, time_limit=t_sub_i,
                                                    M_modes=M, T=0.005,
                                                    verbose=True, label=f"subM{M}")
            try_eval(a_opt, f"subM{M}")

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_ps = remaining * 0.15
        print(f"  Time allocated: {t_ps:.0f}s")
        a_opt, obj = peak_shave(best_seq, time_limit=t_ps, verbose=True, label="PS1")
        try_eval(a_opt, "PS1")

    # === Phase 6: Second FISTA pass with different params ===
    print(f"\n=== Phase 6: FISTA pass 2 ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_f2 = remaining * 0.20
        print(f"  Time allocated: {t_f2:.0f}s")
        a_opt, obj = fista_symmetric(best_seq, time_limit=t_f2, T_init=0.001,
                                     T_decay=0.9999, L_init=3.0, verbose=True, label="FISTA2")
        try_eval(a_opt, "FISTA2")

        # Also try from a different initial
        if time.monotonic() < deadline - 60:
            a_init2 = symmetrize(interpolate_seq(best_seq, int(len(best_seq) * 0.85)))
            a_opt, obj = fista_symmetric(a_init2, time_limit=60, T_init=0.002,
                                         T_decay=0.9998, L_init=4.0, verbose=False, label="FISTA3")
            try_eval(a_opt, "FISTA3")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 120:
            t_multi = remaining * 0.30
            n_candidates = [int(len(best_seq) * f) for f in [0.7, 0.8, 0.9, 1.0, 1.1, 1.2]]
            n_candidates = sorted(set(n for n in n_candidates if 100 < n < 3000))
            t_per = t_multi / max(len(n_candidates), 1)
            print(f"  Testing {len(n_candidates)} lengths, {t_per:.0f}s each")

            for n_try in n_candidates:
                if time.monotonic() > deadline - 30:
                    break
                a_interp = symmetrize(interpolate_seq(best_seq, n_try))
                # Quick FISTA
                t_alloc = min(t_per, deadline - time.monotonic() - 10)
                if t_alloc < 15:
                    break
                a_opt, obj = fista_symmetric(a_interp, time_limit=t_alloc, T_init=0.002,
                                             T_decay=0.9997, L_init=4.0,
                                             verbose=False, label=f"multin_{n_try}")
                try_eval(a_opt, f"multin_{n_try}")

    # === Phase 8: Final polished FISTA ===
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_final = min(remaining - 10, 120)
            print(f"  Final polish: {t_final:.0f}s")
            a_opt, obj = fista_symmetric(best_seq, time_limit=t_final, T_init=0.0005,
                                         T_min=5e-6, T_decay=0.9999, L_init=2.0,
                                         verbose=True, label="final")
            try_eval(a_opt, "final_polish")

            # Quick MW polish
            if time.monotonic() < deadline - 20:
                t_mw_f = min(40, deadline - time.monotonic() - 10)
                if t_mw_f > 10:
                    a_opt, obj = multiplicative_weights(best_seq, time_limit=t_mw_f,
                                                        top_k=20, T_init=0.001,
                                                        lr_init=0.005, verbose=False,
                                                        label="MW_final")
                    try_eval(a_opt, "MW_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(n_base) / n_base
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    elif not np.isfinite(fs):
        best_seq = np.ones(n_base) / n_base
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: score={best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
