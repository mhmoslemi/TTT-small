from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lp_refine(f, max_iters=50, tol=1e-8):
    """Iterative LP refinement: given f, find g that minimizes max(f*g) subject to g>=0, sum(g)=1."""
    n = len(f)
    f = f / f.sum()  # normalize
    a = f.copy()
    
    # Build constraint matrix A for f*g <= M
    # (f*g)_k = sum_j f_j * g_{k-j} for valid j
    # For each k (0 to 2n-2), the coefficient of g_i in (f*g)_k is f_{k-i} if 0 <= k-i < n
    # So A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    
    # LP: minimize M
    # subject to: A @ g <= M  (i.e., A @ g - M <= 0)
    #             sum(g) = 1
    #             g >= 0
    #
    # Variables: [g_0, ..., g_{n-1}, M]
    # c = [0, ..., 0, 1]
    # A_ineq: [A | -I] @ [g; M] <= 0
    # A_eq: [1...1 0] @ [g; M] = 1
    # bounds: g_i >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ineq = np.zeros((2*n-1, n+1))
    A_ineq[:,:n] = A
    A_ineq[:,n] = -1.0
    
    A_eq = np.zeros((1, n+1))
    A_eq[0,:n] = 1.0
    
    res = linprog(c, A_ineq=A_ineq, b_ineq=np.zeros(2*n-1),
                  A_eq=A_eq, b_eq=np.array([1.0]),
                  bounds=[(0, None)]*n + [(0, None)],
                  method='highs')
    
    if res.success:
        g = res.x[:n]
        M_new = res.x[n]
        return g, M_new
    return a, np.inf

    n = len(f_init)
    f = f_init.copy() / f_init.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    # Precompute the structure of the LP
    # A matrix: A[k,i] = f[k-i]... but wait, A changes each iteration because f changes!
    
    for it in range(n_iters):
        # Build A matrix for current f
        A = np.zeros((2*n-1, n))
        for i in range(n):
            A[i:i+n, i] = f
        
        # Solve LP
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ineq = np.hstack([A, -np.eye(2*n-1)])
        A_eq = np.zeros((1, n+1))

                      A_eq=A_eq, b_eq=np.array([1.0]),
                      bounds=[(0, None)]*(n+1), method='highs')
        if not res.success:
            break
        
        g = res.x[:n]
        
        # Update
        f_new = (1 - t_init) * f + t_init * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_obj

We want to minimize max_k conv(a,a)_k subject to a >= 0, sum a = 1.

conv(a,a)_k = sum_i a_i * a_{k-i} is a quadratic form in a (specifically, a bilinear form). The maximum of quadratic forms is in general non-convex.

"""
Iterative LP-based optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.

Key idea: For a fixed sequence f, finding g that minimizes max_k (f*g)_k
subject to g>=0, sum(g)=1 is a linear program. By iterating this best-response
dynamics (f <- (1-t)f + t*g), we converge to a local optimum of the
self-convolution minimization. This is exact at each step (unlike subgradient
methods) and exploits the linear structure for efficiency.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(f):
    """Build dense convolution matrix: A[k,i] = f[k-i] if 0<=k-i<n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def lp_step(f):
    """Solve LP: minimize M s.t. (f*g)_k <= M for all k, g>=0, sum(g)=1.
    Returns (g, M) or (None, inf) on failure."""
    n = len(f)
    A_conv = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ineq = np.zeros((2 * n - 1, n + 1))
    A_ineq[:, :n] = A_conv
    A_ineq[:, n] = -1.0
    b_ineq = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ineq=A_ineq, b_ineq=b_ineq,
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if res.success:
        g = res.x[:n]
        M = res.x[n]
        return g, M
    return None, np.inf


def iterative_lp(f_init, time_limit, t_init=0.5, seed=42, verbose=True):
    """Iterative LP best-response optimization. Returns (best_f, best_max_conv)."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(f_init)
    f = np.maximum(f_init, 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t = t_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g, M_lp = lp_step(f)
        if g is None:
            if verbose:
                print(f"    LP failed at iter {it}, stopping")
            break

        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s

        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve == 0:
            t = min(t * 1.03, 0.95)
        elif no_improve > 30:
            t = max(t * 0.8, 0.05)

        f = f_new

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={it}, best_obj={best_obj:.10f}, t={t:.3f}, "
                  f"M_lp={M_lp:.10f}, elapsed={elapsed:.1f}s")

        if no_improve > 300:
            f = best_f + rng.standard_normal(n) * 0.003
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            t = 0.5
            no_improve = 0
            if verbose:
                print(f"    Perturbation restart at iter {it}")

    return best_f, best_obj


def coordinate_polish(a0, time_limit=30.0):
    """Coordinate descent: try scaling each element by various multipliers."""
    t0 = time.monotonic()
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a0, np.inf
    a = a / s
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5])
    indices = np.arange(n)

    sweeps = 0
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        sweeps += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old = a[j]
            if old < 1e-15:
                continue
            best_j = best_obj
            best_val = old
            denom = 1.0 - old
            if denom < 1e-10:
                continue
            for mult in multipliers:
                new_val = old * mult
                if new_val < 0.0 or new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / denom
                if scale <= 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                a_trial[indices != j] *= scale
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial = a_trial / ss
                obj = fast_convolve(a_trial).max()
                if obj < best_j - 1e-14:
                    best_j = obj
                    best_val = new_val
            if best_val != old:
                scale = (1.0 - best_val) / denom
                a[j] = best_val
                a[indices != j] *= scale
                ss = a.sum()
                a = a / ss
                if best_j < best_obj - 1e-14:
                    best_obj = best_j
                    best_a = a.copy()
                    improved = True

    return best_a, best_obj


def make_gaussian(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    return a / a.sum()


def make_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_uniform(n):
    return np.ones(n) / n


def make_triangular(n):
    x = np.linspace(0, 1, n)
    a = 1.0 - 2.0 * np.abs(x - 0.5)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_asymmetric(n, alpha=0.3):
    """Asymmetric beta-like shape."""
    x = np.linspace(0.001, 0.999, n)
    a = x ** alpha * (1.0 - x) ** (1.0 - alpha)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f_norm)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result
