import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_peak_gradient(a, k, n):
    """Compute gradient of conv[k] with respect to a. conv[k] = sum_j a[j]*a[k-j]."""
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        other = k - j
        if 0 <= other < n:
            if j == other:
                grad[j] += 2.0 * a[j]
            else:
                grad[j] += a[other]
    return grad


def conv_peak_gradient_fast(a, k, n):
    """Vectorized gradient of conv[k] w.r.t. a."""
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    j_idx = np.arange(js, je + 1)
    other_idx = k - j_idx
    valid = (other_idx >= 0) & (other_idx < n)
    if not valid.any():
        return grad
    jv = j_idx[valid]
    ov = other_idx[valid]
    np.add.at(grad, jv, a[ov])
    diagonals = (jv == ov)
    if diagonals.any():
        diag_pos = jv[diagonals]
        grad[diag_pos] += a[diag_pos]
    return grad


def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP: (f*g)[k] for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

        current_obj = 2.0 * n * M

            no_improve = 0

        lr *= decay

        if verbose and it % 200 == 0:
            print(f"    Subgrad iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, no_improve={no_improve}")

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """
    LP-based Frank-Wolfe with adaptive M expansion and line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP direction
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 8.0:
                # Perturbation strategy
                conv_cur = fast_convolve(f)
                k_peak = int(np.argmax(conv_cur))
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                f = best_f.copy()
                f[js:je + 1] *= 0.85
                diff = 1.0 - f.sum()
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je + 1] = False
                if non_peak.any():
                    w = f[non_peak]
                    wsum = w.sum()
                    if wsum > 1e-15:
                        f[non_peak] += diff * w / wsum
                f = np.maximum(f, 0)
                fsum = f.sum()
                if fsum > 1e-15:
                    f = f / fsum
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-13:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.05, 5.0)
            if no_improve > 80:
                rng = np.random.default_rng(it * 13 + 1)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            if diff > 1e-4:
                no_improve = 0
            elif diff > 1e-6:
                no_improve += 1
            else:
                no_improve += 3

        # Line search: try several step sizes
        best_alpha = 0.0
        best_alpha_obj = 2.0 * n * M
        step_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        if diff < 1e-3:
            step_candidates = [0.3, 0.5, 0.7, 0.9, 1.0, 1.5, 2.0]

        for alpha in step_candidates:
            a_clip = min(alpha, 2.0)
            f_new = f + a_clip * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_alpha_obj:
                best_alpha_obj = obj_new
                best_alpha = a_clip

        if best_alpha < 1e-8:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 4.0)
        else:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            current_obj = 2.0 * n * M
            if current_obj < best_obj - 1e-14:
                best_obj = current_obj
                best_f = f.copy()

        if verbose and it % 50 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}, m_exp={m_exp:.2f}, diff={diff:.2e}")

        if no_improve > 150:
            break

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each element, try adjusting it to reduce the peak.
    More targeted than full gradient methods.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    sweep = 0

    while time.monotonic() - t0 < time_limit and no_improve < 30:
        conv = fast_convolve(f)
        M = conv.max()
        k_peak = int(np.argmax(conv))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        improved = False

        # Find the most "excess" element among peak contributors
        contributors = np.arange(js, je + 1)
        products = f[contributors] * f[k_peak - contributors]
        order = np.argsort(products)[::-1]

        for oi in range(min(8, len(order))):
            j = int(contributors[order[oi]])
            old_val = f[j]
            if old_val < 1e-12:
                continue

            # Try reducing this element
            for mult in [0.3, 0.5, 0.7, 0.85, 0.95]:
                new_val = old_val * mult
                delta = old_val - new_val
                if delta < 1e-15:
                    continue

                trial = f.copy()
                trial[j] = new_val
                # Redistribute to non-peak elements
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je + 1] = False
                if non_peak.any():
                    w = f[non_peak]
                    wsum = w.sum()
                    if wsum > 1e-15:
                        trial[non_peak] += delta * w / wsum
                trial = np.maximum(trial, 0)
                tsum = trial.sum()
                if tsum < 1e-15:
                    continue
                trial = trial / tsum

                conv_t = fast_convolve(trial)
                obj_t = 2.0 * n * conv_t.max()
                if obj_t < best_obj - 1e-13:
                    best_obj = obj_t
                    best_f = trial.copy()
                    f = trial
                    improved = True
                    break

        if not improved:
            no_improve += 1
            # Try a small random perturbation
            if no_improve > 5:
                rng = np.random.default_rng(sweep * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                conv = fast_convolve(f)
                obj = 2.0 * n * conv.max()
                if obj < best_obj:
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0

        else:
            no_improve = 0
            sweep += 1

        if verbose and sweep % 10 == 0:
            print(f"    CoordDescent sweep {sweep}: obj={best_obj:.10f}")

    return best_f, best_obj


def multiplicative_descent(f0, time_limit, verbose=False):
    """Mirror descent with multiplicative weight updates targeting top peaks."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    temperature = 0.003
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        k_actual = min(30, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = 2.0 * n * fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temperature *= 1.01

        if no_improve > 200:
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = max(0.003, lr_init_val)
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, f[0] / new_n)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_hann(n):
    """Hann window."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flattop(n):
    """Flattop window (highly flat top, good rolloff)."""
    x = np.linspace(0, 1, n)
    a = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * x) +
         0.277263158 * np.cos(4 * np.pi * x) - 0.083578947 * np.cos(6 * np.pi * x) +
         0.006947368 * np.cos(8 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n, separation=0.3):
    """Bimodal sequence."""
    c1 = n * (0.5 - separation / 2)
    c2 = n * (0.5 + separation / 2)
    sigma = n * 0.08
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence, return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


lr_init_val = 0.003


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining subgradient descent, LP-Frank-Wolfe,
    coordinate descent, and multiplicative descent with multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_hann(768)
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "hann_default")

    n_parent = len(parent)
    print(f"\nStarting search. Parent n={n_parent}, score={best_score:.6f}")

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse Initializations ===")
    init_candidates = [parent.copy()]

    for n_try in [640, 700, 768, 800, 850, 900, 1024]:
        if n_try == n_parent:
            init_candidates.append(parent.copy())
        else:
            init_candidates.append(interpolate_sequence(parent, n_try))
        init_candidates.append(make_hann(n_try))
        init_candidates.append(make_gaussian(n_try, 0.2))
        init_candidates.append(make_gaussian(n_try, 0.3))
        init_candidates.append(make_flattop(n_try))
        init_candidates.append(make_bimodal(n_try, 0.2))

    best_init = None
    best_init_obj = float('inf')
    for ic in init_candidates:
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        conv = fast_convolve(a_norm)
        obj = 2.0 * len(a_norm) * conv.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        best_seq, best_score = try_sequence(a_norm, best_seq, best_score, f"init_n{len(a_norm)}")

    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: Subgradient optimization on best init ===
    print(f"\n=== Phase 2: Subgradient (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 200)
        print(f"  Subgrad: t={t_alloc:.0f}s")
        a_opt, obj = subgradient_optimize(best_init, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "subgrad_main")

    # === Phase 3: LP-FW on best ===
    base = best_seq if best_seq is not None else best_init
    if best_seq is not None:
        print(f"\n=== Phase 3: LP-FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t_alloc = min(rem, 250)
            print(f"  LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw_main")

    # === Phase 4: Subgradient again (different local min?) ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Subgradient round 2 (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 400
        if rem > 50:
            t_alloc = min(rem, 120)
            print(f"  Subgrad2: t={t_alloc:.0f}s")
            a_opt, obj = subgradient_optimize(best_seq, time_limit=t_alloc, top_k=10, lr_init=0.01)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "subgrad_2")

    # === Phase 5: Coordinate descent polish ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Coord Descent (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  CoordDesc: t={t_alloc:.0f}s")
            a_opt, obj = coord_descent_polish(best_seq, time_limit=t_alloc, verbose=True)

    # === Phase 6: Multi-length LP-FW exploration ===
    base = best_seq if best_seq is not None else best_init
    print(f"\n=== Phase 6: Multi-length LP-FW ===")
    for target_n in [720, 780, 832, 896, 1024]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 80:
            break
        f_interp = interpolate_sequence(base, target_n)
        best_seq, best_score = try_sequence(f_interp, best_seq, best_score, f"interp_n{target_n}")
        t_alloc = min(rem * 0.35, 100)
        if t_alloc < 25:
            continue
        print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")

        # Subgradient on this length too

        if rem2 > 40:

        rem = deadline - time.monotonic() - 150
        if rem > 60:
            t_alloc = min(rem, 200)
            print(f"  Deep LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "deep_lpfw")

    # === Phase 8: Perturbed variants ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Perturbed LP-FW ===")
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            t_alloc = min(rem / (3 - p), 80)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                # Blend with uniform
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            else:
                # Mirror blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()

            print(f"  Variant {p}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, f"perturb_{p}")

    # === Phase 9: Final polish ===
    if best_seq is not None:
        print(f"\n=== Phase 9: Final Polish ===")
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t1 = min(rem * 0.35, 60)
            t2 = min(rem * 0.35, 60)
            t3 = max(0, rem - t1 - t2 - 5)

            print(f"  Final subgrad: t={t1:.0f}s")
            a_opt, obj = subgradient_optimize(best_seq, time_limit=t1, top_k=8, lr_init=0.005)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_subgrad")

            rem2 = deadline - time.monotonic() - 10
            if rem2 > 10 and best_seq is not None:
                t2_actual = min(t2, rem2 - 5)
                if t2_actual > 5:
                    print(f"  Final coord: t={t2_actual:.0f}s")
                    a_opt, obj = coord_descent_polish(best_seq, time_limit=t2_actual)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_coord")

            rem3 = deadline - time.monotonic() - 3
            if rem3 > 5 and best_seq is not None:
                t3_actual = min(t3, rem3)
                if t3_actual > 3:
                    print(f"  Final LP-FW: t={t3_actual:.0f}s")
                    a_opt, obj = lp_fw_optimize(best_seq, time_limit=t3_actual)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
