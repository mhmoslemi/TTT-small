def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, c, top_frac=0.1, temperature=0.01):
    """Compute weighted subgradient of max(c) using top peaks with softmax."""
    n = len(f)
    len_c = len(c)
    k_top = max(5, int(len_c * top_frac))
    peaks = np.argpartition(c, -k_top)[-k_top:]
    pv = c[peaks]
    scaled = pv / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[k - idx] + f[idx]
        sg[js:je + 1] += w[i] * contrib
    return sg

        else:
            f_new = f_new / s

                    obj_new = obj_trial

                if lr < 1e-8:
                    break
                continue

        if obj_new < best_obj - 1e-14:

            best_f = f.copy()

            lr = min(lr * 1.002, 0.05)
            if verbose and iter_count % 200 == 0:

            tau = max(tau_init, tau * 0.5)

def solve_lp_direction(f, M, n, active_lags=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k (or active lags), g >= 0."""
    c_obj = -np.ones(n)
    
    if active_lags is not None:
        lags = active_lags
    else:
        lags = range(2 * n - 1)
    
    n_lags = len(lags) if isinstance(lags, (list, np.ndarray)) else (2 * n - 1)
    
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    
    # Only use active constraints for speed
    if active_lags is not None and len(active_lags) < (2 * n - 1) // 2:
        A_ub = A_ub[active_lags, :]
        b_ub = b_ub[active_lags]
    
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with active set refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find near-active lags
        conv = fast_convolve(f)
        M = conv.max()
        threshold = M * 0.95
        active_lags = np.where(conv >= threshold)[0].tolist()
        
        g = solve_lp_direction(f, M * m_exp, n, active_lags)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m_exp = 1.0
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Fine line search
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(5):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            m_exp = 1.0

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)

        if no_improve > 100:
            break

    return best_f, best_obj


def coordinate_descent_fine(f0, time_limit, verbose=False):
    """Fine coordinate descent with exact 1D optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        rng = np.random.default_rng(epoch * 1000 + 7)
        order = rng.permutation(n)
        improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]

            # Try multiple candidate values
            candidates = []
            if old_val > 0:
                candidates = [0.0, old_val * 0.5, old_val * 0.7, old_val * 1.3,
                             old_val * 1.5, old_val * 0.3, old_val * 2.0]
            else:
                candidates = [0.01, 0.005, 0.02, 0.001]

            best_j_obj = best_obj
            best_j_val = old_val

            for new_val in candidates:
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and epoch > 1:
            break

    return best_f, best_obj


def make_asymmetric_init(n, variant=0, seed=42):
    """Generate asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 100)

    if variant == 0:
        a = np.exp(-np.arange(n) / (n * 0.15))
    elif variant == 1:
        a = np.exp(-np.arange(n)[::-1] / (n * 0.15))
    elif variant == 2:
        x = np.arange(n) / n
        a = np.exp(-3 * (x - 0.2) ** 2 / (0.15 ** 2)) + 0.3 * np.exp(-3 * (x - 0.8) ** 2 / (0.1 ** 2))
    elif variant == 3:
        a = np.linspace(0.01, 1.0, n)
    elif variant == 4:
        x = np.arange(n) / n
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif variant == 5:
        a = np.cos(np.pi * np.arange(n) / (2 * n)) ** 2 + 0.05
    elif variant == 6:
        a = np.ones(n)
        a[:n//3] *= 1.5
        a[2*n//3:] *= 0.5
        a += rng.standard_normal(n) * 0.1
        a = np.maximum(a, 0.01)
    elif variant == 7:
        a = np.log1p(np.arange(n) + 1)
    elif variant == 8:
        a = (np.arange(n) + 1) ** (-0.5)
    elif variant == 9:
        # Power law with modulation
        x = np.arange(n) / n
        a = (x + 0.1) ** (-0.3) * (1 + 0.3 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif variant == 10:
        # Two exponential bumps
        x = np.arange(n) / n
        a = np.exp(-8 * (x - 0.25) ** 2) + 0.4 * np.exp(-12 * (x - 0.75) ** 2)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage optimization:
diverse asymmetric initialization -> smoothed mirror descent -> active-set
LP-Frank-Wolfe -> coordinate descent. Explores multiple sequence lengths
with interpolation and random restarts to escape local minima.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_mirror_descent(f0, time_limit, verbose=False,
                            tau_init=10.0, tau_decay=1.004,
                            lr_init=0.005, lr_decay=0.9996,
                            restart_every=250, seed=42):
    """Adaptive smoothed mirror descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()

        len_c = len(conv)
        k_top = max(10, int(len_c * 0.12))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] + f[idx]
            sg[js:je + 1] += w[i] * contrib

        sg_mean = sg.mean() + 1e-15
        f_new = f * np.exp(-lr * (sg - sg_mean))
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.01:
            found = False
            for damp in [0.5, 0.25, 0.1, 0.05, 0.02]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    found = True
                    break
            if not found:
                lr *= 0.5
                if lr < 1e-9:
                    break
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.001, 0.04)
            if verbose and iter_count % 200 == 0:
                print(f"      SMD iter {iter_count}: obj={best_obj:.10f}, tau={tau:.1f}", flush=True)
        else:
            no_improve += 1

        tau *= tau_decay
        lr *= lr_decay

        if no_improve > restart_every:
            rng = np.random.default_rng(iter_count * 7 + seed)
            noise_scale = 0.002 + 0.001 * (no_improve / restart_every)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.4
            tau = max(tau_init, tau * 0.5)
            no_improve = 0

    return best_f, best_obj


def solve_lp_direction(f, M, n, active_lags=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for specified lags, g >= 0."""
    c_obj = -np.ones(n)

    if active_lags is not None and len(active_lags) < (2 * n - 1):
        n_lags = len(active_lags)
        A_ub = np.zeros((n_lags, n))
        b_ub = np.zeros(n_lags)
        for i, k in enumerate(active_lags):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[i, j] = f[k - j]
            b_ub[i] = M
    else:
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)

    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, seed=42):
    """LP-based Frank-Wolfe with active set for speed."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        M = conv.max()
        threshold = M * 0.9
        active_lags = np.where(conv >= threshold)[0].tolist()
        if len(active_lags) > 300:
            top_k = np.argpartition(conv, -300)[-300:]
            active_lags = top_k.tolist()

        g = solve_lp_direction(f, M * m_exp, n, active_lags)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                rng = np.random.default_rng(it * 17 + seed)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m_exp = 1.0
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            no_improve += 1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 80:
                rng = np.random.default_rng(it * 31 + seed)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(5):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            m_exp = 1.0

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}", flush=True)
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)

        if no_improve > 80:
            break

    return best_f, best_obj


def coordinate_descent_fine(f0, time_limit, verbose=False, seed=42):
    """Coordinate descent with multiple candidate values per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        rng = np.random.default_rng(epoch * 1000 + seed)
        order = rng.permutation(n)
        improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]

            candidates = [0.0]
            if old_val > 1e-10:
                candidates.extend([old_val * 0.3, old_val * 0.5, old_val * 0.7,
                                   old_val * 1.3, old_val * 1.5, old_val * 2.0])
            else:
                candidates.extend([0.001, 0.005, 0.01, 0.02])

            best_j_obj = best_obj
            best_j_val = old_val

            for new_val in candidates:
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                if new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def make_asymmetric_init(n, variant, seed):
    """Generate asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 100)

    if variant == 0:
        a = np.exp(-np.arange(n) / (n * 0.15))
    elif variant == 1:
        a = np.exp(-np.arange(n)[::-1] / (n * 0.15))
    elif variant == 2:
        x = np.arange(n) / n
        a = np.exp(-3 * (x - 0.2) ** 2 / (0.15 ** 2)) + 0.3 * np.exp(-3 * (x - 0.8) ** 2 / (0.1 ** 2))
    elif variant == 3:
        a = np.linspace(0.01, 1.0, n)
    elif variant == 4:
        x = np.arange(n) / n
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif variant == 5:
        a = np.cos(np.pi * np.arange(n) / (2 * n)) ** 2 + 0.05
    elif variant == 6:
        a = np.ones(n)
        a[:n // 3] *= 1.5
        a[2 * n // 3:] *= 0.5
        a = a + rng.standard_normal(n) * 0.1
        a = np.maximum(a, 0.01)
    elif variant == 7:
        a = np.log1p(np.arange(n) + 1)
    elif variant == 8:
        a = (np.arange(n) + 1) ** (-0.5)
    elif variant == 9:
        x = np.arange(n) / n
        a = (x + 0.1) ** (-0.3) * (1 + 0.3 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif variant == 10:
        x = np.arange(n) / n
        a = np.exp(-8 * (x - 0.25) ** 2) + 0.4 * np.exp(-12 * (x - 0.75) ** 2)

        a = np.exp(-4 * (x - 0.15) ** 2) * (1 + 0.2 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                        t_lpfw = min(rem3, 40.0)
                        a_lpfw, _ = lp_fw_optimize(best_seq, time_limit=t_lpfw, seed=seed)
                        best_seq, best_score = try_sequence(a_lpfw, best_seq, best_score, f"lpfw_n{target_n}")

    # === Phase 5: Random restarts ===
    print("\n=== Phase 5: Random Restarts ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(3):
            rem = deadline - time.monotonic() - 80.0
            if rem < 15:
                break
            t_alloc = min(rem / max(1, 3 - r), 50.0)
            if t_alloc < 10:
                t_alloc = 10

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
            elif r == 1:
                v = int(rng.integers(0, 12))
                asym = make_asymmetric_init(n_best, variant=v, seed=seed + r * 10)
                f_init = 0.6 * best_seq + 0.4 * asym
                f_init = f_init / f_init.sum()
            else:
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()

            print(f"  Restart {r} (n={n_best}): t={t_alloc:.0f}s", flush=True)
            a_opt, _ = smoothed_mirror_descent(f_init, time_limit=t_alloc * 0.5,
                                               tau_init=5.0, tau_decay=1.003,
                                               lr_init=0.004, seed=seed + r)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, f"restart_{r}")

            rem2 = deadline - time.monotonic() - 60.0
            if rem2 > 10 and len(best_seq) == n_best:
                a_lpfw, _ = lp_fw_optimize(best_seq, time_limit=min(rem2, 25.0), seed=seed + r)
                best_seq, best_score = try_sequence(a_lpfw, best_seq, best_score, f"restart_{r}_lfw")

    # === Phase 6: Coordinate descent ===
    print("\n=== Phase 6: Coordinate Descent ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20.0
        if rem > 10:
            t_alloc = min(rem, 50.0)
            print(f"  CD (n={len(best_seq)}): t={t_alloc:.0f}s", flush=True)
            a_opt, obj = coordinate_descent_fine(best_seq, time_limit=t_alloc, seed=seed)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "cd")

    # === Phase 7: Final LP-FW squeeze ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5.0
        if rem > 8:
            t_alloc = min(rem, 30.0)
            print(f"  Final LP-FW: t={t_alloc:.0f}s", flush=True)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, seed=seed + 99)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
