def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)

    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0)

    it = 0
    
    # FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute FFT
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

            # Inverse FFT
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            ss = a_new.sum()
            if ss < 1e-10:
                break
            a_new = a_new / ss
            
            new_obj = fast_convolve(a_new).max()
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    {label} SF iter {it}: obj={best_obj:.10f}, alpha={alpha:.3f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Decay alpha (tighten the cap)
        alpha = max(alpha * alpha_decay, 1.01)
        
        if no_improve > 200:

            rng = np.random.default_rng(it * 31337 + 7)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.001, 1e-15)
            a = a / a.sum()

            a = a / a.sum()

            no_improve = 0
    
    return best_a, best_obj

def coord_descent_polish(a0, time_limit, verbose=False, label=""):
    """
    Fine coordinate descent: adjust one coordinate at a time with
    multiple trial values, maintaining sum=1 via scaling others.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            # Don't try to make a single coordinate too large
            max_frac = 0.1
            if old_val > max_frac:
                continue  # Already large, skip
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try several multipliers
            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > max_frac:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def peak_shave(a0, time_limit, verbose=False, label=""):
    """
    Direct peak shaving: identify top peaks, find most contributing positions,
    and optimally redistribute mass.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        improved = False
        
        # Find top 5 peaks
        K = min(5, len(conv))
        peak_indices = np.argsort(conv)[-K:]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = a[idx] * a[k - idx]
            
            # Try reducing the top product
            for top_contrib in range(min(3, len(products))):
                j_max = idx[np.argsort(products)[-top_contrib-1]]
                old_val = a[j_max]
                if old_val < 1e-10:
                    continue
                
                best_trial = None
                best_trial_obj = M
                
                # Try moving mass to various positions
                for frac in [0.3, 0.5, 0.7]:
                    new_val = old_val * frac
                    diff = old_val - new_val
                    if diff < 1e-14:
                        continue
                    
                    # Find best position to receive the mass
                    # Try positions far from peak k
                    candidates = []
                    for m in range(0, n, max(1, n // 20)):
                        if m == j_max:
                            continue
                        # Check if m contributes to peak k
                        if js <= m <= je:
                            continue
                        candidates.append(m)
                    
                    if not candidates:
                        candidates = [m for m in range(n) if m != j_max and not (js <= m <= je)]
                    
                    for m in candidates[:10]:
                        trial = a.copy()
                        trial[j_max] = new_val
                        trial[m] += diff
                        trial = np.maximum(trial, 0.0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_trial_obj - 1e-14:
                            best_trial_obj = obj
                            best_trial = trial
                
                if best_trial is not None:
                    a = best_trial
                    improved = True
                    if best_trial_obj < best_obj - 1e-14:
                        best_obj = best_trial_obj
                        best_a = a.copy()
                        if verbose and it % 50 == 0:
                            print(f"    {label} PS iter {it}: obj={best_obj:.10f}")
                    break  # Move to next peak
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0
    
    return best_a, best_obj

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated multiplicative weight
(mirror descent), frequency-domain spectral flattening, peak shaving, and aggressive
multi-restart. Key innovations: Nesterov momentum for O(1/t^2) convergence rate,
spectral flattening exploiting the DFT structure of auto-convolution, adaptive
peak-balancing that encourages multiple simultaneous active peaks, and diverse
multi-restart with symmetrization and shape blending.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k.
    db_k/da_i = 2*a_{k-i} for valid i (where 0<=k-i<n).
    """
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx_arr = np.arange(js, je + 1)
        g[idx_arr] += 2.0 * w_k * a[k - idx_arr]
    return g

def softmax_weights(values, temp):
    """Compute softmax weights over peak values."""
    scaled = values / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(values))
        w[np.argmax(values)] = 1.0
    else:
        w /= ws
    return w

def get_top_peaks(conv, top_k):
    """Get indices of top-k peaks in convolution."""
    K = min(top_k, len(conv))
    if K < len(conv):
        return np.argpartition(conv, -K)[-K:]
    else:
        return np.arange(len(conv))

def nesterov_mw(a0, time_limit, top_k=20, temp_init=0.0002, lr_init=0.008,
                 verbose=False, label=""):
    """
    Nesterov-accelerated multiplicative weight (mirror descent) on the simplex.
    Minimizes max_k (a*a)_k subject to a>=0, sum(a)=1.
    Uses exponential smoothing momentum for O(1/t^2) convergence.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    y = a.copy()
    a_prev = a.copy()
    t_param = 1.0

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        y_conv = fast_convolve(y)
        M = y_conv.max()

        peak_indices = get_top_peaks(y_conv, top_k)
        pv = y_conv[peak_indices]
        w = softmax_weights(pv, temp)

        g = compute_subgrad(y, peak_indices, w)

        # Backtracking line search
        best_lr = lr * 0.1
        best_trial_obj = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-10:
                continue
            log_update = -trial_lr * g
            log_update = np.clip(log_update, -50, 50)
            a_trial = y * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_conv = fast_convolve(a_trial)
            trial_obj = trial_conv.max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        log_update = -best_lr * g
        log_update = np.clip(log_update, -50, 50)
        a_new = y * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        alpha = (t_param - 1.0) / t_new

        a_prev = a.copy()
        a = a_new
        y = a + alpha * (a - a_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        t_param = t_new

        conv_new = fast_convolve(a)
        new_obj = conv_new.max()

        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    {label} NMW iter {it}: obj={best_obj:.10f}, lr={best_lr:.6f}, t={t_param:.1f}")
        else:
            no_improve += 1

        temp = min(temp * 1.001, 0.03)
        lr = max(lr * 0.99995, 1e-8)

        if no_improve > 200:
            rng = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.001 + 0.001 * min(no_improve / 2000, 1.0)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 1e-15)
            a = a / a.sum()
            y = a.copy()
            a_prev = a.copy()
            t_param = 1.0
            lr = lr_init * 0.5
            no_improve = 0

        if no_improve > 1500:
            break

        if it % 400 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()
                y = a.copy()
                a_prev = a.copy()
                t_param = 1.0

    return best_a, best_obj

def spectral_flattening(a0, time_limit, alpha_init=4.0, alpha_decay=0.99995,
                       verbose=False, label=""):
    """
    Frequency-domain spectral flattening: cap peaks of |A|^2 (the periodogram)
    to reduce time-domain auto-convolution peaks. Exploits the identity:
    linear auto-conv = IDFT(|DFT(a_padded)|^2).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    alpha = alpha_init
    no_improve = 0
    it = 0

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Target level for non-DC components
        non_dc = power[1:]
        target_level = np.mean(non_dc) * alpha

        A_new = A.copy()
        changed = False
        for m in range(1, len(A)):
            if power[m] > target_level:
                A_new[m] = A[m] * np.sqrt(target_level / power[m])
                changed = True

        if not changed:
            no_improve += 1
        else:
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            ss = a_new.sum()
            if ss < 1e-10:
                break
            a_new = a_new / ss

            new_obj = fast_convolve(a_new).max()
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"    {label} SF iter {it}: obj={best_obj:.10f}, alpha={alpha:.4f}")
            else:
                no_improve += 1

        alpha = max(alpha * alpha_decay, 1.005)

        if no_improve > 100:
            rng = np.random.default_rng(it * 31337 + 7)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-15)
            a = a / a.sum()
            alpha = alpha_init
            no_improve = 0

        if alpha < 1.01:
            rng = np.random.default_rng(it * 271 + 11)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 1e-15)
            a = a / a.sum()
            alpha = alpha_init * 0.7
            no_improve = 0

    return best_a, best_obj

def coord_descent_polish(a0, time_limit, verbose=False, label=""):
    """
    Fine coordinate descent: adjust one coordinate at a time with
    multiple trial multipliers, maintaining sum=1 via rescaling others.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.05:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Try a broader sweep: increase small coordinates
            improved = False
            for j in range(n):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[j]
                if old_val > 0.001:
                    continue
                target = 0.001
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - target) / denom
                trial = a.copy()
                trial[j] = target
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_obj - 1e-14:
                    a = trial
                    best_obj = val
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj

def peak_shave(a0, time_limit, verbose=False, label=""):
    """
    Direct peak shaving: identify top peaks, find most contributing pairs,
    and redistribute mass to reduce the peak.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        improved = False

        K = min(4, len(conv))
        peak_indices = np.argsort(conv)[-K:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = a[idx] * a[k - idx]
            top_contribs = np.argsort(products)[-3:]

            for tc in top_contribs:
                j_max = idx[tc]
                old_val = a[j_max]
                if old_val < 1e-10:
                    continue

                best_trial = None
                best_trial_obj = M

                for frac in [0.3, 0.5, 0.7, 0.85]:
                    new_val = old_val * frac
                    diff = old_val - new_val
                    if diff < 1e-14:
                        continue

                    # Find positions that don't contribute to peak k
                    non_contrib_mask = np.ones(n, dtype=bool)
                    non_contrib_mask[js:je + 1] = False
                    non_contrib_mask[j_max] = False

                    nc_indices = np.where(non_contrib_mask)[0]
                    if len(nc_indices) == 0:
                        continue

                    # Pick target positions: low values among non-contributors
                    nc_sorted = nc_indices[np.argsort(a[nc_indices])[:8]]

                    for m in nc_sorted:
                        trial = a.copy()
                        trial[j_max] = new_val
                        trial[m] += diff
                        trial = np.maximum(trial, 0.0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_trial_obj - 1e-14:
                            best_trial_obj = obj
                            best_trial = trial

                if best_trial is not None:
                    a = best_trial
                    improved = True
                    if best_trial_obj < best_obj - 1e-14:
                        best_obj = best_trial_obj
                        best_a = a.copy()
                        if verbose and it % 50 == 0:
                            print(f"    {label} PS iter {it}: obj={best_obj:.10f}")
                    break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

    return best_a, best_obj

def rectilinear_flow(a0, time_limit, top_k=20, temp_init=0.001, dt_init=0.004,

    multiplicative descent, can escape different local minima.
    """

    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()

    it = 0

        w = softmax_weights(pv, temp)

        a = a - best_dt * g_centered

        new_obj = fast_convolve(a).max()
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj

def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence normalized to sum=1."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bimodal(n, c1=0.3, c2=0.7, sigma=0.08):
    """Bimodal Gaussian."""
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = a / a.sum()
    return a
