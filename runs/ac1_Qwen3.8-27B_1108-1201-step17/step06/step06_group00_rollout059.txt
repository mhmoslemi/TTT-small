"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated subgradient
descent combined with LP-based refinement (following the state-of-the-art
approach), coordinate descent polish, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, w):
    """
    Compute subgradient g[i] = sum_j a[j] * w[i+j] via FFT.
    a: length n, w: length 2n-1
    Returns: g of length n
    """
    n = len(a)
    # Need N >= (n-1) + (2n-2) + 1 = 3n-2 to avoid circular wraparound
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]
    return g


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute objective locally (assumes a is non-negative with sum > 0)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def subgradient_optimize_fft(a0, time_limit, lr_init=0.01, momentum=0.9,
                             n_top=None, seed=42, verbose=False):
    """
    FFT-accelerated projected subgradient method with Nesterov momentum.
    Works on the simplex (sum=1).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    no_improve = 0
    it_count = 0
    
    if n_top is None:
        n_top = max(10, n // 10)
    
    while time.monotonic() - t0 < time_limit:
        # Compute convolution
        b = fast_convolve(a)
        b_max = b.max()
        
        # Select top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        
        # Equal weights on top peaks (or use threshold)
        # Use a soft approach: weight proportional to how close to max
        top_vals = b[top_idx]
        threshold = b_max - 1e-10  # nearly exact peaks
        active = top_vals >= threshold
        if not active.any():
            active[0] = True
        w = np.zeros(len(b))
        w[top_idx[active]] = 1.0 / active.sum()
        
        # Compute subgradient via FFT
        g = compute_subgradient(a, w)
        
        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        # Evaluate
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it_count % 10000 == 0:
                print(f"    it={it_count}, obj={obj_new:.8f}")
        else:
            no_improve += 1
        
        a = a_new
        it_count += 1
        
        # Adaptive step size
        if no_improve > 5000:
            a = best_a + rng.standard_normal(n) * 0.001
            a = project_simplex(a)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.01)
            no_improve = 0
        elif no_improve > 2000:
            cur_lr *= 0.9
        elif no_improve > 1000:
            cur_lr *= 0.95
    
    return best_a, best_obj * 2 * n  # Return the actual objective (with sum=1, it's 2n*max_b)

    subject to conv(f,g) <= M and g >= 0. Then blend.

    n = len(f)
    f = f.copy().astype(np.float64)
    f = f / f.sum()
    
    best_a = f.copy()
    best_obj = fast_convolve(f).max()
    
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        
        # Build LP: maximize sum(g) subject to conv(f,g)_k <= M, g >= 0
        # conv(f,g)_k = sum_j f[j] * g[k-j] for valid j,k-j
        # For each k in 0..2n-2: sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j] <= M
        
        # Objective: c = -1 (minimize -sum(g))
        c = -np.ones(n)
        
        # Constraints: A @ g <= b_lp
        # Row k: for j in range(max(0, k-n+1), min(n-1, k)+1): A[k, k-j] += f[j]
        # b_lp[k] = M
        
        A_rows = []
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            row = np.zeros(n)
            for j in range(lo, hi + 1):
                row[k - j] = f[j]
            A_rows.append(row)
        
        A = np.array(A_rows)
        b_lp = np.full(2 * n - 1, M)
        
        # Solve LP
        try:
            res = linprog(c, A_ub=A, b_ub=b_lp, bounds=[(0, None)] * n,
                          method='highs')
            if res.success:
                g = res.x
                g_sum = g.sum()
                if g_sum > 1e-10:
                    # Normalize g to sum=1
                    g_norm = g / g_sum
                    
                    # Line search for blending parameter t
                    best_t = 0
                    best_blend_obj = best_obj
                    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        blend = (1 - t) * best_a + t * g_norm
                        blend = np.maximum(blend, 0)
                        blend_sum = blend.sum()
                        if blend_sum < 1e-10:
                            continue
                        blend = blend / blend_sum
                        b_blend = fast_convolve(blend)
                        obj_blend = b_blend.max()
                        if obj_blend < best_blend_obj:
                            best_blend_obj = obj_blend
                            best_t = t
                    
                    if best_t > 0:
                        f = (1 - best_t) * best_a + best_t * g_norm
                        f = np.maximum(f, 0)
                        f = f / f.sum()
                        b_f = fast_convolve(f)
                        if b_f.max() < best_obj:
                            best_obj = b_f.max()
                            best_a = f.copy()
                            if verbose:
                                print(f"    LP iter {iteration}: t={best_t}, obj={best_obj:.8f}")
                        
                        f = best_a.copy()
                    else:
                        # No improvement, try smaller t
                        improved = False
                        for t in [0.01, 0.02, 0.05]:
                            blend = (1 - t) * best_a + t * g_norm
                            blend = np.maximum(blend, 0)
                            blend = blend / blend.sum()
                            b_blend = fast_convolve(blend)
                            if b_blend.max() < best_obj - 1e-12:
                                f = blend
                                best_obj = b_blend.max()
                                best_a = f.copy()
                                improved = True
                                break
                        if not improved:
                            break
                else:
                    break
            else:
                break
        except Exception as e:
            if verbose:
                print(f"    LP failed: {e}")
            break
    
    return best_a, best_obj * 2 * n


def coordinate_descent(a0, time_limit, multipliers=None):
    """
    Coordinate descent: for each element, try scaling it while rescaling others.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0])
    
    improved_global = True
    passes = 0
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj * 2 * n
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 10:
                    continue
                
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    
    return best_a, best_obj * 2 * n


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = int(r * n // 2)
    if taper > 0:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated subgradient
descent with Nesterov momentum, LP-based refinement (following the state-of-the-art
active-set approach), coordinate descent polish, and multi-length exploration.

Key innovations over previous approach:
- FFT-accelerated subgradient computation (O(n log n) per step)
- Direct LP-based refinement for principled peak reduction
- Unified FFT size for convolution and subgradient (saves compute)
- Aggressive multi-length exploration with LP polish at each length
- Adaptive restart strategy with diversity injection
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_and_subgradient(a, w):
    """
    Compute both auto-convolution of a and subgradient g[i] = sum_j a[j]*w[i+j].
    Uses a single FFT size for efficiency.
    Returns (b, g) where b is the convolution (length 2n-1) and g is length n.
    """
    n = len(a)
    # Need fft_size >= 3n-2 for subgradient, which also covers convolution
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    b_full = irfft(A * A, fft_size)
    b = b_full[:2 * n - 1]
    W = rfft(w, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]
    return b, g


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def subgradient_optimize_fft(a0, time_limit, lr_init=0.005, momentum=0.9,
                             n_top=None, seed=42, restart_interval=3000,
                             kick_size=0.003, decel_factor=0.92):
    """
    FFT-accelerated projected subgradient with Nesterov momentum.
    Optimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    Returns (best_a, best_max_b).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_local_max_b(a)
    cur_lr = lr_init
    no_improve = 0
    it_count = 0

    if n_top is None:
        n_top = max(10, n // 10)

    while time.monotonic() - t0 < time_limit:
        # Compute convolution and identify peaks
        b, _ = convolve_and_subgradient(a, np.zeros(2 * n - 1))
        b_max = b.max()

        # Select active peaks (exact or near-exact)
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        top_vals = b[top_idx]
        # Use peaks within 1% of max for robustness
        threshold = b_max * 0.999
        active_mask = top_vals >= threshold
        if not active_mask.any():
            active_mask[0] = True

        w = np.zeros(2 * n - 1)
        active_indices = top_idx[active_mask]
        w[active_indices] = 1.0 / len(active_indices)

        # Compute subgradient via FFT
        _, g = convolve_and_subgradient(a, w)

        # Nesterov momentum step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        # Evaluate new point
        obj_new = eval_local_max_b(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it_count += 1

        # Adaptive strategy
        if no_improve >= restart_interval:
            # Restart with perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.05)
            no_improve = 0
            kick_size *= 0.7
        elif no_improve >= restart_interval // 2:
            cur_lr *= decel_factor
        elif no_improve >= restart_interval // 4:
            cur_lr *= 0.97

    return best_a, best_obj


def lp_refine(f, max_iters=30, time_limit=60.0, verbose=False):
    """
    LP-based refinement: solve max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Blend with current f to reduce max convolution.
    Returns (best_a, best_max_b) where a is normalized to sum=1.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_local_max_b(a)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = fast_convolve(a).max()

        # Build LP constraint matrix
        # Row k: sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j]*g[k-j] <= M
        num_constraints = 2 * n - 1
        A = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[k, k - j] = a[j]

        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)  # minimize -sum(g)

        try:
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            # Line search for blending
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj_blend = eval_local_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t <= 0.0:
                # Try very small t
                for t in [0.01, 0.02, 0.03]:
                    blend = (1.0 - t) * best_a + t * g_norm
                    blend = np.maximum(blend, 0.0)
                    blend = blend / blend.sum()
                    obj_blend = eval_local_max_b(blend)
                    if obj_blend < best_obj - 1e-13:
                        best_t = t
                        best_blend_obj = obj_blend
                        break

            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = eval_local_max_b(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.2f}, max_b={best_obj:.10f}")
                else:
                    a = best_a.copy()
            else:
                break

        except Exception:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """
    Coordinate descent on the simplex: scale one element, rescale others.
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_local_max_b(a)

    if multipliers is None:
        multipliers = np.array([0.2, 0.4, 0.6, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0, 5.0])

    improved_global = True
    pass_count = 0
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        pass_count += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 20.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_local_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    """Gaussian window initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    """Raised cosine initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    """Tukey window initialization."""
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_triangular_init(n):
    """Triangular (Bartlett) window."""
    a = np.ones(n)
    half = n // 2
    a[:half] = np.linspace(0.1, 1.0, half)
    a[half:] = np.linspace(1.0, 0.1, n - half)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

            if verbose:

    # Phase B: LP refinement

            print(f"  CD: n={n_cd}, t={t_cd:.0f}s")
            a_cd, obj_cd = coordinate_descent(best_seq, time_limit=t_cd)
            update_best(a_cd, "cd")

    # === Phase 6: Final micro-polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Micro-polish (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_final = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            # Very fine subgradient
            t_micro = min(rem * 0.5, 25.0)
            a_micro, obj_micro = subgradient_optimize_fft(
                best_seq, time_limit=t_micro,
                lr_init=0.0003, momentum=0.96,
                n_top=max(30, n_final // 6), seed=seed + 777777
            )
            update_best(a_micro, "micro_sg")

            # Quick LP
            rem = deadline - time.monotonic() - 10
            if rem > 5:
                a_micro_lp, obj_micro_lp = lp_refine(
                    best_seq, max_iters=10, time_limit=min(rem, 15.0)
                )
                update_best(a_micro_lp, "micro_lp")

            # Tiny CD
            rem = deadline - time.monotonic() - 5
            if rem > 5:
                a_micro_cd, obj_micro_cd = coordinate_descent(
                    best_seq, time_limit=rem,
                    multipliers=np.array([0.7, 0.85, 0.95, 1.05, 1.15, 1.3])
                )
                update_best(a_micro_cd, "micro_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
