def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    k_top = max(10, int(len_c * top_frac))
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[peaks]

    scaled = temperature * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    smooth_val = conv[peaks].max() / temperature + np.log(ws) / temperature

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[k - idx] + f[idx]
        sg[js:je + 1] += w[i] * contrib

    return smooth_val, sg


def momentum_projected_gradient(f0, time_limit, verbose=False,
                                 lr_init=0.05, momentum_init=0.9,
                                 temp_init=5.0, temp_decay=1.005,
                                 top_frac=0.1):
    """
    Nesterov momentum + projected gradient descent on simplex.
    Uses smooth max approximation with increasing temperature.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)  # momentum velocity
    t0 = time.monotonic()
    lr = lr_init
    mu = momentum_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov look-ahead
        y = f + mu * v
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys

        # Compute smooth max gradient at y
        smooth_val, sg = compute_smooth_max_grad(y, temperature=temp, top_frac=top_frac)

        # Gradient step
        v = mu * v - lr * sg
        f_new = y + v
        f_new = project_simplex(f_new)

        # Evaluate
        obj_new = fast_convolve(f_new).max()

        # Backtracking if needed
        if obj_new > best_obj * 1.02:
            for damp in [0.5, 0.25, 0.1]:
                f_trial = f + damp * (f_new - f)
                f_trial = np.maximum(f_trial, 0)
                ts = f_trial.sum()
                if ts < 1e-15:
                    continue
                f_trial = f_trial / ts
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < best_obj:
                    f_new = f_trial
                    obj_new = obj_trial
                    break
            else:
                lr *= 0.5
                v *= 0.5
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            # Slowly increase learning rate
            lr = min(lr * 1.01, 0.1)
            if verbose and it % 100 == 0:
                print(f"      MPGD iter {it}: obj={best_obj:.10f}, lr={lr:.4f}, temp={temp:.1f}")
        else:
            no_improve += 1

        # Adapt temperature (increase to focus on true max)
        temp *= temp_decay

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 7)
            noise = rng.standard_normal(n) * 0.002
            f = project_simplex(best_f + noise)
            v = np.zeros(n)
            lr = lr_init * 0.5
            temp = max(temp_init, temp * 0.5)
            no_improve = 0

        if lr < 1e-8:
            break

    return best_f, best_obj


def smooth_mirror_descent(f0, time_limit, verbose=False,
                          tau_init=5.0, lr_init=0.01, tau_decay=1.003,
                          lr_decay=0.9998, restart_every=150):
    """Adaptive smoothed multiplicative mirror descent."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        len_c = len(conv)
        k_top = max(10, int(len_c * 0.15))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            sg[js:je + 1] += w[i] * (f[k - idx] + f[idx])

        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.03:
            for damp in [0.5, 0.25, 0.1]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    break
            else:
                lr *= 0.5
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"      SMD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        tau *= tau_decay
        lr *= lr_decay

        if no_improve > restart_every:
            rng = np.random.default_rng(it * 7 + 13)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.3
            tau = tau_init
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj


def greedy_peak_shaving(f0, time_limit, verbose=False, top_k=5):
    """
    Greedy combinatorial peak shaving: for the worst peak, identify which
    f[j] values contribute most and reduce them selectively.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find top-k peaks
        peak_indices = np.argsort(conv)[-top_k:]
        peak_indices = [int(k) for k in peak_indices if conv[k] > M * 0.8]

        if not peak_indices:
            peak_indices = [int(np.argmax(conv))]

        # For each peak, compute contribution of each f[j]
        contrib = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            # Contribution of f[j] to peak k is f[k-j] (the other factor)
            contrib[js:je + 1] += f[k - idx]

        # Find the j with highest contribution (most "costly")
        # But also consider how reducing it affects other peaks
        j_candidates = np.where(contrib > contrib.max() * 0.5)[0]
        if len(j_candidates) == 0:
            j_candidates = [np.argmax(contrib)]

        best_j = -1
        best_delta = 0
        best_val = M

        for j in j_candidates[:20]:  # Limit to top 20 candidates
            if f[j] < 1e-10:
                continue
            # Try reducing f[j] by various amounts
            for frac in [0.01, 0.02, 0.05, 0.1, 0.15]:
                delta = f[j] * frac
                if delta < 1e-12:
                    continue
                f_trial = f.copy()
                f_trial[j] -= delta
                f_trial[j] = max(f_trial[j], 0)
                # Redistribute to maintain sum
                deficit = delta
                other_mask = np.ones(n, dtype=bool)
                other_mask[j] = False
                other_sum = f_trial[other_mask].sum()
                if other_sum < 1e-15:
                    continue
                f_trial[other_mask] += deficit * (f_trial[other_mask] / other_sum)
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_j = j
                    best_delta = delta
                    best_trial = f_trial.copy()

        if best_j >= 0:
            f = best_trial
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                if verbose and it % 50 == 0:
                    print(f"      GPS iter {it}: obj={best_obj:.10f}")
        else:
            # No improvement found, try smaller perturbations or restart
            if it % 100 == 0:
                rng = np.random.default_rng(it * 3 + 1)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 0)
                f = f / f.sum()
            if it > 500:
                break

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.3
            if m_exp > 20.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.2
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 100:
                m_exp = min(m_exp * 1.2, 10.0)
            if no_improve > 300:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Fine line search
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(5):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 10.0)

        if no_improve > 300:
            break

    return best_f, best_obj


def coordinate_descent_fine(f0, time_limit, verbose=False):
    """Coordinate descent with local 1D optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        improved = False
        rng = np.random.default_rng(epoch * 1000 + 42)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            deltas = [old_val * d for d in [-0.2, -0.1, -0.05, 0.05, 0.1, 0.2, 0.3, -0.3]]
            for delta in deltas:
                new_val = old_val + delta
                if new_val < 0 or new_val > 0.3:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def make_asymmetric_init(n, variant=0, seed=42):
    """Generate asymmetric initial sequences with many variants."""
    rng = np.random.default_rng(seed + variant * 137)
    x = np.arange(n) / n

    if variant == 0:
        a = np.exp(-x / 0.15)
    elif variant == 1:
        a = np.exp(-(1 - x) / 0.15)
    elif variant == 2:
        a = np.exp(-x / 0.3) * (1 + 0.4 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif variant == 3:
        a = np.linspace(0.05, 1.0, n)
    elif variant == 4:
        a = np.exp(-30 * (x - 0.25)**2) + 0.6 * np.exp(-30 * (x - 0.75)**2)
    elif variant == 5:
        a = np.cos(np.pi * x / 2)**2 + 0.1
    elif variant == 6:
        a = (1 - x)**2 + 0.2 * x
    elif variant == 7:
        a = np.log1p(x * n) / np.log1p(n)
    elif variant == 8:
        a = (x + 0.1)**(-0.3)
    elif variant == 9:
        # Two-scale asymmetric
        a = np.exp(-x/0.2) * 0.7 + np.exp(-(x - 0.5)**2 / 0.05) * 0.3
    elif variant == 10:
        # Sigmoid ramp
        a = 1 / (1 + np.exp(-20 * (x - 0.3)))
    elif variant == 11:
        # Triangle asymmetric
        a = np.where(x < 0.4, x / 0.4, (1 - x) / 0.6)
        a = np.maximum(a, 0.01)
    elif variant == 12:
        # Gaussian mixture (asymmetric)
        a = np.exp(-(x - 0.2)**2 / 0.02) + 0.5 * np.exp(-(x - 0.8)**2 / 0.08)
    else:
        a = np.ones(n) + rng.standard_normal(n) * 0.3
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining momentum-projected gradient,
    smoothed mirror descent, greedy peak shaving, LP-Frank-Wolfe,
    coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    init_candidates = []

    # Try lengths around parent
    if parent is not None:
        parent_n = len(parent)
        test_lengths = [max(256, parent_n - 128), parent_n, parent_n + 128,
                        max(256, parent_n - 64), parent_n + 64,
                        parent_n * 2 // 3, parent_n * 4 // 3]
    else:
        test_lengths = [512, 768, 1024, 1536]

    test_lengths = sorted(set(max(128, int(l)) for l in test_lengths))[:6]

    for n_try in test_lengths:
        for v in range(12):
            a_init = make_asymmetric_init(n_try, variant=v, seed=seed)
            init_candidates.append((a_init, f"asym_{v}_n{n_try}"))

    if parent is not None:
        init_candidates.append((parent, "parent"))

    # Score initializations by fast_convolve objective
    scored_inits = []
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        scored_inits.append((obj, a_norm, label))

    scored_inits.sort(key=lambda x: x[0])
    top_inits = scored_inits[:6]
    print(f"  Top inits: {[f'{l}:{o:.4f}' for o, _, l in top_inits]}")

    # Quick SMD on top initializations to get better starting points
    print("\n=== Phase 2: Quick SMD on Top Inits ===")
    refined_inits = []
    for idx, (obj_init, a_init, label) in enumerate(top_inits[:4]):
        rem = deadline - time.monotonic() - 800
        if rem < 20:
            break
        t_alloc = min(rem / max(1, 4 - idx), 60)
        if t_alloc < 10:
            t_alloc = 10
        print(f"  SMD {label} (n={len(a_init)}): {t_alloc:.0f}s")
        a_opt, obj = smooth_mirror_descent(a_init, time_limit=t_alloc)
        sc = try_seq(a_opt, f"smd_quick_{label}")
        refined_inits.append((a_opt, obj, label))

    # Also try MPGD on top inits
    print("\n=== Phase 3: Momentum Projected Gradient ===")
    for idx, (a_init, obj_init, label) in enumerate(top_inits[:3]):
        rem = deadline - time.monotonic() - 600
        if rem < 20:
            break
        t_alloc = min(rem / max(1, 3 - idx), 100)
        if t_alloc < 15:
            t_alloc = 15
        print(f"  MPGD {label} (n={len(a_init)}): {t_alloc:.0f}s")
        a_opt, obj = momentum_projected_gradient(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"mpgd_{label}")

    # === Phase 4: Greedy Peak Shaving on best ===
    print("\n=== Phase 4: Greedy Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  GPS (n={len(best_seq)}): {t_alloc:.0f}s")
            a_opt, obj = greedy_peak_shaving(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "gps")

    # === Phase 5: LP-Frank-Wolfe on best ===
    print("\n=== Phase 5: LP-Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_alloc = min(rem, 180)
            print(f"  LP-FW (n={len(best_seq)}): {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        base_n = len(base)
        target_lengths = [l for l in test_lengths if abs(l - base_n) > 32]
        # Also try some nearby lengths
        target_lengths.extend([base_n + 32, base_n - 32, base_n + 64, base_n - 64])
        target_lengths = sorted(set(l for l in target_lengths if l >= 128))[:4]

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.35, 80)
                if t_alloc < 15:
                    t_alloc = 15
                print(f"  SMD n={target_n}: {t_alloc:.0f}s")
                a_opt, obj = smooth_mirror_descent(f_interp, time_limit=t_alloc)
                try_seq(a_opt, f"smd_n{target_n}")

    # === Phase 7: MPGD on current best ===
    print("\n=== Phase 7: MPGD Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  MPGD best (n={len(best_seq)}): {t_alloc:.0f}s")
            a_opt, obj = momentum_projected_gradient(best_seq, time_limit=t_alloc,
                                                      lr_init=0.02, temp_init=10.0,
                                                      temp_decay=1.002, verbose=True)
            try_seq(a_opt, "mpgd_refine")

    # === Phase 8: Squeeze with SMD + LP-FW ===
    print("\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 40:
            t1 = min(rem * 0.5, 60)
            print(f"  SMD squeeze: {t1:.0f}s")
            a_opt, obj = smooth_mirror_descent(best_seq, time_limit=t1,
                                               tau_init=20.0, lr_init=0.005,
                                               tau_decay=1.002, restart_every=80)
            try_seq(a_opt, "smd_squeeze")

        rem = deadline - time.monotonic() - 60
        if rem > 30 and best_seq is not None:
            t2 = min(rem, 60)
            print(f"  LP-FW squeeze: {t2:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t2)
            try_seq(a_opt, "lpfw_squeeze")

    # === Phase 9: Coordinate Descent ===
    print("\n=== Phase 9: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  CD (n={len(best_seq)}): {t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_fine(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 10: Final restarts ===
    print("\n=== Phase 10: Final Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            t_alloc = min(rem / max(1, 3 - r), 40)
            if t_alloc < 10:
                t_alloc = 10

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
            elif r == 1:
                # Blend with reversed
                f_init = 0.6 * best_seq + 0.4 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Blend with random asymmetric
                asym = make_asymmetric_init(n_best, variant=rng.integers(0, 13), seed=seed + r)
                f_init = 0.8 * best_seq + 0.2 * asym
                f_init = f_init / f_init.sum()

            print(f"  Restart {r}: {t_alloc:.0f}s")
            a_opt, obj = smooth_mirror_descent(f_init, time_limit=t_alloc,
                                               tau_init=10.0, lr_init=0.005,
                                               tau_decay=1.003, restart_every=60)
            try_seq(a_opt, f"restart_{r}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
