If we define b_k = conv(a,a)[k], then by the convolution theorem:

In the frequency domain: minimize max(b_k) subject to a >= 0, A(0) = 1.

This is equivalent to: minimize max(b_k) subject to a >= 0, sum(a) = 1.

The b_k are the inverse DFT of |A(f)|^2. So we want |A(f)|^2 to be as flat as possible.

A spectral approach:
1. Start with some a (nonnegative, sum=1)
2. Compute A(f) = DFT(a)
3. Compute P(f) = |A(f)|^2
4. The peak of the inverse DFT of P(f) corresponds to the max of b_k
5. To reduce the peak, we want to reduce P(f) at frequencies that contribute most to the peak
6. Update: A_new(f) = A(f) / sqrt(P(f) + eps) (this flattens the spectrum)
7. Project back to nonnegative: a_new = max(Re(IDFT(A_new)), 0), renormalize

This is similar to the "Gerchberg-Saxton" or "error reduction" algorithms used in phase retrieval and X-ray crystallography.

```python
def spectral_flatten(a, n_iters=100, eps=1e-6):
    """Flatten the power spectrum to reduce auto-correlation peak."""
    a = a.copy()
    n = len(a)
    fft_size = 1 << (2*n-2).bit_length()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A)**2
        # The inverse DFT of P gives the auto-correlation
        # We want to reduce the peak, so we "compress" P
        # Using: P_new = P^(alpha) for alpha < 1 compresses the dynamic range
        # Or: P_new = P / (P + threshold)
        
        # Method: iterative thresholding
        # Find the current peak
        b = np.fft.irfft(A * A, fft_size)[:2*n-1]
        peak = b.max()
        
        # Reduce high-power frequencies
        # A_new = A * exp(-beta * contribution_to_peak)
        # The contribution of frequency f to conv[k] is |A(f)|^2 * exp(2πifk/fft_size)
        # For the peak lag k*, the contribution is proportional to |A(f)|^2 * exp(2πifk*/fft_size)
        
        k_star = np.argmax(b)
        # Reduce |A(f)|^2 where it contributes to the peak
        phase_contrib = np.exp(2j * 2*np.pi * k_star * np.arange(len(A)) / fft_size)
        contrib = np.abs(A)**2 * np.real(phase_contrib)
        contrib = np.maximum(contrib, 0)
        
        # Update
        beta = 0.1
        A_new = A * np.exp(-beta * contrib / (peak + eps))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
        a = a_new
    
    return a
```

def spectral_flatten(a, n_iters=50, beta_decay=0.995, verbose=False):
    """Flatten power spectrum to reduce auto-correlation peak."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a / max(len(a), 1)
    a = a / s
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    beta = 0.3

    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        # Current power spectrum
        P = np.abs(A) ** 2
        # The auto-correlation is IFFT of P
        # We want to reduce the peak of IFFT(P)
        # Strategy: reduce P at frequencies that contribute most to the peak

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Contribution of each frequency to the peak
        freqs = np.arange(len(A))
        phase = np.exp(2j * np.pi * peak_k * freqs / fft_size)
        contrib = P * np.real(phase)
        contrib = np.maximum(contrib, 0)

        # Update: reduce high-contribution frequencies
        A_new = A * np.exp(-beta * contrib / (peak_val * len(A) + 1e-10))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-10:
            break
        a_new = a_new / sn

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Accept with some probability (simulated annealing)
            if np.random.default_rng(it).random() < 0.3:
                a = a_new
            else:
                a = best_a.copy()

        beta *= beta_decay

    return best_a

def multiplicative_w descent(f0, time_limit, verbose=False):
    """Multiplicative weights / mirror descent with peak-aware subgradients."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Use top-K peaks for subgradient
        K = min(10, len(conv))
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_vals = conv[peak_indices]
        # Soft weights
        temp = M * 0.1
        weights = np.exp(peak_vals / temp)
        weights /= weights.sum()

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(K):
            k = int(peak_indices[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * 2.0 * f[k - idx]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 20:
            lr *= 0.9
            if no_improve > 100:
                # Restart with perturbation
                rng = np.random.default_rng(it * 13)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 1e-12)
                f = f / f.sum()
                lr = 0.01
                no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):

    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def bundle_subgradient_descent(a, time_limit, verbose=False):
    """Bundle method: maintain top peaks and use barycenter subgradient."""
    f = np.maximum(a.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Top 5 peaks
        K = min(5, len(conv))
        peak_indices = np.argsort(conv)[-K:]
        peak_vals = conv[peak_indices]

        # Softmax weights (temperature-based)
        temp = 0.05 * M
        exp_vals = np.exp((peak_vals - peak_vals.max()) / temp)
        weights = exp_vals / exp_vals.sum()

        # Weighted subgradient
        sg = np.zeros(n)
        for i in range(K):
            k = int(peak_indices[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * 2.0 * f[k - idx]

        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        # Additive update with projection
        step = lr * sg / (M + 1e-10)
        f_new = f - step
        f_new = np.maximum(f_new, 1e-12)
        f_new = f_new / f_new.sum()

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    BSD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            # If no improvement, try smaller step
            if no_improve % 5 == 0:
                lr *= 0.95

        if no_improve > 150:
            rng = np.random.default_rng(it * 41 + 3)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-12)
            f = f / f.sum()
            lr = 0.02
            no_improve = 0

        if lr < 1e-6:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines LP-Frank-Wolfe, spectral flattening, bundle subgradient descent,
    and peak-targeted coordinate descent with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception as e:
        print(f"  No parent available: {e}")

    if parent is None:
        parent = make_gaussian(768, 0.25)
        try_seq(parent, "default")

    base_n = len(parent) if parent is not None else 768

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Initializations ===")
    t0 = time.monotonic()

    init_list = []
    for n_try in [512, 640, 768, 896, 1024]:
        init_list.append((f'g{n_try}', make_gaussian(n_try, 0.25)))
        init_list.append((f'b{n_try}', make_bell(n_try, 0.5)))
        init_list.append((f'c{n_try}', make_cosine_mod(n_try, 1, 0.2)))
        init_list.append((f'm{n_try}', make_multi_cosine(n_try, 4, 0.2, seed=n_try)))

    # Evaluate initializations quickly
    scored_inits = []
    for label, a_init in init_list:
        sc = objective_normalized(a_init)
        scored_inits.append((sc, a_init, label))
    scored_inits.sort(key=lambda x: x[0])

    # Try a few with evaluate_sequence
    for sc, a_init, label in scored_inits[:5]:
        try_seq(a_init, f"init_{label}")

    # Add parent
    if parent is not None:
        sc_parent = objective_normalized(parent)
        scored_inits.insert(0, (sc_parent, parent, "parent"))

    print(f"  Best init: {scored_inits[0][2]}, obj={scored_inits[0][0]:.6f}")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP-FW on best initializations ===
    print("\n=== Phase 2: LP-Frank-Wolfe ===")
    rem = deadline - time.monotonic() - 400
    if rem > 60:
        for trial_idx in range(min(3, len(scored_inits))):
            rem = deadline - time.monotonic() - 350
            if rem < 30:
                break
            sc_init, a_init, label = scored_inits[trial_idx]
            t_alloc = min(rem * 0.5, 250)
            print(f"  LP-FW [{label}] n={len(a_init)}: alloc={t_alloc:.0f}s (init_obj={sc_init:.6f})")
            a_opt, obj = lp_fw_optimize(a_init, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"lpfw_{label}")

    # === Phase 3: Spectral flattening on best ===
    print("\n=== Phase 3: Spectral Flattening ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_alloc = min(rem * 0.3, 60)
            print(f"  Spectral: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            # Convert to normalized for spectral method
            a_norm = best_seq / best_seq.sum()
            a_spec = spectral_flatten(a_norm, n_iters=100, beta_decay=0.998, verbose=True)
            # Scale back to original scale
            a_result = a_spec * best_seq.sum()
            try_seq(a_result, "spectral")

    # === Phase 4: Bundle subgradient descent ===
    print("\n=== Phase 4: Bundle Subgradient ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.4, 80)
            print(f"  BSD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_norm = best_seq / best_seq.sum()
            a_opt, obj = bundle_subgradient_descent(a_norm, time_limit=t_alloc, verbose=True)
            a_result = a_opt * best_seq.sum()
            try_seq(a_result, "bsd")

    # === Phase 5: Peak coordinate descent ===
    print("\n=== Phase 5: Peak Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_alloc = min(rem * 0.5, 60)
            print(f"  PCD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_norm = best_seq / best_seq.sum()
            a_opt, obj = peak_coordinate_descent(a_norm, time_limit=t_alloc, verbose=True)
            a_result = a_opt * best_seq.sum()
            try_seq(a_result, "pcd")

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length ===")
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [640, 512, 896, 1024, 800, 700]:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq / best_seq.sum(), target_n)
            sc = try_seq(f_interp * best_seq.sum(), f"interp_n{target_n}")
            if sc < best_score + 0.02:
                # Optimize at this length
                rem = deadline - time.monotonic() - 60
                if rem > 20:
                    t_alloc = min(rem * 0.3, 40)
                    a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc)
                    try_seq(a_opt * best_seq.sum(), f"lpfw_n{target_n}")

    # === Phase 7: LP-FW with perturbations ===
    print("\n=== Phase 7: Perturbed LP-FW ===")
    if best_seq is not None:
        n_best = len(best_seq)
        a_norm = best_seq / best_seq.sum()
        perturbations = []

        for shift in [1, 2, 3, 5, 8]:
            perturbations.append(('shift_{}'.format(shift), np.roll(a_norm, shift)))

        perturbations.append(('flip', a_norm[::-1].copy()))

        for mix in [0.05, 0.1, 0.2]:
            perturbations.append(('mix_{}'.format(mix),
                                  (1 - mix) * a_norm + mix * (np.ones(n_best) / n_best)))

        for amp in [0.002, 0.005, 0.01]:
            noise = np.random.default_rng(seed + int(amp * 10000)).standard_normal(n_best) * amp
            perturbations.append(('noise_{}'.format(amp), np.maximum(a_norm + noise, 0)))
            perturbations[-1] = (perturbations[-1][0], perturbations[-1][1] / max(perturbations[-1][1].sum(), 1e-15))

        for label, f_pert in perturbations:
            rem = deadline - time.monotonic() - 40
            if rem < 10:
                break
            t_alloc = min(rem * 0.25, 30)
            if t_alloc < 5:
                break
            a_opt, obj = lp_fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt * best_seq.sum(), f"pert_{label}")

    # === Phase 8: Final multiplicative descent ===
    print("\n=== Phase 8: Final MW Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem * 0.6, 30)
            a_norm = best_seq / best_seq.sum()
            a_opt, obj = multiplicative_w_descent(a_norm, time_limit=t_alloc, verbose=True)
            try_seq(a_opt * best_seq.sum(), "mw_final")

    # === Phase 9: Very fine LP-FW polish ===
    print("\n=== Phase 9: Final LP-FW Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            t_alloc = min(rem, 12)
            a_norm = best_seq / best_seq.sum()
            a_opt, obj = lp_fw_optimize(a_norm, time_limit=t_alloc)
            try_seq(a_opt * best_seq.sum(), "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - (deadline - max(1.0, float(budget_s) - 30.0)):.1f}s")

    return [float(x) for x in best_seq]
