"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using ramp-based initialization and
accelerated mirror descent with multi-peak subgradients, adaptive temperature,
and multi-length exploration. Key insight: non-uniform (ramp/exponential) 
sequences dramatically outperform uniform-like ones for n > 50.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for a (assumes sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def make_ramp(n):
    """Linear ramp: a[i] = i+1, normalized."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a /= a.sum()
    return a


def make_power_law(n, p):
    """Power law: a[i] = (i+1)^p, normalized."""
    a = np.arange(1, n + 1, dtype=np.float64) ** p
    a /= a.sum()
    return a


def make_exponential(n, rate):
    """Exponential ramp: a[i] = exp(rate * i / n), normalized."""
    a = np.exp(rate * np.arange(n) / n)
    a /= a.sum()
    return a


def make_gaussian_asym(n, center_frac=0.7, sigma_frac=0.25):
    """Asymmetric Gaussian centered away from middle."""
    x = np.arange(n) / n
    center = center_frac
    a = np.exp(-0.5 * ((x - center) / sigma_frac) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_ramp(n, freq=1, amp=0.3):
    """Ramp modulated by cosine."""
    base = np.arange(1, n + 1, dtype=np.float64)
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
    a = base * mod
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    temperature = temp_init
    it = 0

    # Momentum (Nesterov in mirror space)
    f_mom = f.copy()
    momentum = 0.9

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Get top peaks with softmax weighting
        k_actual = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient (weighted combination)
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Nesterov: evaluate at lookahead point
        f_look = f + momentum * (f - f_mom)
        f_look = np.maximum(f_look, 1e-15)
        f_look = f_look / f_look.sum()

        # Recompute subgradient at lookahead
        conv_look = fast_convolve(f_look)
        M_look = conv_look.max()
        peaks_look = np.argpartition(conv_look, -k_actual)[-k_actual:]
        pv_look = conv_look[peaks_look]
        scaled_look = pv_look / temperature
        scaled_look -= scaled_look.max()
        w_look = np.exp(scaled_look)
        ws_look = w_look.sum()
        if ws_look < 1e-15:
            w_look = np.zeros(len(peaks_look))
            w_look[np.argmax(pv_look)] = 1.0
        else:
            w_look /= ws_look

        sg_look = np.zeros(n)
        for i in range(len(peaks_look)):
            if w_look[i] < 1e-12:
                continue
            k = int(peaks_look[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg_look[idx] += 2.0 * w_look[i] * f_look[k - idx]

        # Mirror descent step with backtracking
        best_trial = None
        best_trial_obj = M
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 2]:
            f_try = f_look * np.exp(-lr_try * sg_look)
            f_try = np.maximum(f_try, 1e-15)
            f_try = f_try / f_try.sum()
            obj_try = fast_convolve(f_try).max()
            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_trial = f_try
                lr = lr_try

        if best_trial is None:
            f_try = f * np.exp(-lr * 0.5 * sg)
            f_try = np.maximum(f_try, 1e-15)
            f_try = f_try / f_try.sum()
            best_trial = f_try
            best_trial_obj = fast_convolve(f_try).max()

        # Update momentum
        f_mom = f.copy()
        f = best_trial

        co = best_trial_obj
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            no_improve += 1

        # Temperature annealing
        temperature *= temp_growth

        # Periodic restart to escape local minima
        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            noise_type = it % 4
            if noise_type == 0:
                f = best_f + rng.standard_normal(n) * 0.005
            elif noise_type == 1:
                shift = rng.integers(1, max(2, n // 16))
                f = np.roll(best_f, shift)
            elif noise_type == 2:
                # Blend with ramp
                ramp = make_ramp(n)
                f = 0.8 * best_f + 0.2 * ramp
            else:
                # Scale perturbation
                f = best_f * np.exp(rng.standard_normal(n) * 0.01)
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            f_mom = f.copy()
            lr = max(lr_init * 0.5, lr * 2)
            no_improve = 0
            temperature = temp_init * 2

        # Decay learning rate slowly
        lr *= 0.9999
        if lr < 1e-10:
            break

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Targeted coordinate descent: identifies top peaks and performs
    mass redistribution to reduce them.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 80:
            # Try a perturbation
            rng = np.random.default_rng(it * 43 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            no_improve = 0
            if it > 5000:
                break

        conv = fast_convolve(f)
        M = conv.max()

        # Get top 5 peaks
        top_peaks = np.argpartition(conv, -5)[-5:]
        found_improvement = False

        for peak_k in top_peaks:
            peak_k = int(peak_k)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            contrib_idx = np.arange(js, je + 1)
            if len(contrib_idx) == 0:
                continue

            products = f[contrib_idx] * f[peak_k - contrib_idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Find top-3 contributors
            top_contrib = np.argsort(products)[-3:]
            for tc in top_contrib:
                src = contrib_idx[tc]
                partner = peak_k - src

                if src == partner:
                    continue

                # Reduce source, add to non-contributing coords
                for frac in [0.05, 0.1, 0.2, 0.3, 0.4]:
                    diff = f[src] * frac
                    if diff < 1e-15:
                        continue
                    trial = f.copy()
                    trial[src] -= diff
                    # Add to partner's "opposite" region (away from this peak)
                    # Non-contributing indices
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[contrib_idx] = False
                    if non_contrib.sum() < 3:
                        continue
                    # Add proportionally to existing values
                    nc_mask = non_contrib
                    nc_vals = trial[nc_mask]
                    nc_sum = nc_vals.sum()
                    if nc_sum < 1e-15:
                        continue
                    trial[nc_mask] += diff * (nc_vals + 1e-15) / (nc_sum + 1e-15)
                    trial = np.maximum(trial, 1e-15)
                    ts = trial.sum()
                    trial = trial / ts
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        found_improvement = True
                        break
                if found_improvement:
                    break
            if found_improvement:
                break

        if not found_improvement:
            no_improve += 1
        else:
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    CD iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def lp_direction_solve(f, M, n, max_iter=50):
    """
    Simple iterative LP-like direction: find g that maximizes sum(g)
    subject to (f*g)[k] <= M for all k, g >= 0.
    Uses projected gradient as approximation.
    """
    # Start with uniform
    g = np.ones(n) / n
    for _ in range(max_iter):
        # Compute violations
        fg = np.convolve(f, g)
        violations = np.maximum(fg - M, 0)
        # Gradient of constraint violation
        grad = np.zeros(n)
        for k in range(len(fg)):
            if violations[k] > 0:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad[idx] += violations[k] * f[k - idx]
        # Update: move away from violations
        g = g - 0.1 * grad
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
    return g


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Parametric initialization (ramp, exponential, power law)
    2. Accelerated mirror descent
    3. Coordinate descent polish
    4. Multi-length exploration
    5. Final squeeze
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Parametric Initialization ===
    print("\n=== Phase 1: Parametric Initialization ===")
    t0 = time.monotonic()

    n_values = [512, 640, 768, 896, 1024, 1280]
    init_candidates = []

    for n_try in n_values:
        # Ramp family
        init_candidates.append((f"ramp_{n_try}", make_ramp(n_try)))
        init_candidates.append((f"pow0.5_{n_try}", make_power_law(n_try, 0.5)))
        init_candidates.append((f"pow1.5_{n_try}", make_power_law(n_try, 1.5)))
        init_candidates.append((f"exp1_{n_try}", make_exponential(n_try, 1.0)))
        init_candidates.append((f"exp2_{n_try}", make_exponential(n_try, 2.0)))
        init_candidates.append((f"exp3_{n_try}", make_exponential(n_try, 3.0)))
        init_candidates.append((f"gauss_asym_{n_try}", make_gaussian_asym(n_try, 0.7, 0.25)))
        init_candidates.append((f"cos_ramp_{n_try}", make_cosine_ramp(n_try, 1, 0.3)))
        init_candidates.append((f"cos_ramp2_{n_try}", make_cosine_ramp(n_try, 2, 0.2)))

    # Add parent
    if parent is not None:
        init_candidates.append(('parent', parent))

    # Evaluate all
    best_init = None
    best_init_obj = float('inf')
    for label, a_init in init_candidates:
        obj = objective_normalized(a_init)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_init
        if obj < 1.6:
            try_seq(a_init, f"init_{label}")

    print(f"  Best init: obj_norm={best_init_obj:.6f}")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Main Mirror Descent Optimization ===
    print("\n=== Phase 2: Accelerated Mirror Descent ===")
    rem = deadline - time.monotonic() - 400

    # Sort and pick top candidates
    scored_inits = []
    for label, a_init in init_candidates:
        obj = objective_normalized(a_init)
        scored_inits.append((obj, a_init, label))
    scored_inits.sort(key=lambda x: x[0])

    # Optimize top 3 with mirror descent
    for trial_idx in range(min(3, len(scored_inits))):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        obj_init, a_init, label = scored_inits[trial_idx]
        t_alloc = min(rem * 0.5, 400)
        print(f"  MD [{label}] n={len(a_init)}: alloc={t_alloc:.0f}s (init_obj={obj_init:.6f})")
        a_opt, obj = accelerated_mirror_descent(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"md_{label}")

    # Also optimize the best found so far
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_alloc = min(rem * 0.4, 200)
            print(f"  MD [best_seq] n={len(best_seq)}: alloc={t_alloc:.0f}s")
            a_opt, obj = accelerated_mirror_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "md_best")

    # === Phase 3: Coordinate Descent Polish ===
    print("\n=== Phase 3: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.5, 120)
            print(f"  CD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 4: Multi-Length Exploration ===
    print("\n=== Phase 4: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_ns = [512, 640, 768, 896, 1024, 1280, 1536]
        for target_n in target_ns:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            # Also try ramp at this length
            ramp_n = make_ramp(target_n)
            try_seq(ramp_n, f"ramp_n{target_n}")

            # If ramp is good, optimize it
            ramp_obj = objective_normalized(ramp_n)
            if ramp_obj < 1.5:
                t_alloc = min(rem * 0.3, 60)
                if t_alloc > 10:
                    a_opt, obj = accelerated_mirror_descent(ramp_n, time_limit=t_alloc)
                    try_seq(a_opt, f"md_ramp_n{target_n}")

    # === Phase 5: Combined Polish (MD + CD alternating) ===
    print("\n=== Phase 5: Combined Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        # Alternate MD and CD
        for round_idx in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            t_md = min(rem * 0.4, 60)
            if t_md > 5:
                a_opt, obj = accelerated_mirror_descent(best_seq, time_limit=t_md,
                                                        lr_init=0.003, n_peaks=10,
                                                        temp_init=0.0005)
                try_seq(a_opt, f"md_polish{round_idx}")

            rem = deadline - time.monotonic() - 10
            if rem < 5:
                break
            t_cd = min(rem * 0.4, 40)
            if t_cd > 3:
                a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_cd)
                try_seq(a_opt, f"cd_polish{round_idx}")

    # === Phase 6: Final Squeeze with varied params ===
    print("\n=== Phase 6: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            # Try with different hyperparameters
            t_alloc = min(rem, 20)
            a_opt, obj = accelerated_mirror_descent(best_seq, time_limit=t_alloc,
                                                    lr_init=0.001, n_peaks=50,
                                                    temp_init=0.0001, temp_growth=1.01)
            try_seq(a_opt, "final_squeeze")

    # Fallback
    if best_seq is None:
        best_seq = make_ramp(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    total_time = time.monotonic() - (deadline - max(1.0, float(budget_s) - 20.0))
    print(f"Total time: {total_time:.1f}s")
    return [float(x) for x in best_seq]
