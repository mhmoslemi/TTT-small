import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def build_lp_matrix(f, n):
    """Build constraint matrix for LP: rows k, cols j, entry f[k-j] if valid."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def gradient_descent_peaks(f0, time_limit, verbose=False):

    Uses the analytical gradient: d(a*a)[k]/da_m = 2*a_{k-m}.

                    f = trial
                    improved = True
                    no_improve = 0

            # The gradient at the peak tells us which coordinate to reduce

def peak_equalization_cd(f0, time_limit, verbose=False):
    """
    Targeted coordinate descent that specifically equalizes the top peaks.
    For each top peak, identifies the top contributing (j, k-j) pairs and
    reduces them while redistributing to non-contributing positions.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k in peak_indices:
            k = int(k)
            if conv[k] < M * 0.8:
                continue

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Try reducing the top-3 contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(3, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-10:
                    continue

                        continue

                    if abs(d) < 1e-12:

                    trial = f.copy()
                    trial[j_max] = new_val

                        trial[non_contrib] += d * w
                    elif d < 0:

                        w = np.maximum(trial[non_contrib], 1e-10)
                        w = w / w.sum()
                        trial[non_contrib] += d * w

                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial = trial / ts

                    new_obj = 2.0 * n * fast_convolve(trial).max()
                    if new_obj < best_obj - 1e-13:

                        best_f = trial.copy()
                        f = trial

                        break

        if not improved:

            if no_improve % 10 == 0:

    best_obj = 2.0 * n * M

    no_improve = 0
    it = 0
    momentum_dir = np.zeros(n)
    momentum_coeff = 0.3

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP direction with slightly relaxed M
        m_relax = 1.0
        g = None
        for attempt in range(4):
            m_try = M * (1.0 + 0.05 * attempt)
            g = solve_lp_direction(f, m_try, n)
            if g is not None and g.sum() > 1e-10:
                break
            if time.monotonic() - t0 > time_limit:
                break
        if g is None or g.sum() < 1e-10:
            # Perturbation to escape
            if no_improve > 20:
                rng = np.random.default_rng(it * 17 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0

            if no_improve > 80:
                break
            continue

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            no_improve += 1
            continue
        g = g / gs

        # Apply momentum
        direction = momentum_coeff * momentum_dir + (1 - momentum_coeff) * (g - f)
        momentum_dir = direction.copy()

        # Backtracking line search
        alpha = 1.0
        improved = False
        for _ in range(8):

            f_new = np.maximum(f_new, 0.0)

                f = f_new
                M = new_M
                improved = True
                current_obj = 2.0 * n * M
                if current_obj < best_obj - 1e-14:
                    best_obj = current_obj
                    best_f = f.copy()
                    no_improve = 0
                break

            if no_improve > 60:

                g = solve_lp_direction(f, m_try, n)
                if g is not None and g.sum() > 1e-10:

                    g = g / g.sum()
                    direction = g - f
                    for alpha_try in [0.5, 0.3, 0.2, 0.1]:
                        f_new = np.maximum(f + alpha_try * direction, 0)
                        sn = f_new.sum()
                        if sn < 1e-15:
                            continue
                        f_new = f_new / sn

                        if new_M < M:

def multiplicative_descent(f0, time_limit, verbose=False):
    """Mirror descent with multiplicative updates targeting peaks."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    temperature = 0.003
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Weight peaks by their value
        n_peaks = min(20, len(conv))
        peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            partners = kk - idx
            sg[idx] += 2.0 * w[i] * f[partners]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = 2.0 * n * fast_convolve(f).max()
        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temperature *= 1.005

        if no_improve > 100:
            rng = np.random.default_rng(it * 37 + 3)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.003
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=3.0):
    """Asymmetric exponential decay."""
    x = np.arange(n)
    a = np.exp(-decay * x / n)
    a = a / a.sum()
    return a


def make_bimodal_asymmetric(n, center1=0.3, center2=0.7, width=0.1):
    """Bimodal asymmetric sequence."""
    x = np.arange(n) / n
    a = np.exp(-((x - center1) / width) ** 2) + 0.8 * np.exp(-((x - center2) / width) ** 2)
    a = a / a.sum()
    return a


def make_bell(n, width=0.7):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

    print(f"=== Phase 1: LP-FW on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem, 120)
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(parent, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw_p1")

    # === Phase 2: Gradient-based peak CD ===
    if best_seq is not None:
        print(f"=== Phase 2: Gradient CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_alloc = min(rem, 100)
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = gradient_descent_peaks(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "gradcd_p2")

    # === Phase 3: Peak equalization CD ===
    if best_seq is not None:
        print(f"=== Phase 3: Peak Eq CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = peak_equalization_cd(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "peqcd_p3")

    # === Phase 4: Multi-length exploration ===
    base = best_seq if best_seq is not None else parent
    print(f"=== Phase 4: Multi-length (base n={len(base)}) ===")
    target_lengths = [len(base) - 32, len(base) - 16, len(base) + 16, len(base) + 32,
                      len(base) - 64, len(base) + 64]
    target_lengths = [t for t in target_lengths if 100 <= t <= 2000]
    # Remove duplicates
    seen = set()
    unique_lengths = []
    for t in target_lengths:
        if t not in seen:
            seen.add(t)
            unique_lengths.append(t)

    for target_n in unique_lengths:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 150
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        best_seq, best_score = try_sequence(f_interp, best_seq, best_score, f"interp_n{target_n}")

        # Quick LP-FW on interpolated
        t_alloc = min(rem * 0.3, 50)
        if t_alloc < 15:
            continue
        print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, f"lpfw_n{target_n}")

        # Quick peak CD
        rem2 = deadline - time.monotonic() - 100
        if rem2 > 15 and best_seq is not None and len(best_seq) == target_n:
            t_alloc2 = min(rem2 * 0.3, 30)
            if t_alloc2 > 5:
                a_opt2, obj2 = peak_equalization_cd(best_seq, time_limit=t_alloc2, verbose=False)
                best_seq, best_score = try_sequence(a_opt2, best_seq, best_score, f"peq_n{target_n}")

    # === Phase 5: Diverse initializations ===
    print(f"=== Phase 5: Diverse inits ===")
    n_best = len(best_seq) if best_seq is not None else n_parent
    diverse_inits = [
        make_asymmetric(n_best, 4.0),
        make_asymmetric(n_best, 6.0),
        make_bimodal_asymmetric(n_best, 0.35, 0.65, 0.12),
        make_bimodal_asymmetric(n_best, 0.25, 0.75, 0.08),
        make_bell(n_best, 0.6),
        make_bell(n_best, 0.85),
    ]
    for i, di in enumerate(diverse_inits):
        rem = deadline - time.monotonic() - 80
        if rem < 20:
            break
        # Interpolate base with diverse init
        blend = 0.5 * (best_seq if best_seq is not None else parent) + 0.5 * di
        blend = blend / blend.sum()
        best_seq, best_score = try_sequence(blend, best_seq, best_score, f"blend_{i}")

        t_alloc = min(rem * 0.2, 30)
        if t_alloc > 5:
            a_opt, obj = lp_fw_optimize(blend, time_limit=t_alloc, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, f"lpfw_div{i}")

    # === Phase 6: Deep optimization on best ===
    if best_seq is not None:
        print(f"=== Phase 6: Deep optimization (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_alloc = min(rem, 80)
            print(f"  LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "deep_lpfw")

    # === Phase 7: Final polish ===
    if best_seq is not None:
        print(f"=== Phase 7: Final polish (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            # Split remaining time among methods
            t1 = rem * 0.3
            t2 = rem * 0.3
            t3 = rem * 0.3
            t4 = rem * 0.1

            print(f"  Peak CD: t={t1:.0f}s")
            a_opt, obj = peak_equalization_cd(best_seq, time_limit=t1)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_peq")

            rem2 = deadline - time.monotonic() - 15
            if rem2 > 5 and best_seq is not None:
                t2a = min(t2, rem2 - 5)
                print(f"  Grad CD: t={t2a:.0f}s")
                a_opt, obj = gradient_descent_peaks(best_seq, time_limit=t2a)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_gradcd")

            rem3 = deadline - time.monotonic() - 10
            if rem3 > 5 and best_seq is not None:
                t3a = min(t3, rem3 - 5)
                print(f"  MD: t={t3a:.0f}s")
                a_opt, obj = multiplicative_descent(best_seq, time_limit=t3a)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_md")

            rem4 = deadline - time.monotonic() - 5
            if rem4 > 3 and best_seq is not None:
                t4a = min(t4, rem4)
                print(f"  Final LP-FW: t={t4a:.0f}s")
                a_opt, obj = lp_fw_optimize(best_seq, time_limit=t4a)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_lpfw2")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
