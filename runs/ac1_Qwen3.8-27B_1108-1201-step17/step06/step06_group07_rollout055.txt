def coordinate_descent(a0, time_limit, verbose=False, label=""):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_obj

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g

            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()
        if lr < 1e-9:
            break
    return best_a, best_obj


def solve_lp_direction(a, M, top_k_constraints=None):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k (or top-k most active), g >= 0.
    Returns the optimal g vector, or None if infeasible/failed.
    """
    n = len(a)
    num_constraints = 2 * n - 1

    if top_k_constraints is not None and top_k_constraints < num_constraints:
        # Only use the most active constraints
        conv_cross = fast_convolve(a)  # proxy for activity
        # Actually we need (a*g)_k but g is unknown. Use a*a as proxy for which k are tight
        # Better: use the convolution a*a to identify tight constraints
        aa_conv = fast_convolve(a)
        active_idx = np.argsort(aa_conv)[-top_k_constraints:]
    else:
        active_idx = np.arange(num_constraints)

    K = len(active_idx)
    # Build constraint matrix for active constraints only
    C = np.zeros((K, n))
    for row_i, k in enumerate(active_idx):
        k = int(k)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                C[row_i, i] = a[j]

    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
    except Exception:
        return None

    if not result.success:
        return None
    return result.x


def lp_optimize(a0, time_limit, verbose=False, label=""):
    """
    LP-based iterative optimization. At each step:
    1. Compute M = max(a*a)
    2. Solve LP to find g maximizing sum(g) subject to (a*g)_k <= M
    3. Blend a with normalized g
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lp_fail_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Try LP with varying numbers of constraints for speed
        if it < 50:
            g = solve_lp_direction(a, M, top_k_constraints=None)
        elif it < 200:
            g = solve_lp_direction(a, M, top_k_constraints=200)
        else:
            g = solve_lp_direction(a, M, top_k_constraints=100)

        if g is None:
            lp_fail_count += 1
            if lp_fail_count > 10:
                # Fall back to mirror descent step
                K = min(30, len(conv))
                peak_indices = np.argpartition(conv, -K)[-K:]
                pv = conv[peak_indices]
                scaled = pv / 0.001
                scaled -= scaled.max()
                w = np.exp(scaled)
                w /= w.sum()
                g = compute_subgradient(a, peak_indices, w)
                a = a * np.exp(np.clip(-0.01 * g, -50, 50))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                lp_fail_count = 0
                no_improve += 1
                continue
        else:
            lp_fail_count = 0
            g_sum = g.sum()
            if g_sum < 1e-10:
                no_improve += 1
                continue
            g_norm = g / g_sum

            # Line search on blend parameter
            best_t = 0
            best_trial_obj = M
            for t in [0.5, 0.3, 0.7, 1.0, 0.2, 0.8]:
                trial = (1 - t) * a + t * g_norm
                trial = np.maximum(trial, 0)
                s = trial.sum()
                if s < 1e-10:
                    continue
                trial = trial / s
                trial_obj = fast_convolve(trial).max()
                if trial_obj < best_trial_obj - 1e-15:
                    best_trial_obj = trial_obj
                    best_t = t

            if best_t == 0:
                for t in [0.1, 0.05]:
                    trial = (1 - t) * a + t * g_norm
                    trial = np.maximum(trial, 0)
                    s = trial.sum()
                    if s < 1e-10:
                        continue
                    trial = trial / s
                    trial_obj = fast_convolve(trial).max()
                    if trial_obj < best_trial_obj - 1e-15:
                        best_trial_obj = trial_obj
                        best_t = t

            if best_t > 0:
                a = (1 - best_t) * a + best_t * g_norm
                a = np.maximum(a, 0)
                a = a / a.sum()
                new_obj = fast_convolve(a).max()
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    no_improve = 0
                    if verbose and it % 100 == 0:
                        print(f"    {label} LP iter {it}: obj={best_obj:.10f}, t={best_t:.2f}")
                else:
                    no_improve += 1
            else:
                no_improve += 1

        # Symmetrize periodically
        if it % 30 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 0)
            s = a_sym.sum()
            if s > 1e-10:
                a_sym /= s
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj - 1e-15:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()
                no_improve = 0

        # Restart if stuck
        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 3)
            noise = rng.standard_normal(n) * 0.003
            a = np.maximum(best_a + noise, 1e-10)
            a = a / a.sum()
            no_improve = 0

        if no_improve > 500:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with multiplier search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in [0.85, 0.92, 0.97, 1.03, 1.08, 1.15]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence normalized to sum=1."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def symmetrize(a):
    """Symmetrize a sequence (make it palindromic)."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining LP-based direction finding,
    mirror descent, and coordinate descent. The LP approach solves
    max sum(g) s.t. (a*g)_k <= M at each step, giving exact optimal
    blend directions that outperform gradient-based methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_score = p_score
            best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse Initializations ===
    print("=== Phase 1: Initialization ===")
    n_values = [768]
    if parent is not None:
        n_values = [len(parent), 700, 800, 832, 896]
    else:
        n_values = [768, 700, 800, 832, 896, 960]

    best_init = None
    best_init_obj = float('inf')

    for n_try in n_values:
        for sigma in [0.2, 0.25, 0.3, 0.35, 0.4, 0.45]:
            a = make_gaussian(n_try, sigma)
            obj = try_seq(a, f"g_n{n_try}_s{sigma}")
            c_obj = 2 * n_try * fast_convolve(a).max()
            if c_obj < best_init_obj:
                best_init_obj = c_obj
                best_init = a

        a = make_bell(n_try, 0.5)
        obj = try_seq(a, f"bell_n{n_try}")
        c_obj = 2 * n_try * fast_convolve(a).max()
        if c_obj < best_init_obj:
            best_init_obj = c_obj
            best_init = a

    if parent is not None:
        obj = try_seq(parent, "parent")
        c_obj = 2 * len(parent) * fast_convolve(parent).max()
        if c_obj < best_init_obj:
            best_init_obj = c_obj
            best_init = parent

        # Also try symmetrized parent
        a_sym = symmetrize(parent)
        obj = try_seq(a_sym, "parent_sym")
        c_obj = 2 * len(parent) * fast_convolve(a_sym).max()
        if c_obj < best_init_obj:
            best_init_obj = c_obj
            best_init = a_sym

    if best_init is None:
        best_init = make_gaussian(768, 0.3)

    print(f"  Best init: score={best_score:.6f}, raw_obj={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: LP-based Optimization (main, largest allocation) ===
    print(f"\n=== Phase 2: LP Optimization ===")
    remaining = deadline - time.monotonic()
    t_lp = remaining * 0.50
    print(f"  Time allocated: {t_lp:.0f}s")
    a_opt, obj = lp_optimize(best_init, time_limit=t_lp, verbose=True, label="LP_main")
    try_seq(a_opt, "LP_main")

    # === Phase 3: Mirror Descent Refinement ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_md = remaining * 0.25
        print(f"  Time allocated: {t_md:.0f}s")
        a_opt, obj = mirror_descent(best_seq, time_limit=t_md, top_k=30,
                                    temp_init=0.0003, lr_init=0.008,
                                    verbose=True, label="MD_main")
        try_seq(a_opt, "MD_main")

    # === Phase 4: LP Pass 2 (symmetrized start) ===
    print(f"\n=== Phase 4: LP Pass 2 (sym) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_lp2 = remaining * 0.30
        print(f"  Time allocated: {t_lp2:.0f}s")
        a_sym = symmetrize(best_seq)
        a_opt, obj = lp_optimize(a_sym, time_limit=t_lp2 * 0.6, verbose=True, label="LP_sym")
        try_seq(a_opt, "LP_sym")

        if time.monotonic() < deadline - 30:
            a_opt, obj = mirror_descent(best_seq, time_limit=t_lp2 * 0.4, top_k=20,
                                        temp_init=0.0001, lr_init=0.005,
                                        verbose=True, label="MD_fine")
            try_seq(a_opt, "MD_fine")

    # === Phase 5: Coordinate Descent Polish ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = min(remaining * 0.40, 120)
        print(f"  Time allocated: {t_cd:.0f}s")
        a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd,
                                        verbose=True, label="CD")
        try_seq(a_opt, "CD")

    # === Phase 6: Multi-restart with perturbation ===
    print(f"\n=== Phase 6: Multi-restart ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_restarts = 3
        t_per = min(remaining * 0.30 / n_restarts, 80)
        if t_per > 20:
            for r in range(n_restarts):
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                if r == 0:
                    f_init = best_seq.copy()
                elif r == 1:
                    noise = rng.standard_normal(len(best_seq)) * 0.003
                    f_init = np.maximum(best_seq + noise, 1e-15)
                    f_init = f_init / f_init.sum()
                else:
                    f_init = symmetrize(best_seq)
                    noise = rng.standard_normal(len(f_init)) * 0.002
                    f_init = np.maximum(f_init + noise, 1e-15)
                    f_init = f_init / f_init.sum()

                t_alloc = min(t_per, remaining - 30)
                if t_alloc < 10:
                    break
                a_opt, obj = lp_optimize(f_init, time_limit=t_alloc,
                                         verbose=False, label=f"restart_{r}")
                try_seq(a_opt, f"restart_{r}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_polish = min(remaining - 10, 60)
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_polish,
                                            verbose=False, label="polish")
            try_seq(a_opt, "polish")

            # Try symmetrized final
            a_s = symmetrize(best_seq)
            try_seq(a_s, "final_sym")

            # Final MD micro-polish
            if remaining > 30:
                a_opt, obj = mirror_descent(best_seq, time_limit=min(remaining - 15, 40),
                                            top_k=10, temp_init=0.00005, lr_init=0.002,
                                            verbose=False, label="micro_MD")
                try_seq(a_opt, "micro_MD")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
