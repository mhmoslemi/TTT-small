    """

    candidates = generate_candidate_pool(parent) if parent is not None else generate_uniform_sequence(kwargs.pop('seed', 42))

    # Hybrid optimization via gradient descent at varying lengths and adaptive cooling
    # We'll test different Length families separately
    best_obj = float('inf')
    best_sequence = None
    length_list = [len(c) for c in candidates]  # Get all lengths

    # Main optimization loop
    for seq in candidates:
        # Convolution-based gradient for penalty function
        conv_seq = fast_convolve(seq)
        # Compute the derivative of the objective (2*n*max(conv_seq)/sum_sq)
        grad = compute_gradient_fast(conv_seq, n=len(seq))
        
        # Project gradient
        seq_simplex = project_simplex(seq)
        # With the called method of the evaluator: score is evaluated from normalized candidate
        # But wait, the Adam is for real penalties
        obj = 2 * len(seq_simplex) * np.max(conv_seq) / (np.sum(seq_simplex)**2)
        
        # If this candidate improves, update
        if obj < best_obj:
            best_obj = obj
            best_sequence = seq_simplex

    # Convert back to original sequence (clipping if needed by the Adam internal defense)
    if best_sequence is None:
        # fallback
        best_sequence = generate_uniform_sequence(kwargs.pop('default_length',32))
        best_obj = float(evaluate_sequence(best_sequence.tolist()))

    # Final check and ordering
    best_obj_val = evaluate_sequence(best_sequence.tolist())
    if best_obj_val < best_obj - 1e-10:  # if we found a better one internally
        best_obj = best_obj_val
        best_sequence = best_sequence

    # Return the best sequence in float list form
    return best_sequence.tolist()
