    p_pad = np.concatenate([p, np.zeros(n - 1)])
    # Window sums: for j = 0, ..., n-1: sum_{i=0}^{n-1} p_pad[j+i]
    cumsum = np.concatenate([[0], np.cumsum(p_pad)])
    window_sums = cumsum[n:n + n] - cumsum[:n]  # sum of p_pad[j:j+n] for j=0..n-1
    L = 2.0 * max(window_sums.max(), 1e-10)
    lr = 1.0 / L
    
    best_val = np.inf
    best_a = a.copy()
    
    for it in range(n_inner):
        gy = inner_gradient(y, p)
        a_new = project_simplex(y - lr * gy)
        
        # Compute objective
        val = 0.5 * np.dot(a_new, inner_gradient(a_new, p))
        
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_map * t_map)) / 2.0
        y = a_new + ((t_map - 1.0) / t_new) * (a_new - a)
        a = a_new
        t_map = t_new
    
    return best_a, best_val


def minimax_search(n, time_limit, a_init=None, verbose=False):
    """
    Minimax (Hedge + FISTA) for min_{a in simplex} max_k (a*a)[k].
    """
    rng = np.random.default_rng(12345)
    
    if a_init is None:
        a = np.ones(n) / n
    else:
        a = a_init.copy()
        a = np.maximum(a, 0)
        a /= a.sum()
    
    p = np.ones(2 * n - 1) / (2 * n - 1)
    
    best_obj = auto_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    lr_outer = 0.3  # step size for p update
    tau = 0.05  # temperature for exponentiated gradient
    
    # We'll use a schedule: first few hundred iters use more inner steps,
    # then reduce.
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Adaptive inner iterations
        if it < 100:
            n_inner = 50
        elif it < 500:
            n_inner = 30
        elif it < 2000:
            n_inner = 15
        else:
            n_inner = 10
        
        # Inner: FISTA
        a, _ = solve_inner_fista(p, a, n_inner)
        
        # Compute convolution
        conv = auto_convolve(a)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            if verbose and it % 200 == 0:
                print(f"      iter {it}: score={2*n*best_obj:.6f}")
        
        # Outer: exponentiated gradient update on p
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        
        # Update p
        p = (1 - lr_outer) * p + lr_outer * w
        p = np.maximum(p, 1e-30)
        p /= p.sum()
        
        # Adaptive schedule
        if it % 500 == 0:
            lr_outer = max(0.1, lr_outer * 0.95)
            # Decrease temperature to focus on peaks
            tau = max(0.005, tau * 0.9)
    
    return best_a


def subgradient_polish(a, time_limit, verbose=False):
    """
    Direct subgradient descent on f(a) = max_k (a*a)[k].
    Subgradient = 2 * Q_S * a where S = argmax set.
    Uses Polyak-like step size.
    """
    n = len(a)
    a = a.copy()
    best_obj = auto_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    lr = 0.01
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = auto_convolve(a)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"      SubGrad iter {it}: score={2*n*best_obj:.6f}")
        
        # Find active set (top peaks)
        # Use all peaks within 1% of max for a "smoothed" subgradient
        threshold = M * 0.99
        active = conv >= threshold
        if not active.any():
            active = np.zeros(len(conv), dtype=bool)
            active[np.argmax(conv)] = True
        
        # Compute subgradient (average over active peaks)
        # Subgrad = (2 / |S|) * sum_{k in S} Q_e_k * a
        # Q_e_k * a has components: [Q_e_k * a][m] = a[k-m] (if 0<=k-m<n)
        # So sum over k in S: sum_k [Q_e_k a][m] = sum_{k in S} a[k-m] (if valid)
        # = (active * a_rev)[m + n - 1] ... let me think
        # 
        # (Q_e_k a)[m] = a[k-m] if 0 <= k-m < n, else 0
        # sum_{k in S} (Q_e_k a)[m] = sum_{k in S} a[k-m] * 1[0<=k-m<n]
        # = sum_{j=0}^{n-1} a[j] * (sum_{k in S} 1[k = m+j])
        # = sum_{j=0}^{n-1} a[j] * active[m+j] (where active is 0-indexed)
        # This is the same as the inner_gradient but with p = active (normalized)
        # Actually it's (active * a_rev)[m+n-1]
        
        active_count = active.sum()
        a_rev = a[::-1]
        conv_sub = full_convolve(active.astype(float), a_rev)
        subgrad = (2.0 / active_count) * conv_sub[n-1:2*n-1]
        
        # Step
        a_new = project_simplex(a - lr * subgrad)
        
        # Check improvement
        conv_new = auto_convolve(a_new)
        M_new = conv_new.max()
        
        if M_new < M:
            a = a_new
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 200:
                break
        
    return best_a


def hadamard_init(n):
    """Initialize using a smoothed version of a 'flat-top' window."""
    # Use a raised cosine (Hann) window, which has good frequency properties
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def gaussian_init(n, center=0.4, sigma=0.25):
    """Asymmetric Gaussian initialization."""
    x = np.arange(n)
    mu = n * center
    s = n * sigma
    a = np.exp(-0.5 * ((x - mu) / s) ** 2)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def bell_init(n, width=0.6):
    """Flat-topped bell (Hann-like but wider)."""
    x = (np.arange(n) - n / 2.0) / (n / 2.0)
    a = np.cos(np.pi * x / 2.0) ** 2 * (np.abs(x) < width)
    a += 0.1  # small floor
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def interpolate_seq(a, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result
