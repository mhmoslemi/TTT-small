"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using peak-cycling projected gradient,
L-BFGS with low-temperature softmax, stochastic basin hopping, and coordinate
descent polish. Key novelty: cycling through individual auto-convolution peaks
and doing smooth quadratic optimization on each, combined with adaptive
weighting of active peaks.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_local(a):
    """Compute 2n*max(conv)/(sum)^2 locally (a assumed nonnegative)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def subgrad_at_peak(a, k):
    """Compute gradient of (a*a)[k] with respect to a.
    Returns vector g where g[i] = 2*a[k-i] (for valid k-i)."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def peak_cycling_optimize(a0, time_limit, n_peaks=10, lr_init=0.01, lr_decay=0.999,
                          patience=50, seed=42):
    """
    Peak-cycling: for each of the top peaks, do projected gradient descent
    on the smooth quadratic (a*a)[k]. Track best by true max objective.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Find top peaks
        b = fast_convolve(a)
        b_max = b.max()
        if b_max < best_obj * (1 - 1e-10):
            best_obj = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Get top peaks (those close to the max)
        threshold = best_obj * 0.95  # peaks within 5% of best
        active_peaks = np.where(b >= threshold)[0]
        if len(active_peaks) > n_peaks:
            # Take the top n_peaks
            peak_vals = b[active_peaks]
            top_idx_in_active = np.argsort(peak_vals)[::-1][:n_peaks]
            active_peaks = active_peaks[top_idx_in_active]
        
        # Cycle through active peaks
        for k in active_peaks:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            
            # Do a few projected gradient steps on (a*a)[k]
            for step in range(5):
                if time.monotonic() - t0 > time_limit:
                    break
                g = subgrad_at_peak(a, k)
                # Project gradient onto tangent space of simplex (remove mean component)
                g_tan = g - g.mean()
                a_new = project_simplex(a - cur_lr * g_tan)
                
                # Check if this helped (evaluate the specific peak)
                b_new = fast_convolve(a_new)
                if b_new[k] < b[k] - 1e-15:
                    a = a_new
                else:
                    break  # This step didn't help for this peak
        
        # Decay learning rate
        cur_lr *= lr_decay
        if cur_lr < 1e-6:
            cur_lr = lr_init * 0.1
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
        
        if no_improve > patience:
            # Restart with perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            cur_lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def weighted_subgrad_optimize(a0, time_limit, n_peaks=20, lr_init=0.005,
                              momentum=0.9, temp=0.001, seed=42):
    """
    Projected subgradient with adaptive temperature-weighted peaks.
    Uses exponential weighting of peaks based on their value.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        # Compute weights: focus on peaks near the max
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        # Compute weighted subgradient
        g = np.zeros(n)
        # Only use significant weights
        for k in range(len(b)):
            if w[k] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * 2.0 * a[k - idx]
        
        # Nesterov momentum step
        v = momentum * v - cur_lr * g
        g_tan = v - v.mean()
        a_new = project_simplex(a + g_tan)
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive adjustments
        if no_improve > 2000:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, 1e-6)
            no_improve = 0
        elif no_improve > 1000:
            cur_lr *= 0.95
        else:
            cur_lr = max(cur_lr * 0.9999, 1e-7)
        
        # Slightly grow temperature
        temp = min(temp * 1.001, 0.1)
    
    return best_a, best_obj


def softmax_objective_and_grad(z, T, n):

    a = softmax(z), objective = 2n * logsumexp(T * conv(a,a)) / T

    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)

    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()

    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(np.conj(W) * A, fft_size)[:n]
    # cross[i] = sum_k conj(W)_k * ... hmm, let me think more carefully.
    # We want: grad_a[i] = 2n * sum_k w[k] * a[k-i] (for valid k-i)
    # This is the cross-correlation of w and a.
    # Using FFT: cross_corr[i] = sum_k w[k+i] * a[i]... no.

    # We want sum_m w[m+i] * a[m] = np.correlate(a, w, 'full') shifted appropriately.

    # grad_a[i] = 2n * sum_k w[k] * a[k-i] for 0 <= k-i < n

    # but we can do it in O(n) total using the structure):

    # This is a sliding dot product, computable via FFT.
    
    # Let rev_a[j] = a[n-1-j]. Then:
    # sum_j w[i+j] * a[j] = sum_j w[i+j] * rev_a[n-1-j]
    # = (w * rev_a)[n-1+i]  (convolution)
    
    rev_a = a[::-1]

def fast_convolve_size(x, y, fft_size):
    """Full convolution of x and y using given FFT size."""
    A = rfft(x, fft_size)
    B = rfft(y, fft_size)
    return irfft(A * B, fft_size)[:len(x) + len(y) - 1]


def lbgfs_refine(a0, T, time_limit, maxiter=10000):
    """L-BFGS in log-space with soft-max objective."""

    a0_norm = a0 / a0.sum()
    z0 = np.log(np.maximum(a0_norm, 1e-300))
    
    best_obj, _ = softmax_objective_and_grad(z0, T, n)
    best_z = z0.copy()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = softmax_objective_and_grad(z, T, n)
        return obj

            return np.zeros_like(z)
        _, g = softmax_objective_and_grad(z, T, n)
        return g

            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,

        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x.copy()

        pass
    
    # Convert back
    z_shifted = best_z - best_z.max()
    a_out = np.exp(z_shifted)
    a_out = a_out / a_out.sum()
    return a_out, best_obj


def coordinate_polish(a0, time_limit, n_sweeps=200):
    """Coordinate descent: try scaling individual elements."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5])
    
    sweep = 0
    while sweep < n_sweeps and time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / denom
                if scale <= 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                s = a_trial.sum()
                if s < 0.01:
                    continue
                a_trial /= s
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - old_val)
                if scale > 1e-10:
                    scale = (1.0 - best_j_val) / scale
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved and sweep > 10:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def basin_hopping(a0, time_limit, n_restarts=50, noise_scale=0.01,
                  local_time=5.0, seed=42):
    """
    Basin hopping: perturb + local optimize, accept if improved.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        # Perturb
        noise = rng.standard_normal(n) * noise_scale
        a_pert = project_simplex(a + noise)
        
        # Local optimize (quick weighted subgradient)
        t_local_start = time.monotonic()
        t_local = min(local_time, time_limit - (time.monotonic() - t0) - 60)
        if t_local < 1:
            break
        
        a_opt, _ = weighted_subgrad_optimize(
            a_pert, time_limit=t_local, n_peaks=10,
            lr_init=0.01, momentum=0.9, temp=0.005,
            seed=seed + restart
        )
        
        b = fast_convolve(a_opt)
        obj = b.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_opt.copy()
            a = a_opt.copy()
            print(f"    Basin hop {restart}: new best {obj:.8f}")
        else:
            # Keep the perturbed point for next restart ( Langevin-like)
            a = a_pert
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using peak-cycling projected gradient,
L-BFGS with very low-temperature softmax, adaptive weighted subgradient with
Nesterov momentum, stochastic basin hopping, multi-length exploration, and
coordinate descent polish. Key novelty: cycling through individual auto-convolution
peaks and doing smooth quadratic optimization on each, combined with very-low-T
softmax for near-exact max tracking in L-BFGS.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_local(a):
    """Compute 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def subgrad_at_peak(a, k):
    """Gradient of (a*a)[k] w.r.t. a: g[i] = 2*a[k-i] for valid indices."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def peak_cycling_optimize(a0, time_limit, n_peaks=15, lr_init=0.015,
                          lr_decay=0.9998, patience=40, seed=42):
    """
    Peak-cycling: for each of the top peaks, do projected gradient descent
    on the smooth quadratic (a*a)[k]. Track best by true max objective.
    Uses backtracking to ensure each step helps.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    no_improve = 0
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        if b_max < best_obj - 1e-13:
            best_obj = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Identify active peaks (within 10% of best)
        threshold = best_obj * 0.92
        active_peaks = np.where(b >= threshold)[0]
        if len(active_peaks) > n_peaks:
            peak_vals = b[active_peaks]
            top_idx = np.argsort(peak_vals)[::-1][:n_peaks]
            active_peaks = active_peaks[top_idx]
        # Sort by peak value descending
        if len(active_peaks) > 0:
            order = np.argsort(b[active_peaks])[::-1]
            active_peaks = active_peaks[order]

        # Cycle through active peaks
        peak_improved_any = False
        for k in active_peaks:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj

            # Try to reduce this specific peak with backtracking
            current_peak_val = b[k]
            g = subgrad_at_peak(a, k)
            g_tan = g - g.mean()  # project onto tangent space of simplex
            g_norm = np.linalg.norm(g_tan)
            if g_norm < 1e-15:
                continue

            # Try a few learning rates
            found_step = False
            for lr_factor in [1.0, 0.5, 0.25, 0.1]:
                lr_try = cur_lr * lr_factor
                a_new = project_simplex(a - lr_try * g_tan)
                b_new_k = fast_convolve(a_new)[k]
                if b_new_k < current_peak_val - 1e-15:
                    a = a_new
                    peak_improved_any = True
                    found_step = True
                    # Also check overall max
                    b_full = fast_convolve(a)
                    if b_full.max() < best_obj - 1e-13:
                        best_obj = b_full.max()
                        best_a = a.copy()
                        no_improve = 0
                    break

            if not found_step:
                stuck_count += 1

        # Decay learning rate
        cur_lr *= lr_decay
        if cur_lr < 1e-7:
            cur_lr = lr_init * 0.2
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            stuck_count = 0

        if no_improve > patience:
            # Restart with perturbation
            noise_scale = 0.002 + 0.003 * (no_improve // (patience * 2))
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            cur_lr = lr_init
            no_improve = 0
            stuck_count = 0
        elif stuck_count > len(active_peaks) * 3:
            # All peaks stuck, try a different perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            cur_lr = max(cur_lr, lr_init * 0.5)
            stuck_count = 0

    return best_a, best_obj


def weighted_subgrad_optimize(a0, time_limit, n_peaks=20, lr_init=0.006,
                              momentum=0.92, temp_init=0.001, temp_growth=1.003,
                              kick_interval=2000, kick_size=0.003, seed=42):
    """
    Projected subgradient with Nesterov momentum and temperature-weighted peaks.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        # Process in small batches for time checking
        batch_size = 20
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break
            it_count += 1
            b = fast_convolve(a)

            # Compute weights with temperature
            scaled = b / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum

            # Compute weighted subgradient efficiently
            g = np.zeros(n)
            # Only process significant weights
            significant = w > 1e-8
            k_indices = np.where(significant)[0]
            for k in k_indices:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w[k] * 2.0 * a[k - idx]

            # Nesterov momentum step
            v = momentum * v - cur_lr * g
            v_tan = v - v.mean()
            a_new = project_simplex(a + v_tan)

            b_new = fast_convolve(a_new)
            obj_new = b_new.max()

            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            a = a_new

        # Periodic adjustments
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.6, lr_init * 0.05)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.97

        cur_temp = min(cur_temp * temp_growth, 0.05)

    return best_a, best_obj


def softmax_obj_grad(z, T, n):
    """
    Soft-max smoothed objective and gradient in log-space.
    a = softmax(z), obj = 2n/T * logsumexp(T * conv(a,a))
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()

    # Gradient: grad_a[i] = 2n * sum_k w[k] * a[k-i]
    # This is cross-correlation: (w star a)[i] = sum_k w[k+i] * a[k]
    # Using FFT: IFFT(conj(FFT(w)) * FFT(a))
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(np.conj(W) * A, fft_size)[:n]
    grad_a = 2.0 * n * cross

    # Convert to z-space gradient
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lbgfs_refine(a0, T, time_limit, maxiter=10000):
    """L-BFGS in log-space with soft-max objective."""
    t0 = time.monotonic()
    n = len(a0)
    a0_norm = a0 / a0.sum()
    z0 = np.log(np.maximum(a0_norm, 1e-300))

    best_obj, _ = softmax_obj_grad(z0, T, n)
    best_z = z0.copy()
    eval_count = [0]

    def func(z):
        eval_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = softmax_obj_grad(z, T, n)
        return obj

    def jac(z):
        eval_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = softmax_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x.copy()
            best_obj = final_obj
    except Exception:
        pass

    z_shifted = best_z - best_z.max()
    a_out = np.exp(z_shifted)
    a_out = a_out / a_out.sum()
    return a_out, best_obj


def coordinate_polish(a0, time_limit, n_sweeps=200):
    """Coordinate descent: try scaling individual elements."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5])

    sweep = 0
    while sweep < n_sweeps and time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / denom
                if scale <= 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                s = a_trial.sum()
                if s < 0.01:
                    continue
                a_trial /= s
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and sweep > 5:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def basin_hopping(a0, time_limit, n_restarts=80, noise_scale=0.008,
                  local_time=3.0, seed=42):
    """
    Basin hopping: perturb + local optimize, accept if improved.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    consecutive_rejects = 0

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        # Perturb (adapt noise scale)
        cur_noise = noise_scale * (1.0 + 0.5 * consecutive_rejects / max(1, n_restarts // 4))
        noise = rng.standard_normal(n) * cur_noise
        a_pert = project_simplex(a + noise)

        # Local optimize
        t_local = min(local_time, time_limit - (time.monotonic() - t0) - 60)
        if t_local < 1:
            break

        a_opt, _ = weighted_subgrad_optimize(
            a_pert, time_limit=t_local, n_peaks=15,
            lr_init=0.01, momentum=0.9, temp_init=0.003,
            temp_growth=1.005, kick_interval=500, kick_size=0.002,
            seed=seed + restart * 31
        )

        b = fast_convolve(a_opt)
        obj = b.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_opt.copy()
            a = a_opt.copy()
            consecutive_rejects = 0
            print(f"    Basin hop {restart}: new best {obj:.8f}")
        else:
            consecutive_rejects += 1
            # Keep perturbed point for diversity
            if consecutive_rejects % 5 == 0:
                a = a_pert

    return best_a, best_obj


def make_gaussian_init(n, sigma=0.3):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    """Raised cosine window."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    """Tukey window."""
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    t = np.linspace(0, 1, taper)
    a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
    a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Quick subgradient warm-up from parent
    2. Peak-cycling projected gradient (novel)
    3. L-BFGS with very low temperature (T=3,5,10)
    4. Basin hopping with adaptive noise
    5. Multi-length + diverse initialization exploration
    6. Final polish (L-BFGS + coordinate descent + micro peak-cycling)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1152) / 1152
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Quick subgradient warm-up ===
    print(f"\n=== Phase 1: Subgradient warm-up (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 1300
    if rem > 30:
        t_alloc = min(rem, 120.0)
        print(f"  Allocating {t_alloc:.0f}s")
        a_sg, _ = weighted_subgrad_optimize(
            parent, time_limit=t_alloc, n_peaks=25,
            lr_init=0.008, momentum=0.93,
            temp_init=0.001, temp_growth=1.004,
            kick_interval=1500, kick_size=0.002, seed=seed
        )
        try_seq(a_sg, "sg_warmup")

    # === Phase 2: Peak-cycling (NOVEL) ===
    print(f"\n=== Phase 2: Peak-cycling (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 1000
        if rem > 30:
            t_alloc = min(rem, 400.0)
            print(f"  Peak-cycling, t={t_alloc:.0f}s")
            a_pc, _ = peak_cycling_optimize(
                best_seq, time_limit=t_alloc, n_peaks=20,
                lr_init=0.02, lr_decay=0.9997, patience=35,
                seed=seed + 100
            )
            try_seq(a_pc, "peak_cycling_1")

            # Second round of peak cycling (finer)
            rem = deadline - time.monotonic() - 600
            if rem > 20:
                t_alloc2 = min(rem, 150.0)
                print(f"  Peak-cycling round 2, t={t_alloc2:.0f}s")
                a_pc2, _ = peak_cycling_optimize(
                    best_seq, time_limit=t_alloc2, n_peaks=30,
                    lr_init=0.005, lr_decay=0.9999, patience=50,
                    seed=seed + 200
                )
                try_seq(a_pc2, "peak_cycling_2")

    # === Phase 3: L-BFGS with very low temperature ===
    print(f"\n=== Phase 3: L-BFGS low-T (best={best_score:.6f}) ===")
    if best_seq is not None:
        for T in [3, 5, 10, 20]:
            rem = deadline - time.monotonic() - 500
            if rem < 15:
                break
            t_alloc = min(rem * 0.4, 100.0)
            if t_alloc < 10:
                continue
            print(f"  L-BFGS T={T}, t={t_alloc:.0f}s")
            a_lb, _ = lbgfs_refine(best_seq, T=T, time_limit=t_alloc, maxiter=8000)
            try_seq(a_lb, f"lbgfs_T{T}")

    # === Phase 4: Basin hopping ===
    print(f"\n=== Phase 4: Basin hopping (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 250.0)
            print(f"  Basin hopping, t={t_alloc:.0f}s")
            a_bh, _ = basin_hopping(
                best_seq, time_limit=t_alloc,
                n_restarts=150, noise_scale=0.005,
                local_time=2.5, seed=seed + 300
            )
            try_seq(a_bh, "basin_hop")

    # === Phase 5: Multi-length + diverse init ===
    print(f"\n=== Phase 5: Multi-length/diverse (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Try different lengths
        target_lengths = [int(n_best * 0.75), int(n_best * 1.25), int(n_best * 1.5),
                          n_best // 2, n_best * 2]
        target_lengths = sorted(set(x for x in target_lengths if 256 <= x <= 4096 and x != n_best),
                                key=lambda x: abs(x - n_best))

        for target_n in target_lengths[:3]:
            rem = deadline - time.monotonic() - 250
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 50.0)
            if t_alloc < 8:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = weighted_subgrad_optimize(
                f_interp, time_limit=t_alloc, n_peaks=15,
                lr_init=0.005, momentum=0.9,
                temp_init=0.002, temp_growth=1.005,
                kick_interval=800, kick_size=0.003,
                seed=seed + target_n
            )
            try_seq(a_opt, f"len_n{target_n}")

            if best_seq is not None and len(best_seq) == target_n:
                rem2 = deadline - time.monotonic() - 200
                if rem2 > 15:
                    a_pc3, _ = peak_cycling_optimize(
                        best_seq, time_limit=min(rem2 * 0.4, 30.0),
                        n_peaks=15, lr_init=0.01, seed=seed + target_n + 1
                    )
                    try_seq(a_pc3, f"pc_n{target_n}")

        # Diverse initializations at parent length
        inits = [
            ("gauss", make_gaussian_init(n_parent, 0.25)),
            ("cosine", make_cosine_init(n_parent)),
            ("tukey", make_tukey_init(n_parent, 0.4)),
            ("tukey_wide", make_tukey_init(n_parent, 0.7)),
        ]
        for name, init_a in inits[:2]:
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 30.0)
            if t_alloc < 8:
                continue
            print(f"  Diverse init: {name}, t={t_alloc:.0f}s")
            a_opt, _ = weighted_subgrad_optimize(
                init_a, time_limit=t_alloc, n_peaks=15,
                lr_init=0.01, momentum=0.9,
                temp_init=0.005, temp_growth=1.01,
                kick_interval=500, kick_size=0.005,
                seed=seed + hash(name) % 10000
            )
            try_seq(a_opt, f"diverse_{name}")

    # === Phase 6: Deep L-BFGS on best ===
    print(f"\n=== Phase 6: Deep L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        for T in [3, 5]:
            rem = deadline - time.monotonic() - 80
            if rem < 10:
                break
            t_alloc = min(rem * 0.5, 50.0)
            if t_alloc < 8:
                continue
            print(f"  Deep L-BFGS T={T}, t={t_alloc:.0f}s")
            a_fin, _ = lbgfs_refine(best_seq, T=T, time_limit=t_alloc, maxiter=10000)
            try_seq(a_fin, f"deep_lbgfs_T{T}")

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: CD polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd = min(rem, 45.0)
            print(f"  CD, t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd, n_sweeps=150)
            try_seq(a_cd, "cd_polish")

    # === Phase 8: Final micro peak-cycling ===
    print(f"\n=== Phase 8: Final micro (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 25.0)
            a_micro, _ = peak_cycling_optimize(
                best_seq, time_limit=t_alloc, n_peaks=30,
                lr_init=0.003, lr_decay=0.99995, patience=50,
                seed=seed + 777
            )
            try_seq(a_micro, "micro_pc")

            # One more L-BFGS pass
            rem = deadline - time.monotonic() - 5
