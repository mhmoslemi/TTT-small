    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated mirror descent with
soft-max smoothing, LP-Few-Wolfe refinement, and coordinate descent polishing.
Key innovation: replaces expensive LP solves with O(n log n) soft-max gradient
evaluations, enabling 100x more iterations within the time budget.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def soft_max_conv(a, beta):
    """Compute smooth approximation of max(conv) and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    K = len(conv)
    # Numerical stability
    MC = conv.max()
    shifted = beta * (conv - MC)
    # Clip to prevent overflow
    shifted = np.clip(shifted, -100, 100)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    # Gradient: sg[j] = beta * sum_k w[k] * d(conv_k)/da[j]
    # d(conv_k)/da[j] = a[k-j] if 0 <= k-j < n
    # sg = beta * cross-correlation of w with a (in a specific sense)
    # Actually: sg[j] = beta * sum_k w[k] * a[k-j] (when 0<=k-j<n)
    # This is a convolution-like operation
    # sg[j] = beta * (w * a_flipped)[j + n - 1] where a_flipped = reverse(a)
    a_flip = a[::-1]
    # conv(w, a_flip)[j + n - 1] = sum_k w[k] * a_flip[j+n-1-k] = sum_k w[k] * a[k-j]
    # We need the full linear convolution of w (length 2n-1) and a_flip (length n)
    # Result has length 3n-2, and we want indices n-1 to 2n-2 (corresponding to j=0 to n-1)
    total_size = K + n - 1
    fft_sz = 1 << (total_size - 1).bit_length()
    W = np.fft.rfft(w, fft_sz)
    AF = np.fft.rfft(a_flip, fft_sz)
    cross = np.fft.irfft(W * AF, fft_sz)[:total_size]
    sg = beta * cross[n - 1: 2 * n - 1]
    obj = MC + np.log(exp_vals.sum()) / beta
    return obj, sg


def accelerated_mirror_descent(f0, time_limit, beta_start=5.0, beta_end=50.0,
                                eta_scale=1.0, verbose=False):
    """
    Accelerated mirror descent with soft-max smoothing.
    Minimizes 2n * max(conv(a,a)) over the simplex {a >= 0, sum a = 1}.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    t = 1
    f_prev = f.copy()
    total_iters = 0
    no_improve = 0

    # Adaptive beta schedule
    n_total_target = int(time_limit * 500)  # estimate total iterations
    beta = beta_start

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.1:
            break

        t += 1
        total_iters += 1

        # Increase beta gradually
        progress = min(1.0, total_iters / max(1, n_total_target * 0.7))
        beta = beta_start + (beta_end - beta_start) * progress

        # Nesterov extrapolation
        if t == 2:
            coeff = 0.0
        else:
            coeff = (t - 2) / (t - 1)
        z = f + coeff * (f - f_prev)
        z = np.maximum(z, 1e-15)
        zs = z.sum()
        z = z / zs

        # Compute soft-max gradient
        _, sg = soft_max_conv(z, beta)

        # Mirror descent step with adaptive eta
        # Lipschitz constant estimate: L ~ beta * max(z) * 2 (rough)
        eta = eta_scale / np.sqrt(t + 1)
        # Clip gradient to prevent extreme updates
        sg_max = np.max(np.abs(sg))
        if sg_max > 1e-10:
            sg = sg / sg_max * min(sg_max, 2.0)

        f_new = z * np.exp(-eta * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        f_prev = f
        f = f_new

        # Evaluate
        if total_iters % 10 == 0 or total_iters < 100:
            obj = 2.0 * n * fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                if verbose and total_iters % 1000 == 0:
                    print(f"    AMD iter {total_iters}: obj={best_obj:.10f}, beta={beta:.2f}")
            else:
                no_improve += 1

        if total_iters % 5000 == 0 and no_improve > 20000:
            # Restart with perturbation
            rng = np.random.default_rng(total_iters)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 1e-15)
            f = f / f.sum()
            f_prev = f.copy()
            no_improve = 0
            t = 1

    return best_f, best_obj


def standard_fw(f0, time_limit, verbose=False):
    """
    Standard Frank-Wolfe: at each step, move towards the unit vector
    corresponding to the minimum subgradient component. Very fast (O(n log n) per step).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Find subgradient (use all active peaks averaged)
        active = np.where(conv >= M - 1e-15)[0]
        n_active = len(active)
        sg = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += f[k - np.arange(js, je + 1)]
        sg /= n_active

        # FW direction: towards e_{argmin(sg)}
        i_star = int(np.argmin(sg))

        # Step size: 2/(t+2) for standard FW
        alpha = 2.0 / (it + 2.0)
        # Also try a few step sizes
        best_val = M
        best_a = alpha
        for a_try in [alpha, alpha * 0.5, alpha * 2.0, 0.01, 0.05]:
            trial = (1 - a_try) * f + a_try * (1.0 if i_star == 0 else 0.0) * np.ones(1)[0]
            # Actually construct properly
            trial = f.copy()
            # Move towards e_i_star
            # e_i_star has 1 at position i_star, 0 elsewhere
            # (1-a)*f + a*e_i_star: at position i_star, value is a; elsewhere, (1-a)*f[j]
            # But this doesn't maintain sum=1 in general... actually it does:
            # sum = a + (1-a)*sum(f) = a + (1-a)*1 = 1. Good.
            # But it might make some entries negative if f[j] < 0... no, f >= 0.
            # Actually: trial[j] = (1-a)*f[j] for j != i_star, trial[i_star] = a
            # Wait no: (1-a)*f + a*e = [(1-a)*f[0], ..., (1-a)*f[i_star]+a, ..., (1-a)*f[n-1]]
            # That's not right either. Let me be explicit:
            trial = (1.0 - a_try) * f
            trial[i_star] += a_try
            tv = fast_convolve(trial).max()
            if tv < best_val:
                best_val = tv
                best_a = a_try

        f = (1.0 - best_a) * f
        f[i_star] += best_a
        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs > 1e-15:
            f = f / fs

        if it % 50 == 0:
            obj = 2.0 * n * fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 50

        if no_improve > 5000:
            # Try a perturbation
            rng = np.random.default_rng(it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    m_levels = [1.0, 1.02, 1.05, 1.1, 1.15, 1.25, 1.5, 2.0]
    m_idx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        g = None
        for level_try in range(m_idx, len(m_levels)):
            M_try = conv.max() * m_levels[level_try]
            g = solve_lp_direction(f, M_try, n)
            if g is not None and g.sum() > 1e-12:
                m_idx = level_try
                break
            m_idx = level_try + 1

        if g is None or g.sum() < 1e-12:
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            m_idx = 0
            no_improve += 1
            if no_improve > 100:
                break
            continue

        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            if no_improve > 80:
                break
            continue
        no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            continue

        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        M = conv.max()
        f = f_new
        m_idx = 0

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Fine-grained coordinate descent: try moving small amounts of mass
    between pairs of coordinates, accepting only improvements.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 200:
            break

        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Find contributing indices for the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        if len(contrib_idx) == 0:
            break

        products = f[contrib_idx] * f[peak_k - contrib_idx]
        total_contrib = products.sum()
        if total_contrib < 1e-15:
            break

        # Try moving mass from the largest contributor to non-contributors
        max_prod_i = int(contrib_idx[np.argmax(products)])
        partner = peak_k - max_prod_i

        # Determine source (larger of the pair)
        if f[max_prod_i] >= f[partner]:
            source = max_prod_i
        else:
            source = partner

        # Non-contributing coordinates
        mask = np.ones(n, dtype=bool)
        mask[contrib_idx] = False
        if not mask.any():
            mask[:] = True
            mask[source] = False

        nc_indices = np.where(mask)[0]
        if len(nc_indices) == 0:
            break

        found_improve = False
        for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3]:
            diff_amt = f[source] * frac
            if diff_amt < 1e-15:
                continue
            trial = f.copy()
            trial[source] -= diff_amt
            # Distribute to non-contributors
            nc_vals = trial[nc_indices]
            nc_sum = nc_vals.sum()
            if nc_sum > 1e-15:
                trial[nc_indices] += diff_amt * nc_vals / nc_sum
            else:
                trial[nc_indices] += diff_amt / len(nc_indices)

            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts > 1e-15:
                trial = trial / ts
            obj = 2.0 * n * fast_convolve(trial).max()
            if obj < best_obj - 1e-13:
                f = trial
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                found_improve = True
                break

        if not found_improve:
            # Try all top peaks
            top_peaks = np.argsort(conv)[-min(10, len(conv)):]
            for pk in top_peaks:
                pk = int(pk)
                if pk == peak_k:
                    continue
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                cidx2 = np.arange(js2, je2 + 1)
                if len(cidx2) == 0:
                    continue
                prods2 = f[cidx2] * f[pk - cidx2]
                src2 = int(cidx2[np.argmax(prods2)])
                partner2 = pk - src2
                if f[src2] >= f[partner2]:
                    source2 = src2
                else:
                    source2 = partner2

                mask2 = np.ones(n, dtype=bool)
                mask2[cidx2] = False
                if mask2.sum() < 2:
                    continue
                nc2 = np.where(mask2)[0]

                for frac2 in [0.02, 0.05, 0.1, 0.2]:
                    diff2 = f[source2] * frac2
                    if diff2 < 1e-15:
                        continue
                    trial = f.copy()
                    trial[source2] -= diff2
                    nc_vals2 = trial[nc2]
                    nc_sum2 = nc_vals2.sum()
                    if nc_sum2 > 1e-15:
                        trial[nc2] += diff2 * nc_vals2 / nc_sum2
                    trial = np.maximum(trial, 0.0)
                    trial = trial / trial.sum()
                    obj2 = 2.0 * n * fast_convolve(trial).max()
                    if obj2 < best_obj - 1e-13:
                        f = trial
                        best_obj = obj2
                        best_f = f.copy()
                        no_improve = 0
                        found_improve = True
                        break
                if found_improve:
                    break

        if not found_improve:
            no_improve += 1
            # Try a small random perturbation occasionally
            if no_improve % 20 == 0:
                rng = np.random.default_rng(it * 7 + 3)
                f = project_simplex(f + rng.standard_normal(n) * 0.001)
                no_improve = 0

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.3, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Diverse initializations + fast accelerated mirror descent
    2. LP-FW refinement
    3. Coordinate descent polishing
    4. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = try_seq(parent, "parent")
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        try_seq(parent, "default")

    print(f"\n=== Phase 1: Accelerated Mirror Descent (multiple inits) ===")
    t0 = time.monotonic()

    # Generate diverse initializations
    inits = []
    for n_try in [640, 768, 896]:
        inits.append((f'gauss_{n_try}', make_gaussian(n_try, 0.25)))
        inits.append((f'bell_{n_try}', make_bell(n_try, 0.5)))
        inits.append((f'cos_{n_try}', make_multi_cosine(n_try, 4, 0.25, seed=n_try)))
        inits.append((f'uniform_{n_try}', np.ones(n_try) / n_try))
    inits.append(('parent', parent))

    # Quick evaluate all inits
    scored = []
    for label, a_init in inits:
        sc = objective_normalized(a_init)
        scored.append((sc, label, a_init))
        if sc < 2.05:
            try_seq(a_init, f"init_{label}")
    scored.sort(key=lambda x: x[0])

    # Run AMD on top 4 initializations
    rem = deadline - time.monotonic()
    t_per = max(30, (rem * 0.35) // 4)
    for i in range(min(4, len(scored))):
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        sc, label, a_init = scored[i]
        t_alloc = min(t_per, rem - 10)
        print(f"  AMD [{label}] n={len(a_init)}: alloc={t_alloc:.0f}s (init_obj={sc:.6f})")
        a_opt, obj = accelerated_mirror_descent(a_init, time_limit=t_alloc,
                                                 beta_start=5.0, beta_end=80.0,
                                                 eta_scale=1.5, verbose=True)
        try_seq(a_opt, f"amd_{label}")

    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s, best={best_score:.6f}")

    # === Phase 2: Standard FW (very fast, many iterations) ===
    print(f"\n=== Phase 2: Standard FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.15, 120)
            print(f"  FW: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = standard_fw(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "fw")

    # === Phase 3: LP-FW refinement ===
    print(f"\n=== Phase 3: LP-FW Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            t_alloc = min(rem * 0.25, 300)
            print(f"  LP-FW: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 4: AMD again on best (with different params) ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.15, 120)
            print(f"\n=== Phase 4: AMD Round 2 ===")
            print(f"  AMD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = accelerated_mirror_descent(best_seq, time_limit=t_alloc,
                                                     beta_start=20.0, beta_end=200.0,
                                                     eta_scale=0.8, verbose=True)
            try_seq(a_opt, "amd2")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coordinate Descent Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.2, 150)
            print(f"  CD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 6: LP-FW with perturbations ===
    print(f"\n=== Phase 6: LP-FW with Perturbations ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        perturbations = []
        for shift in [1, 2, 3]:
            perturbations.append((f'shift{shift}', np.roll(best_seq, shift)))
        for mix in [0.05, 0.1, 0.2]:
            perturbations.append((f'mix{mix}', (1 - mix) * best_seq + mix * (np.ones(n_best) / n_best)))

        for label, f_p in perturbations:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            f_p = project_simplex(f_p)
            t_alloc = min(rem * 0.25, 30)
            if t_alloc < 5:
                break
            print(f"  LP-FW [{label}]: alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_p, time_limit=t_alloc)
            try_seq(a_opt, f"pert_{label}")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length ===")
    if best_seq is not None:
        for target_n in [1024, 896, 768, 640, 512]:
            if abs(target_n - len(best_seq)) < 50:
                continue
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_{target_n}")
            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.2, 30)
                if t_alloc > 5:
                    a_opt, obj = accelerated_mirror_descent(f_interp, time_limit=t_alloc,
                                                             beta_start=10.0, beta_end=100.0,
                                                             eta_scale=1.0)
                    try_seq(a_opt, f"amd_n{target_n}")

    # === Phase 8: Final coordinate descent ===
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            t_alloc = min(rem * 0.5, 60)
            print(f"  CD Final: alloc={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd_final")

        # One more AMD pass
        rem = deadline - time.monotonic()
        if rem > 10:
            t_alloc = min(rem * 0.4, 30)
            print(f"  AMD Final: alloc={t_alloc:.0f}s")
            a_opt, obj = accelerated_mirror_descent(best_seq, time_limit=t_alloc,
                                                     beta_start=50.0, beta_end=300.0,
                                                     eta_scale=0.5)
            try_seq(a_opt, "amd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
