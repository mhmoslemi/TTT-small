def spectral_iteration(a, n_iters=100):
    n = len(a)
    N = 1 << (2*n - 1).bit_length()  # FFT size
    a = np.pad(a, (0, N - n))
    for _ in range(n_iters):
        A = np.fft.fft(a)
        # We want |A|^2 to be as flat as possible
        # But we need to preserve the DC component (sum of a)
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        # Target: flat magnitude
        target_mag = np.ones(N) * A_mag.mean()
        # But preserve DC
        target_mag[0] = A_mag[0]
        # Update
        A_new = target_mag * np.exp(1j * A_phase)
        a = np.fft.ifft(A_new).real
        a = a[:n]
        a = np.maximum(a, 0)
        a = a / a.sum()
    return a

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def objective_val(a):

def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0)

        for m_frac in [1.0, 1.001, 1.005, 1.01, 1.02, 1.05, 1.1, 1.15, 1.25, 1.5, 2.0, 3.0]:
            M_try = M * m_frac
            b_ub = np.full(2 * n - 1, M_try)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': max(5.0, min(30.0, rem * 0.3))})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                if g.sum() > 1e-12:
                    break
                g = None
        if g is None:
            # Perturb and restart
            rng = np.random.default_rng(it * 31 + 7)
            strategy = it % 4
            if strategy == 0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            elif strategy == 1:
                f = project_simplex(0.8 * best_f + 0.2 * best_f[::-1])
            elif strategy == 2:
                shift = rng.integers(1, max(2, n // 10))
                f = project_simplex(np.roll(best_f, shift))
            else:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve += 1
            if no_improve > 60:
                break
            continue

        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            # Small perturbation to break symmetry
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(f + rng.standard_normal(n) * 0.001)
            conv = fast_convolve(f)
            M = conv.max()
            continue

        no_improve = 0
        best_alpha, best_val = line_search(f, g, M)

        if best_alpha < 1e-10:
            no_improve += 1
            continue

        f_new = f + best_alpha * (g - f)
        f_new = norm_seq(f_new)
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        f = f_new
        M = M_new

        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 10 == 0:
                print(f"    LP-FW it={it}: obj={best_obj:.10f}, alpha={best_alpha:.4f}")

    return best_f, best_obj


def peak_directed_descent(f0, time_limit, verbose=True):
    """Targeted coordinate descent: reduce mass at peak-contributing positions."""
    f = norm_seq(f0.astype(np.float64))
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break
        if no_improve > 100:
            break
        it += 1

        conv = fast_convolve(f)
        M = conv.max()
        # Get top few peaks
        n_peaks = min(5, 2 * n - 1)
        top_peaks = np.argsort(conv)[-n_peaks:]

        found_improve = False
        for pk in top_peaks:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            contrib = np.arange(js, je + 1)
            if len(contrib) == 0:
                continue
            products = f[contrib] * f[pk - contrib]
            total = products.sum()
            if total < 1e-15:
                continue

            # Find the most contributing pair
            max_idx = contrib[np.argmax(products)]
            partner = pk - max_idx

            # Determine source (larger of the pair)
            if f[max_idx] >= f[partner]:
                source = max_idx
            else:
                source = partner

            # Target: non-contributing positions
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[contrib] = False
            if non_contrib.sum() < 2:
                non_contrib[:] = True
                non_contrib[source] = False

            # Try various fractions
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                diff = f[source] * frac
                if diff < 1e-15:
                    continue
                trial = f.copy()
                trial[source] -= diff
                nc_vals = trial[non_contrib]
                nc_sum = nc_vals.sum()
                if nc_sum < 1e-15:
                    # Distribute uniformly
                    trial[non_contrib] += diff / non_contrib.sum()
                else:
                    trial[non_contrib] += diff * nc_vals / nc_sum
                trial = norm_seq(trial)
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    found_improve = True
                    break
            if found_improve:
                break

        if not found_improve:
            # Try swap-based moves
            rng = np.random.default_rng(it * 43 + 11)
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                trial = f.copy()
                trial[i], trial[j] = f[j], f[i]
                trial = norm_seq(trial)
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    found_improve = True
                    break
            if not found_improve:
                no_improve += 1
                # Perturb
                if no_improve % 20 == 0:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)

        if verbose and it % 20 == 0:
            print(f"    PTD it={it}: obj={best_obj:.10f}")

    return best_f, best_obj


def multiplicative_descent(f0, time_limit, verbose=True, lr_init=0.003, decay=0.9998):
    """Mirror descent with subgradient of max(conv)."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = lr_init
    temp = 0.001
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break
        it += 1

        conv = fast_convolve(f)
        M = conv.max()

        # Weight peaks by their excess over the mean
        mean_conv = conv.mean()
        excess = np.maximum(conv - mean_conv, 0)
        top_k = min(50, 2 * n - 1)
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_excess = excess[peak_idx]
        if peak_excess.max() < 1e-15:
            break
        # Temperature-based weighting
        scaled = peak_excess / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws

        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            partners = k - idx
            sg[idx] += 2.0 * w[i] * f[partners]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MD it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= decay
        temp *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(it * 53 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.005, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.5
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def spectral_equalize(f0, time_limit, verbose=True):
    """Iterative spectral equalization: alternate between time-domain (non-neg)
    and frequency-domain (flatten |A|^2)."""
    a = norm_seq(f0.astype(np.float64))
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    # Initial frequency domain
    a_padded = np.zeros(N)
    a_padded[:n] = a
    A = np.fft.fft(a_padded)

    # Target: flatten |A|^2 while preserving DC
    target_dc = A[0].real  # = sum(a) = 1

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break
        it += 1

        # Time domain: enforce non-negativity and normalization
        a = np.fft.ifft(A).real
        a = a[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            if verbose and it % 20 == 0:
                print(f"    Spectral it={it}: obj={best_obj:.10f}")

        # Frequency domain: modify |A| to flatten |A|^2
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)

        # Compute IFFT(|A|^2) to see current convolution shape
        conv_freq = np.fft.ifft(np.abs(A) ** 2).real

        # Identify which frequencies contribute most to the peak
        peak_k = np.argmax(fast_convolve(a))
        # In frequency domain, the contribution of frequency m to conv[peak_k]
        # is |A[m]|^2 * exp(2*pi*i*m*peak_k/N)
        phases = np.exp(2j * np.pi * np.arange(N) * peak_k / N)
        contrib = np.abs(A) ** 2 * np.real(phases)

        # Reduce |A| at frequencies that most contribute to the peak
        # and increase at those that reduce it
        target_mag = np.abs(A).copy()
        # Soft equalization
        mean_mag_sq = np.mean(np.abs(A) ** 2)
        blend = 0.05  # small blend to avoid oscillation
        for m in range(1, N // 2):
            if contrib[m] > 0:
                target_mag[m] *= (1.0 - blend * min(1.0, contrib[m] / (mean_mag_sq + 1e-15)))
            else:
                target_mag[m] *= (1.0 + blend * min(1.0, -contrib[m] / (mean_mag_sq + 1e-15)))

        # Reconstruct A with modified magnitudes
        phase_A = np.angle(A)
        A_new = target_mag * np.exp(1j * phase_A)
        A_new[0] = target_dc + 0j  # preserve DC
        # Ensure Hermitian symmetry (for real signal)
        A_new[1:] = (A_new[1:] + np.conj(A_new[N - np.arange(1, N // 2 + 1)])) / 2 if N % 2 == 0 else A_new[1:]

        # Simpler: just use the magnitude modification
        A = target_mag * np.exp(1j * phase_A)
        A[0] = target_dc + 0j

    return best_a, best_obj


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f)
    return norm_seq(f_interp)


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return norm_seq(a)


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    return norm_seq(np.maximum(a, 0))


def make_multi_cosine(n, n_freqs=5, max_amp=0.2, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.02, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    return norm_seq(np.maximum(a, 0))


def make_bell(n, width_frac=0.5):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    return norm_seq(np.maximum(a, 0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimizer combining LP-Frank-Wolfe, peak-directed descent,
    multiplicative descent, spectral equalization, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01 or len(a) == 0:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = try_seq(parent, "parent")
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768)
        try_seq(parent, "default_gauss")

    n_best = len(best_seq) if best_seq is not None else len(parent)
    print(f"Starting optimization: best_n={n_best}, best_score={best_score:.6f}")

    # === Phase 1: LP-FW on parent (main workhorse) ===
    print("\n=== Phase 1: LP-FW Primary (600s) ===")
    alloc = min(600.0, (deadline - time.monotonic()) * 0.4)
    if alloc > 30:
        a_opt, obj = lp_fw_optimize(parent, time_limit=alloc, verbose=True)
        try_seq(a_opt, "lpfw_primary")
        if best_seq is not None:
            parent = best_seq
            n_best = len(best_seq)

    # === Phase 2: Peak-directed descent ===
    print("\n=== Phase 2: Peak-Directed Descent (250s) ===")
    alloc = min(250.0, (deadline - time.monotonic()) * 0.3)
    if alloc > 20:
        a_opt, obj = peak_directed_descent(parent, time_limit=alloc, verbose=True)
        try_seq(a_opt, "ptd")
        if best_seq is not None:
            parent = best_seq
            n_best = len(best_seq)

    # === Phase 3: Multiplicative descent ===
    print("\n=== Phase 3: Multiplicative Descent (200s) ===")
    alloc = min(200.0, (deadline - time.monotonic()) * 0.25)
    if alloc > 15:
        a_opt, obj = multiplicative_descent(parent, time_limit=alloc, verbose=True)
        try_seq(a_opt, "md")
        if best_seq is not None:
            parent = best_seq
            n_best = len(best_seq)

    # === Phase 4: LP-FW refinement (tighter) ===
    print("\n=== Phase 4: LP-FW Refinement (250s) ===")
    alloc = min(250.0, (deadline - time.monotonic()) * 0.25)
    if alloc > 20:
        a_opt, obj = lp_fw_optimize(parent, time_limit=alloc, verbose=True)
        try_seq(a_opt, "lpfw_refine")
        if best_seq is not None:
            parent = best_seq
            n_best = len(best_seq)

    # === Phase 5: Spectral equalization ===
    print("\n=== Phase 5: Spectral Equalization (120s) ===")
    alloc = min(120.0, (deadline - time.monotonic()) * 0.15)
    if alloc > 15:
        a_opt, obj = spectral_equalize(parent, time_limit=alloc, verbose=True)
        try_seq(a_opt, "spectral")
        if best_seq is not None:
            parent = best_seq
            n_best = len(best_seq)

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length Exploration (150s) ===")
    target_lengths = []
    base_n = len(best_seq) if best_seq is not None else 768
    for tl in [base_n - 32, base_n + 32, base_n - 64, base_n + 64, 1024, 640, 512]:
        if 256 <= tl <= 2048 and tl != base_n and tl not in target_lengths:
            target_lengths.append(tl)

    for tl in target_lengths[:4]:
        rem = (deadline - time.monotonic())
        if rem < 15:
            break
        f_interp = interpolate_seq(best_seq if best_seq is not None else parent, tl)
        sc = try_seq(f_interp, f"interp_n{tl}")
        if sc < best_score + 0.02:
            alloc = min(50.0, rem * 0.3)
            if alloc > 10:
                a_opt, obj = lp_fw_optimize(f_interp, time_limit=alloc, verbose=False)
                try_seq(a_opt, f"lpfw_n{tl}")
                if best_seq is not None:
                    parent = best_seq

    # === Phase 7: Perturbation + LP-FW ===
    print("\n=== Phase 7: Perturbation Search (120s) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed)
        perturbations = []
        for shift in [1, 2, 3, 5, 8]:
            perturbations.append(('shift', np.roll(best_seq, shift)))
        perturbations.append(('flip', best_seq[::-1]))
        for mix in [0.05, 0.1, 0.15, 0.2]:
            perturbations.append((f'mix{mix}', (1 - mix) * best_seq + mix * (np.ones(n_best) / n_best)))
        for amp in [0.002, 0.005, 0.01, 0.02]:
            noise = rng.standard_normal(n_best) * amp
            perturbations.append((f'noise{amp}', np.maximum(best_seq + noise, 0)))
        # Shifted cosine blend
        for freq in [1, 2, 3]:
            for amp in [0.05, 0.1]:
                cos_mod = make_cosine_mod(n_best, freq, amp)
                perturbations.append((f'cos{freq}_{amp}', (1 - amp) * best_seq + amp * cos_mod))

        for label, f_pert in perturbations:
            rem = (deadline - time.monotonic())
            if rem < 10:
                break
            f_pert = project_simplex(f_pert)
            sc = try_seq(f_pert, f"pert_{label}")
            if sc < best_score + 0.01:
                alloc = min(30.0, rem * 0.4)
                if alloc > 8:
                    a_opt, obj = lp_fw_optimize(f_pert, time_limit=alloc, verbose=False)
                    try_seq(a_opt, f"lpfw_pert_{label}")
                    if best_seq is not None:
                        parent = best_seq

    # === Phase 8: Final polish ===
    print("\n=== Phase 8: Final Polish (remaining time) ===")
    if best_seq is not None:
        # Quick PTD
        rem = (deadline - time.monotonic()) - 30
        if rem > 15:
            alloc = min(60.0, rem)
            a_opt, obj = peak_directed_descent(best_seq, time_limit=alloc, verbose=False)
            try_seq(a_opt, "ptd_final")

        # Quick MD
        rem = (deadline - time.monotonic()) - 15
        if rem > 10:
            alloc = min(30.0, rem)
            a_opt, obj = multiplicative_descent(best_seq, time_limit=alloc, verbose=False,
                                                lr_init=0.001, decay=0.9999)
            try_seq(a_opt, "md_final")

        # Final LP-FW squeeze
        rem = (deadline - time.monotonic()) - 5
        if rem > 8:
            alloc = min(rem, 30.0)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=alloc, verbose=False)
            try_seq(a_opt, "lpfw_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
