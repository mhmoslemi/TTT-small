```python
import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

def logsumexp_objective_and_grad(z, tau):
    f = softmax(z)
    n = len(f)
    c = fast_convolve(f)
    # F = tau * log(sum(exp(c/tau)))
    c_scaled = c / tau
    c_max = c_scaled.max()
    exp_shifted = np.exp(c_scaled - c_max)
    log_sum = c_max + np.log(exp_shifted.sum())
    F = tau * log_sum
    # Weights
    w = exp_shifted / exp_shifted.sum()
    # Cross-correlation h_i = sum_j w[j+i] * f[j]
    L = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, L)
    Ff = np.fft.rfft(f, L)
    h = np.fft.irfft(W * np.conj(Ff), L)[:n]
    # Gradient w.r.t. z
    mean_fh = np.dot(f, h)
    grad_z = 2.0 * f * (h - mean_fh)
    return F, grad_z

def optimize_logsumexp(f_init, tau_schedule, max_iter_per_tau=200, time_limit=300):
    """Optimize using logsumexp with continuation."""
    f = f_init.copy()
    f = np.maximum(f, 1e-10)
    f = f / f.sum()
    best_f = f.copy()
    best_obj = 2.0 * len(f) * fast_convolve(f).max()
    
    z = np.log(f + 1e-15)  # initial z (since f = softmax(z), z = log(f) up to a constant)
    
    t0 = time.monotonic()
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj_grad(z_eval):
            return logsumexp_objective_and_grad(z_eval, tau)
        
        result = minimize(
            lambda z_eval: obj_grad(z_eval)[0],
            z,
            jac=lambda z_eval: obj_grad(z_eval)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter_per_tau, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        
        z = result.x
        f_new = softmax(z)
        obj = 2.0 * len(f_new) * fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
        f = f_new
        print(f"  tau={tau:.6f}, obj={obj:.10f}, L-BFGS iters={result.nit}")
    
    return best_f, best_obj

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining
logsumexp-continuation optimization (smooth approximation of the max) with
LP-Frank-Wolfe refinement. The logsumexp method uses L-BFGS on a softmax
parameterization with FFT-based gradient computation, enabling efficient
optimization of very long sequences. Multi-length exploration seeks better
local minima.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Compute softmax of z."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_convolve_grad(z, tau):
    """
    Compute the logsumexp of the auto-convolution and its gradient w.r.t. z.
    f = softmax(z), c = f*f (linear convolution), F = tau * log(sum(exp(c/tau)))
    """
    f = softmax_fn(z)
    n = len(f)
    c = fast_convolve(f)
    c_scaled = c / tau
    c_max = c_scaled.max()
    exp_shifted = np.exp(c_scaled - c_max)
    log_sum_exp = c_max + np.log(exp_shifted.sum())
    F = tau * log_sum_exp
    # Softmax weights
    w = exp_shifted / exp_shifted.sum()
    # Cross-correlation: h[i] = sum_j w[j+i] * f[j] for i in [0, n-1]
    L = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, L)
    Ff = np.fft.rfft(f, L)
    h = np.fft.irfft(W * np.conj(Ff), L)[:n]
    # Gradient w.r.t. z: 2 * f * (h - dot(f, h))
    dot_fh = np.dot(f, h)
    grad_z = 2.0 * f * (h - dot_fh)
    return F, grad_z


def lse_obj_func(z, tau):
    """Objective function for scipy.optimize.minimize."""
    F, _ = lse_convolve_grad(z, tau)
    return F


def lse_grad_func(z, tau):
    """Gradient function for scipy.optimize.minimize."""
    _, g = lse_convolve_grad(z, tau)
    return g


def objective_value(f):
    """Compute 2*n*max(conv) for sum-normalized f."""
    n = len(f)
    c = fast_convolve(f)
    return 2.0 * n * c.max()


def logsumexp_optimize(f_init, time_limit, verbose=True, verbose_tag="lse"):
    """
    Optimize using logsumexp continuation with L-BFGS.
    f_init should be sum-normalized and non-negative.
    """
    f = np.maximum(f_init.copy(), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = objective_value(f)
    
    # Initial tau based on current convolution spread
    c = fast_convolve(f)
    c_max = c.max()
    c_min = c.min()
    spread = c_max - c_min
    if spread < 1e-15:
        spread = c_max
    tau = c_max * 0.3
    
    # We'll decrease tau geometrically
    tau_min = c_max * 0.0005
    tau_ratio = 0.7  # multiply tau by this each step
    
    t0 = time.monotonic()
    it_tau = 0
    
    # Initial z: log(f) (up to a constant, which softmax handles)
    z = np.log(f + 1e-30)
    
    while time.monotonic() - t0 < time_limit:
        it_tau += 1
        if tau < tau_min:
            break
        
        # Run L-BFGS for this tau
        max_iter = 300 if it_tau <= 3 else 200
        options = {
            'maxiter': max_iter,
            'ftol': 1e-14,
            'gtol': 1e-10,
            'maxls': 50
        }
        
        try:
            result = minimize(
                lse_obj_func, z,
                jac=lse_grad_func,
                args=(tau,),
                method='L-BFGS-B',
                options=options
            )
            z_new = result.x
        except Exception as e:
            if verbose:
                print(f"  [{verbose_tag}] L-BFGS error at tau={tau:.8f}: {e}")
            tau *= tau_ratio
            continue
        
        f_new = softmax_fn(z_new)
        obj_new = objective_value(f_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
        
        # Update z for next iteration
        # Only accept if the logsumexp objective decreased
        F_new, _ = lse_convolve_grad(z_new, tau)
        F_old, _ = lse_convolve_grad(z, tau)
        if F_new <= F_old + 1e-10:
            z = z_new
        
        if verbose:
            print(f"  [{verbose_tag}] tau={tau:.8f}, max_conv={c_max:.8f}, obj={best_obj:.10f}, "
                  f"lte_iters={result.nit}, msg={result.message[:40]}")
        
        # Decrease tau
        tau *= tau_ratio
        # Also update c_max estimate
        c = fast_convolve(f_new)
        c_max = c.max()
        tau_min = c_max * 0.0005
        
        # Check if we're making progress
        if verbose:
            print(f"  [{verbose_tag}]   new tau={tau:.8f}, tau_min={tau_min:.8f}")
        
        # If tau is very small relative to c_max, we're in the "sharp" regime
        # and should switch to LP-FW
        if tau < c_max * 0.005:
            break
    
    return best_f, best_obj


def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_lp_matrix(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, m_expand_start=1.0):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    m_exp = m_expand_start
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                # Perturbation
                rng = np.random.default_rng(it * 17 + 3)
                scale = 0.003 * min(m_exp, 3.0)
                f = np.maximum(best_f + rng.standard_normal(n) * scale, 0)
                f = f / f.sum()
                c = fast_convolve(f)
                M = c.max()
                m_exp = 1.0
                no_improve = 0
            continue
        
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.05
            continue
        g = g / g_sum
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.1, 5.0)
            if no_improve > 80:
                break
            continue
        else:
            if diff > 1e-5:
                no_improve = 0
            else:
                no_improve += 1
        
        # Line search
        best_alpha = 0
        best_alpha_obj = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            c_new = fast_convolve(f_new)
            M_new = c_new.max()
            if M_new < best_alpha_obj:
                best_alpha_obj = M_new
                best_alpha = alpha
        
        if best_alpha > 0:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0)
            f = f / f.sum()
            c = fast_convolve(f)
            M = c.max()
            m_exp = 1.0
            
            current_obj = 2.0 * n * M
            if current_obj < best_obj:
                best_obj = current_obj
                best_f = f.copy()
        
        if verbose and it % 10 == 0:
            print(f"  LP-FW iter {it}: obj={best_obj:.10f}, M={M:.10f}, m_exp={m_exp:.2f}")
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def peak_shaving(f0, time_limit):
    """Peak-targeted coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = 2.0 * n * c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 50:
        c = fast_convolve(f)
        M = c.max()
        improved = False
        
        top_k = min(8, len(c))
        peak_indices = np.argsort(c)[-top_k:]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue
            
            order = np.argsort(products)[::-1]
            for oi in range(min(3, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue
                
                for mult in [0.6, 0.8, 1.2, 1.5]:
                    new_val = old_val * mult
                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        non_peak = np.ones(n, dtype=bool)
                        non_peak[js:je + 1] = False
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w = f[non_peak]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
        
        if not improved:
            no_improve += 1
            # Try reducing the main peak contributors
            k_peak = int(np.argmax(fast_convolve(f)))
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            trial = f.copy()
            trial[js:je + 1] *= 0.95
            d = f.sum() - trial.sum()
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[js:je + 1] = False
            if non_contrib.sum() > 0:
                nc_w = f[non_contrib]
                nw = nc_w.sum()
                if nw > 1e-15:
                    trial[non_contrib] += d * nc_w / nw
            trial = np.maximum(trial, 0)
            trial = trial / trial.sum()
            obj = 2.0 * n * fast_convolve(trial).max()
            if obj < best_obj:
                f = trial
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj
