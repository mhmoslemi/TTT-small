```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric (palindromic) parameterization to halve problem size
2. L-BFGS on smoothed max with adaptive temperature and FFT-based gradients
3. LP-based direction finding for refinement
4. Fine coordinate descent polish
Exploits the fact that the optimal solution is palindromic.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def make_symmetric(x):
    """Create palindromic sequence from first half."""
    return np.concatenate([x, x[::-1]])

def w_to_a(w):
    """Convert log-parameter w (length m) to palindromic sequence a (length 2m) on simplex."""
    m = len(w)
    w_shifted = w - w.max()
    e = np.exp(w_shifted)
    x = e / (2.0 * e.sum())
    return make_symmetric(x)

def a_to_w(a):
    """Convert palindromic sequence a (sum=1, length 2m) to log-parameter w (length m)."""
    m = len(a) // 2
    x = a[:m]
    return np.log(np.maximum(x, 1e-30))

def compute_grad_a_fft(a, weights):
    """Compute gradient of sum_k weights[k]*conv_k w.r.t. a, using FFT."""
    n = len(a)
    # (h convolve a_flipped)[i + n - 1] = sum_j weights[i+j] * a[j]
    conv = np.convolve(weights, a[::-1], mode='full')
    return 2.0 * conv[n-1:n-1+n]

def smoothed_obj_grad(w, tau=20.0):
    """Objective and gradient for smoothed max of convolution, w.r.t. log-parameters w."""
    a = w_to_a(w)
    n = len(a)
    m = n // 2
    x = a[:m]
    conv = fast_convolve(a)
    
    scaled = tau * conv
    scaled_max = scaled - scaled.max()
    exp_s = np.exp(scaled_max)
    log_sum_exp = scaled.max() + np.log(exp_s.sum())
    f = log_sum_exp / tau
    
    weights = exp_s / exp_s.sum()
    grad_a = compute_grad_a_fft(a, weights)
    
    # Chain rule: grad_w[l] = x[l] * (grad_a[l] + grad_a[2m-1-l] - 2*(S1+S2))
    # where S1 = dot(grad_a[:m], x), S2 = dot(grad_a[m:][::-1], x)
    S1 = np.dot(grad_a[:m], x)
    S2 = np.dot(grad_a[m:][::-1], x)
    grad_w = x * (grad_a[:m] + grad_a[m:][::-1] - 2.0 * (S1 + S2))
    
    return f, grad_w

def lbbfgs_symmetric(a_init, time_limit, tau_start=5.0, tau_end=200.0, verbose=True, label=""):
    """
    L-BFGS optimization of smoothed max on symmetric parameterization.
    a_init: initial palindromic sequence (sum=1)
    Returns: (optimized_a, best_objective, best_w)
    """
    from scipy.optimize import minimize
    
    m = len(a_init) // 2
    w = a_to_w(a_init)
    
    best_a = a_init.copy()
    best_obj = fast_convolve(a_init).max()
    best_w = w.copy()
    
    t0 = time.monotonic()
    
    # Adaptive temperature: increase tau over time
    n_steps = 200
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        # Anneal temperature
        frac = step / max(n_steps - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Run L-BFGS for a few iterations at this temperature
        result = minimize(
            smoothed_obj_grad, w, args=(tau,),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': 50, 'gtol': 1e-6, 'ftol': 1e-10}
        )
        
        w = result.x
        a = w_to_a(w)
        obj = fast_convolve(a).max()
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_w = w.copy()
        
        if verbose and step % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} step {step}: tau={tau:.1f}, obj={best_obj:.10f}, "
                  f"smoothed={result.fun:.10f}, time={elapsed:.1f}s")
    
    return best_a, best_obj, best_w

def lp_direction_sym(a, M):
    """
    LP: find symmetric direction g maximizing sum(g) s.t. (a*g)_k <= M, g >= 0.
    a is palindromic, length n. Returns g (palindromic) or None.
    """
    n = len(a)
    m = n // 2
    
    # Variable: y of length m, g = make_symmetric(y)
    # Constraint: (a*g)_k <= M for k = 0, ..., 2n-2
    # By symmetry, we only need k = 0, ..., n-1
    
    # (a*g)_k = sum_j a_j * g_{k-j}
    # g_p = y_{min(p, n-1-p)} for p in [0, n-1]
    
    C = np.zeros((n, m))
    for k in range(n):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            p = k - j
            y_idx = min(p, n - 1 - p)
            C[k, y_idx] += a[j]
    
    c_obj = -np.ones(m)  # maximize sum(y) = minimize -sum(y)
    
    res = linprog(c_obj, A_ub=C, b_ub=np.full(n, M),
                  bounds=[(0, None)] * m, method='highs')
    
    if res.status != 0:
        return None
    
    y = res.x
    g = make_symmetric(y)
    return g

def lp_iterate(a, time_limit, max_iters=200, verbose=True, label=""):
    """
    LP-based direction finding with line search.
    """
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a /= a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        g = lp_direction_sym(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        # Rescale g to have same sum as a
        g_rescaled = g * (a.sum() / g_sum)
        
        # Line search: find t that minimizes max(((1-t)*a + t*g_rescaled)^2)
        # (1-t)*a + t*g_rescaled has sum = 1 for all t
        best_t = 0.0
        best_t_obj = M
        
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - t) * a + t * g_rescaled
            a_trial = np.maximum(a_trial, 0)
            a_trial /= a_trial.sum()
            obj_trial = fast_convolve(a_trial).max()
            if obj_trial < best_t_obj:
                best_t_obj = obj_trial
                best_t = t
        
        if best_t < 1e-10:
            # Try smaller steps
            for t in [0.01, 0.02, 0.03, 0.05]:
                a_trial = (1 - t) * a + t * g_rescaled
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < best_t_obj:
                    best_t_obj = obj_trial
                    best_t = t
        
        if best_t < 1e-10:
            break
        
        a = (1 - best_t) * a + best_t * g_rescaled
        a = np.maximum(a, 0)
        a /= a.sum()
        
        # Symmetrize
        a = 0.5 * (a + a[::-1])
        a /= a.sum()
        
        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose and it % 10 == 0:
                print(f"    {label} LP iter {it}: obj={best_obj:.10f}, t={best_t:.3f}")
        elif it > 20:
            break
    
    return best_a, best_obj
