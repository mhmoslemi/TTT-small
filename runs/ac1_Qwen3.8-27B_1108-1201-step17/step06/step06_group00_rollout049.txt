def fast_convolve(a, b=None):
    """Full convolution via FFT. If b is None, auto-convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def fast_autoconv(a):
    """Auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fixed-point iteration
with precomputed sparse structure, adaptive damping, multi-length exploration,
symmetry exploitation, and coordinate descent polish.

Core idea: At optimality, f must satisfy f is a fixed point of the mapping
f -> LP_best_response(f), where the LP solves:
  max sum(g) s.t. conv(f,g)_k <= ||f*f||_inf for all k, g >= 0
This is a principled approach since at a local minimum, f is a best response
to itself (complementary slackness).

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_autoconv(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return float('inf')
    b = fast_autoconv(a)
    return float(2.0 * n * b.max() / (S * S))


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def precompute_lag_col(n):
    """Precompute (lag, col) pairs for the convolution constraint matrix."""
    lags = []
    cols = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            lags.append(k)
            cols.append(j)
    return np.array(lags, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_fast(f, M, lag_col, active_mask=None):
    """
    Solve LP: max sum(g) s.t. conv(f,g)_k <= M for active lags, g >= 0.
    Uses precomputed sparse structure.
    """
    n = len(f)
    lags, cols = lag_col
    
    if active_mask is not None:
        # Filter to active constraints
        # lag_col has one entry per (lag, col) pair
        # We need to filter by lag
        active_set = set(active_mask.tolist())
        keep = np.array([lag in active_set for lag in lags], dtype=bool)
        lags_f = lags[keep]
        cols_f = cols[keep]
        n_constr = len(active_mask)
    else:
        lags_f = lags
        cols_f = cols
        n_constr = 2 * n - 1
    
    if len(lags_f) == 0:
        return np.zeros(n)
    
    # Values: A[entry] = f[lag - col]
    values = f[lags_f - cols_f]
    
    A = csr_matrix((values, (lags_f, cols_f)), shape=(n_constr, n))
    
    c = -np.ones(n)
    b = np.full(n_constr, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    
    if res.status != 0:
        return None
    
    return res.x


def lp_iteration(f, lag_col, step_size=0.5, active_fraction=1.0):
    """One LP-based fixed-point iteration. f normalized to sum=sqrt(2n)."""
    n = len(f)
    target_sum = np.sqrt(2.0 * n)
    
    S = f.sum()
    if S < 1e-10:
        return f
    f = f / S * target_sum
    
    ff = fast_autoconv(f)
    M = ff.max()
    
    if M < 1e-15:
        return f
    
    # Select active lags
    num_lags = 2 * n - 1
    n_active = max(5, int(active_fraction * num_lags))
    
    if n_active < num_lags:
        # Top n_active lags by convolution value
        top_idx = np.argpartition(ff, -n_active)[-n_active:]
        active_mask = top_idx
    else:
        active_mask = None
    
    g = solve_lp_fast(f, M, lag_col, active_mask)
    if g is None:
        return f
    
    g_sum = g.sum()
    if g_sum < 1e-10:
        return f
    g = g / g_sum * target_sum
    
    f_new = (1.0 - step_size) * f + step_size * g
    f_new = np.maximum(f_new, 0.0)
    
    S_new = f_new.sum()
    if S_new < 1e-10:
        return f
    f_new = f_new / S_new * target_sum
    
    return f_new

    t0 = time.monotonic()

    best_f = f.copy()
    best_obj = compute_objective(f)
    
    step = step_init
    frac = active_fraction
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        f = lp_iteration(f, lag_col, step_size=step, active_fraction=frac)
        it += 1
        
        obj = compute_objective(f)
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"      LP it={it}, obj={obj:.8f}, step={step:.3f}, frac={frac:.2f}, t={elapsed:.0f}s")
        else:
            no_improve += 1
        
        # Adaptive step size
        if no_improve > 30:
            step = max(0.05, step * 0.85)
            no_improve = 0
        elif no_improve == 0:
            step = min(0.9, step * 1.03)
        
        # Grow active fraction (use more constraints as we converge)

            if S > 1e-10:

            step = step_init
            frac = active_fraction

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_polish(a0, time_limit):
    """Coordinate descent: scale individual elements to reduce max convolution."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        return a0, compute_objective(a0)
    a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_maxconv = fast_autoconv(a).max()
    
    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0])
    
    max_passes = 20
    for _pass in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, 2.0 * n * best_maxconv
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_val = old_val
            best_j_maxconv = best_maxconv
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial_sum = a_trial.sum()
                if a_trial_sum < 1e-15:
                    continue
                a_trial /= a_trial_sum
                mc = fast_autoconv(a_trial).max()
                if mc < best_j_maxconv - 1e-15:
                    best_j_maxconv = mc
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a /= a.sum()
                if best_j_maxconv < best_maxconv - 1e-15:
                    best_maxconv = best_j_maxconv
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, 2.0 * n * best_maxconv


def make_gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization, normalized to sum=1."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    """Raised cosine window, normalized to sum=1."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

        t_alloc = min(rem, 500.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, obj_lp = lp_optimize(
            parent, time_limit=t_alloc,
            step_init=0.5, active_fraction=0.7,
            verbose=True, seed=seed
        )
        try_seq(a_lp, f"lp_parent_n{n_parent}")
    
    # === Phase 2: Re-run LP with different parameters ===
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 400
        if rem > 100:
            print(f"\n=== Phase 2: LP refined (n={n_best}) ===")
            t_alloc = min(rem, 300.0)
            print(f"  Budget: {t_alloc:.0f}s")
            # Use higher active fraction for more precise optimization
            a_lp2, obj_lp2 = lp_optimize(
                best_seq, time_limit=t_alloc,
                step_init=0.3, active_fraction=1.0,
                n_active_grow=1.0, verbose=True,
                seed=seed + 1000
            )
            try_seq(a_lp2, f"lp_refined_n{n_best}")
    
    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = set()
        for factor in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
            tl = int(n_parent * factor)
            tl = max(256, min(3000, tl))
            target_lengths.add(tl)
        
        # Also try some round numbers
        for tl in [512, 768, 1024, 1536, 2048]:
            if 256 <= tl <= 3000:
                target_lengths.add(tl)
        
        # Sort by distance from current best length
        n_cur = len(best_seq)
        target_lengths = sorted(target_lengths, key=lambda x: abs(x - n_cur))
        
        for target_n in target_lengths[:5]:
            if target_n == n_cur:
                continue
            rem = deadline - time.monotonic() - 200
            if rem < 60:
                break
            
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 120.0)
            if t_alloc < 30:
                continue
            
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_optimize(
                f_interp, time_limit=t_alloc,
                step_init=0.5, active_fraction=0.7,
                verbose=False, seed=seed + target_n % 10000
            )
            try_seq(a_opt, f"length_n{target_n}")
    
    # === Phase 4: Diverse initializations ===
    print(f"\n=== Phase 4: Diverse initializations (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_base = len(best_seq)
        if n_base < 256:
            n_base = 1024
        
        templates = {
            "gauss_0.2": make_gaussian_init(n_base, 0.2),
            "gauss_0.4": make_gaussian_init(n_base, 0.4),
            "cosine": make_cosine_init(n_base),
            "uniform": make_uniform_init(n_base),
            "sym_gauss": make_symmetric_init(n_base, make_gaussian_init(n_base, 0.3)),
            "sym_cosine": make_symmetric_init(n_base, make_cosine_init(n_base)),
        }
        
        # Add perturbed best
        templates["perturbed_best"] = project_simplex(
            best_seq + rng.standard_normal(n_base) * 0.02)
        
        for name, init_a in templates.items():
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t_alloc = min(rem * 0.25, 60.0)
            if t_alloc < 15:
                continue
            
            print(f"  {name} (t={t_alloc:.0f}s)")
            a_opt, _ = lp_optimize(
                init_a, time_limit=t_alloc,
                step_init=0.5, active_fraction=0.7,
                verbose=False, seed=seed + hash(name) % 50000
            )
            try_seq(a_opt, f"diverse_{name}")
    
    # === Phase 5: Final LP polish on best ===
    print(f"\n=== Phase 5: Final LP polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem, 150.0)
            print(f"  Final LP n={n_best}, t={t_alloc:.0f}s")
            a_fin, _ = lp_optimize(
                best_seq, time_limit=t_alloc,
                step_init=0.2, active_fraction=1.0,
                n_active_grow=1.0, verbose=True,
                seed=seed + 99999
            )
            try_seq(a_fin, "final_lp")
    
    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_cd = min(rem, 80.0)
            print(f"  CD polish t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd_polish")
    
    # === Phase 7: One more LP pass if time remains ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 40:
            t_alloc = min(rem, 80.0)
            n_best = len(best_seq)
            print(f"\n=== Phase 7: Final final LP (t={t_alloc:.0f}s) ===")
            a_ff, _ = lp_optimize(
                best_seq, time_limit=t_alloc,
                step_init=0.1, active_fraction=1.0,
                verbose=False, seed=seed + 123456
            )
            try_seq(a_ff, "final_final_lp")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")
    
    return [float(x) for x in best_seq]
