```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) momentum subgradient, (2) sparse-constraint LP-Frank-Wolfe,
(3) structured simulated annealing, (4) pairwise coordinate descent,
(5) adaptive multi-start.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / n
    idx = int(np.nonzero(cond)[0][-1])
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def compute_subgradient(f, peak_indices, peak_weights):
    """Compute weighted subgradient at given peaks."""
    n = len(f)
    g = np.zeros(n, dtype=np.float64)
    for k, w in zip(peak_indices, peak_weights):
        k = int(k)
        if w < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * w * f[k - idx]
    return g


def momentum_subgradient(f0, time_limit, lr_init=0.01, momentum=0.9, 
                         num_peaks=20, verbose=False):
    """Subgradient method with heavy-ball momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = obj_normalized(f)
    v = np.zeros(n, dtype=np.float64)  # momentum
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Select top peaks
        K = min(num_peaks, len(conv))
        sorted_idx = np.argpartition(conv, -K)[-K:]
        peak_vals = conv[sorted_idx]
        
        # Exponential weights based on proximity to max
        temp = 0.02
        w = np.exp((peak_vals - M) / temp)
        w = w / w.sum()
        
        # Compute subgradient
        g = compute_subgradient(f, sorted_idx, w)
        
        # Momentum update
        v = momentum * v + g
        f_new = f - lr * v
        f_new = project_simplex(f_new)
        
        obj = obj_normalized(f_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Adaptive learning rate
        if no_improve > 30:
            lr *= 0.7
            no_improve = 0
        lr = max(lr, 1e-6)
        
        if verbose and it % 100 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        
        if lr < 1e-7:
            break
    
    return best_f, best_obj


def sparse_lp_fw(f0, time_limit, num_constraints=80, verbose=False):
    """LP-Frank-Wolfe using only top-K constraints (sparse)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = obj_normalized(f)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_mult = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max() * m_mult
        
        # Select top constraints
        K = min(num_constraints, 2 * n - 1)
        sorted_idx = np.argpartition(conv, -K)[-K:]
        sorted_idx = sorted(sorted_idx, key=lambda x: -conv[int(x)])
        
        # Build sparse LP: max sum(g) s.t. (f*g)[k] <= M for k in sorted_idx, g >= 0
        # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
        # This is a matrix A @ g <= M where A[k_idx, j] = f[j] if k[j]-j is valid
        
        A_ub = np.zeros((K, n), dtype=np.float64)
        b_ub = np.full(K, M, dtype=np.float64)
        
        for row, k in enumerate(sorted_idx):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            A_ub[row, js:je+1] = f[js:je+1][::-1]
        
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(1.0, (time_limit - (time.monotonic()-t0)) / max(1, int((time_limit-0))))})
        
        if not res.success:
            m_mult = min(m_mult * 1.1, 5.0)
            continue
        
        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_mult = min(m_mult * 1.05, 3.0)
            continue
        g = g / g_sum
        
        # Check if direction is trivial
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 2
            if no_improve > 30:
                m_mult = min(m_mult * 1.05, 4.0)
            if no_improve > 100:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_alpha_val = obj_normalized(f)
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_trial = (1 - alpha) * f + alpha * g
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            val = obj_normalized(f_trial)
            if val < best_alpha_val:
                best_alpha_val = val
                best_alpha = alpha
        
        if best_alpha < 1e-8:
            no_improve += 1
            m_mult = min(m_mult * 1.03, 2.0)
        else:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0)
            f = f / f.sum()
            m_mult = 1.0
            
            cur_obj = obj_normalized(f)
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if verbose and it % 20 == 0:
            print(f"    SLW iter {it}: obj={best_obj:.10f}, m_mult={m_mult:.2f}")
        
        if no_improve > 150:
            break
    
    return best_f, best_obj


def simulated_annealing(f0, time_limit, verbose=False):
    """Structured simulated annealing with peak-targeted perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_obj = obj_normalized(f)
    
    t0 = time.monotonic()
    T = 0.05  # initial temperature
    cooling = 0.9997
    rng = np.random.default_rng(12345)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit and T > 1e-5:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        k_peak = int(np.argmax(conv))
        
        # Identify peak contributors
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        
        # Perturb peak contributors more
        noise = rng.standard_normal(n) * 0.01
        noise[js:je+1] *= 2.0  # double perturbation at peak
        f_new = project_simplex(f + noise)
        
        new_obj = obj_normalized(f_new)
        delta = new_obj - obj_normalized(f)
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            f = f_new
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
        else:
            no_improve += 1
        
        T *= cooling
        
        if no_improve > 200:
            T *= 0.5  # faster cooling when stuck
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    SA iter {it}: obj={best_obj:.10f}, T={T:.6f}")
    
    return best_f, best_obj


def pairwise_cd(f0, time_limit, verbose=False):
    """Pairwise coordinate descent: shift mass between pairs of indices."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_obj = obj_normalized(f)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit and no_improve < 50:
        it += 1
        conv = fast_convolve(f)
        k_peak = int(np.argmax(conv))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        
        # Try shifting mass from peak contributors to non-peak positions
        improved = False
        
        # Get sorted peak contributors by their contribution
        idx_peak = np.arange(js, je + 1)
        products = f[idx_peak] * f[k_peak - idx_peak]
        top_contrib = idx_peak[np.argsort(products)[::-1]][:5]
        
        non_peak_mask = np.ones(n, dtype=bool)
        non_peak_mask[js:je+1] = False
        non_peak_indices = np.where(non_peak_mask)[0]

            if f[i] < 1e-6:
                continue

            if len(non_peak_indices) == 0:

                f_trial[i] -= delta

                if f_trial[i] < 0 or f_trial[j] < 0:

                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                obj = obj_normalized(f_trial)

                f_trial[i] -= delta
                f_trial[j] += delta
                f_trial = np.maximum(f_trial, 0)

                    best_obj = obj

                    f = f_trial
                    improved = True
                    break

                i = top_contrib[rng.integers(len(top_contrib))] if len(top_contrib) > 0 else 0
                if f[i] < 1e-6:
                    continue
                j = non_peak_indices[rng.integers(len(non_peak_indices))] if len(non_peak_indices) > 0 else 0

                obj = obj_normalized(f_trial)

def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
