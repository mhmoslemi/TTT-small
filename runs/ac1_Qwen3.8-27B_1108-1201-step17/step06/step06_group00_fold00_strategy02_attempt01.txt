def project_simplex(v):
    """Euclidean projection onto the simplex."""
    n = len(v)
    sorted_vals = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_vals) - sum(sorted_vals) / n
    rho = np.argmin(cssv >= 0) + 1
    if rho <= 1:
        return np.ones(n) / n
    else:
        theta = (cssv[rho-1] / rho)
        return np.maximum(v - theta, 0.0)
