LP-based optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses constraint generation LP to find optimal descent directions,
combined with diverse initializations and multi-scale exploration.
"""

import cvxpy as cp
from scipy.fft import rfft, irfft
import time

# Global variables for tracking
_best_seq_global = None
_best_score_global = float('inf')

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute objective locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))

    t = cp.Variable()

        hi = min(n - 1, k)

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, b):
    """Linear cross-correlation: (a*b)[k] = sum_i a[i]*b[k-i]."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local_maxconv(a):
    """Return max of auto-convolution (assuming sum(a)=1)."""
    b = fast_convolve(a)
    return float(b.max())


def make_tukey(n, r=0.5):
    """Tukey (tapered cosine) window, normalized to sum 1."""
    a = np.ones(n)
    taper = int(r * n // 2)
    if taper > 0:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped, normalized to sum 1."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

    """Raised cosine, normalized to sum 1."""
    x = np.linspace(0, 1, n)

    a = np.maximum(a, 1e-15)
    return a / a.sum()


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, a_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            w_sel = w[top_idx]
            w_sel /= w_sel.sum()
            w_full = np.zeros(len(b))
            w_full[top_idx] = w_sel
            w = w_full

        # Subgradient
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * a[k - idx]

        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        obj_new = eval_local_maxconv(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new

        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95

        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate descent: try scaling each element."""
    t0 = time.monotonic()
    a = a0.copy() / a0.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = eval_local_maxconv(a)

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.8, 2.5])
    improved_global = True

    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = eval_local_maxconv(a_trial)
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase LP-based optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Solver list (try in order)
    solver_list = [cp.GLOP, cp.CBC, cp.GLPK]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = make_tukey(1152, 0.5)
        print("No parent, using Tukey n=1152")

    n_parent = len(parent)
    print(f"\n{'='*60}")
    print(f"Starting optimization. Parent n={n_parent}, score~{best_score:.6f}")
    print(f"{'='*60}\n")

    # === Phase 1: LP optimization on parent (main) ===
    print(f"=== Phase 1: LP minimize_cross on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 10:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj_opt = lp_optimize(
            parent, time_limit=t_alloc,
            solver_list=solver_list,
            method="minimize_cross",
            top_k_frac=0.05,
            verbose=True
        )
        try_seq(a_opt, "lp_mincross_parent")

    # === Phase 2: LP state-of-art on current best ===
    print(f"\n=== Phase 2: LP state-of-art (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.2, 300.0)
        if t_alloc > 10:
            print(f"  Allocating {t_alloc:.0f}s")
            a_opt, obj_opt = lp_optimize(
                best_seq, time_limit=t_alloc,
                solver_list=solver_list,
                method="state_of_art",
                top_k_frac=0.05,
                verbose=True
            )
            try_seq(a_opt, "lp_sota_best")

    # === Phase 3: Diverse initializations with LP ===
    print(f"\n=== Phase 3: Diverse initializations (best={best_score:.6f}) ===")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits = {
        "tukey_0.3": make_tukey(n_base, 0.3),
        "tukey_0.5": make_tukey(n_base, 0.5),
        "tukey_0.7": make_tukey(n_base, 0.7),
        "tukey_1.0": make_tukey(n_base, 1.0),
        "gauss_0.25": make_gaussian(n_base, 0.25),
        "gauss_0.4": make_gaussian(n_base, 0.4),
        "cosine": make_cosine(n_base),
    }

    for name, init_a in inits.items():
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.08, 40.0)
        if t_alloc < 5:
            continue
        # Quick subgradient warmup then LP
        t_sg = t_alloc * 0.3
        t_lp = t_alloc * 0.7
        a_warm, _ = subgradient_optimize(
            init_a, time_limit=t_sg,
            lr_init=0.01, momentum=0.9,
            temp_init=0.005, temp_growth=1.05,
            kick_interval=500, kick_size=0.005,
            n_top=max(10, n_base // 20),
            seed=seed + hash(name) % 10000
        )
        try_seq(a_warm, f"sg_{name}")
        if best_seq is not None and len(best_seq) == n_base:
            a_opt, _ = lp_optimize(
                best_seq, time_limit=t_lp,
                solver_list=solver_list,
                method="minimize_cross",
                top_k_frac=0.05,
                verbose=False
            )
            try_seq(a_opt, f"lp_{name}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.75, 1.25, 1.5, 2.0, 0.5]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths and tl != n_parent:
                target_lengths.append(tl)

        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            f_interp = interpolate_to_length(best_seq, target_n)
            t_alloc = min(rem * 0.15, 60.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_optimize(
                f_interp, time_limit=t_alloc,
                solver_list=solver_list,
                method="minimize_cross",
                top_k_frac=0.05,
                verbose=False
            )
            try_seq(a_opt, f"length_n{target_n}")

    # === Phase 5: Deep LP on best ===
    print(f"\n=== Phase 5: Deep LP on best (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.2, 300.0)
        if t_alloc > 10:
            print(f"  Deep LP n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = lp_optimize(
                best_seq, time_limit=t_alloc,
                solver_list=solver_list,
                method="minimize_cross",
                top_k_frac=0.08,
                verbose=True
            )
            try_seq(a_opt, "deep_lp")

    # === Phase 6: State-of-art LP on best ===
    print(f"\n=== Phase 6: SOTA LP on best (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.15, 150.0)
        if t_alloc > 10:
            print(f"  SOTA LP t={t_alloc:.0f}s")
            a_opt, _ = lp_optimize(
                best_seq, time_limit=t_alloc,
                solver_list=solver_list,
                method="state_of_art",
                top_k_frac=0.08,
                verbose=True
            )
            try_seq(a_opt, "sota_lp_deep")

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: CD polish (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.1, 60.0)
        if t_cd > 5:
            print(f"  CD t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd_polish")

    # === Phase 8: Final micro-LP ===
    print(f"\n=== Phase 8: Final micro-LP (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.5, 60.0)
        if t_alloc > 5:
            print(f"  Micro LP t={t_alloc:.0f}s")
            a_opt, _ = lp_optimize(
                best_seq, time_limit=t_alloc,
                solver_list=solver_list,
                method="minimize_cross",
                top_k_frac=0.1,
                verbose=False
            )
            try_seq(a_opt, "micro_lp")

            # Perturbation + re-LP
            rem = deadline - time.monotonic()
            if rem > 10:
                n_best = len(best_seq)
                for attempt in range(3):
                    rem = deadline - time.monotonic()
                    if rem < 5:
                        break
                    noise = rng.standard_normal(n_best) * 0.002
                    a_pert = project_simplex(best_seq + noise)
                    t_alloc = min(rem * 0.3, 20.0)
                    a_p, _ = lp_optimize(
                        a_pert, time_limit=t_alloc,
                        solver_list=solver_list,
                        method="minimize_cross",
                        top_k_frac=0.05,
                        verbose=False
                    )
                    try_seq(a_p, f"perturb_lp_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
