    Iteratively flatten the spectrum of a to reduce auto-correlation peaks.
    """

    a = a / a.sum()
    n = len(a)

    for it in range(n_iter):

        power = np.abs(A)**2
        # Target: flatten the power spectrum (excluding DC)
        # DC component is fixed: |A[0]|^2 = (sum a)^2 = 1
        power_non_dc = power[1:]
        target_avg = power_non_dc.mean()
        # Blend current and target
        power_new = power.copy()

        # Project to non-negative simplex

        a_new = a_new / a_new.sum()
        a = a_new
    
    return a

The auto-correlation c[k] = conv(a,a)[k] for k = 0, ..., 2n-2.

The peak of c is what we want to minimize. The "flatness" of c is related to the "flatness" of |A|^2.

1. Compute A = FFT_L(a)
2. Compute |A|^2

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_flatten(a, time_limit, alpha_init=0.3, verbose=False):
    """
    Iterative spectral flattening: work in frequency domain to flatten the power spectrum,
    which helps reduce auto-correlation peaks.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    L = 1 << (2 * n - 1).bit_length()
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    alpha = alpha_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        A = np.fft.rfft(a, L)
        power = np.abs(A) ** 2
        phase = np.angle(A)
        
        # Target: flatten non-DC power
        power_non_dc = power[1:]
        # Use a "water-filling" target: reduce power at high frequencies that contribute to peaks
        # Simple approach: blend towards flat
        target = np.median(power_non_dc)  # Use median for robustness
        power_new = power.copy()
        power_new[1:] = (1 - alpha) * power[1:] + alpha * target
        
        # Reconstruct
        A_new = np.sqrt(np.maximum(power_new, 0)) * np.exp(1j * phase)
        A_new[0] = A[0]  # Preserve DC (which determines the sum)
        a_new = np.fft.irfft(A_new, L)[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        # Evaluate
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    SpecFlat iter {it}: obj={2*n*obj:.8f}")
        else:
            no_improve += 1
            # Adapt alpha
            alpha *= 0.98
            if alpha < 0.01:
                alpha = 0.01
        
        a = a_new
        
        if no_improve > 200:
            # Try different strategy: flatten more aggressively at specific frequencies
            if verbose:
                print(f"    SpecFlat: restarting at iter {it}")
            a = best_a.copy()
            alpha = 0.1
            no_improve = 0
    
    return best_a, 2 * n * best_obj


def spectral_flatten_weighted(a, time_limit, verbose=False):
    """
    Weighted spectral flattening: identify which frequencies contribute most to the peaks
    and reduce them selectively.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        return a, 2 * n * fast_convolve(a).max()
    a = a / s
    
    L = 1 << (2 * n - 1).bit_length()
    n_freq = L // 2 + 1
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Identify peak lags
        peak_threshold = 0.8 * M
        peak_lags = np.where(conv >= peak_threshold)[0]
        
        A = np.fft.rfft(a, L)
        power = np.abs(A) ** 2
        phase = np.angle(A)
        
        # For each frequency m, compute how much it "builds up" at the peak lags
        # contribution of freq m to lag k: |A[m]|^2 * cos(2*pi*m*k/L) (approximately, for the real part)
        # We want to reduce frequencies that have positive contribution to peak lags
        reduction = np.zeros(n_freq)
        for m in range(1, n_freq):
            # Sum of contributions to peak lags
            k_vals = peak_lags.astype(float)
            contributions = power[m] * np.cos(2 * np.pi * m * k_vals / L)
            reduction[m] = contributions.sum()
        
        # Reduce power at frequencies with high positive contribution to peaks
        reduction = np.maximum(reduction, 0)
        if reduction[1:].max() > 1e-15:
            # Scale reduction to be a fraction of the power
            scale = 0.1 * power[1:].max() / (reduction[1:].max() + 1e-15)
            power_new = power.copy()
            power_new[1:] = np.maximum(power[1:] - scale * reduction[1:], 0)
        else:
            # Fallback: simple flattening
            target = np.median(power[1:])
            power_new = power.copy()
            power_new[1:] = 0.9 * power[1:] + 0.1 * target
        
        A_new = np.sqrt(power_new) * np.exp(1j * phase)
        A_new[0] = A[0]
        a_new = np.fft.irfft(A_new, L)[:n]
        
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        obj = fast_convolve(a_new).max()

            best_a = a_new.copy()
            if verbose and it % 20 == 0:

def adam_optimize_powermean(f0, time_limit, p=200, lr_init=0.005, verbose=False):
    """
    Adam optimization on the power-mean smooth surrogate.
    F(a) = (sum_k conv[a,k]^p)^(1/p) ≈ max(conv) for large p.
    """
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, F_val, conv = compute_smooth_grad(f, p, tau=None)
        M = conv.max()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Adaptive LR
        if no_improve > 50:
            lr *= 0.9
        if no_improve > 200:
            lr *= 0.7
        if no_improve > 500:
            # Restart from best with perturbation
            rng = np.random.default_rng(t * 7 + 3)
            noise = 0.003 * min(no_improve / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.5
            if verbose:
                print(f"    Adam-PM restart at {t}: obj={2*n*best_obj:.8f}")
        
        if verbose and t % 200 == 0:
            print(f"    Adam-PM iter {t}: obj={2*n*best_obj:.8f}, lr={lr:.6f}")
    
    return best_f, 2 * n * best_obj


def lbfgs_optimize(f0, time_limit, verbose=False):

    """

        nonlocal n_evals, tau

            return 0, np.zeros_like(x)

        w = np.exp(scaled)

        constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0, 'jac': lambda x: np.ones_like(x)}]

            options={'maxiter': 1000, 'maxfun': 5000, 'ftol': 1e-10, 'gtol': 1e-8}
        )
        
        f_new = np.maximum(result.x, 0)
        ss = f_new.sum()
        if ss > 1e-15:
            f_new = f_new / ss
            obj = fast_convolve(f_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f_new.copy()
                if verbose:
                    print(f"    L-BFGS round {r}: obj={2*n*obj:.8f}")
            f = f_new
        
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
    
    return best_f, 2 * n * best_obj


def subgradient_heavyball(f0, time_limit, verbose=False):
    """
    Subgradient method with heavy-ball momentum on the actual max objective.
    Uses a "soft" subgradient (weighted average over near-max positions).
    """
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    momentum = np.zeros(n)
    mu = 0.9  # Heavy-ball coefficient
    step_size = 0.01
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Soft subgradient: weight positions by exp(conv[k] / tau)
        tau = 0.02 * M
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute subgradient: grad[j] = 2 * sum_k w[k] * a[k-j] (approximately)
        w_rev = w[::-1]
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        W = np.fft.rfft(w_rev, fft_size)
        conv_r = np.fft.irfft(A * W, fft_size)
        h = conv_r[n - 1:2 * n - 1][::-1]
        grad = 2.0 * h
        
        # Heavy-ball update
        momentum = mu * momentum - step_size * grad
        f_new = project_simplex(f + momentum)
        
        # Adaptive step size
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            step_size = min(step_size * 1.05, 0.05)
        else:
            no_improve += 1
            step_size *= 0.95
            if no_improve > 100:
                step_size = max(step_size * 0.8, 0.001)
        
        f = f_new
        
        # Decay momentum slightly
        mu = max(0.8, mu * 0.9999)
        
        if no_improve > 300:
            # Restart
            rng = np.random.default_rng(it * 11 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            momentum = np.zeros(n)
            step_size = 0.01
            no_improve = 0
            if verbose:
                print(f"    SubGrad restart at {it}: obj={2*n*best_obj:.8f}")
        
        if verbose and it % 100 == 0:
            print(f"    SubGrad iter {it}: obj={2*n*best_obj:.8f}, step={step_size:.6f}")
    
    return best_f, 2 * n * best_obj


def local_search(f0, time_limit, verbose=False):
    """
    Diverse local search: coordinate descent, pairwise exchange, 
    block moves, and random perturbations.
    """
    f = np.maximum(f0.astype(np.float64), 0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        
        # Strategy 1: Coordinate descent on random elements
        if it % 3 == 0:
            for _ in range(10):
                j = rng.integers(0, n)
                old_val = f[j]
                if old_val < 1e-10:
                    continue
                for frac in [0.3, 0.5, 0.7, 1.3, 1.5]:
                    trial = f.copy()
                    trial[j] = old_val * frac
                    diff = old_val - trial[j]
                    # Redistribute
                    if diff != 0:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        if f[mask].sum() > 1e-10:
                            trial[mask] += diff * f[mask] / f[mask].sum()
                    trial = np.maximum(trial, 0)
                    ss = trial.sum()
                    if ss < 1e-10:
                        continue
                    trial = trial / ss
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
        
        # Strategy 2: Pairwise exchange
        elif it % 3 == 1:
            for _ in range(5):
                j, k = rng.integers(0, n, size=2)
                if j == k:
                    continue
                for eps in [0.001, 0.005, 0.01, 0.02]:
                    trial = f.copy()
                    transfer = eps * min(f[j], f[k])
                    if transfer < 1e-12:
                        continue
                    trial[j] -= transfer
                    trial[k] += transfer
                    if trial[j] < 0:
                        continue
                    ss = trial.sum()
                    if ss < 1e-10:
                        continue
                    trial = trial / ss
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
        
        # Strategy 3: Block move (move mass from a block to another)
        elif it % 3 == 2:
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            if abs(j - k) < 2:
                continue
            for eps in [0.3, 0.5, 0.7]:
                trial = f.copy()
                # Move mass from neighborhood of j to neighborhood of k
                r = max(1, n // 20)
                js = slice(max(0, j - r), min(n, j + r + 1))
                ks = slice(max(0, k - r), min(n, k + r + 1))
                mass = eps * trial[js].sum()
                if mass < 1e-12:
                    continue
                trial[js] -= mass * f[js] / f[js].sum() if f[js].sum() > 1e-10 else 0
                trial[ks] += mass * f[ks] / f[ks].sum() if f[ks].sum() > 1e-10 else 0
                trial = np.maximum(trial, 0)
                ss = trial.sum()
                if ss < 1e-10:
                    continue
                trial = trial / ss
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
        
        if not improved and it % 50 == 0:
            # Random perturbation
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
    
    return best_f, 2 * n * best_obj


def make_parametric_init(n, center=0.4, width=0.3, osc_freq=0, osc_amp=0.0):
    """
    Parametric initialization with optional oscillatory component.
    a[j] = gaussian(j; center, width) * (1 + osc_amp * sin(2*pi*osc_freq*j/n))
    """
    x = np.arange(n) / n
    x_center = x - center
    a = np.exp(-0.5 * (x_center / width) ** 2)
    if osc_amp > 0 and osc_freq > 0:
        a = a * (1 + osc_amp * np.sin(2 * np.pi * osc_freq * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_variational_init(n, seed=42):
    """
    Create a diverse set of parametric initializations.
    """
    rng = np.random.default_rng(seed)
    inits = []
    
    for center in [0.3, 0.4, 0.5, 0.6]:
        for width in [0.2, 0.25, 0.3, 0.4]:
            for osc_freq in [0, 2, 3, 4, 6]:
                for osc_amp in [0, 0.1, 0.2, 0.3]:
                    a = make_parametric_init(n, center, width, osc_freq, osc_amp)
                    if a.sum() > 1e-10:
                        inits.append(a)
    
    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
