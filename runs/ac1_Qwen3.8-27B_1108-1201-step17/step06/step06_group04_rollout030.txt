If it's achieved at multiple points k1, ..., kk, then the subdifferential is the convex hull of {grad c[k1], ..., grad c[kk]}, and any point in this convex hull is a valid subgradient. The standard choice is the average.

But for minimization, we want to use a subgradient that points in the direction of steepest descent. The most aggressive choice is the gradient of the single largest peak, not the average. This is because reducing the largest peak is the most important.

So I'll use: g = grad c[k*] where k* = argmax c.

If multiple peaks are tied (within numerical precision), I'll use the average.

```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(f):
    """Compute 2*n*max(conv) for sum-normalized f."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max(), conv
```

```python
def subgradient(f, conv, active_set=None):
    """Compute subgradient of max(conv) w.r.t. f."""
    n = len(f)
    if active_set is None:
        max_val = conv.max()
        active_set = np.where(conv >= max_val - 1e-12)[0]
    
    g = np.zeros(n)
    for k in active_set:
        k = int(k)
        # grad c[k][i] = f[k-i] if 0 <= k-i <= n-1
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            g[lo:hi+1] += f[k-hi:k-lo+1][::-1]
    g /= len(active_set)
    # Project onto tangent space (sum = 0)
    g -= g.mean()
    return g
```

```python
            k_lo = max(0, i)  # k-i >= 0 => k >= i
            k_hi = min(2*n-2, i+n-1)  # k-i <= n-1 => k <= i+n-1

            # For k in [k_lo, k_hi]:
            #   if k == 2i: quadratic term
            #   else: linear term = 2*f[k-i]
            
            # We want to minimize max(conv[k] + delta_k(x)) over x >= 0
            # where delta_k(x) is as above.
            
            # Strategy: binary search or ternary search on x
            # The function max_k (conv[k] + delta_k(x)) is convex in x (max of linear/quadratic functions)
            # So we can use ternary search or golden section.
            
            # Let's use a quick approach: try a few candidate x values.
            # The optimal x is where the max is minimized, which is typically
            # where two or more conv[k] values are equal (at the optimal x).
            
            # Simple approach: golden section search
            x_max = max(x_old * 2, 3.0/n)
            
            def eval_x(x):
                # Update conv in O(n)
                delta = 2*(x - x_old)
                conv_new = conv.copy()  # This is O(n) but we can avoid the copy
                # Actually, let's compute the max directly
                max_val = conv.max()  # Start with original max
                for k in range(k_lo, k_hi+1):
                    ki = k - i
                    if 0 <= ki <= n-1:
                        if k == 2*i:
                            new_ck = conv[k] + x*x - x_old*x_old
                        else:
                            new_ck = conv[k] + 2*(x - x_old)*f[ki]
                        if new_ck > max_val:
                            max_val = new_ck
                # Also consider unaffected k
                # The max over unaffected k is just the max of conv outside [k_lo, k_hi]
                if k_lo > 0:
                    max_val = max(max_val, conv[:k_lo].max())
                if k_hi < 2*n-2:
                    max_val = max(max_val, conv[k_hi+1:].max())
                return max_val
            
            # Golden section search
            gr = (np.sqrt(5)-1)/2
            lo = 0.0
            hi = x_max
            c = hi - gr*(hi-lo)
            d = lo + gr*(hi-lo)
            fc = eval_x(c)
            fd = eval_x(d)
            best_x = x_old
            best_v = eval_x(x_old)
            
            for _ in range(25):
                if fc < fd:
                    hi = d; d = c; fd = fc
                    c = hi - gr*(hi-lo)
                    fc = eval_x(c)
                else:
                    lo = c; c = d; fc = fd
                    d = lo + gr*(hi-lo)
                    fd = eval_x(d)
                if abs(hi-lo) < 1e-12:
                    break
            
            # Also check endpoints and original
            for xc in [0.0, x_old, (lo+hi)/2]:
                vc = eval_x(xc)
                if vc < best_v:
                    best_v = vc
                    best_x = xc
            
            if best_x != x_old:
                f[i] = best_x
                # Update conv
                for k in range(k_lo, k_hi+1):
                    ki = k - i
                    if 0 <= ki <= n-1:
                        if k == 2*i:
                            conv[k] += best_x*best_x - x_old*x_old
                        else:
                            conv[k] += 2*(best_x - x_old)*f[ki]
                x_old = best_x  # Not needed since we moved on
                
                obj = conv.max() * 2 * n
                # Wait, we need to account for normalization. Since we're changing f[i],
                # the sum changes, so we need to renormalize.
                # This complicates things. Let me handle normalization separately.
            
            # Renormalize
            s = f.sum()
            if s < 1e-15:
                break
            f = f / s
            conv = fast_convolve(f)  # Recompute after normalization
            
            obj = 2*n*conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                improved = True
        
        if not improved:
            no_improve += 1
    
    return best_f, best_obj
```

```python
def obj_of_x_vec(x, conv, f, i, k_lo, k_hi, n):
    scale = 1.0 + (x - f[i])
    if abs(scale) < 1e-15:
        return 1e20
    scale2 = scale * scale
    
    # Affected k values
    k_affected = np.arange(k_lo, k_hi+1)
    ki_vals = k_affected - i  # = k - i
    
    # Linear term: 2*(x - f[i]) * f[ki_vals]
    linear = 2.0 * (x - f[i]) * f[ki_vals]
    
    # Quadratic term at k = 2i
    quad = np.zeros(len(k_affected))
    mask = (k_affected == 2*i)
    quad[mask] = x*x - f[i]*f[i]
    
    # New conv at affected k
    new_conv = (conv[k_affected] + linear + quad) / scale2
    
    # Unaffected k
    max_unaffected = 0.0
    if k_lo > 0:
        max_unaffected = max(max_unaffected, conv[:k_lo].max())
    if k_hi < 2*n-2:
        max_unaffected = max(max_unaffected, conv[k_hi+1:].max())
    max_unaffected /= scale2
    
    return max(new_conv.max(), max_unaffected)
```
