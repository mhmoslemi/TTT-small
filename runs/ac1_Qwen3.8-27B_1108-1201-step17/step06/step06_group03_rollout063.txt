"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using corrected subgradient mirror descent,
LP-Frank-Wolfe, pairwise exchange local search, and multi-length exploration.
Key fix: correct subgradient is 2*a[k-j] (not a[k-j]+a[j] as in previous version).

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_correct_subgradient(a, conv, n_peaks=20, temperature=0.1):
    """
    Compute weighted subgradient of max(conv) using the correct formula:
    d/da[j] (a*a)[k] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    len_c = len(conv)
    k_top = min(n_peaks, len_c)
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[peaks]
    
    # Softmax weights
    scaled = pv / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        # For position j, the gradient contribution is 2*a[k-j] if 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(n-1, k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        # a[k-j] for j in [js, je]
        # k-j ranges from k-je to k-js
        mirror_idx = k - np.arange(js, je + 1)
        sg[js:je+1] += w[i] * 2.0 * a[mirror_idx]
    
    return sg

def smoothed_mirror_descent_v2(a0, time_limit, verbose=False,
                                tau_init=1.0, tau_decay=1.005,
                                lr_init=0.02, lr_decay=0.9999,
                                restart_every=150, n_peaks=20):
    """
    Smoothed mirror descent with CORRECT subgradient.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Compute correct subgradient
        sg = compute_correct_subgradient(a, conv, n_peaks=n_peaks, temperature=1.0/tau)
        
        # Multiplicative update
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        obj_new = fast_convolve(a_new).max()
        if obj_new < M:
            a = a_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
                if verbose and iter_count % 200 == 0:
                    print(f"      SMD2 iter {iter_count}: obj={best_obj:.10f}")
        else:
            # Backtracking
            alpha = lr
            found = False
            for _ in range(5):
                alpha *= 0.5
                a_trial = a * np.exp(-alpha * sg)
                a_trial = np.maximum(a_trial, 1e-15)
                a_trial = a_trial / a_trial.sum()
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < M:
                    a = a_trial
                    if obj_trial < best_obj - 1e-14:
                        best_obj = obj_trial
                        best_a = a.copy()
                        no_improve = 0
                    found = True
                    break
            if not found:
                no_improve += 1
        
        # Adaptive parameters
        tau *= tau_decay
        lr *= lr_decay
        
        # Restart
        if no_improve > restart_every:
            rng = np.random.default_rng(iter_count * 7 + 13)
            noise = rng.standard_normal(n) * 0.003
            a = np.maximum(best_a + noise, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.5
            tau = tau_init * 5
            no_improve = 0
            
        if lr < 1e-8:
            break
    
    return best_a, best_obj

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j] for valid indices
    # For row k: sum over j where 0<=j<n and 0<=k-j<n of f[j] * g[k-j]
    # = sum over j in [max(0,k-n+1), min(n-1,k)] of f[j] * g[k-j]
    # Let i = k-j, so j = k-i, and i ranges over [max(0,k-n+1), min(n-1,k)]
    # So (f*g)[k] = sum_i f[k-i] * g[i] for i in [max(0,k-n+1), min(n-1,k)]
    # This means: A[k, i] = f[k-i] for valid (k,i)
    
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[k, i] = f[k - i]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_fw_optimize_v2(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with correct gradient awareness."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)

    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(a, M, n)
        if g is None:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        g = g / g_sum
        
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 80:
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + 7)

                a = a / a.sum()
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0
            continue
        else:
            no_improve = 0
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = np.linspace(0.05, 1.0, 20)
        for alpha in alphas:
            ft = a + alpha * (g - a)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            a_new = a + best_alpha * (g - a)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            conv = fast_convolve(a_new)
            M = conv.max()
            a = a_new
            
            if M < best_obj - 1e-14:

                best_a = a.copy()
                if verbose and it % 10 == 0:
                    print(f"      LP-FW2 iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 80:
                rng = np.random.default_rng(it * 43 + 11)
                a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-15)
                a = a / a.sum()
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0
    
    return best_a, best_obj

def pairwise_exchange(a0, time_limit, verbose=False, n_trials_per_round=200):

        return a, float('inf')

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(n_trials_per_round):
            if time.monotonic() - t0 > time_limit:

            # Pick two random coordinates
            i, j = rng.integers(0, n, size=2)

                    break

                    continue
                if a[i] < 1e-15 and a[j] < 1e-15:
                    continue
                
                total = a[i] + a[j]
                if total < 1e-15:
                    continue
                best_frac = a[i] / total
                best_local_obj = fast_convolve(a).max()
                
                for frac in [best_frac + d for d in [-0.2, -0.1, -0.05, 0.05, 0.1, 0.2]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using corrected subgradient mirror descent,
LP-Frank-Wolfe, targeted pairwise exchange, and multi-length exploration.
Key fix over previous version: correct subgradient is 2*a[k-j] (not a[k-j]+a[j]).
This yields a natural 'mirror repulsion' update that directly reduces peak overlaps.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_correct_subgradient(a, conv, n_peaks=30, temperature=0.1):
    """
    Compute weighted subgradient of max(conv) using correct formula:
    d/da[j] (a*a)[k] = 2*a[k-j] if 0 <= k-j < n, else 0.
    Uses softmax weighting over top peaks for smoothness.
    """
    n = len(a)
    len_c = len(conv)
    k_top = min(n_peaks, len_c)
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[peaks]

    scaled = pv / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        mirror_idx = k - np.arange(js, je + 1)
        sg[js:je + 1] += w[i] * 2.0 * a[mirror_idx]

    return sg


def smoothed_mirror_descent_v2(a0, time_limit, verbose=False,
                                tau_init=2.0, tau_decay=1.003,
                                lr_init=0.03, lr_min=1e-7,
                                restart_every=200, n_peaks=30):
    """Smoothed mirror descent with CORRECT subgradient and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(a)
        M = conv.max()

        sg = compute_correct_subgradient(a, conv, n_peaks=n_peaks, temperature=1.0 / tau)

        # Momentum
        momentum = mu * momentum + (1 - mu) * sg
        eff_sg = momentum

        # Multiplicative update
        a_new = a * np.exp(-lr * eff_sg)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        obj_new = fast_convolve(a_new).max()
        if obj_new < M:
            a = a_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
                if verbose and iter_count % 500 == 0:
                    print(f"      SMD2 iter {iter_count}: obj={best_obj:.10f}")
        else:
            # Backtracking line search
            found = False
            for _ in range(6):
                lr *= 0.5
                a_trial = a * np.exp(-lr * eff_sg)
                a_trial = np.maximum(a_trial, 1e-15)
                a_trial = a_trial / a_trial.sum()
                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < M:
                    a = a_trial
                    if obj_trial < best_obj - 1e-14:
                        best_obj = obj_trial
                        best_a = a.copy()
                        no_improve = 0
                    found = True
                    break
            if not found:
                no_improve += 1
                lr = max(lr, lr_min)

        tau *= tau_decay

        # Restart with perturbation
        if no_improve > restart_every:
            rng = np.random.default_rng(iter_count * 7 + 13)
            noise_scale = min(0.01, 0.003 + 0.003 * no_improve / restart_every)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 1e-15)
            a = a / a.sum()
            momentum = np.zeros(n)
            lr = lr_init * 0.5
            tau = tau_init * 3
            no_improve = 0

        if lr < lr_min:
            break

    return best_a, best_obj


def build_lp_matrix(n):
    """Pre-build the LP constraint matrix for auto-convolution constraints."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A[k, i] = 1.0  # placeholder, will be scaled by f
    return A


def solve_lp_direction(f, M, n, A_template=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    if A_template is None:
        A_ub = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            A_ub[k, idx] = f[k - idx]
    else:
        # A_template has 1s where valid; scale rows by appropriate f values
        A_ub = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            A_ub[k, idx] = f[k - idx]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def gradient_directed_local_search(a0, time_limit, verbose=False):
    """
    Use the correct gradient to identify which mirror pairs to split.
    The gradient 2*a[k-j] tells us: at position j, the 'partner' is at k-j.
    We specifically try to reduce a[j] * a[k-j] for the hottest pairs.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()

        it += 1

        M = conv.max()
        k_peak = int(np.argmax(conv))

        # The peak is sum_j a[j]*a[k_peak-j] for j in valid range
        # To reduce it, we want to reduce products a[j]*a[k_peak-j]
        # Strategy: for the largest such product, transfer mass from one to the other

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        products = a[idx] * a[k_peak - idx]

        # Try modifying the top few product pairs
        top_pairs = np.argsort(products)[-min(10, len(products)):]
        improved = False

        for p in top_pairs:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(idx[p])
            j_mirror = k_peak - j
            if j == j_mirror:
                continue  # can't split a single coordinate

            total = a[j] + a[j_mirror]
            if total < 1e-15:
                continue

            cur_frac = a[j] / total
            best_frac = cur_frac
            best_local = M

            # Try making more asymmetric (move mass to one side)
            for frac in [0.05, 0.1, 0.2, 0.8, 0.9, 0.95,
                        max(0.01, cur_frac - 0.3), max(0.01, cur_frac - 0.15),
                        min(0.99, cur_frac + 0.15), min(0.99, cur_frac + 0.3)]:
                if abs(frac - cur_frac) < 0.01:
                    continue
                a_copy = a.copy()
                a_copy[j] = frac * total
                a_copy[j_mirror] = (1.0 - frac) * total
                val = fast_convolve(a_copy).max()
                if val < best_local - 1e-14:
                    best_local = val
                    best_frac = frac

            if abs(best_frac - cur_frac) > 0.005:
                a[j] = best_frac * total
                a[j_mirror] = (1.0 - best_frac) * total
                if best_local < best_obj - 1e-14:

                    best_a = a.copy()
                    improved = True

        # Occasionally do random perturbations

            a = np.maximum(a + noise, 1e-15)

        if not improved and it > 100:

            stuck_count = getattr(gradient_directed_local_search, '_stuck', 0) + 1
            gradient_directed_local_search._stuck = stuck_count
            if stuck_count > 50:
                break

    return best_a, best_obj

        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp /= ss
    return a_interp


def coordinate_descent_refine(a0, time_limit, verbose=False):
    """Fine coordinate descent for final polish."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    round_num = 0

    while time.monotonic() - t0 < time_limit:
        round_num += 1
        improved = False
        rng = np.random.default_rng(1000 + round_num)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj

            old_val = a[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_j_val = old_val
            best_j_obj = fast_convolve(a).max()

            # Try multiplicative changes
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.3]:
                new_val = old_val * factor
                if new_val < 0 or new_val > 0.95:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if abs(best_j_val - old_val) > 1e-14:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break
        if verbose and round_num % 3 == 0:
            print(f"      CD round {round_num}: obj={best_obj:.10f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Diverse asymmetric initializations + quick SMD screening
    2. Correct-gradient SMD on best candidates
    3. LP-Frank-Wolfe refinement
    4. Gradient-directed local search (mirror pair splitting)
    5. Targeted pairwise exchange
    6. Multi-length exploration
    7. Coordinate descent final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Generate and screen initializations ===
    print("=== Phase 1: Initialization & Screening ===")
    test_lengths = [512, 640, 768, 896, 1024]
    init_candidates = []

    for n_try in test_lengths:
        for v in range(12):
            a_init = make_asymmetric_init(n_try, variant=v, seed=seed)
            init_candidates.append((a_init, f"init_v{v}_n{n_try}"))

    if parent is not None:
        init_candidates.append((parent, "parent"))

    # Quick screen using fast_convolve
    scored = []
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        scored.append((obj, a_norm, label))

    scored.sort(key=lambda x: x[0])
    top_n = 5
    top_inits = scored[:top_n]
    print(f"  Top {top_n} inits: " + ", ".join(f"{l}:{o:.4f}" for o, _, l in top_inits))

    # Evaluate top inits with actual score
    for obj, a_init, label in top_inits:
        try_seq(a_init, label)

    # === Phase 2: SMD with correct gradient on top candidates ===
    print("\n=== Phase 2: SMD (correct gradient) ===")
    for idx in range(min(3, len(top_inits))):
        rem = deadline - time.monotonic() - 600
        if rem < 20:
            break
        obj_init, a_init, label = top_inits[idx]
        t_alloc = min(rem / max(1, 3 - idx), 250)
        if t_alloc < 10:
            t_alloc = 10
        print(f"  SMD on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
        a_opt, obj = smoothed_mirror_descent_v2(a_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"smd_{label}")

    # Also run SMD on parent if we have one
    if parent is not None and best_seq is not None and len(parent) == len(best_seq):
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  SMD on parent (n={len(parent)}): t={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent_v2(parent, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "smd_parent")

    # === Phase 3: LP-Frank-Wolfe ===
    print("\n=== Phase 3: LP-Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            t_alloc = min(rem, 250)
            print(f"  LP-FW (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 4: Gradient-directed local search ===
    print("\n=== Phase 4: Gradient-Directed Local Search ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  GDLS (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = gradient_directed_local_search(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "gdls")

    # === Phase 5: Targeted pairwise exchange ===
    print("\n=== Phase 5: Targeted Pairwise Exchange ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 180)
            print(f"  PWE (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = targeted_pairwise_exchange(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "pwe")

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [base_n + 128, base_n + 256, base_n // 2 + 128]:
            if target_n <= 0 or target_n > 2048:
                continue
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.05:
                t_alloc = min(rem * 0.5, 80)
                if t_alloc < 10:
                    t_alloc = 10
                print(f"  SMD n={target_n}: t={t_alloc:.0f}s")
                a_opt, obj = smoothed_mirror_descent_v2(f_interp, time_limit=t_alloc)
                try_seq(a_opt, f"smd_n{target_n}")

    # === Phase 7: Another round of SMD + LP-FW on current best ===
    print("\n=== Phase 7: Second Optimization Round ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.5, 100)
            if t_alloc > 10:
                print(f"  SMD round 2: t={t_alloc:.0f}s")
                a_opt, obj = smoothed_mirror_descent_v2(
                    best_seq, time_limit=t_alloc,
                    tau_init=5.0, lr_init=0.01, restart_every=100)
                try_seq(a_opt, "smd_r2")

        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem * 0.6, 80)
            if t_alloc > 10:
                print(f"  LP-FW round 2: t={t_alloc:.0f}s")
                a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "lpfw_r2")

    # === Phase 8: Coordinate descent final polish ===
    print("\n=== Phase 8: Coordinate Descent Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 60)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_refine(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd_final")

    # === Phase 9: One more PWE squeeze ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_alloc = min(rem, 25)
            print(f"  Final PWE: t={t_alloc:.0f}s")
            a_opt, obj = targeted_pairwise_exchange(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "pwe_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
