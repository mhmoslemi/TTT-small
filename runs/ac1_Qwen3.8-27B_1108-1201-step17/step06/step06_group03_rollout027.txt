def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, f[0])
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_active_lp(f, M, active_indices, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0."""
    K = len(active_indices)
    if K == 0:
        return None

    c_obj = -np.ones(n)

    # Build constraint matrix: each row is the shifted f for the corresponding lag
    A_ub = np.zeros((K, n))
    for i, k in enumerate(active_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        A_ub[i, js:je + 1] = f[k - js:je + 1][::-1]

    b_ub = np.full(K, M)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None
