"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain flattening (Gerchberg-Saxton variant)
2. LP-Frank-Wolfe with adaptive M relaxation
3. Swap-based coordinate descent
4. Multi-length exploration
5. Simulated annealing for global search
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective_normalized(a):
    """For sum(a)=1, returns 2*n*max(conv)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def gs_flatten(a, n_iter=50, alpha=0.5):
    """Mild Gerchberg-Saxton flattening of the power spectrum."""
    a = a.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    
    for _ in range(n_iter):
        # Pad and compute DFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        
        # Compute target magnitude (average power)
        # Keep DC fixed
        mag2 = np.abs(A) ** 2
        target = mag2[1:].mean()  # average power excluding DC
        
        # Apply soft flattening
        new_mag2 = mag2.copy()
        new_mag2[0] = mag2[0]  # keep DC
        new_mag2[1:] = mag2[1:] + alpha * (target - mag2[1:])
        new_mag2 = np.maximum(new_mag2, 0)
        
        # Reconstruct with new magnitudes, keeping phases
        phase = np.angle(A)
        new_A = np.sqrt(new_mag2) * np.exp(1j * phase)
        new_A[0] = A[0]  # keep DC exactly
        
        a_new = np.fft.irfft(new_A, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def swap_descent(f0, time_limit, verbose=False):

    # conv[k] involves a[i]*a[k-i], so swapping a[i] and a[j] affects
    # conv[k] for k where i or j is involved

        if no_improve > 100:
            break

            break

        max_prod_pos = np.argmax(products)

                if obj < best_swap_obj - 1e-15:
                    best_swap_obj = obj
                    best_swap = trial_dst
            if best_swap is not None:
                f[src], f[best_swap] = f[best_swap], f[src]
                best_obj = best_swap_obj
                best_f = f.copy()

            else:
                no_improve += 1
            continue

        best_swap = None
        best_swap_obj = best_obj

        candidates = set()
        candidates.add(dst)
        # Add some random positions
        rng = np.random.default_rng(hash((src, dst, no_improve)) % (2**32))
        candidates.update(rng.choice(n, size=min(20, n), replace=False))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain flattening (Gerchberg-Saxton variant) for initialization
2. LP-Frank-Wolfe with adaptive M relaxation and dual-guided line search
3. Swap-based coordinate descent for fine-tuning
4. Multi-peak subgradient descent
5. Multi-length exploration and simulated annealing for global search
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """For sum(a)=1, returns 2*n*max(conv)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def gs_flatten(a, n_iter=30, alpha=0.3):
    """Mild Gerchberg-Saxton flattening of the power spectrum."""
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    for _ in range(n_iter):
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, fft_size)

        mag2 = np.abs(A) ** 2
        if len(mag2) > 1:
            target = mag2[1:].mean()
        else:
            target = mag2[0]

        new_mag2 = mag2.copy()
        new_mag2[0] = mag2[0]
        if len(new_mag2) > 1:
            new_mag2[1:] = mag2[1:] + alpha * (target - mag2[1:])
        new_mag2 = np.maximum(new_mag2, 1e-20)

        phase = np.angle(A)
        new_A = np.sqrt(new_mag2) * (np.cos(phase) + 1j * np.sin(phase))
        new_A[0] = A[0]

        a_new = np.fft.irfft(new_A, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

    return a


def solve_lp_direction(f, M, n):

    c = -np.ones(n)

    for j in range(n):

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

            elif noise_type == 1:
                shift = rng.integers(1, max(2, n // 10))
                f = project_simplex(np.roll(best_f, shift))
            elif noise_type == 2:
                f = project_simplex(0.85 * best_f + 0.15 * best_f[::-1])
            elif noise_type == 3:
                f = gs_flatten(best_f, n_iter=10, alpha=0.2)
            else:
                # Blend with a shifted version
                shift = rng.integers(1, max(2, n // 20))
                f = project_simplex(0.9 * best_f + 0.1 * np.roll(best_f, shift))

            m_idx = 0
            no_improve += 1
            if no_improve > 60:
                break
            continue

        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            if no_improve > 50:
                break
            continue
        no_improve = 0

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M_cur
        alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]

        for alpha in alphas:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0 and best_alpha < 1.0:
            idx = alphas.index(best_alpha)
            lo = alphas[max(0, idx - 1)]
            hi = alphas[min(len(alphas) - 1, idx + 1)]
            if lo == hi:
                lo = max(0.0, best_alpha - 0.05)
                hi = min(1.0, best_alpha + 0.05)

            for _ in range(8):
                a1 = hi - 0.618 * (hi - lo)
                a2 = lo + 0.618 * (hi - lo)
                ft1 = np.maximum(f + a1 * (g - f), 0.0)
                ft2 = np.maximum(f + a2 * (g - f), 0.0)
                ft1 /= ft1.sum()
                ft2 /= ft2.sum()
                v1 = fast_convolve(ft1).max()
                v2 = fast_convolve(ft2).max()
                if v1 < v2:
                    hi = a2
                else:
                    lo = a1
                mid = (lo + hi) / 2.0
                ftm = np.maximum(f + mid * (g - f), 0.0)
                ftm /= ftm.sum()
                vm = fast_convolve(ftm).max()
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid

        if best_alpha < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            continue

        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        f = f_new
        conv = fast_convolve(f)
        M = conv.max()
        m_idx = 0

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def swap_descent(f0, time_limit, verbose=False):
    """Coordinate descent using swap operations targeting peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 200:
            break

        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        if len(contrib_idx) == 0:
            break

        products = f[contrib_idx] * f[peak_k - contrib_idx]
        max_prod_pos = np.argmax(products)
        src = int(contrib_idx[max_prod_pos])
        partner = int(peak_k - src)

        # Generate candidate swap positions
        rng = np.random.default_rng(it * 31 + 17)
        candidates = set()
        candidates.add(partner)
        candidates.update(int(x) for x in rng.choice(n, size=min(30, n), replace=False))
        # Add evenly spaced positions
        candidates.update(int(x) for x in range(0, n, max(1, n // 30)))

        best_swap = None
        best_swap_obj = best_obj

        for trial_dst in candidates:
            if trial_dst == src:
                continue
            f[src], f[trial_dst] = f[trial_dst], f[src]
            obj = fast_convolve(f).max()
            f[src], f[trial_dst] = f[trial_dst], f[src]
            if obj < best_swap_obj - 1e-15:
                best_swap_obj = obj
                best_swap = trial_dst

        if best_swap is not None:
            f[src], f[best_swap] = f[best_swap], f[src]
            if best_swap_obj < best_obj - 1e-14:
                best_obj = best_swap_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    Swap iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

    return best_f, best_obj


def multi_peak_subgrad(f0, time_limit, verbose=False, k_peaks=10):
    """Subgradient descent using top-k peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 300:
            break

        conv = fast_convolve(f)
        top_k = min(k_peaks, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_values = conv[peak_indices]

        # Weight peaks by their value (emphasize higher peaks)
        w = peak_values ** 2
        w /= w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peak_indices)):
            if w[i] < 1e-12:
                continue
            k = int(peak_indices[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Projected gradient step
        f_new = np.maximum(f - lr * sg, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 50 == 0:
                print(f"    MSPG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve % 50 == 0:
            lr *= 0.8
        if lr < 1e-8:
            break

    return best_f, best_obj


def simulated_annealing(f0, time_limit, verbose=False):
    """Simulated annealing with diverse moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_obj = best_obj

    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    T_init = 0.01
    T_final = 0.00005
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_final / T_init) ** progress

        f_save = f.copy()
        move_type = rng.random()

        if move_type < 0.35:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                f[i], f[j] = f[j], f[i]
        elif move_type < 0.6:
            i = int(rng.integers(n))
            delta = rng.standard_normal() * max(T, 0.001)
            f[i] = max(0.0, f[i] + delta)
        elif move_type < 0.8:
            i = int(rng.integers(0, max(1, n - 10)))
            length = int(rng.integers(2, min(15, n - i)))
            delta = rng.standard_normal(length) * max(T, 0.001)
            f[i:i + length] = np.maximum(f[i:i + length] + delta, 0.0)
        else:
            # Block swap
            i = int(rng.integers(0, n - 10))
            j = int(rng.integers(0, n - 10))
            if abs(i - j) > 10:
                length = int(min(5, n - i, n - j))
                f[i:i + length], f[j:j + length] = f[j:j + length].copy(), f[i:i + length].copy()

        s = f.sum()
        if s < 1e-15:
            f = f_save
            continue
        f = f / s
        obj = fast_convolve(f).max()

        if obj < cur_obj or rng.random() < np.exp(-min(50.0, (obj - cur_obj) / max(T, 1e-10))):
            cur_obj = obj
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
                if verbose and it % 200 == 0:
                    print(f"    SA iter {it}: best={best_obj:.10f}, T={T:.6f}")
        else:
            f = f_save

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_cosine_mod(n, freq, amp, phase=0.0):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.25, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining LP-FW, swap descent,
    multi-peak subgradient, frequency-domain methods, and simulated annealing.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    total_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_multi_cosine(768, 3, 0.2, seed=42)
        try_seq(parent, "default_parent")
        print(f"  No parent available, using default. Score: {best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Generate diverse candidates ===
    print(f"\n=== Phase 1: Diverse Candidates ===")
    t0 = time.monotonic()

    candidates = []
    candidates.append(("parent", parent))

    # GS-flattened versions
    for alpha_gs in [0.2, 0.4, 0.6]:
        candidates.append((f"gs_a{alpha_gs}", gs_flatten(parent, n_iter=20, alpha=alpha_gs)))

    # Noise perturbations
    for amp in [0.002, 0.005, 0.01]:
        rng = np.random.default_rng(seed + int(amp * 10000))
        candidates.append((f"noise_{amp}", project_simplex(parent + rng.standard_normal(n_parent) * amp)))

    # Shifts
    for shift in [1, 2, 3, 5, 8]:
        candidates.append((f"shift_{shift}", project_simplex(np.roll(parent, shift))))

    # Reversal blends
    for mix in [0.1, 0.2, 0.3]:
        candidates.append((f"rev_{mix}", project_simplex((1 - mix) * parent + mix * parent[::-1])))

    # Multi-cosine at parent length
    for nc in [2, 3, 5]:
        candidates.append((f"mc_{nc}", make_multi_cosine(n_parent, nc, 0.2, seed=n_parent + nc)))

    # Different lengths
    for target_n in [512, 640, 896, 1024]:
        if target_n != n_parent:
            interp = interpolate_sequence(parent, target_n)
            candidates.append((f"interp_{target_n}", interp))
            # Also GS on interpolated
            candidates.append((f"gs_interp_{target_n}", gs_flatten(interp, n_iter=15, alpha=0.3)))

    # Evaluate all candidates
    scored_candidates = []
    for label, cand in candidates:
        cand = np.maximum(cand, 0.0)
        sc = cand.sum()
        if sc < 1e-10:
            continue
        cand_norm = cand / cand.sum()
        obj = objective_normalized(cand_norm)
        scored_candidates.append((obj, label, cand_norm))

    scored_candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(scored_candidates)} candidates")
    print(f"  Top 5: ", end="")
    for obj, label, _ in scored_candidates[:5]:
        print(f"{label}({obj:.6f}) ", end="")
    print()
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # Evaluate top candidates with evaluate_sequence
    for obj, label, cand in scored_candidates[:10]:
        try_seq(cand, f"init_{label}")

    # === Phase 2: LP-FW on top candidates ===
    print(f"\n=== Phase 2: LP-FW Optimization ===")

    # Try top 4 candidates with LP-FW
    for trial_idx in range(min(4, len(scored_candidates))):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        obj_init, label, cand = scored_candidates[trial_idx]
        t_alloc = min(rem * 0.35, 180)
        if t_alloc < 20:
            t_alloc = 20
        print(f"  LP-FW [{label}] n={len(cand)}: alloc={t_alloc:.0f}s (init={obj_init:.6f})")
        a_opt, obj_opt = lp_fw_optimize(cand, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"lpfw_{label}")

    # === Phase 3: Swap descent on best ===
    print(f"\n=== Phase 3: Swap Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.4, 150)
            print(f"  Swap descent: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = swap_descent(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "swap1")

    # === Phase 4: Multi-peak subgradient ===
    print(f"\n=== Phase 4: Multi-Peak Subgradient ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.4, 100)
            print(f"  MSPG: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = multi_peak_subgrad(best_seq, time_limit=t_alloc, k_peaks=15, verbose=False)
            try_seq(a_opt, "mspg")

    # === Phase 5: GS + LP-FW refinement ===
    print(f"\n=== Phase 5: GS + LP-FW Refinement ===")
    if best_seq is not None:
        for alpha_gs in [0.15, 0.3, 0.5]:
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            f_gs = gs_flatten(best_seq, n_iter=15, alpha=alpha_gs)
            t_alloc = min(rem * 0.3, 80)
            if t_alloc < 15:
                t_alloc = 15
            print(f"  GS(a={alpha_gs}) + LP-FW: alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_gs, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"gs_lp_{alpha_gs}")

    # === Phase 6: Simulated annealing ===
    print(f"\n=== Phase 6: Simulated Annealing ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_alloc = min(rem * 0.4, 100)
            print(f"  SA: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = simulated_annealing(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "sa")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [n_parent, 512, 640, 896, 1024, 640]:
            if target_n == base_n or target_n in [512, 640, 896, 1024][:0]:
                pass
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.25, 50)
                if t_alloc > 10:
                    a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
                    try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 8: Final LP-FW polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem * 0.6, 60)
            print(f"  Final LP-FW: alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final_lp")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem * 0.5, 30)
            print(f"  Final swap: alloc={t_alloc:.0f}s")
            a_opt, obj = swap_descent(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "final_swap")

        # Very fine MSPG
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            t_alloc = min(rem, 15)
            print(f"  Final MSPG: alloc={t_alloc:.0f}s")
            a_opt, obj = multi_peak_subgrad(best_seq, time_limit=t_alloc, k_peaks=5, verbose=False)
            try_seq(a_opt, "final_mspg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  FALLBACK used: score={best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - total_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Total time: {total_time:.1f}s / {budget_s}s")

    return [float(x) for x in best_seq]
