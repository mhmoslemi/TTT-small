from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(f):
    """Build the matrix A where (A @ g)[k] = (f*g)[k]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, j] = f[k - j]
    return A

def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def lp_optimize(f_init, time_limit, max_iter=1000, t_init=0.5):
    """LP-based iterative optimization."""
    t0 = time.monotonic()
    f = f_init.copy().astype(np.float64)
    s = f.sum()
    f = f / s
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        g = lp_direction(f, M)
        if g is None:
            break

        if g_sum < 1e-10:

        g = g / g_sum  # normalize to sum 1

        t = t_init
        for _ in range(20):
            f_new = (1 - t) * f + t * g
            peak_new = fast_convolve(f_new).max()

LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key insight: At each iteration, solve an LP to find the best direction g
that, when convolved with the current f, doesn't exceed the current peak M.
This gives a principled descent direction that is better than simple subgradients.

The LP is: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_rows(f):
    """Build the rows of the convolution matrix efficiently.
    Returns (row_indices, col_indices, values) for sparse matrix construction.
    A[k, m] = f[k-m] if 0 <= k-m < n, else 0.
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for m in range(n):
        # For column m, non-zeros are at rows k = m, m+1, ..., m+n-1
        # A[k, m] = f[k-m]
        for j in range(n):
            rows.append(m + j)
            cols.append(m)
            vals.append(f[j])
    return np.array(rows), np.array(cols), np.array(vals)


def lp_solve(f, M):
    """Solve the LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns g or None if LP fails."""
    n = len(f)
    rows, cols, vals = build_conv_rows(f)
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                  bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9, 
                         temp_init=0.001, kick_interval=1000, kick_size=0.003):
    """Subgradient method with momentum on the simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        b_max = b.max()
        
        # Subgradient at the max positions
        max_idx = np.where(b >= b_max - 1e-10)[0]
        if len(max_idx) == 0:
            max_idx = np.array([b.argmax()])
        
        # Average subgradient over near-max positions
        g = np.zeros(n)
        for k in max_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += a[k - idx] / len(max_idx)
        
        # Nesterov momentum
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive step size
        if it % kick_interval == 0:
            a = project_simplex(best_a + np.random.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.8, lr_init * 0.1)
    
    return best_a, best_obj

        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:

                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)

                val = fast_convolve(a_trial).max()

                    best_obj = val
                    best_a = a_trial.copy()
                    improved = True
                    # Update a to the new best
                    a = a_trial.copy()
                    break

            break

def interpolate_sequence(f, new_n):
    """Interpolate a probability sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Core idea: At each iteration, solve a linear program to find the best non-negative
direction g such that conv(f,g) doesn't exceed the current peak M. This provides
a principled descent direction superior to plain subgradients. The LP is:
    max sum(g)  s.t.  (f*g)[k] <= M for all k,  g >= 0
Combined with subgradient polishing, coordinate descent, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_rc(n):
    """Precompute row/col index arrays for the (2n-1) x n convolution matrix."""
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    for m in range(n):
        sl = slice(m * n, (m + 1) * n)
        rows[sl] = m + np.arange(n, dtype=np.int32)
        cols[sl] = m
    return rows, cols


def make_conv_matrix(f, rc):
    """Build the sparse convolution matrix A where (A@g)[k] = (f*g)[k]."""
    rows, cols = rc
    n = len(f)
    vals = np.tile(f, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve(f, rc, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = make_conv_matrix(f, rc)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

        f = f / ss
    n = len(f)
    rc = build_conv_rc(n)
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    stagnant = 0

    if verbose:

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = fast_convolve(f).max()

def subgradient_optimize(a0, time_limit, lr_init=0.004, momentum=0.9,
                         temp_init=0.001, temp_growth=1.015,
                         kick_interval=800, kick_size=0.003, n_top=None, seed=42):
    """Subgradient method with Nesterov momentum and temperature-weighted peaks."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        b_max = b.max()

        # Temperature-weighted weights over peaks
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= ws

        if n_top is not None and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx]
            wsel_sum = w_sel.sum()
            if wsel_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w_full = np.zeros(len(b))
                w_full[top_idx] = w_sel / wsel_sum
                w = w_full

        # Compute weighted subgradient
        g = np.zeros(n)
        active = np.where(w > 1e-12)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * a[k - idx]

        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Periodic adjustments
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.96

        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate descent: try scaling individual elements."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    multipliers = np.array([0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0])
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
        if not improved:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a probability sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
