import time
from scipy.optimize import linprog
from scipy import sparse

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(a):
    """Build sparse A_ub matrix for the LP."""
    n = len(a)
    n_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_fallback(a, M, n, top_k=20):
    """Subgradient-based direction as LP fallback."""
    conv = fast_convolve(a)
    K = min(top_k, len(conv))
    if K < len(conv):
        peak_indices = np.argpartition(conv, -K)[-K:]
    else:
        peak_indices = np.arange(len(conv))

    pv = conv[peak_indices]
    pmax = pv.max()
    pmin = pv.min()
    if pmax - pmin > 1e-15:
        w = (pv - pmin) / (pmax - pmin)
    else:
        w = np.ones(len(peak_indices)) / len(peak_indices)
    w = w / w.sum()

    # Compute subgradient
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = w[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]

    # Project to get a direction on the simplex
    direction = a - 0.01 * (g - g.mean())
    direction = project_simplex(direction)
    return direction

def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with binary search on each coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, fast_convolve(a).max()
    a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.6, 0.8, 0.95, 1.05, 1.2, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):

    a = np.maximum(a, 0)

    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""

    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a

def make_cosine(n):

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    n_base = 768
    if parent is not None:
        n_base = len(parent)

    # === Phase 1: Diverse Initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    init_candidates = []
    for n_try in [n_base, 512, 1024]:
        init_candidates.append((make_gaussian(n_try, 0.25), f"gauss25_n{n_try}"))
        init_candidates.append((make_gaussian(n_try, 0.35), f"gauss35_n{n_try}"))
        init_candidates.append((make_bell(n_try, 0.5), f"bell_n{n_try}"))
        init_candidates.append((make_cosine(n_try), f"cos_n{n_try}"))
        init_candidates.append((make_uniform(n_try), f"uni_n{n_try}"))

    if parent is not None:
        init_candidates.append((parent, "parent"))
        init_candidates.append((symmetrize(parent), "parent_sym"))

    best_init = None
    best_init_obj = float('inf')
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        try_seq(a_norm, f"init_{label}")
        obj = 2 * len(a_norm) * fast_convolve(a_norm).max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm

    if best_init is None:
        best_init = make_gaussian(n_base, 0.3)

    print(f"  Best init: obj={best_init_obj:.6f}")

    # === Phase 2: LP-based optimization (main) ===
    print(f"\n=== Phase 2: LP Optimization (main) ===")
    remaining = deadline - time.monotonic()
    t_lp = remaining * 0.45
    print(f"  Time: {t_lp:.0f}s")
    a_opt, obj = lp_optimize(best_init, time_limit=t_lp, verbose=True, label="LP_main")
    try_seq(a_opt, "LP_main")

    # === Phase 3: Continue LP from symmetrized best ===
    print(f"\n=== Phase 3: LP on symmetrized ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_lp2 = remaining * 0.20
        a_sym = symmetrize(best_seq)
        a_opt, obj = lp_optimize(a_sym, time_limit=t_lp2, verbose=True, label="LP_sym")
        try_seq(a_opt, "LP_sym")

        # Also try LP on parent with different params if available
        if parent is not None and time.monotonic() < deadline - 60:
            t_lp3 = min(60, (deadline - time.monotonic()) - 60)
            if t_lp3 > 30:
                a_opt, obj = lp_optimize(parent, time_limit=t_lp3, verbose=False, label="LP_parent")
                try_seq(a_opt, "LP_parent")

    # === Phase 4: Subgradient optimization ===
    print(f"\n=== Phase 4: Subgradient ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_sg = remaining * 0.15
        a_opt, obj = subgradient_optimize(best_seq, time_limit=t_sg, top_k=40,
                                          temp_init=0.0003, lr_init=0.008,
                                          verbose=True, label="SG")
        try_seq(a_opt, "SG")

    # === Phase 5: Multi-restart with LP ===
    print(f"\n=== Phase 5: Multi-restart LP ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_restarts = 3
        t_per = min(remaining * 0.15 / n_restarts, 80)
        if t_per > 20:
            for r in range(n_restarts):
                remaining = deadline - time.monotonic()
                if remaining < 40:
                    break
                if r == 0:
                    f_init = best_seq.copy()
                elif r == 1:
                    noise = rng.standard_normal(len(best_seq)) * 0.003
                    f_init = np.maximum(best_seq + noise, 1e-15)
                    f_init = f_init / f_init.sum()
                else:
                    # Blend with cosine
                    f_init = 0.85 * best_seq + 0.15 * make_cosine(len(best_seq))
                    f_init = f_init / f_init.sum()

                t_alloc = min(t_per, remaining - 30)
                if t_alloc < 15:
                    break
                a_opt, obj = lp_optimize(f_init, time_limit=t_alloc, verbose=False,
                                         label=f"restart_{r}")
                try_seq(a_opt, f"restart_{r}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coord Descent Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = min(remaining - 20, 120)
        if t_cd > 10:
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd,
                                            verbose=True, label="CD")
            try_seq(a_opt, "CD")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_polish = min(remaining - 10, 100)
            a_opt, obj = lp_optimize(best_seq, time_limit=t_polish, verbose=False,
                                     label="polish")
            try_seq(a_opt, "polish")
            # Try symmetrized final
            a_s = symmetrize(best_seq)
            try_seq(a_s, "final_sym")

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(768, 0.3)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
