"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimization with log-sum-exp
smoothing, adaptive temperature annealing, symmetrization, and multi-length search.
Key improvements: per-coordinate adaptive learning rates (Adam), exact gradient
computation via FFT cross-correlation, and polynomial bump initializations.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, w, fft_size):
    """Compute gradient of sum_k w_k*(a*a)_k with respect to a.
    Returns grad[j] = 2 * sum_k w[k] * a[k-j]"""
    n = len(a)
    N = fft_size
    W = np.fft.rfft(w, N)
    A_rev = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(W * A_rev, N)
    # grad[j] = 2 * conv[j + n - 1]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def adam_optimize(a0, time_limit, T0=0.05, T_end=0.0005, lr=0.002,
                  top_k_clip=200, verbose=False, label=""):
    """Adam optimization on the smoothed max-convolution objective."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    conv_size = 2 * n - 1
    fft_size = 1 << (3 * n).bit_length()

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    temp = T0
    t0 = time.monotonic()
    no_improve = 0
    lr_current = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(a)
        M = conv.max()

        if M < best_obj - 1e-12:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
            if verbose and t % 200 == 0:
                print(f"  {label} Adam iter {t}: obj={best_obj:.10f}, T={temp:.6f}, lr={lr_current:.6f}")
        else:
            no_improve += 1

        # Compute softmax weights - optionally restrict to top peaks for efficiency
        K = min(top_k_clip, conv_size)
        if K < conv_size:
            peak_idx = np.argpartition(conv, -K)[-K:]
            w = np.zeros(conv_size)
            pv = conv[peak_idx]
            scaled = pv / temp
            scaled -= scaled.max()
            ew = np.exp(scaled)
            ws = ew.sum()
            if ws < 1e-15:
                ew[np.argmax(pv)] = 1.0
                ws = 1.0
            w[peak_idx] = ew / ws
        else:
            scaled = conv / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()

        # Compute gradient
        grad = compute_gradient(a, w, fft_size)

        # Clip gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 10.0:
            grad = grad / gnorm * 10.0

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        a_new = a - lr_current * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if np.isfinite(obj_new) and obj_new < M * 5:
            a = a_new.copy()
        else:
            lr_current *= 0.5

        # Adaptive learning rate
        if no_improve > 100:
            lr_current *= 0.7
            no_improve = 0

        # Cool temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        temp = T0 * (T_end / T0) ** (progress ** 0.7)

        # Periodic symmetrization
        if t % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = project_simplex(a_sym)
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()
                m = np.zeros(n)
                v = np.zeros(n)

        # Restart if stuck
        if no_improve > 800:
            rng = np.random.default_rng(t * 6151 + 7)
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise)
            m = np.zeros(n)
            v = np.zeros(n)
            lr_current = lr
            no_improve = 0

        if lr_current < 1e-10:
            break

    return best_a, best_obj


def mirror_descent(a0, time_limit, top_k=30, temp_init=0.0005, lr_init=0.01,
                   verbose=False, label=""):
    """Multiplicative weight mirror descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        g = np.zeros(n)
        for idx in range(len(peak_indices)):
            k = int(peak_indices[idx])
            w_k = w[idx]
            if w_k < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]

        # Backtracking
        best_lr = lr * 0.1
        best_trial_obj = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-12:
                continue
            log_update = -trial_lr * g
            log_update = np.clip(log_update, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        log_update = -best_lr * g
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        conv = fast_convolve(a)
        new_obj = conv.max()

        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        temp = min(temp * 1.002, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * 0.9999, 1e-8)

        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            noise = rng.standard_normal(n) * 0.002
            a = np.maximum(best_a + noise, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0

        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        if lr < 1e-9:
            break

    return best_a, best_obj


def peak_targeting(a0, time_limit, verbose=False, label=""):
    """Targeted peak reduction by redistributing mass from peak contributors."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        improved = False

        K = min(5, len(conv))
        peak_indices = np.argsort(conv)[-K:]

        for kp in peak_indices:
            k = int(kp)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            products = a[idx] * a[k - idx]
            j_max = idx[np.argmax(products)]

            old_val = a[j_max]
            if old_val < 1e-10:
                continue

            best_trial_obj = M
            best_trial = None

            for frac in [0.5, 0.7, 0.85, 0.95]:
                new_val = old_val * frac
                diff = old_val - new_val
                if diff < 1e-14:
                    continue
                trial = a.copy()
                trial[j_max] = new_val
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc = a[non_contrib]
                    nc_sum = nc.sum()
                    if nc_sum > 1e-15:
                        trial[non_contrib] += diff * nc / nc_sum
                        trial = np.maximum(trial, 0.0)
                        ss = trial.sum()
                        if ss > 1e-10:
                            trial = trial / ss
                            obj = fast_convolve(trial).max()
                            if obj < best_trial_obj - 1e-14:
                                best_trial_obj = obj
                                best_trial = trial.copy()

            if best_trial is not None:
                a = best_trial
                improved = True
                if best_trial_obj < best_obj - 1e-14:
                    best_obj = best_trial_obj
                    best_a = a.copy()
                    if verbose and it % 20 == 0:
                        print(f"  {label} PT iter {it}: obj={best_obj:.10f}")

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    return best_a, best_obj


def fine_cd(a0, time_limit, verbose=False, label=""):

    a = np.maximum(a0.astype(np.float64), 0.0)

                new_val = old_val * mult
                if new_val < 0:

                denom = 1.0 - old_val

                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val

                trial = trial / ss
                val = fast_convolve(trial).max()

                if best_j_obj < best_obj - 1e-14:

                    improved = True

def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_poly_bump(n, p=2):
    """Polynomial bump: f(x) = (1-x^2)^p on [-1,1] normalized."""
    x = np.linspace(-1, 1, n)
    a = (1 - x ** 2) ** p
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n, width_frac=0.8):
    """Raised cosine window."""
    x = np.arange(n) / n
    margin = (1 - width_frac) / 2
    t = (x - margin) / width_frac
    t = np.clip(t, 0, 1)
    a = 0.5 * (1 + np.cos(np.pi * (2 * t - 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym

    for i in range(n_adam):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        _, a_init, label = best_inits[i]
        t_alloc = min(t_per_adam, remaining - 30)
        if t_alloc < 30:
            break
        print(f"  Adam on {label} ({len(a_init)} elements), time={t_alloc:.0f}s")
        a_opt, obj = adam_optimize(a_init, time_limit=t_alloc, T0=0.05, T_end=0.0003,
                                   lr=0.002, verbose=True, label=f"Adam_{label}")
        try_seq(a_opt, f"Adam_{label}")

    # Phase 2: Mirror descent from best
    print("\n=== Phase 2: Mirror Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_md = remaining * 0.20
        if t_md > 60:
            a_opt, obj = mirror_descent(best_seq, time_limit=t_md, top_k=40,
                                        temp_init=0.0003, lr_init=0.008,
                                        verbose=True, label="MD_main")
            try_seq(a_opt, "MD_main")

    # Phase 3: Adam with different params from symmetrized best
    print("\n=== Phase 3: Adam (symmetric, different params) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_a2 = remaining * 0.15
        if t_a2 > 60:
            a_sym = symmetrize(best_seq)
            a_opt, obj = adam_optimize(a_sym, time_limit=t_a2, T0=0.02, T_end=0.0001,
                                       lr=0.001, top_k_clip=50,
                                       verbose=True, label="Adam_sym")
            try_seq(a_opt, "Adam_sym")

    # Phase 4: Peak targeting
    print("\n=== Phase 4: Peak Targeting ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_pt = remaining * 0.10
        if t_pt > 30:
            a_opt, obj = peak_targeting(best_seq, time_limit=t_pt, verbose=True, label="PT")
            try_seq(a_opt, "PT")

    # Phase 5: Fine coordinate descent
    print("\n=== Phase 5: Fine CD ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = remaining * 0.15
        if t_cd > 30:
            a_opt, obj = fine_cd(best_seq, time_limit=t_cd, verbose=True, label="CD")
            try_seq(a_opt, "CD")

    # Phase 6: Multi-restart Adam from best
    print("\n=== Phase 6: Multi-restart ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_restarts = 4
        t_per_r = min(remaining * 0.15 / n_restarts, 60)
        if t_per_r > 20:
            for r in range(n_restarts):
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                if r == 0:
                    f_init = symmetrize(best_seq)
                elif r == 1:
                    noise = rng.standard_normal(len(best_seq)) * 0.002
                    f_init = np.maximum(best_seq + noise, 1e-15)
                    f_init = f_init / f_init.sum()
                elif r == 2:
                    # Blend with a poly bump
                    pb = make_poly_bump(len(best_seq), 3)
                    f_init = 0.8 * best_seq + 0.2 * pb
                    f_init = f_init / f_init.sum()
                else:
                    # Slightly different length: interpolate
                    new_n = len(best_seq) + 64
                    old_x = np.linspace(0, 1, len(best_seq))
                    new_x = np.linspace(0, 1, new_n)
                    f_interp = np.interp(new_x, old_x, best_seq)
                    f_init = np.maximum(f_interp, 0)
                    f_init = f_init / f_init.sum()

                t_alloc = min(t_per_r, remaining - 30)
                if t_alloc < 15:
                    break
                print(f"  Restart {r}: n={len(f_init)}, time={t_alloc:.0f}s")
                a_opt, obj = adam_optimize(f_init, time_limit=t_alloc, T0=0.01, T_end=0.0002,
                                           lr=0.001, top_k_clip=100,
                                           verbose=False, label=f"R{r}")
                try_seq(a_opt, f"restart_{r}")

    # Phase 7: Final polish with very fine Adam
    print("\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_polish = min(remaining - 10, 120)
            print(f"  Polish time: {t_polish:.0f}s")
            a_opt, obj = adam_optimize(best_seq, time_limit=t_polish, T0=0.005, T_end=0.00005,
                                       lr=0.0005, top_k_clip=30,
                                       verbose=True, label="polish")
            try_seq(a_opt, "polish")

            # Try symmetrized
            a_s = symmetrize(best_seq / best_seq.sum())
            try_seq(a_s, "final_sym")

            # Fine CD on polished
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                a_opt, obj = fine_cd(best_seq, time_limit=min(remaining2 - 5, 60),
                                     verbose=False, label="final_cd")
                try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(768, 0.25)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
