```python
import cvxpy as cp
import numpy as np

def solve_lp_direction(a, M, n):
    """Solve: max sum(g) s.t. g >= 0, conv(a,g) <= M element-wise"""
    g = cp.Variable(n, nonneg=True)
    
    constraints = []
    # conv(a, g)[k] = sum_j a[j] * g[k-j] for valid j
    for k in range(2*n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        # sum_{j=js}^{je} a[j] * g[k-j] <= M
        # Let i = k-j, so i ranges from k-je to k-js
        # = sum_{i=k-je}^{k-js} a[k-i] * g[i] <= M
        i_start = k - je
        i_end = k - js
        expr = sum(a[k-i] * g[i] for i in range(i_start, i_end + 1))
        constraints.append(expr <= M)
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    if g.value is None:
        return None
    return np.maximum(g.value, 0)
```

        return np.ones(n) / n
    return a / ss


def lp_optimize(a0, time_limit, verbose=False, label="", m_factor=1.0,
                restart_interval=50, noise_scale=0.002):
    """Main LP-based optimization loop with backtracking and restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a0, float('inf')
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lp_fail_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        conv = fast_convolve(a)
        M = conv.max()
        current_obj = 2 * n * M

        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    {label} it{it}: obj={best_obj:.8f}")

        # Solve LP
        M_bound = M * m_factor
        g = lp_direction(a, M_bound, n)

        if g is None:
            # Try with larger bound
            g = lp_direction(a, M_bound * 1.05, n)
            if g is None:
                lp_fail_count += 1
                no_improve += 1
                if no_improve > 30:
                    # Restart from best with noise
                    rng = np.random.default_rng(it * 7919 + 13)
                    a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 1e-15)
                    a = a / a.sum()
                    conv = fast_convolve(a)
                    M = conv.max()
                    no_improve = 0
                    lp_fail_count = 0
                continue

        # Try different step sizes (backtracking)
        improved = False
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_conv = fast_convolve(a_trial)
            trial_M = trial_conv.max()
            trial_obj = 2 * n * trial_M
            if trial_obj < current_obj - 1e-14:
                a = a_trial
                M = trial_M
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > restart_interval:
                # Restart from best with perturbation
                rng = np.random.default_rng(it * 31337 + 7)
                scale = noise_scale * (1.0 + 0.5 * min(no_improve / restart_interval, 3.0))
                a = np.maximum(best_a + rng.standard_normal(n) * scale, 1e-15)
                a = a / a.sum()
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0

                # Also try symmetric version occasionally
                if it % (2 * restart_interval) == 0:
                    a_sym = 0.5 * (a + a[::-1])
                    a_sym = np.maximum(a_sym, 1e-15)
                    a_sym = a_sym / a_sym.sum()
                    sym_obj = 2 * n * fast_convolve(a_sym).max()
                    if sym_obj < best_obj:
                        best_obj = sym_obj
                        best_a = a_sym.copy()

    return best_a, best_obj


def mirror_descent_polish(a0, time_limit, top_k=20, temp_init=0.0002,
                          lr_init=0.005, verbose=False, label=""):
    """Fast multiplicative weight mirror descent for final polishing."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} MD it{it}: obj={2*n*best_obj:.8f}")

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        g = np.zeros(n)
        for idx in range(len(peak_indices)):
            k = int(peak_indices[idx])
            w_k = w[idx]
            if w_k < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]

        # Update with backtracking
        best_lr = lr * 0.1
        best_trial = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-10:
                continue
            log_update = np.clip(-trial_lr * g, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj <= best_trial:
                best_lr = trial_lr
                best_trial = trial_obj
                if trial_obj < M * 0.999:
                    break

        log_update = np.clip(-best_lr * g, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        conv = fast_convolve(a)
        new_obj = conv.max()
        if new_obj >= best_obj + 1e-12:
            no_improve += 1
        else:
            no_improve = 0

        temp = min(temp * 1.003, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * 0.9999, 1e-9)

        if no_improve > 400:
            rng = np.random.default_rng(it * 104729 + 3)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.001, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0

        if lr < 1e-10:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-based primary search with diverse non-symmetric
    initializations, multiple sequence lengths, and gradient-based polishing.
    Key innovation: uses exact LP solutions for direction finding (no subgradient
    approximation) and explicitly avoids symmetrization to exploit the
    non-symmetric nature of optimal solutions.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    n_base = 768
    if parent is not None:
        n_base = len(parent)

    # === Phase 1: LP optimization from parent (main time allocation) ===
    print("=== Phase 1: LP from parent ===")
    if parent is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.35, 600)
        print(f"  Time: {t_alloc:.0f}s, n={len(parent)}")
        a_opt, obj = lp_optimize(parent, time_limit=t_alloc, verbose=True,
                                 label="LP1", m_factor=1.0, restart_interval=30)
        try_seq(a_opt, "LP_parent")

    # === Phase 2: LP from diverse non-symmetric initializations ===
    print("\n=== Phase 2: LP from diverse inits ===")
    inits = []
    for n_try in [n_base, 768]:
        inits.append((make_exponential(n_try, 0.5), f"exp05_n{n_try}"))
        inits.append((make_exponential(n_try, 1.0), f"exp10_n{n_try}"))
        inits.append((make_exponential(n_try, 2.0), f"exp20_n{n_try}"))
        inits.append((make_exponential(n_try, 3.0), f"exp30_n{n_try}"))
        inits.append((make_asymmetric_gauss(n_try), f"asymG_n{n_try}"))
        inits.append((make_triangle(n_try), f"tri_n{n_try}"))
        inits.append((make_sigmoid(n_try), f"sig_n{n_try}"))

    if parent is not None:
        inits.append((parent, "parent"))

    # Quick evaluation of inits
    best_init = None
    best_init_score = float('inf')
    for a_init, label in inits:
        a_norm = a_init / a_init.sum() if a_init.sum() > 0 else None
        if a_norm is None:
            continue
        sc = try_seq(a_norm, f"init_{label}")
        if sc < best_init_score:
            best_init_score = sc
            best_init = a_norm.copy()

    if best_init is None and parent is not None:
        best_init = parent.copy()

    # Run LP on best init (if different from parent)
    if best_init is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.25, 400)
        if t_alloc > 15:
            print(f"  Best init score: {best_init_score:.6f}")
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = lp_optimize(best_init, time_limit=t_alloc, verbose=True,
                                     label="LP2", m_factor=1.0, restart_interval=30)
            try_seq(a_opt, "LP_best_init")

    # === Phase 3: LP with different m_factor (looser constraint) ===
    print("\n=== Phase 3: LP with m_factor > 1 ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.15, 200)
        if t_alloc > 10:
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = lp_optimize(best_seq, time_limit=t_alloc, verbose=True,
                                     label="LP3", m_factor=1.1, restart_interval=20,
                                     noise_scale=0.001)
            try_seq(a_opt, "LP_mfactor")

    # === Phase 4: Try different sequence lengths ===
    print("\n=== Phase 4: Different lengths ===")
    lengths_to_try = [640, 896, 1024, 1200, 512]
    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.08, 90)
        if t_alloc < 20:
            break

        # Initialize: interpolate parent or use exponential
        if parent is not None:
            old_x = np.linspace(0, 1, parent.size)
            new_x = np.linspace(0, 1, n_try)
            a_init = np.interp(new_x, old_x, parent)
        else:
            a_init = make_exponential(n_try, 0.8)
        a_init = np.maximum(a_init, 0)
        if a_init.sum() < 1e-10:
            a_init = make_exponential(n_try, 0.8)
        a_init = a_init / a_init.sum()

        print(f"  n={n_try}, time={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a_init, time_limit=t_alloc, verbose=True,
                                 label=f"LP_n{n_try}", m_factor=1.0, restart_interval=25)
        try_seq(a_opt, f"LP_n{n_try}")

    # === Phase 5: Gradient-based polish (mirror descent) ===
    print("\n=== Phase 5: MD Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.15, 180)
        if t_alloc > 10:
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = mirror_descent_polish(best_seq, time_limit=t_alloc,
                                               top_k=20, temp_init=0.0001,
                                               lr_init=0.003, verbose=True,
                                               label="polish")
            try_seq(a_opt, "MD_polish")

    # === Phase 6: Final LP polish on best ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining - 15, 120)
        if t_alloc > 10:
            print(f"\n=== Phase 6: Final LP polish ===")
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = lp_optimize(best_seq, time_limit=t_alloc, verbose=True,
                                     label="final", m_factor=1.0,
                                     restart_interval=20, noise_scale=0.0005)
            try_seq(a_opt, "final_LP")

            # Also try MD polish on the LP result
            if time.monotonic() < deadline - 20:
                t_md = min(deadline - time.monotonic() - 15, 60)
                if t_md > 5:
                    a_opt2, obj2 = mirror_descent_polish(a_opt, time_limit=t_md,
                                                         top_k=10, temp_init=0.00005,
                                                         lr_init=0.002, verbose=False,
                                                         label="final_md")
                    try_seq(a_opt2, "final_MD")

    # === Phase 7: Try symmetrized version (just in case) ===
    if best_seq is not None:
        a_sym = 0.5 * (best_seq + best_seq[::-1])
        a_sym = np.maximum(a_sym, 0)
        a_sym = a_sym / a_sym.sum()
        try_seq(a_sym, "sym_check")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
