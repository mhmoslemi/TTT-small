Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothed gradient descent with Nesterov acceleration,
subgradient refinement, and multi-length exploration.
Key improvement over LP-FW: much faster per-iteration cost (O(n log n) vs LP solve),
allowing far more iterations in the time budget.

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.array([max(v[0], 0.0)]) if n == 1 else np.zeros(0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, peaks):
    """Compute the averaged subgradient of max_k (a*a)[k] at the given peak set."""
    n = len(a)
    s = np.zeros(n)
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # s[lo:hi+1] += 2 * a[k-lo : k-hi-1 : -1]
        # Equivalently: s[lo:hi+1] += 2 * np.flip(a[k-hi : k-lo+1])
        s[lo:hi+1] += 2.0 * a[hi:lo-1:-1]
    s /= len(peaks)
    return s

def smoothed_objective_and_grad(a, conv, tau):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    F_smooth = tau * log(sum_k exp(conv[k]/tau))
    grad F_smooth = (1/sum_k exp(conv[k]/tau)) * sum_k exp(conv[k]/tau) * grad(conv[k])
    """
    n = len(a)
    # Numerical stability
    max_val = conv.max()
    shifted = (conv - max_val) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    F = tau * (np.log(sum_exp) + max_val / tau)
    
    # Gradient: weighted average of subgradients
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-15:
            continue
        kk = int(k)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        grad[lo:hi+1] += weights[k] * 2.0 * a[hi:lo-1:-1]
    
    return F, grad

def lse_gradient_descent(a0, time_limit, tau_init=1e-3, tau_decay=0.98, verbose=False):
    """
    Log-sum-exp smoothed gradient descent with Nesterov acceleration.
    Optimizes on the simplex (sum=1, a>=0).
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    tau = tau_init
    step = 1.0 / (2.0 * n)  # Initial step size (Lipschitz constant estimate)
    it = 0
    no_improve = 0
    
    # Nesterov momentum
    y = f.copy()
    beta = 0.9
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute smoothed gradient at current point
        conv_y = fast_convolve(y)
        F, grad = smoothed_objective_and_grad(y, conv_y, tau)
        
        # Nesterov step
        f_new = y - step * grad
        f_new = project_simplex(f_new)
        
        # Momentum update
        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta**2)) / 2.0
        y = f_new + (beta - 1.0) / (beta + 1.0) * (f_new - f)
        y = project_simplex(y)
        
        # Check improvement
        conv_f = fast_convolve(f_new)
        obj_val = 2.0 * n * conv_f.max()
        
        if obj_val < best_obj - 1e-12:
            best_obj = obj_val
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive step size
        if no_improve > 10:
            step *= 0.7
            no_improve = 0
        elif no_improve == 0 and it > 10:
            step = min(step * 1.05, 1.0 / (2.0 * n) * 10)
        
        # Decay temperature
        if it % 5 == 0:
            tau *= tau_decay
            if tau < 1e-8:
                tau = 1e-8
        
        # Periodic subgradient refinement
        if it % 100 == 0:
            f_sub = subgradient_step(best_f, max_iter=5)
            conv_sub = fast_convolve(f_sub)
            obj_sub = 2.0 * n * conv_sub.max()
            if obj_sub < best_obj:
                best_obj = obj_sub
                best_f = f_sub.copy()
                f = best_f.copy()
                y = best_f.copy()
        
        f = f_new
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD iter {it}: obj={best_obj:.10f}, tau={tau:.2e}, step={step:.2e}, elapsed={elapsed:.0f}s")

def subgradient_step(f0, max_iter=10, n_peaks=5):
    """
    A few subgradient steps with backtracking on the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s
    n = len(f)
    
    step = 1.0 / (4.0 * n)
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    
    for _ in range(max_iter):
        conv = fast_convolve(f)
        M = conv.max()
        
        # Get top peaks
        k_actual = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        peaks = peaks[conv[peaks] > M * 0.99]  # Only near-max peaks
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(conv))])
        
        # Compute subgradient
        sgrad = compute_subgradient(f, peaks)
        
        # Projected subgradient step
        f_trial = f - step * sgrad
        f_trial = project_simplex(f_trial)
        
        conv_trial = fast_convolve(f_trial)
        obj_trial = 2.0 * n * conv_trial.max()
        
        if obj_trial < best_obj:
            best_obj = obj_trial
            best_f = f_trial.copy()
            f = f_trial
        else:
            step *= 0.5
            if step < 1e-10:
                break
    
    return best_f

"""
Hybrid gradient-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LSE-smoothed Nesterov-accelerated gradient descent, subgradient refinement
with Barzilai-Borwein steps, targeted peak shaving, and multi-length exploration.
Key advantage over LP-FW: O(n^2) per iteration (vs expensive LP solve), enabling
far more optimization steps within the time budget.
"""

import numpy as np
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peaks):
    """Averaged subgradient of max_k (a*a)[k] at given peak indices."""
    n = len(a)
    s = np.zeros(n)
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        s[lo:hi + 1] += 2.0 * a[hi:lo - 1:-1]
    s /= len(peaks)
    return s


def lse_gradient_descent(a0, time_limit, tau_init=5e-4, tau_floor=1e-6,
                         tau_decay=0.97, verbose=False):
    """
    Log-sum-exp smoothed Nesterov-accelerated gradient descent on the simplex.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    step = 0.5 / n
    it = 0
    no_improve = 0

    y = f.copy()
    beta = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Compute LSE-smoothed gradient at y
        conv_y = fast_convolve(y)
        max_v = conv_y.max()
        shifted = (conv_y - max_v) / tau
        exp_shifted = np.exp(shifted)
        sum_exp = exp_shifted.sum()
        weights = exp_shifted / sum_exp

        grad = np.zeros(n)
        for k in range(len(conv_y)):
            w = weights[k]
            if w < 1e-14:
                continue
            kk = int(k)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            grad[lo:hi + 1] += w * 2.0 * y[hi:lo - 1:-1]

        # Nesterov update
        f_new = y - step * grad
        f_new = project_simplex(f_new)

        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = (beta - 1.0) / (beta + 1.0)
        y_new = f_new + momentum * (f_new - f)
        y_new = project_simplex(y_new)

        # Evaluate
        conv_f = fast_convolve(f_new)
        obj_val = 2.0 * n * conv_f.max()

        if obj_val < best_obj - 1e-13:
            best_obj = obj_val
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step
        if no_improve > 20:
            step *= 0.6
            no_improve = 0
        elif no_improve == 0:
            step = min(step * 1.03, 2.0 / n)

        # Decay temperature periodically
        if it % 20 == 0:
            tau = max(tau * tau_decay, tau_floor)

        # Periodic subgradient refinement every 200 iters
        if it % 200 == 0:
            f_sub = subgradient_refine(best_f, n_iters=10)
            conv_sub = fast_convolve(f_sub)
            obj_sub = 2.0 * n * conv_sub.max()
            if obj_sub < best_obj - 1e-14:
                best_obj = obj_sub
                best_f = f_sub.copy()
                f = best_f.copy()
                y = best_f.copy()
                beta = 1.0

        f = f_new
        y = y_new

        if verbose and it % 2000 == 0:
            print(f"    LSE iter {it}: obj={best_obj:.10f}, tau={tau:.1e}, "
                  f"step={step:.1e}, t={elapsed:.0f}s")

    return best_f, best_obj


def subgradient_refine(f0, n_iters=20, n_peaks=8):
    """Subgradient descent with Barzilai-Borwein adaptive step."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()

    step = 1.0 / (3.0 * n)
    prev_f = f.copy()
    prev_sg = None

    for it in range(n_iters):
        conv = fast_convolve(f)
        M = conv.max()

        k_act = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        mask = conv[peaks] >= M * 0.995
        if mask.sum() == 0:
            peaks = np.array([int(np.argmax(conv))])
        else:
            peaks = peaks[mask]

        sg = compute_subgradient(f, peaks)

        # BB step size
        if prev_sg is not None:
            s_dir = f - prev_f
            y_dir = sg - prev_sg
            sy = s_dir @ y_dir
            yy = y_dir @ y_dir
            if yy > 1e-20:
                if sy > 1e-15:
                    step = sy / yy
                else:
                    step = (s_dir @ s_dir) / max(sy, -1e-10) if sy < -1e-15 else step
                step = np.clip(step, 1e-8, 5.0 / n)

        f_trial = f - step * sg
        f_trial = project_simplex(f_trial)

        conv_t = fast_convolve(f_trial)
        obj_t = 2.0 * n * conv_t.max()

        if obj_t < best_obj:
            best_obj = obj_t
            best_f = f_trial.copy()
            prev_f = f.copy()
            prev_sg = sg.copy()
            f = f_trial
        else:
            step *= 0.5
            if step < 1e-12:
                break

    return best_f

def interpolate_sequence(f, new_n):
    """Linear interpolation to new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_int = np.interp(new_x, old_x, f)
    f_int = np.maximum(f_int, 0)
    ss = f_int.sum()
    if ss > 1e-15:
        f_int /= ss
    else:
        f_int = np.ones(new_n) / new_n
    return f_int


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tilted_gaussian(n, sigma_frac=0.25, tilt=0.3):
    """Gaussian with linear tilt to break symmetry."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a *= (1.0 + tilt * x / (n / 2.0))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bilateral(n, sigma_frac=0.2, tilt_frac=0.5):
    """Bilateral: two overlapping Gaussians with asymmetry."""
    x = np.arange(n)
    center1 = n * (0.5 - tilt_frac / 4.0)
    center2 = n * (0.5 + tilt_frac / 4.0)
    sigma = n * sigma_frac
    a1 = np.exp(-0.5 * ((x - center1) / sigma) ** 2)
    a2 = np.exp(-0.5 * ((x - center2) / sigma) ** 2)
    a = 0.6 * a1 + 0.4 * a2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

            print(f"  Restart {p}: t={t_alloc:.0f}s")
            f_opt, obj_opt = optimize_at_length(f_init, t_alloc, verbose=False)
            best_seq, best_score = try_sequence(f_opt, best_seq, best_score,
                                                 f"restart_{p}")

    # === Phase 5: Deep refinement ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Deep Refinement (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 40
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  Deep: t={t_alloc:.0f}s")

            # Extended LSE with lower temperature
            t_lse = t_alloc * 0.5
            f_lse, obj_lse = lse_gradient_descent(best_seq, time_limit=t_lse,
                                                   tau_init=2e-4, tau_floor=5e-7,
                                                   tau_decay=0.96, verbose=True)
            best_seq, best_score = try_sequence(f_lse, best_seq, best_score, "deep_lse")

            # Peak shaving
            rem2 = deadline - time.monotonic() - 20
            if rem2 > 15 and best_seq is not None:
                t_ps = min(rem2 * 0.5, 40)
                f_ps, obj_ps = peak_shaving(best_seq, time_limit=t_ps, verbose=True)
                best_seq, best_score = try_sequence(f_ps, best_seq, best_score, "deep_peak")

            # Final subgradient
            rem3 = deadline - time.monotonic() - 10
            if rem3 > 10 and best_seq is not None:
                n_iter = max(20, int(rem3 * 80))
                f_sg = subgradient_refine(best_seq, n_iters=n_iter, n_peaks=10)
                conv_sg = fast_convolve(f_sg)
                obj_sg = 2.0 * len(f_sg) * conv_sg.max()
                if obj_sg < best_score / 2.0:  # rough check
                    best_seq, best_score = try_sequence(f_sg, best_seq, best_score, "deep_sg")

    # === Phase 6: Asymmetric length sweep (quick) ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Quick Length Fine-tune ===")
        n_b = len(best_seq)
        for delta in [-16, 16, -8, 8]:
            rem = deadline - time.monotonic() - 15
            if rem < 20:
                break
            t_new = n_b + delta
            if t_new < 128:
                continue
            f_new = interpolate_sequence(best_seq, t_new)
            best_seq, best_score = try_sequence(f_new, best_seq, best_score,
                                                 f"len_{t_new}")
            # Quick LSE pass
            t_quick = min(rem * 0.3, 25)
            if t_quick > 5:
                f_q, _ = lse_gradient_descent(f_new, time_limit=t_quick,
                                               tau_init=3e-4, tau_floor=1e-6)
                best_seq, best_score = try_sequence(f_q, best_seq, best_score,
                                                     f"lseq_{t_new}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
