def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_descent(f0, time_limit, temperature_init=0.001, verbose=False):
    """Nesterov-accelerated gradient descent with softmax approximation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    y = f.copy()  # Nesterov momentum point
    v = np.zeros(n)  # Momentum
    beta = 0.9  # Momentum parameter
    
    temperature = temperature_init
    lr = 0.01 / n  # Initial learning rate (scaled by n)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break
        
        # Nesterov step: y = f + beta * v
        y = f + beta * v
        
        # Compute gradient at y
        grad, current_max = softmax_gradient(y, temperature)
        
        # Gradient step
        f_new = y - lr * grad
        f_new = project_simplex(f_new)
        
        # Update momentum
        v = f_new - y
        f = f_new
        
        # Check improvement
        obj = 2.0 * n * fast_convolve(f).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    ND iter {it}: obj={best_obj:.10f}, temp={temperature:.6f}")
        else:
            no_improve += 1
        
        # Adapt temperature (annealing)
        temperature *= 1.001
        
        # Adapt learning rate
        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
        
        # Periodic restart with perturbation
        if no_improve > 500:
            rng = np.random.default_rng(it * 31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            y = f.copy()
            v = np.zeros(n)
            lr = 0.01 / n
            temperature = temperature_init
            no_improve = 0
        
        # Decay momentum
        beta = min(0.99, beta * 1.0001)
    
    return best_f, best_obj

def peak_shaving(f0, time_limit, verbose=False):

    n = len(f)

    best_f = f.copy()

    while time.monotonic() - t0 < time_limit and no_improve < 100:
        conv = fast_convolve(f)
        M = conv.max()
        
        # Get top 10 peaks
        top_k = min(10, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        
        improved = False
        for peak_k in peak_indices:
            peak_k = int(peak_k)
            if conv[peak_k] < M - 1e-10:
                break
            
            # Find contributing indices
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            contrib_idx = np.arange(js, je + 1)
            if len(contrib_idx) == 0:
                continue
            
            products = f[contrib_idx] * f[peak_k - contrib_idx]
            
            # Find the top contributing coordinate
            max_prod_pos = np.argmax(products)
            source = contrib_idx[max_prod_pos]
            
            # Try reducing source and redistributing
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                diff = f[source] * frac

def make_flat_target_init(n, seed=0):
    """Initialize by trying to match a flat auto-correlation target."""
    rng = np.random.default_rng(seed)
    # Start with a slightly perturbed uniform
    a = np.ones(n) + rng.standard_normal(n) * 0.05
    a = np.maximum(a, 0)

def make_sinusoidal(n, freq, amp, phase=0, seed=0):
    """Sinusoidal modulation of uniform."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a

def make_multisinusoidal(n, n_freqs=5, max_amp=0.3, seed=0):
    """Multi-frequency sinusoidal modulation."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    x = np.arange(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.02, max_amp) / (m ** 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * m * x / n + phase)
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a

def make_binary_lite(n, density=0.5, seed=0):
    """Binary-like sequence (smoothed)."""
    rng = np.random.default_rng(seed)
    b = (rng.random(n) < density).astype(float)
    # Smooth it
    kernel = np.ones(5) / 5
    a = np.convolve(b, kernel, mode='same')
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n + 1)
    new_x = np.linspace(0, 1, new_n + 1)
    f_padded = np.concatenate([f, [f[-1]]])  # Pad for interpolation
    f_interp = np.interp(new_x, old_x, f_padded)
    f_interp = f_interp[:new_n]
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_gradient(f, conv, temperature):
    """
    Compute the gradient of the softmax-smoothed max of conv(f,f).
    g[j] = 2 * sum_k w[k] * f[k-j] where w[k] = softmax(conv/T)[k]
    """
    n = len(f)
    conv_len = len(conv)

    # Softmax weights
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(conv_len)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum

    # g[j] = 2 * sum_{l=0}^{n-1} f[l] * w[j+l]
    # This equals 2 * (w corr f)[j] where (w corr f)[j] = sum_k w[k] * f[k-j]
    # Compute via FFT: ifft(FFT(w, N) * conj(FFT(f, N)))[:n]
    N = 1 << (n + conv_len - 1).bit_length()
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    corr = np.fft.irfft(W * np.conj(F), N)
    grad = 2.0 * corr[:n]

    return grad


def fista_smoothed(f0, time_limit, temp_init=0.002, temp_final=0.0001, verbose=False):
    """FISTA with softmax-smoothed max-conv objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv0 = fast_convolve(f)
    best_obj = 2.0 * n * conv0.max()
    best_f = f.copy()
    t0 = time.monotonic()

    y = f.copy()
    t_k = 1.0

    # Initial step size - estimate smoothness
    temp = temp_init
    lr = 2.0 / (n * temp)

    it = 0
    no_improve = 0
    total_iters_budget = int(time_limit / 0.02)  # rough estimate

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.3:
            break

        # Animate temperature
        progress = min(1.0, it / max(1, total_iters_budget))
        temp = temp_init * (temp_final / temp_init) ** progress

        # Compute gradient at y
        conv_y = fast_convolve(y)
        grad = compute_soft_gradient(y, conv_y, temp)

        # Gradient step
        f_new = y - lr * grad

        # Proximal step: project onto simplex
        f_new = project_simplex(f_new)

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = f_new + ((t_k - 1.0) / t_new) * (f_new - f)
        t_k = t_new

        # Update
        f = f_new

        # Evaluate
        obj = 2.0 * n * fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
            # Perturbation to escape
            if no_improve > 800:
                rng = np.random.default_rng(it * 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                y = f.copy()
                t_k = 1.0
                lr = 2.0 / (n * temp)
                no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    FISTA iter {it}: obj={best_obj:.10f}, temp={temp:.6f}, lr={lr:.8f}")

        if no_improve > 2000:
            break

    return best_f, best_obj


def multiplicative_fista(f0, time_limit, temp_init=0.002, verbose=False):
    """Multiplicative mirror descent with Nesterov acceleration."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv0 = fast_convolve(f)
    best_obj = 2.0 * n * conv0.max()
    best_f = f.copy()
    t0 = time.monotonic()

    y = f.copy()
    t_k = 1.0
    temp = temp_init
    lr = 0.05

    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.3:
            break

        progress = min(1.0, it / 5000.0)
        temp = temp_init * (0.0002 / temp_init) ** progress

        conv_y = fast_convolve(y)
        grad = compute_soft_gradient(y, conv_y, temp)

        # Multiplicative update (mirror descent with entropy OMD)
        f_new = y * np.exp(-lr * grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        # Nesterov
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = f_new + ((t_k - 1.0) / t_new) * (f_new - f)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        t_k = t_new

        f = f_new

        obj = 2.0 * n * fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-5:
                rng = np.random.default_rng(it * 23)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-12)
                f = f / f.sum()
                y = f.copy()
                t_k = 1.0
                lr = 0.05
                no_improve = 0

        if verbose and it % 300 == 0:
            print(f"    MFISTA iter {it}: obj={best_obj:.10f}, temp={temp:.6f}, lr={lr:.6f}")

        if no_improve > 3000:
            break

    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """Direct peak shaving: reduce top-contributing pairs and redistribute."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 150:
        conv = fast_convolve(f)
        M = conv.max()

        top_k = min(15, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        improved = False
        for pk in peak_indices:
            pk = int(pk)
            if conv[pk] < M - 1e-8:
                break

            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            contrib_idx = np.arange(js, je + 1)
            if len(contrib_idx) == 0:
                continue

            products = f[contrib_idx] * f[pk - contrib_idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Try reducing the top-3 contributing positions
            top_contrib = np.argsort(products)[::-1][:3]
            for tc in top_contrib:
                source = contrib_idx[tc]
                partner = pk - source

                for frac in [0.05, 0.1, 0.2, 0.3]:
                    diff = f[source] * frac
                    if diff < 1e-15:
                        continue

                    trial = f.copy()
                    trial[source] -= diff

                    # Redistribute to positions NOT contributing to this peak
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[contrib_idx] = False
                    if non_contrib.sum() < 2:
                        non_contrib[:] = True
                        non_contrib[source] = False

                    nc_idx = np.where(non_contrib)[0]
                    # Weight by inverse of their current contribution to top peaks
                    trial[nc_idx] += diff / len(nc_idx)

                    trial = np.maximum(trial, 0.0)
                    trial = trial / trial.sum()

                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

                if improved:
                    break

            if improved:
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 50 == 0:
            pass

    return best_f, best_obj


def make_multisinusoidal(n, n_freqs=5, max_amp=0.25, seed=0):
    """Multi-frequency sinusoidal modulation of uniform."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    x = np.arange(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.02, max_amp) / np.sqrt(m)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * m * x / n + phase)
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_cosine_mod(n, freq, amp, phase=0.0):
    """Single cosine modulation."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    return a


def make_two_point(n, pos_frac=0.3, width=0.1):
    """Two concentrated bumps."""
    x = np.arange(n) / n
    centers = [pos_frac, 1.0 - pos_frac]
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-0.5 * ((x - c) / (width / np.sqrt(2 * np.pi))) ** 2)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 1e-12)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    print(f"  Top init: {all_candidates[0][2]}, obj={all_candidates[0][0]:.6f}")
    print(f"  Phase 1: {time.monotonic()-t0:.1f}s, best={best_score:.6f}")

    # === Phase 2: FISTA on top candidates ===
    print("\n=== Phase 2: FISTA Optimization ===")
    tried_lengths = set()
    for sc, a_c, label in all_candidates[:6]:
        rem = deadline - time.monotonic() - 120
        if rem < 30:
            break
        n_c = len(a_c)
        if n_c in tried_lengths:
            continue
        tried_lengths.add(n_c)

        t_alloc = min(rem * 0.35, 150)
        if t_alloc < 20:
            break
        print(f"  FISTA [{label}] n={n_c}: {t_alloc:.0f}s (init_obj={sc:.6f})")
        a_opt, obj = fista_smoothed(a_c, time_limit=t_alloc, temp_init=0.003, verbose=True)
        try_seq(a_opt, f"fista_{label}")

    # === Phase 3: Multiplicative FISTA ===
    print("\n=== Phase 3: Multiplicative Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_alloc = min(rem * 0.4, 120)
            print(f"  MFISTA n={len(best_seq)}: {t_alloc:.0f}s")
            a_opt, obj = multiplicative_fista(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "mfista")

    # === Phase 4: Peak shaving ===
    print("\n=== Phase 4: Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem * 0.5, 80)
            print(f"  PS n={len(best_seq)}: {t_alloc:.0f}s")
            a_opt, obj = peak_shaving(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_shave")

    # === Phase 5: Multi-length refinement ===
    print("\n=== Phase 5: Multi-length Refinement ===")
    if best_seq is not None:
        base_n = len(best_seq)
        targets = sorted(set([max(256, base_n // 2), base_n, min(4096, base_n * 3 // 2),
                               min(4096, base_n * 2)]))
        for tn in targets:
            if tn == base_n:
                continue
            rem = deadline - time.monotonic() - 40
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, tn)
            sc = try_seq(f_interp, f"interp_{tn}")
            if sc <= best_score + 0.03:
                t_alloc = min(rem * 0.4, 60)
                if t_alloc > 10:
                    a_opt, obj = fista_smoothed(f_interp, time_limit=t_alloc,
                                                temp_init=0.002, verbose=False)
                    try_seq(a_opt, f"fista_interp_{tn}")

    # === Phase 6: Final combined polish ===
    print("\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t1 = min(rem * 0.4, 40)
            print(f"  FISTA polish: {t1:.0f}s")
            a_opt, obj = fista_smoothed(best_seq, time_limit=t1,
                                       temp_init=0.001, temp_final=0.0001, verbose=True)
            try_seq(a_opt, "polish_fista")

        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t2 = min(rem * 0.4, 30)
            print(f"  PS polish: {t2:.0f}s")
            a_opt, obj = peak_shaving(best_seq, time_limit=t2)
            try_seq(a_opt, "polish_ps")

        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t3 = min(rem * 0.5, 20)
            print(f"  MFISTA polish: {t3:.0f}s")
            a_opt, obj = multiplicative_fista(best_seq, time_limit=t3,
                                             temp_init=0.0008, verbose=False)
            try_seq(a_opt, "polish_mfista")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
