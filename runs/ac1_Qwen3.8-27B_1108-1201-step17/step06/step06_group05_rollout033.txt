Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using symmetric LP-based Frank-Wolfe
with cutting planes, multiple length exploration, and coordinate polishing.

import numpy as np
from scipy.optimize import linprog
import time


def full_from_half(h, n):
    """Expand half-sequence to full symmetric sequence."""
    return np.concatenate([h, h[::-1]])


def half_from_full(f, n):
    """Extract half-sequence from full symmetric sequence."""
    return f[:n//2].copy()


def convolve_first_half(h, n):
    """Compute (f*f)[0..n-1] where f is symmetric with half-representation h."""
    f = full_from_half(h, n)
    # Use FFT for speed
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    c = np.fft.irfft(F * F, fft_size)[:size]
    return c[:n]  # First half by symmetry


def build_lp_matrix(h_f, n):
    """Build the constraint matrix A for the LP.
    A[k, i] = coefficient of h_g[i] in (f*g)[k]
    where f is symmetric with half-rep h_f, g is symmetric with half-rep h_g.
    """
    half_n = n // 2
    A = np.zeros((n, half_n))
    f = full_from_half(h_f, n)
    for i in range(half_n):
        # g[i] = h_g[i] and g[n-1-i] = h_g[i]
        # (f*g)[k] += f[k-i] * h_g[i] + f[k-(n-1-i)] * h_g[i]
        # = f[k-i] * h_g[i] + f[k-n+1+i] * h_g[i]
        for k in range(n):
            val = 0.0
            ki = k - i
            if 0 <= ki < n:
                val += f[ki]
            ki2 = k - n + 1 + i
            if 0 <= ki2 < n:
                val += f[ki2]
            A[k, i] = val
    return A


def build_lp_matrix_fast(h_f, n):

    half_n = n // 2
    f = full_from_half(h_f, n)
    A = np.zeros((n, half_n))
    for i in range(half_n):

        # f[k-i] (if 0<=k-i<n) + f[k-(n-1-i)] (if 0<=k-(n-1-i)<n)

        idx1 = ks - i  # k - i

        A[:, i] = np.where((idx1 >= 0) & (idx1 < n), f[idx1.clip(0, n-1)], 0) + \
                  np.where((idx2 >= 0) & (idx2 < n), f[idx2.clip(0, n-1)], 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric LP-based Frank-Wolfe (main workhorse)
2. Cutting-plane acceleration
3. Multi-length exploration
4. Coordinate descent polishing
"""

import numpy as np
from scipy.optimize import linprog
import time

# All helper functions at top level

def sym_full(h):
    """Expand half to full symmetric sequence."""
    return np.concatenate([h, h[::-1]])

def sym_half(f):
    """Extract half from full symmetric sequence."""
    return f[:len(f)//2].copy()

def conv_first_half_fft(h):
    """Compute (f*f)[0..n-1] for symmetric f, using FFT."""
    f = sym_full(h)
    n = len(f)
    size = 2*n - 1
    fs = 1 << (size-1).bit_length()
    F = np.fft.rfft(f, fs)
    c = np.fft.irfft(F*F, fs)[:size]
    return c[:n]

def conv_full_fft(h):
    """Compute full (f*f) for symmetric f, using FFT."""
    f = sym_full(h)
    n = len(f)
    size = 2*n - 1
    fs = 1 << (size-1).bit_length()
    F = np.fft.rfft(f, fs)
    return np.fft.irfft(F*F, fs)[:size]

def build_A(h_f):
    """Build the LP constraint matrix A (n x n//2) for symmetric FW."""
    n = 2 * len(h_f)
    hn = n // 2
    f = sym_full(h_f)
    A = np.zeros((n, hn))
    for i in range(hn):
        # For each k in 0..n-1: A[k,i] = f[k-i] (if valid) + f[k-(n-1-i)] (if valid)
        # Vectorized:
        ks = np.arange(n)
        i1 = ks - i
        i2 = ks - (n - 1 - i)
        valid1 = (i1 >= 0) & (i1 < n)
        valid2 = (i2 >= 0) & (i2 < n)
        col = np.zeros(n)
        col[valid1] = f[i1[valid1]]
        col[valid2] += f[i2[valid2]]
        A[:, i] = col
    return A

def solve_lp_direction(h_f, M, active_idx):
    """Solve LP: max sum(h_g) s.t. A[active_idx] @ h_g <= M, h_g >= 0."""
    n = 2 * len(h_f)
    hn = n // 2
    A_full = build_A(h_f)
    A = A_full[active_idx, :]
    b = np.full(len(active_idx), M)
    c_obj = -np.ones(hn)
    bounds = [(0, None)] * hn
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fw_iteration(h, active_idx, max_iter_cutting=50):
    """One FW iteration with cutting planes. Returns new h_direction (normalized) or None."""
    n = 2 * len(h)
    # Current max
    c = conv_first_half_fft(h)
    M = c.max()
    
    # Find active constraints (most violated)
    # Start with top constraints
    n_active = min(len(c), max(50, n // 4))
    active_idx = np.argpartition(c, -n_active)[-n_active:]
    
    for _ in range(max_iter_cutting):
        h_g = solve_lp_direction(h, M, active_idx)
        if h_g is None:
            # LP infeasible, expand active set
            violated = np.where(c > M + 1e-10)[0]
            if len(violated) == 0:
                break
            # Add most violated
            active_idx = np.union1d(active_idx, violated[:10])
            continue
        
        # Check if all constraints are satisfied
        A_full = build_A(h)
        lhs = A_full @ h_g
        max_violation = lhs.max() - M
        if max_violation < 1e-10:
            break
        # Add violated constraints
        violated = np.where(lhs > M + 1e-10)[0]
        active_idx = np.union1d(active_idx, violated[:20])
    
    if h_g is None or h_g.sum() < 1e-15:
        return None, M
    
    # Normalize
    h_g = h_g / h_g.sum()
    return h_g, M

def line_search_sym(h1, h2, n):
    """Find alpha in [0,1] minimizing max(conv_first_half((1-a)*h1 + a*h2))."""
    best_alpha = 0.0
    best_val = conv_first_half_fft(h1).max()
    
    # Grid search
    for alpha in np.linspace(0.05, 1.0, 20):
        h_t = (1 - alpha) * h1 + alpha * h2
        val = conv_first_half_fft(h_t).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    # Refine
    if best_alpha > 0:
        lo = max(0, best_alpha - 0.05)
        hi = min(1.0, best_alpha + 0.05)
        for _ in range(15):
            mid = (lo + hi) / 2
            h_t = (1 - mid) * h1 + mid * h2
            val = conv_first_half_fft(h_t).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid
    
    return best_alpha, best_val

def symmetric_lp_fw(h0, time_limit, verbose=False):
    """Main symmetric LP-FW optimizer."""
    h = h0.copy().astype(np.float64)
    h = np.maximum(h, 0)
    s = h.sum()
    if s < 1e-15:
        h = np.ones(len(h0)) / len(h0)
    else:
        h = h / s
    
    n = 2 * len(h)
    c = conv_first_half_fft(h)
    best_obj = 2 * n * c.max()
    best_h = h.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        h_g, M = fw_iteration(h, None)
        if h_g is None:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Check fixed point
        if np.max(np.abs(h_g - h)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        no_improve = 0
        
        # Line search
        alpha, val = line_search_sym(h, h_g, n)
        
        if alpha > 1e-8:
            h_new = (1 - alpha) * h + alpha * h_g
            h_new = np.maximum(h_new, 0)
            h_new = h_new / h_new.sum()
            
            obj = 2 * n * conv_first_half_fft(h_new).max()
            if obj < best_obj:
                best_obj = obj
                best_h = h_new.copy()
            h = h_new
            
            if verbose and it % 20 == 0:

                print(f"    Sym LP-FW iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")
        else:
            no_improve += 1
            if no_improve > 50:
                break
    
    return best_h, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric LP-based Frank-Wolfe with full constraint set (main workhorse)
2. Multi-length exploration with diverse symmetric initializations
3. Strengthened line search via golden section
4. Coordinate descent polishing on the half-sequence
Key innovation: Exploiting sequence symmetry to halve the LP dimensions,
enabling far more FW iterations within the time budget.
"""

import numpy as np
from scipy.optimize import linprog
import time


def sym_full(h):
    """Expand half to full symmetric sequence."""
    return np.concatenate([h, h[::-1]])


def sym_half(f):
    """Extract first half from full symmetric sequence."""
    return f[:len(f) // 2].copy()


def conv_first_half_fft(h):
    """Compute (f*f)[0..n-1] for symmetric f with half-rep h, using FFT."""
    f = sym_full(h)
    n = len(f)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fs)
    c = np.fft.irfft(F * F, fs)[:size]
    return c[:n]


def conv_full_fft(h):
    """Compute full (f*f) for symmetric f with half-rep h, using FFT."""
    f = sym_full(h)
    n = len(f)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fs)
    return np.fft.irfft(F * F, fs)[:size]


def build_A(h_f):
    """Build LP constraint matrix A (n x n//2) for symmetric FW direction."""
    n = 2 * len(h_f)
    hn = n // 2
    f = sym_full(h_f)
    A = np.zeros((n, hn))
    for i in range(hn):
        ks = np.arange(n)
        i1 = ks - i
        i2 = ks - (n - 1 - i)
        valid1 = (i1 >= 0) & (i1 < n)
        valid2 = (i2 >= 0) & (i2 < n)
        col = np.zeros(n)
        col[valid1] = f[i1[valid1]]
        col[valid2] += f[i2[valid2]]
        A[:, i] = col
    return A


def solve_lp_direction(h_f, M):
    """Solve LP: max sum(h_g) s.t. A @ h_g <= M, h_g >= 0. Returns h_g or None."""
    n = 2 * len(h_f)
    hn = n // 2
    A = build_A(h_f)
    b = np.full(n, M)
    c_obj = -np.ones(hn)
    bounds = [(0, None)] * hn
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def line_search_sym(h1, h2):
    """Golden section + grid search for best alpha mixing h1 and h2."""
    c1 = conv_first_half_fft(h1)
    best_val = c1.max()
    best_alpha = 0.0

    # Coarse grid
    for alpha in np.linspace(0.02, 1.0, 25):
        h_t = (1 - alpha) * h1 + alpha * h2
        val = conv_first_half_fft(h_t).max()
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha

    # Golden section refinement
    if best_alpha > 0.01:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = max(0.0, best_alpha - 0.08)
        hi = min(1.0, best_alpha + 0.08)
        for _ in range(30):
            if hi - lo < 1e-9:
                break
            d = hi - lo
            x1 = hi - gr * d
            x2 = lo + gr * d
            h_t1 = (1 - x1) * h1 + x1 * h2
            h_t2 = (1 - x2) * h1 + x2 * h2
            f1 = conv_first_half_fft(h_t1).max()
            f2 = conv_first_half_fft(h_t2).max()
            if f1 < f2:
                hi = x2
                if f1 < best_val:
                    best_val = f1
                    best_alpha = x1
            else:
                lo = x1
                if f2 < best_val:
                    best_val = f2
                    best_alpha = x2
        # Check midpoint
        mid = (lo + hi) / 2.0
        h_m = (1 - mid) * h1 + mid * h2
        fm = conv_first_half_fft(h_m).max()
        if fm < best_val:
            best_val = fm
            best_alpha = mid

    return best_alpha, best_val


def symmetric_lp_fw(h0, time_limit, verbose=False):
    """Symmetric LP-based Frank-Wolfe optimizer."""
    h = h0.copy().astype(np.float64)
    h = np.maximum(h, 0)
    s = h.sum()
    if s < 1e-15:
        h = np.ones(len(h0)) / len(h0)
    else:
        h = h / s

    n = 2 * len(h)
    c = conv_first_half_fft(h)
    best_obj = 2.0 * n * c.max()
    best_h = h.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    stagnation_counter = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        c = conv_first_half_fft(h)
        M = c.max()

        h_g = solve_lp_direction(h, M * (1.0 + 1e-12))
        if h_g is None:
            no_improve += 1
            if no_improve > 100:
                break
            continue

        g_sum = h_g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            if no_improve > 100:
                break
            continue

        h_g = h_g / g_sum

        # Fixed point check
        diff = np.max(np.abs(h_g - h))
        if diff < 1e-14:
            no_improve += 1
            if no_improve > 80:
                break
            continue

        no_improve = 0

        # Line search
        alpha, val = line_search_sym(h, h_g)

        if alpha < 1e-10:
            no_improve += 1
            if no_improve > 80:
                break
            continue

        h_new = (1 - alpha) * h + alpha * h_g
        h_new = np.maximum(h_new, 0)
        sn = h_new.sum()
        if sn < 1e-15:
            break
        h_new = h_new / sn

        obj = 2.0 * n * conv_first_half_fft(h_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h_new.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        h = h_new

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SymLPFW iter {it}: obj={best_obj:.10f}, M={M:.8f}, alpha={alpha:.4f}, t={elapsed:.1f}s")

        if stagnation_counter > 300:
            # Perturb to escape
            rng = np.random.default_rng(it * 7 + 3)
            h = np.maximum(best_h + rng.standard_normal(len(best_h)) * 0.001, 0)
            h = h / h.sum()
            stagnation_counter = 0
            no_improve = 0

    return best_h, best_obj


def make_hann_half(hn):
    """Hann window half-sequence of length hn."""
    n = 2 * hn
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return sym_half(a)


def make_gaussian_half(hn, sigma_frac=0.3):
    """Gaussian half-sequence."""
    n = 2 * hn
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return sym_half(a)


def make_flattened_half(hn, power=2):
    """Flattened-top window: minimize peak of convolution."""
    n = 2 * hn
    x = np.linspace(-1, 1, n)
    a = 1.0 - np.abs(x) ** (2 * power)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return sym_half(a)


def make_uniform_half(hn):
    """Uniform half-sequence."""
    return np.ones(hn) / (2 * hn)


def coordinate_polish_half(h0, time_limit):
    """Coordinate descent on the half-sequence to reduce max convolution."""
    h = h0.copy().astype(np.float64)
    h = np.maximum(h, 0)
    s = h.sum()
    if s < 1e-15:
        return h0, 2.0 * 2 * len(h0) * conv_first_half_fft(h0).max()
    h = h / s
    n = 2 * len(h)

    best_h = h.copy()
    best_obj = 2.0 * n * conv_first_half_fft(h).max()
    t0 = time.monotonic()
    no_full_pass_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_full_pass_improve > 3:
            break
        improved_this_pass = False
        c = conv_first_half_fft(h)

        for j in range(len(h)):
            if time.monotonic() - t0 > time_limit:
                break
            old = h[j]
            if old < 1e-14:
                continue
            best_j_val = old
            best_j_obj = best_obj
            for mult in [0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]:
                new_val = old * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                h_t = h.copy()
                h_t[j] = new_val
                h_t = np.maximum(h_t, 0)
                st = h_t.sum()
                if st < 1e-15:
                    continue
                h_t = h_t / st
                val = 2.0 * n * conv_first_half_fft(h_t).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old:
                h[j] = best_j_val
                h = np.maximum(h, 0)
                h = h / h.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_h = h.copy()
                improved_this_pass = True

        if not improved_this_pass:
            no_full_pass_improve += 1
        else:
            no_full_pass_improve = 0

    return best_h, best_obj


def evaluate_symmetric(h):
    """Evaluate the objective for a symmetric half-sequence."""
    f = sym_full(h)
    s = f.sum()
    if s < 1e-10:
        return float('inf')
    n = len(f)
    c = conv_full_fft(h)
    max_b = c.max()
    return 2.0 * n * max_b / (s ** 2)


def interpolate_half(h, new_hn):
    """Interpolate a half-sequence to a new length."""
    old_n = len(h)
    if old_n == new_hn:
        return h.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_hn)
    h_new = np.interp(new_x, old_x, h)
    h_new = np.maximum(h_new, 0)
    s = h_new.sum()
    if s > 1e-15:
        h_new = h_new / s
    return h_new
