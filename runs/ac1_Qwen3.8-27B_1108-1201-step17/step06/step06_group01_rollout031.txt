def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

def fast_autocorr(a):

    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def peak_shave(a, max_rounds=50000, time_limit=60.0):

def momentum_subgrad(a0, time_limit, top_k=30, lr_init=0.005, verbose=False):

def interpolate_sequence(f, new_n):

    s = f.sum()
    if s < 1e-15:

    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    # Adding mass at position m affects c[k] for k in [m, m+n-1] (from the a[m]*a[...] terms)
    # and c[k] for k in [n-1-m, n-1+n-1-m] (from the a[...]*a[m] terms)

import numpy as np

def fast_autocorr(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute objective 2*n*max_b/(sum_a)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def momentum_subgradient(a0, time_limit, top_k=20, lr_init=0.008, verbose=True):
    """Momentum-based subgradient with simplex projection."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    lr = lr_init
    mom = 0.9
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    fft_size = 1 << (2 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1
        B = rfft(a, fft_size)
        b = irfft(B * B, fft_size)[:2 * n - 1]

        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()

        # Compute subgradient efficiently
        sg = np.zeros(n)
        for idx_i in range(len(peaks)):
            if w[idx_i] < 1e-12:
                continue
            k = int(peaks[idx_i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # sg[j] += 2 * w * a[k-j] for j in [js, je]
            sg[js:je+1] += 2.0 * w[idx_i] * a[k - js:k - js + (je - js + 1)][::-1]

        v = mom * v - lr * sg
        a_new = a + v
        a_new = project_simplex(a_new)

        B2 = rfft(a_new, fft_size)
        b2 = irfft(B2 * B2, fft_size)[:2 * n - 1]
        score = 2.0 * n * b2.max()

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.004)
            no_improve = 0

        if verbose and it % 2000 == 0:
            print(f"    MomSG it={it} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score

def stochastic_perturbation_search(a0, time_limit, n_restarts=15, verbose=True):
    """
    For each restart, perturb a random subset of coordinates, then do
    hill-climbing with single-coordinate moves.
    """
    n = len(a0)
    best_a = a0.copy().astype(np.float64)
    best_score = compute_score(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    fft_size = 1 << (2 * n - 2).bit_length()

    for restart in range(n_restarts):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        a = best_a.copy()
        # Perturb random subset
        n_pert = max(3, n // 20)
        positions = rng.choice(n, size=n_pert, replace=False)
        a[positions] *= (1.0 + rng.standard_normal(n_pert) * 0.1)
        a = np.maximum(a, 0)
        a = a / a.sum()

        current_score = compute_score(a)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        # Hill climbing: random coordinate moves
        local_rem = rem * 0.8 / max(1, n_restarts - restart)
        local_t0 = time.monotonic()
        stagnant = 0

        while time.monotonic() - local_t0 < local_rem and stagnant < 100:
            i = rng.integers(0, n)
            m = rng.integers(0, n)
            if i == m:
                continue
            frac = rng.uniform(0.01, 0.15)
            delta = a[i] * frac
            if delta < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= delta
            a_trial[m] += delta
            if a_trial[i] < 0:
                continue
            a_trial = a_trial / a_trial.sum()
            B = rfft(a_trial, fft_size)
            new_peak = irfft(B * B, fft_size)[:2 * n - 1].max()
            new_score = 2.0 * n * new_peak
            if new_score < current_score - 1e-14:
                a = a_trial
                current_score = new_score
                stagnant = 0
                if new_score < best_score - 1e-13:
                    best_score = new_score
                    best_a = a.copy()
            else:
                stagnant += 1

        if verbose and (restart + 1) % 5 == 0:
            print(f"    PertSearch restart={restart+1}/{n_restarts}: best={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length preserving non-negativity and normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        print(f"No parent, using uniform n=768, score={best_score:.6f}")

    # === Phase 1: Aggressive peak shaving (most important) ===
    print(f"\n=== Phase 1: Peak Shaving ===")
    remaining = total_budget - (time.monotonic() - t_start)
    t_alloc = min(remaining * 0.35, 500.0)
    a1, s1 = peak_shave_aggressive(best_seq, t_alloc, verbose=True)
    best_seq, best_score = check_and_update(a1, "phase1_peaks", best_seq, best_score)
    print(f"  Phase 1 done: {best_score:.6f}")

    # === Phase 2: Momentum subgradient ===
    print(f"\n=== Phase 2: Momentum Subgradient ===")
    remaining = total_budget - (time.monotonic() - t_start)
    t_alloc = min(remaining * 0.2, 300.0)
    a2, s2 = momentum_subgradient(best_seq, t_alloc, verbose=True)
    best_seq, best_score = check_and_update(a2, "phase2_momsg", best_seq, best_score)
    print(f"  Phase 2 done: {best_score:.6f}")

    # === Phase 3: Peak shaving again on improved result ===
    print(f"\n=== Phase 3: Peak Shaving Round 2 ===")
    remaining = total_budget - (time.monotonic() - t_start)
    t_alloc = min(remaining * 0.2, 300.0)
    a3, s3 = peak_shave_aggressive(best_seq, t_alloc, verbose=True)
    best_seq, best_score = check_and_update(a3, "phase3_peaks", best_seq, best_score)
    print(f"  Phase 3 done: {best_score:.6f}")

    # === Phase 4: Stochastic perturbation search ===
    print(f"\n=== Phase 4: Stochastic Perturbation Search ===")
    remaining = total_budget - (time.monotonic() - t_start)
    t_alloc = min(remaining * 0.15, 200.0)
    a4, s4 = stochastic_perturbation_search(best_seq, t_alloc, n_restarts=12)
    best_seq, best_score = check_and_update(a4, "phase4_perturb", best_seq, best_score)
    print(f"  Phase 4 done: {best_score:.6f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration ===")
    remaining = total_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    alt_lengths = [n_best + d for d in [-128, -64, -32, 32, 64, 128] if 128 <= n_best + d <= 4096]

    per_length = min(remaining * 0.6 / max(len(alt_lengths), 1), 80.0)
    for alt_n in alt_lengths[:4]:
        rem = total_budget - (time.monotonic() - t_start) - 60
        if rem < 30:
            break
        t_alloc = min(per_length, rem * 0.5)
        print(f"  Trying n={alt_n} (budget={t_alloc:.0f}s)")
        init = interpolate_sequence(best_seq, alt_n)
        a_alt, _ = peak_shave_aggressive(init, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)
        # Quick momentum on the alt result
        rem = total_budget - (time.monotonic() - t_start) - 60
        if rem > 20 and len(best_seq) == alt_n:
            a_alt2, _ = momentum_subgradient(best_seq, min(rem * 0.3, 40.0), verbose=False)
            best_seq, best_score = check_and_update(a_alt2, f"alt_mom_n{alt_n}", best_seq, best_score)
        print(f"  After n={alt_n}: best={best_score:.6f}")

    # === Phase 6: Final combined polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = total_budget - (time.monotonic() - t_start)
    if remaining > 20:
        t_ps = min(remaining * 0.5, 100.0)
        a_f1, _ = peak_shave_aggressive(best_seq, t_ps, verbose=False)
        best_seq, best_score = check_and_update(a_f1, "final_peaks", best_seq, best_score)

    remaining = total_budget - (time.monotonic() - t_start)
    if remaining > 15:
        t_ms = min(remaining * 0.4, 50.0)
        a_f2, _ = momentum_subgradient(best_seq, t_ms, verbose=False)
        best_seq, best_score = check_and_update(a_f2, "final_mom", best_seq, best_score)

    remaining = total_budget - (time.monotonic() - t_start)
    if remaining > 10:
        a_f3, _ = peak_shave_aggressive(best_seq, min(remaining * 0.6, 30.0), verbose=False)
        best_seq, best_score = check_and_update(a_f3, "final_final", best_seq, best_score)

    # Final verification with the actual evaluator
    if best_seq is None:
        best_seq = np.ones(512) / 512
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
