import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

def lp_fw_single(f, M, line_search_points=20):
    """One LP-FW iteration with detailed line search."""
    n = len(f)
    g = lp_direction(f, M, n)
    if g is None:
        return f, M, 0.0
    g_sum = g.sum()
    if g_sum < 1e-15:
        return f, M, 0.0
    g = g / g_sum
    
    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in np.linspace(0.01, 1.0, line_search_points):
        ft = (1 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0)
        s = ft.sum()
        if s < 1e-15:
            continue
        ft = ft / s
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha < 1e-10:
        return f, M, 0.0
    
    # Golden section refinement
    lo = max(0, best_alpha - 0.05)
    hi = min(1.0, best_alpha + 0.05)
    for _ in range(8):
        m1 = lo + (hi - lo) * 0.382
        m2 = lo + (hi - lo) * 0.618
        ft1 = np.maximum((1-m1)*f + m1*g, 0); ft1 /= ft1.sum()
        ft2 = np.maximum((1-m2)*f + m2*g, 0); ft2 /= ft2.sum()
        v1 = fast_convolve(ft1).max()
        v2 = fast_convolve(ft2).max()
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    
    best_alpha = (lo + hi) / 2
    f_new = np.maximum((1-best_alpha)*f + best_alpha*g, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f, M, 0.0
    f_new = f_new / s
    M_new = fast_convolve(f_new).max()
    return f_new, M_new, best_alpha

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe with
min-max LP direction finding, golden section line search, multi-strategy
perturbation, and coordinate descent polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lp_minmax_direction(f, n):
    """
    Solve: min t s.t. (f*g)[k] <= t for all k, sum(g) = 1, g >= 0.
    Returns (g, t) or (None, None) if infeasible.
    """
    # Variables: g[0], ..., g[n-1], t (total n+1 variables)
    # Objective: minimize t -> c = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # coefficient for t
    
    # Constraints: (f*g)[k] - t <= 0 for k = 0, ..., 2n-2
    # (f*g)[k] = sum_{j: valid} g[j] * f[k-j]
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    for j in range(n):
        # g[j] appears in (f*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j]
        A_ub[j:j+n, j] = f
    
    # -t coefficient in each row
    A_ub[:, n] = -1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        g = res.x[:n]
        t_val = res.x[n]
        return g, t_val
    return None, None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-Frank-Wolfe with min-max direction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g, t_star = lp_minmax_direction(f, n)
        
        if g is None:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            continue
        g = g / g_sum
        
        # Check if we're at a fixed point
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        no_improve = 0
        
        # Line search: minimize max(((1-a)*f + a*g) * ((1-a)*f + a*g))
        # over alpha in [0, 1]
        best_alpha = 0.0
        best_val = M
        
        # Coarse search
        for alpha in np.linspace(0.02, 0.98, 20):
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            s_ft = ft.sum()
            if s_ft < 1e-15:
                continue
            ft = ft / s_ft
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha < 1e-10:
            no_improve += 5
            continue
        
        # Fine search (golden section)
        lo = max(0.0, best_alpha - 0.04)
        hi = min(1.0, best_alpha + 0.04)
        for _ in range(10):
            m1 = lo + 0.382 * (hi - lo)
            m2 = lo + 0.618 * (hi - lo)
            ft1 = np.maximum((1-m1)*f + m1*g, 0)
            ft1 /= ft1.sum()
            ft2 = np.maximum((1-m2)*f + m2*g, 0)
            ft2 /= ft2.sum()
            v1 = fast_convolve(ft1).max()
            v2 = fast_convolve(ft2).max()
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        
        best_alpha = (lo + hi) / 2
        f_new = np.maximum((1-best_alpha)*f + best_alpha*g, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new
        M_new = fast_convolve(f_new).max()
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"      LP-FW iter {it}: obj={best_obj:.10f}, t={elapsed:.1f}s")
        
        f = f_new
    
    return best_f, best_obj


def smoothed_mirror_descent(f0, time_limit, verbose=False,
                            tau_init=5.0, tau_decay=1.003,
                            lr_init=0.01, lr_decay=0.9998, restart_every=200):
    """Adaptive smoothed mirror descent."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0
    best_tau = tau
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        len_c = len(conv)
        k_top = max(10, int(len_c * 0.2))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] + f[idx]
            sg[js:je + 1] += w[i] * contrib
        
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.05:
            for damp in [0.5, 0.25, 0.1]:
                f_trial = f * (1-damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    break
            else:
                lr *= 0.5
                continue
        
        f = f_new
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            best_tau = tau
        else:
            no_improve += 1
        
        tau *= tau_decay
        lr *= lr_decay
        
        if no_improve > restart_every:
            rng = np.random.default_rng(iter_count * 7 + 13)
            noise = 0.003 + 0.002 * (no_improve / restart_every)
            f = np.maximum(best_f + rng.standard_normal(n) * noise, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.3
            tau = best_tau
            no_improve = 0
        
        if lr < 1e-7:
            break
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with golden section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000 + 1)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            
            old_val = f[j]
            if old_val < 1e-12:
                # Try setting it to a small value
                candidates = [1e-4, 1e-3, 0.01]
            else:
                candidates = [old_val * d for d in [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.1]]
                candidates.append(0.0)
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for new_val in candidates:
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_asymmetric_init(n, variant=0, seed=42):

    if variant == 0:

    elif variant == 1:
        a = np.exp(-np.arange(n)[::-1] / (n * 0.15))

        a = np.maximum(a, 0)

    elif variant == 5:
        a = np.cos(np.pi * np.arange(n) / (2 * n)) ** 2 + 0.1
    elif variant == 6:
        a = np.ones(n)
        a[:n//3] *= 1.5
        a[2*n//3:] *= 0.5
        a += rng.standard_normal(n) * 0.1
        a = np.maximum(a, 0)

    elif variant == 8:

    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using min-max LP-based Frank-Wolfe,
smoothed mirror descent, coordinate descent, and multi-length exploration.
Key innovation: uses a cleaner LP formulation (min t s.t. (f*g)[k]<=t, sum(g)=1, g>=0)
to find the optimal FW direction at each step, combined with golden-section line search.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lp_minmax_direction(f, n):
    """
    Solve: min t s.t. (f*g)[k] <= t for all k, sum(g) = 1, g >= 0.
    Returns (g, t) or (None, None) if unsuccessful.
    """
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    for j in range(n):
        A_ub[j:j + n, j] = f

    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        g = res.x[:n]
        t_val = res.x[n]
        return g, t_val
    return None, None


def lp_fw_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

    n = len(f)

            no_improve += 1
            if no_improve > 200:
                break
            continue

        if g_sum < 1e-15:
            no_improve += 1
            continue
        g = g / g_sum

        gap = np.max(np.abs(g - f))

            continue

        no_improve = 0

        # Coarse line search
        best_alpha = 0.0

def smoothed_mirror_descent(f0, time_limit, verbose=False,
                            tau_init=5.0, tau_decay=1.003,
                            lr_init=0.01, lr_decay=0.9998, restart_every=200):
    """Adaptive smoothed mirror descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0
    best_tau = tau

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()

        len_c = len(conv)
        k_top = max(10, int(len_c * 0.2))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] + f[idx]
            sg[js:je + 1] += w[i] * contrib

        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.05:
            took_step = False
            for damp in [0.5, 0.25, 0.1, 0.05]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    took_step = True
                    break
            if not took_step:
                lr *= 0.5
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            best_tau = tau
        else:
            no_improve += 1

        tau *= tau_decay
        lr *= lr_decay

        if no_improve > restart_every:
            rng_local = np.random.default_rng(iter_count * 7 + 13)
            noise_scale = 0.003 + 0.002 * min(no_improve / restart_every, 3.0)
            f = np.maximum(best_f + rng_local.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.3
            tau = best_tau
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with candidate-based search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        rng_local = np.random.default_rng(pass_count * 97 + 3)
        order = rng_local.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Generate candidate values
            candidates = [0.0]
            if old_val > 1e-12:
                candidates.extend([old_val * d for d in [0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 0.1]])
            else:
                candidates.extend([1e-4, 1e-3, 0.01])

            for new_val in candidates:
                if new_val < 0 or new_val > 0.4:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = np.maximum(f, 0)
                s_f = f.sum()
                if s_f < 1e-15:
                    break
                f = f / s_f
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem * 0.12, 70)
            if t_alloc < 15:
                t_alloc = 15

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = np.maximum(best_seq + noise, 1e-12)
                f_init = f_init / f_init.sum()
                label = "perturb_small"
            elif r == 1:
                asym = make_asymmetric_init(n_best, variant=rng.integers(0, 11), seed=seed + r * 7)
                f_init = 0.85 * best_seq + 0.15 * asym
                f_init = f_init / f_init.sum()
                label = "blend_asym"
            elif r == 2:
                # Try a fresh asymmetric init at the same length
                f_init = make_asymmetric_init(n_best, variant=rng.integers(0, 11),
                                              seed=seed + 200 + r)
                label = "fresh_asym"
            else:
                # Moderate perturbation
                noise = rng.standard_normal(n_best) * 0.01
                f_init = np.maximum(best_seq + noise, 1e-12)
                f_init = f_init / f_init.sum()
                label = "perturb_med"

            print(f"  Restart {r} ({label}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc * 0.6)
            _try(a_opt, f"restart_{r}")

            rem2 = deadline - time.monotonic()
            if rem2 > 20:
                a_opt2, obj2 = smoothed_mirror_descent(a_opt, time_limit=min(rem2 * 0.5, 30),
                                                       tau_init=15.0, lr_init=0.004)
                _try(a_opt2, f"restart_smd_{r}")

    # === Phase 6: Coordinate descent ===
    print("\n=== Phase 6: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.2, 100)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            _try(a_opt, "cd")

    # === Phase 7: Final LP-FW squeeze ===
    print("\n=== Phase 7: Final LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.3, 80)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            _try(a_opt, "final_lpfw")

    # === Phase 8: Final SMD with low temperature ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t_alloc = min(rem, 40)
            print(f"  Final SMD (low tau): t={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                                  tau_init=30.0, lr_init=0.002,
                                                  tau_decay=1.001, restart_every=60)
            _try(a_opt, "final_smd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={elapsed:.1f}s)")
    return [float(x) for x in best_seq]
