```python
def lp_direction(f, M):
    """Find g that maximizes sum(g) subject to (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    # Build constraint matrix A where (A @ g)_k = (f*g)_k
    # (f*g)_k = sum_{j} f_{k-j} * g_j for valid j
    # So A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, use the banded structure
    # A is (2n-1) x n, banded
    
    import cvxpy as cp
    g = cp.Variable(n)
    constraints = [g >= 0]
    
    # For each k, (f*g)_k = sum_j A[k,j] * g_j <= M
    # Build the constraints
    for k in range(2*n - 1):
        # (f*g)_k = sum_j f_{k-j} * g_j where 0 <= k-j < n and 0 <= j < n
        # So j ranges from max(0, k-n+1) to min(k, n-1)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            # sum_{j=js}^{je} f[k-j] * g_j <= M
            coeffs = f[k - np.arange(js, je+1)]
            constraints.append(coeffs @ g[js:je+1] <= M)
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLPK)  # or CBC, or ECOS
    
    if g.value is None:
        return None
    return np.maximum(g.value, 0)
```

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0)

    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, float(v[0]))])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_full(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            A[k, js:je + 1] = f[k - np.arange(js, je + 1)]
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return f.copy()


def solve_lp_reduced(f, M, K=200, max_outer=5):
    """Solve LP with reduced constraint set, adding violated constraints."""
    n = len(f)
    conv = fast_convolve(f)
    peak_indices = np.argpartition(conv, -K)[-K:]
    peak_indices = np.sort(peak_indices)

    for outer in range(max_outer):
        Kc = len(peak_indices)
        A = np.zeros((Kc, n))
        for i, k in enumerate(peak_indices):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                A[i, js:je + 1] = f[k - np.arange(js, je + 1)]
        b = np.full(Kc, M)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 20})
            if not res.success:
                break
        except Exception:
            break
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-12:
            break
        fg = cross_convolve(f, g)
        violations = np.where(fg > M * (1.0 + 1e-8))[0]
        if len(violations) == 0:
            return g
        new_indices = np.unique(np.concatenate([peak_indices, violations]))
        if len(new_indices) > K * 2:
            top = new_indices[np.argsort(fg[new_indices])]
            peak_indices = np.sort(top[-K:])
        else:
            peak_indices = np.sort(new_indices)
    return solve_lp_full(f, M)


def lp_based_optimize(f0, time_limit, use_reduced=True, verbose=False, label=""):
    """LP-based optimization: iteratively find best blend direction via LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr_blend = 0.3

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = fast_convolve(f).max()

        if use_reduced:
            g_raw = solve_lp_reduced(f, M, K=150)
        else:
            g_raw = solve_lp_full(f, M)

        s = g_raw.sum()
        if s < 1e-10:
            break
        g = g_raw / s

        if np.allclose(g, f, atol=1e-10):
            no_improve += 5
            if no_improve > 30:
                break
            continue

        best_t = 0.0
        best_trial = M
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            ss = f_trial.sum()
            if ss < 1e-12:
                continue
            f_trial = f_trial / ss
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-14:
                best_trial = obj
                best_t = t

        if best_t < 1e-6:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        new_obj = fast_convolve(f).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 20 == 0:
                print(f"    {label} LP iter {it}: obj={best_obj:.10f}, t={best_t:.3f}")
        else:
            no_improve += 1

    return best_f, best_obj


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def mirror_descent(a0, time_limit, top_k=30, temp_init=0.0003, lr_init=0.008,
                   verbose=False, label=""):
    """Multiplicative weight mirror descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = compute_subgradient(a, peak_indices, w)

        best_lr = lr * 0.1
        best_trial_obj = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-12:
                continue
            log_update = np.clip(-trial_lr * g, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_conv = fast_convolve(a_trial)
            trial_obj = trial_conv.max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        log_update = np.clip(-best_lr * g, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        new_obj = fast_convolve(a).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} MD iter {it}: obj={best_obj:.10f}, lr={best_lr:.6f}")
        else:
            no_improve += 1

        temp = min(temp * 1.002, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * 0.9999, 1e-10)

        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.001 + 0.002 * min(no_improve / 5000, 1.0)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0

        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        if lr < 1e-10:
            break

    return best_a, best_obj


def rectilinear_flow(a0, time_limit, top_k=30, temp_init=0.001, dt_init=0.004,
                     verbose=False, label=""):
    """Rectilinear (additive) flow on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    dt = dt_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = compute_subgradient(a, peak_indices, w)
        g_centered = g - g.mean()

        best_dt = dt * 0.1
        best_trial_obj = M * 1.01
        for trial_dt in [dt, dt * 0.5, dt * 0.25, dt * 0.1]:
            if trial_dt < 1e-12:
                continue
            a_trial = a - trial_dt * g_centered
            a_trial = project_simplex(a_trial)
            trial_conv = fast_convolve(a_trial)
            trial_obj = trial_conv.max()
            if trial_obj <= best_trial_obj:
                best_dt = trial_dt
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        a = a - best_dt * g_centered
        a = project_simplex(a)

        new_obj = fast_convolve(a).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} RF iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        temp = min(temp * 1.003, 0.05)
        dt = max(dt * 0.9999, 1e-10)

        if no_improve > 200:
            rng = np.random.default_rng(it * 104729 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            dt = dt_init * 0.3
            no_improve = 0

        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = project_simplex(a_sym)
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        if dt < 1e-11:
            break

    return best_a, best_obj


def fine_coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent: adjust one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-13:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 0.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped initial sequence, normalized to sum=1."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n, c1=0.3, c2=0.7, sigma=0.08):
    """Bimodal Gaussian."""
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = a / a.sum()
    return a


def make_fourier_flat(n, bandwidth=0.7):
    """Fourier-based: flat magnitude in passband, smooth rolloff."""
    # Design frequency response with flat magnitude
    k = np.arange(n // 2 + 1)
    # Raised cosine window in frequency
    edge = int(n * bandwidth / 2)
    A = np.zeros(n // 2 + 1)
    A[0] = 1.0
    for i in range(1, n // 2 + 1):
        if i <= edge:
            A[i] = 1.0
        else:
            t = (i - edge) / max(1, (n // 2 - edge))
            A[i] = 0.5 * (1 + np.cos(np.pi * t))
    # Convert to full spectrum (real, even)
    full = np.zeros(n)
    full[:n // 2 + 1] = A
    if n % 2 == 0:
        full[n // 2:] = A[n // 2 - 1:0:-1]
    else:
        full[(n // 2 + 1):] = A[n // 2 - 1:0:-1]
    a = irfft(full, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def symmetrize(a):
    """Make sequence palindromic."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: LP-based direction finding + mirror descent
    + rectilinear flow + coordinate descent, with diverse initializations
    across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Generate and evaluate diverse initializations ===
    print("=== Phase 1: Initializations ===")
    n_candidates = [768, 1024]
    if parent is not None:
        n_candidates = [len(parent), 1024, 768]

    init_pool = []
    for n_try in n_candidates:
        for sigma in [0.25, 0.35, 0.45]:
            a = make_gaussian(n_try, sigma)
            init_pool.append((a, f"gauss_s{sigma}_n{n_try}"))
        a = make_bell(n_try, 0.5)
        init_pool.append((a, f"bell_n{n_try}"))
        a = make_bell(n_try, 0.7)
        init_pool.append((a, f"bell_w_n{n_try}"))
        a = make_bimodal(n_try)
        init_pool.append((a, f"bimodal_n{n_try}"))
        a = make_fourier_flat(n_try, 0.6)
        init_pool.append((a, f"fourier_n{n_try}"))
        a = make_fourier_flat(n_try, 0.8)
        init_pool.append((a, f"fourier_w_n{n_try}"))

    if parent is not None:
        init_pool.append((parent, "parent"))
        init_pool.append((symmetrize(parent), "parent_sym"))

    # Evaluate all initializations, keep top few
    init_scores = []
    for a_init, label in init_pool:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        score = try_seq(a_norm, f"init_{label}")
        if np.isfinite(score):
            init_scores.append((score, a_norm, label))

    init_scores.sort(key=lambda x: x[0])
    print(f"  Top init: {init_scores[0][2]} with score {init_scores[0][0]:.6f}")

    # Pick top 4 distinct initializations for optimization
    top_inits = []
    seen_n = set()
    for score, a, label in init_scores:
        n = len(a)
        if n not in seen_n:
            top_inits.append((a, label))
            seen_n.add(n)
        if len(top_inits) >= 4:
            break
    if len(top_inits) < 3 and init_scores:
        for score, a, label in init_scores:
            if (a, label) not in top_inits:
                top_inits.append((a, label))
            if len(top_inits) >= 3:
                break

    # === Phase 2: LP-based optimization (main new ingredient) ===
    print("\n=== Phase 2: LP-based Optimization ===")
    remaining = deadline - time.monotonic()
    t_lp_total = remaining * 0.35
    n_starts = min(len(top_inits), 3)
    t_per_lp = t_lp_total / max(1, n_starts)

    for idx, (a_init, label) in enumerate(top_inits[:n_starts]):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 30)
        if t_alloc < 30:
            break
        print(f"  LP start {idx+1}/{n_starts} [{label}], time={t_alloc:.0f}s")
        a_opt, obj = lp_based_optimize(a_init, time_limit=t_alloc,
                                       use_reduced=True, verbose=True,
                                       label=f"LP_{label}")
        try_seq(a_opt, f"LP_{label}")
        # Also try symmetrized
        a_sym = symmetrize(a_opt)
        try_seq(a_sym, f"LP_sym_{label}")

    # === Phase 3: Mirror Descent polishing on best ===
    print("\n=== Phase 3: Mirror Descent Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_md = remaining * 0.25
        print(f"  MD time: {t_md:.0f}s")
        a_opt, obj = mirror_descent(best_seq, time_limit=t_md, top_k=30,
                                    temp_init=0.0002, lr_init=0.006,
                                    verbose=True, label="MD_main")
        try_seq(a_opt, "MD_main")
        a_sym = symmetrize(a_opt)
        try_seq(a_sym, "MD_sym")

    # === Phase 4: Rectilinear Flow ===
    print("\n=== Phase 4: Rectilinear Flow ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_rf = remaining * 0.15
        print(f"  RF time: {t_rf:.0f}s")
        a_opt, obj = rectilinear_flow(best_seq, time_limit=t_rf, top_k=30,
                                      temp_init=0.0008, dt_init=0.003,
                                      verbose=True, label="RF")
        try_seq(a_opt, "RF")

    # === Phase 5: LP-based optimization round 2 (from best) ===
    print("\n=== Phase 5: LP Round 2 ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_lp2 = remaining * 0.15
        if t_lp2 > 30:
            print(f"  LP2 time: {t_lp2:.0f}s")
            a_opt, obj = lp_based_optimize(best_seq, time_limit=t_lp2,
                                           use_reduced=True, verbose=True,
                                           label="LP2")
            try_seq(a_opt, "LP2")
            a_sym = symmetrize(a_opt)
            try_seq(a_sym, "LP2_sym")

    # === Phase 6: Fine Coordinate Descent ===
    print("\n=== Phase 6: Coordinate Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = min(remaining * 0.10, 120)
        if t_cd > 20:
            print(f"  CD time: {t_cd:.0f}s")
            a_opt, obj = fine_coordinate_descent(best_seq, time_limit=t_cd,
                                                 verbose=True, label="CD")
            try_seq(a_opt, "CD")

    # === Phase 7: Multi-restart with perturbations ===
    print("\n=== Phase 7: Multi-restart ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_restarts = 3
        t_per = min(remaining * 0.12 / n_restarts, 60)
        if t_per > 20:
            for r in range(n_restarts):
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                if r == 0:
                    f_init = best_seq.copy()
                elif r == 1:
                    noise = rng.standard_normal(len(best_seq)) * 0.003
                    f_init = np.maximum(best_seq + noise, 1e-15)
                    f_init = f_init / f_init.sum()
                else:
                    f_init = 0.85 * best_seq + 0.15 * (np.ones(len(best_seq)) / len(best_seq))
                    f_init = f_init / f_init.sum()

                t_alloc = min(t_per, remaining - 30)
                if t_alloc < 15:
                    break
                # Alternate between MD and LP
                if r % 2 == 0:
                    a_opt, obj = mirror_descent(f_init, time_limit=t_alloc, top_k=20,
                                                temp_init=0.0001, lr_init=0.004,
                                                label=f"rest_{r}")
                else:
                    a_opt, obj = lp_based_optimize(f_init, time_limit=t_alloc,
                                                   use_reduced=True, label=f"rest_{r}")
                try_seq(a_opt, f"restart_{r}")

    # === Phase 8: Final fine polish ===
    print("\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_polish = min(remaining - 10, 90)
            a_opt, obj = mirror_descent(best_seq, time_limit=t_polish, top_k=10,
                                        temp_init=0.00005, lr_init=0.002,
                                        verbose=True, label="polish")
            try_seq(a_opt, "polish")
            a_sym = symmetrize(best_seq)
            try_seq(a_sym, "final_sym")
            # Try a final LP step
            if time.monotonic() < deadline - 15:
                a_opt, obj = lp_based_optimize(best_seq, time_limit=min(30, deadline - time.monotonic() - 10),
                                               use_reduced=True, label="LP_final")
                try_seq(a_opt, "LP_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
