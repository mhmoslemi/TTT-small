import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

    max_c = c.max()
    if max_c < 1e-15:
        return np.zeros(n), c
    c_norm = c / max_c
    w = c_norm ** (p - 1)
    # Gradient: grad_j = 2 * sum_k w_k * f_{k-j} * max_c^(p-1) / p * (1/max_c)^(1/p) * ...
    # Simpler: grad of sum(c^p) w.r.t. f_j = 2p * sum_k c_k^(p-1) * f_{k-j}
    # We'll use the gradient of (1/p)*sum(c^p) for cleaner scaling
    # grad_j = 2 * sum_k c_k^(p-1) * f_{k-j}
    w_full = w * (max_c ** (p - 1))
    w_rev = w_full[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g, c


def lse_gradient(f, beta):
    """Gradient of log-sum-exp smoothed max: (1/beta)*log(sum(exp(beta*c_k)))."""
    n = len(f)
    c = fast_convolve(f)
    scaled = beta * c
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g, c


def pnorm_gd(f0, time_limit, p_schedule=None, verbose=False):
    """
    P-norm gradient descent with Nesterov momentum, backtracking,
    and structured restarts.
    """
    if p_schedule is None:
        p_schedule = [4, 8, 16, 32, 64, 128, 256]

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for p in p_schedule:
        stage_start = time.monotonic()
        stage_time = max(5.0, min(time_limit / len(p_schedule),
                                   (deadline_remaining(time_limit, t0) * 0.9) / max(1, len(p_schedule) - p_schedule.index(p))))
        # Simplify: allocate proportional time
        stage_time = min(time_limit / len(p_schedule) * 1.5,
                         (time_limit - (time.monotonic() - t0)) * 0.85 / max(1, len(p_schedule.index(p)) + 1))
        stage_time = max(5.0, stage_time)

        momentum = np.zeros(n)
        mu = 0.85
        lr = 1.0 / (n ** 0.5 * p ** 0.3)
        no_improve = 0
        it = 0
        temp = 0.005  # annealing temperature

        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.88:
            it += 1
            g, c = pnorm_gradient(f, p)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break

            # Nesterov-like: look-ahead
            f_look = f + mu * momentum
            g_look, _ = pnorm_gradient(f_look, p) if it % 5 == 0 else g

            # Update
            momentum = mu * momentum - lr * (g_look / g_norm)
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)

            # Evaluate
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()

            if obj_trial <= best_obj + 1e-10:
                f = f_trial
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    lr = min(lr * 1.03, 0.5 / p)
                    temp *= 0.999
                else:
                    no_improve += 1
                    lr *= 0.98
            else:
                # Backtrack
                lr *= 0.5
                momentum = mu * momentum - lr * (g / g_norm)
                f_trial = f + momentum
                f_trial = project_simplex(f_trial)
                c_trial = fast_convolve(f_trial)
                obj_trial = c_trial.max()
                if obj_trial <= c.max() + 1e-10:
                    f = f_trial
                    if obj_trial < best_obj - 1e-13:
                        best_obj = obj_trial
                        best_f = f.copy()
                        no_improve = 0
                else:
                    no_improve += 1
                    lr *= 0.7
                momentum = mu * momentum * 0.8

            # Structured restart when stuck
            if no_improve > 60:
                # Peak-targeted perturbation
                c_peak = fast_convolve(best_f)
                peak_k = np.argmax(c_peak)
                # Find elements contributing to the peak
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                peak_idx = np.arange(js, je + 1)
                contrib = best_f[peak_idx] * best_f[peak_k - peak_idx]
                contrib_norm = contrib / (contrib.sum() + 1e-15)

                # Perturb peak-contributing elements more
                noise = rng.standard_normal(n) * temp
                noise[peak_idx] *= 2.0
                f = project_simplex(best_f + noise)
                momentum = np.zeros(n)
                lr = 1.0 / (n ** 0.5 * p ** 0.3)
                no_improve = 0
                temp = max(temp * 0.9, 0.001)

            if lr < 1e-10:
                break

        if verbose:
            print(f"    P-norm p={p}: best={best_obj:.12f}, time={time.monotonic()-t0:.1f}s")

    return best_f, best_obj


def lse_gd(f0, time_limit, beta_schedule=None, verbose=False):
    """Log-sum-exp gradient descent with momentum."""
    if beta_schedule is None:
        beta_schedule = [4, 8, 16, 32, 64, 128, 256, 512]

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()

    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    for beta in beta_schedule:
        stage_time = max(5.0, min(60.0, (time_limit - (time.monotonic() - t0)) * 0.8 / max(1, len(beta_schedule) - beta_schedule.index(beta))))
        stage_start = time.monotonic()

        momentum = np.zeros(n)
        mu = 0.8
        lr = 1.0 / (beta * n ** 0.5)
        no_improve = 0
        it = 0
        temp = 0.003

        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.88:
            it += 1
            g, c = lse_gradient(f, beta)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break

            momentum = mu * momentum - lr * (g / g_norm)
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()

            if obj_trial <= c.max() + 1e-10:
                f = f_trial
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    lr = min(lr * 1.02, 1.0 / (beta * 0.5))
                    temp *= 0.998
                else:
                    no_improve += 1
                    lr *= 0.98
            else:
                lr *= 0.5
                momentum = mu * momentum - lr * (g / g_norm)
                f_trial = project_simplex(f + momentum)
                c_trial = fast_convolve(f_trial)
                obj_trial = c_trial.max()
                if obj_trial <= c.max() + 1e-10:
                    f = f_trial
                    if obj_trial < best_obj - 1e-13:
                        best_obj = obj_trial
                        best_f = f.copy()
                        no_improve = 0
                else:
                    no_improve += 1
                    lr *= 0.7

            if no_improve > 80:
                c_peak = fast_convolve(best_f)
                peak_k = np.argmax(c_peak)
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                peak_idx = np.arange(js, je + 1)
                noise = rng.standard_normal(n) * temp
                noise[peak_idx] *= 2.5
                f = project_simplex(best_f + noise)
                momentum = np.zeros(n)
                lr = 1.0 / (beta * n ** 0.5)
                no_improve = 0
                temp = max(temp * 0.85, 0.0005)

            if lr < 1e-12:
                break

        if verbose:
            print(f"    LSE beta={beta}: best={best_obj:.12f}, time={time.monotonic()-t0:.1f}s")

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 30})
    if res.success:
        return res.x
    return None


def lpfw_polish(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe for high-precision polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp = min(m_exp * 1.15, 8.0)
            if m_exp > 7.5:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                c = fast_convolve(f)
                M = c.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp = min(m_exp * 1.1, 8.0)
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                m_exp = min(m_exp * 1.08, 5.0)
            if no_improve > 120:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                c = fast_convolve(f)
                M = c.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.05, 1.0, 16):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            c = fast_convolve(f_new)
            M = c.max()
            f = f_new
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.12f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)

        if no_improve > 150:
            break

    return best_f, best_obj


def multiplicative_update(f0, time_limit, verbose=False):
    """Multiplicative weight update targeting peaks."""
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()

        # Find top peaks
        k_peaks = min(20, len(c))
        peak_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        peak_vals = c[peak_idx]
        excess = np.maximum(peak_vals - c.mean(), 0)
        if excess.sum() < 1e-15:
            excess = np.ones(len(peak_idx))
        w_peaks = excess / excess.sum()

        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w_peaks[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w_peaks[i] * f[k - idx]

        f = f * np.exp(-lr * 2.0 * sg)
        f = np.maximum(f, 1e-12)
        f = f / f.sum()

        c_new = fast_convolve(f)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.03)
        else:
            no_improve += 1
            lr *= 0.97

        if no_improve > 150:
            noise = rng.standard_normal(n) * 0.002
            f = np.maximum(best_f + noise, 1e-10)
            f = f / f.sum()
            lr = 0.004
            no_improve = 0

        if lr < 1e-7:
            break

        if verbose and it % 100 == 0:
            print(f"    MW iter {it}: obj={obj_new:.10f}, best={best_obj:.10f}")

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_super_gaussian(n, power=6):
    x = (np.arange(n) - n / 2.0) / (n / 2.0)
    a = np.exp(-np.abs(x) ** power)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hann(n):
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def make_two_peak(n, sep=0.3):
    center = n / 2.0
    offset = n * sep / 2.0
    x = np.arange(n)
    a = np.exp(-((x - (center - offset)) / (n * 0.08)) ** 2) + \
        np.exp(-((x - (center + offset)) / (n * 0.08)) ** 2)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization with P-norm/LSE gradient descent,
    LP-FW polishing, multiplicative updates, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10 or len(a) < 2:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # === Get parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if np.isfinite(p_score) and p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.3)
        try_seq(parent, "default_init")

    n_parent = len(parent)

    # === Phase 1: Quick diverse initialization screening ===
    print("=== Phase 1: Initialization Screening ===")
    t1 = time.monotonic()

    init_pool = [
        ("parent", parent.copy()),
        ("gauss_25", make_gaussian(n_parent, 0.25)),
        ("gauss_35", make_gaussian(n_parent, 0.35)),
        ("super_g6", make_super_gaussian(n_parent, 6)),
        ("super_g10", make_super_gaussian(n_parent, 10)),
        ("hann", make_hann(n_parent)),
        ("two_peak", make_two_peak(n_parent, 0.3)),
    ]

    best_init = None
    best_init_obj = float('inf')
    for name, a in init_pool:
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss < 1e-10:
            continue
        a = a / ss
        c = fast_convolve(a)
        obj = len(a) * c.max()  # relative score (without factor 2)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a
        try_seq(a, f"init_{name}")

    # Random perturbations of parent
    for trial in range(8):
        scale = [0.005, 0.01, 0.02, 0.01, 0.015, 0.008, 0.012, 0.003][trial]
        noise = rng.standard_normal(n_parent) * scale
        a_pert = np.maximum(parent + noise, 0)
        a_pert = a_pert / a_pert.sum()
        c = fast_convolve(a_pert)
        obj = n_parent * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_pert
        try_seq(a_pert, f"init_pert{trial}")

    print(f"  Best init obj: {best_init_obj:.8f}")
    print(f"  Phase 1 time: {time.monotonic()-t1:.1f}s")

    # === Phase 2: P-norm GD on best init ===
    print(f"\n=== Phase 2: P-norm GD (n={len(best_init)}) ===")
    t2 = time.monotonic()
    rem = deadline - time.monotonic() - 600
    t_alloc = max(30, min(rem * 0.35, 400))
    print(f"  Time allocation: {t_alloc:.0f}s")

    a_opt, obj = pnorm_gd(best_init, time_limit=t_alloc, verbose=True)
    try_seq(a_opt, "pnorm_gd")

    # === Phase 3: LSE GD ===
    print(f"\n=== Phase 3: LSE GD ===")
    rem = deadline - time.monotonic() - 500
    if rem > 60:
        t_alloc = min(rem * 0.3, 250)
        a_opt2, obj2 = lse_gd(best_init, time_limit=t_alloc, verbose=True)
        try_seq(a_opt2, "lse_gd")
        # Also from parent
        if not np.allclose(best_init, parent):
            rem2 = deadline - time.monotonic() - 400
            if rem2 > 40:
                a_opt3, obj3 = lse_gd(parent, time_limit=min(rem2 * 0.3, 100))
                try_seq(a_opt3, "lse_parent")

    # === Phase 4: Multiplicative update ===
    print(f"\n=== Phase 4: Multiplicative Update ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            t_alloc = min(rem * 0.2, 150)
            a_opt, obj = multiplicative_update(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "multiplicative")

    # === Phase 5: LP-FW Polish ===
    print(f"\n=== Phase 5: LP-FW Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.4, 250)
            print(f"  Time allocation: {t_alloc:.0f}s")
            a_opt, obj = lpfw_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length Exploration ===")
    base = best_seq if best_seq is not None else best_init

    target_lengths = []
    if n_parent != 512:
        target_lengths.append(512)
    if n_parent != 768:
        target_lengths.append(768)
    if n_parent != 1024:
        target_lengths.append(1024)
    if n_parent != 1536:
        target_lengths.append(1536)
    if n_parent != 2048:
        target_lengths.append(2048)

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 150
        if rem < 60:
            break

        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")

        # Quick optimization
        t_gd = min(rem * 0.25, 80)
        if t_gd > 20:
            a_opt, obj = pnorm_gd(f_interp, time_limit=t_gd,
                                   p_schedule=[8, 16, 32, 64])
            try_seq(a_opt, f"pnorm_n{target_n}")

        rem2 = deadline - time.monotonic() - 100
        if rem2 > 30:
            t_lpfw = min(rem2 * 0.4, 80)
            a_opt, obj = lpfw_polish(a_opt, time_limit=t_lpfw)
            try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 7: Deep LP-FW with diverse restarts on best ===
    print(f"\n=== Phase 7: Deep LP-FW ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(5):
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break
            t_alloc = min(rem / (6 - p), 120)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                # Peak-targeted perturbation
                c_pk = fast_convolve(best_seq)
                peak_k = np.argmax(c_pk)
                js = max(0, peak_k - n_best + 1)
                je = min(n_best - 1, peak_k)
                f_init = best_seq.copy()
                f_init[js:je+1] *= 0.95
                f_init = project_simplex(f_init)
            elif p == 3:
                # Blend with super-Gaussian
                sg = make_super_gaussian(n_best, 8)
                f_init = 0.85 * best_seq + 0.15 * sg
                f_init = f_init / f_init.sum()
            else:
                # Scale-invariant reshape
                f_init = np.power(best_seq, 1.1)
                f_init = f_init / f_init.sum()

            a_opt, obj = lpfw_polish(f_init, time_limit=t_alloc, verbose=(p == 0))
            try_seq(a_opt, f"deep_lpfw_{p}")

    # === Phase 8: Final squeeze ===
    print(f"\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t_alloc = min(rem, 60)
            a_opt, obj = lpfw_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_squeeze")

        # Quick P-norm with very high p
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, obj = pnorm_gd(best_seq, time_limit=min(rem, 30),
                                   p_schedule=[64, 128, 256])
            try_seq(a_opt, "final_pnorm")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
