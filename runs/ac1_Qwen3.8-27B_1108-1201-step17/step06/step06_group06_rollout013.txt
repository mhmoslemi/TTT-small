def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return out


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = f[j]
        A_ub[k, n] = -1.0  # coefficient of M
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g[i] >= 0, M free (or M >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g = res.x[:n]
            M_star = res.x[n]
            # Ensure g is on simplex
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs < 1e-15:
                return None, None
            g = g / gs
            return g, M_star
    except Exception:
        pass
    return None, None


def lp_fw_direct(f0, time_limit, verbose=False):
    """
    LP-Frank-Wolfe with direct M-minimization LP.
    At each step: solve min M s.t. (f*g)[k]<=M, sum(g)=1, g>=0.
    Then take a step from f towards g.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP
        g, M_lp = lp_minimize_M(f, n)
        if g is None:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Check if g is close to f (convergence)
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        no_improve = 0
        
        # Line search: find best alpha in [0, 1]
        best_alpha = 0.0
        best_val = M
        
        # Coarse search
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val - 1e-14:
                best_val = val
                best_alpha = alpha
        
        # Fine search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
        
        if it % 100 == 0 and verbose:
            print(f"    LP-FW iter {it}: current={2*n*M:.8f}, best={2*n*best_obj:.8f}")
    
    return best_f, 2 * n * best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """
    Peak shaving: identify the peak coefficient and try to reduce it
    by adjusting the contributing elements.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        
        # Find contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        products = f[idx] * f[peak_k - idx]
        
        # Try to reduce the peak by adjusting the largest contributing pair
        j_max = idx[np.argmax(products)]
        j_other = peak_k - j_max
        
        if j_max == j_other:
            # Diagonal: reduce f[j_max]^2
            old = f[j_max]
            improved = False
            for factor in [0.99, 0.98, 0.97, 0.95, 0.93]:
                trial = f.copy()
                trial[j_max] = old * factor
                diff = old - trial[j_max]
                # Redistribute to elements that don't contribute to the peak
                mask = np.ones(n, dtype=bool)
                # Avoid elements that contribute to the top 3 peaks
                top_peaks = np.argsort(conv)[-3:]
                for pk in top_peaks:
                    js2 = max(0, int(pk) - n + 1)
                    je2 = min(n - 1, int(pk))
                    mask[js2:je2+1] = False
                if mask.sum() > 0:
                    weights = f[mask]
                    if weights.sum() > 1e-15:
                        trial[mask] += diff * weights / weights.sum()
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
            if not improved:
                no_improve += 1
            else:
                no_improve = 0
        else:
            # Off-diagonal: reduce f[j_max] * f[j_other]
            old_jm = f[j_max]
            old_jo = f[j_other]
            improved = False
            for scale in [0.99, 0.98, 0.97, 0.95]:
                trial = f.copy()
                trial[j_max] = old_jm * scale
                diff = old_jm - trial[j_max]
                # Redistribute away from peak-contributing positions
                mask = np.ones(n, dtype=bool)
                mask[j_max] = False
                mask[j_other] = False
                top_peaks = np.argsort(conv)[-3:]
                for pk in top_peaks:
                    js2 = max(0, int(pk) - n + 1)
                    je2 = min(n - 1, int(pk))
                    mask[js2:je2+1] = False
                if mask.sum() > 0:
                    weights = f[mask]
                    if weights.sum() > 1e-15:
                        trial[mask] += diff * weights / weights.sum()
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
            if not improved:
                no_improve += 1
            else:
                no_improve = 0
        
        if no_improve > 50:
            # Perturb and restart
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"    Peak-shaving iter {it}: obj={2*n*best_obj:.8f}")
    
    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
