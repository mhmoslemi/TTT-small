    Returns the optimal g_0.

    cols = []

            cols.append(j)
            vals.append(f[j])

    M: current max convolution value
    t: step size (if None, use line search)
    Returns new point.

        return f

    if g_sum < 1e-15:

        best_obj = eval_objective(f)

            obj_try = eval_objective(f_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_t = t_try
        t = best_t

        return f
    
    f_new = (1 - t) * f + t * g
    f_new = np.maximum(f_new, 0)
    f_new = f_new / f_new.sum()
    return f_new

"""
LP-based optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses a Frank-Wolfe type algorithm with LP direction finding,
combined with coordinate descent and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def eval_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))

def precompute_lp_indices(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_lp(f, M, rows, cols):
    n = len(f)
    vals = f[cols]
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    if res.status != 0:
        return None, res.status
    return res.x, 0

    M = fast_convolve(f).max()
    
    rows, cols = precompute_lp_indices(n)
    
    if verbose:

    rng = np.random.default_rng(seed)

    step_mult = 1.0

def coordinate_polish(f_init, time_limit, verbose=True):
    t0 = time.monotonic()
    a = project_simplex(f_init.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 0.9, 1.1, 1.3, 1.5, 2.0])
    n_sweeps = 0
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        n_sweeps += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old = a[j]
            if old < 1e-14:
                continue
            best_j = old
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = old * mult
                if new_val > 0.9999:
                    continue
                denom = 1.0 - old
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j = new_val
            if best_j != old:

                scale = (1.0 - best_j) / denom

                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    if verbose:
        print(f"  CD polish: {n_sweeps} sweeps, best={best_obj:.8f}")
    
    return best_a, best_obj
