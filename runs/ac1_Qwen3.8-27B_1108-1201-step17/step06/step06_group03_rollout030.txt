def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(v):
    """Numerically stable softmax."""
    v = v - v.max()
    e = np.exp(v)
    return e / e.sum()


def lse_objective_and_grad(v, tau):
    """
    Compute LSE-tau smoothed objective value and gradient with respect to v.
    f = softmax(v), objective approximates max(f*f).
    """
    f = softmax(v)
    n = len(f)
    conv = fast_convolve(f)
    
    # LSE value
    scaled = tau * conv
    scaled -= scaled.max()
    log_sum_exp = scaled.max() + np.log(np.exp(scaled).sum())
    lse_val = log_sum_exp / tau
    
    # Weights for gradient
    w = np.exp(scaled)
    w /= w.sum()
    
    # Gradient of LSE wrt f: grad_f[m] = 2 * sum_k w[k] * f[k-m] * I(0<=k-m<n)
    # This is 2 * cross_correlation(w, f)[m]
    # Using convolution: cross_corr[m] = sum_j w[m+j]*f[j]
    # = convolve(w, f[::-1], 'full')[len(w)-1-m]
    conv_result = np.convolve(w, f[::-1], mode='full')

    # grad_f[m] = 2 * conv_result[2n-2-m] for m=0,...,n-1
    grad_f = 2.0 * conv_result[2 * n - 2 - np.arange(n)]
    
    # Gradient wrt v: grad_v[j] = f[j] * (grad_f[j] - dot(grad_f, f))
    s = np.dot(grad_f, f)
    grad_v = f * (grad_f - s)
    
    return lse_val, grad_v

def reduced_lp_direction(f, M, K=80):
    """
    Solve reduced LP: max sum(g) s.t. (f*g)[k] <= M for top-K peaks, g >= 0.
    Returns the direction g (normalized to sum 1) or None.
    """
    n = len(f)
    conv = fast_convolve(f)
    
    # Get top-K peak positions
    k_top = min(K, len(conv))
    peak_idx = np.argpartition(conv, -k_top)[-k_top:]
    peak_idx = np.sort(peak_idx)
    
    # Build constraint matrix: A[i, j] = f[peak_idx[i] - j] if valid
    A = np.zeros((len(peak_idx), n))
    b = np.full(len(peak_idx), M)
    for i, k in enumerate(peak_idx):
        js = max(0, int(k) - n + 1)
        je = min(n - 1, int(k))
        if je >= js:
            A[i, js:je + 1] = f[k - np.arange(js, je + 1)]
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None

def multiplicative_descent(f0, time_limit, lr_init=0.01, verbose=False):
    """Multiplicative (mirror) descent with adaptive peak weighting."""
    f = np.maximum(f0.astype(np.float64), 1e-12)

        M = conv.max()

        w = np.exp(scaled)
        w /= w.sum()

        f_new = f * np.exp(-lr * grad_f / max(grad_f.max(), 1e-10))

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(int((time.monotonic() - t0) * 1000) % 100000 + 1)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try relative perturbations
            for frac in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]:
                new_val = old_val * (1.0 + frac)
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                ft = f.copy()
                ft[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] *= scale
                ft = np.maximum(ft, 0)
                ss = ft.sum()
                if ss < 1e-15:
                    continue
                ft /= ss
                val = fast_convolve(ft).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    """
    Multi-strategy optimization combining L-BFGS (softmax parameterization),
    reduced LP-Frank-Wolfe, multiplicative descent, and coordinate descent.
    Explores multiple sequence lengths with interpolation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 50.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {score:.6f} (n={len(a)}) ***")
        return score
    
    # Try parent sequence
    if 'height_sequence_1' in dir() or True:
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            p = np.clip(p, 0.0, 1000.0)
            if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
                parent = p / p.sum()
                p_score = float(evaluate_sequence(parent.tolist()))
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
                if p_score < best_score:
                    best_score = p_score
                    best_seq = parent.copy()
        except Exception:
            parent = None
    
    # === Phase 1: Generate and evaluate initializations ===
    print("=== Phase 1: Initializations ===")
    init_candidates = []
    test_lengths = [512, 640, 768, 896, 1024]
    
    for n_try in test_lengths:
        for v in range(11):
            a_init = make_init(n_try, variant=v, seed=seed)
            init_candidates.append((a_init, f"v{v}_n{n_try}"))
    
    if best_seq is not None:
        init_candidates.append((best_seq, "parent"))
    
    # Score inits by fast convolution objective
    scored_inits = []
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        scored_inits.append((obj, a_norm, label))
    
    scored_inits.sort(key=lambda x: x[0])
    top_inits = scored_inits[:6]
    print(f"  Top inits: {[f'{l}:{o:.4f}' for o, _, l in top_inits[:4]]}")
    
    # === Phase 2: L-BFGS optimization on top inits ===
    print("\n=== Phase 2: L-BFGS Optimization ===")
    for idx, (obj_init, a_init, label) in enumerate(top_inits[:4]):
        rem = deadline - time.monotonic() - 600
        if rem < 30:
            break
        t_alloc = min(rem / max(1, 4 - idx), 180)
        if t_alloc < 20:
            t_alloc = 20
        print(f"  L-BFGS on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
        a_opt, _ = lbfgs_optimize(a_init, time_limit=t_alloc,
                                   tau_start=15.0, tau_end=250.0,
                                   n_phases=5, verbose=True)
        try_seq(a_opt, f"lbfgs_{label}")
    
    # === Phase 3: Multiplicative descent on best ===
    print("\n=== Phase 3: Multiplicative Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  MD on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "md_best")
    
    # === Phase 4: LP-FW with reduced constraints ===
    print("\n=== Phase 4: LP-FW (Reduced) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 60:
            t_alloc = min(rem, 200)
            K = 60 if len(best_seq) <= 640 else 80
            print(f"  LP-FW on best (n={len(best_seq)}, K={K}): t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, K=K, verbose=True)
            try_seq(a_opt, "lpfw")
    
    # === Phase 5: L-BFGS refinement after LP-FW ===
    print("\n=== Phase 5: L-BFGS Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  L-BFGS refine (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_optimize(best_seq, time_limit=t_alloc,
                                       tau_start=50.0, tau_end=400.0,
                                       n_phases=4, verbose=True)
            try_seq(a_opt, "lbfgs_refine")
    
    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [512, 640, 896, 1024, 1280]:
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break
            if len(base) == target_n:
                continue
            f_interp = interpolate_seq(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.005:
                t_alloc = min(rem * 0.35, 80)
                if t_alloc < 15:
                    t_alloc = 15
                print(f"  L-BFGS n={target_n}: t={t_alloc:.0f}s")
                a_opt, _ = lbfgs_optimize(f_interp, time_limit=t_alloc,
                                           tau_start=20.0, tau_end=200.0,
                                           n_phases=4, verbose=False)
                try_seq(a_opt, f"lbfgs_n{target_n}")
    
    # === Phase 7: Random restarts ===
    print("\n=== Phase 7: Random Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(3):
            rem = deadline - time.monotonic() - 150
            if rem < 25:
                break
            t_alloc = min(rem / max(1, 3 - r), 80)
            if t_alloc < 15:
                t_alloc = 15
            
            if r == 0:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = np.maximum(best_seq + noise, 1e-12)
                f_init /= f_init.sum()
            elif r == 1:
                asym = make_init(n_best, variant=int(rng.integers(0, 11)), seed=seed + r * 7)
                f_init = 0.6 * best_seq + 0.4 * asym
                f_init = np.maximum(f_init, 1e-12)
                f_init /= f_init.sum()
            else:
                # Reverse blend
                f_init = 0.4 * best_seq + 0.6 * best_seq[::-1]
                f_init = np.maximum(f_init, 1e-12)
                f_init /= f_init.sum()
            
            print(f"  Restart {r}: t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_optimize(f_init, time_limit=t_alloc,
                                       tau_start=10.0, tau_end=200.0,
                                       n_phases=4, verbose=False)
            try_seq(a_opt, f"restart_{r}")
    
    # === Phase 8: Coordinate descent polish ===
    print("\n=== Phase 8: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd")
    
    # === Phase 9: Final L-BFGS squeeze ===
    print("\n=== Phase 9: Final L-BFGS ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 40)
            print(f"  Final L-BFGS: t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_optimize(best_seq, time_limit=t_alloc,
                                       tau_start=100.0, tau_end=500.0,
                                       n_phases=3, verbose=False)
            try_seq(a_opt, "final_lbfgs")
    
    # Fallback
    if best_seq is None:
        best_seq = make_init(512, variant=0, seed=seed)
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
