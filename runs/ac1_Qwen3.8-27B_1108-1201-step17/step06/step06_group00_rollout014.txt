Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Pressure Equalization (PE) iterations,
a novel fixed-point method that equalizes the "pressure" (weighted cross-correlation)
across the sequence support. Combined with adaptive temperature annealing, multiple
diverse initializations, spectral polishing, and multi-length exploration.

The PE method is based on the KKT conditions: at the optimum, the weighted
cross-correlation of the active convolution peaks with the sequence must be
constant on the support. The update a_new[j] ∝ a[j] / pressure[j] drives
toward this equilibrium.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_weights(w, a):
    """
    Compute c[j] = sum_k w[k] * a[k-j] for j = 0..n-1.
    This is the 'pressure' at position j.
    Uses FFT for efficiency.
    """
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    Wf = rfft(w, fft_size)
    Af = rfft(a, fft_size)
    result = irfft(np.conj(Wf) * Af, fft_size)
    return result[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally (assumes sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def pressure_equalize(a, T, p, max_iters, support_min=1e-10):
    """
    Core Pressure Equalization (PE) iteration.
    
    Parameters:
    - a: sequence (sum=1, a>=0)
    - T: temperature for softmax weights
    - p: power parameter for the update (1 = full equalization, <1 = tempered)
    - max_iters: maximum number of PE iterations
    - support_min: minimum value to maintain (avoid numerical issues)
    
    Returns: updated sequence
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a /= a.sum()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    eps = 1e-15
    
    for it in range(max_iters):
        b = fast_convolve(a)
        b_max = b.max()
        
        # Softmax weights over convolution positions
        log_w = b / T - b_max / T
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        # Compute pressure
        c = cross_corr_weights(w, a)
        c = np.maximum(c, eps)
        
        # PE update: a_new[j] ∝ a[j] * c[j]^(-p)
        a_new = a * np.power(c, -p)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new /= a_sum
        
        # Enforce non-negativity (should already be satisfied)
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        
        a = a_new
        
        # Track best
        if it % 10 == 0 or it == max_iters - 1:
            obj = b.max()  # use previous b as approximation
            # Actually let's recompute
            obj = fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a

                cooling_rate=0.999, checkpoint_interval=100, seed=42):

    PE optimization with temperature annealing.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    no_improve_count = 0
    batch_size = 50
    
    while time.monotonic() - t0 < time_limit:
        a, _ = _pe_batch(a, T, p, batch_size)
        
        b = fast_convolve(a)
        obj = b.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Cool temperature
        T *= cooling_rate
        if T < T_end:
            T = T_end
        
        # Periodic restart to escape local minima
        if no_improve_count > 200:
            # Try a perturbation
            noise_level = 0.001 * (1 + no_improve_count / 100)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_level, 0)
            a /= a.sum()
            no_improve_count = 0
            # Reset temperature
            T = max(T_init * 0.1, T_end)
    
    return best_a, best_obj


def _pe_batch(a, T, p, n_iters):
    """Run n_iters PE iterations."""
    n = len(a)
    eps = 1e-15
    
    for _ in range(n_iters):
        b = fast_convolve(a)
        b_max = b.max()
        
        log_w = (b - b_max) / T
        w = np.exp(log_w)
        w_sum = w.sum()

        c = np.maximum(c, eps)
        
        a_new = a * np.power(c, -p)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new /= a_sum
        a = np.maximum(a_new, 0)
        a /= a.sum()
    
    return a, b.max()
