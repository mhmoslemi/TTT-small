def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fft_size)[:size]

def fft_cross_correlation(w, a):
    """Compute (w ⋆ a)[k] = sum_m w[k+m]*a[m] for k=0..len(a)-1 using FFT."""
    n = len(a)
    b = a[::-1]  # reversed
    # Need convolution of w (length 2n-1) and b (length n)
    conv_len = len(w) + n - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    B = np.fft.rfft(b, fft_size)
    C = np.fft.irfft(W * B, fft_size)
    # g[j] = C[j + n - 1] for j = 0, ..., n-1
    return C[n-1 : 2*n-1]

    grad = (2.0 / tau) * corr

            if best_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    a[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()
                if best_obj_j < best_obj - 1e-14:
                    best_obj = best_obj_j
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            no_improve_rounds += 1
            if no_improve_rounds >= 2:
                break
        else:
            no_improve_rounds = 0
    
    return best_a, best_obj

def subgradient_descent(a0, time_limit, verbose=False, lr_init=0.01):
    """Subgradient descent targeting the max peak directly."""
    a = np.maximum(a0, 1e-10).copy()
    a /= a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find top peaks
        k_top = max(3, len(conv) // 20)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        
        # Compute subgradient (average over top peaks)
        sg = np.zeros(n)
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[js:je+1] += (a[k - idx] + a[idx]) / len(peaks)
        
        # Multiplicative update
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()
        
        obj_new = fast_convolve(a_new).max()
        if obj_new < M:
            a = a_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.01, 0.05)
            else:
                no_improve += 1
        else:
            lr *= 0.8
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-10)
            a /= a.sum()
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def multi_tau_lbfgs(a0, time_limit, verbose=True):
    """Run L-BFGS with decreasing tau (annealing smoothing)."""
    a = a0.copy()
    n = len(a)
    t0 = time.monotonic()
    
    # Start with moderate smoothing, decrease to very sharp
    tau_schedule = [5.0, 10.0, 25.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        if verbose:
            conv = fast_convolve(a)
            print(f"    L-BFGS tau={tau:.0f}, current max_conv={conv.max():.10f}, time_left={remaining:.0f}s")
        
        a, res = lbfgs_optimize(a, tau, maxiter=min(200, int(remaining * 50)))
        
        if verbose:
            conv = fast_convolve(a)
            print(f"    After tau={tau:.0f}: max_conv={conv.max():.10f}, "
                  f"iters={res.nit}, status={res.status}")
    
    return a
