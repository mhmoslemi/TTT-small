```python
import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time

def solve_lp(f, M, n):
    """
    Solve: maximize sum(b) s.t. b >= 0, (f*b)(k) <= M for all k.
    Returns optimal b.
    """
    # c = -ones(n) (since linprog minimizes)
    c = -np.ones(n)
    
    # Build constraint matrix A such that A @ b <= M
    # A has shape (2n-1, n)
    # A[k, j] = f(k-j) if k-j in [0,n-1], else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            fi = k - j  # index into f
            if 0 <= fi < n:
                if f[fi] > 0:
                    rows.append(k)
                    cols.append(j)
                    vals.append(f[fi])
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return -result.fun  # This is sum(b) at optimum
    return None
```

    c[n] = 1.0  # minimize M_new

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
combined with mirror descent and coordinate descent. The LP formulation
directly finds the optimal probability direction that minimizes cross-correlation
peak, providing a more principled descent direction than subgradient methods.

import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_indices(n):
    """Pre-compute row/col indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_direction_lp(f, n, rows, cols):
    """
    Solve: min M_new
           s.t. (f*g)(k) <= M_new for all k
                sum(g) = 1
                g >= 0
    Returns (g, M_new) or (None, None) on failure.
    """

    # Objective: minimize M_new
    c = np.zeros(n + 1)
    c[n] = 1.0

    # Build A_ub: (f*g)(k) - M_new <= 0 for k = 0, ..., 2n-2
    A_rows = []
    A_cols = []
    A_vals = []

    A_cols.extend(cols[nonzero].tolist())
    A_vals.extend(f_vals[nonzero].tolist())

    A_rows.extend(list(range(2 * n - 1)))
    A_cols.extend([n] * (2 * n - 1))
    A_vals.extend([-1.0] * (2 * n - 1))
    
    A_ub = sparse.csc_matrix((A_vals, (A_rows, A_cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub,
                     A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        g = np.maximum(result.x[:n], 0.0)
        s = g.sum()
        if s > 1e-15:
            g /= s
        M_new = result.x[n]
        return g, M_new
    return None, None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding,
combined with mirror descent and coordinate descent for polishing.

Key innovation: At each step, solve a linear program to find the optimal
probability direction g that minimizes the cross-correlation peak with the
current sequence f. This provides a more principled descent direction than
subgradient methods, as it accounts for the full constraint structure.
"""

import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[int(rho)] - 1.0) / (int(rho) + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_data(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        for j in range(js, je):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_direction(f, n, rows, cols, obj_perturb=None):
    """Solve min M s.t. (f*g)(k)<=M, sum(g)=1, g>=0. Returns (g, M_val)."""
    nvar = n + 1
    c = np.zeros(nvar)
    c[n] = 1.0
    if obj_perturb is not None:
        c[:n] += obj_perturb

    f_idx = rows - cols
    f_vals = f[f_idx]
    nz = f_vals > 1e-15
    ar = list(rows[nz]) + list(range(2 * n - 1))
    ac = list(cols[nz]) + [n] * (2 * n - 1)
    av = list(f_vals[nz]) + [-1.0] * (2 * n - 1)

    A_ub = sparse.csc_matrix((av, (ar, ac)), shape=(2 * n - 1, nvar))
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={"presolve": True})
    except Exception:
        return None, None

    if res.success:
        g = np.maximum(res.x[:n], 0.0)
        s = g.sum()
        if s > 1e-12:
            g /= s
        return g, float(res.x[n])
    return None, None


def lp_optimize(f_init, time_limit, verbose=True, label="LP"):
    n = len(f_init)
    f = np.maximum(f_init, 0.0).astype(np.float64)
    ss = f.sum()
    if ss < 1e-15:
        f = np.ones(n) / n
    else:
        f /= ss

    rows, cols = build_lp_data(n)
    conv = fast_convolve(f)
    M = float(conv.max())
    best_f = f.copy()
    best_obj = 2.0 * n * M

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        # Solve LP
        perturb = None
        if no_improve > 50:
            perturb = rng.standard_normal(n) * 0.001

        g, M_new = solve_lp_direction(f, n, rows, cols, perturb)
        if g is None:
            break

        # Line search
        best_t = 0.0
        best_trial = best_obj
        for t in [0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            ft = (1.0 - t) * f + t * g
            ft = np.maximum(ft, 0.0)
            s2 = ft.sum()
            if s2 < 1e-15:
                continue
            ft /= s2
            obj_t = 2.0 * n * float(fast_convolve(ft).max())
            if obj_t < best_trial - 1e-14:
                best_trial = obj_t
                best_t = t

        if best_t > 0.0:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0.0)
            f /= f.sum()
            new_obj = 2.0 * n * float(fast_convolve(f).max())
            if new_obj < best_obj - 1e-13:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and it % 50 == 0:
            print(f"  {label} it={it} obj={best_obj:.10f} M_new={M_new:.8f} "
                  f"t={best_t:.4f} stuck={no_improve}")

        if no_improve > 80:

            fs = np.maximum(fs, 0.0)
            fs /= fs.sum()
            so = 2.0 * n * float(fast_convolve(fs).max())
            if so < best_obj - 1e-12:
                best_obj = so
                best_f = fs.copy()
                f = fs.copy()
                no_improve = 0
            else:
                f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 0.0)
                f /= f.sum()
                no_improve = 0

    return best_f, best_obj


def compute_subgrad(a, peak_idx, weights):
    n = len(a)
    g = np.zeros(n)
    for i in range(len(peak_idx)):
        k = int(peak_idx[i])
        w = weights[i]
        if w < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        g[js:je + 1] += 2.0 * w * a[k - idx]
    return g
