"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach focused on:
1. Multi-length exploration (interpolating to longer sequences for more freedom)
2. FFT-based Adam with aggressive temperature scheduling and SA restarts
3. LP-Frank-Wolfe with adaptive relaxation and dual-guided step sizes
4. Peak-targeted coordinate descent for final shaving
Key improvements over previous: much longer sequences (up to 3072), faster LP
solving, dual-guided Adam, and iterative peak reduction.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):
    """
    Compute gradient of log-sum-exp smooth approximation of max(conv(f,f)/tau).
    Returns (gradient vector, convolution).
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # h[j] = sum_k w[k] * f[k-j]
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv


def adam_optimize(f0, time_limit, tau_init=None, lr_init=0.004, verbose=False):
    """Adam with log-sum-exp smoothing and simulated-annealing restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.1 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    tau_floor = 1e-8
    sa_counter = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = compute_smooth_grad(f, tau)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            sa_counter += 1
        else:
            no_improve += 1

        f = f_new

        # Adaptive temperature: exponential decay with SA reheating
        tau *= 0.9995
        if tau < tau_floor:
            tau = tau_floor

        # Simulated annealing: periodic reheating
        if no_improve > 80:
            lr *= 0.8
        if no_improve > 300 + 50 * sa_counter:
            # SA restart: reheat tau and perturb
            rng = np.random.default_rng(t * 7 + sa_counter * 13)
            noise_scale = 0.001 + 0.004 * min(sa_counter / 5.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            sa_counter += 1
            lr = lr_init * (0.5 + 0.5 / sa_counter)
            tau = 0.06 * fast_convolve(f).max()
            if verbose:
                print(f"    SA restart #{sa_counter} at iter {t}: obj={2*n*best_obj:.8f}")

        if verbose and t % 1000 == 0:
            print(f"    Adam iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, lr={lr:.5f}, sa={sa_counter}")

    return best_f, 2 * n * best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Exploits Toeplitz structure for efficiency."""
    c = -np.ones(n)
    # Build constraint matrix using the Toeplitz structure
    # Row k: f*g at position k = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[k-j] * g[j] <= M
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n))
    for j in range(n):
        # g[j] appears in rows k=j to k=j+n-1 with coefficient f[k-j]
        k_start = j
        k_end = min(j + n, n_cons)
        A_ub[k_start:k_end, j] = f[:k_end - j]
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x, (res.ineqlin_to_dual if hasattr(res, 'ineqlin_to_dual') else None)
        # Try to get dual info
        if hasattr(res, 'mult_s') and res.mult_s is not None:
            return res.x, res.mult_s
        return res.x, None
    except Exception:
        pass
    return None, None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation and dual guidance."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    lp_count = 0
    fail_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        # Solve LP with current M * m_exp
        M_try = M * m_exp
        g, dual = solve_lp_direction(f, M_try, n)
        lp_count += 1

        if g is None:
            fail_count += 1
            m_exp *= 1.15
            if m_exp > 8.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                fail_count = 0
            continue

        fail_count = 0
        g_sum = g.sum()
        if g_sum < 1e-12:
            m_exp *= 1.1
            continue
        g = g / g_sum

        # Check if direction is useful
        if np.max(np.abs(g - f)) < 5e-6:
            no_improve += 1
            m_exp *= 1.05
            if no_improve > 30:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        no_improve = 0

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = max(1.0, m_exp * 0.95)

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 5 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 6.0)

        if no_improve > 60:
            break

    return best_f, 2 * n * best_obj


def peak_shaving_cd(f0, time_limit, verbose=False):
    """Peak-shaving coordinate descent: directly targets top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get top peaks
        n_peaks = min(8, 2 * n - 1)
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]

            if len(idx) == 0:
                continue

            j_max = idx[np.argmax(products)]
            j_other = pk - j_max
            prod_val = f[j_max] * f[j_other]

            if prod_val < 1e-15:
                continue

            if j_max == j_other:
                # Diagonal: reduce f[j_max]
                for factor in [0.5, 0.7, 0.85, 0.9]:
                    trial = f.copy()
                    old_val = trial[j_max]
                    trial[j_max] *= factor
                    diff = old_val - trial[j_max]
                    # Redistribute to elements far from all peaks
                    contrib = np.zeros(n, dtype=bool)
                    for pk2 in peak_indices[:4]:
                        js2 = max(0, int(pk2) - n + 1)
                        je2 = min(n - 1, int(pk2))
                        contrib[js2:je2 + 1] = True
                    free = ~contrib
                    if free.sum() > 3:
                        trial[free] += diff * f[free] / f[free].sum()
                    else:
                        trial = f.copy()  # abort
                        continue
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial /= ts
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
            else:
                # Off-diagonal: reduce the product f[j_max]*f[j_other]
                for scale in [0.4, 0.6, 0.75, 0.85]:
                    trial = f.copy()
                    old_jm = trial[j_max]
                    trial[j_max] = old_jm * scale
                    diff = old_jm - trial[j_max]
                    # Redistribute to "safe" positions
                    contrib = np.zeros(n, dtype=bool)
                    for pk2 in peak_indices[:4]:
                        js2 = max(0, int(pk2) - n + 1)
                        je2 = min(n - 1, int(pk2))
                        contrib[js2:je2 + 1] = True
                    free = ~contrib
                    free[j_max] = False
                    if free.sum() > 3:
                        trial[free] += diff * f[free] / f[free].sum()
                    else:
                        trial = f.copy()
                        continue
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial /= ts
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                rng = np.random.default_rng(int(time.monotonic() * 10000))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it % 20 == 0:
            print(f"    PeakCD iter {it}: obj={2*n*best_obj:.8f}")

        if no_improve > 50:
            break

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_chirp(n, beta=1.0):
    """Chirp-based initialization with good spectral properties."""
    x = np.arange(n, dtype=np.float64)
    phi = beta * x * x / n
    a = np.abs(np.cos(phi)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.3):
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    x = np.arange(n) - n / 2.0
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = 1.0 - t ** 6
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def dual_guided_adam(f0, time_limit, verbose=False):

        try_seq(a_opt, "adam_main")

        if time.monotonic() < deadline - 300:
            a_opt2, obj2 = adam_optimize(best_seq, time_limit=t2b,
                                         tau_init=0.04 * fast_convolve(best_seq).max(),
                                         lr_init=0.002, verbose=True)
            try_seq(a_opt2, "adam_fine")

        if time.monotonic() < deadline - 200:
            a_opt3, obj3 = dual_guided_adam(best_seq, time_limit=t2c, verbose=False)
            try_seq(a_opt3, "dual_adam")

    # === Phase 3: LP-FW refinement ===
    print(f"\n=== Phase 3: LP-FW Refinement (n={len(best_seq)}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        t3 = max(0, min(rem, 700))
        if t3 > 30:
            print(f"  Time: {t3:.0f}s, n={len(best_seq)}")

            # Split: main LP-FW + perturbed restart
            t3a = t3 * 0.7
            t3b = t3 * 0.3

            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t3a, verbose=True)
            try_seq(a_opt, "lpfw_main")

            if time.monotonic() < deadline - 200:
                perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.004)
                a_opt2, obj2 = lp_fw_optimize(perturbed, time_limit=t3b, verbose=True)
                try_seq(a_opt2, "lpfw_pert")

    # === Phase 4: Peak-shaving coordinate descent ===
    print(f"\n=== Phase 4: Peak-Shaving CD (n={len(best_seq)}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        t4 = max(0, min(rem, 350))
        if t4 > 30:
            print(f"  Time: {t4:.0f}s")
            a_opt, obj = peak_shaving_cd(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "peak_cd")

    # === Phase 5: Final combined polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            # Quick Adam
            t5a = min(rem * 0.5, 40)
            if t5a > 10:
                a_opt, obj = adam_optimize(
                    best_seq, time_limit=t5a,
                    tau_init=0.02 * fast_convolve(best_seq).max(),
                    lr_init=0.001, verbose=False)
                try_seq(a_opt, "final_adam")

            # Quick LP-FW
            rem2 = deadline - time.monotonic() - 10
            t5b = max(0, min(rem2, 30))
            if t5b > 10:
                a_opt, obj = lp_fw_optimize(best_seq, time_limit=t5b, verbose=False)
                try_seq(a_opt, "final_lpfw")

            # Quick peak shaving
            rem3 = deadline - time.monotonic() - 5
            t5c = max(0, min(rem3, 20))
            if t5c > 5:
                a_opt, obj = peak_shaving_cd(best_seq, time_limit=t5c, verbose=False)
                try_seq(a_opt, "final_cd")

    # === Multi-length second attempt if time remains ===
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Bonus: Try longer sequence ===")
        current_n = len(best_seq)
        for target_n in [current_n * 2, current_n * 3]:
            if target_n > 5000:
                break
            rem = deadline - time.monotonic() - 30
            if rem < 40:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"bonus_interp_n{target_n}")
            # Quick Adam on this
            t_bonus = min(rem - 20, 60)
            if t_bonus > 20:
                a_opt, obj = adam_optimize(f_interp, time_limit=t_bonus, lr_init=0.003)
                try_seq(a_opt, f"bonus_adam_n{target_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
