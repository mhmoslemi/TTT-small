def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_from_conv(conv, n):
    """Compute objective from convolution (assuming sum=1)."""
    return 2.0 * n * conv.max()


def project_nonneg_sum1(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def make_gaussian(n, center_frac=0.5, sigma_frac=0.2):
    """Gaussian-shaped sequence."""
    x = np.arange(n)
    c = n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - c) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_parabolic(n, p=1.0):

    x = 2.0 * np.arange(n) / n - 1.0
    a = np.maximum(1.0 - x ** 2, 0.0) ** p
    s = a.sum()
    if s > 1e-15:

    return a

def spectral_init(n, tau=0.5, n_iter=50, seed=0):
    """
    Initialize via spectral method:
    1. Start with random non-negative sequence.
    2. Compute power spectrum.
    3. Flatten the spectrum (raise low values, lower high values).
    4. Take IFFT, clip to non-negative.
    5. Iterate.
    """
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n)) + 0.1
    a = a / a.sum()
    
    for it in range(n_iter):
        # Compute power spectrum
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Flatten: target is geometric mean of P
        log_P = np.log(P + 1e-30)
        target = np.exp(np.median(log_P))
        
        # Blend current and target
        P_new = (1.0 - tau) * P + tau * target
        
        # Take IFFT, keep real part, clip
        A_new = np.sqrt(P_new)  # magnitude
        # Preserve phase
        phase = np.angle(A)
        A_new = A_new * np.exp(1j * phase)
        
        # IFFT
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        a_new = np.maximum(a_new - a_new.min(), 0.0)  # Remove DC offset
        s = a_new.sum()
        if s < 1e-15:
            a_new = a  # Revert
        else:
            a_new /= s
            a = a_new
    
    return a


def solve_lp_direction(f, M, n):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    c = -np.ones(n)
    # Build sparse constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, j + n)
        A_ub[start:end, j] = f[:end - start]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                   options={'presolve': True, 'time_limit': max(0.5, 30.0)})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None

        # grad[i] = 2 * sum_{j} f[j] * p[j+i] * (1/tau)

        # But M is n x n and we're doing this every iteration... O(n^2) per iteration.
        # For n=1024, that's 1M operations per iteration. At ~100M ops/sec, that's ~10ms.
        # Acceptable for a few hundred iterations.
        
        # Actually, let me use the FFT approach properly:
        # grad[i] = sum_{j=0}^{n-1} f[j] * p[j+i]
        # Let's define v[0..2n-2] = p[0..2n-2] (the softmax weights)
        # And w[0..n-1] = f[0..n-1]
        # grad[i] = sum_j w[j] * v[j+i] for i = 0, ..., n-1
        # This is the cross-correlation: (w *_ v)[i] = sum_j w[j] * v[j+i]
        # Using FFT: cross_corr = ifft(fft(w, N) * conj(fft(v, N)))... 
        # No, the standard FFT-based cross-correlation is:
        # (x *_ y)[k] = sum_n x[n+k] * y[n] = ifft(conj(FFT(x)) * FFT(y))[k]  (circular)
        # For linear, we need to zero-pad.
        
        # Let me just do it the straightforward way with a padded FFT:
        # We want c[i] = sum_{j=0}^{n-1} f[j] * p[j+i] for i=0,...,n-1

        # Define v of length 2n-1: v[j] = p[j] for j=0..2n-2, 0 otherwise.
        # c[i] = sum_j u[j] * v[j+i]
        # = (v * reverse(u))[i + (n-1)]  where * is linear convolution of length 2n-1 and n
        # Hmm, let me just verify with a small example.

        # c[i] = sum_j u[j] * v[j+i]
        # Let's reverse v: v_rev[k] = v[n-1-k]... no, v has length 2n-1.

        # c = Ifft(fft(v, N) * fft(reverse(u), N)) ... I'll just test numerically.
        
        # FOR NOW, let me use a simpler approach: compute the gradient using the
        # convolution matrix explicitly. For n <= 2048, this is O(n^2) which is fine.

        # I'm overcomplicating this. Let me just directly compute:

        # For the smoothed version: grad[i] = (2/tau) * sum_k p[k] * f[k-i]
        # where f[k-i] = 0 if k-i not in [0, n-1].

        # grad = (2/tau) * T @ f where T[i,k] = 1 if k-i in [0,n-1], i.e., T is a band matrix.

        # grad[i] = sum_{k=max(i,0)}^{min(i+n-1, 2n-2)} p[k] * f[k-i]
        # Let j = k-i: grad[i] = sum_{j=0}^{min(n-1, 2n-2-i)} p[i+j] * f[j]
        # = <p[i:i+min(n,2n-1-i)], f[:min(n,2n-1-i)]>
        
        # This is the cross-correlation of p and f. Using the "overlap and add" FFT method:
        # Or more simply, use scipy.signal.fftconvolve or just do it with a loop for small n.
        
        # For n up to 2048, I'll use the direct O(n^2) computation.
        # Actually, I can do it in O(n log n) using FFT:
        # grad[i] = sum_j f[j] * p[j+i] = (f * p_rev)[i + n - 1]... 
        # No. Let me just use:
        # Define a vector v of length 2n-1: v = p
        # Define a vector u of length n: u = f  
        # The quantity we want is c[i] = sum_{j=0}^{n-1} u[j] * v[j+i] for i=0..n-1
        # This is the "cross-correlation" which can be computed as:
        # c = ifft(fft(v, 4n) * conj(fft(u, 4n)))[n-1 : 2n-1] ... I need to be careful.

        # (p ⊛ f)[i] = sum_j p[j+i] * f[j] = grad[i] * (tau/2)
        # We need (p ⊛ f)[i] for i = 0, ..., n-1
        # FFT method: pad p and f to length L >= (2n-1) + n - 1 = 3n-2
        # FFT(p ⊛ f) = conj(FFT(p)) * FFT(f)  ... wait, is it conj(FFT(x))*FFT(y) or FFT(x)*conj(FFT(y))?
        
        # The cross-correlation (x ⊛ y)[k] = sum_n x[n+k] * conj(y[n])
        # In our case, both are real, so: (x ⊛ y)[k] = sum_n x[n+k] * y[n]
        # FFT: DFT(x ⊛ y) = conj(DFT(x)) * DFT(y)  ... let me verify.
        # DFT(x ⊛ y)[k] = sum_m (x ⊛ y)[m] * exp(-2πimk/L)
        # = sum_m sum_n x[n+m] y[n] * exp(-2πimk/L)
        # Let p = n+m: = sum_p sum_n x[p] y[n] * exp(-2πi(p-n)k/L)
        # = (sum_p x[p] exp(-2πipk/L)) * (sum_n y[n] exp(2πink/L))
        # = X[k] * conj(Y[k])
        
        # So: (x ⊛ y) = ifft(FFT(x) * conj(FFT(y)))
        
        # In our case: x = p, y = f
        # grad[i] * tau/2 = (p ⊛ f)[i] = ifft(FFT(p) * conj(FFT(f)))[i]

        F_fft = np.fft.rfft(f, L)
        cross = np.fft.irfft(P_fft * np.conj(F_fft), L)[:n]
        
        grad = (2.0 / tau) * cross
        
        # Multiplicative update (mirror descent on simplex)
        f_new = f * np.exp(-eta * grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        # Compute new objective
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        # Nesterov update
        # f_mom = f_new + momentum * (f_new - f)
        f_mom = f_new + momentum * (f_new - f)
        f_mom = project_nonneg_sum1(f_mom)
        
        # Actually let me use a simpler momentum:
        # f = f_new (just use the new value)
        f = f_new
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    SMD iter {it}: obj={best_obj:.10f}, tau={tau:.6f}")
        else:
            no_improve += 1
        
        # Adaptive temperature
        if no_improve > 30:
            tau = max(tau * 0.9, tau_min)
            no_improve = 0
        
        # Adaptive learning rate
        if it % 100 == 0:
            eta *= 0.95
        
        if momentum > 0.5 and it > 200:
            momentum *= 0.999
        
        if no_improve > 200:
            break
    
    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """
    LP-based Frank-Wolfe with away-step for faster convergence.
    """
    f = project_nonneg_sum1(f0.copy())
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    # Track active set for away-step
    active_set = set()
    
    m_levels = [1.0, 1.02, 1.05, 1.1, 1.2, 1.5, 2.0, 3.0]
    m_idx = 0
    
    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3.0:
            break
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Solve LP direction
        g = None
        for level in range(m_idx, len(m_levels)):
            M_try = M * m_levels[level]
            g = solve_lp_direction(f, M_try, n)
            if g is not None and g.sum() > 1e-12:
                m_idx = level
                break
            m_idx = level + 1
        
        if g is None or g.sum() < 1e-12:
            # Perturb and try again
            rng = np.random.default_rng(it * 7 + 13)
            f = project_nonneg_sum1(f + rng.standard_normal(n) * 0.002)
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        g = g / g.sum()
        
        # Check if direction is meaningful
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        no_improve = 0
        
        # Line search: find optimal alpha in [0, 1]
        # Objective along direction: phi(alpha) = max(conv(f + alpha*(g-f)))
        best_alpha = 0.0
        best_val = M
        
        # Coarse grid
        alphas = [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine search around best_alpha
        if 0 < best_alpha < 1:
            step = max(0.005, best_alpha * 0.1)
            for da in [-2*step, -step, step, 2*step]:
                a_try = best_alpha + da
                if a_try <= 0 or a_try >= 1.01:
                    continue
                ft = f + a_try * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = a_try
        
        if best_alpha < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            continue
        
        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        
        f = f_new
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f.copy()
            m_idx = 0
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: identify peaks and reduce contributing mass.
    """
    f = project_nonneg_sum1(f0.copy())
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 100:
        conv = fast_convolve(f)
        M = conv.max()
        
        # Get top peaks
        k_peaks = np.argsort(conv)[-5:][::-1]
        
        improved = False
        for k in k_peaks:
            k = int(k)
            # Contributing indices
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contrib = np.arange(js, je + 1)
            
            if len(contrib) == 0:
                continue
            
            products = f[contrib] * f[k - contrib]
            
            if products.sum() < 1e-15:
                continue
            
            # Find top contributor
            src = contrib[np.argmax(products)]
            
            # Try reducing src and distributing to non-contributing coords
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[contrib] = False
            
            if non_contrib.sum() < 3:
                # Too many contributors, try a different approach
                non_contrib[:] = True
                non_contrib[src] = False
            
            best_trial = None
            best_trial_obj = M
            
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
                diff = f[src] * frac
                if diff < 1e-12:
                    continue
                trial = f.copy()
                trial[src] -= diff
                nc = trial[non_contrib]
                nc_sum = nc.sum()
                if nc_sum < 1e-12:
                    continue
                # Distribute proportionally to existing values (preserves shape)
                trial[non_contrib] += diff * (nc / nc_sum)
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-15:
                    best_trial_obj = obj
                    best_trial = trial
            
            if best_trial is not None:
                f = best_trial
                improved = True
                if best_trial
