This is actually a very natural approach! Let me formalize:

**Mirror Descent with Entropy Regularization:**
- Regularizer: Ω(a) = sum_j a[j] * log(a[j]) (negative entropy)
- Bregman divergence: D(a, b) = sum_j [a[j] log(a[j]/b[j]) - a[j] + b[j]]
- Update: a_{k+1} = argmin_{a ≥ 0, sum a = 1} { <g_k, a> + (1/η) D(a, a_k) }
- Closed form: a_{k+1}[j] = a_k[j] * exp(-η * g_k[j]) / Z (where Z normalizes)

This is the "multiplicative weights" update! And it's known to converge for convex problems. Our problem is non-convex, but mirror descent can still work well in practice for non-convex problems (it's like gradient descent in the "exponential parametrization").

    return f_interp


def nesterov_mirror_descent(a_init, time_limit, temp_init=0.05, temp_end=0.005,
                            eta_init=0.05, verbose=True, momentum=0.9):
    """
    Nesterov-accelerated mirror descent with entropy regularizer.
    Update: a_{k+1} = softmax(log(a_k) - eta * (momentum * v + g_k))
    where g_k is the soft-gradient and v is the momentum accumulator.
    """
    n = len(a_init)
    a = normalize_a(a_init.copy() + 1e-10)  # Ensure strictly positive
    z = np.log(a)  # Log-space parameterization
    
    best_a = a.copy()
    best_score = eval_score(a)
    
    v = np.zeros(n)  # Momentum in gradient space
    t0 = time.monotonic()
    iters = 0
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Anneal temperature and learning rate
        progress = min(1.0, iters / max(1, int(time_limit * 10)))  # Rough estimate
        # Actually, let's use time-based annealing
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        temp = temp_init * (temp_end / temp_init) ** elapsed_frac
        eta = eta_init * (1.0 / (1.0 + 0.5 * elapsed_frac * iters / 100))
        
        # Compute gradient
        g = compute_soft_gradient(a, temp, peak_count=max(10, n // 4))
        
        # Momentum update
        v = momentum * v + g
        
        # Mirror descent step in log-space
        z = z - eta * v
        # Renormalize in log-space (subtract max for numerical stability)
        z = z - z.max()
        a_new = np.exp(z)
        a_new = a_new / a_new.sum()
        
        score = eval_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
        
        a = a_new
        
        if patience > 200:
            # Restart: add noise in log-space
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            z = np.log(np.maximum(a, 1e-10)) + rng.standard_normal(n) * 0.05
            z = z - z.max()
            a = np.exp(z)
            a = a / a.sum()
            v = np.zeros(n)
            patience = 0
        
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD iter={iters} ({elapsed:.0f}s): score={best_score:.6f} temp={temp:.5f}")
    
    return best_a, best_score


def peak_shave_optimize(a_init, time_limit, verbose=True):

    Peak-shaving optimization: iteratively reduce the maximum convolution peak
    by adjusting the contributing elements.
    """

    best_a = a.copy()
    best_score = eval_score(a)

        b_inner[-1] = -np.inf
        peak_k = int(np.argmax(b_inner))
        peak_val = b[peak_k]

            break
        
        # Find contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Find the top contributing pairs
        top_idx = np.argsort(contribs)[::-1][:5]
        
        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = int(peak_k - i)
            
            if a[i] < 1e-12:
                continue
            
            # Try reducing a[i] and redistributing
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-14:
                    continue

                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask[j] = False  # Don't add to the paired element

                if others.sum() < 1e-14:

                a_trial = normalize_a(a_trial)
                
                score = eval_score(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break

        if not improved:

                            continue
                        # Reduce others proportionally

                        mask[ci] = False
                        others = a[mask]
                        if others.sum() < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[ci] += diff
                        a_trial[mask] = others - diff * (others / others.sum())
                        a_trial = normalize_a(a_trial)
                        score = eval_score(a_trial)

Minimizes 2*n*max(conv(a,a))/(sum(a)^2) using Nesterov-accelerated mirror descent
(entropy-regularized multiplicative updates), peak-shaving local search,
coordinate descent polish, and multi-start with diverse asymmetric initializations.
Key improvement over previous: much faster per-iteration cost (O(n log n) vs LP-based),
allowing many more iterations; adaptive temperature and learning rate; simulated
annealing restarts to escape local minima; exploits sequence asymmetry for lower peaks.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-convolution of sequences a and b via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_gradient(a, temp, peak_count=None):
    """
    Compute gradient of soft-max smoothed max auto-convolution.
    Uses FFT-based cross-correlation for O(n log n) complexity.
    """
    n = len(a)
    b = fast_convolve(a)

    if peak_count is not None and peak_count < len(b):
        idx = np.argpartition(b, -peak_count)[-peak_count:]
        b_masked = np.full_like(b, -1e30)
        b_masked[idx] = b[idx]
    else:
        b_masked = b

    max_b = b_masked.max()
    scaled = (b_masked - max_b) / temp
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum

    # g[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    # = 2 * (reverse(a) * w)[n-1+j]
    u = a[::-1]
    conv_uw = cross_convolve(u, w)
    g = 2.0 * conv_uw[n - 1:2 * n - 1]
    return g

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    rounds = 0
    while time.monotonic() - t0 < time_limit:
        rounds += 1
        any_improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-13:
                continue

            # Coarse search
            for factor in coarse_factors:
                new_val = a[j] * factor
                diff = new_val - a[j]
                if abs(diff) < 1e-14:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                others_sum = a[mask].sum()
                if others_sum < 1e-13:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                a_trial[mask] = a[mask] - diff * (a[mask] / others_sum)
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-13:
                    continue
                a_trial /= s
                score = eval_score(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    any_improved = True
                    break

            # Fine search (only if coarse found something nearby)
            if a[j] > 1e-13:
                for factor in fine_factors:
                    new_val = a[j] * factor
                    diff = new_val - a[j]
                    if abs(diff) < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    others_sum = a[mask].sum()
                    if others_sum < 1e-13:
                        continue
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    a_trial[mask] = a[mask] - diff * (a[mask] / others_sum)
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-13:
                        continue
                    a_trial /= s
                    score = eval_score(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        any_improved = True
                        break

        if verbose and rounds % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD round={rounds} ({elapsed:.0f}s): score={best_score:.6f}")

        if not any_improved and rounds > 20:
            break

    return best_a, best_score


def log_space_polish(a_init, time_limit, verbose=True):
    """
    L-BFGS optimization in log-space with soft-max smoothing.
    Uses scipy.optimize.minimize with L-BFGS-B.
    """
    from scipy.optimize import minimize
    n = len(a_init)
    a = normalize_a(a_init.copy() + 1e-12)
    z0 = np.log(a)
    z0 = z0 - z0.max()

    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stopped = [False]

    # Use a fixed temperature for the smoothing
    T = 100.0

    def objective(z):
        if time.monotonic() - t0 > time_limit * 0.9:
            stopped[0] = True
            return 1e15
        a_temp = np.exp(z - z.max())
        a_temp = a_temp / a_temp.sum()
        score = eval_score(a_temp)
        if score < best_score:
            best_score = score
            best_a = a_temp.copy()
        return score

    def gradient(z):
        if time.monotonic() - t0 > time_limit * 0.9:
            stopped[0] = True
            return np.zeros_like(z)
        a_temp = np.exp(z - z.max())
        a_temp = a_temp / a_temp.sum()
        # Gradient of soft-max smoothed objective
        b = fast_convolve(a_temp)
        max_b = b.max()
        scaled = (b - max_b) / T
        w = np.exp(scaled)
        w = w / w.sum()
        # Cross-correlation gradient
        u = a_temp[::-1]
        conv_uw = cross_convolve(u, w)
        grad_a = 2.0 * conv_uw[n - 1:2 * n - 1]
        # Chain rule for softmax: grad_z = a * (grad_a - sum(grad_a * a))
        S = np.dot(grad_a, a_temp)
        grad_z = a_temp * (grad_a - S)
        return grad_z

    try:
        res = minimize(
            fun=objective,
            x0=z0,
            jac=gradient,
            method='L-BFGS-B',
            options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': 50000, 'disp': False}
        )
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a

    score = eval_score(a_out)
    if score < best_score:
        best_score = score
        best_a = a_out.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    LogL-BFGS ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def get_initializations(parent, n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        if len(parent) == n:
            inits.append(("parent", parent.copy()))
        else:
            inits.append(("parent_interp", interpolate_sequence(parent, n)))

    # Uniform
    inits.append(("uniform", np.ones(n) / n))

    # Uniform + random perturbation
    for scale in [0.1, 0.3]:
        pert = rng.standard_normal(n) * scale
        a = np.ones(n) / n + pert
        a = normalize_a(a)
        inits.append((f"uniform_pert_{scale}", a))

    # Exponential ramp (left-heavy)
    x = np.linspace(0, 1, n)
    a = np.exp(2.0 * x)
    a = normalize_a(a)
    inits.append(("ramp_left", a))

    # Exponential ramp (right-heavy)
    a = np.exp(-2.0 * x)
    a = normalize_a(a)
    inits.append(("ramp_right", a))

    # Sinusoidal modulation
    for freq in [1, 2, 3]:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * x)
        a = normalize_a(a)
        inits.append((f"sin_{freq}", a))

    # Random (smooth)
    a = np.abs(rng.standard_normal(n)) + 0.1
    # Smooth it
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    a = normalize_a(a)
    inits.append(("random_smooth", a))

    # Sparse peaks
    m = max(3, n // 100)
    positions = rng.choice(n, size=m, replace=False)
    a = np.zeros(n)
    a[positions] = 1.0
    a = normalize_a(a + 0.01)
    # Slightly smooth
    kernel2 = np.ones(3) / 3
    a = np.convolve(a, kernel2, mode='same')
    a = normalize_a(a)
    inits.append(("sparse", a))

    return inits


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate via evaluate_sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with diverse strategies:
    1. Mirror descent (multiplicative updates) from multiple initializations
    2. Peak-shaving local search
    3. Coordinate descent polish
    4. L-BFGS in log-space
    5. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Try multiple lengths
    n_values = [768]
    if parent is not None:
        n_parent = len(parent)
        if n_parent not in n_values:
            n_values.insert(0, n_parent)
    # Add nearby lengths
    for n_alt in [640, 896, 1024, 512, 1280]:
        if n_alt not in n_values:
            n_values.append(n_alt)
    n_values = n_values[:5]  # Limit to 5 lengths

    print(f"\n=== Multi-phase optimization, budget={budget_s:.0f}s ===")
    print(f"Lengths to try: {n_values}")

    for n_try in n_values:
        rem = deadline - time.monotonic()
        if rem < 120:
            print(f"  Skipping n={n_try}: only {rem:.0f}s remaining")
            break

        print(f"\n--- n={n_try}, remaining={rem:.0f}s ---")

        # Generate initializations
        inits = get_initializations(parent, n_try, rng)

        # Allocate time among initializations
        n_inits = len(inits)
        time_per_init = min(rem * 0.7 / max(1, n_inits), 300.0)

        for init_name, init_a in inits:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(time_per_init, rem * 0.8)
            if t_alloc < 20:
                break

            print(f"  [{init_name}] n={n_try}, t={t_alloc:.0f}s")

            # Phase 1: Mirror descent
            a_md, score_md = mirror_descent_optimize(
                init_a, t_alloc * 0.5,
                temp_init=0.06, temp_end=0.003,
                eta_init=0.06, momentum=0.88,
                restart_every=200, verbose=False
            )
            best_seq, best_score = check_and_update(a_md, f"MD_{init_name}_n{n_try}", best_seq, best_score)

            # Phase 2: Peak shaving
            rem = deadline - time.monotonic()
            if rem > 30:
                t_ps = min(rem * 0.25, 80.0)
                a_ps, score_ps = peak_shave_optimize(a_md, t_ps, verbose=False)
                best_seq, best_score = check_and_update(a_ps, f"PS_{init_name}_n{n_try}", best_seq, best_score)

            # Phase 3: Coordinate descent
            rem = deadline - time.monotonic()
            if rem > 20:
                t_cd = min(rem * 0.15, 50.0)
                a_cd, score_cd = coordinate_descent_polish(a_ps if 'a_ps' in dir() else a_md, t_cd, verbose=False)
                best_seq, best_score = check_and_update(a_cd, f"CD_{init_name}_n{n_try}", best_seq, best_score)

    # === Final polish on overall best ===
    print(f"\n=== Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Log-space L-BFGS
        rem = deadline - time.monotonic()
        if rem > 30:
            t_lb = min(rem * 0.5, 120.0)
            print(f"  L-BFGS log-space: n={n_best}, t={t_lb:.0f}s")
            a_lg, score_lg = log_space_polish(best_seq, t_lb, verbose=True)
            best_seq, best_score = check_and_update(a_lg, "final_lbgfs", best_seq, best_score)

        # Another round of mirror descent on the polished solution
        rem = deadline - time.monotonic()
        if rem > 30:
            t_md2 = min(rem * 0.4, 80.0)
            print(f"  MD round 2: n={n_best}, t={t_md2:.0f}s")
            a_md2, score_md2 = mirror_descent_optimize(
                best_seq, t_md2,
                temp_init=0.02, temp_end=0.001,
                eta_init=0.03, momentum=0.92,
                restart_every=400, verbose=False
            )
            best_seq, best_score = check_and_update(a_md2, "MD2_final", best_seq, best_score)

        # Peak shave
        rem = deadline - time.monotonic()
        if rem > 20:
            t_ps2 = min(rem * 0.5, 60.0)
            a_ps2, _ = peak_shave_optimize(best_seq, t_ps2, verbose=False)
            best_seq, best_score = check_and_update(a_ps2, "PS2_final", best_seq, best_score)

        # Coordinate descent
        rem = deadline - time.monotonic()
        if rem > 15:
            t_cd2 = min(rem * 0.6, 40.0)
            a_cd2, _ = coordinate_descent_polish(best_seq, t_cd2, verbose=False)
            best_seq, best_score = check_and_update(a_cd2, "CD2_final", best_seq, best_score)

        # Try neighboring lengths
        rem = deadline - time.monotonic()
        if rem > 60:
            for factor in [0.8, 1.2, 1.5]:
                alt_n = int(n_best * factor)
                if alt_n < 128 or alt_n > 4096 or alt_n == n_best:
                    continue
                rem = deadline - time.monotonic()
                if rem < 40:
                    break
                t_alt = min(rem * 0.4, 60.0)
                init_alt = interpolate_sequence(best_seq, alt_n)
                a_alt, _ = mirror_descent_optimize(
                    init_alt, t_alt,
                    temp_init=0.04, temp_end=0.002,
                    eta_init=0.05, momentum=0.9,
                    restart_every=300, verbose=False
                )
                best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
