def peak_targeted_second_order(sequence, time_limit):
    """Identifies peak contributors and redistributes mass."""
    try:
        conv = np.convolve([float(x) for x in sequence], [float(x) for x in sequence], mode='full')
        n = len(sequence)
    except:
        return sequence, float(evaluate_sequence(sequence))
    
    peak_index = np.argmax(conv)
    # Identify the specific contribution indices
    contrib_indices = []
    for i in range(len(sequence)):
        if abs(sequence[i]) < 1e-12:
            continue
        for j in range(len(sequence)):
            if i + j == peak_index and j < len(sequence):
                contrib_indices.append((i, j))
                break
    
    # Calculate redistributions to reduce peak contributions
    # For each peak contributor, reduce its value and distribute to non-contributing coordinates
    result_candidate = sequence.copy()
    total_contrib = 0.0
    
    try:
        for idx_pair in contrib_indices:
            idx1, idx2 = idx_pair
            reduction = 0.15 * result_candidate[idx1] * result_candidate[idx2] / (n + len(contrib_indices))
            result_candidate[idx1] -= reduction
            result_candidate[idx2] -= reduction
            
            # Find a target index (could be a non-contributing for centroid shift)
            target_idx = np.argmax([abs(sequence[i] - result_candidate[i]) for i in range(n)])
            result_candidate[target_idx] += reduction
    except Exception as e:
        print(f"Redistribution error: {e}.")

    max_iter = int(time_limit / 12.0)

            grad = calculate_gradient(result_candidate, i)

        pass
        
    return result_candidate, float(evaluate_sequence(result_candidate))
