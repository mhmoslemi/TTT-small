import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_grad(f, tau):
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv

(2) Nesterov-accelerated mirror descent with adaptive temperature,
(3) Peak-chasing refinement with targeted mass redistribution,
(4) Interpolation-based length adaptation,
(5) Final coordinate descent polish.
Key innovations: entropy-regularized mirror descent on the simplex,
adaptive "peak chasing" that switches optimization focus to newly
emergent peaks, and spectral annealing for multi-scale exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):
    """Compute gradient of log-sum-exp smoothed max-convolution."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv


def mirror_descent_step(x, grad, eta):
    """Multiplicative weights (mirror descent) step on the simplex."""
    n = len(x)
    log_x = np.full(n, -700.0)
    mask = x > 1e-300
    log_x[mask] = np.log(x[mask])
    log_x -= log_x.max()
    unnormalized = np.exp(log_x - eta * grad)
    s = unnormalized.sum()
    if s < 1e-300:
        return x.copy()
    return unnormalized / s


def nesterov_mirror_descent(f0, time_limit, tau_init=None, lr_init=0.01,
                            verbose=False):
    """Nesterov-accelerated mirror descent on the smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.1 * conv.max()
    else:
        tau = tau_init

    # Nesterov variables
    y = f.copy()  # extrapolated point
    t = 1.0
    eta = lr_init
    m_eta = 0.0
    v_eta = 0.0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    it = 0
    total_iter = 20000  # rough estimate for scaling

    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = min(it / total_iter, 1.0)
        # Cosine temperature schedule
        tau = max(1e-8 * conv.max() if conv.max() > 0 else 1e-8,
                  tau_init * 0.1 * (0.5 + 0.5 * np.cos(np.pi * frac)))
        if tau <= 0:
            tau = 1e-7

        # Compute gradient at y (Nesterov lookahead)
        grad_y, _ = compute_smooth_grad(y, tau)

        # Adam-momentum on eta
        m_eta = beta1 * m_eta + (1 - beta1) * grad_y
        v_eta = beta2 * v_eta + (1 - beta2) * grad_y * grad_y
        m_hat = m_eta / (1 - beta1 ** it)
        v_hat = v_eta / (1 - beta2 ** it)
        grad_norm = np.sqrt(np.mean(v_hat)) + eps

        # Adaptive learning rate
        if no_improve > 50:
            eta = lr_init * (0.5 ** (no_improve // 100))
        eta = np.clip(eta, 1e-6, 0.5)

        # Mirror descent step on x
        f_new = mirror_descent_step(f, grad_y, eta)

        # Nesterov extrapolation
        t_new = 1.0 / (1.0 + np.sqrt(1 - 0.1 / (t + 1)))
        mu = (1 - t_new) / t if t > 0 else 0
        y = f_new + mu * (f_new - f)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        t = t_new
        f = f_new

        conv_new = fast_convolve(f)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            # Restart from best with perturbation
            rng = np.random.default_rng(it * 31 + 17)
            noise = rng.standard_normal(n) * (0.002 + 0.003 * min(no_improve / 500.0, 2.0))
            f = project_simplex(best_f + noise)
            y = f.copy()
            t = 1.0
            m_eta = np.zeros(n)
            v_eta = np.zeros(n)
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    NMD restart at iter {it}: obj={2*n*best_obj:.8f}")

        if verbose and it % 1000 == 0:
            conv_check = fast_convolve(best_f)
            print(f"    NMD iter {it}: obj={2*n*conv_check.max():.8f}, "
                  f"tau={tau:.2e}, eta={eta:.5f}")

    return best_f, 2 * n * best_f and 2 * n * fast_convolve(best_f).max()


def adam_smooth_optimize(f0, time_limit, tau_init=None, lr_init=0.003, verbose=False):
    """Adam optimization on log-sum-exp smoothed max-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.08 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = compute_smooth_grad(f, tau)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Cosine temperature decay
        progress = min(t / 5000.0, 1.0)
        tau = max(tau_init * 0.02, tau_init * 0.5 * (0.5 + 0.5 * np.cos(np.pi * progress)))
        if tau < 1e-7:
            tau = 1e-7

        if no_improve > 80:
            lr *= 0.85
        if no_improve > 300:
            rng = np.random.default_rng(t * 13 + 7)
            noise_scale = 0.002 + 0.004 * min(no_improve / 1000.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.8
            tau = 0.04 * fast_convolve(f).max()
            if verbose:
                print(f"    Adam restart at iter {t}: obj={2*n*best_obj:.8f}")

        if verbose and t % 500 == 0:
            print(f"    Adam iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}")

    return best_f, 2 * n * best_obj


def peak_chasing_optimize(f0, time_limit, verbose=False):
    """
    Peak-chasing: alternates between reducing the current peak and
    preventing other peaks from growing. Uses targeted mass transfers.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get top peaks (up to 5)
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]
        peak_indices = peak_indices[::-1]  # descending

        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max

            # Find target: element with low contribution to all top peaks
            # and low correlation with the peak
            peak_contrib = np.zeros(n)
            for pk2 in peak_indices[:3]:
                pk2 = int(pk2)
                js2 = max(0, pk2 - n + 1)
                je2 = min(n - 1, pk2)
                for jj in range(js2, je2 + 1):
                    peak_contrib[jj] += f[jj] * f[pk2 - jj]

            # Candidate targets: low peak_contrib, not j_max or j_other
            candidates = np.where((peak_contrib < 0.3 * M) &
                                  (np.arange(n) != j_max) &
                                  (np.arange(n) != j_other))[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != j_max)[0]

            # Try to reduce the peak by moving mass from j_max to a candidate
            old_val = f[j_max]
            if old_val < 1e-12:
                continue

            best_trial = None
            best_trial_obj = M

            for frac_transfer in [0.1, 0.2, 0.3, 0.4]:
                transfer_amount = old_val * frac_transfer
                # Distribute to candidates proportional to their current value
                cand_vals = f[candidates]
                if cand_vals.sum() < 1e-15:
                    dist = np.ones(len(candidates)) / len(candidates)
                else:
                    dist = cand_vals / cand_vals.sum()

                trial = f.copy()
                trial[j_max] -= transfer_amount
                for ci in range(len(candidates)):
                    trial[candidates[ci]] += transfer_amount * dist[ci]
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial /= ts
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-15:
                    best_trial_obj = obj
                    best_trial = trial.copy()

            if best_trial is not None and best_trial_obj < best_obj - 1e-14:
                f = best_trial
                best_obj = best_trial_obj
                best_f = f.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Try small gradient step
                tau = 0.03 * M
                grad, _ = compute_smooth_grad(f, tau)
                lr = 0.001 / (np.max(np.abs(grad)) + 1e-10)
                f = project_simplex(f - lr * grad)
                conv_new = fast_convolve(f)
                if conv_new.max() < best_obj - 1e-14:
                    best_obj = conv_new.max()
                    best_f = f.copy()
                    no_improve = 0
                else:
                    f = best_f.copy()

            if no_improve > 100:
                rng = np.random.default_rng(it * 41 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
        else:
            no_improve = 0

        if no_improve > 500:
            break

    return best_f, 2 * n * best_f and 2 * n * fast_convolve(best_f).max()


def coordinate_polish(f0, time_limit, verbose=False):
    """Fine coordinate descent polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Try pairs of coordinates
        step = max(1, n // 200)
        for i in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                break
            j = min(i + n // 2, n - 1)
            if i == j:
                continue

            old_i = f[i]
            old_j = f[j]
            if old_i < 1e-12 and old_j < 1e-12:
                continue

            for di in [-0.1, -0.05, 0.05, 0.1]:
                for dj in [-di, -0.5 * di, 0.5 * di, di]:
                    new_i = old_i + di * max(old_i, 0.01)
                    new_j = old_j + dj * max(old_j, 0.01)
                    if new_i < 0 or new_j < 0:
                        continue
                    trial = f.copy()
                    trial[i] = new_i
                    trial[j] = new_j
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial /= ts
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_f, 2 * n * fast_convolve(best_f).max()


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = _solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                m_exp = min(m_exp * 1.15, 5.0)
            if no_improve > 150:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.15, 5.0)

        if no_improve > 200:
            break

    return best_f, 2 * n * best_obj


def _solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, center_frac=0.4, sigma_frac=0.3):
    """Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_sigmoid(n, sharpness=3.0):
    """Sigmoid-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-sharpness * (x - 0.4)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Flat-topped bell."""
    x = np.arange(n) - n / 2.0
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = 1.0 - t ** 6
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_phasor(n, freq=1.0, asymmetric=True):
    """Phasor-based initialization (exploits frequency structure)."""
    x = np.arange(n) / n
    if asymmetric:
        a = np.abs(np.sin(np.pi * freq * x)) ** 2
        a = a * np.exp(-1.5 * x)
    else:
        a = np.abs(np.sin(np.pi * freq * x)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization with multi-length exploration,
    Nesterov mirror descent, peak chasing, and LP refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Initialization and Multi-Length Exploration ===
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # Generate diverse initializations at multiple lengths
    lengths_to_try = [512, 768, 1024, 1536, 2048]
    initializations = []

    for n_try in lengths_to_try:
        initializations.append((make_gaussian(n_try, 0.35, 0.28), f"gauss_{n_try}"))
        initializations.append((make_sigmoid(n_try, 3.0), f"sigmoid_{n_try}"))
        initializations.append((make_bell(n_try, 0.6), f"bell_{n_try}"))
        initializations.append((make_phasor(n_try, 1.0), f"phasor_{n_try}"))
        # Exponential decay
        a_exp = np.exp(-np.arange(n_try) / (n_try * 0.35))
        a_exp /= a_exp.sum()
        initializations.append((a_exp, f"exp_{n_try}"))

    # Quick evaluation of initializations
    max_init_time = 80
    for a_init, label in initializations:
        if time.monotonic() - t_start > max_init_time:
            break
        try_seq(a_init, label)

    # Also try interpolating parent to other lengths
    if parent is not None:
        for n_try in lengths_to_try:
            if time.monotonic() - t_start > max_init_time:
                break
            if n_try == len(parent):
                continue
            f_interp = interpolate_sequence(parent, n_try)
            try_seq(f_interp, f"interp_parent_{n_try}")

    work_seq = best_seq if best_seq is not None else make_gaussian(768, 0.35, 0.28)
    work_n = len(work_seq)
    print(f"  Best init: n={work_n}, score={best_score:.6f}")

    # === Phase 2: Main Optimization (Nesterov Mirror Descent + Adam) ===
    print(f"\n=== Phase 2: Main Optimization (n={work_n}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.45, 700)
    if t2 > 60:
        print(f"  Time allocated: {t2:.0f}s")

        # Run Nesterov Mirror Descent
        t2a = t2 * 0.5
        a_opt, _ = nesterov_mirror_descent(work_seq, time_limit=t2a,
                                           lr_init=0.015, verbose=True)
        try_seq(a_opt, "nmd_main")

        # Run Adam from the result
        if time.monotonic() < deadline - rem * 0.55:
            rem2 = deadline - time.monotonic() - rem * 0.55
            t2b = min(rem2, t2 * 0.4)
            if t2b > 30:
                a_opt2, _ = adam_smooth_optimize(best_seq, time_limit=t2b,
                                                 tau_init=0.05 * fast_convolve(best_seq).max(),
                                                 lr_init=0.003, verbose=True)
                try_seq(a_opt2, "adam_after_nmd")

    # === Phase 3: Try Other Lengths ===
    print(f"\n=== Phase 3: Multi-Length Exploration ===")
    if best_seq is not None:
        for target_n in [1024, 1536, 2048, 640]:
            rem = deadline - time.monotonic() - 200
            if rem < 60:
                break
            if target_n == len(best_seq):
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_{target_n}")

            # Quick optimization at this length
            t_phase = min(rem * 0.2, 120)
            if t_phase > 25:
                a_opt, _ = nesterov_mirror_descent(f_interp, time_limit=t_phase * 0.6,
                                                   lr_init=0.01, verbose=False)
                try_seq(a_opt, f"nmd_{target_n}")
                if time.monotonic() < deadline - 180:
                    t_phase2 = min((deadline - time.monotonic() - 180) * 0.5, 60)
                    if t_phase2 > 15:
                        a_opt2, _ = adam_smooth_optimize(
                            best_seq, time_limit=t_phase2,
                            tau_init=0.03 * fast_convolve(best_seq).max(),
                            lr_init=0.002, verbose=False)
                        try_seq(a_opt2, f"adam_{target_n}")

    # === Phase 4: Peak Chasing ===
    print(f"\n=== Phase 4: Peak Chasing ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 40:
            t4 = min(rem, 250)
            print(f"  Time allocated: {t4:.0f}s")
            a_opt, _ = peak_chasing_optimize(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "peak_chase")

    # === Phase 5: LP-FW Refinement ===
    print(f"\n=== Phase 5: LP-FW Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t5 = min(rem, 180)
            print(f"  Time allocated: {t5:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 6: Final Adam Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t6 = min(rem, 80)
            a_opt, _ = adam_smooth_optimize(
                best_seq, time_limit=t6,
                tau_init=0.015 * fast_convolve(best_seq).max(),
                lr_init=0.001, verbose=False)
            try_seq(a_opt, "final_adam")

    # === Phase 7: Coordinate Descent Polish ===
    print(f"\n=== Phase 7: Coordinate Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 25
        if rem > 15:
            t7 = min(rem, 60)
            a_opt, _ = coordinate_polish(best_seq, time_limit=t7, verbose=False)
            try_seq(a_opt, "coord_polish")

    # === Phase 8: One more NMD from best ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t8 = min(rem, 45)
            a_opt, _ = nesterov_mirror_descent(
                best_seq, time_limit=t8,
                lr_init=0.005, verbose=False)
            try_seq(a_opt, "final_nmd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
