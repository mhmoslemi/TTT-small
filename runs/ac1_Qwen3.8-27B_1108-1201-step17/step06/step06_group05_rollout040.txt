Actually, one more important insight: in the LP-FW method, the LP direction gives us a point g on the simplex. The FW step is f_new = (1-alpha)*f + alpha*g. The optimal alpha can be found by line search. But a better approach (accelerated FW) uses a "moreau envelope" step or a "Greedy FW" (GFW) step where we only update the coordinates that change the most.

**Greedy Frank-Wolfe (GFW)**: Instead of taking a step towards the full LP solution g, we identify the single coordinate j that has the largest gap (c_j - <g, c> where c is the gradient) and take a step towards the unit vector e_j. This is much faster than a full LP solve and often works well in practice.

def compute_subgradient_fft(f, c, top_k=None):

    Compute subgradient of max(c) where c = f*f, using FFT.
    If top_k is specified, only use top-k active constraints.
    Returns subgradient of length n.

    c_len = len(c)
    
    if top_k is not None and top_k < c_len:
        idx = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros(c_len)
        w[idx] = 1.0 / top_k

def subgradient_descent(f0, time_limit, verbose=False):
    """Fast projected subgradient descent with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    lr = 1.0 / np.sqrt(n)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        sg = compute_subgradient_fft(f, c)
        
        f_trial = f - lr * sg
        f_trial = project_simplex(f_trial)
        
        c_trial = fast_convolve(f_trial)
        obj_trial = c_trial.max()
        
        if obj_trial < best_obj - 1e-14:
            f = f_trial
            best_obj = obj_trial
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.01, 1.0 / np.sqrt(n) * 2)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 100:
                # Restart
                f = project_simplex(best_f + np.random.default_rng(it).standard_normal(n) * 0.005)
                lr = 1.0 / np.sqrt(n)
                no_improve = 0
        
        if verbose and it % 200 == 0:
            print(f"    SGD iter {it}: obj={c.max():.10f}, best={best_obj:.10f}, lr={lr:.4f}")
    
    return best_f, best_obj

def make_raised_cosine(n):
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_optimal_init(n):
    """
    Create initial sequence based on flat-convolution theory.
    The ideal f has |F(omega)|^2 = constant in a band.
    For non-negative f, the best approximation is a raised cosine or
    a specific polynomial shape.
    """
    # Use a power of raised cosine (between triangle and uniform)
    x = np.linspace(-1, 1, n)
    # Try different powers
    best_a = None
    best_obj = float('inf')
    for power in [2, 3, 4, 5]:
        a = (0.5 * (1 + np.cos(np.pi * x))) ** power
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a /= s
            c = fast_convolve(a)
            obj = 2 * n * c.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a
    return best_a if best_a is not None else make_raised_cosine(n)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def coordinate_polish(f0, time_limit):
    """Coordinate descent for final refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj
