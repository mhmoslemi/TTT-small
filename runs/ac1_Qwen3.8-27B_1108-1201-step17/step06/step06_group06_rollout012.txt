def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Diverse initialization with evaluation
2. Curriculum-based smooth optimization with adaptive temperature and hard top-k subgradient
3. Minimax game with exact QP inner loop (CVXOPT) exploiting Hankel structure
4. LP-Frank-Wolfe refinement
5. Multi-length exploration and final polish
Key innovations: Hard top-k subgradient for sharper descent, explicit QP solvers
for the minimax inner loop, and a curriculum temperature schedule that avoids
premature convergence to individual peaks.

import numpy as np
from scipy.optimize import linprog
import time
import cvxopt

cvxopt.solvers.options['show_progress'] = False


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_grad(f, conv, tau):
    """Compute gradient of sum softmax(conv/tau) * conv via FFT cross-correlation."""
    n = len(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # grad[j] = 2 * sum_m w[m] * f[m-j] (where 0<=m-j<n)
    # This is cross-correlation: grad[j] = 2 * sum_i w[i+j] * f[i]
    # Compute via FFT: cross_corr = ifft(FFT(w_padded) * conj(FFT(f_padded)))
    # Need zero-padding for linear cross-correlation
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w, fft_size)
    # cross_corr[j] = sum_i w[i+j] * f[i] = ifft(conj(FFT(w)) * FFT(f))[j]
    cc = np.fft.irfft(np.conj(W) * F, fft_size)
    grad = 2.0 * cc[:n]
    return grad, w


def hard_topk_grad(f, conv, k):
    """Compute subgradient of top-k mean of conv. O(k*n) direct computation."""
    n = len(f)
    # Find top k indices
    if k >= len(conv):
        active = np.arange(len(conv))
    else:
        active = np.argsort(conv)[-k:]
    grad = np.zeros(n)
    for m in active:
        m = int(m)
        # grad[j] += 2*f[m-j] for j where 0 <= m-j < n
        j_start = max(0, m - n + 1)
        j_end = min(n, m + 1)
        for j in range(j_start, j_end):
            grad[j] += 2.0 * f[m - j]
    grad /= k
    return grad, active


def curriculum_optimize(f0, time_limit, verbose=False):
    """
    Curriculum-based optimization:
    - Start with high temperature (smooth), gradually decrease
    - Use Adam for smooth phase, switch to hard top-k subgradient near the end
    - Adaptive learning rate with restarts
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    tau = 0.1 * M
    best_obj = M
    best_f = f.copy()

    lr = 0.003
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    no_improve = 0
    k_hard = 5  # switch to hard top-k when tau is low enough

    t0 = time.monotonic()
    use_hard = False

    while time.monotonic() - t0 < time_limit:
        t += 1
        rem = time_limit - (time.monotonic() - t0)

        # Determine if we should use hard or soft gradient
        use_hard = tau < 0.02 * best_obj

        if use_hard:
            grad, active = hard_topk_grad(f, conv, k_hard)
        else:
            grad, _ = soft_grad(f, conv, tau)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        conv = conv_new

        # Adaptive temperature decay
        decay = 0.9995 if not use_hard else 0.999
        tau *= decay
        if tau < 0.005 * best_obj:
            tau = 0.005 * best_obj

        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.9
        elif no_improve > 200:
            # Restart from best with perturbation
            rng = np.random.default_rng(t * 13 + 7)
            noise = 0.003 + 0.002 * min(no_improve / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            conv = fast_convolve(f)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = 0.003
            tau = 0.05 * conv.max()

        if verbose and t % 200 == 0:
            print(f"    Curriculum iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, lr={lr:.5f}, hard={use_hard}")

    return best_f, 2 * n * best_obj


def build_hankel(p):
    """Build the Hankel matrix Q with Q[i,j] = p[i+j]."""
    n = len(p) // 2 + (len(p) % 2)  # p has length 2n-1
    n = (len(p) + 1) // 2
    Q = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            Q[i, j] = p[i + j]
    return Q


def build_hankel_fast(p):
    """Build Hankel matrix Q[i,j] = p[i+j] efficiently using row construction."""
    n = (len(p) + 1) // 2
    Q = np.zeros((n, n))
    for i in range(n):
        Q[i, :] = p[i:i + n]
    return Q


def solve_qp_inner(p, n, max_time=5.0):
    """
    Solve min a^T Q(p) a / 2 s.t. sum(a)=1, a>=0.
    Uses CVXOPT if Q(p) is PSD, otherwise projected gradient.
    Returns the optimal a.
    """
    Q = build_hankel_fast(p)
    Q = (Q + Q.T) / 2.0  # Ensure symmetry (should already be symmetric)

    # Check PSD via Cholesky
    is_psd = False
    try:
        np.linalg.cholesky(Q + 1e-10 * np.eye(n))
        is_psd = True
    except np.linalg.LinAlgError:
        is_psd = False

    if is_psd:
        # Solver QP: min 0.5 * a^T P a + q^T a s.t. G a <= h, A a = b
        P = cvxopt.matrix(Q)
        q = cvxopt.matrix(np.zeros(n))
        # a >= 0: -a <= 0
        G = cvxopt.matrix(-np.eye(n))
        h = cvxopt.matrix(np.zeros(n))
        # sum(a) = 1
        A = cvxopt.matrix(np.ones((1, n)))
        b = cvxopt.matrix(np.ones(1))
        try:
            sol = cvxopt.solvers.qpsol(P, q, G, h, A, b)
            if sol['status'] == 'optimal':
                a = np.array(sol['x']).flatten()
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-10:
                    a /= ss
                    return a
        except Exception:
            pass

    # Fallback: projected gradient with BB step
    a = np.ones(n) / n
    best_a = a.copy()
    best_val = a @ Q @ a
    t0 = time.monotonic()
    lr = 1.0 / (2.0 * np.linalg.norm(Q, ord=2) + 1e-10)

    for it in range(200):
        if time.monotonic() - t0 > max_time:
            break
        grad = Q @ a
        a_new = project_simplex(a - lr * grad)
        val = a_new @ Q @ a_new
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()
        
        # BB step size update
        if it > 0:
            s_vec = a_new - a
            y_vec = Q @ a_new - grad
            ss = s_vec @ s_vec
            ys = s_vec @ y_vec
            if ss > 1e-20:
                if ys > 1e-15:
                    lr = ss / ys
                else:
                    lr = ss / (y_vec @ y_vec + 1e-15)
                lr = min(lr, 1.0 / (0.1 * np.linalg.norm(Q, ord=2) + 1e-10))
        a = a_new

    return best_a


def minimax_qp(f0, time_limit, verbose=False):
    """
    Minimax game: alternate between inner QP (min a^T Q(p) a) and outer MW update.
    Uses exact QP solver when possible.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    p = np.ones(2 * n - 1) / (2 * n - 1)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    lr_p = 0.2
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        # Inner: solve QP
        a_opt = solve_qp_inner(p, n, max_time=min(3.0, rem * 0.5))
        if a_opt is not None:
            f = a_opt

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    Minimax-QP iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1

        if no_improve > 30:
            # Perturb p
            rng = np.random.default_rng(it * 7 + 3)
            p = p + rng.standard_normal(2 * n - 1) * 0.01
            p = np.maximum(p, 0)
            p /= p.sum()
            no_improve = 0

        # Outer: multiplicative weight update
        eta = 0.05 + 0.03 * (1.0 / (1.0 + it * 0.01))
        scaled = conv / (eta + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        p = (1 - lr_p) * p + lr_p * w
        p /= p.sum()

        # Decay lr_p
        if it % 50 == 0:
            lr_p = max(0.05, lr_p * 0.95)

    return best_f, 2 * n * best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp, g >= 0
        g = _solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 10:
                m_exp = min(m_exp * 1.05, 10.0)
            if no_improve > 50:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0
            m_exp = 1.0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 10.0)

        if no_improve > 80:
            break

    return best_f, 2 * n * best_obj


def _solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def peak_cd(f0, time_limit, verbose=False):

            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max

            if j_max == j_other:
                old_val = f[j_max]
                if old_val < 1e-10:
                    continue
                for factor in [0.5, 0.7, 0.85]:
                    trial = f.copy()
                    trial[j_max] *= factor
                    diff = old_val - trial[j_max]
                    contrib_mask = np.ones(n, dtype=bool)
                    for pk2 in peak_indices[:2]:
                        js2 = max(0, int(pk2) - n + 1)
                        je2 = min(n - 1, int(pk2))
                        contrib_mask[js2:je2 + 1] = False
                    if contrib_mask.sum() > 0:
                        cm_sum = f[contrib_mask].sum()
                        if cm_sum > 1e-15:
                            trial[contrib_mask] += diff * f[contrib_mask] / cm_sum
                    trial = np.maximum(trial, 0)
                    ss = trial.sum()
                    if ss > 1e-15:
                        trial /= ss
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
            else:
                old_jm = f[j_max]
                for scale in [0.6, 0.8]:
                    trial = f.copy()
                    trial[j_max] = old_jm * scale
                    diff = old_jm - trial[j_max]
                    mask = np.ones(n, dtype=bool)
                    mask[j_max] = False
                    mask[j_other] = False
                    if mask.sum() > 0:
                        ms = f[mask].sum()
                        if ms > 1e-15:
                            trial[mask] += diff * f[mask] / ms
                    trial = np.maximum(trial, 0)
                    ss = trial.sum()
                    if ss > 1e-15:
                        trial /= ss
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0
        else:
            no_improve = 0

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_bell(n, width_frac=0.5, asym=0.0):
    """Bell-shaped initialization with optional asymmetry."""
    x = np.arange(n)
    center = n * (0.5 + asym)
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x - center) / half, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-15:
        a /= ss
    return a


def make_gaussian(n, center_frac=0.5, sigma_frac=0.25):
    """Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-15:
        a /= ss
    return a


def make_exp(n, rate=0.5):
    """Exponential decay."""
    a = np.exp(-np.arange(n) / (n * rate))
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining curriculum Adam, minimax QP,
    LP-FW, and peak CD with diverse initialization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    candidates = []

    # Try to use parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            sc = try_seq(parent, "parent")
            candidates.append((sc, parent))
    except Exception:
        pass

    # Diverse initializations
    inits = []
    for n_try in [512, 768, 1024]:
        inits.append(make_bell(n_try, 0.5, 0.0))
        inits.append(make_bell(n_try, 0.4, 0.05))
        inits.append(make_gaussian(n_try, 0.5, 0.25))
        inits.append(make_gaussian(n_try, 0.45, 0.2))
        inits.append(make_exp(n_try, 0.5))
        inits.append(np.ones(n_try) / n_try)
        # Bimodal
        b = make_gaussian(n_try, 0.3, 0.15)
        b += 0.5 * make_gaussian(n_try, 0.7, 0.15)
        b /= b.sum()
        inits.append(b)

    for i, a_init in enumerate(inits):
        if time.monotonic() > deadline - 1500:
            break
        sc = try_seq(a_init, f"init_{i}")
        candidates.append((sc, a_init.copy()))

    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done in {time.monotonic()-t_start:.1f}s, best={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 2: Curriculum optimization on top candidates ===
    print(f"\n=== Phase 2: Curriculum Optimization ===")
    top_candidates = [c[1] for c in candidates[:3]]
    if parent is not None and parent not in top_candidates:
        top_candidates.insert(0, parent)

    for i, f_start in enumerate(top_candidates[:3]):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.25, 250)
        print(f"  Candidate {i} (n={len(f_start)}), time={t_alloc:.0f}s")
        a_opt, obj = curriculum_optimize(f_start, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"curriculum_{i}")

    # === Phase 3: Minimax QP ===
    print(f"\n=== Phase 3: Minimax QP ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t3 = min(rem, 300)
            print(f"  Time allocated: {t3:.0f}s")
            a_opt, obj = minimax_qp(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "minimax_qp")

            # Try from a perturbed start
            if time.monotonic() < deadline - 250:
                rem2 = deadline - time.monotonic() - 200
                t3b = min(rem2, 100)
                perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.005)
                a_opt2, obj2 = minimax_qp(perturbed, time_limit=t3b, verbose=False)
                try_seq(a_opt2, "minimax_qp_pert")

    # === Phase 4: LP-FW ===
    print(f"\n=== Phase 4: LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t4 = min(rem, 300)
            print(f"  Time allocated: {t4:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "lpfw")

            if time.monotonic() < deadline - 150:
                rem2 = deadline - time.monotonic() - 100
                t4b = min(rem2, 100)
                perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.003)
                a_opt2, obj2 = lp_fw_optimize(perturbed, time_limit=t4b, verbose=False)
                try_seq(a_opt2, "lpfw_pert")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 1280, 640, 960]:
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.001:
                t5 = min(rem * 0.3, 80)
                if t5 > 20:
                    a_opt, obj = curriculum_optimize(f_interp, time_limit=t5, lr_init=0.003)
                    try_seq(a_opt, f"curriculum_n{target_n}")
                    rem2 = deadline - time.monotonic() - 60
                    if rem2 > 30:
                        a_opt2, obj2 = lp_fw_optimize(best_seq, time_limit=min(rem2, 60), verbose=False)
                        try_seq(a_opt2, f"lpfw_n{target_n}")

    # === Phase 6: Peak CD ===
    print(f"\n=== Phase 6: Peak CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t6 = min(rem, 100)
            a_opt, obj = peak_cd(best_seq, time_limit=t6, verbose=True)
            try_seq(a_opt, "peak_cd")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t7 = min(rem, 45)
            a_opt, obj = curriculum_optimize(
                best_seq, time_limit=t7, verbose=False)
            try_seq(a_opt, "curriculum_final")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t8 = min(rem, 30)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t8, verbose=False)
            try_seq(a_opt, "lpfw_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
