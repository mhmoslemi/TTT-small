Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Diverse structured initialization (Chebyshev, Gaussian, polynomial, multimodal)
2. p-norm smoothed FISTA for fast global convergence
3. L-BFGS on p-norm smoothed objective for fast local convergence
4. LP-based Frank-Wolfe for high-precision polishing
5. Active-set refinement focusing on critical peaks
6. Multi-length exploration

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def pnorm_objective(f, p):
    """Compute (1/p) * sum_k (f*f)_k^p"""
    c = fast_convolve(f)
    return np.sum(c**p) / p

def pnorm_gradient(f, p):
    """Compute gradient of (1/p) * sum_k (f*f)_k^p with respect to f."""
    n = len(f)
    c = fast_convolve(f)
    # d(phi)/d(f_j) = sum_k c_k^{p-1} * 2 * f_{k-j} (where valid)
    # This is 2 * correlate(c^{p-1}, f)
    w = c ** (p - 1) if p > 1 else np.ones_like(c)
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    # g_j = 2 * sum_i f_i * w_{j+i} = 2 * conv[2n-2-j]... 
    # Let me re-derive: conv[k] = sum_i f[i] * w_rev[k-i]
    # w_rev[m] = w[len(w)-1-m] = w[2n-2-m]
    # So conv[k] = sum_i f[i] * w[2n-2-(k-i)] = sum_i f[i] * w[2n-2-k+i]
    # We want g_j/2 = sum_i f[i] * w[j+i]
    # So 2n-2-k+i = j+i => 2n-2-k = j => k = 2n-2-j
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g, c

def pnorm_fista(f0, time_limit, p_start=4, p_end=64, n_phases=6):
    """FISTA (accelerated proximal gradient) on p-norm smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()
    
    t0 = time.monotonic()
    p_values = np.geomspace(p_start, p_end, n_phases)
    
    for p in p_values:
        if time.monotonic() - t0 > time_limit * 0.9:
            break
        
        # Compute Lipschitz constant (approximate)
        c_curr = fast_convolve(f)
        max_c = c_curr.max()
        # L ≈ 2 * p * max_c^{p-1} * max(f) * n (rough bound)
        L = 2.0 * p * max(max_c, 1e-10)**(p-1) * np.max(f) * n
        L = max(L, 1e-10)
        step = 1.0 / L
        
        # FISTA

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def pnorm_obj_and_grad(f, p):
    """Compute p-norm smoothed objective and gradient."""
    n = len(f)
    c = fast_convolve(f)
    obj = np.sum(c**p) / p
    if p > 1:
        w = c ** (p - 1)
    else:
        w = np.ones_like(c)
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return obj, g, c

def solve_lp_direction(f, M, n, active_indices=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    if active_indices is not None:
        num_constraints = len(active_indices)
        A_ub = np.zeros((num_constraints, n))
        b_ub = np.full(num_constraints, M)
        for idx, k in enumerate(active_indices):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            A_ub[idx, js:je+1] = f[js:je+1]
    else:
        num_constraints = 2 * n - 1
        A_ub = np.zeros((num_constraints, n))
        b_ub = np.full(num_constraints, M)
        for j in range(n):
            A_ub[j:j + n, j] = f
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def coordinate_polish(f0, time_limit, verbose=False):
    """Coordinate descent with bisection for optimal value."""
    f = np.maximum(f0.astype(np.float64), 0.0)

        c = fast_convolve(f)

            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try log-spaced multipliers
            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                new_val = old_val * mult

                denom = 1.0 - old_val
                if denom < 1e-12:

                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)

                ss = f_trial.sum()
                if ss < 1e-15:

                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth

        it = 0
        no_improve = 0
        lr = 0.5 / (np.linalg.norm(f) * np.max(fast_convolve(f)) * n ** 0.5 + 1e-10)
        
        stage_start = time.monotonic()
        stage_time = time_limit / len(beta_schedule) * 0.8
        
        while time.monotonic() - stage_start < stage_time:
            it += 1
            c = fast_convolve(f)
            scaled = beta * c
            scaled -= scaled.max()
            exp_scaled = np.exp(scaled)
            w = exp_scaled / exp_scaled.sum()
            
            w_rev = w[::-1].copy()
            conv_len = n + len(w_rev) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            F = np.fft.rfft(f, fft_size)
            W = np.fft.rfft(w_rev, fft_size)
            conv = np.fft.irfft(F * W, fft_size)[:conv_len]
            indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
            g = 2.0 * conv[indices]
            
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            
            momentum = mu * momentum - lr * g
            f_trial = project_simplex(f + momentum)
            
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()
            
            if obj_trial <= c.max() + 1e-10:
                f = f_trial
                if obj_trial < best_obj - 1e-14:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    lr = min(lr * 1.03, 1.0 / (g_norm * 0.3 + 1e-10))
                else:
                    no_improve += 1
            else:
                no_improve += 1
                lr *= 0.6
                momentum *= 0.5
            
            if no_improve > 60:
                rng = np.random.default_rng(it * 13 + int(beta) * 7)
                noise_scale = 0.003 / np.sqrt(beta / 4.0)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                momentum = np.zeros(n)
                lr = 0.3 / (n ** 0.5 + 1e-10)
                no_improve = 0
    
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining diverse initialization, p-norm FISTA,
    L-BFGS, active-set LP-FW, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = make_gaussian(768, 0.3)
    
    n_parent = len(parent)
    
    # === Phase 1: Quick screening of initializations ===
    print("=== Phase 1: Initialization Screening ===")
    t1 = time.monotonic()
    
    init_candidates = []
    init_candidates.append(("parent", parent))
    init_candidates.append(("gauss_025", make_gaussian(n_parent, 0.25)))
    init_candidates.append(("gauss_035", make_gaussian(n_parent, 0.35)))
    init_candidates.append(("gauss_045", make_gaussian(n_parent, 0.45)))
    init_candidates.append(("bell_4", make_bell(n_parent, 0.6, 4)))
    init_candidates.append(("bell_6", make_bell(n_parent, 0.7, 6)))
    init_candidates.append(("bell_8", make_bell(n_parent, 0.75, 8)))
    init_candidates.append(("raised_cos", make_raised_cosine(n_parent)))
    init_candidates.append(("cheb", make_chebyshev_like(n_parent, 0.5)))
    init_candidates.append(("cheb2", make_chebyshev_like(n_parent, 0.8)))
    init_candidates.append(("multimodal3", make_multimodal(n_parent, 3)))
    init_candidates.append(("poly", make_polynomial(n_parent)))
    
    best_init = None
    best_init_obj = float('inf')
    for name, a in init_candidates:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        c = fast_convolve(a)
        obj = 2 * len(a) * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a
        sc = try_seq(a, f"init_{name}")
    
    # Random perturbations
    for trial in range(8):
        noise = rng.standard_normal(n_parent) * 0.02
        a_pert = np.maximum(parent + noise, 0)
        a_pert = a_pert / a_pert.sum()
        c = fast_convolve(a_pert)
        obj = 2 * n_parent * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_pert
        try_seq(a_pert, f"pert{trial}")
    
    print(f"  Best init: obj={best_init_obj:.8f} ({time.monotonic()-t1:.1f}s)")
    
    # === Phase 2: p-norm FISTA ===
    print(f"\n=== Phase 2: p-norm FISTA (n={n_parent}) ===")
    t2 = time.monotonic()
    rem = deadline - time.monotonic() - 800
    t_alloc = max(30, min(rem, 350))
    print(f"  Allocation: {t_alloc:.0f}s")
    a_opt, obj = pnorm_fista(best_init, time_limit=t_alloc, verbose=True)
    try_seq(a_opt, "fnorm")
    
    # Also try from parent if different
    if not np.allclose(best_init, parent, atol=1e-6):
        rem = deadline - time.monotonic() - 600
        if rem > 80:
            t_alloc2 = min(rem * 0.3, 150)
            a_opt2, obj2 = pnorm_fista(parent, time_limit=t_alloc2)
            try_seq(a_opt2, "fnorm_parent")
    
    print(f"  Phase 2 done ({time.monotonic()-t2:.1f}s)")
    
    # === Phase 3: L-BFGS with p-norm ===
    print(f"\n=== Phase 3: L-BFGS p-norm ===")
    t3 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t_alloc = min(rem * 0.5, 250)
            print(f"  Allocation: {t_alloc:.0f}s")
            a_opt, obj = lbfgs_optimize(best_seq, time_limit=t_alloc, p=32, verbose=True)
            try_seq(a_opt, "lbfgs_p32")
            
            # Also try higher p
            rem2 = deadline - time.monotonic() - 400
            if rem2 > 40:
                a_opt2, obj2 = lbfgs_optimize(best_seq, time_limit=min(rem2*0.4, 100), p=64, verbose=True)
                try_seq(a_opt2, "lbfgs_p64")
    print(f"  Phase 3 done ({time.monotonic()-t3:.1f}s)")
    
    # === Phase 4: LP-FW with active set ===
    print(f"\n=== Phase 4: LP-FW Active Set ===")
    t4 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_alloc = min(rem * 0.6, 350)
            print(f"  Allocation: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw")
    print(f"  Phase 4 done ({time.monotonic()-t4:.1f}s)")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    t5 = time.monotonic()
    base = best_seq if best_seq is not None else best_init
    target_lengths = [1024, 640, 896, 512]
    
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 100:
            break
        
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")
        
        # Quick p-norm FISTA
        t_gd = min(rem * 0.25, 80)
        if t_gd > 30:
            a_opt, obj = pnorm_fista(f_interp, time_limit=t_gd)
            try_seq(a_opt, f"fnorm_n{target_n}")
        
        # LP-FW
        rem2 = deadline - time.monotonic() - 150
        if rem2 > 50:
            t_lpfw = min(rem2 * 0.4, 80)
            a_opt2, obj2 = lp_fw_optimize(a_opt, time_limit=t_lpfw)
            try_seq(a_opt2, f"lpfw_n{target_n}")
    
    print(f"  Phase 5 done ({time.monotonic()-t5:.1f}s)")
    
    # === Phase 6: Gradient descent with LSE smoothing ===
    print(f"\n=== Phase 6: LSE Gradient Descent ===")
    t6 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 60:
            t_alloc = min(rem * 0.5, 150)
            a_opt, obj = gradient_descent_smoothed(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lse_gd")
    print(f"  Phase 6 done ({time.monotonic()-t6:.1f}s)")
    
    # === Phase 7: Deep LP-FW with diverse restarts ===
    print(f"\n=== Phase 7: Deep LP-FW ===")
    t7 = time.monotonic()
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 50:
                break
            t_alloc = min(rem / (3 - p), 100)
            
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.001
                f_init = project_simplex(best_seq + noise)
            else:
                rc = make_raised_cosine(n_best)
                f_init = 0.85 * best_seq + 0.15 * rc
                f_init = f_init / f_init.sum()
            
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(p == 0))
            try_seq(a_opt, f"deep_lpfw_{p}")
    print(f"  Phase 7 done ({time.monotonic()-t7:.1f}s)")
    
    # === Phase 8: Final coordinate polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    t8 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 60)
            a_opt, obj = coordinate_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "coord_final")
        
        # Final LP-FW squeeze
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem, 30)
            a_opt, obj = lp_fw_optimize(a_opt, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")
    print(f"  Phase 8 done ({time.monotonic()-t8:.1f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Fallback to uniform
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
