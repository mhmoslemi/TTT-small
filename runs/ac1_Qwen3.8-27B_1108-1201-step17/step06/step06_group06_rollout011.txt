    grad = (2.0 / tau) * h
    return grad, c, smooth_val


def nesterov_adam_optimize(f0, time_limit, tau_init=None, lr_init=0.003,
                           tau_decay=0.9995, tau_min=1e-6, verbose=False):
    """
    Nesterov-accelerated Adam on the log-sum-exp smoothed objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.06 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    f_prev = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Nesterov extrapolation
        y = f + 0.5 * (f - f_prev) if t > 1 else f.copy()

        grad_y, conv_y, _ = smooth_gradient(y, tau)
        M_y = conv_y.max()

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * grad_y * grad_y
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_prev = f.copy()
        f_new = y - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive temperature
        tau = max(tau * tau_decay, tau_min)

        # Adaptive learning rate with restarts
        if no_improve > 50:
            lr *= 0.9
        if no_improve > 200:
            # Restart from best with perturbation
            rng = np.random.default_rng(t * 31 + 7)
            noise_scale = 0.003 + 0.005 * min(no_improve / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            f_prev = f.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * (0.5 + 0.5 * np.random.default_rng(t).random())
            tau = max(0.04 * fast_convolve(f).max(), tau_min)
            if verbose:
                print(f"      Restart at t={t}: obj={2*n*best_obj:.8f}")

        if verbose and t % 200 == 0:
            print(f"      t={t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, lr={lr:.5f}")

    return best_f, 2 * n * best_obj


def subgradient_nesterov(f0, time_limit, verbose=False):
    """
    Pure subgradient method with Nesterov momentum on the unsmoothed max.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    f_prev = f.copy()
    t0 = time.monotonic()
    t = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Nesterov
        momentum = 0.3 * (1.0 / (1.0 + 0.01 * t))
        if t == 1:
            y = f.copy()
        else:
            beta_t = (1.0 - np.sqrt(1 + 4 * (t - 1) ** 2)) / (2 * (t - 1)) + 0.5
            beta_t = 0.3 + 0.7 / (1 + 0.05 * t)
            y = f + beta_t * (f - f_prev)

        conv_y = fast_convolve(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient at y: g[m] = 2*y[k_star - m] (approx, ignoring diagonal)
        # with bounds: only where 0 <= k_star - m <= n-1
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for m in range(js, je + 1):
            val = 2.0 * y[k_star - m]
            if 2 * m == k_star:
                val = y[m]
            g[m] = val

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Step size: diminishing
        step = 2.0 / (np.linalg.norm(f) * g_norm * (1 + 0.1 * t))
        step = min(step, 0.05)

        f_prev = f.copy()
        f_new = project_simplex(y - step * g)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if no_improve > 200:
            rng = np.random.default_rng(t * 13 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            f_prev = f.copy()
            no_improve = 0

        if verbose and t % 100 == 0:
            print(f"      SubGrad t={t}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def peak_cd_refine(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent:
    At each step, identify the top peaks, find the dominant contributing pairs,
    and redistribute mass to reduce those peaks while maintaining sum=1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve_rounds = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get top peaks (up to 10)
        n_peaks = min(10, 2 * n - 1)
        peak_indices = np.argsort(conv)[-n_peaks:][::-1]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            top_contrib_idx = idx[np.argsort(products)][-3:]

            for j_a in top_contrib_idx:
                j_b = pk - j_a
                if j_a < 0 or j_a >= n or j_b < 0 or j_b >= n:
                    continue
                if f[j_a] < 1e-10 or f[j_b] < 1e-10:
                    continue

                # Try reducing the product by scaling down the larger element
                # and redistributing to elements that don't contribute to peaks
                mass_to_move = 0.0
                if f[j_a] >= f[j_b]:
                    for frac in [0.05, 0.02, 0.01, 0.005]:
                        mass_to_move = f[j_a] * frac
                        break
                else:
                    for frac in [0.05, 0.02, 0.01, 0.005]:
                        mass_to_move = f[j_b] * frac
                        break

                if mass_to_move < 1e-12:
                    continue

                # Find "safe" elements to redistribute to
                unsafe = np.zeros(n, dtype=bool)
                for pk2 in peak_indices[:5]:
                    js2 = max(0, int(pk2) - n + 1)
                    je2 = min(n - 1, int(pk2))
                    unsafe[js2:je2 + 1] = True
                unsafe[j_a] = True
                if j_b >= 0 and j_b < n:
                    unsafe[j_b] = True

                safe_mask = ~unsafe & (f > 1e-12)
                if safe_mask.sum() < 3:
                    safe_mask = ~unsafe
                    if safe_mask.sum() < 1:
                        continue

                # Try different reduction strategies
                for red_frac in [0.3, 0.5, 0.7]:
                    trial = f.copy()
                    if f[j_a] >= f[j_b]:
                        reduce_amt = f[j_a] * red_frac
                        trial[j_a] -= reduce_amt
                    else:
                        reduce_amt = f[j_b] * red_frac
                        trial[j_b] -= reduce_amt
                    trial = np.maximum(trial, 0)

                    # Redistribute proportionally to safe elements
                    safe_vals = trial[safe_mask]
                    if safe_vals.sum() < 1e-15:
                        safe_vals = f[safe_mask].copy()
                        if safe_vals.sum() < 1e-15:
                            continue
                    trial[safe_mask] += reduce_amt * safe_vals / safe_vals.sum()
                    trial = np.maximum(trial, 0)
                    trial_sum = trial.sum()
                    if trial_sum < 1e-15:
                        continue
                    trial /= trial_sum

                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve_rounds += 1
            if no_improve_rounds > 5:
                # Perturb
                rng = np.random.default_rng(it * 7 + 1)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve_rounds = 0
        else:
            no_improve_rounds = 0

        if verbose and it % 10 == 0:
            print(f"      PeakCD it={it}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def dual_based_direction(f, n):
    """
    Compute a search direction using the dual structure.
    The subgradient at the peak is ~2*flip_shift(a). We find a "composite"
    direction from multiple peaks weighted by their excess.
    """
    conv = fast_convolve(f)
    M = conv.max()
    # Weight peaks by their excess above the mean
    target = conv.mean()
    excess = np.maximum(conv - target, 0)
    if excess.sum() < 1e-15:
        return 2.0 * f[::-1], conv

    w = excess / excess.sum()

    # Composite gradient: sum_k w[k] * 2*f[k-j]
    # = 2 * PF(f, w) where PF is the "peak filter"
    n_f = len(f)
    f_rev = f[::-1]
    total_len = n_f + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    F_rev = np.fft.rfft(f_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(F_rev * W, fft_size)
    h = conv_result[n_f - 1: 2 * n_f - 1]

    return 2.0 * h, conv


def dual_descent(f0, time_limit, verbose=False):
    """
    Descent direction from the dual (composite subgradient from all peaks).
    Uses backtracking line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        direction, conv = dual_based_direction(f, n)
        M = conv.max()

        d_norm = np.linalg.norm(direction)
        if d_norm < 1e-15:
            break

        # Backtracking line search
        step = 1.0 / (d_norm * np.linalg.norm(f) + 1e-8)
        step = min(step, 0.5)
        improved = False

        for _ in range(20):
            f_new = project_simplex(f - step * direction)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                f = f_new
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_f = f.copy()
                improved = True
                no_improve = 0
                break
            step *= 0.5
            if step < 1e-10:
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 19 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0

        if verbose and it % 20 == 0:
            print(f"      DualDescent it={it}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        a = np.maximum(1 - np.abs(x) / (n * 0.45), 0)
    elif shape_type
