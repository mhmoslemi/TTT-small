    # So grad[i] = sum_l a[l]*(w[i+l] + w[i-l])
    
    # This is like a "two-sided" weighting. We can compute it with FFT:
    # Define w2[m] = w[m] for m in [0, 2n-1], and extend by zero.
    # grad[i] = sum_l a[l] * (w2[i+l] + w2[i-l])
    # = sum_l a[l]*w2[i+l] + sum_l a[l]*w2[i-l]
    # = (a * w2)_cross[+i] + (a * w2)_cross[-i]
    # where cross-correlation is defined as (f*g)_cross[m] = sum_l f[l]*g[l+m]
    
    # Using FFT: the cross-correlation (f*g)_cross[m] = ifft(conj(FFT(f)) * FFT(g))[m]
    # (with appropriate circular vs linear considerations)
    
    # For linear cross-correlation with zero-padding:
    # (f*g)_cross[m] for m in [-n+1, n-1] can be computed as:
    # ifft(conj(fft(f, N)) * fft(g, N))[m + (N-1)//2] ... this is getting messy.
    
    # Simplest approach: just compute grad = 2 * first n elements of conv(w, a)
    # Let me verify with a small example.
    # a = [a0, a1] (n=2), w = [w0, w1, w2] (length 3 = 2n-1)
    # c = conv(a,a) = [a0^2, 2*a0*a1, a1^2]
    # sum w[k]*c[k] = w0*a0^2 + w1*2*a0*a1 + w2*a1^2
    # d/da0 = 2*w0*a0 + 2*w1*a1
    # d/da1 = 2*w1*a0 + 2*w2*a1
    # 
    # Now conv(w, a) = [w0*a0, w0*a1+w1*a0, w1*a1+w2*a0, w2*a1] (length 4)
    # First n=2 elements: [w0*a0, w0*a1+w1*a0]
    # 2 * that = [2*w0*a0, 2*(w0*a1+w1*a0)]
    # 
    # But the actual gradient is [2*w0*a0 + 2*w1*a1, 2*w1*a0 + 2*w2*a1]
    # These don't match! So conv(w, a) is NOT the right approach.
    
    # The correct approach: grad = 2 * "auto-weighted correlation"
    # Actually, let me reconsider. The gradient of <w, conv(a,a)> w.r.t. a is:
    # It's 2 * T_a^T * w, where T_a is the convolution matrix (T_a * b = conv(a,b)).
    # T_a is a (2n-1) x n matrix with T_a[k,j] = a[k-j] if 0<=k-j<n else 0.
    # So (T_a^T * w)[i] = sum_k T_a[k,i]*w[k] = sum_k a[k-i]*w[k] (valid k)
    # = sum_{k=i}^{i+n-1} a[k-i]*w[k] = sum_{l=0}^{n-1} a[l]*w[l+i]
    
    # So grad[i] = 2 * sum_{l=0}^{n-1} a[l]*w[l+i]
    
    # This is the cross-correlation of a and w! Specifically:
    # grad[i] = 2 * (a ⋆ w)[i] where (a ⋆ w)[i] = sum_l a[l]*w[l+i]
    
    # We can compute this as: ifft(conj(fft(a, N)) * fft(w, N))[0:n] * 2
    # where N >= 2n-1.
    
    N = 1 << ((2 * n - 1) - 1).bit_length()
    A_fft = rfft(a, N)
    W_fft = rfft(w, N)
    grad_half = irfft(np.conj(A_fft) * W_fft, N)[:n]
    return 2.0 * grad_half


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def soft_max_obj_and_grad(z, T, n, time_limit_check=None, t0=None):
    """
    Compute the soft-max smoothed objective and its gradient in log-space.
    obj = (2*n/T) * log(sum exp(T*conv(a,a)))
    where a = softmax(z) (probability vector).
    """
    a = softmax_from_z(z)
    c = fast_convolve(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    log_sum_exp = max_Tc + np.log(exp_Tc.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    # Gradient w.r.t. a:
    w = exp_Tc / exp_Tc.sum()  # softmax weighting on convolution lags
    grad_a = cross_corr_grad(w, a)
    
    # Convert to gradient w.r.t. z (log-space):
    # a = softmax(z), so da_i/dz_j = a_i*(delta_ij - a_j)
    # grad_z_j = sum_i grad_a_i * da_i/dz_j = sum_i grad_a_i * a_i*(delta_ij - a_j)
    # = a_j * (sum_i grad_a_i * a_i - sum_i grad_a_i * a_i * a_j)
    # Wait: = a_j * (grad_a_j * a_j + sum_{i!=j} grad_a_i * a_i) - a_j * a_j * sum_i grad_a_i * a_i
    # Hmm, let me redo.
    # grad_z_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = a_j * grad_a_j - a_j * sum_i grad_a_i * a_i
    # = a_j * (grad_a_j - a * grad_a . a / a_j) ... no
    # = a_j * grad_a_j - a_j * (a @ grad_a)
    # = a_j * (grad_a_j - a @ grad_a)
    # Wait that's not right either.
    # grad_z_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = a_j * grad_a_j - a_j * sum_i grad_a_i * a_i
    # = a_j * (grad_a_j - dot(a, grad_a))
    # Hmm wait: sum_i grad_a_i * a_i * delta_ij = grad_a_j * a_j
    # sum_i grad_a_i * a_i * a_j = a_j * sum_i grad_a_i * a_i = a_j * dot(a, grad_a)
    # So grad_z_j = grad_a_j * a_j - a_j * dot(a, grad_a) = a_j * (grad_a_j - dot(a, grad_a))
    
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lbgfs_optimize(a0, T, time_limit, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        obj, _ = soft_max_obj_and_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        _, g = soft_max_obj_and_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        a_out = a0
    
    return a_out


def exact_subgradient_step(a, time_limit=None):
    """
    One step of subgradient descent using the exact peak of the auto-correlation.
    Returns the updated a.
    """
    n = len(a)
    c = fast_convolve(a)
    peak = c.max()
    
    # Find all lags achieving (near) the maximum
    threshold = peak * (1 - 1e-6)
    w = (c >= threshold).astype(np.float64)
    w = w / w.sum()
    
    # Subgradient direction (pointing toward decreasing the peak)
    grad = cross_corr_grad(w, a)
    
    return grad, peak


def subgradient_refine(a0, time_limit, lr_init=0.01, momentum=0.9, 
                        top_k=None, annealing=0.9995):
    """
    Subgradient method with momentum for fine-tuning the auto-correlation peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    step_count = 0
    
    if top_k is None:
        top_k = max(10, n // 50)
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        # Use top-k peaks for robustness
        k_actual = min(top_k, len(c))
        peaks = np.argpartition(c, -k_actual)[-k_actual:]
        pv = c[peaks]
        
        # Softmax weighting
        temp = 0.001 * (pv.max() - pv.min() + 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w_full = np.zeros(len(c))
            w_full[peaks[np.argmax(pv)]] = 1.0
        else:
            w_full = np.zeros(len(c))
            w_full[peaks] = w / ws
        
        grad = cross_corr_grad(w_full, a)
        
        v = momentum * v - lr * grad
        
        # Project onto simplex
        a_candidate = a + v
        a_new = simplex_project(a_candidate)
        a_new = a_new / a_new.sum()
        
        max_c = fast_convolve(a_new).max()
        score = 2.0 * n * max_c
        step_count += 1
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a  # keep old (don't accept)
            no_improve += 1
        
        lr *= annealing
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 5, 0.003)
            no_improve = 0
        
        if step_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f} step={step_count}")
    
    return best_a, best_score


def adaptive_temperature_optimize(a0, time_limit, T_start=1000, T_end=10, 
                                   T_decay=0.98, lbgfs_budget_frac=0.7):
    """
    Run L-BFGS with decreasing temperature (coarse to fine).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    T = T_start
    phase = 0
    while time.monotonic() - t0 < time_limit * lbgfs_budget_frac and T > T_end:
        phase += 1
        phase_time = min(50.0, (time_limit * lbgfs_budget_frac - (time.monotonic() - t0)) * 0.5)
        if phase_time < 5:
            break
        
        print(f"    AdaptiveT phase={phase} T={T:.0f} budget={phase_time:.0f}s")
        a_new = lbgfs_optimize(a, T, phase_time, maxiter=3000)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        
        T *= T_decay
    
    return best_a, best_score


def coordinated_peak_reduction(a0, time_limit):
    """
    Directly reduce the peak by adjusting the elements that contribute most to it.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        # Find the global peak
        peak_k = int(np.argmax(c))
        peak_val = c[peak_k]
        
        if peak_val <= best_score / (2.0 * n) + 1e-15:
            break
        
        # Find contributing elements
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Sort by contribution
        order = np.argsort(contribs)[::-1]
        top_count = min(5, len(order))
        
        improved = False
        for ci in range(top_count):
            i = idx[order[ci]]
            j = peak_k - i
            if i == j:
                # Diagonal: reducing a[i] reduces the peak quadratically
                for factor in [0.98, 0.95, 0.9, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    diff = a[i] - a_trial[i]
                    if diff < 1e-15:
                        continue
                    # Redistribute mass to other elements (preferably ones that don't contribute to the peak)
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    # Also mask out elements that contribute to the peak
                    for ci2 in range(top_count):
                        i2 = idx[order[ci2]]
                        j2 = peak_k - i2
                        mask[i2] = False
                        mask[j2] = False
                    if mask.sum() < 1:
                        mask[i] = True  # at least keep i valid
                        mask[i] = False
                        others = a[mask]
                    else:
                        others = a[mask]
                    if others.sum() < 1e-15:
                        continue
                    a_trial[mask] = others + diff * others / others.sum()
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Off-diagonal: reduce both a[i] and a[j]
                for f1 in [0.98, 0.95, 0.9]:
                    for f2 in [0.98, 0.95, 0.9]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            # Try a small random perturbation
            rng = np.random.default_rng(iteration)
            a_trial = a + rng.standard_normal(n) * 0.001
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                # Stuck, break
                break
        
        iteration += 1
        if iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakReduce {elapsed:.0f}s: score={best_score:.6f} iter={iteration}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def one_sided_init(n, alpha=1.0):
    """Create a one-sided triangular-like initialization."""
    x = np.linspace(0, 1, n)
    a = (x ** alpha)
    a = a / a.sum()
    return a


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase gradient-based optimization:
    1. Initialize from parent or structured inits at multiple lengths.
    2. Coarse-to-fine L-BFGS with adaptive temperature.
    3. Subgradient polishing with momentum.
    4. Direct peak reduction.
    5. Multi-length exploration with interpolation.
    6. Final micro-polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        # Create initial sequence
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Initial uniform: n={n_init}, score={best_score:.6f}")
    
    # === Phase 1: Coarse-to-fine optimization at multiple lengths ===
    print("\n=== Phase 1: Adaptive Temperature L-BFGS ===")
    
    # Determine which lengths to try
    n_parent = len(parent)
    n_values = [512, 768, 1024, 1536, 2048]
    if n_parent not in n_values:
        n_values.insert(0, n_parent)
    n_values = sorted(set(n_values))
    
    # Prioritize by proximity to parent length
    n_values_sorted = sorted(n_values, key=lambda x: abs(x - n_parent))
    
    for n_try in n_values_sorted:
        rem = deadline - time.monotonic() - 100
        if rem < 30:
            break
        
        t_alloc = min(200.0, rem * 0.6)
        if t_alloc < 20:
            continue
        
        print(f"\n  Optimizing n={n_try}, budget={t_alloc:.0f}s")
        
        # Multiple initializations
        inits = []
        if parent is not None:
            inits.append(interpolate_sequence(parent, n_try))
        # One-sided initialization (exploits asymmetry)
        inits.append(one_sided_init(n_try, alpha=1.5))
        inits.append(one_sided_init(n_try, alpha=0.5))
        # Random
        r = rng.standard_normal(n_try)
        r = np.maximum(r, 0)
        inits.append(r / r.sum())
        
        for init_idx, init in enumerate(inits):
            rem = deadline - time.monotonic() - 50
            if rem < 20:
                break
            
            t_phase = min(t_alloc / (len(inits) - init_idx), rem * 0.8)
            if t_phase < 15:
                break
            
            print(f"    Init {init_idx}/{len(inits)}, budget={t_phase:.0f}s")
            
            # Adaptive temperature optimization
            a_opt, score_opt = adaptive_temperature_optimize(
                init, t_phase, T_start=500, T_end=5, 
                T_decay=0.95, lbgfs_budget_frac=0.8
            )
            best_seq, best_score = check_and_update(a_opt, f"adaptT_n{n_try}_i{init_idx}", best_seq, best_score)
    
    # === Phase 2: Subgradient polishing ===
    print(f"\n=== Phase 2: Subgradient Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_sg = min(rem * 0.5, 150.0)
            print(f"  SubGrad budget={t_sg:.0f}s")
            a_sg, _ = subgradient_refine(best_seq, t_sg, lr_init=0.005, 
                                          momentum=0.9, top_k=max(20, len(best_seq)//30))
            best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)
    
    # === Phase 3: Direct peak reduction ===
    print(f"\n=== Phase 3: Peak Reduction (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_pr = min(rem * 0.6, 100.0)
            print(f"  PeakReduce budget={t_pr:.0f}s")
            a_pr, _ = coordinated_peak_reduction(best_seq, t_pr)
            best_seq, best_score = check_and_update(a_pr, "peak_reduce", best_seq, best_score)
    
    # === Phase 4: Try larger sequences ===
    print(f"\n=== Phase 4: Larger Sequences (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        larger_n = [n_best * 2, int(n_best * 1.5), n_best + 512]
        larger_n = [x for x in larger_n if x <= 8192 and x > n_best]
        larger_n = sorted(set(larger_n), reverse=True)
        
        for n_large in larger_n[:2]:
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            t_alloc = min(rem * 0.5, 150.0)
            print(f"  Trying n={n_large}, budget={t_alloc:.0f}s")
            init_large = interpolate_sequence(best_seq, n_large)
            a_lg, score_lg = adaptive_temperature_optimize(
                init_large, t_alloc, T_start=500, T_end=5,
                T_decay=0.95, lbgfs_budget_frac=0.8
            )
            best_seq, best_score = check_and_update(a_lg, f"larger_n{n_large}", best_seq, best_score)
            
            if best_seq is not None and len(best_seq) == n_large:
                # Quick subgradient polish
                rem2 = deadline - time.monotonic() - 20
                if rem2 > 10:
                    a_sg2, _ = subgradient_refine(best_seq, min(rem2*0.4, 50), 
                                                   lr_init=0.003, momentum=0.9)
                    best_seq, best_score = check_and_update(a_sg2, f"sg_large_n{n_large}", 
                                                             best_seq, best_score)
    
    # === Phase 5: Final refinement with low temperature L-BFGS ===
    print(f"\n=== Phase 5: Final L-BFGS (best={best_score:.6f}) ===")

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature L-BFGS in
log-space (softmax parametrization), combined with subgradient polishing,
direct peak reduction, and multi-length exploration. Key insight: sequence
asymmetry allows the auto-correlation peak to be spread below the 1/n barrier,
pushing the score below 2 toward the theoretical limit of 1.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def cross_corr_grad(w, a):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Uses the identity: grad[i] = 2 * sum_l a[l]*w[l+i] (valid indices).
    Computed via FFT as 2 * linear cross-correlation of a and w.
    """
    n = len(a)
    N = 1 << ((2 * n - 1) - 1).bit_length()
    A_fft = rfft(a, N)
    W_fft = rfft(w, N)
    grad_half = irfft(np.conj(A_fft) * W_fft, N)[:n]
    return 2.0 * grad_half


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def soft_max_obj_and_grad(z, T, n):
    """
    Soft-max smoothed objective and gradient in log-space.
    obj = (2*n/T) * log(sum_k exp(T*c[k])) where c = conv(softmax(z), softmax(z))
    """
    a = softmax_from_z(z)
    c = fast_convolve(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    log_sum_exp = max_Tc + np.log(exp_Tc.sum())
    obj = (2.0 * n / T) * log_sum_exp

    w = exp_Tc / exp_Tc.sum()
    grad_a = cross_corr_grad(w, a)
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lbgfs_optimize(a0, T, time_limit, maxiter=3000):
    """L-BFGS optimization in log-space with soft-max smoothing at temperature T."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit or stopped[0]:
            stopped[0] = True
            return 1e15
        obj, _ = soft_max_obj_and_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit or stopped[0]:
            stopped[0] = True
            return np.zeros_like(z)
        _, g = soft_max_obj_and_grad(z, T, n)
        return g

    maxfun = min(maxiter * 50, int((time_limit / 0.005)) + 100)
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxfun, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0.copy()

    return a_out


def adaptive_temperature_optimize(a0, time_limit, T_start=2000.0, T_end=3.0,
                                  T_decay=0.9, n_phases=None):
    """
    Run L-BFGS with decreasing temperature (coarse to fine annealing).
    Each phase uses a smaller T for sharper peak resolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if n_phases is None:
        n_phases = max(3, int(np.log(T_start / T_end) / np.log(1.0 / T_decay)))

    T = T_start
    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit:
            break
        if T < T_end:
            T = T_end

        remaining = time_limit - (time.monotonic() - t0)
        phase_time = min(remaining / (n_phases - phase), remaining * 0.4)
        if phase_time < 3:
            break

        # Scale maxiter based on phase time and problem size
        est_eval_time = max(0.001, n * 0.00001)
        maxiter_phase = max(50, int(phase_time / (est_eval_time * 20)))
        maxiter_phase = min(maxiter_phase, 5000)

        a_new = lbgfs_optimize(a, T, phase_time, maxiter=maxiter_phase)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:

            best_a = a_new.copy()
            a = a_new

            print(f"      AdaptT n={n} phase={phase+1}/{n_phases} T={T:.1f} "
                  f"({elapsed:.0f}s): score={best_score:.6f}")

        T *= T_decay

    return best_a, best_score


def subgradient_refine(a0, time_limit, lr_init=0.008, momentum=0.92,
                       top_k=None, anneal=0.9997):
    """Subgradient method with momentum for fine-tuning the auto-correlation peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    step_count = 0

    if top_k is None:
        top_k = max(10, n // 40)

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        k_actual = min(top_k, len(c))
        peaks = np.argpartition(c, -k_actual)[-k_actual:]
        pv = c[peaks]
        pv_range = pv.max() - pv.min()
        temp = max(0.0005 * pv_range, 1e-10)
        scaled = (pv - pv.max()) / temp
        w_peaks = np.exp(scaled)
        ws = w_peaks.sum()
        if ws < 1e-15:
            w_full = np.zeros(len(c))
            w_full[peaks[np.argmax(pv)]] = 1.0
        else:
            w_full = np.zeros(len(c))
            w_full[peaks] = w_peaks / ws

        grad = cross_corr_grad(w_full, a)
        v = momentum * v - lr * grad
        a_candidate = a + v
        a_new = simplex_project(a_candidate)
        a_new = a_new / a_new.sum()

        max_c = fast_convolve(a_new).max()
        score = 2.0 * n * max_c
        step_count += 1

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= anneal
        if no_improve > 150:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 4, 0.002)
            no_improve = 0

        if step_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad {elapsed:.0f}s: score={best_score:.6f} "
                  f"lr={lr:.6f} step={step_count}")

    return best_a, best_score

                for f1 in factors[:4]:
                    for f2 in factors[:4]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            rng = np.random.default_rng(iteration + 42)
            a_trial = a + rng.standard_normal(n) * 0.002
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break

        iteration += 1
        if iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakRed {elapsed:.0f}s: score={best_score:.6f} iter={iteration}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def one_sided_init(n, alpha=1.0):
    """Create a one-sided power-law initialization (exploits asymmetry)."""
    x = np.linspace(0.001, 1.0, n)
    a = x ** alpha
    a = a / a.sum()
    return a


def cosine_tapered_init(n, alpha=1.0):
    """Cosine-tapered one-sided initialization."""
    x = np.linspace(0, 1, n)
    a = (x ** alpha) * (0.5 + 0.5 * np.cos(np.pi * x))
    a = a + 1e-10
    a = a / a.sum()
    return a


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase gradient-based optimization for minimizing the auto-correlation
    peak ratio. Uses adaptive-temperature L-BFGS in softmax log-space as the
    main workhorse, with subgradient and direct peak reduction as refinement.
    Explores multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Initial uniform: n={n_init}, score={best_score:.6f}")

    # ============ Phase 1: Adaptive T L-BFGS at primary lengths ============
    print("\n=== Phase 1: Adaptive Temperature L-BFGS ===")

    n_parent = len(parent)
    # Choose lengths to explore
    n_candidates = set()
    for base in [512, 768, 1024, 1280, 1536, 2048]:
        n_candidates.add(base)
    if n_parent not in n_candidates:
        n_candidates.add(n_parent)

    n_list = sorted(n_candidates, key=lambda x: abs(x - n_parent))[:5]

    for n_try in n_list:
        rem = deadline - time.monotonic() - 120
        if rem < 30:
            break
        t_alloc = min(250.0, rem * 0.55)
        if t_alloc < 20:
            continue

        print(f"\n  --- n={n_try}, budget={t_alloc:.0f}s ---")

        # Multiple initializations
        inits = []
        init_labels = []
        # Parent interpolation (best prior)
        inits.append(interpolate_sequence(parent, n_try))
        init_labels.append("parent")
        # One-sided power law (exploits asymmetry)
        inits.append(one_sided_init(n_try, alpha=1.5))
        init_labels.append("onesided1.5")
        inits.append(one_sided_init(n_try, alpha=0.7))
        init_labels.append("onesided0.7")
        # Cosine tapered
        inits.append(cosine_tapered_init(n_try, alpha=1.0))
        init_labels.append("cosine")

        n_inits = len(inits)
        for ii in range(n_inits):
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            t_phase = min(t_alloc / (n_inits - ii), rem * 0.75)
            if t_phase < 10:
                break

            print(f"    [{init_labels[ii]}] T-anneal, budget={t_phase:.0f}s")
            a_opt, _ = adaptive_temperature_optimize(
                inits[ii], t_phase, T_start=1000.0, T_end=3.0, T_decay=0.88
            )
            best_seq, best_score = check_and_update(
                a_opt, f"AT_n{n_try}_{init_labels[ii]}", best_seq, best_score
        )

    # ============ Phase 2: Subgradient polishing ============
    print(f"\n=== Phase 2: Subgradient Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_sg = min(rem * 0.45, 180.0)
            n_best = len(best_seq)
            print(f"  n={n_best}, budget={t_sg:.0f}s")
            a_sg, _ = subgradient_refine(
                best_seq, t_sg, lr_init=0.006, momentum=0.93,
                top_k=max(15, n_best // 30)
            )
            best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

    # ============ Phase 3: Direct peak reduction ============
    print(f"\n=== Phase 3: Peak Reduction (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_pr = min(rem * 0.5, 120.0)
            print(f"  budget={t_pr:.0f}s")
            a_pr, _ = peak_reduction(best_seq, t_pr)
            best_seq, best_score = check_and_update(a_pr, "peakred", best_seq, best_score)

    # ============ Phase 4: Explore larger sequences ============
    print(f"\n=== Phase 4: Larger Sequences (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        larger_options = sorted(set([
            int(n_best * 1.5), int(n_best * 2), n_best + 512,
            2048, 3072, 4096
        ]), reverse=True)
        larger_options = [x for x in larger_options if x > n_best and x <= 4096]

        for n_large in larger_options[:2]:
            rem = deadline - time.monotonic() - 40
            if rem < 30:
                break
            t_alloc = min(rem * 0.5, 200.0)
            if t_alloc < 20:
                continue

            print(f"  n={n_large}, budget={t_alloc:.0f}s")
            init_large = interpolate_sequence(best_seq, n_large)
            a_lg, _ = adaptive_temperature_optimize(
                init_large, t_alloc, T_start=1000.0, T_end=3.0, T_decay=0.88
            )
            best_seq, best_score = check_and_update(
                a_lg, f"larger_n{n_large}", best_seq, best_score
            )

            if best_seq is not None and len(best_seq) == n_large:
                rem2 = deadline - time.monotonic() - 25
                if rem2 > 15:
                    a_sg2, _ = subgradient_refine(
                        best_seq, min(rem2 * 0.4, 60),
                        lr_init=0.004, momentum=0.93
                    )
                    best_seq, best_score = check_and_update(
                        a_sg2, f"sg_n{n_large}", best_seq, best_score
                )
                rem3 = deadline - time.monotonic() - 15
                if rem3 > 10:
                    a_pr2, _ = peak_reduction(best_seq, min(rem3 * 0.5, 40))
                    best_seq, best_score = check_and_update(
                        a_pr2, f"pr_n{n_large}", best_seq, best_score
                )

    # ============ Phase 5: Final low-T L-BFGS refinement ============
    print(f"\n=== Phase 5: Final L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for T_final in [3, 5, 10, 20]:
            rem = deadline - time.monotonic() - 15
            if rem < 8:
                break
            t_alloc = min(rem * 0.35, 50.0)
            if t_alloc < 5:
                break
            print(f"  T={T_final}, n={n_best}, budget={t_alloc:.0f}s")
            a_fin = lbgfs_optimize(best_seq, T_final, t_alloc, maxiter=3000)
            best_seq, best_score = check_and_update(
                a_fin, f"final_T{T_final}", best_seq, best_score
            )

    # ============ Phase 6: Ultra-fine peak shaving ============
    print(f"\n=== Phase 6: Final Peak Shave (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_ps, _ = peak_reduction(best_seq, rem * 0.85)
            best_seq, best_score = check_and_update(a_ps, "final_ps", best_seq, best_score)

        # One more subgradient pass with very low lr
        rem = deadline - time.monotonic() - 3
        if rem > 5:
            a_sg_final, _ = subgradient_refine(
                best_seq, rem * 0.7, lr_init=0.001, momentum=0.95,
                top_k=5
            )
            best_seq, best_score = check_and_update(
                a_sg_final, "final_sg", best_seq, best_score
            )

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification with the actual evaluator
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Should not happen, but use computed score
        c = fast_convolve(best_seq)
        best_score = 2.0 * len(best_seq) * c.max() / (best_seq.sum() ** 2)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
