import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_full_conv(a):
    """Full convolution vector via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = A * A
    return np.fft.irfft(C, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_max_conv(a, top_k=5):
    """
    Compute subgradient of max_k conv(a,a)[k] using top-k peaks.
    Returns (gradient, max_value, peak_indices).
    */
    n = len(a)
    conv = compute_full_conv(a)
    M = conv.max()
    
    # Get top-k peak indices
    if top_k >= len(conv):
        peak_indices = np.arange(len(conv))
    else:
        peak_indices = np.argsort(conv)[-top_k:]
    
    # Weight peaks by how close they are to max
    weights = np.zeros(len(peak_indices))
    for i, pk in enumerate(peak_indices):
        # Exponential weight: peaks closer to max get more weight
        gap = M - conv[pk]
        weights[i] = np.exp(-gap / (0.01 * M + 1e-15))
    weights /= weights.sum()
    
    # Subgradient: g[m] = sum_i w_i * 2 * a[pk_i - m]
    g = np.zeros(n)
    for i, pk in enumerate(peak_indices):
        pk = int(pk)
        # a[pk - m] for valid m: m in [max(0, pk-n+1), min(n-1, pk)]
        m_start = max(0, pk - n + 1)
        m_end = min(n - 1, pk)
        if m_end >= m_start:
            # a[pk - m] for m in [m_start, m_end] is a[pk-m_end : pk-m_start+1][::-1]
            idx = pk - np.arange(m_start, m_end + 1)
            g[m_start:m_end + 1] += 2.0 * weights[i] * a[idx]
    
    return g, M, peak_indices


def nesterov_subgradient(a0, time_limit, lr_init=0.005, momentum=0.9,
                          top_k=5,anneal_alpha=0.9995, verbose=False):
    """
    Nesterov subgradient descent with simulated annealing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum velocity
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    temperature = 0.01
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Nesterov look-ahead
        a_look = project_simplex(a + momentum * v)
        
        # Compute subgradient at look-ahead point
        g, M_look, peaks = subgradient_max_conv(a_look, top_k=top_k)
        
        # Update velocity
        v = momentum * v - lr * g
        
        # Update position
        a_new = project_simplex(a + v)
        
        # Evaluate
        conv = fast_convolve(a_new)
        obj = conv.max()
        
        # Simulated annealing: occasionally accept worse solutions
        if obj < best_obj:
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                no_improve = 0
            a = a_new
        else:
            # Accept with probability exp(-(obj - best_obj) / temperature)
            delta = obj - best_obj
            if delta < temperature * 10:
                accept_prob = np.exp(-delta / (temperature + 1e-15))
                if rng.random() < accept_prob:
                    a = a_new
                else:
                    a = best_a.copy()
            else:
                a = best_a.copy()
            no_improve += 1
        
        # Cooling
        temperature *= anneal_alpha
        
        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.9
            if no_improve > 200:
                # Restart with perturbation
                noise_scale = 0.003 + 0.005 * min(temperature, 0.01)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                v = np.zeros(n)
                lr = lr_init * 0.8
                no_improve = 0
                temperature = 0.005
        
        if verbose and t % 200 == 0:
            score = 2 * n * best_obj
            print(f"    Nesterov iter {t}: score={score:.8f}, lr={lr:.6f}, temp={temperature:.6f}")
    
    return best_a, 2 * n * best_obj

            P = np.fft.rfft(p_rev, fft_size)
            conv_r = np.fft.irfft(A * P, fft_size)
            # result[j] = sum_k p[k] * a[k-j], for j in [0, n-1]
            # conv_r[i] = sum_j p_rev[j] * a[i-j] = sum_j p[conv_len-1-j] * a[i-j]
            # Let m = conv_len-1-j, then j = conv_len-1-m
            # conv_r[i] = sum_m p[m] * a[i-(conv_len-1-m)] = sum_m p[m] * a[i-conv_len+1+m]
            # We want result[j] = sum_m p[m] * a[m-j]
            # Set i-conv_len+1 = -j, i = conv_len-1-j
            # So result[j] = conv_r[conv_len-1-j]
            result = conv_r[conv_len - 1 : conv_len - 1 + n][::-1]
            
            grad = 2.0 * result
            
            # Multiplicative update: a *= exp(-lr * grad), then normalize
            a = a * np.exp(-inner_lr * grad / (grad.max() + 1e-15))
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
            else:
                a = a / s
        
        conv = fast_convolve(a)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            if verbose and it % 50 == 0:
                print(f"    MW iter {it}: score={2*n*best_obj:.8f}")
        
        # Outer: update p based on current convolution values
        scaled = conv / (0.05 * M + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        p = (1 - lr_outer) * p + lr_outer * w
        p = p / p.sum()
        
        # Adaptive outer lr
        if it % 200 == 0:
            lr_outer = max(0.05, lr_outer * 0.9)
        
        # Occasionally perturb
        if it % 300 == 0:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
    
    return best_a, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_window(n, window_type='kaiser', beta=5.0):
    """Create various window functions as initializations."""
    if window_type == 'kaiser':
        x = np.linspace(-1, 1, n)
        from scipy.special import i0
        a = i0(beta * np.sqrt(np.maximum(1 - x**2, 0)))
    elif window_type == 'nuttall':
        x = np.arange(n)
        a = (0.36358 - 0.48917 * np.cos(2*np.pi*x/n) + 
             0.13659 * np.cos(4*np.pi*x/n) - 0.01064 * np.cos(6*np.pi*x/n))
    elif window_type == 'flat_top':
        x = np.linspace(-1, 1, n)
        a = 1.0 - 1.93 * np.abs(x) + 1.29 * x**2 - 0.388 * x**3 + 0.028 * x**4
    elif window_type == 'gaussian':
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x**2)
    elif window_type == 'hamming':
        x = np.arange(n)
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x / n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def asymmetric_gaussian(n, center=0.4, sigma=0.25):
    """Asymmetric Gaussian."""
    x = np.arange(n) - n * center
    a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def oscillatory_window(n, freq=3.0):
    """Window with controlled oscillation."""
    base = np.hanning(n)
    osc = 1.0 + 0.1 * np.cos(2 * np.pi * freq * np.arange(n) / n)
    a = base * osc
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization:
    Phase 1: Diverse initialization (30s)
    Phase 2: Nesterov subgradient + annealing (500s)
    Phase 3: Peak surgery (350s)
    Phase 4: Multiplicative weights (250s)
    Phase 5: Multi-length exploration (350s)
    Phase 6: Coordinate refinement (200s)
    Phase 7: Final polish (100s + buffer)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score
    
    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    # Diverse initializations
    inits = []
    for n_try in [512, 768, 1024]:
        inits.append(asymmetric_gaussian(n_try, 0.35, 0.25))
        inits.append(asymmetric_gaussian(n_try, 0.5, 0.2))
        inits.append(make_window(n_try, 'kaiser', 4.0))
        inits.append(make_window(n_try, 'nuttall'))
        inits.append(make_window(n_try, 'flat_top'))
        inits.append(oscillatory_window(n_try, 2.0))
        inits.append(oscillatory_window(n_try, 5.0))
        # Exponential decay
        a_exp = np.exp(-np.arange(n_try) / (n_try * 0.3))
        a_exp /= a_exp.sum()
        inits.append(a_exp)
        # Double bump
        x = np.linspace(0, 1, n_try)
        a_double = np.exp(-20*(x-0.3)**2) + np.exp(-20*(x-0.7)**2)
        a_double /= a_double.sum()
        inits.append(a_double)
    
    for i, a_init in enumerate(inits):
        if time.monotonic() > deadline - 1500:
            break
        try_seq(a_init, f"init_{i}")
    
    # Use best as working sequence
    work_seq = best_seq if best_seq is not None else parent
    if work_seq is None:
        work_seq = asymmetric_gaussian(768, 0.4, 0.25)
        try_seq(work_seq, "fallback_init")
    
    work_n = len(work_seq)
    print(f"  Working with n={work_n}, score={best_score:.6f}")
    
    # === Phase 2: Nesterov subgradient + annealing ===
    print(f"\n=== Phase 2: Nesterov Subgradient (n={work_n}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 500)
    if t2 > 60:
        print(f"  Time allocated: {t2:.0f}s")
        a_opt, obj = nesterov_subgradient(work_seq, time_limit=t2 * 0.7,
                                           lr_init=0.006, momentum=0.85,
                                           top_k=5, verbose=True)
        try_seq(a_opt, "nesterov_main")
        
        # Second run with different params
        if time.monotonic() < deadline - 600:
            rem2 = deadline - time.monotonic() - 600
            t2b = min(rem2, 180)
            a_opt2, obj2 = nesterov_subgradient(best_seq, time_limit=t2b,
                                                 lr_init=0.003, momentum=0.9,
                                                 top_k=8, verbose=False)
            try_seq(a_opt2, "nesterov_fine")
    
    # === Phase 3: Peak surgery ===
    print(f"\n=== Phase 3: Peak Surgery ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t3 = min(rem, 350)
            print(f"  Time allocated: {t3:.0f}s")
            a_opt, obj = peak_surgery(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "peak_surgery")
            
            # From perturbed start
            if time.monotonic() < deadline - 400:
                rem2 = deadline - time.monotonic() - 400
                t3b = min(rem2, 150)
                perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.003)
                a_opt2, obj2 = peak_surgery(perturbed, time_limit=t3b, verbose=False)
                try_seq(a_opt2, "peak_surgery_pert")
    
    # === Phase 4: Multiplicative weights ===
    print(f"\n=== Phase 4: Multiplicative Weights ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 40:
            t4 = min(rem, 250)
            print(f"  Time allocated: {t4:.0f}s")
            a_opt, obj = multiplicative_weight_update(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "mw")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        # Try nearby lengths and significantly different lengths
        target_lengths = []
        base_n = len(base)
        for dl in [64, 128, 256, -128, -64]:
            target_lengths.append(base_n + dl)
        target_lengths.extend([512, 1024, 1280, 1536, 2048])
        target_lengths = [tl for tl in target_lengths if 256 <= tl <= 3000]
        
        seen_lengths = set()
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 100
            if rem < 40:
                break
            if target_n in seen_lengths:
                continue
            seen_lengths.add(target_n)
            
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            
            # Quick Nesterov on interpolated
            t5 = min(rem * 0.25, 80)
            if t5 > 15:
                a_opt, obj = nesterov_subgradient(f_interp, time_limit=t5,
                                                   lr_init=0.004, momentum=0.85,
                                                   top_k=5, verbose=False)
                try_seq(a_opt, f"nesterov_n{target_n}")
    
    # === Phase 6: Coordinate refinement ===
    print(f"\n=== Phase 6: Coordinate Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 30:
            t6 = min(rem, 200)
            print(f"  Time allocated: {t6:.0f}s")
            a_opt, obj = coordinate_refinement(best_seq, time_limit=t6, verbose=True)
            try_seq(a_opt, "coord_refine")
    
    # === Phase 7: Final polish (alternate methods) ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t7 = min(rem, 80)
            # Quick Nesterov with very small lr
            a_opt, obj = nesterov_subgradient(best_seq, time_limit=t7 * 0.5,
                                               lr_init=0.001, momentum=0.9,
                                               top_k=3, verbose=False)
            try_seq(a_opt, "final_nesterov")
            
            rem = deadline - time.monotonic() - 10
            if rem > 15:
                t8 = min(rem, 40)
                a_opt2, obj2 = peak_surgery(best_seq, time_limit=t8, verbose=False)
                try_seq(a_opt2, "final_surgery")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
