import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_grad(f, tau):
    """Compute gradient of log(sum exp(conv/tau))."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # h[j] = sum_{l} w[j+l] * f[l]
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    
    grad = (2.0 / tau) * h
    return grad, conv

def make_geometric(n, rate=0.7):

    if s > 1e-15:

    a = a * ramp

    s = a.sum()
    if s > 1e-15:

def two_coord_descent(f0, time_limit, verbose=False):
    """Exact 2-coordinate descent with ternary search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(42)
    it = 0
    
    def eval_max(conv):
        return conv.max()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Pick a random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        c = f[i] + f[j]
        if c < 1e-12:
            continue
        
        # Ternary search on x = f[i], f[j] = c - x, x in [0, c]
        lo, hi = 0.0, c
        for _ in range(25):  # ~25 iterations gives precision ~c/2^25
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            
            f[i], f[j] = m1, c - m1
            conv1 = fast_convolve(f)
            val1 = conv1.max()
            
            f[i], f[j] = m2, c - m2
            conv2 = fast_convolve(f)
            val2 = conv2.max()
            
            if val1 < val2:
                hi = m2
            else:
                lo = m1
        
        # Evaluate at the optimum
        x_opt = (lo + hi) / 2.0
        f[i], f[j] = x_opt, c - x_opt
        conv = fast_convolve(f)
        val = conv.max()
        
        if val < best_obj - 1e-14:
            best_obj = val
            best_f = f.copy()
            no_improve = 0
        else:
            # Revert
            f = best_f.copy()
            no_improve += 1
        
        if no_improve > 200:
            # Perturb
            noise = rng.standard_normal(n) * 0.002
            f = project_simplex(best_f + noise)
            no_improve = 0
            if fast_convolve(f).max() < best_obj:
                best_obj = fast_convolve(f).max()
                best_f = f.copy()
        
        if verbose and it % 100 == 0:
            obj_score = 2 * n * best_obj
            print(f"    2CD iter {it}: obj={obj_score:.8f}")
    
    return best_f, 2 * n * best_obj

        it += 1

        M = conv.max()
        k_star = int(np.argmax(conv))
        
        # Subgradient: g[j] = 2 * a[k_star - j] (with bounds)
        # (Q_k * a)[j] = a[k-j] if 0 <= k-j < n, else 0
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            g[j] = 2.0 * f[k_star - j]
        
        # Multiplicative update
        f = f * np.exp(-eta * g)
        f = np.maximum(f, 1e-30)
        f = f / f.sum()

        eta = 1.0 / (n * np.sqrt(it + 1))

            best_obj = obj
            best_f = f.copy()

            obj_score = 2 * n * best_obj
            print(f"    Mirror iter {it}: obj={obj_score:.8f}, eta={eta:.5f}")
    
    return best_f, 2 * n * best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Geometric (one-sided) initialization - asymmetric distributions reduce central peak.
2. FISTA (Nesterov-accelerated) with adaptive log-sum-exp temperature.
3. Exact 2-coordinate descent with warm-start ternary search on peak-relevant pairs.
4. Simulated annealing with structured restarts.
5. Multi-length exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):
    """Compute gradient of log(sum exp(conv/tau)). Returns (grad, conv)."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv


def make_geometric(n, rate=0.7):
    """Geometric (exponential decay) initialization - one-sided reduces central peak."""
    a = rate ** np.arange(n, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_power_law(n, alpha=2.0):
    """Power law decay initialization."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** (-alpha)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric_gauss(n, center_frac=0.3, sigma_frac=0.25):
    """Asymmetric Gaussian initialization."""
    x = np.arange(n, dtype=np.float64) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def fista_optimize(f0, time_limit, tau_init=None, verbose=False):
    """
    FISTA (Nesterov-accelerated proximal gradient) on the log-sum-exp smoothed
    objective over the simplex.
    """
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.04 * conv.max()
    else:
        tau = tau_init

    y = f.copy()
    t_k = 1.0
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv = compute_smooth_grad(y, tau)
        M = conv.max()

        # Step size: use 1/L where L ~ 2/tau (conservative) but adaptively smaller
        step = tau * 0.4
        f_new = y - step * grad
        f_new = project_simplex(f_new)

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = f_new + ((t_k - 1.0) / t_new) * (f_new - f)
        y = np.maximum(y, 0.0)
        y = y / max(y.sum(), 1e-15)
        f = f_new
        t_k = t_new

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f
