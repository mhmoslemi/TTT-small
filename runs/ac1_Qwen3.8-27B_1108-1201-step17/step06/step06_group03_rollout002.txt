"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smoothed mirror descent
with LP-Frank-Wolfe refinement, constraint reduction, and multi-scale exploration.
Key improvements over previous: proper log-sum-exp annealing, active-set LP,
frequency-informed initialization, and aggressive fine-tuning.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_conv_and_grad(f, tau, top_k_frac=0.25):
    """
    Compute smoothed max of convolution and its gradient.
    Uses top-k selection with softmax for efficiency.
    Returns (smoothed_max, gradient).
    """
    n = len(f)
    conv = fast_convolve(f)
    m = len(conv)
    
    # Select top-k peaks
    k = max(10, int(m * top_k_frac))
    # Get indices of top-k values
    thresholds = np.argpartition(conv, -k)[-k:]
    top_vals = conv[thresholds]
    
    # Softmax weighting
    scaled = tau * top_vals
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(k)
        w[np.argmax(top_vals)] = 1.0
    else:
        w = w / w_sum
    
    # Smoothed max approximation
    smoothed = top_vals.max() + np.log(w_sum) / tau
    
    # Compute gradient: d/d f_j [sum_k w_k * conv[k]]
    # conv[k] = sum_i f[i] * f[k-i]
    # d conv[k] / d f_j = f[k-j] + f[j] (if indices valid)
    #   (with the factor of 2 when k-j = j, i.e., k = 2j)
    grad = np.zeros(n)
    for idx in range(k):
        wk = w[idx]
        if wk < 1e-12:
            continue
        k_val = int(thresholds[idx])
        # j ranges from max(0, k_val - n + 1) to min(n-1, k_val)
        j_start = max(0, k_val - n + 1)
        j_end = min(n - 1, k_val)
        # For each valid j: d conv[k_val] / d f[j] = f[k_val - j] + f[j]
        # Note: when k_val - j == j, i.e., k_val = 2j, the derivative is 2*f[j]
        # But our formula f[k_val-j] + f[j] already handles this correctly:
        # when k_val = 2j, f[k_val-j] = f[j], so we get 2*f[j]. Correct!
        js = np.arange(j_start, j_end + 1)
        grad[j_start:j_end+1] += wk * (f[k_val - js] + f[js])
    
    return smoothed, grad, conv


def md_optimize(f0, time_limit, verbose=True, tau0=10.0, tau_end=500.0,
                lr0=0.01, top_k_frac=0.2):
    """
    Mirror Descent with log-sum-exp smoothing and adaptive parameters.
    f0: initial point (will be normalized to sum=1, non-negative)
    Returns (best_f, best_raw_obj)
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    tau = tau0
    lr = lr0
    no_improve = 0
    it = 0
    tau_step = (tau_end / tau0) ** (1.0 / 2000)  # gradual increase over ~2000 iters
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        smoothed, grad, conv = compute_conv_and_grad(f, tau, top_k_frac)
        raw_max = conv.max()
        
        # Mirror descent (multiplicative) update
        # f_new = f * exp(-lr * grad / f) ... no, standard mirror descent:
        # f_new[j] = f[j] * exp(-lr * grad[j])
        # Then project to simplex (normalize and clip to non-negative)
        f_new = f * np.exp(-lr * grad)
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s_new
        
        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < raw_max * 0.999:
            f = f_new
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    MD iter {it}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")
        else:
            # Try with smaller step
            f_trial = (1 - 0.5) * f + 0.5 * f_new
            f_trial = np.maximum(f_trial, 0)
            f_trial = f_trial / f_trial.sum()
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < raw_max:
                f = f_trial
                if obj_trial < best_obj - 1e-12:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
            else:
                no_improve += 1
                lr *= 0.5
        
        if no_improve > 50:
            # Restart with perturbation
            rng = np.random.default_rng(it * 31 + 7)
            noise_level = 0.002 + 0.003 * (no_improve / 100.0)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_level, 0)
            f = f / f.sum()
            lr = lr0 * 0.5
            no_improve = 0
        
        # Adaptive tau
        if it % 20 == 0:
            tau = min(tau * tau_step, tau_end)
        
        if lr < 1e-10:
            lr = lr0 * 0.1
    
    return best_f, best_obj


def lp_direction(f, M, n, active_tol=0.05):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses constraint reduction: only include constraints where f*f is close to M.
    """
    conv_ff = fast_convolve(f)
    
    # Only include "active" constraints (within active_tol of M)
    # This dramatically reduces LP size
    active = conv_ff >= M * (1 - active_tol)
    
    if not active.any():
        active = conv_ff >= M * 0.9
    
    # Also always include a few random constraints for robustness
    n_active = active.sum()
    if n_active < 10:
        # Add some random constraints
        extra = np.random.choice(len(conv_ff), size=10, replace=False)
        active[extra] = True
    
    # Build LP
    # Variables: g[0], ..., g[n-1] >= 0
    # Objective: maximize sum(g) => minimize -sum(g)
    # Constraints: (f*g)[k] <= M for active k
    #   (f*g)[k] = sum_j f[j] * g[k-j] (with valid indices)
    #   = sum_j g[j] * f[k-j] (by symmetry of the convolution structure)
    # Actually: (f*g)[k] = sum_j f[j] * g[k-j] where 0 <= j <= n-1, 0 <= k-j <= n-1
    #            = sum_j g[j] * f[k-j] where 0 <= j <= n-1, 0 <= k-j <= n-1
    
    # So the constraint matrix A has rows indexed by k (active) and columns by j:
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    
    c_obj = -np.ones(n)
    n_constraints = active.sum()
    
    # Build constraint matrix efficiently
    A_ub = np.zeros((n_constraints, n))
    active_indices = np.where(active)[0]
    
    for i, k in enumerate(active_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # A_ub[i, j] = f[k - j] for j in [j_start, j_end]
        js = np.arange(j_start, j_end + 1)
        A_ub[i, j_start:j_end+1] = f[k - js]
    
    b_ub = np.full(n_constraints, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_optimize(f0, time_limit, verbose=True, active_tol=0.05):
    """
    Frank-Wolfe optimization with LP direction finding and constraint reduction.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find LP direction
        g = lp_direction(f, M, n, active_tol)
        if g is None:
            no_improve += 1
            if no_improve > 30:
                # Perturb and retry
                rng = np.random.default_rng(it * 13 + 3)
                f = np.maximum(f + rng.standard_normal(n) * 0.003, 0)
                f = f / f.sum()
                no_improve = 0
            continue
        
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            continue
        g = g / g_sum
        
        # Check if g is significantly different from f
        if np.max(np.abs(g - f)) < 1e-8:
            no_improve += 1
            if no_improve > 50:
                # Try with relaxed M
                g = lp_direction(f, M * 1.01, n, active_tol)
                if g is not None:
                    g = np.maximum(g, 0)
                    gs = g.sum()
                    if gs > 1e-15:
                        g = g / gs
                    no_improve = 0
                else:
                    # Restart
                    rng = np.random.default_rng(it * 17 + 11)
                    f = np.maximum(best_f + rng.standard_normal(n) * 0.005, 0)
                    f = f / f.sum()
                    no_improve = 0
            continue
        else:
            no_improve = 0
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.05, 1.0, 20):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha < 1e-10:
            no_improve += 1
            continue
        
        # Fine line search around best_alpha
        lo = max(0, best_alpha - 0.05)
        hi = min(1.0, best_alpha + 0.05)
        for _ in range(8):
            mid = (lo + hi) / 2
            ft = f + mid * (g - f)
            ft = np.maximum(ft, 0)
            ft = ft / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid
        
        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        obj = fast_convolve(f_new).max()
        f = f_new
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            if verbose and it % 5 == 0:
                print(f"    FW iter {it}: obj={best_obj:.10f}")
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=True):
    """
    Coordinate descent: for each coordinate, find optimal value via
    local search, keeping other coordinates proportionally scaled.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 99993 + 1)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            
            old_val = f[j]
            if old_val < 1e-10:
                continue
            
            best_val = old_val
            best_obj_j = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            # Try several perturbations
            for delta_frac in [-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5]:
                new_val = old_val + delta_frac * old_val
                new_val = max(0.0, min(0.5, new_val))
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_obj_j - 1e-14:
                    best_obj_j = val
                    best_val = new_val
            
            if best_val != old_val:
                scale = (1.0 - best_val) / denom
                f[j] = best_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = np.maximum(f, 0)
                f = f / f.sum()
                best_obj = fast_convolve(f).max()
                best_f = f.copy()
                improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def make_initializations(n, seed=42):

    inits.append((a / a.sum(), "exp_growth"))

    a = np.linspace(0.01, 1.0, n)
    inits.append((a / a.sum(), "ramp"))
    
    # Inverse ramp
    a = np.linspace(1.0, 0.01, n)
    inits.append((a / a.sum(), "inv_ramp"))
    
    # Sine-modulated exponential
    a = np.exp(-x / 0.2) * (1 + 0.5 * np.sin(2 * np.pi * x))
    a = np.maximum(a, 0)
    inits.append((a / a.sum(), "sine_exp"))
    
    # Cosine window
    a = np.cos(np.pi * x / 2) ** 2
    inits.append((a / a.sum(), "cosine"))

    a = np.exp(-20*(x-0.3)**2) + 0.5*np.exp(-20*(x-0.7)**2)
    inits.append((a / a.sum(), "two_bump"))
    
    # Power law
    a = (np.arange(1, n+1)) ** (-0.3)
    inits.append((a / a.sum(), "power_law"))
    
    # Square root
    a = np.sqrt(np.arange(n) + 1)
    inits.append((a / a.sum(), "sqrt"))

    a = rng.standard_normal(n) ** 2
    inits.append((a / a.sum(), "random"))

    inits.append((a / a.sum(), "u_shaped"))

    a = np.exp(-10*(x-0.4)**2) + 0.3*np.exp(-30*(x-0.8)**2)
    inits.append((a / a.sum(), "asym_bump"))
    
    return inits
