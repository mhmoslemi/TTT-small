import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Compute objective locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def build_conv_matrix(f):
    """Build the convolution matrix B such that B @ g = f * g (full)."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for i in range(n):
        B[i:i + n, i] = f
    return B


def solve_lp_direction(f, M, solver=cp.ECOS):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)_k <= M for all k.
    Returns (g_opt, optimal_value).
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    B = build_conv_matrix(f)
    constraints = [B @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None, None
    if g.value is None:
        return None, None
    g_val = np.maximum(g.value, 0)
    return g_val, float(prob.value)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:

                if new_val < 0 or new_val > 0.999:

                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

                denom = 1.0 - old_val

                a[j] = best_j_val

                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:

                    best_a = a.copy()

"""
LP-based iterative optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses linear programming direction-finding (the state-of-the-art LP approach),
combined with multiplicative updates, multi-length exploration, subgradient
backup, and coordinate descent polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def build_conv_matrix(f):
    """Build convolution matrix B such that B @ g = f * g (full mode)."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for i in range(n):
        B[i:i + n, i] = f
    return B


def solve_lp_direction(f, M, solver_name="GLOP"):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)_k <= M for all k.
    Returns (g_opt, optimal_value) or (None, None) on failure.
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    B = build_conv_matrix(f)
    constraints = [B @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    
    solver_map = {
        "GLOP": cp.GLOP,
        "CBC": cp.CBC,
        "GLPK": cp.GLPK,
        "ECOS": cp.ECOS,
        "PDLP": cp.PDLP,
        "GUROBI": cp.GUROBI,
        "MOSEK": cp.MOSEK,
        "SCIP": cp.SCIP,
        "XPRESS": cp.XPRESS,
    }
    
    solver = solver_map.get(solver_name, cp.GLOP)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        # Try fallback solvers
        for fb in [cp.GLOP, cp.CBC, cp.GLPK, cp.ECOS, cp.PDLP]:
            if fb == solver:
                continue
            try:
                prob = cp.Problem(obj, constraints)
                prob.solve(solver=fb)
                break
            except Exception:
                continue
    
    if g.value is None:
        return None, None
    g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    return g_val, float(prob.value)


def lp_optimize(f_init, time_limit, t_init=0.1, min_t=0.005,
                max_iter=1000, solver_name="GLOP", verbose=True):
    """
    LP-based iterative optimization on the simplex (sum=1).
    Minimizes max(f*f) which is equivalent to minimizing 2n*max(f*f)/(sum f)^2.
    """
    t0 = time.monotonic()
    f = f_init.copy().astype(np.float64)
    S = f.sum()
    f = f / S
    n = len(f)
    
    b = fast_convolve(f)
    M = b.max()
    best_f = f.copy()
    best_M = M
    best_obj = 2.0 * n * M
    t_step = t_init
    stagnant = 0
    lp_fail_count = 0
    
    if verbose:
        print(f"    LP start: n={n}, M={M:.10f}, obj={best_obj:.6f}")
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP
        g_opt, lp_val = solve_lp_direction(f, M, solver_name=solver_name)
        
        if g_opt is None:
            lp_fail_count += 1
            if lp_fail_count > 10:
                if verbose:
                    print(f"    LP failing repeatedly, stopping at iter {it}")
                break
            continue
        
        lp_fail_count = 0
        S_g = g_opt.sum()
        
        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: S_g={S_g:.6f}, M={M:.10f}, obj={2*n*M:.6f}, t={t_step:.4f}")
        
        if S_g <= 1.0 + 1e-5:
            stagnant += 1
            if stagnant > 10:
                if verbose:
                    print(f"    Converged at iter {it} (S_g ~ 1)")
                break
            # Nudge M down slightly to find a new direction
            M *= 0.9995
            continue
        
        stagnant = 0
        g_rescaled = g_opt / S_g
        
        # Line search
        best_alpha = t_step
        best_new_obj = float('inf')
        alphas = [t_step, 2 * t_step, min(4 * t_step, 0.5), min(0.8 * t_step, 1.0), 1.0]
        for alpha in alphas:
            alpha = min(alpha, 1.0)
            f_trial = (1.0 - alpha) * f + alpha * g_rescaled
            f_trial = np.maximum(f_trial, 0.0)
            S_trial = f_trial.sum()
            if S_trial < 0.01:
                continue
            f_trial = f_trial / S_trial
            b_trial = fast_convolve(f_trial)
            M_trial = b_trial.max()
            obj_trial = 2.0 * n * M_trial
            if obj_trial < best_new_obj:
                best_new_obj = obj_trial
                best_alpha = alpha
        
        if best_new_obj >= best_obj:
            # No improvement found, reduce step
            t_step = max(min_t, t_step * 0.5)
            stagnant += 1
            if stagnant > 30:
                break
            continue
        
        # Update
        f = (1.0 - best_alpha) * f + best_alpha * g_rescaled
        f = np.maximum(f, 0.0)
        S = f.sum()
        if S < 0.01:
            f = best_f.copy()
            break
        f = f / S
        
        b = fast_convolve(f)
        M_new = b.max()
        obj_new = 2.0 * n * M_new
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_M = M_new
            best_f = f.copy()
            M = M_new
            t_step = min(t_step * 1.3, 0.5)
        else:
            t_step = max(min_t, t_step * 0.7)
            f = best_f.copy()
            M = best_M
    
    if verbose:
        print(f"    LP done: best obj={best_obj:.6f} (n={n})")
    return best_f, best_obj


def multiplicative_update_optimize(f_init, time_limit, max_iter=5000, verbose=True):
    """
    Multiplicative (power-iteration style) update.
    a[i] <- a[i] / (cross_corr[i])^(1/2), normalized.
    """
    t0 = time.monotonic()
    f = f_init.copy().astype(np.float64)
    f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    
    if verbose:
        print(f"    MU start: n={n}, obj={best_obj:.6f}")
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(f)
        b_max = b.max()
        
        # Compute cross-correlation with peak-weighted b
        # Use all peaks with soft weighting
        T = 0.001
        w = np.exp(b / T - b_max / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w /= w_sum
        
        # grad[i] = sum_k w[k] * f[k-i] (valid range)
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w[k] * f[k - idx]
        
        # Multiplicative update
        grad_safe = np.maximum(grad, 1e-15)
        f_new = f / np.sqrt(grad_safe / (grad_safe.mean() + 1e-30))
        f_new = np.maximum(f_new, 0.0)
        S = f_new.sum()
        if S < 0.01:
            break
        f_new = f_new / S
        
        b_new = fast_convolve(f_new)
        obj_new = 2.0 * n * b_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
        f = f_new
    
    if verbose:
        print(f"    MU done: best obj={best_obj:.6f}")
    return best_f, best_obj
