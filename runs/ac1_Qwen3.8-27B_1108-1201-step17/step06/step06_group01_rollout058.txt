Hmm, actually I realize that for the L-BFGS approach, working in log-space is cleaner. Let me define:

- Parameters: z (length n), where a_i = exp(z_i) / sum(exp(z_j))
- Objective: f(z) = 2n * T * log(sum_k exp(conv(softmax(z), softmax(z))_k / T))
- Gradient: computed via chain rule

The gradient computation:
1. a = softmax(z)
2. b = conv(a, a)
3. w_k = exp(b_k/T) / sum_j exp(b_j/T)
4. grad_a_i = 4n * sum_k w_k * a_{k-i} (cross-correlation)
5. grad_z_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)

This is the standard gradient of a function on the simplex parameterized by log-coordinates.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent with 
temperature annealing, combined with L-BFGS polishing and coordinate descent.
Key novelty: Uses multiplicative (mirror) updates in the primal simplex space
with an adaptive temperature schedule for the soft-max smoothing, which provides
a different optimization landscape than the LP-based approach used previously.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def soft_max_lse(b, T):
    """Compute T * log(sum(exp(b/T))) using log-sum-exp for stability."""
    scaled = b / T
    m = scaled.max()
    return T * (m + np.log(np.exp(scaled - m).sum()))

def compute_soft_max_grad(a, b, T):
    """Compute gradient of 2n*T*log(sum(exp(b_k/T))) w.r.t. a (on simplex).
    Returns gradient projected to the tangent space of the simplex."""
    n = len(a)
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    
    # grad_a[i] = 4n * sum_k w[k] * a[k-i]
    # cross-correlation of w (length 2n-1) and a (length n)
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    grad_a = 4.0 * n * correl[:n]
    
    return grad_a

def log_space_objective(z, T, n):
    """Objective in log-space: a = softmax(z), return soft-max smoothed score."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    obj = 2.0 * n * soft_max_lse(b, T)
    return obj, a, b

def log_space_gradient(z, T, n):
    """Gradient in log-space."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    grad_a = compute_soft_max_grad(a, b, T)
    # Gradient in log-space: grad_z = a * (grad_a - sum(a * grad_a))
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    return grad_z

def mirror_descent_optimize(a_init, time_limit, T_start=0.01, T_end=0.001, 
                            eta_start=0.01, eta_end=0.0001, verbose=True):
    """
    Adaptive mirror descent with temperature annealing.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    total_iters_estimate = 5000
    stuck_count = 0
    stuck_threshold = 200
    
    for it in range(1, total_iters_estimate + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealing schedule (exponential)
        progress = min(1.0, it / (0.8 * total_iters_estimate))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Compute mirror descent step
        grad_a = compute_soft_max_grad(a, b, T)
        
        # Multiplicative update
        log_update = -eta * grad_a
        # Clip to prevent numerical issues
        log_update = np.clip(log_update, -5, 5)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()
        
        # If stuck, try a random restart
        if stuck_count > stuck_threshold:
            rng = np.random.default_rng(it)
            perturbation = rng.standard_normal(n) * 0.01
            a = best_a + perturbation
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck_count = 0
            eta = min(eta * 2, eta_start)
            
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it} ({elapsed:.0f}s): T={T:.5f} eta={eta:.6f} "
                  f"score={current_score:.6f} best={best_score:.6f}")
    
    return best_a, best_score

def lbgfs_log_space(a_init, time_limit, T=50, maxiter=5000):
    """L-BFGS in log-space with soft-max smoothing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    z0 = np.log(a) - np.log(a).mean()  # Center for numerical stability
    t0 = time.monotonic()
    stopped = [False]
    
    def func(z):
        if time.monotonic() - t0 > time_limit or stopped[0]:
            stopped[0] = True
            return 1e15
        obj, _, _ = log_space_objective(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit or stopped[0]:
            stopped[0] = True
            return np.zeros(n)
        return log_space_gradient(z, T, n)
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        # Convert back
        z_opt = res.x
        z_shifted = z_opt - z_opt.max()
        a_out = np.exp(z_shifted)
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a_init
    
    return a_out

def coordinate_descent(a_init, time_limit, n_trials_per_coord=5):
    """Coordinate descent: try adjusting each coordinate."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    # Try different adjustment factors
    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        # Use random order for diversity
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * f
                if a_trial[i] < 0:
                    continue
                # Redistribute the change to maintain sum=1
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-15:
                    continue
                # Distribute diff to other elements proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] -= diff * a_trial[mask] / others_sum
                if a_trial.min() < -1e-10:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial
                    best_a = a.copy()
                    improved = True
                    break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent with 
temperature annealing, L-BFGS in log-space, multiplicative weights with exact 
peak subgradients, and coordinate descent polishing. Key novelty: uses 
multiplicative (mirror geometry) updates which explore a different optimization 
landscape than the LP-based approach, combined with careful temperature schedules
and multi-phase refinement.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def compute_soft_max_weights(b, T):
    """Compute softmax weights over auto-correlation values."""
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    return w


def compute_grad_a(a, w):
    """Compute gradient 4n * cross_corr(w, a) where w has length 2n-1, a has length n.
    cross_corr[i] = sum_k w[k] * a[k-i]"""
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    return 4.0 * n * correl[:n]


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def log_space_func(z, T, n):
    """Objective in log-space with soft-max smoothing."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return 2.0 * n * T * lse


def log_space_grad(z, T, n):
    """Gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    w = compute_soft_max_weights(b, T)
    grad_a = compute_grad_a(a, w)
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    return grad_z


def mirror_descent_optimize(a_init, time_limit, T_start=0.01, T_end=0.0005,
                            eta_start=0.02, eta_end=0.0005, max_iters=20000,
                            verbose=True, label=""):
    """Adaptive mirror descent with temperature annealing on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    stuck_count = 0
    stuck_threshold = 300

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Exponential annealing
        progress = min(1.0, it / (0.7 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        # Compute soft-max weights and gradient
        w = compute_soft_max_weights(b, T)
        grad_a = compute_grad_a(a, w)

        # Multiplicative (mirror) update
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        # Stuck detection: random perturbation
        if stuck_count > stuck_threshold:
            rng = np.random.default_rng(it * 7919)
            noise = rng.standard_normal(n) * 0.02
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck_count = 0
            eta = min(eta * 3.0, eta_start * 0.5)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD iter={it} ({elapsed:.0f}s): T={T:.5f} eta={eta:.5f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def multiplicative_weights_exact(a_init, time_limit, eta_start=0.005,
                                 decay=0.9995, max_iters=50000, verbose=True, label=""):
    """Multiplicative weights using exact peak subgradient (no smoothing)."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    eta = eta_start

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        # Find peak
        k_peak = int(np.argmax(b))

        # Subgradient: grad[i] = 2 * a[k_peak - i] for valid indices
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_peak - idx]

        # Project to tangent space of simplex
        grad = grad - np.dot(a, grad)

        # Multiplicative update
        log_update = -eta * grad
        log_update = np.clip(log_update, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        eta *= decay

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MW iter={it} ({elapsed:.0f}s): eta={eta:.6f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def lbgfs_log_space(a_init, time_limit, T=50, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()
    call_count = [0]

    def func(z):
        call_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return log_space_func(z, T, n)

    def jac(z):
        call_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        return log_space_grad(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        z_opt = res.x
        a_out = softmax_from_z(z_opt)
    except Exception:
        a_out = a_init.copy()

    return a_out


def coordinate_descent(a_init, time_limit, verbose=True, label=""):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    factors = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5]
    consecutive_no_improve = 0
    max_no_improve = 200

    while time.monotonic() - t0 < time_limit and consecutive_no_improve < max_no_improve:
        consecutive_no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial[i] = new_val
                # Redistribute to maintain sum=1
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    break

        if not improved_pass:
            consecutive_no_improve += 1
        if verbose:
            elapsed = time.monotonic() - t0
            if elapsed % 10 < 1:
                print(f"    {label} CD ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if rem > 15:
            t_alloc = min(rem * 0.5, 80.0)
            print(f"  CD: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_cd, "coord_descent", best_seq, best_score)

    # === Phase 6: Second round MD + MW from current best ===
    print(f"\n=== Phase 6: Refinement Round 2 (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 40:
            t_alloc = min(rem * 0.4, 100.0)
            print(f"  MD2: budget={t_alloc:.0f}s")
            a_md2, _ = mirror_descent_optimize(
                best_seq, t_alloc, T_start=0.003, T_end=0.0002,
                eta_start=0.008, eta_end=0.0001, label="MD2"
            )
            best_seq, best_score = check_and_update(a_md2, "MD2", best_seq, best_score)

            rem = deadline - time.monotonic() - 80
            if rem > 20:
                t_alloc = min(rem * 0.4, 60.0)
                a_mw2, _ = multiplicative_weights_exact(
                    best_seq, t_alloc, eta_start=0.003, decay=0.9998, label="MW2"
                )
                best_seq, best_score = check_and_update(a_mw2, "MW2", best_seq, best_score)

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.25, 1.5, 0.8, 2.0]:
            alt_n = int(n_best * factor)
            if 128 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic() - 60
            if rem < 40:
                break
            t_alloc = min(rem * 0.4, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = mirror_descent_optimize(
                init_alt, t_alloc, T_start=0.01, T_end=0.001,
                eta_start=0.02, eta_end=0.0005, label=f"alt_n{alt_n}"
            )
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

            if best_seq is not None and len(best_seq) == alt_n:
                rem = deadline - time.monotonic() - 40
                if rem > 15:
                    a_lb_alt = lbgfs_log_space(best_seq, min(rem * 0.5, 40.0), T=30)
                    best_seq, best_score = check_and_update(a_lb_alt, f"lb_alt_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        # Low-temperature L-BFGS
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            for T in [3, 1]:
                rem = deadline - time.monotonic() - 8
                if rem < 8:
                    break
                t_alloc = min(rem * 0.5, 25.0)
                a_fin = lbgfs_log_space(best_seq, t_alloc, T=T, maxiter=4000)
                best_seq, best_score = check_and_update(a_fin, f"final_T{T}", best_seq, best_score)

        # Final peak shave
        rem = deadline - time.monotonic() - 8
        if rem > 8:
            a_ps_fin, _ = peak_shave(best_seq, min(rem * 0.6, 15.0))
            best_seq, best_score = check_and_update(a_ps_fin, "final_ps", best_seq, best_score)

        # Final coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_cd_fin, _ = coordinate_descent(best_seq, min(rem * 0.6, 10.0))
            best_seq, best_score = check_and_update(a_cd_fin, "final_cd", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
