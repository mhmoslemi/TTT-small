"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Exact subgradient descent with randomized peak selection and momentum
2. LP-Frank-Wolfe refinement
3. Multi-length exploration with diverse asymmetric initializations
4. Coordinate descent polishing
Key innovation: uses contrast-aware peak selection (focuses on peaks that are
significantly above the median), momentum in the subgradient, and adaptive
switching between "exploit" (fine-tuning) and "explore" (restart) modes.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_at_peaks(f, conv, peaks):
    """Compute exact subgradient at specified peak positions."""
    n = len(f)
    sg = np.zeros(n)
    for k in peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        sg[js:je + 1] += f[k - idx] + f[idx]
    return sg / len(peaks)


def momentum_subgradient(f0, time_limit, verbose=False, seed=42):
    """
    Subgradient descent with momentum, randomized peak selection,
    and multiplicative weights (mirror descent) update.
    Uses contrast-aware peak selection: weights peaks by how much they
    exceed the median convolution value.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    rng = np.random.default_rng(seed)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.005
    momentum = 0.7
    sg_ema = np.zeros(n)
    no_improve = 0
    it = 0
    k_top = 5

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        med = np.median(conv)
        contrast = M - med

        # Adaptive k_top: larger when contrast is high
        if contrast > 0.1 * M:
            k_top = max(3, int(len(conv) * 0.05))
        else:
            k_top = max(5, int(len(conv) * 0.15))

        # Select top peaks with contrast-aware weighting
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        # Weight by excess over median (contrast-aware)
        w = np.maximum(pv - med, 1e-10)
        w = w / w.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            sg[js:je + 1] += w[i] * (f[k - idx] + f[idx])

        # Momentum
        sg_ema = momentum * sg_ema + (1 - momentum) * sg

        # Multiplicative weights update
        f_new = f * np.exp(-lr * sg_ema)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        # Backtracking line search
        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.01:
            for damp in [0.5, 0.25, 0.1, 0.05]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    break
            else:
                lr *= 0.5
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.005, 0.03)
        else:
            no_improve += 1
            lr *= 0.995

        # Restart logic
        if no_improve > 80:
            # Structured restart: blend with a perturbed version
            if rng.random() < 0.5:
                noise = rng.standard_normal(n) * 0.008
                f = np.maximum(best_f + noise, 1e-15)
            else:
                # Anti-symmetric perturbation
                anti = best_f - best_f[::-1]
                f = best_f + 0.01 * anti
                f = np.maximum(f, 1e-15)
            f = f / f.sum()
            sg_ema = np.zeros(n)
            lr = 0.005
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"      MSD restart at iter {it}: obj={best_obj:.10f}")

        if lr < 1e-7:
            break

    return best_f, best_obj


def lp_solve_direction(f, M_bound, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M_bound for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M_bound)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        if g is not None and g.sum() > 1e-15:
            return g
    return None

                rng = np.random.default_rng(it * 17 + 3)

                m_exp = 1.0

        if g_sum < 1e-15:
            m_exp *= 1.1

                m_exp = 1.0

        else:
            no_improve = 0

        # Golden section line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]

            if val < best_val:

                best_alpha = alpha

        if best_alpha > 1e-10:
            # Refine with bisection
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):

                if st < 1e-15:

                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 10.0)

        if no_improve > 120:

def coordinate_descent(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate j, find the optimal value
    by 1D minimization. Uses the fact that changing a[j] affects
    conv[k] for k in {2j, 2j+1, ..., 2j+n-2} (with bounds).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Try a range of perturbations
            for frac in [-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5, 0.8]:
                new_val = old_val * (1.0 + frac)
                if new_val < 0 or new_val > 0.8:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    ss = f.sum()
                    if ss > 1e-15:
                        f = f / ss
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def make_diverse_init(n, variant, seed=42):

    rng = np.random.default_rng(seed + variant * 137)

    elif variant == 2:
        # Power law decay
        a = (x + 1) ** (-0.3)
    elif variant == 3:

        a = np.exp(-x / (n * 0.08)) * 0.7 + np.exp(-x / (n * 0.5)) * 0.3
    elif variant == 4:

        a = np.maximum(a, 0)

        a = np.exp(-((x - n * 0.3) / (n * 0.25)) ** 2)
    elif variant == 6:

        a = np.exp(-x / (n * 0.25)) * (1 + 0.3 * rng.standard_normal(n))
        a = np.maximum(a, 0)
    elif variant == 7:
        # Geometric progression
        ratio = 0.98
        a = ratio ** x
    elif variant == 8:
        # Logarithmic
        a = np.log1p(x + 1) / np.log1p(n + 1)
    elif variant == 9:
        # Bimodal asymmetric
        a = np.exp(-((x - n*0.2)/(n*0.15))**2) + 0.4*np.exp(-((x - n*0.7)/(n*0.2))**2)
    elif variant == 10:
        # Step-like (high then low)
        a = np.ones(n)
        a[:n//2] = 1.5
        a[n//2:] = 0.5
        a += rng.standard_normal(n) * 0.05

        a = np.exp(-np.abs(x - peak) / (n * 0.3))

    a = np.maximum(a, 0)
    ss = a.sum()

    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        print(f"  MSD on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
        a_opt, obj = momentum_subgradient(a_init, time_limit=t_alloc, verbose=True,
                                          seed=seed + idx)
        try_seq(a_opt, f"msd_{label}")

    # === Phase 3: LP-Frank-Wolfe on Best ===
    print("\n=== Phase 3: LP-Frank-Wolfe ===")
    if best_seq is not None:
        t_phase3_end = time.monotonic() + 400.0
        rem = t_phase3_end - time.monotonic()
        if rem > 60:
            t_alloc = min(rem, 350.0)
            print(f"  LP-FW on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_main")

    # === Phase 4: Multi-Length Exploration ===
    print("\n=== Phase 4: Multi-Length Exploration ===")
    if best_seq is not None:
        t_phase4_end = time.monotonic() + 250.0
        base = best_seq
        target_lengths = [512, 640, 896, 1024, 1280]

        for target_n in target_lengths:
            rem = t_phase4_end - time.monotonic() - 30.0
            if rem < 30:
                break
            if len(base) == target_n:
                continue
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if np.isfinite(sc) and sc < best_score + 0.02:
                t_alloc = min(rem * 0.45, 100.0)
                if t_alloc < 20:
                    t_alloc = 20
                print(f"  MSD n={target_n}: t={t_alloc:.0f}s")
                a_opt, obj = momentum_subgradient(f_interp, time_limit=t_alloc,
                                                  seed=seed + target_n)
                try_seq(a_opt, f"msd_n{target_n}")

    # === Phase 5: Random Restarts ===
    print("\n=== Phase 5: Random Restarts ===")
    if best_seq is not None:
        t_phase5_end = time.monotonic() + 180.0
        n_best = len(best_seq)

        for r in range(5):
            rem = t_phase5_end - time.monotonic() - 20.0
            if rem < 20:
                break
            t_alloc = min(rem / max(1, 5 - r), 60.0)
            if t_alloc < 15:
                t_alloc = 15

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.01
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
            elif r == 1:
                # Blend with different asymmetric init
                asym = make_diverse_init(n_best, variant=rng.integers(0, 12),
                                         seed=seed + 100 + r)
                f_init = 0.6 * best_seq + 0.4 * asym
                f_init = f_init / f_init.sum()
            elif r == 2:
                # Reverse blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            elif r == 3:
                # Larger perturbation
                noise = rng.standard_normal(n_best) * 0.02
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()
            else:
                # Oscillatory perturbation
                osc = 0.01 * np.sin(2 * np.pi * rng.uniform(1, 5) * np.arange(n_best) / n_best)
                f_init = np.maximum(best_seq + osc, 1e-15)
                f_init = f_init / f_init.sum()

            print(f"  Restart {r}: n={n_best}, t={t_alloc:.0f}s")
            a_opt, obj = momentum_subgradient(f_init, time_limit=t_alloc,
                                              seed=seed + 200 + r)
            try_seq(a_opt, f"restart_{r}")

    # === Phase 6: Coordinate Descent Polish ===
    print("\n=== Phase 6: Coordinate Descent ===")
    if best_seq is not None:
        t_phase6_end = time.monotonic() + 80.0
        rem = t_phase6_end - time.monotonic()
        if rem > 15:
            t_alloc = min(rem, 70.0)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 7: Final LP-FW Squeeze ===
    print("\n=== Phase 7: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15.0
        if rem > 15:
            t_alloc = min(rem, 50.0)
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

    # === Phase 8: One more MSD with very small lr ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10.0
        if rem > 10:
            t_alloc = min(rem, 25.0)
            print(f"  Final MSD (low lr): t={t_alloc:.0f}s")
            a_opt, obj = momentum_subgradient(best_seq, time_limit=t_alloc,
                                              seed=seed + 999)
            try_seq(a_opt, "final_msd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
