import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlate(a, w):
    """Compute R[j] = sum_l a[l] * w[l+j] for j=0,...,len(a)-1."""
    n = len(a)
    m = len(w)
    N = 1 << (n + m - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_edge_concentrated(n, alpha):
    """Sequence concentrated at edges: exp(alpha * (2i/n - 1)^2)."""
    x = 2.0 * np.arange(n) / n - 1.0
    a = np.exp(alpha * x * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

    """Sequence with spike at position 0, ones elsewhere."""
    a = np.ones(n)
    a[0] = c
    a /= a.sum()
    return a

    """Sequence with spikes at both ends."""
    a = np.ones(n)

def make_shifted_gaussian(n, center_frac, sigma_frac):
    """Gaussian centered at a fraction of the length."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:

def solve_lp_direction_sparse(f, M, n, top_k=40, margin=1.0):
    """
    Solve LP: max sum(g) s.t. conv(f,g)[k] <= M*margin for top-k peak positions, g >= 0.
    Uses only the top-k constraints for speed.
    """
    conv_ff = fft_convolve(f)
    peak_indices = np.argsort(conv_ff)[-top_k:]
    
    K = len(peak_indices)
    A_ub = np.zeros((K, n))
    for idx, k in enumerate(peak_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[idx, j] = f[k - j]

    bounds = [(0, None)] * n

    return None

    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fft_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    margin = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem_time = time_limit - (time.monotonic() - t0)
        if rem_time < 2:
            break

        g = solve_lp_direction_sparse(f, M, n, top_k=top_k, margin=m_exp)
        
        if g is None:
            m_exp = min(m_exp * 1.15, 5.0)
            if m_exp > 4.0 and it > 50:
                # Perturb and restart
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fft_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp = min(m_exp * 1.1, 5.0)
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fft_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search: golden section
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fft_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fft_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.max(f_new.sum(), 1e-15)
            f_new = f_new / f_new.sum()
            conv_new = fft_convolve(f_new)
            M_new = conv_new.max()
            f = f_new
            M = M_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)

        if no_improve > 150:
            break

    return best_f, 2 * n * best_obj


def mirror_descent_optimize(f0, time_limit, verbose=False):
    """
    Mirror descent (exponentiated gradient) with soft subgradient.
    Uses multiplicative updates to maintain positivity.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fft_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    tau = 0.03 * best_obj
    eta = 0.3
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.1:
            break

        conv = fft_convolve(f)
        M = conv.max()

        # Soft subgradient weights
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # h[j] = sum_k w[k] * f[k-j]
        h = cross_correlate(f, w)
        h = np.maximum(h, 1e-15)

        # Multiplicative update: f_new = f * exp(-eta * gradient)
        # Gradient of soft objective is proportional to 2*h/tau
        log_factor = -eta * h / (tau * f + 1e-30)
        log_factor = np.clip(log_factor, -80, 80)
        f_new = f * np.exp(log_factor)

        # Renormalize
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s_new
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            continue
        f_new = f_new / s_new

        conv_new = fft_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            eta *= 0.9995
        else:
            no_improve += 1
            eta *= 0.75
            if eta < 0.01:
                eta = 0.01

        f = f_new

        # Adaptive temperature
        tau *= 0.9998
        min_tau = 1e-9 * conv.max() + 1e-15
        if tau < min_tau:
            tau = min_tau

        # Restart
        if no_improve > 300:
            rng = np.random.default_rng(it * 7 + 3)
            noise_scale = 0.002 + 0.002 * min(no_improve / 500.0, 2.0)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 0)
            f = f / f.sum()
            eta = 0.3
            tau = 0.03 * fft_convolve(f).max()
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    MD iter {it}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, eta={eta:.4f}")

    return best_f, 2 * n * best_obj


def smooth_adam_optimize(f0, time_limit, verbose=False, lr_init=0.003, tau_init_frac=0.05):
    """Adam optimization on smooth log-sum-exp surrogate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fft_convolve(f)
    tau = tau_init_frac * conv.max()
    best_obj = conv.max()
    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    t = 0
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.05:
            break

        conv = fft_convolve(f)
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Gradient via cross-correlation
        h = cross_correlate(f, w)
        grad = (2.0 / tau) * h

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fft_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        tau *= 0.9997
        if tau < 1e-8:
            tau = 1e-8

        if no_improve > 150:
            lr *= 0.8
        if no_improve > 500:
            rng = np.random.default_rng(t * 13 + 7)
            noise_scale = 0.002 + 0.003 * min(no_improve / 1000.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.6
            tau = 0.04 * fft_convolve(f).max()

        if verbose and t % 500 == 0:
            print(f"    Adam iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}")

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f, left=0, right=0)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization with fast LP-FW, mirror descent, and Adam.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Initialization (40s) ===
    print("=== Phase 1: Initialization ===")
    t_phase1_start = time.monotonic()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score) and p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_shifted_gaussian(512, 0.4, 0.3)
        try_seq(parent, "default_init")

    # Try diverse initializations
    inits = []
    for n_try in [512, 640, 768, 1024]:
        inits.append((make_edge_concentrated(n_try, 2.0), f"edge_a{n_try}"))

        inits.append((make_edge_concentrated(n_try, 10.0), f"edge_c{n_try}"))

    for a_init, label in inits:
        if time.monotonic() > deadline - 1600:

    # === Phase 2: Fast LP-FW (main workhorse, ~1000s) ===

    # === Phase 3: Mirror Descent Polish (~300s) ===

            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t3, verbose=True)
