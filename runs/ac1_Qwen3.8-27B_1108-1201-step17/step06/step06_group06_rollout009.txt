def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(f, tau):
    """Gradient of log(sum exp(conv/tau)). FFT-based."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv


def compute_lp_grad(f, p):
    """Gradient of (sum_k (a*a)_k^p)^(1/p). FFT-based."""
    n = len(f)
    conv = fast_convolve(f)
    conv_clipped = np.maximum(conv, 1e-30)
    if p > 1.0:
        w = conv_clipped ** (p - 1)
    else:
        w = np.ones_like(conv)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(conv) / len(conv)
    else:
        w = w / w_sum
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    F = np.sum(conv_clipped ** p) ** (1.0 / p)
    if p > 1.0:
        scale = 2.0 / (F ** (p - 1.0))
    else:
        scale = 2.0
    grad = scale * h
    return grad, conv, F

def build_hankel_matrix(p, n):
    """Build the Hankel matrix Q where Q[j,l] = p[j+l]."""
    Q = np.zeros((n, n))
    for j in range(n):
        for l in range(n):
            idx = j + l
            if idx < len(p):
                Q[j, l] = p[idx]
    return Q


def build_hankel_fast(p, n):
    """Build Hankel matrix Q[j,l] = p[j+l] using scipy or direct."""
    Q = np.zeros((n, n))
    indices = np.arange(n)
    idx_matrix = indices[:, None] + indices[None, :]
    valid = idx_matrix < len(p)
    Q[valid] = p[idx_matrix[valid]]
    return Q


def solve_inner_qp_cvxpy(Q, n, solve_time=2.0):
    """Solve min a^T Q a s.t. a >= 0, sum(a) = 1 using cvxpy.
    Returns None if not PSD or solver fails."""
    try:
        # Check if Q is PSD (approximately)
        Q_sym = 0.5 * (Q + Q.T)
        # For large n, use a quick check
        if n <= 2000:
            eigvals = np.linalg.eigvalsh(Q_sym)[:5]
            if eigvals[0] < -1e-6:
                # Not PSD, use SCS which can handle indefinite
                pass
        a = cp.Variable(n)
        # Use quad_form which works for symmetric matrices
        objective = cp.Minimize(cp.quad_form(a, cp.psd_wrap(Q_sym)))
        constraints = [a >= 0, cp.sum(a) == 1]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-7)
        if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
            result = np.maximum(a.value, 0)
            s = result.sum()
            if s > 1e-10:
                return result / s
    except Exception:
        pass
    return None


def minimax_qp(f0, time_limit, verbose=False):

def coordinate_descent_polish(f0, time_limit, verbose=False):

            i = rng.integers(0, n)
            j = rng.integers(0, n)

                continue

            if no_improve > 50:
                rng2 = np.random.default_rng(no_improve * 7)
                f = project_simplex(best_f + rng2.standard_normal(n) * 0.001)
                no_improve = 0
        else:
            no_improve = 0
        if verbose and it % 50 == 0:
            print(f"    CD iter {it}: obj={2*n*best_obj:.8f}")
        it = it + 1 if 'it' in dir() else 1
    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, center_frac=0.5, sigma_frac=0.25):
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = 1.0 - t ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, center_frac=0.4, left_sigma=0.2, right_sigma=0.35):
    x = np.arange(n)
    center = n * center_frac
    left = np.exp(-0.5 * np.minimum(x, center) ** 2 / (n * left_sigma) ** 2)
    right = np.exp(-0.5 * np.maximum(x - center, 0) ** 2 / (n * right_sigma) ** 2)
    a = left * right
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_mixture(n, k=3):
    """Mixture of Gaussians."""
    rng = np.random.default_rng(42 + k)
    centers = rng.uniform(0.2, 0.8, k)
    sigmas = rng.uniform(0.1, 0.3, k)
    weights = rng.dirichlet(np.ones(k))
    x = np.arange(n) / n
    a = np.zeros(n)
    for c, s_, w in zip(centers, sigmas, weights):
        a += w * np.exp(-0.5 * ((x - c) / s_) ** 2)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a

        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    parent = None

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")

    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.5, 0.25)

    # Diverse initializations
