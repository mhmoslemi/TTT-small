"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using entropic Frank-Wolfe with
adaptive smoothing, multi-modal exploration, and peak-targeted coordinate descent.
Key innovations: entropic (multiplicative) FW updates, hybrid exact/smoothed
subgradient, structured multi-modal initializations, and adaptive time allocation.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, tau):
    """Compute gradient of log-sum-exp smooth approximation of max-convolution."""
    n = len(a)
    b = fast_convolve(a)
    scaled = b / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, b


def compute_exact_subgrad(a, b, rel_tol=0.01):
    """Compute exact subgradient using active set (lags within rel_tol of max)."""
    n = len(a)
    M = b.max()
    threshold = M * (1 - rel_tol)
    active = np.where(b >= threshold)[0]
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += 2.0 * a[k - j]
    g /= len(active)
    return g


def hybrid_gradient(a, b, tau, n_active_thresh=5):
    """Use exact subgradient when active set is small, else smoothed."""
    M = b.max()
    active_count = np.sum(b >= M * 0.999)
    if active_count <= n_active_thresh:
        return compute_exact_subgrad(a, b, rel_tol=0.001)
    else:
        g, _ = compute_smooth_grad(a, tau)
        return g


def entropic_step(a, g, eta):
    """Entropic (multiplicative) FW step."""
    scaled = -g / eta
    scaled -= scaled.max()
    v = a * np.exp(scaled)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return v / s


def euclidean_fw_direction(a, g):
    """Euclidean FW: return vertex e_{argmin g}."""
    j = np.argmin(g)
    v = np.zeros(len(a))
    v[j] = 1.0
    return v


def line_search(a, v, b_max, max_backtrack=25):
    """Backtracking line search along FW direction."""
    best_gamma = 0.0
    best_a = a.copy()
    best_val = b_max
    gamma = 1.0
    for _ in range(max_backtrack):
        a_new = a + gamma * (v - a)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            gamma *= 0.5
            continue
        a_new = a_new / s
        b_new = fast_convolve(a_new)
        val = b_new.max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            best_gamma = gamma
        gamma *= 0.6
        if gamma < 1e-6:
            break
    return best_a, best_val

def _solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def peak_targeted_cd(f0, time_limit):
    """Coordinate descent targeting peak-contributing elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)

    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        M = b.max()
        improved = False

        # Focus on top 3 peaks
        top_k = min(3, 2 * n - 1)
        peak_indices = np.argsort(b)[-top_k:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max

            if j_max == j_other:
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue
                for factor in [0.5, 0.7, 0.85]:
                    trial = f.copy()
                    trial[j_max] *= factor
                    diff = old_val - trial[j_max]
                    mask = np.ones(n, dtype=bool)
                    for pk2 in peak_indices[:2]:
                        js2 = max(0, int(pk2) - n + 1)
                        je2 = min(n - 1, int(pk2))
                        mask[js2:je2 + 1] = False
                    mask[j_max] = False
                    if mask.sum() > 0 and trial[mask].sum() > 1e-15:
                        trial[mask] += diff * f[mask] / f[mask].sum()
                    trial = np.maximum(trial, 0)
                    st = trial.sum()
                    if st > 1e-15:
                        trial /= st
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
            else:
                old_jm = f[j_max]
                for scale in [0.6, 0.8]:
                    trial = f.copy()
                    diff = old_jm * (1 - scale)
                    trial[j_max] *= scale
                    mask = np.ones(n, dtype=bool)
                    mask[j_max] = False
                    mask[j_other] = False
                    if mask.sum() > 0 and trial[mask].sum() > 1e-15:
                        trial[mask] += diff * f[mask] / f[mask].sum()
                    trial = np.maximum(trial, 0)
                    st = trial.sum()
                    if st > 1e-15:
                        trial /= st
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0
        else:
            no_improve = 0

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_two_bump(n, gap_frac=0.3, width_frac=0.2):
    """Two-bump initialization."""
    x = np.arange(n) / n
    c1 = 0.35
    c2 = 0.35 + gap_frac
    s1 = width_frac / 2
    s2 = width_frac / 2
    a = np.exp(-0.5 * ((x - c1) / s1) ** 2) + np.exp(-0.5 * ((x - c2) / s2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_comb(n, period=4, duty=0.5):
    """Comb pattern initialization."""
    a = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        if phase < duty:
            a[i] = 1.0
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_smooth_comb(n, period=8, width_frac=0.3):
    """Smooth comb with Gaussian bumps at regular intervals."""
    a = np.zeros(n)
    num_bumps = max(2, n // period)
    for i in range(num_bumps):
        center = (i + 0.5) * n / num_bumps
        sigma = n * width_frac / num_bumps
        x = np.arange(n)
        a += np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.25):
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    x = np.arange(n) - n / 2.0
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization (60s)
    Phase 2: Entropic FW main optimization (700s)
    Phase 3: LP-FW refinement (350s)
    Phase 4: Coordinate descent (300s)
    Phase 5: Multi-length + final polish (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    t_phase1 = 60

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_asymmetric_gaussian(768, 0.35, 0.25)
        try_seq(parent, "default_init")

    # Diverse initializations
    n_base = len(parent) if parent is not None else 768
    inits = []
    for n_try in [n_base, 1024, 640]:
        inits.append((make_asymmetric_gaussian(n_try, 0.3, 0.22), f"gauss30_{n_try}"))
        inits.append((make_asymmetric_gaussian(n_try, 0.5, 0.18), f"gauss50_{n_try}"))
        inits.append((make_bell(n_try, 0.45), f"bell_{n_try}"))
        inits.append((make_two_bump(n_try, 0.3, 0.2), f"2bump_{n_try}"))
        inits.append((make_smooth_comb(n_try, 8, 0.25), f"comb_{n_try}"))
        inits.append((np.ones(n_try) / n_try, f"uniform_{n_try}"))

    for i, (a_init, label) in enumerate(inits):
        if time.monotonic() > deadline - (1800 - t_phase1 - 20):
            break
        sc = try_seq(a_init, label)

    work_seq = best_seq if best_seq is not None else parent
    work_n = len(work_seq)
    print(f"  Best init: n={work_n}, score={best_score:.6f}")

    # === Phase 2: Entropic FW (main) ===
    print(f"\n=== Phase 2: Entropic FW (n={work_n}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 700)
    if t2 > 60:
        print(f"  Time: {t2:.0f}s")
        a_opt, obj = entropic_fw_optimize(work_seq, time_limit=t2 * 0.55, verbose=True)
        try_seq(a_opt, "efw_main")

        if time.monotonic() < deadline - 600:
            rem2 = deadline - time.monotonic() - 600
            t2b = min(rem2, 250)
            a_opt2, obj2 = entropic_fw_optimize(best_seq, time_limit=t2b, verbose=False)
            try_seq(a_opt2, "efw_fine")

    # === Phase 3: LP-FW refinement ===
    print(f"\n=== Phase 3: LP-FW Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 450
        if rem > 60:
            t3 = min(rem, 350)
            print(f"  Time: {t3:.0f}s")
            a_opt, obj = lpfw_optimize(best_seq, time_limit=t3 * 0.6, verbose=True)
            try_seq(a_opt, "lpfw")

            if time.monotonic() < deadline - 300:
                rem2 = deadline - time.monotonic() - 300
                t3b = min(rem2, 120)
                perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.004)
                a_opt2, obj2 = lpfw_optimize(perturbed, time_limit=t3b, verbose=False)
                try_seq(a_opt2, "lpfw_pert")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Peak-Targeted CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t4 = min(rem, 250)
            print(f"  Time: {t4:.0f}s")
            a_opt, obj = peak_targeted_cd(best_seq, time_limit=t4)
            try_seq(a_opt, "peak_cd")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length + Polish ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 1536, 1280, 640]:
            rem = deadline - time.monotonic() - 40
            if rem < 30:
                break
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            t5 = min(rem * 0.3, 60)
            if t5 > 15:
                a_opt, obj = adam_smooth_optimize(f_interp, time_limit=t5, lr_init=0.003)
                try_seq(a_opt, f"adam_n{target_n}")

    # === Phase 6: Final Adam polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t6 = min(rem, 45)
            a_opt, obj = adam_smooth_optimize(
                best_seq, time_limit=t6,
                tau_init=0.015 * fast_convolve(best_seq).max(),
                lr_init=0.001)
            try_seq(a_opt, "final_polish")

    # === Phase 7: Final CD squeeze ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t7 = min(rem, 20)
            a_opt, obj = peak_targeted_cd(best_seq, time_limit=t7)
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
