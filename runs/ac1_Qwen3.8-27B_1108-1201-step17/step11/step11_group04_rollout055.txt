**Spectral Flattening (Gerchberg-Saxton):**

The idea: In the frequency domain, the convolution c = a * a corresponds to |A(ω)|^2 where A = FFT(a). We want |A(ω)|^2 to be as flat as possible. So we iteratively:
1. FFT(a) → A
2. Modify |A| to be flatter (e.g., replace |A| with a power-averaged version)
3. IFFT → a_new
4. Project a_new to be non-negative (clip negatives to 0)
5. Normalize

This is the standard "spectral flattening" or "white noise" algorithm.

The key parameter is how aggressively to flatten. A common choice is:
|A_new| = |A|^alpha for alpha < 1 (compresses dynamic range)

Or: |A_new| = |A|^(1-t) * target^t for a blend parameter t.

**My implementation will use the power method:**
|A_new[k]| = |A[k]|^alpha for k ≠ 0 (keep DC component)
where alpha is in (0, 1).

Then IFFT and clip.

This is very fast (one FFT + one IFFT per iteration) and can make significant global improvements.

**Combining spectral flattening with mirror descent:**

We can alternate:
- Do a few spectral flattening steps (fast global improvement)
- Do a few mirror descent steps (local improvement)
- Repeat

This might escape local minima better than either alone.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetry exploitation (a[i]=a[n-1-i]) to halve effective problem size
2. Fast accelerated mirror descent with FFT-based convolution (O(n log n) per iter)
3. Spectral flattening (Gerchberg-Saxton) for global improvement
4. LogSumExp L-BFGS with aggressive temperature annealing for polishing
5. Direct peak shaving for final refinement
6. Multi-n exploration and adaptive time allocation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_from_padded(a_padded, fft_size):
    """Convolution from pre-allocated padded array."""
    A = rfft(a_padded, fft_size)
    size = len(a_padded)
    return irfft(A * A, fft_size)[:2 * size - 1]


def symmetrize(a):
    """Enforce a[i] = a[n-1-i] via averaging."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1].copy())
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_sym / s


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def subgradient_at_k(a, k):
    """Compute subgradient of c_k = conv(a,a)[k] w.r.t. a.
    Result: g[j] = 2*a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def top_k_indices(arr, k):
    """Get indices of top k values."""
    if k >= len(arr):
        return np.arange(len(arr))
    # Use argpartition for speed
    part = np.argpartition(-arr, k)[:k]
    return part


def accelerated_mirror_descent(a_init, time_limit, use_symmetry=True, verbose=True):
    """
    Nesterov-accelerated mirror descent (multiplicative weights) for minimizing
    max_k conv(a,a)[k] over the simplex with non-negativity.
    Uses FFT for O(n log n) convolution per step.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    if use_symmetry:
        a = symmetrize(a)

    y = a.copy()  # lookahead point
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    mu = 1.0  # Nesterov momentum
    eta0 = 0.5
    K_start = min(100, 2 * n - 1)
    K_min = 3
    no_improve = 0
    patience = 50000
    step_scale = 1.0

    fft_size = 1 << (2 * n - 1).bit_length()
    a_padded = np.zeros(fft_size)

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute convolution at y
        a_padded[:] = 0
        a_padded[:n] = y
        c = fast_convolve_from_padded(a_padded, fft_size)

        max_c = c.max()
        if it % 50000 == 0:
            elapsed = time.monotonic() - t0
            score_y = 2.0 * n * max_c
            print(f"    AMD iter={it} ({elapsed:.0f}s): score_a={best_score:.8f} score_y={score_y:.8f} K={max(K_min, K_start // (1 + it//10000))}")

        # Adaptive K: start with many peaks, decrease over time
        K = max(K_min, K_start // (1 + it // 20000))

        # Get top-K peaks
        if K >= len(c):
            active = np.arange(len(c))
        else:
            # Use threshold-based selection for efficiency
            threshold = max_c * (1.0 - 1e-4 * K)
            active = np.where(c >= threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            elif len(active) > K:
                active = top_k_indices(c, K)

        # Compute weighted subgradient
        g = np.zeros(n)
        for k_idx in active:
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * y[k_idx - idx]
        g /= len(active)

        # Step size with decay
        eta = eta0 * step_scale / np.sqrt(1.0 + it * 0.1)

        # Mirror descent step from y
        log_ratio = -eta * g
        log_ratio -= log_ratio.max()  # numerical stability
        a_new = y * np.exp(log_ratio)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        if use_symmetry:
            a_new = symmetrize(a_new)

        # Nesterov momentum update
        mu_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * mu * mu))
        beta = (1.0 - mu) / mu_new if mu_new > 0 else 0.0
        y_new = a_new + beta * (a_new - a)
        y_new = np.maximum(y_new, 0)
        if use_symmetry:
            y_new = symmetrize(y_new)
        else:
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = a_new.copy()
            else:
                y_new = y_new / s_y

        a = a_new
        y = y_new
        mu = mu_new

        # Track best
        a_padded[:] = 0
        a_padded[:n] = a
        c_a = fast_convolve_from_padded(a_padded, fft_size)
        score_a = 2.0 * n * c_a.max()

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive: reduce step size if stuck
        if no_improve >= patience:
            step_scale *= 0.7
            no_improve = 0
            # Perturb to escape
            rng = np.random.default_rng(it)
            noise = 0.01 * step_scale * rng.standard_normal(n)
            if use_symmetry:
                noise = 0.5 * (noise + noise[::-1].copy())
            y = y + noise
            y = np.maximum(y, 0)
            if use_symmetry:
                y = symmetrize(y)
            else:
                s_y = y.sum()
                if s_y > 1e-15:
                    y = y / s_y
            if step_scale < 0.01:
                break

    return best_a, best_score


def spectral_flatten(a, n_iters=100, blend=0.4, use_symmetry=True, deadline=None):
    """
    Iterative spectral flattening. Flattens |FFT(a)| to reduce peak convolution.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    if use_symmetry:
        a = symmetrize(a)

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        # Flatten magnitudes (keep DC)
        mags = np.abs(F)
        non_dc = mags[1:L // 2 + 1]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * np.angle(F[k]))
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * np.angle(F[L - k]))

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        if use_symmetry:
            a_new = symmetrize(a_new)
        else:
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        a = a_new

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. log-domain z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    a = np.maximum(a0, 1e-300)
    z = np.log(a).copy()
    if deadline is None:
        deadline = time.monotonic() + time_limit

    n_stages = len(T_stages)
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        stages_left = n_stages - stage_i
        avail_time = remaining
        iters = max(20, min(500, int(avail_time * 30 / max(1, stages_left))))

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            o, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return o

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out /= a_out.sum()
    a_out = np.maximum(a_out, 0)
    return a_out


def peak_shave_fine(a0, time_limit, use_symmetry=True):
    """
    Very fine-grained multiplicative peak shaving.
    Targets the exact argmax with small steps.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    if use_symmetry:
        a = symmetrize(a)

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.005
    scale = 1.0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        k_star = np.argmax(b)

        # Subgradient at the single peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g = np.zeros(n)
        g[idx] = 2.0 * a[k_star - idx]

        # Small multiplicative step
        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        if use_symmetry:
            a_new = symmetrize(a_new)
        else:
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 200:
            scale *= 0.7
            no_improve = 0
            if scale < 0.001:
                break

    return best_a, best_score


def structured_init(n, rng, use_symmetry=True):
    """Create a diverse structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-3 * (x - 0.5) ** 2) + 0.3 * np.exp(-6 * (x - 0.25) ** 2) + 0.3 * np.exp(-6 * (x - 0.75) ** 2)
    elif choice == 5:
        a = (1 + np.cos(2 * np.pi * 3 * x)) / 2
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / (n / 4.0)
        a = np.sinc(t / 2.0)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 50.0 * (x - 0.5) ** 4)
    elif choice == 8:
        a = np.ones(n) * 0.1 + 0.9 * np.exp(-2.0 * (x - 0.5) ** 2)
    else:
        a = np.abs(np.sin(np.pi * x * 2)) ** 0.5

    a = np.maximum(a, 0)
    if use_symmetry:
        a = symmetrize(a)
    else:
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        a = a / s
    return a


def interpolate_sequence(f, new_n, use_symmetry=True):
    """Interpolate sequence to new length preserving normalization and symmetry."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    if use_symmetry:
        f_interp = symmetrize(f_interp)
    else:
        ss = f_interp.sum()
        if ss > 1e-15:
            f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Key innovations:
    - Symmetry exploitation (a[i]=a[n-1-i])
    - Very fast accelerated mirror descent (O(n log n) per iteration)
    - Combined spectral + mirror descent for global + local improvement
    - Aggressive L-BFGS polishing with very low temperatures
    """
    USE_SYM = True
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, USE_SYM)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        if USE_SYM:
            a = symmetrize(a)
        else:
            s = a.sum()
            if s < 0.01:
                return
            a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)}) at {time.monotonic()-t_start:.0f}s")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}, n={n_best}")

    # === Phase 0: Quick spectral flattening (30s) ===
    print(f"\n=== Phase 0: Spectral Flattening ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_spec = min(30.0, rem * 0.03)
        a_spec = spectral_flatten(best_seq, n_iters=200, blend=0.35,
                                  use_symmetry=USE_SYM, deadline=time.monotonic() + t_spec)
        local_update(a_spec, "spec_init")
        a_spec2 = spectral_flatten(best_seq, n_iters=100, blend=0.5,
                                   use_symmetry=USE_SYM, deadline=time.monotonic() + min(15, t_spec))
        local_update(a_spec2, "spec_strong")

    # === Phase 1: Accelerated Mirror Descent (main workhorse, ~70% of time) ===
    print(f"=== Phase 1: Accelerated Mirror Descent n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_amd = min(rem * 0.55, 900.0)
    if t_amd > 30:
        print(f"  Budget: {t_amd:.0f}s")
        a_amd, _ = accelerated_mirror_descent(best_seq, t_amd, use_symmetry=USE_SYM)
        local_update(a_amd, "amd_main")

        # Second round with different parameters
        rem = deadline - time.monotonic()
        if rem > 60:
            t_amd2 = min(rem * 0.15, 150.0)
            a_amd2, _ = accelerated_mirror_descent(best_seq, t_amd2, use_symmetry=USE_SYM)
            local_update(a_amd2, "amd_round2")

    # === Phase 2: Spectral + AMD alternating (20% of time) ===
    print(f"=== Phase 2: Spectral + AMD Alternating (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t_alt = min(rem * 0.20, 300.0)
        print(f"  Budget: {t_alt:.0f}s")
        current = best_seq.copy()

        for round_i in range(10):
            sub_rem = deadline - time.monotonic()
            if sub_rem < 20:
                break
            t_sub_spec = min(sub_rem * 0.1, 15.0)
            t_sub_amd = min(sub_rem * 0.3, 60.0)

            # Spectral step
            a_sp = spectral_flatten(current, n_iters=50, blend=0.3,
                                    use_symmetry=USE_SYM, deadline=time.monotonic() + t_sub_spec)
            local_update(a_sp, f"alt_spec_{round_i}")
            current = best_seq.copy() if best_seq is not None else current

            # AMD step
            sub_rem = deadline - time.monotonic()
            if sub_rem < 15:
                break
            a_amd = accelerated_mirror_descent(current, min(t_sub_amd, sub_rem * 0.5),
                                               use_symmetry=USE_SYM)
            local_update(a_amd, f"alt_amd_{round_i}")
            current = best_seq.copy() if best_seq is not None else current

    # === Phase 3: LogSumExp L-BFGS polish ===
    print(f"=== Phase 3: L-BFGS LogSumExp Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.20, 350.0)
    if t_lbfgs > 30:
        # Multi-stage temperature annealing
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005,
                    0.003, 0.002, 0.001]
        print(f"  T_stages: {len(T_stages)}, budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages,
                            deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs_fine")

        # If symmetry was used, also try without symmetry
        rem = deadline - time.monotonic()
        if rem > 60:
            T_stages_ns = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_lb_ns = lbgfs_refine(best_seq, min(rem * 0.3, 120.0), T_stages_ns,
                                   deadline=time.monotonic() + min(rem * 0.3, 120.0))
            local_update(a_lb_ns, "lbgfs_nosym")

    # === Phase 4: Direct peak shaving (fine) ===
    print(f"=== Phase 4: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_ps = min(rem * 0.10, 120.0)
    if t_ps > 10:
        print(f"  Budget: {t_ps:.0f}s")
        a_ps, _ = peak_shave_fine(best_seq, t_ps, use_symmetry=USE_SYM)
        local_update(a_ps, "peak_shave")

        # Without symmetry too
        rem = deadline - time.monotonic()
        if rem > 30:
            a_ps2, _ = peak_shave_fine(best_seq, min(rem * 0.3, 60.0), use_symmetry=False)
            local_update(a_ps2, "peak_shave_ns")

    # === Phase 5: Try different n ===
    print(f"=== Phase 5: Different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for target_n in [1000, 1500, 800]:
        if rem < 80:
            break
        if target_n == n_best:
            continue
        t_try = min(rem * 0.15, 150.0)
        if t_try < 30:
            continue
        print(f"  Trying n={target_n}, budget={t_try:.0f}s")
        init_n = interpolate_sequence(best_seq, target_n, USE_SYM)
        # Quick spectral + AMD
        a_sp = spectral_flatten(init_n, n_iters=50, blend=0.3,
                                use_symmetry=USE_SYM, deadline=time.monotonic() + min(10, t_try))
        a_try, _ = accelerated_mirror_descent(a_sp, min(t_try - 15, t_try * 0.7),
                                              use_symmetry=USE_SYM)
        local_update(a_try, f"amd_n{target_n}")

        # Quick L-BFGS if it improved
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == target_n and rem > 40:
            T_quick = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_tlb = lbgfs_refine(best_seq, min(rem * 0.3, 60.0), T_quick,
                                 deadline=time.monotonic() + min(rem * 0.3, 60.0))
            local_update(a_tlb, f"lbgfs_n{target_n}")

    # === Phase 6: Final polish ===
    print(f"=== Phase 6: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very fine L-BFGS
        t_fin = min(rem * 0.3, 60.0)
        if t_fin > 10:
            T_fin = [1.0, 0.5, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
            a_fin = lbgfs_refine(best_seq, t_fin, T_fin,
                                 deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Final peak shave
        t_ps_fin = min(rem * 0.4, 30.0)
        if t_ps_fin > 5:
            a_ps_fin, _ = peak_shave_fine(best_seq, t_ps_fin, use_symmetry=USE_SYM)
            local_update(a_ps_fin, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        # One more AMD round with very small steps
        t_amd_fin = min(rem * 0.6, 20.0)
        if t_amd_fin > 5:
            a_amd_fin, _ = accelerated_mirror_descent(best_seq, t_amd_fin, use_symmetry=USE_SYM)
            local_update(a_amd_fin, "final_amd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        try:
            test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
