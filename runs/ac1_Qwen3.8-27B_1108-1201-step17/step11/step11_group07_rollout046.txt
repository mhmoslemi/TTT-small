```python
import cvxpy as cp
import numpy as np

def solve_lp(a, M):
    """Solve the LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b))
    constraints = []
    
    # Convolution constraints: (a*b)[k] <= M for k = 0..2n-2
    for k in range(2*n - 1):
        # (a*b)[k] = sum_{m: 0<=k-m<n} a[m]*b[k-m]
        # = sum_{m: max(0,k-n+1) <= m <= min(n-1,k)} a[m]*b[k-m]
        terms = []
        for m in range(max(0, k-n+1), min(n, k+1)):
            terms.append(a[m] * b[k-m])
        constraints.append(cp.sum(terms) <= M)
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or SCS, CLARAB etc.
    
    if b.value is None:
        return None
    return np.maximum(b.value, 0)
```
