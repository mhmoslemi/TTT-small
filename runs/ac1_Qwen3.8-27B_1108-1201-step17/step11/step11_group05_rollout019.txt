def fft_size_for(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()


def fast_conv(a):
    """Linear auto-convolution via FFT. Input: 1D array."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fs = fft_size_for(size)
    A = rfft(a, fs)
    return irfft(A * A, fs)[:size]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fs = fft_size_for(size)
    A = rfft(a, fs)
    B = rfft(b, fs)
    return irfft(A * B, fs)[:size]


def score_of(a):
    """Compute score 2*n*max(conv(a,a))/(sum(a))^2 locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

    if s < 1e-10:

    a = a / s

def solve_lp_reduced(a_fixed, k_indices, warm_b=None):

    M_var = cp.Variable()

    rows = []
    cols = []
    vals = []
    n_k = len(k_indices)

    T = np.zeros((n_k, n))
    for idx, k in enumerate(k_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[idx, j] = a_fixed[j]

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:

            continue

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:

        else:
            max_b_all = b.max()
            w = np.exp((b - max_b_all) / max(0.01, eta * max_b * 0.1))
            w = w / w.sum()

        # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] for valid k-i in [0, n-1]
        # This is 2 * cross-correlation of w and a
        # Use FFT: correlate(w, a) at lag i = sum_k w[k] * a[k-i]
        fs = fft_size_for(2 * n)
        W = rfft(w, fs)
        A_rev = rfft(a[::-1], fs)
        g_raw = irfft(W * A_rev, fs)[:2 * n - 1]
        g = 2.0 * g_raw[:n]

        # Mirror descent (multiplicative weights) update
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = normalize_simplex(a_new)
        a = a_new

        if verbose and (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step {step+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def nesterov_alt_lp(a0, time_limit, solver_name=None, verbose=True):
    """
    Alternating LP with Nesterov dual averaging.
    Player 1: a (minimizer), Player 2: b (maximizer in the max).
    """
    a = normalize_simplex(a0)
    n = len(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()

    y = a.copy()  # Nesterov momentum variable
    warm_b = None
    lp_times = []
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t = time.monotonic()
        # Determine which constraints to include
        b_conv = fast_conv(y)
        max_bc = b_conv.max()
        # Top-k active constraints
        k_frac = min(1.0, 50.0 / max(1, it))
        k_top = max(20, int(k_frac * (2 * n - 1)))
        if k_top < 2 * n - 1 and it > 50:
            idx = np.argpartition(b_conv, -k_top)[-k_top:]
            b_resp = solve_lp_reduced(y, idx, warm_b)
        else:
            b_resp = solve_lp_full(y, warm_b)

        lp_times.append(time.monotonic() - lp_t)
        if b_resp is None:
            continue

        warm_b = b_resp
        avg_lp = np.mean(lp_times[-10:]) if len(lp_times) >= 10 else np.mean(lp_times)

        if avg_lp > 3.0 and it > 20:
            if verbose:
                print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping at iter {it+1}")
            break

        # Nesterov dual averaging step
        alpha = 2.0 / (it + 2)
        y_new = (1.0 - alpha) * y + alpha * b_resp
        y_new = normalize_simplex(y_new)

        a_new = ((it + 1) * a + alpha * (it + 2) * b_resp) / (it + 1 + alpha * (it + 2))
        a_new = normalize_simplex(a_new)

        score_a = 2.0 * n * fast_conv(a_new).max()
        score_y = 2.0 * n * fast_conv(y_new).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y_new.copy()

        a = a_new
        y = y_new

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 30:
                # Perturb to escape
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = normalize_simplex(y + noise)
                a = normalize_simplex(a + noise)
                stall = 0

            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={avg_lp*1000:.0f}ms")

        if len(lp_times) > 30 and np.mean(lp_times[-30:]) > 5.0:
            if verbose:
                print(f"    LP consistently slow, stopping at iter {it+1}")
            break

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """
    Coordinate descent: for each coordinate i, try transferring mass from i to j
    (and vice versa) to reduce the peak autocorrelation.
    """
    a = normalize_simplex(a0)
    n = len(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    improved = True
    rng = np.random.default_rng(777)

    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        b = fast_conv(a)
        max_b = b.max()
        k_peak = int(np.argmax(b))

        # Try transferring from high-value coordinates to low-value ones
        # Focus on coordinates that contribute to the peak
        # grad of b[k_peak] at position i = 2*a[k_peak - i] (if valid)
        # So positions i where a[k_peak-i] is large contribute most to the peak

        # Try a sweep: for each i, try moving a small amount from i to a random j
        step_frac = 0.01

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-8:
                continue

            # Try moving mass from i to i+1 (local transfer)
            for j in [i + 1, i - 1, (i + n // 2) % n]:
                j = j % n
                if j == i or a[j] < 0:
                    continue
                # Transfer delta from i to j
                delta = a[i] * step_frac
                if delta < 1e-10:
                    continue
                a_trial = a.copy()
                a_trial[i] -= delta
                a_trial[j] += delta
                if a_trial[i] < 0:
                    continue
                a_trial = normalize_simplex(a_trial)
                sc = 2.0 * n * fast_conv(a_trial).max()
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break

        if verbose and time.monotonic() - t0 - t0 > 60:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): score={best_score:.8f}")

        if not improved:
            # Try random transfers
            for _ in range(100):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or a[i] < 1e-8:
                    continue
                delta = a[i] * step_frac * rng.uniform(0.5, 2.0)
                a_trial = a.copy()
                a_trial[i] -= delta
                a_trial[j] += delta
                if a_trial[i] < 0:
                    continue
                a_trial = normalize_simplex(a_trial)
                sc = 2.0 * n * fast_conv(a_trial).max()
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True

    return best_a, best_score


def structured_init(n, rng, style=None):
    """Generate diverse structured initializations."""
    x = np.linspace(0, 1, n)
    if style is None:
        style = rng.integers(0, 10)

    if style == 0:
        # Dome (Gaussian center)
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif style == 1:
        # Parabolic dome
        a = 1.0 - 4.0 * (x - 0.5) ** 2
    elif style == 2:
        # U-shape (edge concentrated)
        a = (2.0 * x - 1.0) ** 4
    elif style == 3:
        # Cosine dome
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif style == 4:
        # Multi-peak
        a = np.exp(-20*(x-0.25)**2) + np.exp(-20*(x-0.75)**2) + 0.3*np.ones(n)
    elif style == 5:
        # Oscillatory decay
        a = np.exp(-3*x) * (0.5 + 0.5*np.cos(2*np.pi*5*x))
        a = np.maximum(a, 0)
    elif style == 6:
        # Sinc-like (main lobe)
        t = 2.0 * x - 1.0
        a = np.sinc(3.0 * t) ** 2
        a = np.maximum(a, 0)
    elif style == 7:
        # Legendre polynomial based
        a = 1.0 - 1.2*(2*x-1)**2 + 0.5*(2*x-1)**4
        a = np.maximum(a, 0)
    elif style == 8:
        # Concentrated center
        a = np.exp(-8*(x-0.5)**2) + 0.1
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def logsumexp_nesterov(a0, time_limit, T_initial=1.0, T_final=0.01, verbose=True):
    """
    Nesterov-style accelerated mirror descent on LogSumExp approximation.
    Works in log-space for the simplex constraint.
    """
    a = normalize_simplex(a0)
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z -= z.max()
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()

    # Nesterov parameters
    t_nest = 1.0
    z_prev = z.copy()
    z_extrap = z.copy()

    n_steps = 50000
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal temperature
        frac = step / n_steps
        T = T_initial * (T_final / T_initial) ** frac

        # Extrapolation (Nesterov momentum)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest ** 2))
        momentum = (t_nest - 1.0) / t_new
        z_extrap = z + momentum * (z - z_prev)
        z_prev = z.copy()
        t_nest = t_new

        # Convert to simplex
        z_shifted = z_extrap - z_extrap.max()
        exp_z = np.exp(z_shifted)
        a_curr = exp_z / exp_z.sum()
        a_curr = np.maximum(a_curr, 0)
        a_curr = a_curr / a_curr.sum()

        # Compute convolution
        b = fast_conv(a_curr)
        max_b = b.max()

        # LogSumExp approximation
        shifted = (b - max_b) / T
        exp_shifted = np.exp(shifted)
        lse = max_b + T * np.log(exp_shifted.sum())
        cur_score = 2.0 * n * lse

        real_score = 2.0 * n * max_b
        if real_score < best_score:
            best_score = real_score
            best_a = a_curr.copy()

        # Gradient of LogSumExp: (1/T) * sum_k w[k] * grad b[k]
        # where w[k] = exp(b[k]/T) / sum exp(b[j]/T)
        w = exp_shifted / exp_shifted.sum()

        # grad b[k] w.r.t. a[i] = 2*a[k-i] for valid k-i
        # Combined: grad[i] = (1/T) * 2 * sum_k w[k] * a[k-i]
        fs = fft_size_for(2 * n)
        W = rfft(w, fs)
        A_rev = rfft(a_curr[::-1], fs)
        g_raw = irfft(W * A_rev, fs)[:2 * n - 1]
        g_a = (2.0 / T) * g_raw[:n]

        # Step size: for smooth function with Lipschitz L ~ 2/T * max(a)
        # Nesterov step size: 1/(k+1) * (1/L) approximately
        L_est = 4.0 / T
        step_size = 1.0 / (L_est * (1.0 + 0.01 * step / 1000))

        # Mirror descent update in log-space
        # a_new[i] = a_curr[i] * exp(-step_size * g_a[i])
        # In log-space: z_new[i] = log(a_curr[i]) - step_size * g_a[i]
        log_a_curr = np.log(np.maximum(a_curr, 1e-30))
        z_new = log_a_curr - step_size * g_a
        z_new -= z_new.max()  # Centering for numerical stability
        z = z_new

        if verbose and (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LogSumExpNest step {step+1} ({elapsed:.0f}s): best={best_score:.8f} T={T:.4f}")

    return best_a, best_score


def multi_n_refine(a_best, time_limit, n_candidates, verbose=True):
    """Try refining at different sequence lengths."""
    best_a = a_best.copy()
    best_score = score_of(a_best)
    base_n = len(a_best)

    for new_n in n_candidates:
        if time.monotonic() - time_limit + time_limit > 0:  # placeholder
            pass
        remaining = time_limit[0] - time.monotonic() if isinstance(time_limit, tuple) else time_limit - time.monotonic()
        if remaining < 30:
            break

        a_interp = interpolate_sequence(a_best, new_n)
        # Quick subgradient
        t_sub = min(remaining * 0.4, 60.0)
        a_opt, sc = adaptive_subgradient(a_interp, t_sub, eta_mult=0.8, top_k_frac=0.15, verbose=False)
        sc = score_of(a_opt)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            if verbose:
                print(f"    MultiN: n={new_n} improved to {sc:.8f}")

        # Quick LP if time allows
        remaining = time_limit[0] - time.monotonic() if isinstance(time_limit, tuple) else time_limit - time.monotonic()
        if remaining > 60:
            t_lp = min(remaining * 0.4, 80.0)
            a_opt2, _ = nesterov_alt_lp(a_opt, t_lp, verbose=False)
            sc2 = score_of(a_opt2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_opt2.copy()
                if verbose:
                    print(f"    MultiN: n={new_n} LP improved to {sc2:.8f}")

    return best_a, best_score


def diverse_restart_optimize(a_parent, time_limit, rng, verbose=True):
    """Try diverse initializations and optimize each briefly, keep best."""
    n = len(a_parent)
    best_a = a_parent.copy()
    best_score = score_of(a_parent)

    styles_to_try = [0, 1, 3, 5, 7, 8, 2]
    steps_per_init = 200

    for style in styles_to_try:
        if time.monotonic() - time_limit > 0 if isinstance(time_limit, (int, float)) else False:
            break
        remaining = time_limit - time.monotonic()
        if remaining < 10:
            break

        a_init = structured_init(n, rng, style=style)
        # Blend with parent
        blend = 0.3
        a_blend = (1 - blend) * a_parent + blend * a_init
        a_blend = normalize_simplex(a_blend)

        t_sub = min(remaining * 0.3, 30.0)
        a_opt, sc = adaptive_subgradient(a_blend, t_sub, eta_mult=1.0, top_k_frac=0.2, verbose=False)
        sc = score_of(a_opt)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            if verbose:
                print(f"    Restart style {style}: score={sc:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. Load parent, quick subgradient warm-up
    2. Diverse restarts for better basin
    3. Main LP alternating (workhorse)
    4. LogSumExp Nesterov refinement
    5. Multi-n exploration
    6. Coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_of(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, style=0)
        best_score = score_of(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_of(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Quick subgradient warm-up ===
    print(f"\n=== Phase 1: Subgradient Warm-up (score={best_score:.8f}) ===")
    t1 = deadline - time.monotonic()
    t_alloc = min(t1 * 0.08, 90.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sub, _ = adaptive_subgradient(best_seq, t_alloc, eta_mult=1.2, top_k_frac=0.2, verbose=True)
        local_update(a_sub, "subgrad_warm")

    # === Phase 2: Diverse restarts ===
    print(f"\n=== Phase 2: Diverse Restarts (score={best_score:.8f}) ===")
    t2 = deadline - time.monotonic()
    t_alloc = min(t2 * 0.10, 120.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_div, _ = diverse_restart_optimize(best_seq, t_start + t_alloc, rng, verbose=True)
        local_update(a_div, "diverse_restart")

    # === Phase 3: Main LP Alternating ===
    print(f"\n=== Phase 3: LP Alternating (score={best_score:.8f}) ===")
    t3 = deadline - time.monotonic()
    t_alloc = min(t3 * 0.30, 450.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_alt_lp(best_seq, t_start + t_alloc, verbose=True)
        local_update(a_lp, "lp_main")

    # === Phase 4: LogSumExp Nesterov ===
    print(f"\n=== Phase 4: LogSumExp Nesterov (score={best_score:.8f}) ===")
    t4 = deadline - time.monotonic()
    t_alloc = min(t4 * 0.20, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lse, _ = logsumexp_nesterov(best_seq, t_start + t_alloc, T_initial=2.0, T_final=0.05, verbose=True)
        local_update(a_lse, "logsumexp_nest")

    # === Phase 5: Second LP round ===
    print(f"\n=== Phase 5: LP Round 2 (score={best_score:.8f}) ===")
    t5 = deadline - time.monotonic()
    t_alloc = min(t5 * 0.20, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_alt_lp(best_seq, t_start + t_alloc, verbose=True)
        local_update(a_lp2, "lp_round2")

    # === Phase 6: Subgradient fine-tune ===
    print(f"\n=== Phase 6: Subgradient Fine-tune (score={best_score:.8f}) ===")
    t6 = deadline - time.monotonic()
    t_alloc = min(t6 * 0.15, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sub2, _ = adaptive_subgradient(best_seq, t_alloc, eta_mult=0.5, top_k_frac=0.05, verbose=True)
        local_update(a_sub2, "subgrad_fine")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration (score={best_score:.8f}) ===")
    t7 = deadline - time.monotonic()
    if t7 > 120:
        n_candidates = [n_best + 80, n_best + 160, n_best + 240]
        # Filter to reasonable range
        n_candidates = [n for n in n_candidates if 800 <= n <= 3000]
        if len(n_candidates) > 0:
            print(f"  Trying n candidates: {n_candidates}")
            t_alloc = min(t7 * 0.25, 300.0)
            a_multi, _ = multi_n_refine(best_seq, t_start + t_alloc, n_candidates, verbose=True)
            local_update(a_multi, "multi_n")

    # === Phase 8: Coordinate descent polish ===
    print(f"\n=== Phase 8: Coordinate Descent Polish (score={best_score:.8f}) ===")
    t8 = deadline - time.monotonic()
    t_alloc = min(t8 * 0.25, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_polish(best_seq, t_start + t_alloc, verbose=True)
        local_update(a_cd, "coord_desc")

    # === Phase 9: Final subgradient with very small step ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    t9 = deadline - time.monotonic()
    t_alloc = min(t9 * 0.5, 60.0)
    if t_alloc > 5:
        a_fin, _ = adaptive_subgradient(best_seq, t_alloc, eta_mult=0.3, top_k_frac=0.03, verbose=False)
        local_update(a_fin, "final_subgrad")

    # Final verification
    if best_seq is not None:
        fs = score_of(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    print(f"Returning sequence of length {len(result)}")
    return result
