Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated 
log-sum-exp gradient descent with Adam optimizer and adaptive 
temperature schedule, followed by coordinate descent polishing.
"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return ifft(A * A, fft_size).real[:size]

def compute_lse_grad(a, T):
    """
    Compute gradient of 2n * LSE_T(conv(a,a)) with respect to a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    b = fast_convolve(a)
    
    # Softmax weights (numerically stable)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    
    # grad_a[m] = 2 * sum_k w[k] * a[k-m] for valid k-m
    # = 2 * cross_correlate(w, a)[m]
    # Using FFT: cross_correlate(w, a) = ifft(fft(w) * conj(fft(a)))
    
    L = 1 << (2 * n).bit_length()  # Must be >= 2n-1
    w_padded = np.zeros(L, dtype=np.float64)
    w_padded[:len(w)] = w
    a_padded = np.zeros(L, dtype=np.float64)
    a_padded[:n] = a
    
    W = fft(w_padded)
    A = fft(a_padded)
    corr = ifft(W * np.conj(A)).real
    
    # corr[m] = sum_k w[k] * a[k-m] (circular, but L >= 2n-1 makes it linear)
    # We need corr[0:n] for the gradient
    grad_a = 2.0 * corr[:n]
    return grad_a, b.max()

def lse_value(a, T):
    """Compute 2n * LSE_T(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_b.sum())
    return 2.0 * n * lse

def true_score(a):
    """Compute the actual objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)

    a = a / s
    
    best_a = a.copy()

    t0 = time.monotonic()

                    m = np.zeros(n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated log-sum-exp
gradient descent with Adam optimizer, adaptive temperature annealing,
coordinate descent polishing, and multi-scale exploration.

Key innovations over prior approach:
- Full log-sum-exp gradient (uses info from ALL convolution positions, not just peak)
- O(n log n) gradient via FFT cross-correlation
- Adam with adaptive learning rate and cosine scheduling
- Aggressive multi-scale exploration to escape local optima
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return ifft(A * A, fft_size).real[:size]


def true_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def compute_lse_grad_and_val(a, T):
    """
    Compute gradient and value of 2n * LSE_T(conv(a,a)).
    Returns (gradient, objective_value, max_b).
    Uses FFT cross-correlation for O(n log n) gradient.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    # Softmax weights
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    
    # LSE value: 2n * (max_b + T * log(sum(exp_b)))
    lse_val = max_b + T * np.log(exp_b.sum())
    obj = 2.0 * n * lse_val
    
    # Gradient via FFT cross-correlation
    # grad_a[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    L = 1 << (3 * n - 1).bit_length()
    w_padded = np.zeros(L, dtype=np.float64)
    w_padded[:len(w)] = w
    a_padded = np.zeros(L, dtype=np.float64)
    a_padded[:n] = a
    
    W = fft(w_padded)
    A = fft(a_padded)
    corr = ifft(W * np.conj(A)).real
    
    grad_a = 2.0 * corr[:n]
    return grad_a, obj, max_b


def adam_lse_optimize(a0, time_limit, T_stages=None, base_lr=0.3, verbose=True):
    """
    Adam optimization on log-sum-exp smoothed objective with temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = true_score(a)
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 30.0, 18.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05]
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0
    no_improve = 0

            if s < 1e-15:

                s = 1.0
            a = a / s

                    no_improve = 0

                    s = a.sum()
                    if s < 1e-15:
                        a = best_a.copy()
                        s = 1.0

def gradient_descent_nesterov(a0, time_limit, T_stages=None, base_lr=0.1):
    """
    Nesterov-accelerated gradient descent on LSE objective.
    Alternative optimizer with different dynamics.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:

        s = 1.0
    a = a / s
    
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_score = true_score(a)
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.25, 0.15, 0.1, 0.07]
    
    rng = np.random.default_rng(777)
    no_improve = 0
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        stage_start = time.monotonic()
        remaining = len(T_stages) - T_stages.index(T)
        time_per_stage = max(5.0, (time_limit - (stage_start - t0)) / remaining)
        
        it = 0
        while time.monotonic() - stage_start < time_per_stage:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov lookahead
            alpha = 2.0 / (np.sqrt(it + 1) + 2.0)
            y = a + alpha * (a - y + a)  # simplified
            y = (1 + alpha) * a - alpha * y
            
            grad_y, _, _ = compute_lse_grad_and_val(y, T)
            
            # Gradient step on y
            eta = base_lr / (np.sqrt(it + 1) + 1.0)
            y = y - eta * grad_y
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
                sy = 1.0
            y = y / sy
            
            # Update a
            a = alpha * y + (1 - alpha) * a
            a = np.maximum(a, 0)
            sa = a.sum()
            if sa < 1e-15:
                a = y.copy()
                sa = 1.0
            a = a / sa
            it += 1
            
            if it % 10 == 0:
                score = true_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 10
                if no_improve > 500:
                    a = best_a + rng.standard_normal(n) * 0.003
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    y = a.copy()
                    no_improve = 0
        
        print(f"    Nesterov T={T:.2f}: score={best_score:.8f}")
    
    return best_a, best_score

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        improved = False
        
        # Try reducing top contributors
        top_m = min(8, len(idx))

            for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98]:

                a_trial = a_trial / a_trial.sum()

def interpolate_seq(f, new_n):
    """Cubic-like interpolation (using linear for stability) to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = true_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, seed)
        best_score = true_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    def try_update(arr, label):
        nonlocal best_seq, best_score
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = true_score(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")
    
    # === Phase 1: Adam LSE (main workhorse) ===
    print(f"\n=== Phase 1: Adam LSE (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 750.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        T_stages_1 = [40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.3, 0.95, 0.7, 0.5, 0.38, 0.28, 0.2, 0.14, 0.1, 0.07, 0.05]
        a1, _ = adam_lse_optimize(best_seq, t1, T_stages=T_stages_1, base_lr=0.4)
        try_update(a1, "adam_p1")
    
    # === Phase 2: Nesterov from perturbed starts ===
    print(f"\n=== Phase 2: Nesterov variants (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 250.0)
    if t2 > 20:
        n_starts = 3
        t_per_start = t2 / n_starts
        for si in range(n_starts):
            if time.monotonic() > deadline - 15:
                break
            # Different perturbation for each start
            if si == 0:
                a_start = best_seq + rng.standard_normal(len(best_seq)) * 0.01
            elif si == 1:
                # Slightly different profile
                x = np.linspace(0, 1, len(best_seq))
                a_start = best_seq * (1.0 + 0.1 * np.cos(4 * np.pi * x))
            else:
                # Mirror-mix
                a_start = 0.7 * best_seq + 0.3 * best_seq[::-1]
                a_start = np.maximum(a_start, 0)
            
            a_start = a_start / a_start.sum()
            t_sub = min(t_per_start, deadline - time.monotonic() - 5)
            if t_sub < 10:
                continue
            T_nest = [20.0, 12.0, 7.0, 4.5, 3.0, 2.0, 1.4, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1]
            a2, _ = gradient_descent_nesterov(a_start, t_sub, T_stages=T_nest, base_lr=0.08)
            try_update(a2, f"nesterov_{si}")
    
    # === Phase 3: Multi-scale exploration ===
    print(f"\n=== Phase 3: Multi-scale (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 30:
        n_best = len(best_seq)
        # Try several different lengths
        candidates = sorted(set([
            max(400, n_best - 400),
            max(500, n_best - 200),
            n_best + 300,
            n_best + 600,
            max(800, int(n_best * 1.5)),
        ]))
        candidates = [c for c in candidates if c != n_best and c <= 6000]
        per_n = t3 / max(1, len(candidates))
        print(f"  Trying n values: {candidates}, {per_n:.0f}s each")
        
        for n_try in candidates:
            if time.monotonic() > deadline - 20:
                break
            a_interp = interpolate_seq(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 10)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            T_sub = [25.0, 15.0, 9.0, 6.0, 4.0, 2.8, 2.0, 1.4, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05]
            a_sub, _ = adam_lse_optimize(a_interp, t_sub, T_stages=T_sub, base_lr=0.35)
            try_update(a_sub, f"scale_{n_try}")
    
    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 200.0)
    if t4 > 15:
        a_cd, _ = coordinate_descent(best_seq, t4)
        try_update(a_cd, "coord_p4")
    
    # === Phase 5: Final Adam LSE refinement ===
    print(f"\n=== Phase 5: Final refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 120.0)
    if t5 > 10:
        T_final = [8.0, 5.0, 3.5, 2.5, 1.8, 1.3, 0.9, 0.65, 0.45, 0.32, 0.22, 0.15, 0.1, 0.07, 0.05]
        a5, _ = adam_lse_optimize(best_seq, t5, T_stages=T_final, base_lr=0.15)
        try_update(a5, "final_lse")
    
    # === Phase 6: Quick final coord descent ===
    print(f"\n=== Phase 6: Final coord (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.05, 50.0)
    if t6 > 5:
        a6, _ = coordinate_descent(best_seq, t6)
        try_update(a6, "final_coord")
    
    # Final verification using evaluate_sequence
    if best_seq is not None:
        # Verify with the actual evaluator
        test_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        actual = evaluate_sequence(test_list)
        if np.isfinite(actual) and actual < best_score:
            best_score = actual
            print(f"  evaluate_sequence confirmed: {best_score:.8f}")
        elif not np.isfinite(actual):
            print(f"  WARNING: evaluate_sequence returned {actual}")
        
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
