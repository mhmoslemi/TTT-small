"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Nesterov gradient with LogSumExp smoothing and FFT-based gradients
2. Lightweight LP best-response alternation for structural updates
3. Multi-temperature annealing
4. Coordinate descent polish on peak-contributing coordinates
5. Multi-n exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    L = 1 << (size-1).bit_length()
    A = rfft(a, L)
    return irfft(A*A, L)[:size]

def score(a):
    """Compute objective (scale-invariant)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_conv(a)
    return float(2*n*b.max())

def logsumexp_gradient(a, T):
    """Compute LSE objective and gradient using FFT."""
    n = len(a)
    b = fast_conv(a)
    # LSE
    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    lse = T * (b_max/T + np.log(exp_b.sum()))
    obj = 2*n*lse
    
    # Gradient: grad_a[j] = 2 * sum_k w[k] * a[k-j] * 1_{0<=k-j<n}
    # where w = softmax(b/T)
    w = exp_b / exp_b.sum()
    
    # FFT computation
    L = 1 << (3*n - 2).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    # Cross-correlation: c[m] = sum_k w[k]*a[k-m]
    c = irfft(W * np.conj(A), L)
    grad_a = 2.0 * c[:n]
    
    return obj, grad_a

def softmax_project(z):
    """Project z onto simplex via softmax (for unconstrained gradient)."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def nesterov_lse(a0, time_limit, T_schedule=None, max_iters=100000):
    """Nesterov-accelerated gradient descent on LSE-smoothed objective."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    y = a.copy()  # Nesterov momentum variable
    t_nesterov = 1.0
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    if T_schedule is None:
        T_schedule = [30, 15, 8, 4, 2, 1, 0.5, 0.25, 0.1]
    
    # Barzilai-Borwein state
    prev_x = None
    prev_g = None
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit - 5:
            break
        stage_time = time_limit / len(T_schedule)
        stage_end = t0 + (time.monotonic() - t0) + stage_time
        iters_this_stage = 0
        bb_alpha = 1.0 / (2.0 * n)  # initial step size
        
        for it in range(max_iters):
            if time.monotonic() > stage_end or time.monotonic() - t0 > time_limit - 2:
                break
            iters_this_stage += 1
            
            obj, g = logsumexp_gradient(y, T)
            
            # Project gradient onto simplex (remove mean component)
            g = g - g.mean()  # tangent to simplex
            # Also project: g should be in the tangent space
            
            # Barzilai-Borwein step size
            if prev_x is not None and prev_g is not None:
                s_vec = y - prev_x
                y_vec = g - prev_g
                ss = np.dot(s_vec, s_vec)
                sy = np.dot(s_vec, y_vec)
                yy = np.dot(y_vec, y_vec)
                if ss > 1e-15 and yy > 1e-15:
                    if it % 2 == 0:
                        bb_alpha = ss / sy if abs(sy) > 1e-15 else bb_alpha
                    else:
                        bb_alpha = sy / yy
                    bb_alpha = max(1e-10, min(bb_alpha, 10.0 / n))
            
            # Nesterov step
            alpha = bb_alpha
            x_new = y - alpha * g
            x_new = project_simplex(x_new)
            
            # Update Nesterov parameter
            t_new = (1 + np.sqrt(1 + 4*t_nesterov**2)) / 2
            beta = (t_new - 1) / t_nesterov
            y_new = x_new + beta * (x_new - y)
            y_new = project_simplex(y_new)
            
            prev_x = y.copy()
            prev_g = g.copy()
            y = y_new
            t_nesterov = t_new
            
            # Track best
            sc = score(y)
            if sc < best_score:
                best_score = sc
                best_a = y.copy()
            
            if it % 500 == 0:
                print(f"      LSE T={T}, iter={it}, score={best_score:.8f}, obj={obj:.8f}")
        
        # Reset BB state between stages
        prev_x = None
        prev_g = None
    
    return best_a, best_score

def lp_best_response_fast(a_fixed, time_per_trial=0.5):
    """
    Fast LP best response: given fixed a, find b >= 0, sum(b)=1 minimizing max(conv(a,b)).
    Uses a simple subgradient method instead of full LP for speed.
    """
    n = len(a_fixed)
    b = np.ones(n) / n
    t0 = time.monotonic()
    alpha = 1.0 / n
    
    for it in range(5000):
        if time.monotonic() - t0 > time_per_trial:
            break
        # conv(a, b)
        c = np.convolve(a_fixed, b)  # length 2n-1
        max_c = c.max()
        peak_idx = np.argmax(c)
        
        # Subgradient of max_c w.r.t. b
        # dconv[k]/db[j] = a[k-j] if valid
        g = np.zeros(n)
        js = max(0, peak_idx - n + 1)
        je = min(n-1, peak_idx)
        g[js:je+1] = a_fixed[peak_idx - np.arange(js, je+1)]
        
        # Subgradient step with projection
        b = b - alpha * (g - g.mean())
        b = project_simplex(b)
        alpha *= 0.999  # decay
        
    return b

def coordinate_descent_polish(a0, time_limit, n_iter=500000):
    """Coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, score(a)
    a = a / s
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        peak_k = int(np.argmax(b))
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n-1, peak_k)
        idx = np.arange(js, je+1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing top contributors
        top_positions = idx[np.argsort(contribs)[::-1][:5]]
        
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-10:
                continue
            for factor in [0.999, 0.995, 0.99, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] * (1 - factor)
                # Redistribute to others
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    break
                a_trial[mask] = others + diff * others / os
                a_trial = a_trial / a_trial.sum()
                sc = score(a_trial)
                if sc < best_score - 1e-14:

def interpolate_to_n(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Nesterov-accelerated gradient descent on LogSumExp-smoothed objective with FFT-based gradients
2. Lightweight LP best-response alternation for structural updates
3. L-BFGS refinement with LogSumExp smoothing
4. Coordinate descent polish on peak-contributing coordinates
5. Multi-n exploration to find optimal dimension
6. Final Nesterov polish with very low temperature

Key innovations:
- FFT-based gradient computation via convolution correlation trick
- Barzilai-Borwein adaptive step sizes
- Multiple restart mechanisms to escape local optima
- Efficient subgradient handling for the max (using top-K peaks)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = rfft(a, L)
    return irfft(A * A, L)[:size]


def score(a):
    """Compute objective 2*n*max(conv(a,a))/(sum(a))^2 (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max())


def project_simplex(v):
    """Project v onto probability simplex {x: x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_obj_grad(a, T):
    """
    Compute LogSumExp smoothed objective and gradient using FFT.
    Objective: 2*n*LSE(conv(a,a), T)
    Gradient computed via convolution correlation trick.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    lse = b_max + T * np.log(exp_b.sum())
    obj = 2.0 * n * lse

    # Softmax weights
    w = exp_b / exp_b.sum()

    # Gradient: grad_a[j] = 2 * sum_k w[k]*a[k-j] * 1_{0<=k-j<n}
    # = 2 * (w * a_flipped)[j + n - 1] where * is linear convolution
    a_flipped = a[::-1].copy()
    conv_len = (2 * n - 1) + n - 1  # = 3n - 2
    L = 1 << (conv_len - 1).bit_length()
    W = rfft(w, L)
    Af = rfft(a_flipped, L)
    c = irfft(W * Af, L)[:conv_len]
    grad_a = 2.0 * c[n - 1:n - 1 + n]

    return obj, grad_a


def subgradient_step(a, n_peaks=1):
    """Compute subgradient using top-K peaks (for very low T or exact max)."""
    n = len(a)
    b = fast_conv(a)
    # Get top peaks
    if n_peaks >= len(b):
        peaks = np.arange(len(b))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peak_indices = np.argsort(b)[::-1][:n_peaks]
        peak_vals = b[peak_indices]
        # Weight by proximity to max
        w_raw = np.exp((peak_vals - peak_vals[0]) / (0.01 * peak_vals[0] + 1e-15))
        weights = w_raw / w_raw.sum()
        peaks = peak_indices

    g = np.zeros(n)
    for k, w_k in zip(peaks, weights):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += w_k * a[k - idx]
    return 2.0 * g, b.max()


def nesterov_gradient(a0, time_limit, T_schedule=None, max_iters=200000, use_lse=True):
    """
    Nesterov-accelerated gradient descent with adaptive step size.
    Uses LogSumExp smoothing or subgradient depending on T.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    t_nest = 1.0
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    if T_schedule is None:
        T_schedule = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1]

    prev_x = None
    prev_g = None
    bb_alpha = 1.0 / n

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit - 3:
            break

        # Reset BB state at each temperature
        prev_x = None
        prev_g = None
        bb_alpha = 1.0 / n

        stage_budget = time_limit / len(T_schedule)
        stage_end = time.monotonic() + stage_budget

        no_improve = 0
        stage_best = best_score

        for it in range(max_iters):
            if time.monotonic() > stage_end or time.monotonic() - t0 > time_limit - 2:
                break

            # Compute gradient
            if T > 0.01 and use_lse:
                obj, g = lse_obj_grad(y, T)
            else:
                g, max_b = subgradient_step(y, n_peaks=max(1, min(5, n // 100)))
                obj = 2.0 * n * max_b

            # Project gradient to tangent space of simplex
            g = g - g.mean()

            # Barzilai-Borwein step size
            if prev_x is not None and prev_g is not None:
                s_vec = y - prev_x
                y_vec = g - prev_g
                ss = np.dot(s_vec, s_vec)
                sy = np.dot(s_vec, y_vec)
                yy = np.dot(y_vec, y_vec)
                if ss > 1e-20:
                    if yy > 1e-20 and abs(sy) > 1e-20:
                        if it % 2 == 0:
                            bb_alpha = ss / sy
                        else:
                            bb_alpha = sy / yy
                    bb_alpha = max(1e-12, min(bb_alpha, 5.0 / n))

            # Gradient step with projection
            x_new = y - bb_alpha * g
            x_new = project_simplex(x_new)

            # Nesterov update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            beta = (t_new - 1.0) / t_nest
            y_new = x_new + beta * (x_new - y)
            y_new = project_simplex(y_new)

            prev_x = y.copy()
            prev_g = g.copy()
            y = y_new
            t_nest = t_new

            # Evaluate every few iterations
            if it % 20 == 0 or it < 5:
                sc = score(y)
                if sc < best_score:
                    best_score = sc
                    best_a = y.copy()
                    no_improve = 0
                else:
                    no_improve += 1

                if no_improve > 200:
                    # Restart with perturbation
                    y = best_a + rng.standard_normal(n) * (0.05 / np.sqrt(n))
                    y = project_simplex(y)
                    t_nest = 1.0
                    prev_x = None
                    prev_g = None
                    no_improve = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      T={T:.3f}, iter={it+1}, elapsed={elapsed:.0f}s, best={best_score:.8f}")

    return best_a, best_score


def lp_fast_subgradient(a_fixed, n_iters=2000):
    """
    Fast LP best-response via subgradient method.
    Given fixed a (sum=1), find b>=0, sum(b)=1 minimizing max(conv(a,b)).
    """
    n = len(a_fixed)
    b = np.ones(n) / n
    t = 1.0 / np.sqrt(n)

    for it in range(n_iters):
        c = np.convolve(a_fixed, b)
        peak_k = int(np.argmax(c))
        max_c = c[peak_k]

        # Subgradient of max(conv(a,b)) w.r.t. b
        g = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        g[js:je + 1] = a_fixed[peak_k - np.arange(js, je + 1)]

        # Projected subgradient step
        b = b - t * (g - g.mean())
        b = project_simplex(b)
        t *= 0.9995

    return b


def lp_alternation(a0, time_limit, n_rounds=50):
    """LP best-response alternation with Nesterov averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    t_nest = 1.0
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    per_round = time_limit / max(1, n_rounds)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit - 2:
            break
        round_start = time.monotonic()

        # Solve best response
        b_new = lp_fast_subgradient(y, n_iters=max(100, int(5000 * per_round / 5.0)))

        # Nesterov mix
        alpha = 2.0 / (np.sqrt(rnd + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = project_simplex(y)

        # Also try the weighted average
        a_weighted = ((rnd + 1) * a + (rnd + 2) * b_new) / (2 * rnd + 3)
        a_weighted = project_simplex(a_weighted)

        sc_y = score(y)
        sc_a = score(a_weighted)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a_weighted.copy()
        a = a_weighted

        elapsed = time.monotonic() - t0
        if (rnd + 1) % 10 == 0:
            print(f"      LP alt: round={rnd+1}, elapsed={elapsed:.0f}s, best={best_score:.8f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=500):
    """L-BFGS-B on LogSumExp smoothed objective."""
    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 1.5, 0.8, 0.4, 0.2, 0.1]

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 3:
            break

        def fun(z):
            a_t = np.maximum(z, 0)
            st = a_t.sum()
            if st < 1e-15:
                return 1e10
            a_t = a_t / st
            o, _ = lse_obj_grad(a_t, T)
            return o

        def jac(z):
            a_t = np.maximum(z, 0)
            st = a_t.sum()
            if st < 1e-15:
                return np.zeros_like(z)
            a_t = a_t / st
            _, g = lse_obj_grad(a_t, T)
            return g

        try:
            res = minimize(fun, a, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': maxiter_per_stage * 50})
            a_res = np.maximum(res.x, 0)
            sr = score(a_res)
            if sr < best_score:
                best_score = sr
                best_a = a_res.copy()
            a = a_res
        except Exception as e:
            print(f"      L-BFGS error (T={T}): {e}")

        if (time.monotonic() - t0) % 10 < 1:
            print(f"      L-BFGS T={T:.2f}, elapsed={time.monotonic()-t0:.0f}s, best={best_score:.8f}")

    return best_a


def coordinate_descent(a0, time_limit, max_steps=2000000):
    """
    Coordinate descent targeting peak-contributing elements.
    Tries small multiplicative updates on the most influential coordinates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, score(a)
    a = a / s
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Try reductions on top contributors
        top_m = min(8, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.999, 0.995, 0.99, 0.98, 0.97]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    break
                a_trial[mask] = others + diff * others / os
                a_trial = a_trial / a_trial.sum()
                sc = score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 50 == 0 and stall > 100:
                # Try increasing small elements (diversification)
                small_pos = np.argsort(a)[:3]
                for pos in small_pos:
                    a_trial = a.copy()
                    a_trial[pos] += 1e-4 / n
                    # Reduce largest element proportionally
                    a_trial = a_trial / a_trial.sum()
                    sc = score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
                        break
            if stall > 5000:
                break
        else:
            stall = 0

        if (step + 1) % 50000 == 0:
            print(f"      CD: step={step+1}, elapsed={time.monotonic()-t0:.0f}s, best={best_score:.8f}")

    return best_a, best_score


def interpolate_to_n(f, new_n):
    """Interpolate sequence to new length preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initial sequences."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 10)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-5 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.15)
    elif choice == 5:
        a = np.exp(-8 * (x - 0.35) ** 2) + 0.8 * np.exp(-8 * (x - 0.65) ** 2)
    elif choice == 6:
        a = (1 - x**2) * np.cos(np.pi * x / 2) ** 2
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 10.0 * x * (1 - x))
    elif choice == 8:
        a = np.exp(-2.0 * x) * np.cos(np.pi * x / 2) ** 2
        a = np.abs(a)
    else:
        a = np.ones(n) * (1.0 - 0.3 * np.abs(2 * x - 1))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    1. Nesterov LSE gradient descent (main workhorse)
    2. LP best-response alternation (structural)
    3. L-BFGS with LogSumExp smoothing
    4. Coordinate descent polish
    5. Multi-n exploration
    6. Final Nesterov polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = score(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Nesterov LSE (main workhorse, ~45% of time) ===
    print(f"\n=== Phase 1: Nesterov LSE (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_new, s_new = nesterov_gradient(
            best_seq, t1,
            T_schedule=[25.0, 12.0, 6.0, 3.0, 1.5, 0.8, 0.4, 0.2, 0.1, 0.05],
            max_iters=500000
        )
        try_update(a_new, "P1_nesterov")

    # === Phase 2: LP Alternation (structural updates, ~20%) ===
    print(f"\n=== Phase 2: LP Alternation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = lp_alternation(best_seq, t2, n_rounds=40)
        try_update(a_lp, "P2_lp")

    # === Phase 3: Nesterov again (refine after LP, ~15%) ===
    print(f"\n=== Phase 3: Nesterov Refine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 250.0)
    if t3 > 20:
        print(f"  Budget: {t3:.0f}s")
        a_ref, _ = nesterov_gradient(
            best_seq, t3,
            T_schedule=[5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05],
            max_iters=300000
        )
        try_update(a_ref, "P3_refine")

    # === Phase 4: L-BFGS (precision, ~12%) ===
    print(f"\n=== Phase 4: L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 180.0)
    if t4 > 15:
        print(f"  Budget: {t4:.0f}s")
        a_lb = lbgfs_refine(best_seq, t4,
                            T_stages=[8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1],
                            maxiter_per_stage=300)
        try_update(a_lb, "P4_lbgfs")

    # === Phase 5: Coordinate Descent (micro-optimization, ~8%) ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 120.0)
    if t5 > 10:
        a_cd, _ = coordinate_descent(best_seq, t5)
        try_update(a_cd, "P5_cd")

    # === Phase 6: Multi-n exploration (~5%) ===
    print(f"\n=== Phase 6: Multi-n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.05, 80.0)
    if t6 > 20:
        n_candidates = sorted(set([
            max(400, n_best - 100),
            max(400, n_best + 100),
            max(400, n_best // 2 + 100),
            n_best * 2
        ]) - {n_best})
        per_n = t6 / max(1, len(n_candidates))
        print(f"  Trying n values: {n_candidates}, {per_n:.0f}s each")
        for n_try in n_candidates:
            if time.monotonic() > deadline - 5:
                break
            a_interp = interpolate_to_n(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 3)
            if t_sub < 10:
                continue
            a_sub, _ = nesterov_gradient(a_interp, t_sub,
                                         T_schedule=[10.0, 5.0, 2.0, 1.0, 0.5],
                                         max_iters=100000)
            try_update(a_sub, f"P6_n{n_try}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = max(5.0, rem - 3.0)
    if t7 > 5:
        a_fin, _ = nesterov_gradient(best_seq, t7,
                                     T_schedule=[1.0, 0.5, 0.2, 0.1, 0.05],
                                     max_iters=200000)
        try_update(a_fin, "P7_final")

    # Final verification
    if best_seq is not None:
        fs = score(best_seq)
        best_score = fs
        print(f"\n{'='*50}")
        print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        print(f"{'='*50}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
