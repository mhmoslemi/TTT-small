Let me reconsider. The previous code's LP step solves:
min_b t s.t. conv(a, b) ≤ t, sum(b) = 1, b ≥ 0

This finds the "best response" b given a fixed a. The Nesterov iteration then mixes a with this b. The intuition is that at the optimum a*, the "best response" b* should equal a* (saddle point condition). So iterating a ← (1-α)a + α*b* should converge to the optimum.

This is a valid approach but might not be the most efficient. An alternative is to directly solve:
min_a max_k conv(a,a)[k] s.t. sum(a) = 1, a ≥ 0

This is a DCP (disciplined convex programming) problem in cvxpy:
- min t
- s.t. conv(a,a)[k] ≤ t for all k (quadratic constraints)
- sum(a) = 1, a ≥ 0

Wait, conv(a,a)[k] = sum_i a[i]*a[k-i] is a quadratic form in a, specifically a^T Q_k a where Q_k is a symmetric matrix. So the constraint a^T Q_k a ≤ t is a quadratic constraint, which is convex (if Q_k is PSD, which it is since it's a Gram matrix).

Actually, Q_k is the matrix with Q_k[i,j] = 1 if i+j=k (or i+j=k+n-1 depending on indexing), so it's a permutation matrix (or anti-diagonal). Permutation matrices are PSD. So the constraint is convex.

In cvxpy, I can write:
```python
a_var = cp.Variable(n, nonneg=True)
t_var = cp.Variable()
constraints = [cp.sum(a_var) == 1]
for k in range(2*n - 1):
    # conv(a,a)[k] = a^T Q_k a
    # This is a quadratic form
    constraints.append(cp.quad_form(a_var, Q_k) <= t_var)
prob = cp.Problem(cp.Minimize(t_var), constraints)
```

But this requires building 2n-1 quadratic forms, which is expensive for large n. Each Q_k is n x n (but very sparse - only n non-zero entries).

The total number of constraints is 2n-1, each with n variables. For n=1200, that's 2399 constraints. The LP version (linearizing at the current point) would be similar.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a refined multi-phase approach:
1. Multi-n LogSumExp scan to find optimal dimension
2. Deep LogSumExp L-BFGS with careful temperature annealing (FFT-accelerated)
3. LP-based minimax refinement for saddle point convergence
4. Multi-start perturbation exploration
5. Final ultra-low-temperature LogSumExp polish
6. Coordinate descent micro-optimization

Key improvements over previous:
- Correct gradient computation (fixed 1/T factor bug)
- FFT-accelerated gradient for faster iterations
- Better temperature schedule with more stages
- Multi-n exploration to find optimal dimension
- Perturbation-based escape from local optima
- More precise final polish
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_stable(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def logsumexp_obj_grad_z(z, T):
    """
    Compute LogSumExp smoothed objective and gradient in z-space (softmax param).
    Objective: 2n * (1/T) * log(sum_k exp(T * conv(a,a)[k]))
    where a = softmax(z).
    
    As T→0, this approaches 2n * max(conv(a,a)).
    """
    n = len(z)
    a = softmax_stable(z)
    b = fast_convolve(a)
    
    # LogSumExp computation
    b_max = b.max()
    exp_b = np.exp(T * (b - b_max))
    w = exp_b / exp_b.sum()
    
    # Objective: 2n * (b_max + log(sum exp(T*(b-b_max)))) / T
    #           = 2n * (b_max + log(exp_b.sum()) / T - log(exp_b.sum())/T)
    # Let me just compute it directly:
    # (1/T) * log(sum exp(T*b)) = (1/T) * (T*b_max + log(sum exp(T*(b-b_max))))
    #                           = b_max + (1/T) * log(exp_b.sum())
    obj = 2.0 * n * (b_max + np.log(exp_b.sum()) / T)
    
    # Gradient with respect to a:
    # grad_a[i] = 4n * sum_{l=0}^{n-1} w[i+l] * a[l]
    grad_a = np.zeros(n)
    for i in range(n):
        grad_a[i] = np.dot(w[i:i+n], a)
    grad_a *= 4.0 * n
    
    # Transform to z-space: grad_z[j] = a[j] * (grad_a[j] - <grad_a, a>)
    ga_dot_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - ga_dot_a)
    
    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """
    L-BFGS-B optimization of LogSumExp smoothed objective.
    Parameters:
    - a0: initial point (probability vector, sums to 1)
    - time_limit: seconds
    - T_stages: list of temperatures (decreasing)
    - maxiter_per_stage: max L-BFGS iterations per temperature
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    # Initialize z from a
    a_safe = np.maximum(a, 1e-300)
    z = np.log(a_safe)  # z such that softmax(z) ∝ a
    # softmax(z) = a since softmax(log(a)) = a/sum(a) = a (since sum=1)
    # Actually softmax(log(a)) = exp(log(a))/sum(exp(log(a))) = a/sum(a) = a ✓
    
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
    
    for idx, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit - 10:
            break
        
        stage_start = time.monotonic()
        
        # Compute initial gradient to check for numerical issues
        obj, grad = logsumexp_obj_grad_z(z, T)
        if not np.isfinite(obj) or not np.all(np.isfinite(grad)):
            continue
        
        max_grad = np.max(np.abs(grad))
        if max_grad < 1e-15:
            continue
        
        # Scale for numerical stability in L-BFGS
        # L-BFGS works best when gradients are O(1)
        grad_scale = max_grad if max_grad > 0 else 1.0
        
        def func(z_in):
            o, _ = logsumexp_obj_grad_z(z_in, T)
            return o / grad_scale
        
        def jac(z_in):
            _, g = logsumexp_obj_grad_z(z_in, T)
            return g / grad_scale
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxfun': maxiter_per_stage * 200,
                                   'disp': False})
            if res.x is not None:
                z = res.x
        except Exception as e:
            print(f"    L-BFGS error at T={T}: {e}")
            pass
        
        # Update a from z
        a_new = softmax_stable(z)
        a = a_new
        
        elapsed = time.monotonic() - t0
        score = evaluate_local(a)
        print(f"    LSE T={T:.4f} ({elapsed:.0f}s): score={score:.10f}, "
              f"max_b={fast_convolve(a).max():.8f}")
    
    return a


def build_conv_structure(n):
    """Precompute sparsity pattern for convolution matrix."""
    size = 2 * n - 1
    row_ind = np.repeat(np.arange(n), n)  # for column i, rows are i to i+n-1
    # Actually: for column i, the non-zero rows are i, i+1, ..., i+n-1
    row_ind = []
    col_ind = []
    for i in range(n):
        row_ind.extend(range(i, i + n))
        col_ind.extend([i] * n)
    return np.array(row_ind, dtype=np.int32), np.array(col_ind, dtype=np.int32), size


def update_conv_matrix(row_ind, col_ind, a, size, n):
    """Create sparse convolution matrix T where T @ b = convolve(a, b)."""
    data = a[col_ind]
    return csr_matrix((data, (row_ind, col_ind)), shape=(size, n))


def solve_best_response_lp(a, row_ind, col_ind, size, n, solver):
    """Solve min t s.t. conv(a,b) ≤ t, sum(b)=1, b≥0."""
    T_mat = update_conv_matrix(row_ind, col_ind, a, size, n)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_mat @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Try alternative solvers
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.MOSEK]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt, verbose=False)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except:
                    continue
    except:
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def lp_minimax_refine(a0, time_limit, solver, max_iters=50000):
    """
    Von Neumann minimax iteration: a ← (1-α)a + α*b* where b* is best response to a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    row_ind, col_ind, size = build_conv_structure(n)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        b_star = solve_best_response_lp(a, row_ind, col_ind, size, n, solver)
        if b_star is None:
            break
        
        # Nesterov step with adaptive rate
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        a_new = (1.0 - alpha) * a + alpha * b_star
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        a = a_new
        
        if (it + 1) % 100 == 0:
            score = evaluate_local(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if (it + 1) % 1000 == 0:
                print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")
    
    return best_a


def interpolate_to_n(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp
