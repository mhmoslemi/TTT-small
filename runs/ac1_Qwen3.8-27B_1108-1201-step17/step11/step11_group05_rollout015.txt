    Compute the subgradient of sum_k w[k] * (a*a)[k] w.r.t. a.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Nesterov subgradient with FFT-based gradient and simplex projection
2. LP alternating refinement (PDLP warm-started)
3. Multi-length exploration (n=1200, 1500, 1800)
4. Temperature-based active set focusing

The subgradient method is the main workhorse (O(n log n) per step, millions of steps).
LP refinement provides exact best-response for final polishing.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    else:
        result = np.ones(n) / n
    return result


def compute_subgradient(a, w):
    """
    Compute gradient of sum_k w[k] * (a*a)[k] w.r.t. a.
    
    a: array of length n
    w: weights of length 2n-1 (on the convolution output)
    Returns: gradient of length n
    """
    n = len(a)
    C = w + w[::-1]  # C[k] = w[k] + w[2n-2-k]
    C_rev = C[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    CR = np.zeros(fft_size, dtype=np.float64)
    CR[:len(C_rev)] = C_rev
    A = np.zeros(fft_size, dtype=np.float64)
    A[:n] = a
    conv = irfft(rfft(CR) * rfft(A), fft_size)
    return conv[n-1 : 2*n-1]


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k] subject to b>=0, sum(b)=1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for s in [solver, cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0.0, M_var.value))


def nesterov_subgradient(a0, time_limit, top_k_frac=0.1, momentum=0.9):
    """Nesterov accelerated subgradient method with adaptive step size."""
    n = len(a0)
    x = np.array(a0, dtype=np.float64)
    x = np.maximum(x, 0)
    s = x.sum()
    if s < 1e-15:
        x = np.ones(n) / n
    else:
        x = x / s
    
    y = x.copy()  # lookahead point
    x_prev = x.copy()
    
    best_x = x.copy()
    best_score = 2.0 * n * fast_convolve(x).max()
    
    t0 = time.monotonic()
    eta = 1.0 / (1.0 + 0.01 * np.sqrt(n))  # initial step size
    beta = momentum  # momentum parameter
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute convolution at lookahead point
        b = fast_convolve(y)
        max_b = b.max()
        score = 2.0 * n * max_b
        
        if score < best_score:
            best_score = score
            best_x = y.copy()
        
        # Compute active set weights (top-k)
        k = max(1, int(top_k_frac * (2 * n - 1)))
        idx = np.argpartition(b, -k)[-k:]
        b_top = b[idx]
        # Use softmax-like weights
        max_top = b_top.max()
        w_top = np.exp((b_top - max_top) / max(1e-6, max_top * 0.1))
        w_top /= w_top.sum()
        w = np.zeros(2 * n - 1)
        w[idx] = w_top
        
        # Compute subgradient
        g = compute_subgradient(y, w)
        
        # Nesterov step
        t_alpha = 2.0 / (t + 1 + np.sqrt(1 + 4 * t * t))  # or use Beta sequence
        # Standard FISTA: t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2
        # momentum = (t_k - 1) / (t_{k+1})
        
        # Simpler: use heavy ball
        y_new = y + beta * (y - x_prev) - eta * g
        x_new = simplex_project(y_new)
        
        # Adaptive step size (reduce if not improving)
        x_prev = x.copy()
        x = x_new
        
        # Decay step size
        if t % 10000 == 0:
            eta *= 0.8
            beta = min(beta * 1.001, 0.999)
        
        if t % 100000 == 0:
            elapsed = time.monotonic() - t0
            cur_score = 2.0 * n * fast_convolve(x).max()
            print(f"    NSubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f} eta={eta:.6f}")
    
    return best_x, best_score
