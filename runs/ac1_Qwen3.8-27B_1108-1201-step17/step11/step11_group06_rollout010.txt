**Alternative: Direct Nesterov on the max function with subgradients**

The function f(b) = max_k (Tb)_k is convex. Its subgradient at b is:
∂f(b) = convex hull of {T[k,:] : k in argmax(Tb)}

With Nesterov acceleration:
- y_t = (1 - alpha_t) b_t + alpha_t * b_{t-1} (or some variant)
- b_{t+1} = projection onto simplex of (y_t - eta_t * g_t) where g_t in ∂f(y_t)

This converges at O(1/t) rate. The advantage over LP is that we don't need an LP solver - we just need to compute the max and the corresponding gradient.

The subgradient computation:
1. Compute Tb (using FFT, O(n log n))
2. Find the argmax (or set of near-max values)
3. The gradient is the corresponding row(s) of T

This is much faster per iteration than solving an LP! The question is whether the convergence is fast enough.

With Nesterov acceleration and step size eta_t = c/sqrt(t) (for non-smooth) or eta_t = c/(L*t) (for smooth, but max is not smooth), we get:
- O(1/t) rate for Nesterov on convex functions (with non-smooth, it's O(1/sqrt(t)) for standard subgradient, O(1/t) for Nesterov dual averaging)

Actually, for the min-max game formulation:
- Player 1 chooses b in simplex, Player 2 chooses lambda in simplex
- Payoff: b^T T^T lambda = sum_k lambda_k (Tb)_k
- Value of game = min_b max_lambda b^T T^T lambda = min_b max_k (Tb)_k

The Nesterov method for zero-sum games (like those used in game theory) converges at O(1/t) rate. This is what the LP-based approach does, but with the LP solving the "best response" exactly.

Actually, one more important consideration: **the relationship between the LP solution and the dual**.

The LP: min M s.t. T@b <= M*1, 1^T b = 1, b >= 0

The dual: max sum_k lambda_k - mu s.t. T^T lambda <= mu * 1, lambda >= 0

At optimality, the complementary slackness conditions give us:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Vectorized sparse matrix construction for fast LP iterations
2. LP-Nesterov with cvxpy Parameter for zero-recompile re-solves
3. Hybrid LP/gradient: solve LP every K iterations, use dual-guided gradient between
4. Multi-stage LogSumExp L-BFGS with geometric temperature cooling
5. Multi-n exploration (n=1000, 1200, 1500, 1800)
6. Local coordinate search for final improvement

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_indices(n):
    """Build (rows, cols) for the convolution matrix T where T@b = conv(y, b)."""
    size = 2 * n - 1
    # For column j, non-zeros at rows j, j+1, ..., j+n-1 with values y[0], y[1], ..., y[n-1]
    idx_i = np.repeat(np.arange(n), n)  # i = 0,1,...,n-1 repeated n times
    idx_j = np.tile(np.arange(n), n)    # j = 0,0,...,0,1,1,...,1,...
    rows = idx_j + idx_i
    cols = idx_j
    return rows, cols


def make_conv_matrix(y, rows, cols, size, n):
    """Create CSR matrix from pre-built indices and current y values."""
    data = np.tile(y, n)
    return csr_matrix((data, (rows, cols)), shape=(size, n))


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def multiplicative_weights(a0, time_limit, n_steps=1000000, eta0=0.05):
    """Multiplicative weights with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.5)
        else:
            eta *= 0.9
            if eta < 1e-10:
                break
    return best_a, best_score

def lse_logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / (exp_terms.sum() + 1e-300)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


class LSEFunc:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = lse_logsumexp_obj_grad(z, self.n, self.T)
        return obj


class LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = lse_logsumexp_obj_grad(z, self.n, self.T)
        return g


def lse_lbfgs(a0, time_limit, T_stages=None, maxiter_per_stage=1000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    if T_stages is None:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    for stage_T in T_stages:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 3:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': maxiter_per_stage * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    else:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def local_coordinate_search(a, n_trials=10000, t0=None, time_limit=None):
    """Try random coordinate perturbations to improve the score."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    best_score = evaluate_local(a)
    if t0 is None:
        t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for trial in range(n_trials):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        # Pick 1-3 random coordinates to perturb
        n_pert = rng.integers(1, 4)
        indices = rng.integers(0, n, size=n_pert)
        a_trial = a.copy()
        for idx in indices:
            delta = rng.uniform(-0.05, 0.05) * a[idx]
            a_trial[idx] = max(0, a[idx] + delta)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        sc = evaluate_local(a_trial)
        if sc < best_score:
            best_score = sc
            a = a_trial.copy()
    
    return a, best_score
