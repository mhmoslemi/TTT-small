import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_structure(n):
    """Precompute row/col indices for the (2n-1) x n convolution matrix."""
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            count += 1
    return rows[:count], cols[:count]


def fill_conv_matrix_values(a, rows, cols):
    """Fill values for the convolution matrix given current a."""
    n = len(a)
    size = 2 * n - 1
    vals = np.empty(len(rows), dtype=np.float64)
    for i in range(len(rows)):
        k = rows[i]
        j = cols[i]
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
        else:
            vals[i] = 0.0
    return vals


def get_lp_solver():
    """Find a working LP solver, preferring fast interior-point methods."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / (exp_terms.sum() + 1e-300)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


class LSEFunc:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = lse_obj_grad(z, self.n, self.T)
        return obj


class LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = lse_obj_grad(z, self.n, self.T)
        return g


def multiplicative_weights(a0, time_limit, n_steps=500000, eta0=0.05):
    """Adaptive multiplicative weights with dynamic active set size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    active_frac = 0.05  # fraction of top peaks to use
    prev_best = best_score
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Dynamic active set: use top peaks within a relative threshold
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        # Also ensure we use at least a minimum number
        min_active = max(3, int(n * active_frac))
        if len(active) < min_active:
            top_indices = np.argsort(b)[::-1][:min_active]
            active = top_indices
        # Weight by proximity to max
        w_active = (b[active] - max_b + 1e-10) ** 6
        w_active = w_active / (w_active.sum() + 1e-30)
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        # Multiplicative update with adaptive step
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.25)
            stall = 0
        else:
            a = a_new  # still move, but don't update best
            eta *= 0.85
            stall += 1
            if stall > 200:
                # Perturb to escape
                rng = np.random.default_rng(step)
                noise = 0.01 * rng.standard_normal(n)
                a = np.maximum(a + noise, 0)
                a = a / a.sum()
                stall = 0
            if eta < 1e-9:
                break
        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")
    return best_a, best_score


def lp_alternate(a0, time_limit, solver, n_max_iters=1000000):
    """LP alternating with precomputed structure and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)

    # Precompute matrix structure
    rows, cols = build_conv_matrix_structure(n)
    size = 2 * n - 1

    # Initial T matrix
    vals = fill_conv_matrix_values(y, rows, cols)
    T_mat = csr_matrix((vals, (rows, cols)), shape=(size, n))

    lp_times = []
    step_size_base = 0.1  # base step size for y update

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_time = time.monotonic() - lp_t0
        lp_times.append(lp_time)
        if b_new is None:
            continue

        # Stop if LP is too slow
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 5.0:
                print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping at iter {it+1}")
                break

        # Adaptive step size: larger when improving, smaller when stalling
        if stall_count < 5:
            alpha = step_size_base * (1.0 + 0.1 * (it % 10))
        else:
            alpha = step_size_base * 0.5

        # Update y (dual variable / "mixing" sequence)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
        else:
            y = y / sy

        # Update a (primal) using exponential averaging
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Update T matrix values (structure unchanged)
        vals = fill_conv_matrix_values(y, rows, cols)
        T_mat = csr_matrix((vals, (rows, cols)), shape=(size, n))

        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        current_best = min(score_a, score_y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if best_score < prev_best - 1e-10:
            stall_count = 0
        else:
            stall_count += 1
        prev_best = best_score

        # Periodic perturbation to escape local optima
        if stall_count >= 8:
            freq = rng.integers(1, 15)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.linspace(0, 2 * np.pi, n)
            noise = 0.015 * np.cos(freq * x + phase)
            y = np.maximum(y + noise, 0)
            y = y / y.sum()
            vals = fill_conv_matrix_values(y, rows, cols)
            T_mat = csr_matrix((vals, (rows, cols)), shape=(size, n))
            stall_count = 0

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

    """L-BFGS with LogSumExp smoothing and adaptive temperature scheduling."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline_override is not None:
        deadline = deadline_override
    else:
        deadline = t0 + time_limit

    # Adaptive temperature: start high, decrease based on gradient norm
    T = 5.0
    min_T = 0.0005
    T_decay = 0.7

    for stage in range(40):
        if time.monotonic() > deadline - 2:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        func = LSEFunc(n, T, deadline)
        jac = LSEJac(n, T, deadline)
        try:

                fun=func, x0=z, jac=jac, method='L-BFGS-B',

                         'gtol': 1e-20, 'maxfun': 300000,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Adaptive temperature decrease
        if res is not None and hasattr(res, 'optimal') and res.optimal:
            T *= T_decay * 0.8  # faster decrease if converged
        else:
            T *= T_decay
        T = max(T, min_T)

        if (stage + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            a_tmp = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_tmp).max()
            print(f"    LSE stage={stage+1} T={T:.5f} ({elapsed:.0f}s): score={sc:.8f}")

    return softmax_from_z(z)


def greedy_peak_reduction(a0, time_limit, n_iters=2000):
    """Greedy reduction of autocorrelation peaks by mass redistribution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        k_star = np.argmax(b)

        # Find which pairs contribute to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        contributors = list(range(js, je + 1))
        contributions = np.array([a[j] * a[k_star - j] for j in contributors])
        total_contrib = contributions.sum()
        if total_contrib < 1e-30:
            break

        # Identify top contributors
        top_frac = 0.3
        n_top = max(2, int(len(contributors) * top_frac))
        top_local = np.argsort(contributions)[::-1][:n_top]
        top_j = [contributors[i] for i in top_local]

        # Reduce top contributors, add to "safe" positions
        reduction = 0.1 * max(contributions[top_local[0]], 1e-15)
        for j in top_j:
            a[j] = max(a[j] - reduction, 0)

        # Find safe positions: those with low autocorrelation contribution
        # Simply add to positions with smallest a values
        safe_indices = np.argsort(a)[:n_top]
        add_amount = reduction / n_top
        for j in safe_indices:
            a[j] += add_amount

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / s

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if it % 500 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    Greedy {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score

    elif choice == 8:
        # Flat with sinusoidal modulation
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    else:
        # Two-scale structure
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 11 * x + 1)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Fast MW ===
    print(f"\n=== Phase 1: MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights(best_seq, t1, n_steps=500000, eta0=0.04)
        try_update(a_mw, "mw_1")

    # === Phase 2: LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.38, 650.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = lp_alternate(best_seq, t2, solver, n_max_iters=1000000)
        try_update(a_lp, "lp_1")

    # === Phase 3: Greedy peak reduction ===
    print(f"\n=== Phase 3: Greedy (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t3 = min(rem * 0.05, 60.0)
        a_greedy, _ = greedy_peak_reduction(best_seq, t3, n_iters=5000)
        try_update(a_greedy, "greedy")
        # Follow up with LP if greedy improved
        rem = deadline - time.monotonic()
        if rem > 60:
            t3b = min(rem * 0.08, 100.0)
            a_greedy_lp, _ = lp_alternate(a_greedy, t3b, solver, n_max_iters=100000)
            try_update(a_greedy_lp, "greedy_lp")

    # === Phase 4: Palindromic refinement ===
    print(f"\n=== Phase 4: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t4 = min(rem * 0.1, 150.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = lp_alternate(a_pal, t4, solver, n_max_iters=200000)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [1000, 1400, 1600, 2000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if abs(len(best_seq) - ln) < 50:
            continue
        t_ln = min(rem * 0.1, 120.0)
        if t_ln < 30:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln_mw, _ = multiplicative_weights(init_ln, t_ln * 0.25, n_steps=100000, eta0=0.04)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.65)
        if t_lp > 30:
            a_ln_lp, _ = lp_alternate(a_ln_mw if a_ln_mw is not None else init_ln,
                                       t_lp, solver, n_max_iters=100000)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 6: LSE L-BFGS on best ===
    print(f"\n=== Phase 6: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t6 = min(rem * 0.18, 280.0)
        print(f"  Budget={t6:.0f}s")
        a_lse = lse_lbfgs_refine(best_seq, t6, deadline_override=deadline)
        try_update(a_lse, "lse")

    # === Phase 7: Perturbation + re-optimize ===
    print(f"\n=== Phase 7: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.008 * trial)
            t_sub = min(rem * 0.25, 50.0)
            a_opt, _ = lp_alternate(a_pert, t_sub, solver, n_max_iters=50000)
            try_update(a_opt, f"pert{trial}")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t8 = min(rem * 0.35, 100.0)
        a_fin, _ = lp_alternate(best_seq, t8, solver, n_max_iters=200000)
        try_update(a_fin, "final_lp")

    # === Phase 9: Final MW polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t9 = min(rem * 0.5, 30.0)
        a_fin_mw, _ = multiplicative_weights(best_seq, t9, n_steps=100000, eta0=0.015)
        try_update(a_fin_mw, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
