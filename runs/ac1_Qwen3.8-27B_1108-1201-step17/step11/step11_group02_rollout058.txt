def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_sparse_conv_matrix(a_fixed):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    # Each column j has a_fixed[j] in rows j, j+1, ..., j+n-1
    # So row k, col j = a_fixed[j] if j <= k <= j+n-1, i.e., max(0,k-n+1) <= j <= min(n-1,k)
    row_idx = []
    col_idx = []
    data = []

        row_idx.append(rows)
        col_idx.append(np.full(n, j, dtype=np.int32))

    row_idx = np.concatenate(row_idx)
    col_idx = np.concatenate(col_idx)
    data = np.concatenate(data)
    T = csr_matrix((data, (row_idx, col_idx)), shape=(size, n))
    return T

def solve_lp_best_response(a_fixed, solver):

    b_val = np.asarray(b_var.value, dtype=np.float64)

    b_val = np.maximum(b_val, 0)
    s = b_val.sum()

def lse_objective_and_grad(a, tau):
    """
    Compute LogSumExp smoothed objective and gradient.
    Objective: (2n/tau) * LogSumExp(conv(a,a)/tau)
    """
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    exp_shifted = np.exp((conv - max_conv) / tau)
    w = exp_shifted / exp_shifted.sum()
    lse_val = max_conv + tau * np.log(exp_shifted.sum())
    obj = (2.0 * n / tau) * lse_val
    
    # Gradient via cross-correlation
    # grad[i] = (2n/tau) * sum_j w[i+j] * a[j]
    # = (2n/tau) * (w cross-correlated with a)[i]
    L = 1 << (3 * n - 2).bit_length()
    W_pad = rfft(w, L)
    A_pad = rfft(a, L)
    crosscorr = irfft(W_pad * np.conj(A_pad), L)
    grad = (2.0 * n / tau) * crosscorr[:n]
    
    return obj, grad


def lbgfs_refine(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [20.0, 12.0, 7.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.35, 0.2, 0.1]
    
    best_a = a.copy()
    best_true_score = evaluate_local(a)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Compute a good initial tau based on the spread of the convolution
        conv = fast_convolve(a)
        sorted_conv = np.sort(conv)[::-1]
        # Use a fraction of the spread
        T_eff = max(T, (sorted_conv[0] - sorted_conv[min(n, len(sorted_conv)-1)]) * 0.5)
        
        def func(z):
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            obj, _ = lse_objective_and_grad(a_trial, T_eff)
            return obj
        
        def jac(z):
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z)
            a_trial = a_trial / st
            _, grad = lse_objective_and_grad(a_trial, T_eff)
            return grad
        
        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': 500, 'ftol': 1e-15,
                                    'gtol': 1e-10, 'maxfun': 5000,
                                    'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception:
            pass
        
        sc = evaluate_local(a)
        if sc < best_true_score:
            best_true_score = sc
            best_a = a.copy()
    
    return best_a


def mirror_descent_subgrad(a0, time_limit, n_peaks=5, eta0=0.1):
    """
    Mirror descent with subgradient from top peaks.
    Uses the top n_peaks positions of the convolution to compute an averaged subgradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    no_improve = 0
    eta = eta0
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Select top peaks
        sorted_idx = np.argsort(b)[::-1]
        top_peaks = sorted_idx[:n_peaks]
        # Weight by significance
        peak_weights = np.exp((b[top_peaks] - max_b) / (max_b * 0.01 + 1e-15))
        peak_weights /= peak_weights.sum()
        
        # Compute weighted subgradient
        g = np.zeros(n)
        for w_k, k in zip(peak_weights, top_peaks):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w_k * a[k - idx]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Mirror descent step (exponential weights / multiplicative weights)
        log_a = np.log(np.maximum(a, 1e-300))
        # Subtract mean of g for stability
        g_centered = g - np.mean(g)
        log_a_new = log_a - eta * g_centered
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        
        a = a_new
        score = evaluate_local(a)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive step size
        if no_improve > 100:
            eta *= 0.95
            no_improve = 0
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.6f}")
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=100000):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] * (1 - factor)
                if diff < 1e-16:

                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:

                    best_score = score

                    mask[pos] = False

                    others_sum = others.sum()

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.2)
    elif choice == 5:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        # Power law
        a = x ** 0.5 + (1-x) ** 0.5
    else:
        # Flat with bump
        a = np.ones(n)
        center = n // 2
        width = n // 6
        a[center-width:center+width] += np.exp(-((np.arange(2*width+1) - width) ** 2) / (width * width))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


_conv_struct_cache = {}

def get_conv_structure(n):
    """Get or build the sparsity structure for convolution matrix."""
    if n in _conv_struct_cache:
        return _conv_struct_cache[n]

    row_parts = []
    col_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + n))
        col_parts.append(np.full(n, j, dtype=np.int32))
    row_idx = np.concatenate(row_parts)
    col_idx = np.concatenate(col_parts)
    _conv_struct_cache[n] = (row_idx, col_idx, (size, n))
    return _conv_struct_cache[n]

def solve_lp_fast(a_fixed, solver, eps=1e-8):
    """Solve LP: min t s.t. conv(a_fixed, b)_k <= t, sum(b)=1, b>=0."""
    n = len(a_fixed)
    T_sparse = make_conv_sparse(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_sparse @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, eps=eps)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.ECOS, cp.CLARABEL, cp.SCS, cp.PDLP]:

                    prob.solve(solver=alt, eps=1e-7)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

    return b_val / s

    stall = 0

        if time.monotonic() - t0 > time_limit:

        b_new = solve_lp_fast(y, solver)
        if b_new is None:

        # Nesterov momentum update
        it_local = it - restart_iter
        alpha = 2.0 / (np.sqrt(it_local + 1) + 1.0)

        if sy < 1e-15:
            y_new = a.copy()
            sy = 1.0

        weight_old = it_local + 1
        a_new = (weight_old * a + weight_new * b_new) / (weight_old + weight_new)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / sa

        a = a_new

        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            consecutive_no_lp_improve = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            consecutive_no_lp_improve = 0
        else:
            consecutive_no_lp_improve += 1

def power_sum_obj_grad(a, p):
    """
    Power-sum smoothed objective: (2n) * (sum_k conv_k^p)^(1/p)
    This approximates 2n * max(conv) as p -> inf.
    """
    n = len(a)
    conv = fast_convolve(a)
    conv_pos = np.maximum(conv, 1e-30)
    conv_p = conv_pos ** p
    sum_power = conv_p.sum()
    obj = 2.0 * n * sum_power ** (1.0 / p)
    
    # Gradient: d/da_i [ (sum conv^p)^(1/p) ]
    # = (1/p) * (sum conv^p)^(1/p - 1) * p * sum_k conv_k^(p-1) * d(conv_k)/da_i
    # = (sum conv^p)^(1/p - 1) * sum_k conv_k^(p-1) * a_{k-i}
    # For large p, conv_k^(p-1) / conv_k^p = 1/conv_k, so weights are ~ conv_k^p / sum(conv^p)
    
    weights = conv_p / sum_power  # normalized
    # grad_i = obj_factor * sum_k weights[k] * a_{k-i}
    # where obj_factor = 2n * (sum_power)^(1/p - 1)
    obj_factor = 2.0 * n * sum_power ** (1.0 / p - 1.0)
    
    # Cross-correlation: sum_k weights[k] * a[k-i]
    L = 1 << (3 * n - 2).bit_length()
    W = rfft(weights, L)
    A = rfft(a, L)
    crosscorr = irfft(W * np.conj(A), L)
    grad = obj_factor * crosscorr[:n]
    
    return obj, grad

def dual_subgradient_refine(a0, time_limit, n_iters=100000):
    """

    This gives a lower bound on the optimal max convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    size = 2 * n - 1
    # Initialize lambda uniformly
    lam = np.ones(size) / size
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    eta = 0.01
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute A^T lambda: for each j, (A^T lam)_j = sum_k lam[k] * A[k,j]
        # A[k,j] = a[j] if max(0,k-n+1) <= j <= min(n-1, k)
        # So (A^T lam)_j = a[j] * sum_{k: j in [max(0,k-n+1), min(n-1,k)]} lam[k]
        # = a[j] * sum_{k=max(j,0)}^{min(j+n-1, size-1)} lam[k]
        
        # For j in 0..n-1: (A^T lam)_j = a[j] * sum_{k=j}^{j+n-1} lam[k] (clipped to [0,size-1])
        At_lam = np.zeros(n)
        for j in range(n):
            k_start = j
            k_end = min(j + n, size)
            At_lam[j] = a[j] * lam[k_start:k_end].sum()
        
        # Subgradient of min_j (A^T lam)_j: pick the argmin
        min_idx = np.argmin(At_lam)
        # The "gradient" of the dual objective w.r.t. lambda
        # d/d(lam[k]) of (A^T lam)_[min_idx] = A[min_idx, j_min_related]
        # A[min_idx, j] = a[j] if max(0, min_idx-n+1) <= j <= min(n-1, min_idx)
        grad_lam = np.zeros(size)
        j_start = max(0, min_idx - n + 1)
        j_end = min(n - 1, min_idx)
        grad_lam[j_start:j_end+1] = a[j_start:j_end+1]
        
        # Mirror descent on lambda (simplex)
        log_lam = np.log(np.maximum(lam, 1e-300))
        g_centered = grad_lam - grad_lam.sum() / size
        log_lam_new = log_lam - eta * g_centered
        log_lam_new -= log_lam_new.max()
        lam_new = np.exp(log_lam_new)
        lam_new = lam_new / lam_new.sum()
        lam = lam_new
        
        # Extract the primal solution from the LP: b_j proportional to something
        # Actually, the primal solution at optimality is: b_j > 0 only where (A^T lam)_j = min
        # This is the complementary slackness condition
        # For our purposes, we use the "best response" of the primal:
        # b is supported where (A^T lam)_j is at its minimum
        
        min_val = At_lam.min()
        threshold = min_val + 1e-10
        active = np.where(At_lam <= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmin(At_lam)])
        
        # The optimal b at this lambda: b_j = 1/|active| for j in active (if the constraint is tight)
        b_candidate = np.zeros(n)
        b_candidate[active] = 1.0 / len(active)
        
        # Actually this isn't quite right for the primal. Let's just use the Nesterov approach.
        # Instead, let's evaluate the current a and try to improve it.
        
        if (it + 1) % 1000 == 0:
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score


def mirror_descent_multi_peak(a0, time_limit, n_peaks=8, eta0=0.08):
    """Mirror descent using subgradients from top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    for it in range(300000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Select top peaks with exponential weighting
        sorted_idx = np.argsort(b)[::-1]
        top = sorted_idx[:n_peaks]
        pvals = b[top]
        pweights = np.exp(-(max_b - pvals) / (max_b * 0.05 + 1e-20))
        pweights /= pweights.sum()
        
        # Compute weighted subgradient
        g = np.zeros(n)
        for w_k, k in zip(pweights, top):
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            idx = np.arange(js, je + 1)
            g[idx] += w_k * a[k - idx]
        
        # Normalize
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Mirror descent (multiplicative weights)
        log_a = np.log(np.maximum(a, 1e-300))
        g_c = g - g.mean()
        log_a -= eta * g_c
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = a / a.sum()
        
        sc = evaluate_local(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            eta *= 0.9
            no_improve = 0
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
