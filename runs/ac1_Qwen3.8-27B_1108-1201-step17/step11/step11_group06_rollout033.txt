**Approach: Coordinate descent**

**Approach: Frank-Wolfe (Conditional Gradient)**

The constraint set is the simplex {a >= 0, sum(a) = 1}. Frank-Wolfe is:
1. Compute gradient g of the (smoothed) objective
2. Find s = argmin_{s in simplex} g^T s (this is just s = e_j where j = argmin_i g_i)
3. a_new = (1-t) a + t * s

This is O(n) per iteration (plus the cost of computing the gradient, which is O(n^2) naively or O(n log n) with FFT).

Frank-Wolfe converges at rate O(1/t), same as vanilla subgradient. But it's very simple and can be combined with acceleration.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))

                    continue
                try:
                    prob2.solve(solver=s)
                    if prob2.status in ('optimal', 'optimal_inaccurate'):
                        b_val = np.asarray(b_var.value, dtype=np.float64)
                        M_val = float(max(0, M_var.value))
                        break
                except Exception:
                    continue
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
    
    return b_val, float(M_val) if M_val is not None else None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp, deadline):
    """Compute LogSumExp objective and gradient w.r.t. z for softmax parameterization."""
    if time.monotonic() > deadline:
        return 1e15, np.zeros(len(z))
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lse_func_factory(n, T, deadline):
    """Create a callable for LSE objective."""
    class LSEFunc:
        def __init__(self):
            pass
        def __call__(self, z):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z, n, T, deadline)
            return obj
    return LSEFunc()


def lse_jac_factory(n, T, deadline):
    """Create a callable for LSE gradient."""
    class LSEJac:
        def __init__(self):
            pass
        def __call__(self, z):
            if time.monotonic() > deadline:
                return np.zeros(len(z))
            _, g = logsumexp_obj_grad(z, n, T, deadline)
            return g
    return LSEJac()


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=300):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        func = lse_func_factory(n, stage_T, deadline)()
        jac = lse_jac_factory(n, stage_T, deadline)()
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-16, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def multiplicative_weights_adaptive(a0, time_limit, n_steps, eta0=0.05):
    """Fast multiplicative weights with adaptive backtracking step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.75
            if eta < 1e-9:
                break
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, T_mat, full_T, n_max_iters=500000):
    """
    Nesterov dual averaging LP alternating with active-set LP solving.
    Uses warm-starting and adaptive active set size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    lp_times = []
    prev_b = None
    prev_M = None
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat, full_T, prev_b, prev_M)
        lp_times.append(time.monotonic() - lp_t0)
        if b_new is None:
            continue
        prev_b = b_new
        prev_M = M_b
        
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 10.0:
                print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping at iter {it+1}")
                break
        
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        T_mat = build_conv_matrix_sparse(y)
        full_T = build_conv_matrix_sparse(y)
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.03 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                a = y.copy()
                T_mat = build_conv_matrix_sparse(y)
                full_T = build_conv_matrix_sparse(y)
                stall_count = 0
                prev_b = None
                prev_M = None
            if (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={np.mean(lp_times[-20:]):.3f}s")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        a = (x * (1 - x)) ** 0.7 + 0.05
    else:
        a = np.exp(-3 * (x - 0.5) ** 2) + 0.3 * np.exp(-5 * (x - 0.2) ** 2) + 0.3 * np.exp(-5 * (x - 0.8) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses active-set LP alternating, multi-length exploration, and LSE polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_parent = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick MW on parent for warm start ===
    print(f"\n=== Phase 1: Quick MW n={n_parent} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.02, 30.0)
    if t1 > 5:
        a_mw, _ = multiplicative_weights_adaptive(best_seq, t1, n_steps=200000, eta0=0.05)
        try_update(a_mw, "mw_init")

    # === Phases 2-5: LP alternating at different lengths ===
    n_values = [n_parent, 1400, 1600, 1800]
    # Remove duplicates and sort
    n_values = sorted(set(n_values))
    # Ensure parent n is first
    n_values = [n_parent] + [x for x in n_values if x != n_parent]
    
    print(f"\n=== LP Phases: n_values={n_values} ===")
    
    # Allocate time: 40% to parent n, rest split
    total_lp_budget = (deadline - time.monotonic()) * 0.55
    
    for idx, n_val in enumerate(n_values):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        
        if idx == 0:
            t_lp = min(total_lp_budget * 0.4, rem * 0.4)
        else:
            t_lp = min(total_lp_budget * 0.6 / (len(n_values) - 1), rem * 0.3)
        
        if t_lp < 30:
            break
        
        print(f"\n--- LP Phase n={n_val}, budget={t_lp:.0f}s ---")
        
        # Initialize for this n
        if n_val == n_parent:
            init_seq = best_seq.copy()
        else:
            init_seq = interpolate_sequence(best_seq, n_val)
        
        # Quick MW warmup
        t_mw = min(t_lp * 0.08, 20.0)
        if t_mw > 3:
            a_mw_n, _ = multiplicative_weights_adaptive(init_seq, t_mw, n_steps=100000, eta0=0.05)
            try_update(a_mw_n, f"mw_n{n_val}")
            init_seq = a_mw_n
        
        # LP alternating
        rem = deadline - time.monotonic()
        t_lp_actual = min(t_lp - t_mw, rem * 0.5)
        if t_lp_actual > 20:
            try:
                T_mat = build_conv_matrix_sparse(init_seq)
                full_T = build_conv_matrix_sparse(init_seq)
                a_lp, _ = nesterov_lp_refine(init_seq, t_lp_actual, solver, T_mat, full_T, n_max_iters=500000)
                try_update(a_lp, f"lp_n{n_val}")
            except Exception as e:
                print(f"  LP error for n={n_val}: {e}")
    
    # === Phase 6: LSE L-BFGS polish on best ===
    print(f"\n=== Phase 6: LSE Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t6 = min(rem * 0.25, 180.0)
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        print(f"  {len(T_stages)} stages, budget={t6:.0f}s")
        a_lse = lbgfs_refine(best_seq, t6, T_stages, maxiter_per_stage=200)
        try_update(a_lse, "lse_polish")

    # === Phase 7: Palindromic refinement ===
    print(f"\n=== Phase 7: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t7 = min(rem * 0.15, 120.0)
        a_pal = make_palindromic(best_seq)
        try:
            T_mat = build_conv_matrix_sparse(a_pal)
            full_T = build_conv_matrix_sparse(a_pal)
            a_pal_lp, _ = nesterov_lp_refine(a_pal, t7, solver, T_mat, full_T, n_max_iters=200000)
            try_update(a_pal_lp, "pal_lp")
        except Exception as e:
            print(f"  Pal LP error: {e}")

    # === Phase 8: Perturbation escapes ===
    print(f"\n=== Phase 8: Perturbation Escapes (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.008 * trial)
            t_sub = min(rem * 0.2, 50.0)
            if t_sub < 10:
                break
            # Quick MW + short LP
            a_pert_mw, _ = multiplicative_weights_adaptive(a_pert, t_sub * 0.3, n_steps=50000, eta0=0.04)
            try_update(a_pert_mw, f"pert_mw_{trial}")
            rem = deadline - time.monotonic()
            t_lp_sub = min(rem * 0.3, t_sub * 0.6)
            if t_lp_sub > 10:
                try:
                    T_mat = build_conv_matrix_sparse(a_pert_mw)
                    full_T = build_conv_matrix_sparse(a_pert_mw)
                    a_opt, _ = nesterov_lp_refine(a_pert_mw, t_lp_sub, solver, T_mat, full_T, n_max_iters=100000)
                    try_update(a_opt, f"pert_lp_{trial}")
                except Exception:
                    pass

    # === Phase 9: Final LP polish on best ===
    print(f"\n=== Phase 9: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.4, 100.0)
        try:
            T_mat = build_conv_matrix_sparse(best_seq)
            full_T = build_conv_matrix_sparse(best_seq)
            a_fin, _ = nesterov_lp_refine(best_seq, t9, solver, T_mat, full_T, n_max_iters=200000)
            try_update(a_fin, "final_lp")
        except Exception as e:
            print(f"  Final LP error: {e}")

    # === Phase 10: Final MW polish ===
    print(f"\n=== Phase 10: Final MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t10 = min(rem * 0.5, 30.0)
        a_fin_mw, _ = multiplicative_weights_adaptive(best_seq, t10, n_steps=100000, eta0=0.01)
        try_update(a_fin_mw, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
