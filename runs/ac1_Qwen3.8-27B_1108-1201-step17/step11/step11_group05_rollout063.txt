import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            rows.append(i + j)
            cols.append(j)
            vals.append(a[i])
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK, cp.SCS]:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(T_sp, solver, warm_start=None):

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    seen = set()
    for s in solvers_to_try:

        seen.add(s)

        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))

        if np.max(np.abs(a - a_prev_for_T)) > 1e-6:
            T_sp = build_conv_matrix_sparse(a)
            a_prev_for_T = a.copy()

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(T_sp, solver, warm_start=B_prev)
        B_prev = b_new
        lp_time = time.monotonic() - lp_t0

        if b_new is None:
            continue

        # Nesterov dual averaging step
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / ((it + 1 + alpha * (it + 2)))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 30:
                # Perturb y to escape
                noise = 0.02 * np.random.randn(n)
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-10:
                    y = y / sy
                stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} lp_time={lp_time*1000:.1f}ms")

        if (it + 1) > 50 and lp_time > 3.0:
            avg_lp = np.mean([lp_time])
            if avg_lp > 2.0:
                print(f"    LP too slow ({lp_time:.2f}s), stopping after {it+1} iters")
                break

    return best_a, best_score


def adam_logsumexp(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.1):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0

    total_stages = len(T_stages)
    steps_per_stage = max(2000, int(time_limit * 20000 / max(1, total_stages)))

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break

        top_k = max(5, int(top_k_frac * (2 * n - 1)))

        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break

            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute gradient
            if T_stage < 0.01:
                threshold = max_b * (1.0 - 1e-5)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    max_b_all = b.max()
                    exp_all = np.exp((b - max_b_all) / T_stage)
                    w = exp_all / exp_all.sum()

            # Gradient in a-space: g_a[i] = 2 * sum_k w[k] * a[k-i]
            # Use correlate
            g_a_raw = np.correlate(w, a, mode='full')
            g_a = g_a_raw[n - 1:n + n - 1] * 2.0

            # Gradient in z-space (softmax chain rule)
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)

            # Adam update
            lr = lr_init * (lr_decay ** stage_idx)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Centering
            z -= z.mean()

            if (step + 1) % 10000 == 0:
                elapsed = time.monotonic() - t0
                cur_score = 2.0 * n * fast_convolve(softmax_from_z(z)).max()
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {step+1} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f}")

    return best_a, best_score


def multiplicative_subgradient(a0, time_limit, n_steps=200000):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.02
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Adaptive active set
        frac = max(1e-5, 5e-4 / (1 + step / 500))
        threshold = max_b * (1.0 - frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute gradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * x) + 0.15 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = np.random.dirichlet(np.ones(n) * 0.5)
    elif choice == 8:
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 5 * x + 0.7) + 0.1 * np.cos(2 * np.pi * 11 * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
