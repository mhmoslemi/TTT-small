def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_rows(a, rows):
    """Build rows of (2n-1) x n matrix T for specified row indices."""
    n = len(a)
    T = np.zeros((len(rows), n))
    for i, k in enumerate(rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[i, idx] = a[k - idx]
    return T


def get_lp_solver():

def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def peak_shave_fast(a, time_limit, n_steps=200000):
    """Fast peak shaving using FFT-based convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.05
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Weight active peaks by their excess
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient from active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Mirror descent update
        eta_eff = eta * scale
        log_mult = 1.0 - eta_eff * g
        log_mult = np.maximum(log_mult, 1e-15)
        a_new = a * log_mult
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def gradient_logsumexp(a, T_temp):
    """Compute gradient of LogSumExp smoothing of max autocorrelation."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    w = exp_terms / exp_terms.sum()
    
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += w[k] * a[k - idx]
    
    g -= g @ a  # project to sum=0 (since we're on the simplex)
    return g

def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000, top_k=200):
    """Nesterov LP alternating with reduced constraints."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

                # Switch to larger top_k (more constraints = better solution)
                top_k = min(2 * n - 1, top_k * 2)
                print(f"    LP slow ({avg_lp_time:.2f}s), increasing top_k to {top_k}")
                lp_times = []

                print(f"    LP too slow, stopping at iter {it+1}")
                break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Periodic spectral pre-conditioning
        if (it + 1) % 50 == 0:

            sc_spec = 2.0 * n * fast_convolve(a_spec).max()

            else:

            prev_best = best_score

            if stall_count >= 5:

                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)

                y = y + noise

                sy = y.sum()

                elapsed = time.monotonic() - t0

def lse_lbfgs_refine(a0, time_limit, deadline=None):
    """L-BFGS on LogSumExp smoothed objective with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    # Work in log-domain for non-negativity
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    
    T_stages = [5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
    n_stages = len(T_stages)
    
    for stage_i, T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        stages_left = n_stages - stage_i
        iters = max(20, min(500, int(remaining * 30 / stages_left)))
        
        def obj_func(z_in):
            a_tmp = softmax_z(z_in)
            b = fast_convolve(a_tmp)
            max_b = b.max()
            exp_terms = np.exp((b - max_b) / T)
            lse = max_b + T * np.log(exp_terms.sum() + 1e-30)
            return 2.0 * n * lse
        
        def obj_grad(z_in):
            a_tmp = softmax_z(z_in)
            b = fast_convolve(a_tmp)
            max_b = b.max()
            exp_terms = np.exp((b - max_b) / T)
            w = exp_terms / exp_terms.sum()
            
            g_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g_a[idx] += w[k] * a_tmp[k - idx]
            
            S = np.dot(g_a, a_tmp)
            g_z = a_tmp * (g_a - S)
            return 2.0 * n * g_z
        
        try:
            res = minimize(
                fun=obj_func, x0=z, jac=obj_grad, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_result = softmax_z(z)
    return a_result


def softmax_z(z):
    """Convert log-domain to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def structured_init(n, rng):
    """Create structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 2 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / (n / 4)
        a = np.sinc(t / 2)
        a = np.maximum(a, 0)
    elif choice == 7:
        # Optimal for flat autocorrelation: use cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 8:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    else:
        # Random smooth
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)

    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_rows(a, row_indices):

def fast_convolve_weighted(a, b):
    """Compute convolve(a, b) where a and b may be different (both length n)."""

    size = 2 * n - 1

    return irfft(A * B, fft_size)[:size]


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Spectral flattening via Gerchberg-Saxton."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    return a


def project_simplex(v, s=1.0):
    """Project v onto simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho) == 0:
        return np.maximum(v, 0) / max(1e-30, np.maximum(v, 0).sum()) * s
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
