def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov accelerated subgradient with top-K active set (FFT-free, O(n*K))
2. LP alternating with Nesterov averaging and sparse matrices
3. Coordinate descent local refinement with 1D golden-section search
4. Multi-length (n=800..2500), multi-restart exploration
Key insight: subgradient of 2n*max_k (a*a)[k] is 4n * cross_correlation(a, active_indicator),
computed in O(n*K) where K is the number of top active lags.

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_subgrad(a, b, top_k=50, temp_frac=0.05):
    n = len(a)
    max_b = b.max()
    if max_b <= 0:
        return np.zeros(n)
    top_k = min(top_k, len(b))
    top_idx = np.argpartition(b, -top_k)[-top_k:]
    top_vals = b[top_idx]
    temp = max_b * temp_frac
    if temp > 1e-30:
        weights = np.exp((top_vals - max_b) / temp)
    else:
        weights = np.ones(top_k)
    weights /= weights.sum()
    s = np.zeros(n)
    for idx in range(top_k):
        j = top_idx[idx]
        w = weights[idx]
        i_lo = max(0, j - n + 1)
        i_hi = min(n - 1, j)
        if i_lo <= i_hi:
            s[i_lo:i_hi + 1] += w * a[j - i_lo:j - i_hi - 1:-1]
    return 4.0 * n * s


def project_simplex(v, target_sum=1.0):
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v * (target_sum / s)


def nesterov_subgrad(a0, time_limit, top_k=50, temp_frac=0.05, eta0=0.02):
    n = len(a0)
    a = project_simplex(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    v = np.zeros(n)
    avg_a = a.copy()
    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        b_y = fast_convolve(y)
        s_y = compute_subgrad(y, b_y, top_k, temp_frac)
        # Nesterov momentum
        beta = (t + 1.0) / (t + 1.0 + t + 2.0)
        beta = t / (t + 3.0) if t > 0 else 0.0
        v = beta * v + (1 - beta) * s_y
        # Adaptive step size
        fv = evaluate_local(y)
        eta = eta0 / (1.0 + 0.01 * t)
        # Backtracking
        for _ in range(20):
            y_new = project_simplex(y - eta * v)
            b_new = fast_convolve(y_new)
            max_new = b_new.max()
            max_old = b_y.max()
            if max_new <= max_old + eta * np.dot(s_y, y_new - y) + 1e-12:
                break
            eta *= 0.5
        else:
            y_new = y.copy()
        y = y_new
        avg_a = avg_a + (y - avg_a) / (t + 1)
        sc_y = 2.0 * n * fast_convolve(y).max()
        sc_avg = 2.0 * n * fast_convolve(avg_a).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
        if (t + 1) % 5000 == 0:
            print(f"      NS t={t+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} eta={eta:.6f}")
    return best_a, best_score


def build_conv_matrix_sparse(a):
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    return None, None
                return b_val / sv, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None

            step = 0.01 / (a.sum() / n)
            cand_grad = max(0.0, old_val - step * s_i)
            candidates.append(cand_grad)
            candidates.append(cand_grad * 0.5)

            sc = 2.0 * n * b.max()
            if sc < best_score:

        if sweep % 10 == 0:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, choice=1)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick Nesterov subgradient on parent ===
    print(f"\n=== Phase 1: Nesterov Subgrad n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t1 = min(rem * 0.08, 120.0)
        a_ns, _ = nesterov_subgrad(best_seq, t1, top_k=30, temp_frac=0.1, eta0=0.01)
        try_update(a_ns, "ns1")

    # === Phase 2: LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t2 = min(rem * 0.30, 540.0)
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = lp_alternating(best_seq, t2, solver)
        try_update(a_lp, "lp1")

    # === Phase 3: Nesterov subgradient with different params ===
    print(f"\n=== Phase 3: NS refined (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.12, 200.0)
        # Different top_k and temperature
        a_ns2, _ = nesterov_subgrad(best_seq, t3, top_k=100, temp_frac=0.02, eta0=0.005)
        try_update(a_ns2, "ns2")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t4 = min(rem * 0.10, 180.0)
        a_cd, _ = coord_descent(best_seq, t4, n_sweeps=100)
        try_update(a_cd, "cd1")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [800, 1000, 1500, 2000, 2500]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.15, 180.0)
        if t_ln < 40:
            continue
        print(f"  n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_seq(best_seq, ln)
        # Quick NS
        t_ns = min(t_ln * 0.25, 50.0)
        a_ln_ns, _ = nesterov_subgrad(init_ln, t_ns, top_k=30, temp_frac=0.08, eta0=0.01)
        try_update(a_ln_ns, f"ns_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.5)
        if t_lp > 30:
            a_ln_lp, _ = lp_alternating(a_ln_ns if a_ln_ns is not None else init_ln, t_lp, solver)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 6: Different initializations + LP ===
    print(f"\n=== Phase 6: Multi-restart (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_tries = 3
    for trial in range(n_tries):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_sub = min(rem * 0.25, 120.0)
        init = structured_init(n_best, rng, choice=trial + 3)
        # Quick NS to shape it
        t_ns = min(t_sub * 0.3, 40.0)
        a_shaped, _ = nesterov_subgrad(init, t_ns, top_k=20, temp_frac=0.1, eta0=0.02)
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.5, t_sub * 0.6)
        if t_lp > 20:
            a_opt, _ = lp_alternating(a_shaped, t_lp, solver)
            try_update(a_opt, f"restart{trial}")

    # === Phase 7: Final LP polish on best ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t7 = min(rem * 0.30, 150.0)
        a_fin, _ = lp_alternating(best_seq, t7, solver)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final NS + CD ===
    print(f"\n=== Phase 8: Final NS+CD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t8 = min(rem * 0.3, 60.0)
        a_ns_f, _ = nesterov_subgrad(best_seq, t8, top_k=20, temp_frac=0.03, eta0=0.005)
        try_update(a_ns_f, "final_ns")
        rem = deadline - time.monotonic()
        if rem > 20:
            t9 = min(rem * 0.4, 40.0)
            a_cd_f, _ = coord_descent(best_seq, t9, n_sweeps=20)
            try_update(a_cd_f, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
