def evaluate_sequence(seq):
    n = len(seq)
    seq = np.clip(np.array(seq), 0, None)  # Remove upper bound
    s = seq.sum()
    if s < 1e-5: return np.inf
    convolved = np.convolve(seq, seq, mode='full')[:n-1]
    normalized_conv = convolved / (s * s)
    return np.max(normalized_conv)
