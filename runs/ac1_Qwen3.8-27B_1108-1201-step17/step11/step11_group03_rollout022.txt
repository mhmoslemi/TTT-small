import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_q_sparse(k, n):
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)

def solve_active_qcqp_full(a_init, solver, max_iter=50, tol=1e-8):
    """
    Active-set QCQP: iteratively add constraints until no more are violated.
    min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    # Initial active set
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1 - 0.05)
    active = set(np.where(b >= threshold)[0].tolist())
    
    best_a = a.copy()
    best_score = 2.0 * n * max_b
    
    for iteration in range(max_iter):
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]
        
        for k in sorted(active):
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        solved = False
        for s_solver in [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if s_solver is None:
                continue
            try:
                prob.solve(solver=s_solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except:
                continue
        
        if not solved:
            break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check for violated constraints
        b_new = fast_convolve(a_new)
        max_b_new = b_new.max()
        score_new = 2.0 * n * max_b_new
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        # Find newly violated constraints
        violated = np.where(b_new > t_var.value * (1 + tol))[0]
        if len(violated) == 0:
            # Check all constraints
            violated = np.where(b_new > max(b_new[sorted(active)]) * (1 + 1e-6))[0]
            if len(violated) == 0:
                break
        
        # Add violated to active set
        new_active = violated.tolist()
        active.update(new_active)
        
        # Limit active set size
        if len(active) > 50:
            # Keep only the most violated
            b_vals = b_new[sorted(active)]
            # Keep top 50
            active_list = sorted(active)
            b_for_active = b_new[active_list]
            top_indices = np.argsort(b_for_active)[-50:]
            active = set(active_list[i] for i in top_indices)
        
        a = a_new
    
    return best_a, best_score

    active = sorted(set(np.where(b >= threshold)[0].tolist()))
    
    best_a = a.copy()
    best_t = max_b
    converged = False
    
    for round_num in range(max_rounds):
        if verbose:
            print(f"    Round {round_num}: |active|={len(active)}, t={best_t:.10f}")
        
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]
        
        for k in active:
            Q = build_q_sparse(k, n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        solved = False

            if s_solver is None:
                continue
            try:
                prob.solve(solver=s_solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            if verbose:
                print(f"    Failed to solve QCQP in round {round_num}")
            break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new
        
        b_new = fast_convolve(a_new)
        t_new = float(t_var.value)
        max_b_new = b_new.max()
        
        if max_b_new < best_t:
            best_t = max_b_new
            best_a = a_new.copy()
        
        # Check for violated constraints (using the actual convolution, not the constraint)

This is very efficient to compute! For each active lag k, we just need to compute the "anti-diagonal" of a shifted version of a.

1. Compute `g = ∇f(a)` (subgradient)
2. Update: `a_new = proj_simplex(a - lr * g)`
3. Optionally use Nesterov acceleration

The projection onto the simplex can be done in O(n log n) time.

This is much faster than solving an LP at each step! Let me implement this.

Wait, I need to think about the step size. The function `f(a) = max_k g_k(a)` is Lipschitz continuous on the simplex (with respect to the Euclidean metric). The Lipschitz constant can be bounded.

For `g_k(a) = sum_j a[j] a[k-j]`, the gradient is `2 * shift_reflect(a, k)`. The norm of the gradient is `2 * ||shift_reflect(a, k)||_2 <= 2 * ||a||_2 <= 2 * ||a||_1 = 2` (since a is on the simplex, ||a||_1 = 1 and ||a||_2 <= ||a||_1 = 1).

So the Lipschitz constant is at most 2. With Nesterov, the optimal step size is `lr = L / t` where L is the Lipschitz constant.

But wait, we're on the simplex, and the projection matters. The standard Nesterov for constrained problems uses `lr = L / t` and the projection handles the constraint.

Actually, for the simplex constraint, the standard approach is:
- `lr_t = 2 * L / (t * (t+1))` (for the "standard" Nesterov)
- Or `lr_t = L / t` (for the "accelerated" version)

But I need to be more careful. The function f is not convex, so the standard Nesterov analysis doesn't directly apply. However, empirically, it might still work.

Let me try a simpler approach first: **mirror descent with entropy regularization**.

Mirror descent on the simplex uses the KL divergence as the regularizer:
`a_{t+1}[i] = a_t[i] * exp(-lr * g[i]) / sum_j a_t[j] * exp(-lr * g[j])`

This is the multiplicative weights update. It's known to work well for problems on the simplex.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP alternating with Nesterov acceleration (primary method)
2. Mirror descent (multiplicative weights) for fast local polish
3. Symmetry exploitation to reduce problem size
4. Multi-n exploration (n=1199-1210)
5. Active-set refinement for final precision

Key insight: The problem min_{a in simplex} max_k conv(a,a)[k] is non-convex,
but LP alternating converges to the correct value by the minimax theorem.
Mirror descent provides fast local improvements without requiring a solver.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csc_matrix, lil_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def subgradient_of_max_conv(a, rel_tol=1e-8):
    """
    Compute subgradient of f(a) = max_k conv(a,a)[k] at point a.
    Using: ∂g_k/∂a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    Returns (subgradient, active_set, max_b)
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - rel_tol)
    K = np.where(b >= threshold)[0]
    
    g = np.zeros(n)
    for k in K:
        # g[i] += 2 * a[k-i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        g[idx] += 2.0 * a[k - idx]
    
    g /= len(K)
    return g, K, max_b


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def mirror_descent_refine(a0, time_limit, max_iters=10000000):
    """
    Mirror descent (multiplicative weights) for local refinement.
    Uses subgradient of max convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Adaptive learning rate
    base_lr = 0.5
    lr_multiplier = 1.0
    no_improve = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        g, K, max_b = subgradient_of_max_conv(a, rel_tol=1e-10)
        
        # Learning rate: O(1/sqrt(t)) for subgradient methods
        lr = base_lr * lr_multiplier / np.sqrt(t)
        
        # Mirror descent update (multiplicative weights)
        log_update = -lr * g
        # Numerical stability
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new  # still take the step
            no_improve += 1
            if no_improve > 1000:
                lr_multiplier *= 0.5
                no_improve = 0
                if lr_multiplier < 1e-6:
                    break
        
        if t % 100000 == 0:
            print(f"    MD iter={t}: score={best_score:.10f}")
    
    return best_a, best_score
