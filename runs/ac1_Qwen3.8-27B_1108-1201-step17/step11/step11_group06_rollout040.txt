def exponential_gradient_step(a, k_top, eta):
    """One step of top-k exponential gradient."""
    n = len(a)
    C = fast_convolve(a)
    # Find top-k positions
    if k_top < len(C):
        top_idx = np.argpartition(C, -k_top)[-k_top:]
        w = np.zeros(len(C))
        w[top_idx] = C[top_idx] / C[top_idx].sum()
    else:
        w = C / C.sum()
    # Compute gradient
    w_flipped = w[::-1]
    c = np.convolve(a, w_flipped)
    grad = c[2*n-2 : n-2 : -1]
    # Exponential update
    log_mult = -eta * grad
    log_mult -= log_mult.max()
    a_new = a * np.exp(log_mult)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-300:
        return a, C.max()
    a_new = a_new / s
    return a_new, C.max()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]
