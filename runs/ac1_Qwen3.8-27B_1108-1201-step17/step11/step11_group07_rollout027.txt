```python
import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr(w, a):
    """Compute cross-correlation: result[m] = sum_k w[k]*a[k-m] for m=0..n-1"""
    n = len(a)
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    return R[:n]

def lse_smooth_max(c, T):
    """Compute LSE-smoothed maximum and softmax weights."""
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    w = exp_c / exp_c.sum()
    return lse, w

def gradient_lse(a, T):
    """Gradient of LSE-smoothed 2n*max(conv(a,a)) w.r.t. a.
    Assumes sum(a) = 1."""
    n = len(a)
    c = fast_convolve(a)
    lse, w = lse_smooth_max(c, T)
    obj = 2.0 * n * lse
    # grad[m] = 2 * sum_k w[k] * a[k-m]
    g = 2.0 * cross_corr(w, a)
    return obj, g

def active_set_gradient(a, active_lags):
    """Compute gradient of average c[k] over active lags.
    grad[m] = (2/|S|) * sum_{k in S} a[k-m]"""
    n = len(a)
    w = np.zeros(2*n-1)
    w[active_lags] = 1.0 / len(active_lags)
    g = 2.0 * cross_corr(w, a)
    return g

def project_feasible(a):
    """Project to a >= 0, sum a = 1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def evaluate_local(a):
    a = np.maximum(np.asarray(a), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))
```

```python
    return irfft(A * A, fft_size)[:size]

    """Cross-correlation: result[m] = sum_k w[k]*a[k-m] for m=0..n-1."""
    n = len(a)
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    return R[:n]

    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    g = 2.0 * cross_corr(w, a)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gauss-Seidel coordinate descent with exact closed-form 1D sub-solves
2. LSE (LogSumExp) smoothed gradient descent with temperature annealing
3. Spectral-domain perturbation for diverse restarts
4. Adaptive multi-length exploration
Key innovation: exploiting the linearity of each convolution lag in individual
coordinates for exact O(1) sub-solve updates, enabling millions of passes.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def cross_corr(w, a):
    n = len(a)
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    return R[:n]


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_smoothed_obj_grad(a, T):
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    obj = 2.0 * n * (c_max + T * np.log(exp_c.sum()))
    g = 2.0 * cross_corr(w, a)
    return obj, g


def lse_gradient_descent(a0, time_limit, T_init=10.0, T_final=0.01, n_stages=8):
    """LSE-smoothed gradient descent with annealing."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    T = T_init
    for stage in range(n_stages):
        stage_time = time_limit / n_stages
        t_stage = time.monotonic()
        lr = 0.1 / (n ** 0.5)
        
        while time.monotonic() - t_stage < stage_time:
            if time.monotonic() - t0 > time_limit:
                break
            
            _, g = lse_smoothed_obj_grad(a, T)
            g = g - g.mean()  # Project to sum-zero
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-12:
                break
            g = g / g_norm
            
            improved = False
            for scale in [lr, lr*0.5, lr*0.25, lr*0.1]:
                a_new = normalize(a - scale * g)
                sc = evaluate_local(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True
                    break
            
            if not improved:
                lr *= 0.5
                if lr < 1e-8:
                    break
        
        T = max(T_final, T * (T_final / T_init) ** (1.0 / n_stages))
    
    return best_a, best_score


def gauss_seidel_optimize(a0, time_limit, verbose=False):
    """Gauss-Seidel coordinate descent with exact closed-form sub-solves.
    
    Key insight: For fixed target lag k*, c[k*] is linear (or quadratic)
    in each individual coordinate a[m]. This allows exact O(1) sub-solves.
    """
    n = len(a0)
    a = normalize(a0.copy().astype(np.float64))
    S = a.sum()
    
    c = fast_convolve(a)
    k_star = int(np.argmax(c))
    c_ks = float(c[k_star])
    
    best_a = a.copy()
    best_score = float(2.0 * n * c.max() / (S * S))
    t0 = time.monotonic()
    iteration = 0
    stall_count = 0
    prev_best = best_score
    
    while time.monotonic() - t0 < time_limit:
        # Full Gauss-Seidel pass targeting k_star
        for m in range(n):
            km = k_star - m
            if 2 * m == k_star:
                # Quadratic case: c[k*] = rest + a[m]^2
                am = a[m]
                rest = c_ks - am * am
                denom = S - am
                if denom > 1e-15 and rest > 0:
                    t_opt = rest / denom
                elif denom > 1e-15:
                    t_opt = 0.0
                else:
                    continue
                t_opt = max(0.0, t_opt)
                new_c_ks = rest + t_opt * t_opt
                new_S = S - am + t_opt
                if new_c_ks < c_ks - 1e-14 and new_S > 1e-10:
                    c_ks = new_c_ks
                    S = new_S
                    a[m] = t_opt
            elif 0 <= km < n:
                # Linear case: c[k*] = rest + 2*a[m]*a[km]
                ak = a[km]
                if ak < 1e-15:
                    continue
                am = a[m]
                t_opt = S + am - c_ks / ak
                t_opt = max(0.0, t_opt)
                new_c_ks = c_ks + 2.0 * (t_opt - am) * ak
                new_S = S - am
