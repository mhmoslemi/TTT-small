def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation assuming a >= 0, sum > 0."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        lo = max(0, k - n + 1) if False else 0
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    return None, None
                return b_val / sv, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def lp_alternating(a0, time_limit, solver, n_max_iters=200000, verbose=True):
    """
    LP alternating method with Nesterov acceleration.
    Minimizes max_k (p*p)(k) over the simplex via block coordinate descent.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)
    T_mat = build_conv_matrix_sparse(y)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        lt0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_times.append(time.monotonic() - lt0)
        if b_new is None:
            continue
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 10.0:
                if verbose:
                    print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping")
                break
        # Nesterov-style update
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
        else:
            y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        # Rebuild T matrix periodically (not every iter for speed)
        if (it + 1) % 5 == 0:
            T_mat = build_conv_matrix_sparse(y)
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 8:
                # Perturb to escape
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.03 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                T_mat = build_conv_matrix_sparse(y)
                stall = 0
            if (it + 1) % 500 == 0 and verbose:
                print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    return best_a, best_score


def frank_wolfe_away(a0, time_limit, n_max_iters=500000, verbose=True):
    """
    Frank-Wolfe with away steps for min max_k (p*p)(k) over simplex.
    Uses the fact that the linear minimization oracle is trivial.
    """
    n = len(a0)
    p = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    best_p = p.copy()
    best_score = 2.0 * n * fast_convolve(p).max()
    t0 = time.monotonic()
    # Maintain representation p = sum_i alpha_i * e_{i}
    # But for simplicity, just track the support
    support = list(np.where(p > 1e-10)[0])
    if not support:
        support = [0]

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(p)
        max_b = b.max()
        # Find active set
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        # Compute subgradient (average over active)
        g = np.zeros(n)
        for k in active:
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    g[j] += p[ki]
        g /= len(active)
        # FW step: find min of g
        i_min = np.argmin(g)
        g_min = g[i_min]
        # Away step: find max of g in support
        if len(support) > 1:
            g_support = np.array([g[i] for i in support])
            i_away = support[np.argmax(g_support)]
            g_away = g[i_away]
            # Check if away step is better
            if g_away - g_min > 1e-10:
                # Away-forward step
                gamma = 2.0 / (it + 2)
                p_new = p.copy()
                p_new[i_away] -= gamma * p[i_away]
                p_new[i_min] += gamma * (1 - p[i_away])
                p_new = np.maximum(p_new, 0)
                sp = p_new.sum()
                if sp > 1e-15:
                    p_new = p_new / sp
                    score = 2.0 * n * fast_convolve(p_new).max()
                    if score < best_score:
                        best_score = score
                        best_p = p_new.copy()
                    p = p_new
                    # Update support
                    support = list(np.where(p > 1e-8)[0])
                    if i_min not in support and g_min < g[i_away]:
                        support.append(i_min)
                continue
        # Normal FW step
        gamma = 2.0 / (it + 2)
        p_new = (1.0 - gamma) * p + gamma * np.zeros(n)
        p_new[i_min] = gamma
        p_new = np.maximum(p_new, 0)
        score = 2.0 * n * fast_convolve(p_new).max()
        if score < best_score:
            best_score = score
            best_p = p_new.copy()
        p = p_new
        # Update support
        support = list(np.where(p > 1e-8)[0])
        if len(support) > 50:
            # Prune support
            idx = np.argsort(p)[::-1][:30]
            support = list(idx)
            p_pruned = np.zeros(n)
            p_pruned[support] = p[support]
            p = p_pruned / p_pruned.sum()

        if (it + 1) % 5000 == 0 and verbose:
            print(f"    FW iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    return best_p, best_score


def multiplicative_weights(a0, time_limit, n_max_iters=200000, eta0=0.05, verbose=True):
    """Exponential weights / multiplicative weights method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.2)
        else:
            eta *= 0.85
            if eta < 1e-9:
                break
    return best_a, best_score


def lse_objective_grad(z, n, T):
    """Compute LogSumExp objective and gradient."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


class LSEFunc:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = lse_objective_grad(z, self.n, self.T)
        return obj


class LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = lse_objective_grad(z, self.n, self.T)
        return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=300):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 5:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy() / f.sum()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, choice=None):

def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)

    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

                a_lp, _ = lp_alternating(a_mw if a_mw is not None else a_init, t_lp, solver,
                                         n_max_iters=50000, verbose=False)
                try_update(a_lp, f"lp_n{n_small}_c{init_choice}")
        # LSE polish on best for this n
        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == n_small:
            t_lse = min(rem * 0.4, 40.0)
            T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_lse = lbgfs_refine(best_seq, t_lse, T_stages, maxiter_per_stage=200)
            try_update(a_lse, f"lse_n{n_small}")

    # === Phase 2: Medium-n search (n=400 to 800) ===
    print(f"\n=== Phase 2: Medium-n search (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for n_med in [400, 600]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_budget = min(rem * 0.18, 180.0)
        if t_budget < 30:
            continue
        print(f"  n={n_med}, budget={t_budget:.0f}s")
        # Interpolate from best
        if best_seq is not None:
            a_init = interpolate_sequence(best_seq, n_med)
        else:
            a_init = structured_init(n_med, rng, choice=7)
        # MW warm-up
        t_mw = min((deadline - time.monotonic()) * 0.2, 30.0)
        if t_mw > 5:
            a_mw, _ = multiplicative_weights(a_init, t_mw, n_max_iters=100000, eta0=0.05)
            try_update(a_mw, f"mw_n{n_med}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.6, 100.0)
        if t_lp > 20:
            a_lp, _ = lp_alternating(a_mw if a_mw is not None else a_init, t_lp, solver,
                                     n_max_iters=100000, verbose=False)
            try_update(a_lp, f"lp_n{n_med}")

    # === Phase 3: Parent-length optimization (if parent exists) ===
    print(f"\n=== Phase 3: Parent-length LP (score={best_score:.8f}) ===")
    if parent is not None:
        n_parent = len(parent)
        rem = deadline - time.monotonic()
        # Interpolate best to parent length for a better starting point
        if best_seq is not None and len(best_seq) != n_parent:
            a_start = interpolate_sequence(best_seq, n_parent)
        else:
            a_start = parent.copy()
        t_lp = min(rem * 0.25, 240.0)
        if t_lp > 30:
            print(f"  n={n_parent}, budget={t_lp:.0f}s")
            a_lp, _ = lp_alternating(a_start, t_lp, solver, n_max_iters=200000, verbose=True)
            try_update(a_lp, f"lp_parent_n{n_parent}")
            # Palindromic variant
            rem = deadline - time.monotonic()
            if rem > 60:
                a_pal = make_palindromic(a_lp if a_lp is not None else a_start)
                t_pal = min(rem * 0.3, 120.0)
                a_pal_lp, _ = lp_alternating(a_pal, t_pal, solver, n_max_iters=100000, verbose=False)
                try_update(a_pal_lp, f"pal_lp_n{n_parent}")

    # === Phase 4: Frank-Wolfe with away steps ===
    print(f"\n=== Phase 4: FW Away Steps (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50 and best_seq is not None:
        n_fw = len(best_seq)
        t_fw = min(rem * 0.1, 80.0)
        if t_fw > 10:
            a_fw, _ = frank_wolfe_away(best_seq, t_fw, n_max_iters=200000, verbose=False)
            try_update(a_fw, f"fw_n{n_fw}")

    # === Phase 5: LSE L-BFGS fine polishing ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        n_lse = len(best_seq)
        t_lse = min(rem * 0.2, 200.0)
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        print(f"  n={n_lse}, {len(T_stages)} stages, budget={t_lse:.0f}s")
        a_lse = lbgfs_refine(best_seq, t_lse, T_stages, maxiter_per_stage=500)
        try_update(a_lse, "lse_polish")

    # === Phase 6: Perturbation escapes ===
    print(f"\n=== Phase 6: Perturbation escapes (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.008 * trial)
            t_sub = min(rem * 0.2, 60.0)
            # Use MW for quick escape then LP
            a_mw_esc, _ = multiplicative_weights(a_pert, t_sub * 0.3, n_max_iters=50000, eta0=0.04)
            try_update(a_mw_esc, f"pert_mw{trial}")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_lp_esc, _ = lp_alternating(a_mw_esc, t_sub * 0.5, solver,
                                             n_max_iters=30000, verbose=False)
                try_update(a_lp_esc, f"pert_lp{trial}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        # Quick MW
        t_fin = min(rem * 0.3, 40.0)
        a_fin_mw, _ = multiplicative_weights(best_seq, t_fin, n_max_iters=100000, eta0=0.02)
        try_update(a_fin_mw, "final_mw")
        # Quick LSE
        rem = deadline - time.monotonic()
        if rem > 20:
            T_final = [1.0, 0.5, 0.1, 0.05, 0.01, 0.005, 0.001]
            t_lse_fin = min(rem * 0.6, 60.0)
            a_fin_lse = lbgfs_refine(best_seq, t_lse_fin, T_final, maxiter_per_stage=300)
            try_update(a_fin_lse, "final_lse")

    # Final verification with actual evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        actual_score = evaluate_sequence(final_list)
        if np.isfinite(actual_score):
            print(f"\nFINAL (verified): score={actual_score:.8f} n={len(best_seq)}")
        else:
            print(f"\nWARNING: evaluate_sequence returned inf, using local score={best_score:.8f}")
        best_seq = np.asarray(final_list)
    else:
        best_seq = np.ones(300) / 300

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
