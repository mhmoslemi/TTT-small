```python
def mwu_iteration(a, y, eta):
    """
    One MWU iteration.
    a: primal variable (length n), sum = 1
    y: dual variable (length 2n-1), sum = 1
    Returns updated a, y
    """
    n = len(a)
    
    # Compute T @ a (length 2n-1): this is conv(a, a)
    Ta = fast_autocorr(a)  # length 2n-1
    
    # Compute T^T @ y (length n): cross-correlation
    # (T^T y)_j = sum_k y_k * T[k,j] = sum_k y_k * a[k-j] for 0<=k-j<n
    # = sum_{m=0}^{n-1} y[j+m] * a[m] (with bounds)
    # This is cross-correlation of y and a
    # conv(y, a_rev)[n-1+j] where a_rev = a[::-1]
    a_rev = a[::-1]
    conv_len = len(y) + n - 1  # = 2n-1+n-1 = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    Y = rfft(y, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(Y * A_rev, fft_size)
    Ty = conv[n-1:n-1+n]  # length n
    
    # Update y (column player): y_k <- y_k * exp(-eta * Ta_k) / Z
    # Wait, in the game, the column player wants to MAXIMIZE the payoff,
    # so the column player uses: y_k <- y_k * exp(eta * Ta_k) / Z
    # And the row player wants to MINIMIZE, so: a_j <- a_j * exp(-eta * Ty_j) / Z
    
    # Dual update (y maximizes):
    exp_y = np.exp(eta * Ta - eta * Ta.max())
    y_new = exp_y / exp_y.sum()
    
    # Primal update (a minimizes):
    exp_a = np.exp(-eta * Ty - (-eta * Ty).max())  # = exp(-eta * (Ty - Ty.min()))
    # Actually: a_j <- a_j * exp(-eta * Ty_j) / sum_j a_j * exp(-eta * Ty_j)
    # But we can just use: a_j <- exp(-eta * Ty_j) / sum (simpler, forgets old a)
    # Or: a_j <- a_j * exp(-eta * Ty_j) / sum (multiplicative update)
    
    # Let's use the "soft" version: a_j <- a_j * exp(-eta * Ty_j) / Z
    log_a = np.log(np.maximum(a, 1e-300)) - eta * Ty
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = a_new / a_new.sum()
    
    return a_new, y_new
```

- Column player (maximizing): y_k <- y_k * exp(+eta * (T a)_k) / Z

```python
def mwu_nesterov(n, time_limit, init_a=None, eta=None):
    """MWU with Nesterov acceleration for the autocorrelation minimization game."""
    t0 = time.monotonic()
    
    if init_a is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(init_a, 0)
        a = a / a.sum()
    
    y = np.ones(2*n-1) / (2*n-1)
    a_prev = a.copy()
    y_prev = y.copy()
    
    if eta is None:
        eta = sqrt(2 * log(2*n)) / (2*n-1)  # Rough initial guess
    
    best_a = a.copy()
    best_score = 2*n * fast_autocorr(a).max()
    
    avg_a = np.zeros(n)
    avg_y = np.zeros(2*n-1)
    n_avg = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        alpha = 2.0 / (t + 2.0)
        a_ext = a + (1 - alpha) * (a - a_prev)  # = (2-alpha)*a - (1-alpha)*a_prev
        a_ext = np.maximum(a_ext, 0)
        a_ext = a_ext / a_ext.sum()
        
        y_ext = y + (1 - alpha) * (y - y_prev)
        y_ext = np.maximum(y_ext, 0)
        y_ext = y_ext / y_ext.sum()
        
        # Compute payoffs
        Ta = fast_autocorr(a_ext)  # payoff for dual actions (length 2n-1)
        # T^T @ y_ext: cross-correlation
        a_rev = a_ext[::-1]
        conv_len = (2*n-1) + n - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        Ty = irfft(rfft(y_ext, fft_size) * rfft(a_rev, fft_size), fft_size)[n-1:n-1+n]
        
        # MWU updates
        a_new = a * np.exp(-eta * Ty)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        y_new = y * np.exp(eta * Ta - eta * Ta.max())
        y_new = y_new / y_new.sum()
        
        # Update
        a_prev = a
        y_prev = y
        a = a_new
        y = y_new
        
        # Running average
        avg_a += a
        avg_y += y
        n_avg += 1
        
        if n_avg % 100 == 0:
            avg_a_check = avg_a / n_avg
            score = 2*n * fast_autocorr(avg_a_check).max()
            if score < best_score:
                best_score = score
                best_a = avg_a_check.copy()
    
    return best_a, best_score
```

```python
def adam_lse_optimize(a0, time_limit, T_start=50, T_end=0.5, n_steps=5000):
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    a = a / a.sum()
    
    # Adam parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.01
    m = np.zeros(n)
    v = np.zeros(n)
    
    # Temperature annealing
    T = T_start
    
    best_a = a.copy()
    best_score = 2*n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        T = T_start * (T_end/T_start) ** (step / n_steps)
        
        # Compute gradient of LSE
        b = fast_autocorr(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        # Gradient: 2 * sum_j w[i+j] * a[j] (cross-correlation)
        a_rev = a[::-1].copy()
        conv_len = (2*n-1) + n - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        grad = 2.0 * irfft(rfft(w, fft_size) * rfft(a_rev, fft_size), fft_size)[n-1:n-1+n]
        
        # Adam update in the "unconstrained" space, then project
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        score = 2*n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if step % 500 == 0:
            print(f"    Adam step {step}: T={T:.2f}, score={best_score:.6f}")
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast MWU (Multiplicative Weight Update) + Nesterov acceleration on the zero-sum game
2. 1/sqrt(j+1) spectral initialization (theoretical score ~ pi/2)
3. Adam optimizer with LogSumExp-smoothed objective (FFT-based O(n log n) gradients)
4. Multi-length exploration with interpolation transfer
5. Peak-shaving coordinate descent for final polish
Key insight: the 1/sqrt(j+1) sequence has near-constant autocorrelation in the bulk
(integral representation), providing a much better starting point than uniform.
MWU replaces the expensive LP solve with O(n log n) FFT operations per iteration.
"""

import numpy as np
import time
import math
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(y, a):
    """Compute (T^T @ y) = cross-correlation of y (len 2n-1) with a (len n).
    Returns array of length n where result[j] = sum_k y[k] * a[k-j] for valid k-j."""
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = len(y) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Y = rfft(y, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(Y * A_rev, fft_size)
    return conv[n - 1:n - 1 + n]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_a(a):
    """Project onto the simplex {a >= 0, sum(a) = 1}."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def init_inv_sqrt(n):
    """Initialize with a_j proportional to 1/sqrt(j+1)."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(j)
    return a / a.sum()


def init_symmetric_inv_sqrt(n):
    """Symmetric 1/sqrt initialization."""
    j = np.arange(1, n + 1, dtype=np.float64)
    j_sym = np.minimum(j, n + 1 - j)
    a = 1.0 / np.sqrt(j_sym)
    return a / a.sum()


def init_cosine_taper(n, base='inv_sqrt'):
    """Apply cosine taper to a base sequence to reduce edge effects."""
    if base == 'inv_sqrt':
        j = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / np.sqrt(j)
    else:
        a = np.ones(n)
    # Cosine taper
    x = np.arange(n, dtype=np.float64)
    window = np.sin(np.pi * (x + 0.5) / (2.0 * n)) ** 2
    a = a * window
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return a / s


def init_gaussian(n, sigma_frac=0.4):
    """Gaussian-shaped initialization."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - center) / (sigma_frac * n)) ** 2)
    return a / a.sum()


def init_mixture(n, rng):
    """Random mixture of structure."""
    choice = rng.integers(0, 4)
    if choice == 0:
        a = init_inv_sqrt(n)
    elif choice == 1:
        a = init_symmetric_inv_sqrt(n)
    elif choice == 2:
        a = init_cosine_taper(n)
    else:
        a = init_gaussian(n, rng.uniform(0.2, 0.6))
    # Add small random perturbation
    a = a + rng.uniform(0, 0.01, size=n) * a
    return normalize_a(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def mwu_nesterov(a0, time_limit, eta_base=None, verbose=True):
    """
    MWU + Nesterov acceleration for the zero-sum game:
      min_a max_k (a*a)_k  (with sum(a)=1, a>=0)
    Both primal (a, length n) and dual (y, length 2n-1) use multiplicative updates.
    Nesterov extrapolation on the opponent's strategy gives O(1/T) rate.
    """
    n = len(a0)
    a = normalize_a(a0)
    y = np.ones(2 * n - 1) / (2 * n - 1)
    a_prev = a.copy()
    y_prev = y.copy()

    if eta_base is None:
        eta_base = math.sqrt(2.0 * math.log(4 * n)) / (4 * n)

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        # Adaptive learning rate (decreasing)
        eta = eta_base / (1.0 + 0.001 * t)

        # Nesterov extrapolation of opponent's strategy
        if t == 1:
            a_ext = a.copy()
            y_ext = y.copy()
        else:
            momentum = (t - 1.0) / (t + 1.0)
            a_ext = a + momentum * (a - a_prev)
            y_ext = y + momentum * (y - y_prev)
            a_ext = normalize_a(a_ext)
            y_ext = np.maximum(y_ext, 0)
            ys = y_ext.sum()
            if ys < 1e-300:
                y_ext = np.ones(2 * n - 1) / (2 * n - 1)
            else:
                y_ext = y_ext / ys

        # Compute payoffs using extrapolated opponent
        Ta = fast_autocorr(a_ext)  # dual payoff: (T @ a_ext)_k for each k
        Ty = fast_cross_corr(y_ext, a_ext)  # primal payoff: (T^T @ y_ext)_j for each j

        # MWU updates
        # Primal (minimizer): a_j *= exp(-eta * Ty_j)
        log_update_a = -eta * Ty
        log_update_a -= log_update_a.max()
        a_new = a * np.exp(log_update_a)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()

        # Dual (maximizer): y_k *= exp(+eta * Ta_k)
        log_update_y = eta * Ta
        log_update_y -= log_update_y.max()
        y_new = y * np.exp(log_update_y)
        y_new = np.maximum(y_new, 1e-300)
        y_new = y_new / y_new.sum()

        a_prev = a
        y_prev = y
        a = a_new
        y = y_new

        # Running average
        avg_a += a
        n_avg += 1

        # Check score periodically
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score = 2.0 * n * fast_autocorr(avg_check).max()
            if score < best_score:
                best_score = score
                best_a = avg_check.copy()

            # Also check current iterate
            score_cur = 2.0 * n * fast_autocorr(a).max()
            if score_cur < best_score:
                best_score = score_cur
                best_a = a.copy()

            if verbose and n_avg % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      MWU t={n_avg} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def lse_gradient(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a. Uses FFT."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    # grad[i] = 2 * sum_j w[i+j] * a[j] (cross-correlation)
    grad = 2.0 * fast_cross_corr(w, a)
    return grad, w


def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=0.3, verbose=True):
    """
    Adam optimizer on the LogSumExp-smoothed objective with temperature annealing.
    Works in the unconstrained space with simplex projection.
    """
    n = len(a0)
    a = normalize_a(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    # Adam parameters
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    t0 = time.monotonic()

    total_steps_estimate = 3000
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Cosine annealing of temperature
        frac = min(1.0, step / total_steps_estimate)
        T = T_start * (T_end / T_start) ** frac
        # Learning rate: start moderate, decay
        lr = 0.5 / (1.0 + 0.01 * step)

        # Compute gradient
        grad, w = lse_gradient(a, T)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize_a(a_new)

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam step={step} ({elapsed:.0f}s): T={T:.2f}, best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Coordinate-descent peak shaving: reduce top autocorrelation peaks
    by moving mass from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_a(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find target: position with low value, not in peak area
            peak_set = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_set.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_set]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = normalize_a(a_trial)
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = normalize_a(a_trial)
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 300:
                break

    return best_a, best_score

        # Mix with 1/sqrt for diversity
        init_long = 0.7 * init_long + 0.3 * init_inv_sqrt(n_long)
        init_long = normalize_a(init_long)
        a_out, _ = mwu_nesterov(init_long, t4, verbose=False)
        try_update(a_out, f"mwu_long_n{n_long}")

        # Adam on the longer one
        if time.monotonic() - t_global_start < time_budget - 20:
            t4b = min(remaining * 0.10, 120.0)
            if t4b > 10:
                a_out2, _ = adam_lse_optimize(a_out, t4b, T_start=20.0, T_end=0.2)
                try_update(a_out2, f"adam_long_n{n_long}")

    # Phase 5: Peak shaving
    print(f"\n=== Phase 5: Peak shaving ===")
    elapsed = time.monotonic() - t_global_start
    remaining = time_budget - elapsed
    if remaining > 15:
        t5 = min(remaining * 0.30, 200.0)
        if best_seq is not None:
            a_out, _ = peak_shaving(best_seq, t5, seed=777)
            try_update(a_out, "peak_shave_1")

    # Phase 6: Final Adam polish
    print(f"\n=== Phase 6: Final Adam polish ===")
    elapsed = time.monotonic() - t_global_start
    remaining = time_budget - elapsed
    if remaining > 15:
        t6 = min(remaining * 0.30, 150.0)
        if best_seq is not None:
            a_out, _ = adam_lse_optimize(best_seq, t6, T_start=5.0, T_end=0.1)
            try_update(a_out, "adam_final")

    # Phase 7: One more peak shave
    elapsed = time.monotonic() - t_global_start
    remaining = time_budget - elapsed
    if remaining > 10:
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 5 and best_seq is not None:
            a_out, _ = peak_shaving(best_seq, t7, seed=999)
            try_update(a_out, "peak_shave_2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback: 1/sqrt sequence
        fallback = init_inv_sqrt(1600)
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
```
