    """Build the constraint matrix for the active set S."""
    n = len(a)

    for idx, k in enumerate(S):
        for i in range(n):
            j = k - i

    return T

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses:
1. Cutting-plane LP Nesterov acceleration (main workhorse)
2. FFT-based LogSumExp gradient polish between LP phases

Key advantages over previous: faster LPs via cutting planes,
adaptive time allocation, gradient interlacing.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Helper functions (all top-level, no closures)

def fast_autocorr(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve(a, b):
    """Full convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def simplex_project(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. 2011
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_smooth_max(b, T):
    """Compute (1/T) * log(sum(exp(T*b))) and its gradient w.r.t. b."""
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    val = (m + np.log(e.sum())) / T
    w = e / e.sum()
    return val, w

def lse_grad_wrt_a(a, T):
    """
    Compute gradient of (1/T)*log(sum(exp(T*(a*a)_k))) w.r.t. a.
    Returns (smoothed_value, gradient).
    """
    n = len(a)
    b = fast_autocorr(a)
    val, w = lse_smooth_max(b, T)
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # This is 2 * cross_correlate(w, a) centered
    # conv(a_rev, w)[n-1+i]
    a_rev = a[::-1]
    conv = fast_convolve(a_rev, w)[:2 * n - 1]
    grad = 2.0 * conv[n - 1:2 * n - 1]  # This gives n elements
    return val, grad

def get_lp_solver():

def interpolate_sequence(f, new_n):
    """Interpolate a non-negative sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def build_conv_row(a, k):
    """Build the k-th row of the convolution matrix: (a * b)_k as a function of b."""
    n = len(a)
    row = np.zeros(n)
    for i in range(max(0, k - n + 1), min(n, k + 1)):
        row[i] = a[k - i]
    return row

def solve_cutting_lp(a_fixed, active_set, solver, max_constraints=500):
    """
    Solve LP: min M s.t. (a_fixed * b)_k <= M for k in active_set,
    sum(b) = 1, b >= 0.
    Returns (b_solution, M_value) or (None, None) on failure.
    """
    n = len(a_fixed)
    # Limit active set size
    if len(active_set) > max_constraints:
        # Keep the most important constraints (highest (a*b)_k values)
        # For now, just take a random subset
        active_list = list(active_set)[:max_constraints]
    else:
        active_list = sorted(active_set)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix
    constraints = [cp.sum(b_var) == 1]
    for k in active_list:
        row = build_conv_row(a_fixed, k)
        constraints.append(row @ b_var <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, timeout=30)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try alternative solver
        for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, timeout=30)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def cutting_plane_nesterov(a0, time_limit, solver, verbose=True):
    """
    Cutting-plane LP with Nesterov acceleration.
    The key innovation: only enforce the most active constraints in the LP,
    adding new constraints via a separation oracle.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()

    b_init = fast_autocorr(a)
    top_idx = np.argsort(b_init)[::-1][:80]
    active_set = set(int(x) for x in top_idx)
    
    best_a = a.copy()
    best_score = 2.0 * n * b_init.max()
    
    y = a.copy()
    a_prev = a.copy()
    
    # Running average
    avg_a = a.copy()
    n_avg = 1
    
    t_start = time.monotonic()
    iter_count = 0
    lp_fail_count = 0
    
    for t in range(1, 50000):
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit:
            break
        
        # Solve LP with current active set
        b_star, M_val = solve_cutting_lp(y, active_set, solver)
        if b_star is None:
            lp_fail_count += 1
            if lp_fail_count > 10:
                # Reset active set
                b_y = fast_autocorr(y)
                top_idx = np.argsort(b_y)[::-1][:100]
                active_set = set(int(x) for x in top_idx)
                lp_fail_count = 0
            continue
        
        lp_fail_count = 0
        
        # Separation oracle: find most violated constraint
        conv_yb = fast_convolve(y, b_star)
        k_viol = int(np.argmax(conv_yb))
        if conv_yb[k_viol] > M_val + 1e-8:
            active_set.add(k_viol)
            # Re-solve with the new constraint
            b_star, M_val = solve_cutting_lp(y, active_set, solver)
            if b_star is None:
                continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s
        
        y = y_new
        a_prev = a
        a = a_new
        iter_count += 1
        
        # Track best (check a and y)
        b_a = fast_autocorr(a)

def solve_lp_scipy(T, n):
    """
    Solve: min M s.t. T@b <= M, sum(b)=1, b>=0.
    T is (m, n) matrix.
    Returns (b, M) or (None, None).
    """
    m = T.shape[0]
    # Variables: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Inequality: T@b - M <= 0
    # In standard form: A_ub @ x <= b_ub
    A_ub = np.hstack([T, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Bounds: b_i >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    b_val = res.x[:n]
    M_val = res.x[n]
    return b_val, M_val

def fast_autocorr(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve(a, b):
    """Full convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)

def simplex_project(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_grad_wrt_a(a, T):
    """
    Gradient of (1/T)*log(sum(exp(T*(a*a)_k))) w.r.t. a.
    Returns (smoothed_value, gradient).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    val = (m + np.log(e.sum())) / T
    w = e / e.sum()
    # grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    a_rev = a[::-1]
    conv = fast_convolve(a_rev, w)[:2 * n - 1]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return val, grad


def build_constraint_matrix(a, k_list):
    """Build (len(k_list)) x n constraint matrix for LP."""
    n = len(a)
    m = len(k_list)
    T = np.zeros((m, n))
    for idx, k in enumerate(k_list):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            for i in range(lo, hi + 1):
                T[idx, i] = a[k - i]
    return T


def solve_lp_higs(T_mat, n, b_init=None):
    """
    Solve: min M s.t. T@b <= M, sum(b)=1, b>=0.
    T_mat is (m, n). Uses HiGHS via scipy.
    Returns (b, M) or (None, None).
    """
    m = T_mat.shape[0]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([T_mat, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0:
            b_val = np.maximum(res.x[:n], 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(res.x[n])
    except Exception:
        pass
    return None, None

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a_prev = a
        a = a_new
        iter_count += 1

        # Track best
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Running average
        n_avg += 1
        avg_a = avg_a + (a - avg_a) / n_avg
        if n_avg >= 50 and n_avg % 100 == 0:
            score_avg = 2.0 * n * fast_autocorr(avg_a).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_a.copy()

        # Periodically re-jig the active set (keep top 300 + some random)
        if iter_count % 300 == 0:
            conv_check = fast_autocorr(y)
            top_keep = np.argsort(conv_check)[::-1][:250]
            active_list = [int(x) for x in top_keep]

        # Occasionally add random constraints for diversity
        if iter_count % 200 == 0 and len(active_list) < 400:
            random_k = np.random.randint(0, 2 * n - 1, size=10)
            for rk in random_k:
                rk = int(rk)
                if rk not in active_list:
                    active_list.append(rk)

        if verbose and iter_count % 50 == 0:
            print(f"    CPN it={iter_count} ({elapsed:.0f}s): best={best_score:.6f}, |S|={len(active_list)}")

    return best_a, best_score


def gradient_polish(a0, time_limit, T_start=50.0, T_end=0.1, verbose=False):
    """Projected gradient with LogSumExp smoothing and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t_start = time.monotonic()
    eta = 1.0 / (0.5 * n)
    total_iters = 0

    for stage in range(30):
        if time.monotonic() - t_start > time_limit:
            break
        progress = stage / 30.0
        T_val = T_start * (T_end / T_start) ** progress

        stage_iters = 0
        while time.monotonic() - t_start < time_limit:
            val, grad = lse_grad_wrt_a(a, T_val)
            a_trial = simplex_project(a - eta * grad)
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
                eta = min(eta * 1.02, 5.0 / n)
            else:
                eta = eta * 0.75
            stage_iters += 1
            total_iters += 1
            if stage_iters >= 200:
                break

        if verbose:
            print(f"    GP stage={stage}, T={T_val:.2f}: best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t_start = time.monotonic()
    stall = 0

    while time.monotonic() - t_start < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = int(idx[np.argmax(contrib)])

            # Find low-impact targets
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            candidates = [j for j in range(n) if not peak_mask[j] and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]
            dst = int(dst_pool[int(rng.integers(len(dst_pool)))])

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = float(rng.uniform(0.003, 0.02))
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a non-negative sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 5))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.1
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.6 * np.exp(-4 * (x - 0.7) ** 2) + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization with cutting-plane LP Nesterov,
    gradient polish, peak shaving, and multi-length interpolation.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(1600, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 1: Quick optimization at smaller length for good shape
    if remaining > 400:
        n_small = max(300, n_base // 4)
        print(f"\n=== Phase 1: LP Nesterov at n={n_small} ===")
        t1 = min(remaining * 0.05, 80.0)
        init_small = interpolate_sequence(best_seq, n_small)
        a_out, _ = cutting_plane_nesterov(init_small, t1, initial_active_k=60, verbose=False)
        try_update(a_out, f"lp_n{n_small}")
        # Keep the shape for interpolation
        shape = a_out if a_out is not None else init_small

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 2: Main LP Nesterov at target length
    print(f"\n=== Phase 2: LP Nesterov at n={n_base} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.40, 600.0)
    if t2 > 30:
        init2 = interpolate_sequence(shape, n_base) if len(shape) != n_base else shape.copy()
        # Blend with existing best
        init2 = 0.5 * init2 + 0.5 * best_seq
        init2 = np.maximum(init2, 0)
        init2 = init2 / init2.sum()
        a_out, _ = cutting_plane_nesterov(init2, t2, initial_active_k=120, verbose=True)
        try_update(a_out, f"lp_main_n{n_base}")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 3: Gradient polish
    print(f"\n=== Phase 3: Gradient polish at n={n_base} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 10:
        a_out, _ = gradient_polish(best_seq, t3, T_start=30.0, T_end=0.05, verbose=False)
        try_update(a_out, f"grad_n{n_base}")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 4: Second LP Nesterov round (refinement)
    print(f"\n=== Phase 4: LP Nesterov refinement at n={n_base} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.30, 400.0)
    if t4 > 30:
        a_out, _ = cutting_plane_nesterov(best_seq, t4, initial_active_k=200, verbose=True)
        try_update(a_out, f"lp_refine_n{n_base}")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 5: Peak shaving
    print(f"\n=== Phase 5: Peak shaving (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 150.0)
    if t5 > 5:
        a_out, _ = peak_shaving(best_seq, t5, seed=777)
        try_update(a_out, "peak_shave_1")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 6: Try a longer length
    if remaining > 200:
        n_long = n_base + 400
        print(f"\n=== Phase 6: Explore n={n_long} (budget={remaining:.0f}s) ===")
        t6 = min(remaining * 0.30, 300.0)
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = cutting_plane_nesterov(init_long, t6, initial_active_k=150, verbose=False)
        try_update(a_out, f"lp_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and best_seq is not None and len(best_seq) == n_long:
            t6b = min(remaining * 0.2, 60.0)
            a_out, _ = peak_shaving(best_seq, t6b, seed=999)
            try_update(a_out, f"peak_n{n_long}")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 7: Final gradient polish + peak shave on best
    if remaining > 20:
        print(f"\n=== Phase 7: Final polish (budget={remaining:.0f}s) ===")
        t7 = min(remaining * 0.4, 100.0)
        if t7 > 5:
            a_out, _ = gradient_polish(best_seq, t7, T_start=5.0, T_end=0.01, verbose=False)
            try_update(a_out, "final_grad")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t8 = min(remaining * 0.5, 60.0)
            if t8 > 3:
                a_out, _ = peak_shaving(best_seq, t8, seed=1234)
                try_update(a_out, "final_peak")

    # Final verification
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time_left={remaining:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
