"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = (1 - x) ** 1.0  # linear ramp
    elif choice == 1:
        a = (1 - x) ** 2.0  # quadratic ramp
    elif choice == 2:
        a = np.exp(-3 * x)  # exponential decay
    elif choice == 3:
        a = np.exp(-8 * x ** 2)  # narrow Gaussian at left
    elif choice == 4:
        a = np.exp(-5 * (x - 0.3) ** 2)  # Gaussian at 0.3
    elif choice == 5:
        a = np.sin(np.pi * x * 0.7) ** 2  # partial sine
    elif choice == 6:
        a = (1 - x) ** 1.5 * (0.5 + 0.5 * np.sin(np.pi * x))
    elif choice == 7:

    elif choice == 8:

    a = np.maximum(a, 0)

def solve_lp_fast(a_fixed, active_indices=None, solver='highs'):
    """
    Solve LP: max sum(b) s.t. (conv(a_fixed, b))_k <= 1 for active k, b >= 0.
    Returns (b, scale) where the actual M = 1/scale... 
    Actually, let's reformulate: min M s.t. T @ b <= M, sum(b) = 1, b >= 0.
    """
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    if active_indices is not None and len(active_indices) < 2 * n - 1:
        T_active = T[active_indices, :]
    else:
        T_active = T
    
    m_a = T_active.shape[0]
    
    # min M
    # s.t. T_active @ b - M <= 0  (m_a constraints)
    #      sum(b) = 1
    #      b >= 0, M free
    
    # Variables: b[0..n-1], M (total n+1 variables)
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: T_active @ b - M <= 0
    # Row: [T_active, -1] @ [b; M] <= 0
    A_ub = np.zeros((m_a, n + 1))
    A_ub[:, :n] = T_active
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m_a)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Bounds: b >= 0, M free
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    b_val = res.x[:n]
    M_val = res.x[n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_val)

def active_set_game(a0, time_limit, n_active=20, verbose=True):
    """
    Active-set LP game iteration for min_a max_k (a*a)_k.
    Uses a working set of constraints for speed.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Initialize active set with top lags
    b = fast_autocorr(a)
    active = np.argsort(b)[::-1][:n_active]
    
    iter_count = 0
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP with active constraints
        b_star, M_val = solve_lp_fast(a, active)
        if b_star is None:
            # Fall back to full LP
            b_star, M_val = solve_lp_fast(a, None)
            if b_star is None:
                continue
        
        # Check if solution violates non-active constraints
        conv_check = np.convolve(a, b_star)
        violations = np.where(conv_check > M_val * (1 + 1e-6))[0]
        violations = violations[np.isin(violations, active, invert=True)]
        
        if len(violations) > 0:
            # Add violated constraints to active set
            active = np.unique(np.concatenate([active, violations[:n_active // 2]]))
            active = active[:2 * n_active]  # Cap the active set size
            continue
        
        # Update a using step size
        # Use a decreasing step size (like in online learning)
        alpha = 2.0 / (it + 2.0)
        a_new = (1 - alpha) * a + alpha * b_star
        a_new = normalize(a_new)
        
        # Check score
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        iter_count += 1
        
        # Update active set periodically
        if iter_count % 50 == 0:
            b = fast_autocorr(a)
            active = np.argsort(b)[::-1][:n_active]
            
            if verbose and iter_count % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    AS-Game it={iter_count} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
    
    return best_a, best_score

def lse_grad(a, T):
    """Compute gradient of LSE objective w.r.t. a (on probability simplex)."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b

    # (w * a)_k = sum_j w_j a_{k-j}, we want sum_k w_k a_{k-i}
    # = (w * a_rev)_i... let me think.
    # Actually: sum_k w_k * a_{k-i} = (w * a_full)_{i} where a_full is a embedded in a longer array?
    # No: let me use the FFT approach.
    # conv(w, a_rev)[i] = sum_j w_j * a_rev[i-j] = sum_j w_j * a[n-1-(i-j)] = sum_j w_j * a[n-1-i+j]
    # That's not quite what we want.
    # We want: g_i = 2 * sum_k w_k * a_{k-i} (with a_j = 0 for j out of [0,n-1])
    # = 2 * (w * a)_{?}... 
    # (w * a)_m = sum_j w_j a_{m-j}. We want sum_k w_k a_{k-i} = (w * a)_{... }
    # Let m = k, then sum_k w_k a_{k-i} = sum_k w_k a_{k-i}. If we let l = k-i, then k = l+i:
    # = sum_l w_{l+i} a_l = (a * w_shifted)_{i} where w_shifted... 
    # This is getting confusing. Let me just use the direct FFT cross-correlation.
    # 
    # We want g = 2 * cross_corr(w, a) where cross_corr(w, a)[i] = sum_k w_k a_{k-i}
    # = sum_k w_k a_{k-i}. This is the cross-correlation of w and a at lag i.
    # 
    # cross_corr(w, a) = ifft(conj(fft(w)) * fft(a))... no that's for circular correlation.
    # For linear cross-correlation: cross_corr(w, a)[i] = (w * a_rev)[n-1+i] where a_rev = a[::-1]
    # Actually: (w * a_rev)[k] = sum_j w_j a_rev[k-j] = sum_j w_j a[n-1-k+j]
    # We want sum_k w_k a[k-i]. Let's substitute: let j = k-i in the sum: sum_j w_{j+i} a[j]
    # = sum_j a[j] w[j+i]. 
    # (a * w)[j+i] = sum_l a[l] w[j+i-l]. If we set l=j: that's a[j]w[i]. Not quite.
    # 
    # Let me just compute it via FFT of the full arrays.
    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross-correlation: ifft(conj(W) * A) gives circular cross-correlation
    # For linear: we need to be careful about padding
    # Actually, with zero-padding to fft_size >= n + (2n-1) - 1 = 3n-2, we get linear
    conv_len = n + (2*n-1) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    conv = irfft(W * A_rev, fft_size)
    # conv[k] = sum_j w_j * a_rev[k-j] = sum_j w_j * a[n-1-k+j]
    # We want g_i = sum_k w_k * a_{k-i}
    # conv[n-1+i] = sum_j w_j * a[n-1-(n-1+i)+j] = sum_j w_j * a[j-i]

    # actually a[j-i] with j-i in [0,n-1] means j in [i, i+n-1], and w_j is defined for j in [0,2n-2])
    # So conv[n-1+i] = sum_{j=i}^{i+n-1} w_j a[j-i] for valid range
    # But we want sum_k w_k a[k-i] which is the same thing (k and j are dummy variables)
    # Actually wait: w has length 2n-1, a has length n.

    # So the sum is over k in [max(0,i), min(2n-2, i+n-1)].
    # conv[n-1+i] = sum_j w_j a_rev[n-1+i-j] = sum_j w_j a[n-1-(n-1+i-j)] = sum_j w_j a[j-i]

def logsumexp_polish(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=10):
    """LogSumExp-polished optimization using projected gradient on the simplex."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Projected gradient descent with backtracking
        step = 0.1
        for it in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad = lse_grad(a, T)
            
            # Projected gradient step on simplex
            a_new = a - step * grad
            # Project onto simplex (sum=1, a>=0)
            a_new = project_simplex(a_new)
            
            # Check if step is good (using LSE value as proxy)
            b_new = fast_autocorr(a_new)
            b_old = fast_autocorr(a)
            lse_new = T * (b_new.max() + np.log(np.sum(np.exp(T * (b_new - b_new.max())))))
            lse_old = T * (b_old.max() + np.log(np.sum(np.exp(T * (b_old - b_old.max())))))
            
            if lse_new > lse_old:
                step *= 0.5
                if step < 1e-10:
                    break
                continue
            else:
                a = a_new
                step = min(step * 1.1, 1.0)
            
            # Track best
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si} (T={T:.1f}) ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.max(ind[cond])
    theta = cssv[k - 1] / k
    return np.maximum(v - theta, 0)

def peak_shaving_direct(a0, time_limit, seed=42):
    """Direct peak shaving: reduce the maximum autocorrelation peak."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_k = b.argmax()
        
        # Find contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Move mass from highest contributor to a low-position
        src = idx[np.argmax(contrib)]
        
        # Find a good destination (low autocorrelation contribution everywhere)
        # Try a few random destinations
        improved = False
        for attempt in range(20):
            dst = rng.integers(n)
            if dst == src:
                continue
            for frac in [0.002, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = normalize(a_trial)
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            # Larger random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.max(ind[cond])
    theta = cssv[k - 1] / k
    return np.maximum(v - theta, 0)

def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

def mirror_descent(a0, time_limit, eta0=0.5, n_top=1):
    """
    Mirror descent (exponential weights) with max-subgradient.
    Fast: O(n log n) per step via FFT for autocorrelation.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average
    avg_a = np.zeros(n)
    n_avg = 0
    
    # Precompute FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        A = rfft(a, fft_size)
        b = irfft(A * A, fft_size)[:size]
        
        # Find top peaks
        if n_top == 1:
            k_star = b.argmax()
            # Subgradient: g_i = 2 * a_{k_star - i}
            g = np.zeros(n)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            # g[i] = 2 * a[k_star - i] for i in [js, je]
            # Let's vectorize: for i in [js, je], k_star - i ranges from k_star-je to k_star-js
            g[js:je+1] = 2.0 * a[k_star - je : k_star - js + 1][::-1]
        else:
            # Use top-n weighted subgradient
            top_idx = np.argsort(b)[::-1][:n_top]
            top_vals = b[top_idx]
            w = top_vals / top_vals.sum()
            g = np.zeros(n)
            for wi, ki in zip(w, top_idx):
                js = max(0, ki - n + 1)
                je = min(n - 1, ki)
                g[js:je+1] += 2.0 * wi * a[ki - je : ki - js + 1][::-1]
        
        # Adaptive step size
        eta = eta0 / (1.0 + 0.001 * t)
        
        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        # Check score (only every few steps to save time)
        if t % 50 == 0:
            score = 2.0 * n * b.max()  # b is from current a, not a_new

            b_new = irfft(A_new * A_new, fft_size)[:size]
            score = 2.0 * n * b_new.max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
        
        a = a_new
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            A_avg = rfft(avg_check, fft_size)
            b_avg = irfft(A_avg * A_avg, fft_size)[:size]
            score_avg = 2.0 * n * b_avg.max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
    
    return best_a, best_score

    b = fast_autocorr(a)
    active = np.argsort(b)[::-1][:n_active]
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build constraint matrix for active lags
        T_active = np.zeros((len(active), n))
        for row, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # (conv(a, b))_k = sum_j a_{k-j} * b_j for j in [js, je]
            # T_active[row, j] = a_{k-j} for valid j
            T_active[row, js:je+1] = a[k-je : k-js+1][::-1]
        
        # Solve LP: min M s.t. T_active @ b_var <= M, sum(b_var) = 1, b_var >= 0
        m_a = len(active)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        A_ub = np.zeros((m_a, n + 1))
        A_ub[:, :n] = T_active
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m_a)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(-np.inf, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
        except Exception:
            continue
        
        if res.status != 0:
            continue
        
        b_star = np.maximum(res.x[:n], 0)
        M_val = res.x[n]
        s = b_star.sum()
        if s < 1e-15:
            continue
        b_star = b_star / s
        
        # Check violations
        conv_check = np.convolve(a, b_star)
        violations = np.where(conv_check > M_val * (1 + 1e-4))[0]
        non_active_viol = violations[~np.isin(violations, active)]
        
        if len(non_active_viol) > 0:
            # Add violated constraints
            new_active = np.concatenate([active, non_active_viol[:n_active // 3]])
            active = np.unique(new_active)[:2 * n_active]
            continue
        
        # Update
        alpha = 2.0 / (it + 2.0)
        a_new = (1 - alpha) * a + alpha * b_star
        a_new = normalize(a_new)
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if it % 100 == 0:
            b = fast_autocorr(a)
            active = np.argsort(b)[::-1][:n_active]
            if it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP-Game it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def lse_gradCompute(a, T, fft_size):
    """Compute LSE objective value and gradient."""
    n = len(a)

    A = rfft(a, fft_size)
    b = irfft(A * A, fft_size)[:2*n-1]
    
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj = (max_Tb + np.log(exp_Tb.sum())) / T
    
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient: g_i = 2 * sum_k w_k * a_{k-i}
    # Use FFT cross-correlation
    conv_len = n + (2*n-1) - 1

    A_rev = rfft(a[::-1], fft_size)
    W = rfft(w, fft_size)
    conv = irfft(W * A_rev, fft_size)[:conv_len]
    
    grad = 2.0 * conv[n-1:n-1+n]
    return obj, grad, w

def lse_gradient_descent(a0, time_limit, T_start=100.0, T_end=1.0, n_stages=15):
    """LSE-smoothed optimization with adaptive step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = 1.0 / (T * n)  # Initial step size
        max_steps = 500
        
        for it in range(max_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            obj, grad, w = lse_gradCompute(a, T, fft_size)
            
            # Projected gradient step
            a_new = a - step * grad
            a_new = project_simplex(a_new)
            
            # Check improvement
            obj_new, _, _ = lse_gradCompute(a_new, T, fft_size)
            
            if obj_new > obj:
                step *= 0.5
                if step < 1e-12:
                    break
            else:
                a = a_new
                step = min(step * 1.2, 10.0 / (T * n))
            
            # Track best (use actual max, not LSE)
            if it % 20 == 0:
                A_curr = rfft(a, fft_size)
                b_curr = irfft(A_curr * A_curr, fft_size)[:size]
                score = 2.0 * n * b_curr.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score
