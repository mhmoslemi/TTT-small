"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LogSumExp-smoothed L-BFGS in log-domain with temperature annealing
2. Active-set gradient with backtracking line search for final refinement
3. Multi-length exploration (n = 1000 to 2500)
4. Perturbation-based multi-restart
5. Spectral initialization for diverse starting points

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_grad_a(a, w, n):
    """
    Compute grad_a[i] = sum_k w[k] * a[k-i] for i=0,...,n-1.
    This is the cross-correlation of w (length 2n-1) with a (length n).
    """
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    a_rev = a[::-1]
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    return conv[n - 1:2 * n - 1]


def lse_objective_and_grad(z, T, n):
    """
    LogSumExp-smoothed objective and gradient in z-space.
    a = softmax(z), objective = 2n * T * log(sum exp(b[k]/T))
    """
    a = softmax(z)
    b = fast_convolve(a)
    
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp
    lse = max_b + T * math.log(sum_exp)
    obj = 2.0 * n * lse
    
    # Gradient w.r.t. a
    c = compute_grad_a(a, w, n)
    grad_a = (4.0 * n / T) * c
    
    # Gradient w.r.t. z (softmax parameterization)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def active_set_objective_and_grad(a, n, top_k=1):
    """
    Exact max objective (using top-k active set) and gradient.
    """
    b = fast_convolve(a)
    max_b = b.max()
    obj = 2.0 * n * max_b
    
    # Active set: top-k indices
    if top_k >= len(b):
        active = np.arange(len(b))
    else:
        active = np.argpartition(b, -top_k)[-top_k:]
    
    # Equal weights on active set
    w = np.zeros(len(b))
    w[active] = 1.0 / len(active)
    
    c = compute_grad_a(a, w, n)
    grad_a = 4.0 * n * c
    
    return obj, grad_a, b


def run_lbfqsz_annealing(z_init, n, T_schedule, max_iter_per_stage=200, time_limit=None, t0=None):
    """
    Run L-BFGS with temperature annealing.
    z_init: initial z vector
    n: sequence length
    T_schedule: list of temperatures (high to low)
    """
    z = z_init.copy()
    best_z = z.copy()
    best_a = softmax(z)
    best_score = 2.0 * n * fast_convolve(best_a).max()
    
    if t0 is None:
        t0 = time.monotonic()
    
    for T in T_schedule:
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        def fun_and_grad(z_eval):
            return lse_objective_and_grad(z_eval, T, n)
        
        try:
            result = minimize(fun_and_grad, z, method='L-BFGS-B',
                             jac=True, maxiter=max_iter_per_stage,
                             options={'gtol': 1e-8, 'maxls': 50, 'disp': False})
            if result.fun < 2.0 * n * fast_convolve(softmax(z)).max() + 1e-10:
                z = result.x.copy()
        except Exception as e:
            print(f"    L-BFGS error at T={T}: {e}")
        
        a = softmax(z)
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
        
        if time_limit is not None:
            elapsed = time.monotonic() - t0
            if elapsed > time_limit * 0.9:
                break
    
    return best_a, best_score


def active_set_refine(a0, n, time_limit=60.0, max_iters=10000):
    """
    Refine using active-set gradient with backtracking line search.
    Works in the simplex (a >= 0, sum(a) = 1).
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        
        # Find active set (top peaks within 1% of max)
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        
        c = compute_grad_a(a, w, n)
        grad_a = 4.0 * n * c
        
        # Project gradient onto tangent space of simplex
        # Direction: -grad_a projected to sum=0, then ensure a + step*d >= 0
        d = -grad_a
        d = d - d.mean()  # project to sum=0
        
        # Find max step to maintain non-negativity
        neg_mask = d < 0
        if neg_mask.any():
            max_step = -a[neg_mask].min() / d[neg_mask].min()
        else:
            max_step = 1.0
        max_step = min(max_step * 0.99, 1.0)
        
        # Backtracking line search
        alpha = min(0.1, max_step)
        improved = False
        for _ in range(30):
            a_new = a + alpha * d
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            new_score = 2.0 * n * fast_convolve(a_new).max()
            if new_score < cur_score - 1e-12:
                a = a_new
                improved = True
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                break
            alpha *= 0.5
        
        if not improved:
            # Try smaller active set or different step
            break
    
    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:

    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * x) + 0.15 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:

    elif choice == 8:

    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    # Use multiple frequencies
    a_pert = a.copy()
    for _ in range(rng.integers(1, 4)):
        freq = rng.integers(1, 15)
        phase = rng.uniform(0, 2 * np.pi)
        a_pert = a_pert + strength * np.cos(freq * x + phase)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def score_of(a):
    """Compute the score of a sequence (normalized to sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

            if result.x is not None:

                else:

        except Exception:

    # Final check

    if sc_final < best_score:

    """Exact max objective and gradient (using top-k active set) in z-space."""
    a = softmax(z)
    b = fast_convolve(a)
    max_b = b.max()
    obj = 2.0 * n * max_b

    if top_k >= len(b):
        active = np.arange(len(b))
    else:
        active = np.argpartition(b, -top_k)[-top_k:]
    
    w = np.zeros(len(b))
    w[active] = 1.0 / len(active)
    
    c = compute_grad_a(a, w, n)
    grad_a = 4.0 * n * c
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_cross_correl(w, a, n):
    """
    Compute c[i] = sum_k w[k] * a[k-i] for i=0,...,n-1.
    w has length 2n-1, a has length n.
    Uses FFT: c = (w * a_rev)[n-1:2n-1] where * is linear convolution.
    """
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    a_rev = a[::-1]
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    return conv[n - 1:2 * n - 1]


def lse_fun_grad(z, T, n):
    """LogSumExp-smoothed objective and gradient in z-space."""
    a = softmax(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp
    lse = max_b + T * math.log(sum_exp)
    obj = 2.0 * n * lse
    c = compute_cross_correl(w, a, n)
    grad_a = (4.0 * n / T) * c
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def max_fun_grad(z, n, top_k=3):
    """Exact max objective with top-k subgradient in z-space."""
    a = softmax(z)
    b = fast_convolve(a)
    max_b = b.max()
    obj = 2.0 * n * max_b
    k = min(top_k, len(b))
    active = np.argpartition(b, -k)[-k:]
    # Weight by proximity to max
    gaps = max_b - b[active]
    weights = np.exp(-gaps / max(max_b * 0.01, 1e-10))
    weights = weights / weights.sum()
    w = np.zeros(len(b))
    w[active] = weights
    c = compute_cross_correl(w, a, n)
    grad_a = 4.0 * n * c
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def score_of(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    n = len(a)
    b = fast_convolve(a_norm)
    return float(2.0 * n * b.max())


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def run_lbfqsz_anneal(a_init, n, T_schedule, time_limit, max_iter_per_stage=200, gtol=1e-10):
    """
    Run L-BFGS with temperature annealing in z-space.
    Returns best (a, score) found.
    """
    t0 = time.monotonic()
    z = z_from_a(a_init).copy()
    best_a = a_init.copy()
    best_score = score_of(a_init)

    for stage_idx, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break

        def fun_grad_lse(z_eval):
            return lse_fun_grad(z_eval, T, n)

        try:
            result = minimize(fun_grad_lse, z, method='L-BFGS-B',
                              jac=True, maxiter=max_iter_per_stage,
                              options={'gtol': gtol, 'maxls': 30, 'disp': False})
            if result.x is not None:
                z_candidate = result.x.copy()
                a_candidate = softmax(z_candidate)
                sc = score_of(a_candidate)
                if sc < best_score:
                    best_score = sc
                    best_a = a_candidate.copy()
                    z = z_candidate
                else:
                    z = z_candidate  # Still use for next stage
        except Exception:
            pass

    # Also try with exact max at the final z
    try:
        def fun_grad_max(z_eval):
            return max_fun_grad(z_eval, n, top_k=3)

        result = minimize(fun_grad_max, z, method='L-BFGS-B',
                          jac=True, maxiter=200,
                          options={'gtol': 1e-11, 'maxls': 20, 'disp': False})
        if result.x is not None:
            a_final = softmax(result.x)
            sc = score_of(a_final)
            if sc < best_score:
                best_score = sc
                best_a = a_final.copy()
    except Exception:
        pass

    return best_a, best_score


def active_set_refine(a0, n, time_limit=60.0, max_iters=20000):
    """
    Active-set gradient with backtracking line search on the simplex.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Adaptive active set: peaks within 0.5% of max
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        if len(active) > 20:
            active = np.argpartition(b, -20)[-20:]

        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        c = compute_cross_correl(w, a, n)
        grad_a = 4.0 * n * c

        # Project to tangent space of simplex
        d = -grad_a
        d = d - d.mean()

        # Max step for non-negativity
        neg_mask = d < 0
        if neg_mask.any():
            max_step = (a[neg_mask] / (-d[neg_mask])).min() * 0.99
        else:
            max_step = 1.0
        max_step = min(max_step, 0.5)

        # Backtracking line search
        alpha = max_step
        improved = False
        for _ in range(40):
            if alpha < 1e-15:
                break
            a_new = a + alpha * d
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                alpha *= 0.5
                continue
            a_new = a_new / s_new
            new_score = 2.0 * n * fast_convolve(a_new).max()
            if new_score < cur_score - 1e-14:
                a = a_new
                improved = True
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    no_improve = 0
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * x) + 0.15 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = np.sqrt(np.maximum(1 - (2 * x - 1) ** 2, 0))
    elif choice == 8:
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 9:
        a = 1.0 / (0.1 + x * (1 - x))
    elif choice == 10:
        a = np.exp(-3 * (x - 0.5) ** 2)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_sequence(a, rng, strength=0.01):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    a_pert = a.copy()
    n_freqs = rng.integers(1, 4)
    for _ in range(n_freqs):
        freq = rng.integers(1, 20)
        phase = rng.uniform(0, 2 * np.pi)
        a_pert = a_pert + strength * np.cos(freq * x + phase)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s) ***")

            p_score = score_of(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:

        best_score = score_of(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # Temperature schedule (absolute values, relative to b.max() ~ 6e-4)
    # These give T/b.max() from ~0.05 down to ~0.0003
    T_schedule_main = [3e-5, 2e-5, 1.5e-5, 1e-5, 7e-6, 5e-6, 3.5e-6,
                       2.5e-6, 1.5e-6, 1e-6, 7e-7, 5e-7, 3e-7, 2e-7, 1e-7]

    # === Phase 1: LogSumExp L-BFGS Annealing ===
    print(f"\n=== Phase 1: LSE L-BFGS Annealing n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.30, 450.0)
    if t_phase1 > 30:

    # === Phase 2: Re-run with tighter schedule from improved point ===
    print(f"\n=== Phase 2: Fine LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.15, 200.0)
    if t_phase2 > 20:

        a_opt2, sc2 = run_lbfqsz_anneal(best_seq, n_best, T_fine,
