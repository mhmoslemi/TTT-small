import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Local evaluation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def compute_smoothed_gradient(a, T):
    """
    Compute gradient of (1/T) * log(sum exp(T * conv(a,a)))
    Returns (smoothed_value, gradient)
    """
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    max_b = b.max()
    # Stable softmax
    exponents = T * (b - max_b)
    exp_vals = np.exp(exponents)
    w = exp_vals / exp_vals.sum()  # weights, length 2n-1
    
    # smoothed value
    smooth_val = max_b + np.log(exp_vals.sum()) / T
    
    # Gradient: grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # = S1[i] + S2[i]
    # S1[i] = sum_m w[m+i] * a[m]  (cross-correlation)
    # S2[i] = sum_m w[i-m] * a[m]  (convolution)
    
    fft_size = 1 << (3 * n - 2).bit_length()  # enough for conv of length (2n-1) + n - 1 = 3n-2
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    
    # S1: cross-correlation of a with w
    # (a ⋆ w)[i] = sum_m a[m] * w[m+i]
    # FFT: IFFT(conj(FFT(a)) * FFT(w))
    s1_full = irfft(np.conj(A) * W, fft_size)[:n]
    
    # S2: convolution of w and a
    # (w * a)[i] = sum_m w[m] * a[i-m]
    # FFT: IFFT(FFT(w) * FFT(a))
    s2_full = irfft(W * A, fft_size)[:n]
    
    grad = s1_full + s2_full
    return smooth_val, grad

def mirror_descent_smoothed(a0, time_limit, T_start=50.0, T_end=0.05, n_T_steps=20, eta_scale=1.0):
    """
    Mirror descent on LogSumExp smoothed objective.
    a is kept on the probability simplex (a >= 0, sum(a) = 1).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    # T schedule
    T_values = np.geomspace(T_start, T_end, n_T_steps)
    
    for T in T_values:
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        # Adaptive step size: eta ~ 1/(T * ||grad||_inf)
        # Start with a reasonable eta
        eta = eta_scale / (T * n ** 0.5)
        eta = min(eta, 2.0)  # Cap to prevent divergence
        
        # Adam state for adaptive step sizes in log-space
        m = np.zeros(n)
        v = np.zeros(n)
        t_count = 0
        beta1, beta2 = 0.9, 0.999
        
        iters_in_T = 0
        while time.monotonic() - t0 < time_limit - 1:
            iters_in_T += 1
            smooth_val, grad = compute_smoothed_gradient(a, T)
            
            # Mirror descent step (exponential tilting with Adam)
            t_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t_count)
            v_hat = v / (1 - beta2 ** t_count)
            
            # Learning rate
            lr = eta / (np.sqrt(v_hat) + 1e-8)
            lr = np.clip(lr, 1e-6, 1.0)
            # Use mean lr for stability
            lr_mean = float(np.mean(lr))
            
            # Exponential tilting
            log_a = np.log(np.maximum(a, 1e-300))
            log_a -= lr_mean * (m_hat - m_hat.min())  # Shift by min for numerical stability
            log_a -= log_a.max()  # Normalize
            a_new = np.exp(log_a)
            s_new = a_new.sum()
            if s_new < 1e-300:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            
            # Check improvement
            score = evaluate_local(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new
            
            # Periodic check
            if iters_in_T % 100 == 0:
                elapsed = time.monotonic() - t0
                if elapsed > time_limit - 5:
                    break
        
        # Report
        elapsed = time.monotonic() - t0
        print(f"  MD T={T:.3f} ({elapsed:.0f}s): best_score={best_score:.8f}")
    
    return best_a, best_score

                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except:
                    continue
    except:
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val

def lp_alternating(a0, time_limit, solver, n_iters=10000):
    """LP alternating: fix a, solve for b, update a via Nesterov."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov momentum
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        b_new = lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = best_a.copy()
        else:
            y = y / sy
        
        # Weighted average for a
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Check score
        for candidate in [a, y]:
            sc = evaluate_local(candidate)
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
        
        if (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit):
    """Coordinate descent: try reducing each element and redistributing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_idx = int(np.argmax(b))
        
        # Find contributing positions
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_idx - idx]
        
        # Try reducing the top contributor
        top_pos = idx[np.argmax(contribs)]
        improved = False
        
        for factor in [0.999, 0.998, 0.995, 0.99, 0.98]:
            a_trial = a.copy()
            reduction = a_trial[top_pos] * (1 - factor)
            a_trial[top_pos] -= reduction
            if a_trial[top_pos] < 0:
                a_trial[top_pos] = 0
                reduction = a[top_pos]
            # Redistribute to other elements
            mask = np.ones(n, dtype=bool)
            mask[top_pos] = False
            others = a_trial[mask]
            others_sum = others.sum()
            if others_sum < 1e-15:
                break
            a_trial[mask] = others + reduction * others / others_sum
            a_trial = a_trial / a_trial.sum()
            
            sc = evaluate_local(a_trial)
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try a random perturbation
            if step % 100 == 0:
                rng = np.random.default_rng(step)
                noise = rng.standard_normal(n) * 0.005
                a_trial = a + noise
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    sc = evaluate_local(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
        
        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"  CD step={step} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def get_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except:
            continue
    return cp.ECOS

def interpolate_to(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new

def build_conv_matrix(a):
