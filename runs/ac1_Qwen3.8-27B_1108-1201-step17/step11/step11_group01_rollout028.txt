```python
def solve_lp_with_constraints(a_fixed, constraint_indices, solver):
    """Solve the LP with only the specified constraints."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build only the specified rows
    constraints = [cp.sum(b_var) == 1]
    for k in constraint_indices:
        # (a_fixed * b_var)[k] <= M_var
        # This is sum_i a_fixed[i] * b_var[k-i] for valid i
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        row = a_fixed[k - idx]  # coefficients for b_var[idx]
        constraints.append(row @ b_var[idx] <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    return b_val, float(M_var.value), constraint_indices
```

```python
from scipy.optimize import linprog

def solve_lp_linprog(a_fixed, active_indices, n):
    """Solve the best-response LP using scipy linprog."""
    # Variables: b_0, ..., b_{n-1}, M (total n+1 variables)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Inequalities: row_k(b) - M <= 0 for each active k
    A_ub = []
    b_ub = []
    for k in active_indices:
        row = np.zeros(n + 1)
        row[n] = -1.0  # -M
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        row[idx] = a_fixed[k - idx]
        A_ub.append(row)
        b_ub.append(0.0)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, None)] * n + [(None, None)]  # b_i >= 0, M unbounded
    # Actually M should be >= 0 since b >= 0 and a >= 0
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    b_val = res.x[:n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = res.x[n]
    return b_val, M_val
```

```python
2. LSE-smoothed L-BFGS with dual temperature schedules for polishing
3. Multi-length exploration with interpolation
4. Perturbation-based diversification to escape local minima
Key improvements over previous: constraint generation for faster LP iterations,
dual LSE schedules, better time management, adaptive perturbation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import math


def fast_convolve(a):
    """Compute linear autocorrelation a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(b) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Call evaluate_sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_lp_solver():
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_hiGHS(a_fixed, active_indices):
    """Solve best-response LP using scipy HiGHS solver with constraint generation."""
    n = len(a_fixed)
    # Variables: b_0..b_{n-1}, M -> total n+1
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])

    A_ub_rows = []
    for k in active_indices:
        row = np.zeros(n + 1)
        row[n] = -1.0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        row[idx] = a_fixed[k - idx]
        A_ub_rows.append(row)

    A_ub = np.array(A_ub_rows) if A_ub_rows else np.zeros((1, n + 1))
    b_ub = np.zeros(len(A_ub_rows)) if A_ub_rows else np.zeros(1)

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
    except Exception:
        return None, None

    if res.status != 0 or res.x is None:
        return None, None

    b_val = res.x[:n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = max(res.x[n], 0.0)
    return b_val, M_val


def nesterov_lp_constrgen(a0, time_limit, max_iters=500000, n_active=300):
    """Nesterov LP with constraint generation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    all_indices = np.arange(2 * n - 1)
    active_set = list(all_indices[:n_active])  # start with first n_active

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Update active set based on current autocorrelation
        b_curr = fast_convolve(y)
        top_k = np.argsort(b_curr)[::-1][:n_active]
        active_set = list(top_k)

        # Solve LP with these constraints
        b_new, M_lp = solve_lp_hiGHS(y, active_set)
        if b_new is None:
            # Fallback: use uniform
            b_new = np.ones(n) / n
            M_lp = b_curr.max()

        # Check if solution violates non-active constraints
        cross = fast_convolve(np.concatenate([y, b_new]))  # not quite right
        # Actually just check: (y * b_new)_k for non-active k
        ybn = np.convolve(y, b_new)
        max_violation = 0.0
        violated = None
        non_active = set(all_indices) - set(active_set)
        if len(non_active) > 0:
            for k in non_active:
                if ybn[k] > M_lp * 1.001 + 1e-10:
                    if ybn[k] > max_violation:
                        max_violation = ybn[k]
                        violated = k
            if violated is not None:
                active_set.append(violated)
                b_new, M_lp = solve_lp_hiGHS(y, active_set)
                if b_new is None:
                    b_new = np.ones(n) / n
                    M_lp = ybn.max()

        # Nesterov update
        t_k = (1.0 + math.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovCG iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.6f} n_active={len(active_set)}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LSE objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient w.r.t. a: 2n * sum_k w_k * a_{k-j}
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_lse(a0, time_limit, T_start=200.0, T_end=1.0, n_stages=20):
    """L-BFGS with LSE smoothing (decreasing T schedule)."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 40

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            if (si + 1) % 5 == 0:
                print(f"    LSE dec stage={si+1}/{n_stages} T={stage_T:.2f}: score={sc:.8f}")
        except Exception:
            pass

    return best_a, best_score

def subgradient_momentum(a0, time_limit, n_steps=300000, eta_start=0.5, eta_end=0.005, momentum=0.7):
    """Subgradient descent with momentum for final polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    v = np.zeros(n)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Use all "active" lags (within 95% of max)
        active = np.where(b >= max_b * 0.90)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        eta = eta_start * (eta_end / eta_start) ** (step / max(1, n_steps))
        v = momentum * v - eta * grad_z
        z_new = z + v
        a_new = softmax_from_z(z_new)

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        z = z_new
        a = a_new

        if (step + 1) % 50000 == 0:
            print(f"    SubGrad step={step+1}: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, shape_type=0):
    """Generate various structured initializations."""
    if shape_type == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif shape_type == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif shape_type == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif shape_type == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif shape_type == 4:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 2 * np.arange(n) / n)
    elif shape_type == 5:
        # U-shape: high at edges, low in middle
        x = np.linspace(0, 1, n)
        a = (2 * x - 1) ** 2 + 0.1
    elif shape_type == 6:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2) + 0.05
    elif shape_type == 7:
        # W-shape
        x = np.linspace(0, 1, n)
        a = np.cos(2 * np.pi * 1.5 * x) ** 2 + 0.1
    elif shape_type == 8:
        # Random
        a = rng.random(n) + 0.1
    else:
        x = np.linspace(0, 1, n)
        a = np.ones(n) * (1.0 + 0.15 * np.cos(2 * np.pi * 5 * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_sequence(a, rng, scale=0.1):
    """Add Gaussian perturbation to a sequence and re-normalize."""
    n = len(a)
    noise = rng.normal(0, scale, n)
    a_pert = a * (1.0 + noise)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns the best sequence found within budget."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    total_budget = float(budget_s) - 30.0

    print(f" Total budget: {total_budget:.0f}s, deadline: {global_deadline - time.monotonic():.0f}s remaining")

    # Get parent sequence or default init
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f" Parent sequence: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(1200, rng, 0)
        best_score = evaluate_safe(best_seq)
        print(f" Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: score={sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP with constraint generation (40% time) ===
    print(f"\n=== Phase 1: Nesterov LP (n={n_base}, score={best_score:.8f}) ===")
    t1 = 0.38 * total_budget
    a_out, _ = nesterov_lp_constrgen(best_seq, t1, n_active=400)
    try_update(a_out, "nesterov_cg")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: LSE L-BFGS decreasing T (20% time) ===
    print(f"\n=== Phase 2: LSE L-BFGS dec (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.20 * total_budget, remaining * 0.45)
    if t2 > 20:
        a_lse, _ = lbgfs_lse(best_seq, t2, T_start=200.0, T_end=1.0, n_stages=20)
        try_update(a_lse, "lse_dec")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Second Nesterov LP round (15% time) ===
    print(f"\n=== Phase 3: Nesterov LP round 2 (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.15 * total_budget, remaining * 0.40)
    if t3 > 15:
        a_out2, _ = nesterov_lp_constrgen(best_seq, t3, n_active=500)
        try_update(a_out2, "nesterov_cg2")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: LSE L-BFGS increasing T (10% time) ===
    print(f"\n=== Phase 4: LSE L-BFGS inc (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.10 * total_budget, remaining * 0.40)
    if t4 > 15:
        a_lse2, _ = lbgfs_lse_increasing(best_seq, t4, T_start=20.0, T_end=400.0, n_stages=15)
        try_update(a_lse2, "lse_inc")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 5: Multi-length exploration (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5_total = min(0.10 * total_budget, remaining * 0.45)
    if t5_total > 30:
        n_tries = [n_base + 100, n_base - 100, n_base + 200, n_base - 200]
        # Remove non-positive
        n_tries = [x for x in n_tries if x > 100]
        for n_try in n_tries:
            remaining = global_deadline - time.monotonic()
            if remaining < 20:
                break
            t5 = min(t5_total / max(1, len(n_tries)), remaining * 0.4)
            if t5 < 10:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_constrgen(init_try, t5, n_active=200, max_iters=100000)
            try_update(a_try, f"len_{n_try}")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Subgradient polish (5% time) ===
    print(f"\n=== Phase 6: Subgradient polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t6 = min(0.05 * total_budget, remaining * 0.5)
    if t6 > 10:
        a_sg, _ = subgradient_momentum(best_seq, t6, n_steps=200000, eta_start=0.3, eta_end=0.003, momentum=0.75)
        try_update(a_sg, "subgrad")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Perturbation and restart (total remaining) ===
    print(f"\n=== Phase 7: Perturbation & restart (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 30:
        n_perturb = 3
        per_perturb = remaining / (n_perturb + 1)
        for pi in range(n_perturb):
            remaining = global_deadline - time.monotonic()
            if remaining < 15:
                break
            scale = 0.05 * (pi + 1)
            a_pert = perturb_sequence(best_seq, rng, scale=scale)
            # Try a short LSE optimization from the perturbed start
            tp = min(per_perturb, remaining * 0.6)
            if tp > 5:
                a_res, _ = lbgfs_lse(a_pert, tp, T_start=100.0, T_end=1.0, n_stages=10)
                try_update(a_res, f"perturb_{pi}")
            # Also try short Nesterov
            remaining = global_deadline - time.monotonic()
            tp2 = min(per_perturb * 0.5, remaining * 0.4)
            if tp2 > 5:
                a_res2, _ = nesterov_lp_constrgen(a_pert, tp2, n_active=200, max_iters=50000)
                try_update(a_res2, f"perturb_nest_{pi}")
    print(f"  Phase 7 done: best_score={best_score:.8f}")

    # === Final: Quick LSE polish on the absolute best ===
    remaining = global_deadline - time.monotonic()
    if remaining > 10:
        print(f"\n=== Final polish (score={best_score:.8f}) ===")
        t_final = min(remaining * 0.7, 60.0)
        if t_final > 5:
            a_fin, _ = lbgfs_lse(best_seq, t_final, T_start=50.0, T_end=0.5, n_stages=8)
            try_update(a_fin, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
