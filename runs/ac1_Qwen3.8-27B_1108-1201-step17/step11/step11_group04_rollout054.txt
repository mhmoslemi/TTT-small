import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

_GLOBAL_DEADLINE = 0.0


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution conv(a,b) via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_convolution_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def solve_lp_best_response_hig(a_fixed):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Uses scipy linprog with HiGHS for speed.
    min M s.t. T @ b - M <= 0, sum(b) = 1, b >= 0
    Variables: [b_0, ..., b_{n-1}, M]
    """
    n = len(a_fixed)
    T = build_convolution_matrix(a_fixed)
    size = 2 * n - 1

    # Variables: [b_0,...,b_{n-1}, M]
    # Objective: minimize M -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0

    # Inequality: T @ b - M <= 0 -> [T, -1] @ x <= 0
    A_ub = np.zeros((size, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(size)

    # Equality: sum(b) = 1 -> [1,...,1, 0] @ x = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: b >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status != 0:
        return None, None

    b_val = res.x[:n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(max(0, res.x[n]))
    return b_val, M_val

        if step % 30000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiActiveSet {elapsed:.0f}s: score={best_score:.8f} scale={scale:.4f} n_active={len(active)}")

    return best_a, best_score


def spectral_flatten(a, n_iters=100, blend=0.35, deadline=None):
    """Spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        k_range = range(1, L // 2 + 1)
        for k in k_range:
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def logsumexp_obj_grad(z, n, T_temp):
    """LogSumExp objective and gradient w.r.t. log-domain z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    global _GLOBAL_DEADLINE
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit

    n = len(a0)
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline or time.monotonic() > _GLOBAL_DEADLINE:
            break
        remaining = min(deadline, _GLOBAL_DEADLINE) - time.monotonic()
        if remaining < 5:
            break

        def func(z_in):
            if time.monotonic() > min(deadline, _GLOBAL_DEADLINE):
                return 1e15
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac(z_in):
            if time.monotonic() > min(deadline, _GLOBAL_DEADLINE):
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        try:
            iters = max(10, min(500, int(remaining * 30)))
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-22,
                         'gtol': 1e-16, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        if (stage_i + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            obj_val, _ = logsumexp_obj_grad(z, n, stage_T)
            print(f"    L-BFGS stage {stage_i+1}/{len(T_stages)} ({elapsed:.0f}s): T={stage_T}, lse_obj={obj_val:.8f}")

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def peak_shave_refined(a0, time_limit, n_steps=200000):
    """
    Peak shaving with multi-scale active set and adaptive parameters.
    Uses both the direct peak and its "neighborhood" for a smoother update.
    """
    global _GLOBAL_DEADLINE
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.015
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit or time.monotonic() > _GLOBAL_DEADLINE:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Multi-scale: use different thresholds
        # Primary: very close to max
        threshold_primary = max_b * (1.0 - 1e-4)
        active_primary = np.where(b >= threshold_primary)[0]
        if len(active_primary) == 0:
            active_primary = np.array([np.argmax(b)])

        # Secondary: broader neighborhood
        threshold_secondary = max_b * (1.0 - 5e-3)
        active_secondary = np.where(b >= threshold_secondary)[0]

        # Combined gradient
        g = np.zeros(n)
        # Primary peaks (weight 1)
        for k in active_primary:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active_primary)

        # Secondary peaks (weight 0.3)
        if len(active_secondary) > len(active_primary):
            g_sec = np.zeros(n)
            for k in active_secondary:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g_sec[idx] += a[k - idx]
            g_sec /= len(active_secondary)
            g = 0.7 * g + 0.3 * g_sec

        eta_eff = eta * scale
        log_update = -eta_eff * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break

        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave+ {elapsed:.0f}s: score={best_score:.8f} scale={scale:.4f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    else:
        a = 1.0 - np.abs(2 * x - 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Key improvements: HiGHS LP, adaptive active-set mirror descent,
    continuation L-BFGS, multi-scale peak shaving.
    """
    global _GLOBAL_DEADLINE
    rng = np.random.default_rng(seed)
    _GLOBAL_DEADLINE = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ★ NEW BEST [{label}]: {sc:.8f} (n={len(a)}) at {elapsed:.1f}s")

    rem = _GLOBAL_DEADLINE - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 0: Spectral pre-conditioning (30s) ===
    print(f"\n=== Phase 0: Spectral Pre-conditioning ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    if rem > 40:
        t_spec = min(rem * 0.02, 30.0)
        a_spec = spectral_flatten(best_seq, n_iters=80, blend=0.3,
                                  deadline=time.monotonic() + t_spec)
        update_best(a_spec, "spectral_pre")

    # === Phase 1: Nesterov LP (400s) ===
    print(f"\n=== Phase 1: Nesterov LP (score={best_score:.8f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    t_lp = min(rem * 0.25, 400.0)
    if t_lp > 50:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_mw_refine(best_seq, t_lp, n_max_iters=1000000)
        update_best(a_lp, "nesterov_lp_1")

    # === Phase 2: Multiplicative Active Set (400s) ===
    print(f"\n=== Phase 2: Multi Active Set (score={best_score:.8f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    t_mas = min(rem * 0.25, 400.0)
    if t_mas > 30:
        print(f"  Budget: {t_mas:.0f}s")
        a_mas, _ = multiplicative_active_set(best_seq, t_mas, n_steps=500000)
        update_best(a_mas, "multi_active_set")

    # === Phase 3: Continuation LogSumExp L-BFGS (350s) ===
    print(f"\n=== Phase 3: Continuation L-BFGS (score={best_score:.8f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    t_lbfgs = min(rem * 0.22, 350.0)
    if t_lbfgs > 30:
        # Very gradual temperature schedule
        T_stages = [50.0, 25.0, 15.0, 10.0, 7.0, 5.0, 4.0, 3.0, 2.5, 2.0,
                    1.5, 1.2, 1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2,
                    0.15, 0.12, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.025, 0.02,
                    0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        print(f"  T_stages: {len(T_stages)}, budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages,
                            deadline=time.monotonic() + t_lbfgs)
        update_best(a_lb, "lbgfs_continuation")

    # === Phase 4: Peak Shaving (150s) ===
    print(f"\n=== Phase 4: Peak Shaving (score={best_score:.8f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    t_peak = min(rem * 0.12, 150.0)
    if t_peak > 20:
        a_pk, _ = peak_shave_refined(best_seq, t_peak, n_steps=300000)
        update_best(a_pk, "peak_shave_refined")

    # === Phase 5: Second Nesterov LP round (200s) ===
    print(f"\n=== Phase 5: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    t_lp2 = min(rem * 0.15, 200.0)
    if t_lp2 > 30:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_mw_refine(best_seq, t_lp2, n_max_iters=500000)
        update_best(a_lp2, "nesterov_lp_2")

    # === Phase 6: Fine L-BFGS polish (150s) ===
    print(f"\n=== Phase 6: Fine L-BFGS Polish (score={best_score:.8f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    t_fin = min(rem * 0.15, 150.0)
    if t_fin > 20:
        T_stages_fin = [2.0, 1.5, 1.0, 0.7, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15,

        a_fin = lbgfs_refine(best_seq, t_fin, T_stages_fin,
                             deadline=time.monotonic() + t_fin)
        update_best(a_fin, "final_lbgfs")

    # === Phase 7: Try different n (if time permits) ===
    print(f"\n=== Phase 7: Different n (score={best_score:.8f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    if rem > 80:
        for n_try in [1000, 1500, 2000]:
            rem = _GLOBAL_DEADLINE - time.monotonic()
            if rem < 40:
                break
            if len(best_seq) == n_try:
                continue
            t_sub = min(rem * 0.2, 80.0)
            if t_sub < 20:
                break
            print(f"  Trying n={n_try}, budget={t_sub:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_sp = spectral_flatten(init_try, n_iters=20, blend=0.3)
            # Quick Nesterov
            a_try, _ = nesterov_mw_refine(a_sp, t_sub * 0.6, n_max_iters=100000)
            update_best(a_try, f"nesterov_n{n_try}")
            # Quick L-BFGS if improved
            rem = _GLOBAL_DEADLINE - time.monotonic()
            if best_seq is not None and len(best_seq) == n_try and rem > 20:
                T_try = [5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
                a_try2 = lbgfs_refine(best_seq, min(rem * 0.5, 40.0), T_try,
                                      deadline=time.monotonic() + min(rem * 0.5, 40.0))
                update_best(a_try2, f"lbgfs_n{n_try}")

    # === Phase 8: Final peak shave + LP on best (if time) ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.4, 40.0)
        if t_ps > 10:
            a_ps, _ = peak_shave_refined(best_seq, t_ps, n_steps=100000)
            update_best(a_ps, "final_peak")

    rem = _GLOBAL_DEADLINE - time.monotonic()
    if rem > 15:
        t_flp = min(rem * 0.5, 30.0)
        if t_flp > 8:
            a_flp, _ = nesterov_mw_refine(best_seq, t_flp, n_max_iters=50000)
            update_best(a_flp, "final_lp")

    # Final output
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        # Verify
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
