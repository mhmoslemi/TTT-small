def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def subgradient_single_peak(y, k_star):
    """Compute subgradient for single peak lag. g[j] = y[k_star - j] for valid indices."""
    n = len(y)
    g = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    idx = np.arange(js, je + 1)
    g[idx] = y[k_star - idx]
    return g


def subgradient_multi_peak(y, peaks, weights):
    """Compute weighted multi-peak subgradient."""
    n = len(y)
    g = np.zeros(n)
    for k_star, w in zip(peaks, weights):
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] += w * y[k_star - idx]
    return g


def nesterov_mw(a0, time_limit, eta0=0.05, momentum=0.4, multi_peak=3, T_peak=0.05):

    Nesterov-accelerated multiplicative weights with multi-peak subgradient.
    Fast: O(n log n) per iteration via FFT for convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    a_prev = a.copy()
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

        max_b = b.max()

            active = np.where(b >= threshold)[0]
            if len(active) > multi_peak:

                active = active[top_idx]
            w_active = b[active] / max(b[active].sum(), 1e-300)

class LSEFunc:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        n = self.n
        T = self.T
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled = (b - max_b) / T
        scaled -= scaled.max()
        exp_s = np.exp(scaled)
        lse = max_b + T * (np.log(exp_s.sum() + 1e-300))
        return 2.0 * n * lse


class LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        n = self.n
        T = self.T
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled = (b - max_b) / T
        scaled -= scaled.max()
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()
        # Gradient: dLSE/da[j] = sum_k p[k] * a[k-j]
        grad_a = np.zeros(n)
        nonzero = p > 1e-18
        for k in np.where(nonzero)[0]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += p[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return 2.0 * n * a * (grad_a - S) / T

            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        maxiter = min(500, max(20, int(remaining * 2)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': maxiter * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
        a = np.exp(z - z.max())
        a = a / a.sum()
        elapsed = time.monotonic() - t_start
        score = 2.0 * n * fast_convolve(a).max()
        print(f"      LSE T={stage_T}: score={score:.8f} ({elapsed:.0f}s)")

    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        # Bell-shaped with wide base
        a = np.exp(-8 * (x - 0.5) ** 2) + 0.3
    else:
        # Multi-modal
        a = np.exp(-10 * (x - 0.25) ** 2) + np.exp(-10 * (x - 0.75) ** 2) + 0.2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_and_optimize(a0, rng, time_limit, strength=0.02):
    """Perturb sequence and re-optimize with Nesterov MW."""
    n = len(a0)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0, 2 * np.pi)
    freq2 = rng.integers(2, 15)
    phase2 = rng.uniform(0, 2 * np.pi)
    amp2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq * x + phase) + 0.5 * np.sin(freq2 * x + phase2))
    a_pert = np.maximum(a0 + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        a_pert = a0.copy()
    else:
        a_pert = a_pert / s
    a_opt, score = nesterov_mw(a_pert, time_limit, eta0=0.03, momentum=0.3)
    return a_opt, score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov-accelerated MW as main workhorse, with multi-length
    exploration and LSE L-BFGS polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')

    print(f"Budget: {budget_s:.0f}s, deadline in {deadline - time.monotonic():.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 2000
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick Nesterov MW on parent ===
    print(f"\n=== Phase 1: Quick NMW n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.08, 100.0)
    if t1 > 15:
        a_mw, _ = nesterov_mw(best_seq, t1, eta0=0.04, momentum=0.3, multi_peak=3)
        try_update(a_mw, "nmw_1")

    # === Phase 2: Multi-length Nesterov MW exploration ===
    print(f"\n=== Phase 2: Multi-length NMW (score={best_score:.8f}) ===")
    lengths_to_try = [len(best_seq), 2000, 3000]
    # Remove duplicates and sort by priority (closest to current first)
    lengths_to_try = sorted(set(lengths_to_try), key=lambda x: abs(x - len(best_seq)))

    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        if ln == len(best_seq):
            t_ln = min(rem * 0.3, 300.0)
            init_ln = best_seq.copy()
        else:
            t_ln = min(rem * 0.25, 250.0)
            init_ln = interpolate_sequence(best_seq, ln)

        if t_ln < 30:
            continue

        print(f"  n={ln}, budget={t_ln:.0f}s")
        # First quick MW to warm up
        t_warm = t_ln * 0.15
        a_warm, _ = nesterov_mw(init_ln, t_warm, eta0=0.05, momentum=0.4, multi_peak=3, T_peak=0.05)
        try_update(a_warm, f"nmw_warm_n{ln}")

        # Then longer Nesterov MW
        rem = deadline - time.monotonic()
        t_main = min(rem * 0.4, t_ln * 0.7)
        if t_main > 30:
            a_main, _ = nesterov_mw(a_warm, t_main, eta0=0.03, momentum=0.4, multi_peak=5, T_peak=0.03)
            try_update(a_main, f"nmw_main_n{ln}")

    # === Phase 3: Multiple perturbation escapes ===
    print(f"\n=== Phase 3: Perturbation escapes (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_trials = 4
    if rem > 100:
        t_per = min(rem * 0.12, 80.0)
        for trial in range(n_trials):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            print(f"  Trial {trial+1}/{n_trials}, budget={t_per:.0f}s")
            a_opt, _ = perturb_and_optimize(best_seq, rng, t_per, strength=0.01 + 0.01 * trial)
            try_update(a_opt, f"pert{trial}")

    # === Phase 4: Fresh structured starts at best length ===
    print(f"\n=== Phase 4: Fresh starts (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    n_fresh = min(rem * 0.06, 60.0)
    if n_fresh > 15:
        for fs in range(3):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            init_f = structured_init(n_best, rng)
            a_fs, _ = nesterov_mw(init_f, n_fresh, eta0=0.05, momentum=0.4)
            try_update(a_fs, f"fresh{fs}")

    # === Phase 5: LSE L-BFGS polishing ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.3, 400.0)
        print(f"  Budget: {t5:.0f}s, n={len(best_seq)}")
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        a_lse = lse_lbfgs_refine(best_seq, t5, T_stages)
        try_update(a_lse, "lse")

    # === Phase 6: Second LSE pass if time remains ===
    print(f"\n=== Phase 6: Second LSE pass (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t6 = min(rem * 0.2, 150.0)
        T_stages2 = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        a_lse2 = lse_lbfgs_refine(best_seq, t6, T_stages2)
        try_update(a_lse2, "lse2")

    # === Phase 7: Final MW polish ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.4, 60.0)
        a_fin, _ = nesterov_mw(best_seq, t7, eta0=0.01, momentum=0.2, multi_peak=3, T_peak=0.02)
        try_update(a_fin, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    # Ensure output is valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
