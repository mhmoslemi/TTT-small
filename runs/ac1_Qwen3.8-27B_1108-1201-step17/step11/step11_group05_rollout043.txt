def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver, b_warm=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s, warm_start=(b_warm is not None))
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0.0, M_var.value))

    s = a.sum()
    return a / s

def multiplicative_weights_polish(a0, time_limit):
    """Mirror descent / multiplicative weights on the active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0

    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.05
    no_improve = 0
    
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Gradient
        grad_a = np.zeros(n)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(k - i_end, k - i_start + 1)
            grad_a[i_start:i_end+1] += a[idx][::-1]
        grad_a /= len(active)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad_a)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:

            a = a_new

        else:
            no_improve += 1
        
        if no_improve >= 100:
            eta *= 0.9
            no_improve = 0
            if eta < 1e-5:
                break

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver, b_warm=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s, warm_start=(b_warm is not None))
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0.0, M_var.value))


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def make_ramp(n, power=1.0):
    """Monotone power ramp, increasing."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    s = a.sum()
    return a / s


def make_decrease_ramp(n, power=1.0):
    """Monotone power ramp, decreasing."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    a = a[::-1]
    s = a.sum()
    return a / s


def make_exp_ramp(n, rate=2.0):
    """Exponential ramp, increasing."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(rate * x / (n - 1))
    s = a.sum()
    return a / s


def make_wedge_asym(n, alpha=1.0, beta=0.5):
    """Asymmetric wedge: (i+1)^alpha * (n-i)^beta."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def compute_gradient_active(a, active):
    """Compute gradient of mean of b[k] for k in active, w.r.t. a."""
    n = len(a)
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    # grad_a[i] = sum_k w[k] * a[k-i] for valid k-i in [0, n-1]
    # = sum_j a[j] * w[i+j] for j in [0, n-1], i+j in [0, 2n-2]
    # Use FFT: this is cross-correlation
    # want[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # = (a reversed) circularly convolved with w, at appropriate index
    # Simpler: use the fact that this is a sliding dot product
    # Pad a to length 2n-1, compute correlation with w
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    # want[i] = sum_j a[j] * w[i+j] = cross-correlation of a_pad and w at lag -i
    # = (a_pad flipped) correlated with w... let me just use direct FFT
    # want[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # Let u[j] = a[j], v[k] = w[k]
    # want[i] = sum_j u[j] * v[i+j]
    # This is the (i-th) element of the cross-correlation (u ⊛ v) where
    # (u ⊛ v)[i] = sum_j u[j] v[i+j]
    # = IFFT(FFT(u_padded) * conj(FFT(v_padded))) ... no, that's not right either
    # 
    # Let's use: (u ⊛ v)[i] = sum_j u[j] v[i+j]
    # Flip u: u_flip[k] = u[-k] (circular)
    # Then (u ⊛ v)[i] = sum_j u_flip[-j] v[i+j] ... this is getting circular
    #
    # Simplest correct approach for n up to ~2000:
    # want[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # For each i, i+j ranges from i to i+n-1, and w has length 2n-1
    # So we need w[i : i+n] dot a

    # Use FFT-based approach:
    # Define A = a extended to length 2n-1 (zeros after n)
    # want[i] = sum_{j=0}^{n-1} A[j] * w[i+j] = (A * w_rev)[i + n - 1] where w_rev is reversed
    # Actually, let's just use the direct O(n * |active|) approach since |active| is small
    
    grad_a = np.zeros(n)
    for k in active:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # grad_a[i] += a[k-i] for i in [i_start, i_end]
        # a[k-i_start], a[k-i_start-1], ..., a[k-i_end]
        idx = np.arange(k - i_end, k - i_start + 1)
        grad_a[i_start:i_end + 1] += a[idx][::-1]
    grad_a /= len(active)
    return grad_a

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive active set
        if no_improve < 200:
            threshold = max_b * (1.0 - 1e-4)
        elif no_improve < 500:
            threshold = max_b * (1.0 - 5e-4)
        else:
            threshold = max_b * (1.0 - 2e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        # Limit active set size for speed
        if len(active) > 50:
            top_idx = b[active].argsort()[-50:]
            active = active[top_idx]

        grad_a = compute_gradient_active(a, active)

        # Log-space gradient
        S = np.dot(a, grad_a)
        grad_z = a * (grad_a - S)

        # Adam update
        lr = lr_init * (lr_decay ** step)
        if no_improve > 200:
            lr *= 0.5
        if no_improve > 500:
            lr *= 0.5

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()

        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradPolish step={step+1} ({elapsed:.0f}s): best={best_score:.8f} no_improve={no_improve}")

        if no_improve > 2000:
            break

    return best_a, best_score
