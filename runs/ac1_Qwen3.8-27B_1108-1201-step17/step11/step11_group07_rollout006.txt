This is a different approach! It solves an LP at each step to find the best direction to move. The LP is:
maximize sum(b_j)
subject to: b_j >= 0 for all j
             (f * b)_k <= M for all k

where f * b is the cross-correlation (or convolution) of f and b.

This is a linear program! We can solve it with scipy or cvxpy.

The key insight: at each iteration, we find the best "direction" g to move towards, subject to the constraint that the convolution peak doesn't exceed the current M. Then we take a step towards g.

This is essentially a projected gradient method where the projection is onto the set {g >= 0: ||f*g||_inf <= M}.

Let me implement this LP-based approach. The LP has:
- n variables (b_0, ..., b_{n-1})
- 2n-1 inequality constraints: (f*b)_k <= M for all k
- n non-negativity constraints

The (f*b)_k = sum_i f[i]*b[k-i] (with appropriate index ranges).

For n = 1200, this is an LP with 1200 variables and 2399 constraints. This is very manageable for modern LP solvers.

```python
from scipy.optimize import linprog

def solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)_k <= M for all k."""
    n = len(f)
    # Variables: b_0, ..., b_{n-1}
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (f*b)_k <= M for k = 0, ..., 2n-2
    # (f*b)_k = sum_{i: 0<=i<n, 0<=k-i<n} f[i] * b[k-i]
    
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n))
    b_ub = np.full(n_cons, M)
    
    for k in range(n_cons):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i  # b[j]
            A_ub[k, j] = f[i]
    
    # Bounds: b_j >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x
```

```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_direction_lp(f, M, max_feasible=None):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)_k <= M for all k.
    Returns the optimal b vector, or None if infeasible.
    """
    n = len(f)
    n_cons = 2 * n - 1
    
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    # (f*b)_k = sum_i f[i] * b[k-i] for valid i, k-i
    A_ub = np.zeros((n_cons, n))
    for k in range(n_cons):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[k, i] = f[k - i]
    
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    
    try:

        if res.status == 0:
            return res.x
        elif res.status == 2:  # infeasible
            return None
    except Exception:
        pass
    return None
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
combined with L-BFGS polishing and multi-length exploration.
The LP approach solves at each step: max sum(b) s.t. b>=0, (f*b)_k <= M,
which monotonically reduces the peak of the auto-convolution.
Combined with aggressive LSE-LBFGS polishing and peak-targeted refinement.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A_ub_fast(f):
    """Build constraint matrix for LP: A_ub[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j = np.arange(n)[None, :].astype(np.int64)
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    A_ub = np.zeros((2 * n - 1, n))
    A_ub[valid] = f[idx[valid]]
    return A_ub


def solve_direction_lp(f, M, relax=0.0):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)_k <= M*(1+relax) for all k."""
    n = len(f)
    M_eff = M * (1.0 + relax)
    c_obj = -np.ones(n)
    A_ub = build_A_ub_fast(f)
    b_ub = np.full(2 * n - 1, M_eff)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(f0, time_limit, verbose=True):
    """LP-based peak reduction optimization with adaptive step size."""
    n = len(f0)
    f = normalize(f0.copy())
    best_f = f.copy()
    best_score = evaluate_local(f)
    M = fast_convolve(f).max()
    t0 = time.monotonic()
    t = 0.8
    no_improve = 0
    iter_count = 0
    relax = 0.0

    if verbose:
        print(f"  LP start: n={n}, M={M:.10f}, score={best_score:.8f}")

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        g = solve_direction_lp(f, M, relax=relax)
        if g is None:
            relax = min(0.1, relax + 0.01)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        improved = False
        for alpha in [t, t * 0.5, t * 0.25, t * 0.1, t * 0.05]:
            f_new = (1.0 - alpha) * f + alpha * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-15:
                continue
            f_new = f_new / s
            M_new = fast_convolve(f_new).max()
            if M_new < M - 1e-14:
                f = f_new.copy()
                M = M_new
                sc = evaluate_local(f)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                    if verbose and iter_count % 200 == 0:
                        print(f"    LP iter {iter_count}: M={M:.10f}, score={sc:.8f}, t={t:.3f}")
                improved = True
                break

        if improved:
            no_improve = 0
            t = min(1.0, t * 1.03)
            relax = max(0.0, relax - 0.002)
        else:
            no_improve += 1
            t *= 0.7
            if no_improve > 30:
                relax = min(0.05, relax + 0.005)
                no_improve = 0
            if t < 1e-8:
                break

    if verbose:
        print(f"  LP done: {iter_count} iters, M={M:.10f}, best={best_score:.8f}, time={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def lse_obj_grad(a, T):
    """Compute LSE-smoothed objective and gradient. a must have sum=1."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    w = exp_c / exp_c.sum()
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    b = R[:n]
    g = 2.0 * b
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]
    return obj, g


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in log-parameterization."""
    a = softmax_from_z(z)
    obj, grad_a = lse_obj_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_optimize(a0, time_limit, T_stages, maxiter_per_stage=2000):
    """L-BFGS optimization with LogSumExp smoothing and temperature annealing."""
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        iters = max(10, int(remaining * 20))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-16, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def peak_targeted_reduce(a0, time_limit, n_peaks=5, verbose=True):
    """Target the top n_peaks peaks specifically for reduction."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.01

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()
        # Find top peaks
        peak_indices = np.argsort(c)[-n_peaks:]
        T = 0.05
        exp_c = np.exp((c - c_max) / T)
        w = exp_c / exp_c.sum()

        # Gradient with emphasis on peaks
        N = 1 << (len(w) + n - 1).bit_length()
        W = rfft(w, N)
        A = rfft(a, N)
        R = irfft(W * np.conj(A), N)
        b = R[:n]
        g = 2.0 * b
        idx = 2 * np.arange(n)
        valid = idx < (2 * n - 1)
        g[valid] -= a[valid] * w[idx[valid]]

        # Additional gradient from peak constraints
        for pk in peak_indices:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            for m in range(js, je + 1):
                g[m] += 0.1 * a[pk - m]

        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Line search
        improved = False
        for scale in [lr, lr * 0.5, lr * 0.25]:
            a_new = a - scale * g
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if improved:
            lr = min(0.05, lr * 1.01)
        else:
            lr *= 0.8
            if lr < 1e-8:
                break

    if verbose:
        print(f"  Peak-targeted done: best={best_score:.8f}, time={time.monotonic()-t0:.1f}s")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)

    a = np.exp(-8 * (x - 0.5) ** 2)
    candidates.append(normalize(a))

    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))

    a = np.sin(np.pi * x) ** 2
    candidates.append(normalize(a))

    a = (x * (1 - x)) ** 0.3
    candidates.append(normalize(a))

    a = np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)
    candidates.append(normalize(a))

    a = np.exp(-3 * x) * (1 - np.exp(-3 * (1 - x)))
    candidates.append(normalize(a))

    # Beta distribution shaped
    a = x ** 2 * (1 - x) ** 2
    candidates.append(normalize(a))

    # Smooth random
    coeffs = rng.normal(0, 1, 15)
    a = np.zeros(n)
    for i, c_val in enumerate(coeffs):
        a += c_val * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + a.max() + 0.1, 0)
    candidates.append(normalize(a))

    candidates.append(np.ones(n) / n)

    return candidates
```
