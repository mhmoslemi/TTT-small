"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_gradient(a, T):
    """Compute gradient of 2n*LSE(conv(a,a)/T) w.r.t. a using FFT."""
    n = len(a)
    b = fast_convolve(a)
    # Softmax weights
    b_shifted = b - b.max()
    scaled = b_shifted / T
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # c_i = sum_j w_{i+j} * a_j for i=0,...,n-1
    # This is conv(a, w_rev)[n-1:2n-1][::-1] where w_rev is w reversed
    w_rev = w[::-1]
    fft_len = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_len)
    W = rfft(w_rev, fft_len)
    conv_full = irfft(A * W, fft_len)
    
    # c_i = conv_full[2n-2-i] for i=0,...,n-1
    c = conv_full[n-1:2*n-1][::-1]
    
    # Gradient: (4n/T) * c
    grad = (4.0 * n / T) * c
    return grad, w, b


def lse_objective(a, T):
    """Compute the LSE-smoothed objective value."""
    n = len(a)
    b = fast_convolve(a)
    b_shifted = b - b.max()
    scaled = b_shifted / T
    lse = b.max() + T * np.log(np.exp(scaled).sum())
    return 2.0 * n * lse


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, non-negative)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    positive = u > cssv / np.arange(1, n + 1)
    rho = np.sum(positive)
    theta = cssv[rho - 1] / rho
    proj = np.maximum(v - theta, 0)
    s = proj.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return proj / s


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    from scipy.sparse import csr_matrix
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():

    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

    return cp.SCS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver, cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if s is None:
            continue
        try:

            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def multiplicative_weights(a0, time_limit, n_steps=2000000, eta0=0.03):
    """Multiplicative weights with subgradient averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Use averaging for better convergence
    a_avg = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        
        # Use top-k active constraints for better subgradient
        k_max = min(20, len(b))
        top_k = np.argsort(b)[::-1][:k_max]
        w_active = b[top_k]
        w_active = w_active / w_active.sum()
        
        g = np.zeros(n)
        for w, k in zip(w_active, top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Averaging
        a_avg = (step * a_avg + a_new) / (step + 1)
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        score_avg = 2.0 * n * fast_convolve(a_avg).max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 0.15)
        elif score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()
            a = a_avg
            eta = min(eta * 1.005, 0.15)
        else:
            eta *= 0.95
            if eta < 1e-10:
                break
        
        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score


def lp_alternating(a0, time_limit, solver, n_max_iters=500000):
    """LP alternating with Nesterov acceleration and sparse matrices."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    lp_times = []
    T_mat = build_conv_matrix_sparse(y)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_times.append(time.monotonic() - lp_t0)
        if b_new is None:
            continue
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 10.0:
                print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping at iter {it+1}")
                break
        
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        T_mat = build_conv_matrix_sparse(y)
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.01 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def lse_adam_polish(a0, time_limit, T_stages=None, max_iter_per_stage=5000):
    """LSE-smoothed optimization using Adam in log-space with FFT gradients."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    if T_stages is None:
        T_stages = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr_base = 0.01
    
    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        # Scale lr with temperature (smaller T needs smaller steps)
        lr = lr_base * min(1.0, T * 1000)
        
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        stage_best = softmax_from_z(z)
        stage_best_score = 2.0 * n * fast_convolve(stage_best).max()
        
        for it in range(max_iter_per_stage):
            if time.monotonic() >= deadline:
                break
            if time.monotonic() - t0 > time_limit:
                break
            
            a = softmax_from_z(z)
            
            # Compute gradient in z-space
            grad_a, w, b = compute_lse_gradient(a, T)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Adam update
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Check improvement every 50 steps
            if (it + 1) % 50 == 0:
                a_check = softmax_from_z(z)
                score = 2.0 * n * fast_convolve(a_check).max()
                if score < stage_best_score:
                    stage_best_score = score
                    stage_best = a_check.copy()
        
        # Update z from stage best
        if stage_best_score < 2.0 * n * fast_convolve(softmax_from_z(z)).max():
            z = z_from_a(stage_best)
        
        elapsed = time.monotonic() - t0
        print(f"    LSE-Adam stage {stage_idx+1}/{len(T_stages)} T={T:.6f} ({elapsed:.0f}s): score={stage_best_score:.8f}")
    
    return softmax_from_z(z)


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=300):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    if T_stages is None:
        T_stages = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for stage_T in T_stages:
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        def func(z_x):
            if time.monotonic() > deadline:
                return 1e15
            a_tmp = softmax_from_z(z_x)
            return lse_objective(a_tmp, stage_T)
        
        def jac(z_x):
            if time.monotonic() > deadline:
                return np.zeros(len(z_x))
            a_tmp = softmax_from_z(z_x)
            grad_a, _, _ = compute_lse_gradient(a_tmp, stage_T)
            S = np.dot(grad_a, a_tmp)
            return a_tmp * (grad_a - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        # Fourier-based: sum of a few cosines
        a = np.ones(n)
        for f in [2, 3, 5]:
            a += 0.1 / f * np.cos(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
    else:
        # Gaussian mixture
        centers = rng.uniform(0.2, 0.8, size=3)
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-50 * (x - c) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def lse_adam_coordinate(a0, time_limit, T_start=0.001, T_end=0.00001, n_stages=15):
    """Alternative LSE polishing with coordinate-wise updates and momentum."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    T_schedule = np.geomspace(T_start, T_end, n_stages)
    momentum = np.zeros(n)
    beta_m = 0.9
    lr = 0.005
    
    for stage_idx, T in enumerate(T_schedule):
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        # Scale lr inversely with n for stability
        lr_stage = lr / max(1, n // 500)
        
        iters = max(100, int(remaining * 30 / (len(T_schedule) - stage_idx)))
        
        for it in range(iters):
            if time.monotonic() >= deadline:
                break
            
            a = softmax_from_z(z)
            grad_a, w, b = compute_lse_gradient(a, T)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Momentum update
            momentum = beta_m * momentum + (1 - beta_m) * grad_z
            z = z - lr_stage * momentum
            
            # Periodic check
            if (it + 1) % 200 == 0:
                pass
        
        if stage_idx % 3 == 0:
            elapsed = time.monotonic() - t0
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            print(f"    LSE-Coord stage {stage_idx+1}/{n_stages} T={T:.2e} ({elapsed:.0f}s): score={sc:.8f}")
    
    return softmax_from_z(z)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses MW, LP alternating, LSE-Adam polishing, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")
    
    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            old_best = best_score
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)}, was {old_best:.8f})")
    
    # === Phase 1: Quick MW to warm up ===
    print(f"\n=== Phase 1: MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights(best_seq, t1, n_steps=500000, eta0=0.03)

    # === Phase 2: LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = lp_alternating(best_seq, t2, solver, n_max_iters=500000)
        try_update(a_lp, "lp_main")
    
    # === Phase 3: LSE-Adam fine polish ===
    print(f"\n=== Phase 3: LSE-Adam n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t3 = min(rem * 0.15, 250.0)
        T_stages = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        print(f"  {len(T_stages)} stages, budget={t3:.0f}s")
        a_lse = lse_adam_polish(best_seq, t3, T_stages=T_stages, max_iter_per_stage=3000)
        try_update(a_lse, "lse_adam")

    print(f"\n=== Phase 4: Palindromic LP (score={best_score:.8f}) ===")

                a_pol = lse_adam_polish(a_opt, min(rem * 0.3, 30.0),
                                         T_stages=[0.01, 0.005, 0.001, 0.0005],

                try_update(a_pol, f"pert{trial}_lse")
    
    # === Phase 8: Final fine LSE-Adam polish ===
    print(f"\n=== Phase 8: Final LSE-Adam (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t8 = min(rem * 0.25, 120.0)
        T_final = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]

        a_fin_lp, _ = lp_alternating(best_seq, t9, solver, n_max_iters=200000)
        try_update(a_fin_lp, "final_lp")

        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")
    
    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
