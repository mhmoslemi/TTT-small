Let me also think about the "convolution equalization" step more carefully. The idea is:
- Given current a, compute conv = convolution(a, a)

`d/d a[j] sum_k (conv[k] - target)^2 = 2 * sum_k (conv[k] - target) * d conv[k] / da[j]`

This is a smooth quadratic objective, so we can use efficient gradient methods. But note: minimizing the variance of conv is not the same as minimizing the max of conv. However, it's a good proxy, especially when combined with LSE for the actual max.

**Key insight about the optimal structure:**

The convolution `c(k) = sum_i a[i]*a[k-i]` represents the "overlap energy" at each shift. For the max to be minimized, we want all overlap energies to be equal. This is like a "Fourier transform of a top hat" problem.

For a sequence with DFT `A(k)`, the DFT of the (zero-padded) convolution is `A(k)^2`. For the convolution to be constant, we'd need `A(k)^2` to be a delta, i.e., `A(k) = 0` for k≠0. But that means a is uniform, which gives a triangular convolution. The contradiction arises because the "constant convolution" would need `A(k)^2 = 0` for k≠0, but the uniform sequence has `A(k) ≠ 0` for k≠0 (it's a sinc function).

So the problem is fundamentally about making `|A(k)|^2` (which is the power spectrum of the convolution) as concentrated as possible at k=0, while keeping a non-negative.

For a non-negative sequence, the power spectrum `|A(k)|^2` has the constraint that it's the power spectrum of a non-negative signal, which means it must satisfy the "positive real part" condition (the signal must be a valid power spectrum). Actually, that's for the autocorrelation, not the general case.

For our problem, the constraint is simply `a >= 0`. In the frequency domain, this means the inverse DFT of A must be non-negative. This is a set of n linear constraints on the 2n-1 real parameters of A (for real a).

The optimization problem becomes: minimize `max_k c(k)` (or equivalently, minimize the max time-domain value) subject to `IDFT(A) >= 0` and `A(0) = sum(a) = 1` (normalized).

This is a non-convex problem (the non-negativity constraint on the IDFT is linear, but the objective involves the inverse DFT of A^2, which is non-convex in A).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE gradient with aggressive multi-stage temperature annealing
2. Convolution variance minimization as a smooth smoothing pre-pass
3. Multi-length exploration (1200-3000) with interpolated initialization
4. Spectral domain shaping to flatten the power spectrum
5. Population-based diversification with structured mutations
6. Targeted peak-shaving via coordinate updates on active set

import numpy as np
import time
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_obj_and_grad(a, T):
    """
    LSE-smoothed objective and gradient w.r.t. a (a assumed normalized, sum=1).
    obj = 2n * LSE(conv(a,a), T)
    grad[j] = 4n * sum_m a[m] * w[j+m]  (cross-correlation form)
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse_val = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse_val

    w = exp_c / exp_c.sum()

    # cross[j] = sum_m a[m] * w[j+m]
    # = conv(reversed_a, w)[j + n - 1]
    conv_len = 3 * n - 1
    fft_size = next_fast_len(conv_len)
    rev_a = a[::-1]
    B = rfft(rev_a, fft_size)
    W = rfft(w, fft_size)
    bw = irfft(B * W, fft_size)
    cross = bw[n - 1:2 * n - 1]

    grad = 4.0 * n * cross
    return obj, grad


def lse_obj_and_grad_z(z, T):
    """LSE objective and gradient in log-parameterization."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    obj, grad_a = lse_obj_and_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def variance_obj_and_grad(a):
    """
    Objective: sum_k (c(k) - mean(c))^2  (variance of convolution)
    Gradient: 4 * sum_m a[m] * (c(j+m) - mean(c))
    """
    n = len(a)
    c = fast_convolve(a)
    c_mean = c.mean()
    dev = c - c_mean
    obj = np.dot(dev, dev)

    # grad[j] = 4 * sum_m a[m] * (c[j+m] - c_mean)
    # = 4 * cross where now "w" is replaced by "dev"
    conv_len = 3 * n - 1
    fft_size = next_fast_len(conv_len)
    rev_a = a[::-1]
    B = rfft(rev_a, fft_size)
    D = rfft(dev, fft_size)
    bw = irfft(B * D, fft_size)
    cross = bw[n - 1:2 * n - 1]

    grad = 4.0 * cross
    return obj, grad


def variance_minimize(a0, time_budget, n_iters=5000, lr=0.01):
    """Minimize convolution variance via gradient descent in z-space."""
    n = len(a0)
    z = z_from_a(a0).copy()
    best_a = a0.copy()
    best_score = evaluate_local(a0)
    t0 = time.monotonic()
    lr = lr

    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break

        a = softmax_from_z(z)
        _, grad_var = variance_obj_and_grad(a)
        S = np.dot(grad_var, a)
        grad_z = a * (grad_var - S)

        # Line search
        for scale in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            z_new = z - scale * grad_z
            a_new = softmax_from_z(z_new)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = z_new.copy()
            break

        if it % 500 == 0:
            lr *= 0.7

    return best_a, best_score


def lbgfs_optimize(a0, time_budget, T_stages, maxiter_per_stage=5000):
    """L-BFGS optimization with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    for stage_T in T_stages:
        elapsed = time.monotonic() - t0
        if elapsed > time_budget:
            break
        remaining = time_budget - elapsed
        if remaining < 5:
            break

        def func(z_in):
            obj, _ = lse_obj_and_grad_z(z_in, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_obj_and_grad_z(z_in, stage_T)
            return g

        iters = min(maxiter_per_stage, max(10, int(remaining * 50)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def spectral_shape_optimize(a0, time_budget, n_iters=5000):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = next_fast_len(2 * n - 1)
    t0 = time.monotonic()
    lr = 0.003

    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        r = (P - P_mean) / (P_mean + 1e-15)
        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-12
        H[valid] = -r[valid] * P_mean * A[valid] / (P[valid] + 1e-10)
        H[0] = 0.0
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        for scale in [lr, lr * 0.3, lr * 0.1]:
            a_new = a + scale * h
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            break

        if it % 500 == 0:
            lr *= 0.8

    return best_a, best_score


def peak_shave(a0, time_budget, n_rounds=200):
    """Directly target the peak positions to reduce them."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a0)
    t0 = time.monotonic()
    z = z_from_a(a).copy()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_budget:
            break

        c = fast_convolve(a)
        c_max = c.max()
        # Active set: peaks within 1% of max
        threshold = c_max * 0.99
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            active = np.array([n - 1])

        # Compute subgradient from active peaks
        # grad[j] = sum_{k in active} 2*a[k-j] (simplified, equal weights)
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)

        conv_len = 3 * n - 1
        fft_size = next_fast_len(conv_len)
        rev_a = a[::-1]
        B = rfft(rev_a, fft_size)
        W = rfft(w, fft_size)
        bw = irfft(B * W, fft_size)
        cross = bw[n - 1:2 * n - 1]

        grad_a = 2.0 * n * cross  # simplified gradient
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        # Gradient descent in z-space
        for scale in [0.05, 0.02, 0.01, 0.005]:
            z_new = z - scale * grad_z
            a_new = softmax_from_z(z_new)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                z = z_new.copy()
            break

    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)

    # Uniform
    candidates.append(np.ones(n) / n)

    # Gaussian bump
    a = np.exp(-8 * (x - 0.5) ** 2)
    candidates.append(normalize(a))

    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))

    # Sine squared
    a = np.sin(np.pi * x) ** 2
    candidates.append(normalize(a))

    # Power law (edges)
    a = (x * (1 - x)) ** 0.3
    candidates.append(normalize(a))

    # Two bumps
    a = np.exp(-12 * (x - 0.3) ** 2) + np.exp(-12 * (x - 0.7) ** 2)
    candidates.append(normalize(a))

    # U-shaped (edge-weighted)
    a = 1.0 + 2.0 * (2 * x - 1) ** 2
    candidates.append(normalize(a))

    # Harmonic: 1 + alpha*cos
    for alpha in [0.3, 0.6, 0.9]:
        a = 1.0 + alpha * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0)
        candidates.append(normalize(a))

    # Harmonic: 1 + alpha*cos(2 cycles)
    for alpha in [0.3, 0.6]:
        a = 1.0 + alpha * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
        candidates.append(normalize(a))

    # Random smooth
    coeffs = rng.normal(0, 1, 15)
    a = np.zeros(n)
    for i, c_i in enumerate(coeffs):
        a += c_i * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 1.0, 0)
    candidates.append(normalize(a))

    # Truncated exponential (one-sided)
    a = np.exp(-3 * x)
    candidates.append(normalize(a))

    # Symmetric edge peak
    a = np.exp(-5 * x) + np.exp(-5 * (1 - x))
    candidates.append(normalize(a))

    # Periodic with random phase
    for f in [2, 3, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * x + phase)
        a = np.maximum(a, 0)
        candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturb_sequence(a, rng, method='gaussian', strength=0.02):
    """Perturb a sequence using various methods."""
    n = len(a)
    if method == 'gaussian':
        noise = rng.normal(0, strength, n)
        a_new = a + noise
    elif method == 'low_freq':
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        noise = strength * np.cos(freq * x + phase)
        a_new = a + noise
    elif method == 'edge_boost':
        a_new = a.copy()
        edge_width = max(5, n // 20)
        w = rng.uniform(strength * 0.5, strength)
        a_new[:edge_width] += w
        a_new[-edge_width:] += w
        a_new[edge_width:-edge_width] -= 2 * w * edge_width / (n - 2 * edge_width)
    elif method == 'block_swap':
        a_new = a.copy()
        block_size = rng.integers(max(5, n // 50), max(10, n // 20))
        i = rng.integers(0, max(1, n - block_size))
        j = rng.integers(0, max(1, n - block_size))
        if abs(i - j) > block_size:
            a_new[i:i + block_size], a_new[j:j + block_size] = \
                a_new[j:j + block_size].copy(), a_new[i:i + block_size].copy()
    else:
        a_new = a + rng.normal(0, strength, n)

    return normalize(a_new)


def population_search(a_init, time_budget, pop_size=8, rng=None):
    """Population-based search with adaptive mutations."""
    if rng is None:
        rng = np.random.default_rng(789)

    n = len(a_init)
    methods = ['gaussian', 'low_freq', 'edge_boost', 'block_swap']
    population = [normalize(a_init.copy())]
    for i in range(pop_size - 1):
        m = methods[i % len(methods)]
        strength = 0.015 + 0.01 * (i % 3)
        population.append(perturb_sequence(a_init, rng, m, strength))

    scores = [evaluate_local(p) for p in population]
    best_idx = int(np.argmin(scores))
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]

    t0 = time.monotonic()
    gen = 0
    adaptive_strength = 0.015

    while time.monotonic() - t0 < time_budget:
        gen += 1
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break

        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        if scores[0] < best_score:
            best_score = scores[0]
            best_a = population[0].copy()

        keep = max(2, pop_size // 2)
        elites = population[:keep]
        new_pop = list(elites)

        for i in range(pop_size - len(new_pop)):
            parent = elites[int(rng.integers(0, keep))]
            method = methods[int(rng.integers(0, len(methods)))]
            strength = adaptive_strength * (1.0 + 0.5 * np.sin(gen * 0.1))
            child = perturb_sequence(parent, rng, method, strength)
            new_pop.append(child)

        if pop_size > len(new_pop) and len(elites) >= 2:
            t = rng.uniform(0.2, 0.8)
            child = normalize(t * elites[0] + (1 - t) * elites[1])
            new_pop.append(child)

        population = new_pop[:pop_size]
        scores = [evaluate_local(p) for p in population]

        if gen % 30 == 0:
            if scores[0] >= best_score - 1e-10:
                adaptive_strength *= 0.85
            else:
                adaptive_strength = min(0.04, adaptive_strength * 1.1)

    return best_a, best_score

    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, sc_opt = optimize_sequence(best_seq, t_alloc, rng)
        try_update(a_opt, f"opt_n{n_best}")

    # === Phase 2: Try longer lengths ===
    print(f"\n=== Phase 2: Longer lengths (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        lengths_to_try = []
        for n_try in [n_best + 300, n_best + 600, n_best + 1000, int(n_best * 1.5), int(n_best * 2.0)]:
            if 800 <= n_try <= 5000 and n_try not in lengths_to_try:
                lengths_to_try.append(n_try)

        for n_try in lengths_to_try:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_sub = min(rem * 0.25, 200.0)
            if t_sub < 20:
                break
            print(f"  Trying n={n_try} (budget={t_sub:.0f}s)")
            init = interpolate_sequence(best_seq, n_try)
            # Quick variance pass + LSE
            a_sub, sc_sub = optimize_sequence(init, t_sub, rng)
            try_update(a_sub, f"len_{n_try}")

    # === Phase 3: Try shorter lengths (sometimes helps) ===
    print(f"\n=== Phase 3: Shorter lengths (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60 and n_best > 1500:
        for n_try in [n_best - 200, n_best - 400]:
            if n_try < 500:
                continue
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem * 0.2, 80.0)
            if t_sub < 15:
                break
            print(f"  Trying n={n_try}")
            init = interpolate_sequence(best_seq, n_try)
            a_sub, sc_sub = optimize_sequence(init, t_sub, rng)
            try_update(a_sub, f"short_{n_try}")

    # === Phase 4: Population search at best length ===
    print(f"\n=== Phase 4: Population search (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_pop = min(rem * 0.25, 250.0)
        print(f"  Budget: {t_pop:.0f}s")
        a_pop, _ = population_search(best_seq, t_pop, pop_size=10, rng=rng)
        try_update(a_pop, "population")

    # === Phase 5: Re-optimize best found ===
    print(f"\n=== Phase 5: Re-optimize best (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_re = min(rem * 0.3, 200.0)
        a_re, sc_re = optimize_sequence(best_seq, t_re, rng)
        try_update(a_re, "reoptimize")

    # === Phase 6: Final fine LSE polish ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        T_final = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
        t_fin = min(rem * 0.5, 60.0)
        a_fin = lbgfs_optimize(best_seq, t_fin, T_final, maxiter_per_stage=2000)
        try_update(a_fin, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        # Also verify with the actual evaluate_sequence
        test_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        actual = evaluate_sequence(test_list)
        print(f"\nFINAL: local={best_score:.8f}, actual={actual:.8f} (n={len(best_seq)})")
        if actual < best_score:
            best_score = actual
    else:
        print("ERROR: No valid sequence found!")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
