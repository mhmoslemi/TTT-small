Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov LP alternating with periodic QCQP checks
2. Iterative active-set QCQP (main innovation): iteratively adds violated
   quadratic constraints until the KKT conditions are satisfied
3. Individual QP analysis to identify truly active constraints
4. Low-temperature LogSumExp L-BFGS for final precision
5. Multi-n exploration (n=1197..1203)
Key innovation: The iterative active-set method systematically builds up the
correct active constraint set by solving the QCQP, checking all 2n-1
constraints, and adding the most violated one. This converges to the
exact optimum of the full QCQP (all constraints), unlike the fixed
tolerance approach in the previous version.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix, csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.MOSEK, cp.GUROBI]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    else:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_sum = b_val.sum()
    if s_sum < 1e-15:
        return None, None
    b_val = b_val / s_sum
    return b_val, float(max(0, M_var.value))


def solve_qcqp_active(a, active_list, solver):
    """
    Solve: min t s.t. a^T Q_k a <= t for k in active_list, sum(a)=1, a>=0
    Returns (a_optimal, t_value) or (None, None) on failure.
    """
    n = len(a)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active_list:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def iterative_active_qcqp(a0, solver, max_rounds=100, tol=1e-10, time_limit=None, t0=None):
    """
    Iterative active-set QCQP:
    1. Start with initial active set (top peaks)
    2. Solve QCQP
    3. Check ALL 2n-1 constraints
    4. Add most violated if any
    5. Repeat until no violation > tol
    
    This converges to the optimal solution of the full problem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    if t0 is None:
        t0 = time.monotonic()
    if time_limit is not None:
        deadline = t0 + time_limit
    else:
        deadline = float('inf')

    # Initialize active set with top peaks
    b = fast_convolve(a)
    top_k = np.argsort(b)[::-1][:min(10, len(b))]
    active_set = list(set(int(k) for k in top_k))

    best_a = a.copy()
    best_score = 2.0 * n * b.max()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Solve QCQP with current active set
        a_new, t_val = solve_qcqp_active(a, active_set, solver)
        if a_new is None:
            break
        a = a_new

        # Evaluate score
        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Check for violated constraints
        violations = b - t_val
        worst_idx = int(np.argmax(violations))
        worst_viol = violations[worst_idx]

        if worst_viol < tol:
            print(f"    QCQP converged: round={rnd+1}, t={t_val:.10f}, max_b={max_b:.10f}")
            break

        # Add the most violated constraint
        if worst_idx not in active_set:
            active_set.append(worst_idx)

        # Also add any constraints that are close to being violated
        near_violated = np.where(violations > tol * 0.1)[0]
        for k in near_violated:
            k = int(k)
            if k not in active_set:
                active_set.append(k)

        # Periodic pruning: remove constraints that are no longer active
        if len(active_set) > 50:
            # Keep only constraints that are tight (within 1e-6 of t)
            b_cur = fast_convolve(a)
            tight = np.where(b_cur >= t_val - 1e-6)[0]
            active_set = list(set(int(k) for k in tight))
            if len(active_set) == 0:
                active_set = [int(np.argmax(b_cur))]

        if (rnd + 1) % 5 == 0:
            print(f"    QCQP round={rnd+1}: active={len(active_set)}, t={t_val:.8f}, score={score:.10f}")

    return best_a, best_score


def solve_single_qp(k, n, solver):
    """Solve min a^T Q_k a s.t. sum(a)=1, a>=0."""

    a_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Minimize(cp.quad_form(a_var, Q, assume_PSD=True)),
                      [cp.sum(a_var) == 1])
    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:

        try:

                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:

        except Exception:

    return None


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov LP alternating with adaptive restarts and periodic QCQP."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        # Nesterov acceleration
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                # Restart with perturbation
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                amplitude = 0.02 + 0.01 * rng.random()
                noise = amplitude * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

        if len(lp_times) > 200 and np.mean(lp_times[-200:]) > 3.0:
            print(f"    LP too slow, stopping at iter {it+1}")
            break

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        iters = max(10, min(200, int(maxiter_per_stage / (len(T_stages) * 0.1))))
        
        def func(zz):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_obj_grad(zz, n, stage_T)
            return obj

        def jac(zz):
            if time.monotonic() > deadline:
                return np.zeros_like(zz)
            _, g = lse_obj_grad(zz, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-15, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.ECOS, cp.SCS, cp.CLARABEL]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    else:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_sum = b_val.sum()
    if s_sum < 1e-15:
        return None, None
    b_val = b_val / s_sum
    return b_val, float(max(0, M_var.value))


def solve_qcqp_active_set(a, active_list, solver):
    """Solve min t s.t. a^T Q_k a <= t for k in active_list, sum(a)=1, a>=0."""
    n = len(a)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active_list:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def iterative_active_qcqp(a0, solver, max_rounds=100, tol=1e-10, deadline=None):
    """
    Iterative active-set QCQP with constraint cycling.
    Systematically adds violated constraints until optimal.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if deadline is None:
        deadline = float('inf')

    b = fast_convolve(a)
    top_k = np.argsort(b)[::-1][:min(10, len(b))]
    active_set = set(int(k) for k in top_k)

    best_a = a.copy()
    best_score = 2.0 * n * b.max()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        active_list = sorted(active_set)
        a_new, t_val = solve_qcqp_active_set(a, active_list, solver)
        if a_new is None:
            break
        a = a_new

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Check ALL constraints for violation
        violations = b - t_val
        worst_idx = int(np.argmax(violations))
        worst_viol = violations[worst_idx]

        if worst_viol < tol:
            break

        # Add most violated constraint
        active_set.add(worst_idx)

        # Also add near-violated constraints (within 10% of worst)
        threshold = t_val + max(0.01 * worst_viol, tol)
        near_viol = np.where(b >= threshold)[0]
        for k in near_viol:
            active_set.add(int(k))

        # Prune: if active set too large, keep only tight constraints
        if len(active_set) > 80:
            tight = np.where(b >= t_val - 1e-6)[0]
            active_set = set(int(k) for k in tight)
            if len(active_set) == 0:
                active_set = {int(np.argmax(b))}

        if (rnd + 1) % 5 == 0:
            print(f"    QCQP r{rnd+1}: |A|={len(active_set)}, t={t_val:.10f}, score={score:.10f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 12:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                amp = 0.01 + 0.02 * rng.random()
                noise = amp * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                noise2 = amp * 0.5 * np.cos((freq * 2 + 1) * x + rng.uniform(0, 2 * np.pi))
                y = y + noise + noise2
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} lp={avg_lp:.1f}ms")

        if len(lp_times) > 200 and np.mean(lp_times[-200:]) > 3.0:
            break

    return best_a, best_score


def symmetrize(a):
    """Symmetrize sequence: return (a + rev(a))/2, renormalized."""
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj_grad(z, n, T_temp):
    """LogSumExp objective and gradient in z-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_total=5000):
    """L-BFGS with temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    n_stages = len(T_stages)
    iters_per = max(10, maxiter_total // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(zz):
            obj, _ = lse_obj_grad(zz, n, stage_T)
            return obj

        def jac(zz):
            _, g = lse_obj_grad(zz, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-22, 'gtol': 1e-15, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate to new length, preserve normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local score evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization. Key innovations over previous:
    1. Iterative active-set QCQP (converges to exact optimum)
    2. Symmetrization (exploits convexity for free improvement)
    3. More aggressive multi-n search
    4. Better time allocation
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score_state = [float('inf')]
    best_seq_state = [None]
    solver = get_lp_solver()
    print(f"Solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq_state[0] = parent.copy()
            best_score_state[0] = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        init = np.cos(np.pi * x / 2) ** 2
        init = np.maximum(init, 0)
        init = init / init.sum()
        best_seq_state[0] = init
        best_score_state[0] = evaluate_local(init)
        print(f"Default init: n={len(init)}, score={best_score_state[0]:.10f}")

    n_base = len(best_seq_state[0])

    def try_update(a, label):
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score_state[0]:
            best_score_state[0] = sc
            best_seq_state[0] = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Initial QCQP from parent ===
    print(f"\n=== Phase 1: Init QCQP (score={best_score_state[0]:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        a_q, _ = iterative_active_qcqp(best_seq_state[0], solver,
                                        max_rounds=30, tol=1e-8,
                                        deadline=time.monotonic() + min(rem * 0.08, 60))
        try_update(a_q, "qcqp_init")
        # Also try symmetrized
        if best_seq_state[0] is not None:
            a_sym = symmetrize(best_seq_state[0])
            try_update(a_sym, "sym_init")

    # === Phase 2: Nesterov LP alternating ===
    print(f"\n=== Phase 2: LP Alternating n={len(best_seq_state[0])} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 450.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq_state[0], t_alloc, solver, n_max_iters=500000)
        try_update(a_lp, "lp_main")
        if best_seq_state[0] is not None:
            a_sym = symmetrize(best_seq_state[0])
            try_update(a_sym, "sym_after_lp")

    # === Phase 3: Iterative QCQP (main innovation) ===
    print(f"\n=== Phase 3: Iterative QCQP (score={best_score_state[0]:.10f}) ===")
    rem = deadline - time.monotonic()
    t_qcqp = min(rem * 0.20, 250.0)
    if t_qcqp > 30:
        print(f"  Budget: {t_qcqp:.0f}s")
        a_q2, _ = iterative_active_qcqp(best_seq_state[0], solver,
                                         max_rounds=80, tol=1e-11,
                                         deadline=time.monotonic() + t_qcqp)
        try_update(a_q2, "iter_qcqp")
        if best_seq_state[0] is not None:
            a_sym = symmetrize(best_seq_state[0])
            try_update(a_sym, "sym_qcqp")
            # QCQP on symmetrized
            a_q3, _ = iterative_active_qcqp(a_sym, solver,
                                             max_rounds=50, tol=1e-11,
                                             deadline=time.monotonic() + min(rem * 0.05, 40))
            try_update(a_q3, "qcqp_sym")

    # === Phase 4: L-BFGS polish ===
    print(f"\n=== Phase 4: L-BFGS (score={best_score_state[0]:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lbgfs = min(rem * 0.08, 100.0)
    if t_lbgfs > 15:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_lb = lbgfs_refine(best_seq_state[0], t_lbgfs, T_stages, maxiter_total=4000)
        try_update(a_lb, "lbgfs")

    # === Phase 5: Final QCQP on current best ===
    print(f"\n=== Phase 5: Final QCQP (score={best_score_state[0]:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t_fq = min(rem * 0.15, 100.0)
        a_qf, _ = iterative_active_qcqp(best_seq_state[0], solver,
                                         max_rounds=60, tol=1e-12,
                                         deadline=time.monotonic() + t_fq)
        try_update(a_qf, "qcqp_final")
        a_sym = symmetrize(best_seq_state[0])
        try_update(a_sym, "sym_final")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n (score={best_score_state[0]:.10f}) ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq_state[0])
    n_targets = [n_best - 3, n_best - 2, n_best - 1, n_best + 1, n_best + 2, n_best + 3]
    n_targets = [nt for nt in n_targets if nt >= 200 and nt <= 2000]
    per_n = max(20, min(rem / max(1, len(n_targets)) * 0.55, 100))

    for nt in n_targets:
        rem = deadline - time.monotonic()
        if rem < per_n * 0.6:
            break
        print(f"  n={nt} (budget={per_n:.0f}s)")
        init_nt = interpolate_sequence(best_seq_state[0], nt)
        t_lp = per_n * 0.5
        a_lp_nt, _ = nesterov_lp_refine(init_nt, t_lp, solver, n_max_iters=150000)
        try_update(a_lp_nt, f"lp_n{nt}")
        if best_seq_state[0] is not None and len(best_seq_state[0]) == nt:
            rem = deadline - time.monotonic()
            if rem > 20:
                a_q_nt, _ = iterative_active_qcqp(best_seq_state[0], solver,
                                                   max_rounds=40, tol=1e-10,
                                                   deadline=time.monotonic() + min(rem * 0.3, 35))
                try_update(a_q_nt, f"qcqp_n{nt}")
                a_sym_nt = symmetrize(best_seq_state[0])
                try_update(a_sym_nt, f"sym_n{nt}")

    # === Phase 7: Ultimate polish ===
    print(f"\n=== Phase 7: Ultimate Polish (score={best_score_state[0]:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        T_ultra = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        a_u = lbgfs_refine(best_seq_state[0], min(rem * 0.25, 50), T_ultra, maxiter_total=3000)
        try_update(a_u, "lbgfs_ultra")
        a_sym_u = symmetrize(best_seq_state[0])
        try_update(a_sym_u, "sym_ultra")

        rem = deadline - time.monotonic()
        if rem > 20:
            a_qult, _ = iterative_active_qcqp(best_seq_state[0], solver,
                                               max_rounds=40, tol=1e-13,
                                               deadline=time.monotonic() + min(rem * 0.5, 40))
            try_update(a_qult, "qcqp_ultimate")
            a_sym_ult = symmetrize(best_seq_state[0])
            try_update(a_sym_ult, "sym_ultimate")

    # Final
    if best_seq_state[0] is not None:
        fs = evaluate_local(best_seq_state[0])
        if np.isfinite(fs):
            best_score_state[0] = fs
        print(f"\nFINAL: score={best_score_state[0]:.10f} (n={len(best_seq_state[0])})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq_state[0]]
