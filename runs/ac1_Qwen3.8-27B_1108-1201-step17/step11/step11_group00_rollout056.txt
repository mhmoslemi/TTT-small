def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def proj_simplex(v, sum_target=1.0):
    """Project v onto {x : sum(x) = sum_target, x >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - sum_target))[0]
    if len(rho) == 0:
        return np.full(n, sum_target / n)
    rho = rho[-1]
    theta = (cssv[rho] - sum_target) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgrad_fft(a, weights):
    """
    Compute 2 * sum_k weights[k] * a[k-i] for i=0..n-1 using FFT.
    weights has length 2n-1 (corresponding to autocorrelation lags).
    """
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a_rev, fft_size)
    W = rfft(weights, fft_size)
    conv = irfft(A_rev * W, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def mirror_step(a, grad, eta):
    """Multiplicative weight update: a_new[i] ∝ a[i]*exp(-eta*grad[i])."""
    log_a = np.log(np.maximum(a, 1e-300))
    log_new = log_a - eta * grad
    log_new -= log_new.max()
    new = np.exp(log_new)
    s = new.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return new / s


def subgradient_optimize(a0, time_limit, top_k=10, verbose=True):
    """
    Fast subgradient optimization using mirror descent.
    Thousands of iterations in the time limit.

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    ema_a = a.copy()
    ema_alpha = 0.9
    
    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        
        # Find top-k peaks
        top_idx = np.argsort(b)[::-1][:top_k]

        if w_sum < 1e-15:
            weights[top_idx[0]] = 1.0
        else:
            weights /= w_sum
        
        # Compute subgradient
        grad = compute_subgrad_fft(a, weights)

        eta = 1.0 / np.sqrt(it + 1)

            print(f"    SubGrad it={it} ({elapsed:.0f}s): cur={cur_score:.8f}, best={best_score:.8f}")

def get_lp_solver():
    """Find a working LP solver."""
    if not HAS_CP:
        return None
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix (banded/Toeplitz structure)
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    # T has shape (2n-1) x n
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def spectral_flatten(a0, time_limit, beta_start=0.5, verbose=True):
    """
    Spectral flattening: compress the dynamic range of |FFT(a)|^2
    to make the autocorrelation more uniform.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute spectrum
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Flatten: power_new = power^alpha where alpha < 1 compresses dynamic range
        beta = beta_start * (1.0 - it / 200.0) + 0.3  # start at beta_start, end at 0.3
        power_flat = np.sign(power) * np.maximum(power, 1e-30) ** beta
        
        # Reconstruct
        # Need to find A_new such that |A_new|^2 = power_flat and A_new has right phase
        phase = np.angle(A)
        A_new = np.sqrt(power_flat) * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert if it got worse
            a = a  # keep old
        
        if verbose and it % 20 == 0:
            print(f"    SpecFlat it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=15, verbose=True):
    """LogSumExp-smoothed optimization with L-BFGS."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        iters = max(20, min(100, int(30 + 30 * (si / n_stages))))
        
        def func(z_in):
            a = _softmax(z_in)
            b = fast_autocorr(a)
            Tb = stage_T * b
            return (Tb.max() + np.log(np.exp(Tb - Tb.max()).sum())) / stage_T
        
        def jac(z_in):
            a = _softmax(z_in)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = compute_subgrad_fft(a, w)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                                    'maxfun': iters * 200, 'disp': False})
            z = res.x
            a = _softmax(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass
        
        if verbose and si % 5 == 0:
            print(f"    LSE stage={si}/{n_stages} T={stage_T:.3f} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def proj_simplex(v, sum_target=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.nonzero(u * np.arange(1, n + 1) > (cssv - sum_target))[0]
    if len(rho_arr) == 0:
        return np.full(n, sum_target / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - sum_target) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgrad_fft(a, weights):
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1

    A_rev = rfft(a_rev, fft_size)
    W = rfft(weights, fft_size)
    conv = irfft(A_rev * W, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]
