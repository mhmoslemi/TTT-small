"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated multiplicative weights (subgradient) - fast bulk optimization
2. LP alternating with exact best-response - linear convergence near optimum
3. Active-set QCQP for exact convex polish
4. L-BFGS with LogSumExp smoothing for final numerical precision
Key improvement over previous: replaces expensive per-iteration LP with O(n log n)
subgradient steps for the bulk phase, freeing time for more LP iterations and QCQP polish.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the evaluation score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_subgradient(a, active):
    """Compute subgradient of max_k conv(a,a)[k] at a, averaged over active set."""
    n = len(a)
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        partners = k - np.arange(js, je + 1)
        g[js:je + 1] += a[partners]
    g *= 2.0 / len(active)
    return g


def nesterov_mw(a0, budget, s=1.0):
    """
    Nesterov-accelerated multiplicative weights (mirror descent) on the simplex.
    Minimizes max_k conv(a,a)[k] subject to a>=0, sum(a)=s.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    sa = a.sum()
    if sa < 1e-15:
        a = np.ones(n) / n
        sa = 1.0
    a = a / sa  # work with sum=1 internally

    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Adaptive parameters
    active_tol = 0.0  # start tight, widen over time
    eta_scale = 0.5
    rng = np.random.default_rng(123)
    prev_best = best_score
    stale_iters = 0

    it = 0
    while time.monotonic() - t0 < budget:
        it += 1

        # Adaptive active set tolerance (widen over time)
        active_tol = min(0.001, 0.0001 * (it // 500))

        # Compute subgradient at y
        c = fast_convolve(y)
        max_c = c.max()
        threshold = max_c * (1.0 - active_tol)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        g = compute_subgradient(y, active)

        # Step size with adaptation
        eta = eta_scale / np.sqrt(max(it, 1))

        # Multiplicative update (mirror descent step)
        log_update = -eta * g
        log_update -= log_update.max()
        y_new = y * np.exp(log_update)
        total = y_new.sum()
        if total < 1e-300:
            y_new = y.copy()
            total = 1.0
        y_new = y_new / total

        # Nesterov extrapolation
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * y_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_new.copy()
            sy = 1.0
        y = y / sy

        a = y_new

        # Track best
        score = 2.0 * n * fast_convolve(a)
        score_val = score.max() * 2.0 * n  # since sum=1
        score_val = 2.0 * n * score.max()
        if score_val < best_score:
            best_score = score_val
            best_a = a.copy()
            stale_iters = 0
        else:
            stale_iters += 1

        # Periodic restart if stuck
        if stale_iters > 2000 and it % 5000 == 0:
            noise_scale = 0.01 * max(1.0, eta)
            p = rng.dirichlet(np.ones(n))
            y = (1 - noise_scale) * y + noise_scale * p
            y = np.maximum(y, 0)
            y /= y.sum()
            stale_iters = 0

        # Adaptive eta: if not improving, reduce
        if it % 5000 == 0:
            if best_score < prev_best - 1e-12:
                prev_best = best_score
                eta_scale = min(eta_scale * 1.1, 2.0)
            else:
                eta_scale = max(eta_scale * 0.85, 0.05)
                prev_best = best_score

        if it % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter={it} ({elapsed:.0f}s): score={best_score:.10f} eta={eta:.6f} |A|={len(active)}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    a_arr = np.asarray(a_fixed, dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a_arr
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, warm_b=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s_name in solvers_to_try:
        if s_name is None:
            continue
        try:
            extra = {}
            if warm_b is not None and s_name in (cp.CLARABEL, cp.PDLP):
                extra['warm_start'] = True
            prob.solve(solver=s_name, **extra)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None, s_name
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value)), s_name
        except Exception:
            continue
    return None, None, None


def nesterov_lp_alt(a0, budget, solver, max_iters=500000):
    """Nesterov-accelerated LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()

    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    slow_streak = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > budget:
            break

        lp_t0 = time.monotonic()
        b_new, M_b, used_solver = solve_lp_best_response(y, solver, warm_b=a)
        lp_time = time.monotonic() - lp_t0

        if b_new is None:
            continue

        # Track slow LPs
        if lp_time > 2.0:
            slow_streak += 1
        else:
            slow_streak = max(0, slow_streak - 1)
        if slow_streak > 50:
            print(f"    LP too slow, stopping at iter {it+1}")
            break

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1.0) * a + alpha * (it + 2.0) * b_new) / (it + 1.0 + alpha * (it + 2.0))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                p = rng.dirichlet(np.ones(n) * 0.5)
                y = 0.95 * y + 0.05 * p
                y = np.maximum(y, 0)
                y /= y.sum()
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} lp_time={lp_time*1000:.1f}ms")

    return best_a, best_score


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_active_qcqp(a, solver, rel_tol=1e-5, max_active=20):
    """Solve active-set QCQP: min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]

    if len(active) > max_active:
        active = active[np.argsort(b[active])[-max_active:]]

    if len(active) == 0:
        active = np.array([np.argmax(b)])

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s_name in solvers_to_try:
        if s_name is None:
            continue
        try:
            prob.solve(solver=s_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def lse_objective_and_grad(z, n, T_temp):
    """Compute LogSumExp smoothed objective and gradient w.r.t. z (softmax parameterization)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    c = fast_convolve(a)
    max_c = c.max()
    exp_terms = np.exp((c - max_c) / T_temp)
    lse = max_c + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    # Gradient: dLSE/dz_j = 2*T*a_j*(S_j - V) where S_j = sum_k w_k*a[k-j], V = sum_k w_k*c_k
    V = np.dot(w, c)
    # Compute S = correlation of w with a
    S = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        partners = k - np.arange(js, je + 1)
        S[js:je + 1] += w[k] * a[partners]

    grad = 2.0 * T_temp * a * (S - V)
    return obj, grad


def lbgfs_refine(a0, budget, T_stages=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing in log-space."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z -= z.max()  # center for numerical stability
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    n_stages = len(T_stages)
    iters_per_stage = max(30, max(500, int(budget * 50)) // n_stages)

    for stage_T in T_stages:
        elapsed = time.monotonic() - t0
        if elapsed > budget:
            break
        remaining = budget - elapsed
        if remaining < 3:
            break

        def fun(z_val):
            obj, _ = lse_objective_and_grad(z_val, n, stage_T)
            return obj

        def jac(z_val):
            _, g = lse_objective_and_grad(z_val, n, stage_T)
            return g

        try:
            res = minimize(
                fun=fun, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': min(iters_per_stage, int(remaining * 30)),
                         'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': int(remaining * 100), 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        update_best(a_lb, "lbgfs_final")

    # === Phase 6: Try n=1199 ===
    print(f"\n=== Phase 6: Try n=1199 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_1199 = min(rem * 0.15, 150.0)
        print(f"  Budget: {t_1199:.0f}s")
        init_1199 = interpolate_sequence(best_seq, 1199)
        a_1199, _ = nesterov_lp_alt(init_1199, t_1199 * 0.7, solver, max_iters=200000)
        update_best(a_1199, "lp_n1199")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1199 and rem > 10:
            a_q, _ = solve_active_qcqp(best_seq, solver, rel_tol=1e-6, max_active=10)
            update_best(a_q, "qcqp_n1199")

    # === Phase 7: Try n=1201 ===
    print(f"\n=== Phase 7: Try n=1201 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_1201 = min(rem * 0.15, 150.0)
        print(f"  Budget: {t_1201:.0f}s")
        # Interpolate from the n_best solution
        ref = best_seq if len(best_seq) != 1201 else best_seq
        # Use the original best before n changes
        init_1201 = interpolate_sequence(best_seq, 1201)
        a_1201, _ = nesterov_lp_alt(init_1201, t_1201 * 0.7, solver, max_iters=200000)
        update_best(a_1201, "lp_n1201")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1201 and rem > 10:
            a_q, _ = solve_active_qcqp(best_seq, solver, rel_tol=1e-6, max_active=10)
            update_best(a_q, "qcqp_n1201")

    # === Phase 8: Final QCQP on best ===
    print(f"\n=== Phase 8: Final QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 8:
        a_qf, _ = solve_active_qcqp(best_seq, solver, rel_tol=1e-8, max_active=6)
        update_best(a_qf, "final_qcqp")

    # Final verification
    if best_seq is not None:
        fs = compute_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
