       n = len(a)

       b = np.maximum(b, 0)
       s = b.sum()
       if s < 1e-15:
           return None
       return b / s

           score = 2.0 * n * fast_convolve(a).max()

       return best_a, best_score

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_lp_data(a):
    """Build LP data for best response: min M s.t. (a*b)[k]<=M, sum(b)=1, b>=0.
    Returns (c, A_ub, b_ub, A_eq, b_eq, bounds) for scipy linprog."""
    n = len(a)
    size = 2 * n - 1
    # A_ub is size x (n+1): row k has [a[k-j] for valid j, -1]
    A_ub = np.zeros((size, n + 1))
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[k, js:je + 1] = a[k - je:k - js + 1]
        A_ub[k, n] = -1.0
    c = np.zeros(n + 1)
    c[n] = 1.0
    b_ub = np.zeros(size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    return c, A_ub, b_ub, A_eq, b_eq, bounds


def solve_best_response(a):
    """Solve the LP to find b minimizing max_k (a*b)[k] subject to sum(b)=1, b>=0."""
    n = len(a)
    c, A_ub, b_ub, A_eq, b_eq, bounds = build_lp_data(a)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status != 0:
            return None
        b = res.x[:n]
        b = np.maximum(b, 0)
        s = b.sum()
        if s < 1e-15:
            return None
        return b / s
    except Exception:
        return None

def _solve_best_response_cvxpy(a_fixed, solver):
    """Fallback: solve best response LP using cvxpy."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    if solver:
        solvers_to_try.insert(0, solver)
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None
                return b_val / s_sum
        except Exception:
            continue
    return None


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_qcqp_growing(a0, max_active=15, rel_tol=1e-5):
    """
    Active-set QCQP with growing active set.
    Start with top constraint, add violated ones iteratively.
    """
    n = len(a0)
    active = set()

    # Start with the argmax position
    b = fast_convolve(a0)
    k_max = int(np.argmax(b))
    active.add(k_max)
    # Also add nearby positions that are close to max

    for k in range(len(b)):
        if b[k] >= threshold and len(active) < 3:
            active.add(k)

    current_a = a0.copy()
    current_score = 2.0 * n * b.max()

    for iteration in range(max_active + 5):
        if len(active) == 0:
            active.add(k_max)

        # Build and solve QCQP

        for k in sorted(active):

            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        a_sol = None
        solvers_to_try = [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]

                    a_sol = np.maximum(a_sol, 0)
                    s_sum = a_sol.sum()
                    if s_sum < 1e-15:
                        a_sol = None
                    else:
                        a_sol = a_sol / s_sum
                    break
            except Exception:
                continue

        if a_sol is None:
            break

        # Check for violated constraints
        b_new = fast_convolve(a_sol)
        max_b_new = b_new.max()
        t_val = float(max(0, t_var.value))

        worst_k = -1
        worst_violation = 0.0

        current_score = score
    return current_a, current_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, n, T_temp):
    """Compute LogSumExp smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lse_func_wrapper(z, n, T_temp, deadline):
    """Top-level LSE objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = lse_obj_grad(z, n, T_temp)
    return obj


def lse_jac_wrapper(z, n, T_temp, deadline):
    """Top-level LSE gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = lse_obj_grad(z, n, T_temp)
    return g


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    if T_stages is None:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        try:
            res = minimize(
                fun=lse_func_wrapper, x0=z, jac=lse_jac_wrapper,
                args=(n, stage_T, deadline),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            scale *= 0.7
            no_improve = 0
            if scale < 0.002:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    """Find a working LP solver for cvxpy (fallback)."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast scipy-based Nesterov LP alternating, growing active set QCQP,
    L-BFGS LogSumExp, and multiplicative weights polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
        elif np.isfinite(sc) and sc < best_score:
            # Very small improvement, still update
            best_score = sc
            best_seq = a.copy()

    # === Phase 1: Fast Nesterov LP alternating (main workhorse) ===
    print(f"\n=== Phase 1: Fast NLP n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = nesterov_lp_refine(best_seq, t_alloc, max_iters=500000)
        update_best(a_lp, "nlp_phase1")

    # === Phase 2: QCQP polish (growing active set) ===
    print(f"\n=== Phase 2: QCQP Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_qcqp = min(rem * 0.15, 200.0)
    if t_qcqp > 20:
        q_start = time.monotonic()
        # Wide active set
        a_q1, sc_q1 = solve_qcqp_growing(best_seq, max_active=20, rel_tol=1e-4)
        update_best(a_q1, "qcqp_wide")
        # Narrow active set (more precise)
        a_q2, sc_q2 = solve_qcqp_growing(best_seq, max_active=8, rel_tol=1e-6)
        update_best(a_q2, "qcqp_narrow")
        # Single constraint QP
        if rem > 10:
            b_cur = fast_convolve(best_seq)
            k_max = int(np.argmax(b_cur))
            n = len(best_seq)
            Q = build_q_sparse(k_max, n)
            a_var = cp.Variable(n, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.quad_form(a_var, Q, assume_PSD=True)),
                              [cp.sum(a_var) == 1])
            for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS]:
                if s is None:
                    continue
                try:
                    prob.solve(solver=s)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        a_qp = np.asarray(a_var.value, dtype=np.float64)
                        a_qp = np.maximum(a_qp, 0)
                        ss = a_qp.sum()
                        if ss > 1e-15:
                            update_best(a_qp / ss, "single_qp")
                        break
                except Exception:
                    continue
        print(f"  QCQP time: {time.monotonic() - q_start:.1f}s")

    # === Phase 3: Second NLP round from polished point ===
    print(f"\n=== Phase 3: NLP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, sc_lp2 = nesterov_lp_refine(best_seq, t_alloc, max_iters=500000)
        update_best(a_lp2, "nlp_phase3")

    # === Phase 4: QCQP polish round 2 ===
    print(f"\n=== Phase 4: QCQP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_q3, _ = solve_qcqp_growing(best_seq, max_active=10, rel_tol=1e-6)
        update_best(a_q3, "qcqp_round2")

    # === Phase 5: L-BFGS LogSumExp ===
    print(f"\n=== Phase 5: L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lbgfs = min(rem * 0.12, 150.0)
    if t_lbgfs > 20:
        a_lb = lbgfs_refine(best_seq, t_lbgfs,
                            T_stages=[3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005],
                            maxiter_per_stage=5000)
        update_best(a_lb, "lbgfs")

    # === Phase 6: Multiplicative weights ===
    print(f"\n=== Phase 6: Mult Weights (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_ms, _ = multiplicative_subgradient(best_seq, min(rem * 0.2, 25.0), n_steps=100000)
        update_best(a_ms, "mult")
        # Final QCQP
        rem = deadline - time.monotonic()
        if rem > 10:
            a_qf, _ = solve_qcqp_growing(best_seq, max_active=5, rel_tol=1e-7)
            update_best(a_qf, "qcqp_final")

    # === Phase 7: Try n=1199 ===
    print(f"\n=== Phase 7: n=1199 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_1199 = min(rem * 0.15, 180.0)
        print(f"  Budget: {t_1199:.0f}s")
        init_1199 = interpolate_sequence(best_seq, 1199)
        a_1199, _ = nesterov_lp_refine(init_1199, t_1199 * 0.6, max_iters=200000)
        update_best(a_1199, "nlp_1199")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1199 and rem > 15:
            a_q, _ = solve_qcqp_growing(best_seq, max_active=10, rel_tol=1e-5)
            update_best(a_q, "qcqp_1199")

    # === Phase 8: Try n=1201 ===
    print(f"\n=== Phase 8: n=1201 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_1201 = min(rem * 0.15, 180.0)
        print(f"  Budget: {t_1201:.0f}s")
        init_1201 = interpolate_sequence(best_seq, 1201)
        a_1201, _ = nesterov_lp_refine(init_1201, t_1201 * 0.6, max_iters=200000)
        update_best(a_1201, "nlp_1201")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1201 and rem > 15:
            a_q, _ = solve_qcqp_growing(best_seq, max_active=10, rel_tol=1e-5)
            update_best(a_q, "qcqp_1201")

    # === Phase 9: Final polish on best ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        a_final, _ = solve_qcqp_growing(best_seq, max_active=6, rel_tol=1e-8)
        update_best(a_final, "final_qcqp")
        rem = deadline - time.monotonic()
        if rem > 8:
            a_ms2, _ = multiplicative_subgradient(best_seq, min(rem * 0.5, 15.0), n_steps=50000)
            update_best(a_ms2, "final_mult")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
