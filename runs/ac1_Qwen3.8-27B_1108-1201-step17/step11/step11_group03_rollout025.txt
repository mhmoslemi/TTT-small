        # base_conv = conv - 2*ai_old*partner + ai_old^2*[quad_pos]
        # i.e., base_conv[k] = conv[k] - lin[k]*ai_old - ai_old^2*[k==quad_pos]
        base_conv = conv - ai_old * lin
        if quad_pos >= 0:
            base_conv[quad_pos] -= ai_old * ai_old
        
        base_sum = total_sum - ai_old
        
        def obj(x):
            if x < 0:
                x = 0.0
            vals = base_conv + x * lin
            if quad_pos >= 0:
                vals[quad_pos] += x * x
            s = base_sum + x
            if s < 1e-15:
                return 1e15
            return 2.0 * n * vals.max() / (s * s)
        
        x_upper = max(5.0 * ai_old, 0.1 * base_sum, 1e-6)
        
        try:
            result = minimize_scalar(obj, bounds=(0.0, x_upper), method='bounded',
                                     options={'xatol': 1e-14})

        except:
            ai_new = ai_old
        
        # Update conv
        if abs(ai_new - ai_old) > 1e-15:
            conv = conv + (ai_new - ai_old) * lin
            if quad_pos >= 0:
                conv[quad_pos] += (ai_new * ai_new - ai_old * ai_old)
            a[i] = ai_new
            total_sum = total_sum - ai_old + ai_new
    
    return a, conv, total_sum

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP alternating with Nesterov acceleration (main workhorse)
2. Coordinate descent polish (NEW - optimizes one coordinate at a time via 1D convex search)
3. Active-set QCQP polishing
4. Multi-n exploration (1197-1203)
Key new contribution: Coordinate descent exploits the 1D structure of each coordinate's
optimization (max of a linear + one quadratic function over k), solved via golden section.
This provides a different convergence path than LP alternating and can escape its local minima.

import numpy as np
import cvxpy as cp
import time
import math
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)
    consecutive_lp_failures = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            consecutive_lp_failures += 1
            if consecutive_lp_failures >= 5:
                break
            continue
        consecutive_lp_failures = 0

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 12:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 7)
                noise = 0.05 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def golden_section_min(f, a, b, tol=1e-14, max_iter=200):
    """Minimize a 1D function on [a,b] using golden section search."""
    inv_phi = (math.sqrt(5.0) - 1.0) / 2.0
    c = b - inv_phi * (b - a)
    d = a + inv_phi * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - inv_phi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + inv_phi * (b - a)
            fd = f(d)
    return (a + b) / 2.0


def coordinate_descent_refine(a0, time_limit, max_passes=100000):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, the score as a function of x=a_i is:
      2n * max_k [base_k + x*lin_k + x^2*quad_k] / (base_sum + x)^2
    which is unimodal in x >= 0, solvable by golden section.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    total_sum = a.sum()
    if total_sum < 1e-15:
        return a0
    a = a / total_sum
    total_sum = 1.0
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    for p in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n)
        pass_improved = False

        for i in order:
            ai_old = a[i]

            # Build linear coefficient vector: lin[k] = 2*a[k-i] for valid k-i, 0 at k=2i
            start = i
            end = min(i + n, 2 * n - 1)
            lin = np.zeros(2 * n - 1)
            lin[start:end] = 2.0 * a[:end - start]
            quad_pos = 2 * i
            if quad_pos < 2 * n - 1:
                lin[quad_pos] = 0.0
            else:
                quad_pos = -1

            # base_conv = conv - ai_old * lin - ai_old^2 at quad_pos
            base_conv = conv - ai_old * lin
            if quad_pos >= 0:
                base_conv[quad_pos] -= ai_old * ai_old

            base_sum = total_sum - ai_old

            # 1D objective: minimize 2n * max(base_conv + x*lin + x^2*[k=quad_pos]) / (base_sum+x)^2
            def obj_1d(x):
                if x < 0.0:
                    x = 0.0
                vals = base_conv + x * lin
                if quad_pos >= 0:
                    vals[quad_pos] += x * x
                s = base_sum + x
                if s < 1e-15:
                    return 1e15
                return 2.0 * n * vals.max() / (s * s)

            # Upper bound for x
            x_upper = max(8.0 * ai_old, 0.5 * base_sum, 1e-6)
            # Verify x_upper is actually "above" the minimum
            obj_upper = obj_1d(x_upper)
            if obj_upper < best_score:
                # Expand upper bound
                x_upper = x_upper * 4.0

            x_new = golden_section_min(obj_1d, 0.0, x_upper, tol=1e-13, max_iter=150)
            x_new = max(0.0, x_new)

            if abs(x_new - ai_old) > 1e-15:
                # Update conv incrementally
                conv = conv + (x_new - ai_old) * lin
                if quad_pos >= 0:
                    conv[quad_pos] += (x_new * x_new - ai_old * ai_old)
                a[i] = x_new
                total_sum = total_sum - ai_old + x_new
                pass_improved = True

        score = 2.0 * n * fast_convolve(a).max() / (total_sum * total_sum)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 15:
            # Perturb to escape potential plateau
            perturbation = 0.02 * rng.standard_normal(n)
            a += perturbation
            a = np.maximum(a, 0)
            ts = a.sum()
            if ts > 1e-15:
                a = a / ts
                total_sum = 1.0
                conv = fast_convolve(a)
            no_improve = 0

        if (p + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc pass={p+1} ({elapsed:.0f}s): score={best_score:.10f}")

    # Final normalization
    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
    return best_a


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_active_qcqp(a, solver, rel_tol=1e-5, max_active=20):
    """Solve the active-set QCQP: min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()

    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]

    if len(active) > max_active:
        active = active[np.argsort(b[active])[-max_active:]]

    if len(active) == 0:
        active = np.array([np.argmax(b)])

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None
                a_val = a_val / s_sum
                return a_val
        except Exception:
            continue
    return None


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        update_best(a_lp, "lp_phase1")

    # === Phase 2: Coordinate Descent (NEW) ===
    print(f"\n=== Phase 2: Coordinate Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_cd = min(rem * 0.25, 400.0)
    if t_cd > 30:
        print(f"  Budget: {t_cd:.0f}s")
        a_cd = coordinate_descent_refine(best_seq, t_cd)
        update_best(a_cd, "coord_desc1")

    # === Phase 3: QCQP Polish ===
    print(f"\n=== Phase 3: QCQP Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_qcqp = min(rem * 0.08, 100.0)
    if t_qcqp > 15:
        a_q = solve_active_qcqp(best_seq, solver, rel_tol=1e-4, max_active=20)
        update_best(a_q, "qcqp_wide")
        a_q2 = solve_active_qcqp(best_seq, solver, rel_tol=1e-6, max_active=8)
        update_best(a_q2, "qcqp_tight")

    # === Phase 4: Coordinate Descent Round 2 ===
    print(f"\n=== Phase 4: Coord Descent Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_cd2 = min(rem * 0.15, 200.0)
    if t_cd2 > 20:
        print(f"  Budget: {t_cd2:.0f}s")
        a_cd2 = coordinate_descent_refine(best_seq, t_cd2)
        update_best(a_cd2, "coord_desc2")

    # === Phase 5: Try other n values ===
    print(f"\n=== Phase 5: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    n_candidates = [1199, 1201, 1198, 1202, 1197]
    for n_try in n_candidates:
        if time.monotonic() > deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_n = min(rem * 0.15, 150.0)
        if t_n < 30:
            continue
        print(f"  Trying n={n_try} (budget {t_n:.0f}s)")
        init_n = interpolate_sequence(best_seq, n_try)
        # Brief LP
        a_n, _ = nesterov_lp_refine(init_n, t_n * 0.5, solver, n_max_iters=100000)
        update_best(a_n, f"lp_n{n_try}")
        # Brief coord descent
        rem = deadline - time.monotonic()
        if rem > 20 and best_score < 2.0:
            t_cd_n = min(rem * 0.3, 60.0)
            a_cd_n = coordinate_descent_refine(best_seq, t_cd_n)
            update_best(a_cd_n, f"cd_n{n_try}")

    # === Phase 6: Final QCQP + Coord Descent ===
    print(f"\n=== Phase 6: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        a_fq = solve_active_qcqp(best_seq, solver, rel_tol=1e-7, max_active=5)
        update_best(a_fq, "final_qcqp")
        rem = deadline - time.monotonic()
    if rem > 15:
        a_fc = coordinate_descent_refine(best_seq, min(rem * 0.5, 30.0))
        update_best(a_fc, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
