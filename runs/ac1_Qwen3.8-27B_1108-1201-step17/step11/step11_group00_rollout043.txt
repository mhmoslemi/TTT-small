def fast_autocorr(a):

    n = len(a)

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    # Actually: (a_rev * w)[k] = sum_j a_rev[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
    # We want: grad[i] = 2 * sum_j w[i+j] * a[j]
    # Let's use: conv(a, w)[i + 2n - 2 - ... ] hmm this is getting complicated
    # 
    # Simpler: grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is the cross-correlation of a and w, evaluated at specific lags.
    # corr[i] = sum_j a[j] * w[i+j] = (a * w_reversed)[i + len(w) - 1 - (n-1)]... 
    #
    # Let me just use direct FFT approach:
    # Define v[i] = sum_j a[j] * w[i+j] for i = 0, ..., n-1
    # This is a "sliding dot product" which can be computed as:
    # v = ifft(fft(a, N) * conj(fft(w, N)))  ... no, that's cross-correlation
    #
    # Actually, v[i] = sum_j a[j] * w[i+j] is the cross-correlation of w and a:
    # v = (a * w_rev) shifted appropriately
    # where w_rev = w reversed
    #
    # Let's use: v[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # = (a convolved with w)[i + n - 1]  ... no
    #
    # conv(a, w)[k] = sum_j a[j] * w[k-j]
    # We want sum_j a[j] * w[i+j] = conv(a, w)[i + 2*... ]
    #
    # Let's use the "flip" trick:
    # sum_j a[j] * w[i+j] = conv(a[::-1], w)[(2n-2) - (i+j) + j]... 
    #
    # OK let me just be explicit:
    # v[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # = sum_{j=0}^{n-1} a[j] * w[i+j]
    #
    # Let u = a[::-1] (reversed a), so u[m] = a[n-1-m]
    # conv(u, w)[k] = sum_m u[m] * w[k-m] = sum_m a[n-1-m] * w[k-m]
    # Let j = n-1-m, so m = n-1-j:
    # = sum_j a[j] * w[k-(n-1-j)] = sum_j a[j] * w[k-n+1+j]
    #
    # We want sum_j a[j] * w[i+j], so set k-n+1 = i, i.e., k = i + n - 1
    # conv(u, w)[i + n - 1] = sum_j a[j] * w[i+j] = v[i]
    #
    # So: v = conv(a[::-1], w)[n-1 : 2n-1]  (indices n-1 to 2n-2, length n)
    
    a_rev = a[::-1].copy()
    fft_size = 1 << (len(a_rev) + len(w) - 2).bit_length()
    U = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(U * W, fft_size)
    v = conv[n-1 : 2*n-1]  # v[i] = sum_j a[j] * w[i+j]
    return 2.0 * v

def mirror_descent_lse(a0, time_limit, T_start=50.0, T_end=1.0, n_steps=5000, eta=0.5, seed=42):
    """
    Mirror descent with LogSumExp smoothing.
    a_{t+1} ∝ a_t * exp(-eta * grad(f_T)(a_t))
    where f_T(a) = (1/T) * log(sum_k exp(T * conv(a,a)[k]))
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average
    avg_a = np.zeros(n)
    n_avg = 0
    
    # Temperature schedule
    temps = np.geomspace(T_start, T_end, n_steps) if n_steps > 1 else np.array([T_start])
    
    for t in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        T = temps[min(t, len(temps)-1)]
        
        # Compute convolution
        b = fast_autocorr(a)
        
        # Compute weights
        w = compute_lse_weights(b, T)
        
        # Compute gradient
        grad = compute_lse_gradient(a, w)
        
        # Mirror descent update
        # z = log(a), z_new = z - eta * grad, a_new = exp(z_new) / sum(exp(z_new))
        log_a = np.log(a)
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        s = a_new.sum()
        if s < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        
        # Track score
        score = 2.0 * n * b.max()  # use b from before update (slightly stale but fine)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            avg_score = 2.0 * n * fast_autocorr(avg_check).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_check.copy()
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD t={t} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.1f}")
    
    return best_a, best_score

def coord_descent_polish(a0, time_limit, seed=123):
    """Coordinate descent: move mass between pairs to reduce peak."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top peaks
        top_k = np.argsort(b)[::-1][:5]
        
        improved = False
        for k in top_k:
            # Find the positions contributing to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if len(idx) == 0:
                continue
            
            # Find highest contributor
            src = idx[np.argmax(contrib)]
            
            # Find target: position with low value, far from peak
            peak_positions = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_positions.update(range(js2, je2 + 1))
            
            # Try a few random targets
            for attempt in range(20):
                dst = rng.integers(n)
                if dst == src or dst in peak_positions:
                    continue
                
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial /= a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial /= a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score

def spectral_init(n, rng):
    """Initialize using random phases in frequency domain."""
    # Generate random phases
    phases = rng.uniform(0, 2*np.pi, size=n)
    # Construct frequency-domain signal
    A = np.exp(1j * phases)
    # Make it real: ensure conjugate symmetry
    # For a real sequence of length n, we need A[n-k] = conj(A[k])
    # Use rfft-compatible approach
    A_real = np.zeros(n, dtype=complex)
    A_real[0] = 1.0  # DC component positive
    for k in range(1, (n//2) + 1):
        if k < n - k:  # avoid double-counting
            phase = rng.uniform(0, 2*np.pi)
            A_real[k] = np.exp(1j * phase)
            A_real[n-k] = np.exp(-1j * phase)
    # IDFT
    a = np.real(np.fft.ifft(A_real) * n)
    # Clamp and normalize
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def structured_init(n, rng):
    """Structured initialization."""
    choice = rng.integers(0, 8)
    if choice == 0:
        a = np.ones(n) / n
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-5*(x-0.5)**2)
    elif choice == 2:
        a = np.ones(n)
        mask = rng.random(n) > 0.5
        a[~mask] = 0
        if a.sum() < n*0.1:
            a = np.ones(n)
    elif choice == 3:
        x = np.linspace(0, 2*np.pi, n)
        a = 1 + 0.5*np.cos(3*x) + 0.3*np.cos(7*x)
    elif choice == 4:
        # Binary with controlled density
        density = rng.uniform(0.3, 0.7)
        a = (rng.random(n) < density).astype(float)
        if a.sum() < 1:
            a = np.ones(n)
    elif choice == 5:
        # Two peaks
        x = np.linspace(0, 1, n)
        c1, c2 = 0.3, 0.7
        a = np.exp(-50*(x-c1)**2) + np.exp(-50*(x-c2)**2)
    elif choice == 6:
        # Smooth random
        a = np.cumsum(rng.normal(0, 1, n))
        a -= a.min()
        a += 0.1
    else:
        # Triangle window
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - np.abs(x))
        a = a ** rng.uniform(0.5, 2)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interp_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp /= ss
    return a_interp
