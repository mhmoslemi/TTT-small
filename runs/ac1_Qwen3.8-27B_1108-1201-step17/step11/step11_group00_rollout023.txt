The Frank-Wolfe method works for minimizing a convex function over a convex set. Our objective isn't convex, but the LP subproblem (min_b max_k (a*b)_k) is. We can use FW for the LP:

Alternative: compute T @ b using FFT! Since T @ b = convolve(a, b), we can compute it in O(n log n) instead of O(n^2). But the LP solver needs the matrix explicitly...

Unless I use a first-order method for the LP instead of a black-box solver. The LP can be solved with:
- Projected gradient (Frank-Wolfe): O(n log n) per step (using FFT for T @ b)
- Interior point: needs matrix-vector products, which we can do with FFT

Actually, **Frank-Wolfe for the LP** is a great idea:

The LP is: min M s.t. (a*b)_k <= M for all k, sum b = 1, b >= 0

This can be rewritten as: min_b max_k (a*b)_k s.t. sum b = 1, b >= 0

Frank-Wolfe for this:
- Current point: b
- Linearization: find the FW direction
  - The "loss" is f(b) = max_k (a*b)_k
  - Subgradient at b: ∇f(b) = vector where ∇f[j] = a[k*-j] (k* = argmax)
  - FW direction: min_{v in simplex} <∇f, v> = min_j ∇f[j] = min_j a[k*-j]
  - So the FW vertex is e_{j*} where j* = argmin_j a[k*-j]
- Update: b_new = (1-t)*b + t*e_{j*}

This is O(n) per step! And it requires no matrix construction, no solver, just O(n) work.

But FW converges at O(1/t) rate, so we need many steps for high accuracy. However, for the purpose of getting an "approximate best response" in the Nesterov loop, this is perfect!

Let me combine:
- Use FW (or MWU) for fast approximate best responses
- Every K iterations, solve the LP exactly for a precise correction
- This gives the best of both worlds

Actually, I just realized: the MWU I described earlier IS essentially the same as FW with a specific step size schedule. The difference is:
- FW: b_new = (1-t)*b + t*e_{j*} (additive)
- MWU: b_new ∝ b * exp(-eta * grad) (multiplicative)

Both are valid approaches. MWU might have better practical performance for this problem.

import numpy as np

from scipy.fft import rfft, irfft


def fast_convolve(a, b):
    """Full convolution a*b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def mwu_best_response(a_fixed, n_steps=150, eta_start=10.0, eta_end=0.1, b_init=None):
    """Fast MWU approximation of best response b minimizing max_k (a*b)_k."""
    n = len(a_fixed)
    if b_init is not None:
        b = np.maximum(b_init, 1e-15)
        b = b / b.sum()
    else:
        b = np.ones(n) / n

    for t in range(n_steps):
        frac = t / max(1, n_steps - 1)
        eta = eta_start * (eta_end / eta_start) ** frac

        u = fast_convolve(a_fixed, b)
        k_star = int(np.argmax(u))

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        indices = np.arange(js, je + 1)
        grad = np.zeros(n)
        if len(indices) > 0:
            grad[indices] = a_fixed[k_star - indices]

        log_b = np.log(np.maximum(b, 1e-30)) - eta * grad
        log_b -= log_b.max()
        b = np.exp(log_b)
        s = b.sum()
        if s < 1e-30:
            b = np.ones(n) / n
        else:
            b = b / s

    return b

            a_new = y_new + momentum * (y_new - y)

def lse_mirror_descent(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=15):
    """Log-sum-exp optimization via mirror descent (multiplicative weights) with annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        n_inner = max(50, min(500, int(200 + 100 * si / n_stages)))
        etas = np.geomspace(stage_T * 0.1, stage_T * 0.001, n_inner)

        for t in range(n_inner):
            if time.monotonic() - t0 > time_limit:
                break
            eta = etas[t]

            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # grad[i] = 2 * sum_k w[k] * a[k-i] = 2 * (w * a)[i] (cross-correlation)
            conv = fast_convolve(w, a)
            # (w*a)[i] = sum_k w[k] a[i-k]... we need sum_k w[k] a[k-i]
            # That's the cross-correlation: sum_k w[k] a[k-i]
            # = conv(w[::-1], a)[n-1+i] ... let me just compute directly via FFT
            # Actually: sum_k w[k]*a[k-i] for k=0..2n-2, valid when 0<=k-i<n
            # Let's use: result[i] = sum_k w[k] * a[k-i]
            # This is the convolution of w with reversed a, shifted
            # Simpler: use the formula grad[i] = 2 * sum_j w[i+j] * a[j]
            # by the self-adjoint property (w and a have same length structure)
            # Actually the correct formula from lse_obj_grad:
            # grad_a[i] = 2 * sum_j w[i+j] * a[j] for valid indices
            # This equals 2 * (w * a_rev)[n-1+i] where a_rev = a[::-1]
            # Let me use FFT:
            a_rev = a[::-1].copy()
            conv_ac = fast_convolve(a_rev, w)
            grad_a = 2.0 * conv_ac[n - 1:n - 1 + n]

            # Mirror descent update
            log_a = np.log(np.maximum(a, 1e-30)) - eta * grad_a
            log_a -= log_a.max()
            a = np.exp(log_a)
            s = a.sum()
            if s < 1e-30:
                a = np.ones(n) / n
            else:
                a = a / s

        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"        LSE stage {si}: T={stage_T:.2f}, score={score:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving for final polish."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for jj in range(js2, je2 + 1):
                    peak_areas.add(jj)

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]
            dst = int(dst_pool[rng.integers(len(dst_pool))])

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = float(rng.uniform(0.003, 0.04))
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-15:
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, init_type=0):
    """Generate diverse structured initializations."""
    if init_type == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif init_type == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif init_type == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif init_type == 3:
        x = np.linspace(0, 1, n)
        a = (x * (1 - x)) ** 0.8
    elif init_type == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.25 * np.cos(3 * x) + 0.05 * np.cos(9 * x)
    elif init_type == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.2) ** 2) + 0.6 * np.exp(-5 * (x - 0.8) ** 2)
    elif init_type == 6:
        # U-shaped: high at edges
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.cos(np.pi * x) ** 2
    elif init_type == 7:
        # Spectral: flatten the spectrum
        a = np.ones(n) / n
        noise = rng.standard_normal(n)
        noise = np.convolve(noise, np.ones(20) / 20, mode='same')
        a = a + 0.1 * noise
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * x ** 2) + 0.5 * np.exp(-2 * (x - 1) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def lse_gradient_step(a, T, eta):
    """One mirror descent step for LSE objective."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    a_rev = a[::-1].copy()
    conv_ac = fast_convolve(a_rev, w)
    grad_a = 2.0 * conv_ac[n - 1:n - 1 + n]

    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad_a
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s < 1e-30:
        a_new = np.ones(n) / n
    else:
        a_new = a_new / s
    return a_new
