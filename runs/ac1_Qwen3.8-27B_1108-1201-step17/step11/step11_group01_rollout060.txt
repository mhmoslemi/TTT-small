import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_T(a_fixed):
    """Build the convolution matrix T where (T @ b)_k = (a_fixed * b)_k."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T


def build_T_fast(a_fixed):
    """Build T matrix using scipy.signal or vectorized approach."""
    n = len(a_fixed)
    size = 2 * n - 1
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n
    k_idx = np.arange(size)[:, None]  # (2n-1, 1)
    j_idx = np.arange(n)[None, :]  # (1, n)
    diff = k_idx - j_idx  # (2n-1, n)
    mask = (diff >= 0) & (diff < n)
    T = np.where(mask, a_fixed[diff], 0.0)
    return T

            prob.solve(solver=solver_name)

    return cp.PDLP


def solve_lp_br(a_fixed, T, solver):

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]

        # Try alternative solver
        alt = cp.ECOS if solver == cp.PDLP else cp.PDLP
        try:
            prob.solve(solver=alt, verbose=False)
        except Exception:
            return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=500000):
    """Nesterov-accelerated LP alternating minimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Pre-allocate
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute T for current y
        T = build_T_fast(y)
        
        # Solve LP
        b_new, M_lp = solve_lp_br(y, T, solver)
        if b_new is None:
            # Fallback: use subgradient step
            b = fast_convolve(y)
            max_b = b.max()
            active = np.where(b >= max_b * 0.9)[0]
            g = np.zeros(n)
            for k in active:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += y[k - idx]
            g /= (len(active) + 1e-15)
            b_new = g / (g.sum() + 1e-15)

        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        # Check improvement
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)

            best_score = sc_y
            best_a = y.copy()

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0 using HiGHS."""
    n = T.shape[1]

    b_val = np.maximum(b_val, 0)

    if s < 1e-15:

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_hg(a_fixed, T):
    n = T.shape[1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([T, -np.ones((T.shape[0], 1))])
    b_ub = np.zeros(T.shape[0])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    b_val = np.asarray(res.x[:n], dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def nesterov_lp(a0, time_limit, max_iters=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stagnant = 0
    last_improve = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() > _global_deadline - 45:
            break

        T = build_T(y)
        b_new = solve_lp_hg(y, T)
        if b_new is None:
            b = fast_convolve(y)
            max_b = b.max()
            if max_b < 1e-15:

            else:

                g = np.zeros(n)

                    js = max(0, k - n + 1)

                    idx = np.arange(js, je + 1)
                    g[idx] += y[k - idx]

                    b_new = g / gs

        alpha = 2.0 / (t_k + 1.0)

        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        improved = False
        if sc_a < best_score - 1e-12:
            best_score = sc_a
            best_a = a.copy()
            stagnant = 0
            improved = True

        sc_y = score_normalized(y)

            stagnant = 0

        if not improved:

            break
        if time.monotonic() - last_improve > 120 and it > 2000:

    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_objective_grad(a, T_val, n):
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_a -= S
    return obj, grad_a


def lse_lbfgs_polish(a0, time_limit, T_start=50.0, T_end=300.0, n_stages=8):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    z = z_from_a(a).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_normalized(a)

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_z(z_val):
            a_tmp = softmax_from_z(z_val)
            obj, _ = lse_objective_grad(a_tmp, T_val, n)
            return obj

        def jac_z(z_val):
            a_tmp = softmax_from_z(z_val)
            _, grad_a = lse_objective_grad(a_tmp, T_val, n)
            S = np.dot(grad_a, a_tmp)
            grad_z = a_tmp * grad_a
            return grad_z

        try:
            res = sp_minimize(
                fun=func_z, x0=z, jac=jac_z, method='L-BFGS-B',
                options={'maxiter': 50, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': 500, 'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
        except Exception:
            pass

    return best_a


def subgradient_polish(a0, time_limit, n_steps=200000):

    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() > _global_deadline - 30:
            break

        b = fast_convolve(a)
        max_b = b.max()
        active = np.where(b >= max_b * 0.92)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= (len(active) + 1e-15)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        progress = step / max(1, n_steps)
        eta = 0.3 * (0.01 / 0.3) ** progress
        z = z - eta * grad_z
        a = softmax_from_z(z)

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, style=None):

        style = rng.integers(0, 6)
    if style == 0:

        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif style == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif style == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif style == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    else:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

            a_final = lse_lbfgs_polish(best_seq, t6, T_start=200.0, T_end=1000.0, n_stages=6)
            try_update(a_final, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
