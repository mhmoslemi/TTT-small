    Nesterov dual averaging LP alternating with adaptive restarts.
    Uses warm starts and backtracking for aggressive progress.
    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:

    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = eval_local(a)

    prev_best = best_score

    full_update_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_time = time.monotonic() - lp_t0

        if b_new is None:
            continue
        B_prev = b_new

        # Nesterov step size
        alpha = 2.0 / (it + 2)

        # Update y (dual variable)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Update a (primal variable) - Nesterov accelerated
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)

            a = y.copy()

            sc_try = eval_local(a_try)

                a = a_try.copy()
                y = a_try.copy()
                full_update_count += 1

        score_a = eval_local(a)
        score_y = eval_local(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

def mirror_descent_nesterov(a0, time_limit, T_stages, lr_init=0.05, top_k_frac=0.15):
    """
    Nesterov-accelerated mirror descent (multiplicative weights) with soft active set.
    Works in probability simplex with adaptive step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()
    deadline = t0 + time_limit

    y = a.copy()  # Nesterov lookahead
    eta = lr_init
    step_count = 0
    patience = 2000
    last_improve_step = 0

    total_stages = len(T_stages)
    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break

        top_k = max(10, int(top_k_frac * (2 * n - 1)))
        steps_this_stage = max(500, int(remaining * 30000 / max(1, total_stages - stage_idx)))

        for step in range(steps_this_stage):
            if time.monotonic() > deadline:
                break

            it = step_count

            # Compute Nesterov step size
            alpha = 2.0 / (step_count + 2)

            # Compute gradient at y (lookahead point)
            b_y = fast_convolve(y)
            max_b = b_y.max()

            if T < 0.001:
                # Hard active set
                threshold = max_b * (1.0 - 1e-5)
                active = np.where(b_y >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b_y)])
                w = np.zeros(len(b_y))
                w[active] = 1.0 / len(active)
            else:
                # Soft active set with top-k
                if top_k < len(b_y):
                    idx = np.argpartition(b_y, -top_k)[-top_k:]
                    b_top = b_y[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T)
                    w_top = exp_top / (exp_top.sum() + 1e-300)
                    w = np.zeros(len(b_y))
                    w[idx] = w_top
                else:
                    max_b_all = b_y.max()
                    exp_all = np.exp((b_y - max_b_all) / T)
                    w = exp_all / (exp_all.sum() + 1e-300)

            # Gradient: g[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices
            # = 2 * correlate(w, a)[n-1:2n-1] (by symmetry)
            # Using FFT for speed
            fft_size = 1 << (2 * n - 1).bit_length()
            W = rfft(w, fft_size)
            A_rev = rfft(y[::-1], fft_size)
            g_a = irfft(W * A_rev, fft_size)[:2 * n - 1]
            g_a = g_a[n - 1:2 * n - 1]  # This gives sum_k w[k] * y[k-1-j+1]... let me verify

            # Actually, let me use np.correlate which is C-optimized
            g_a = np.correlate(w, y, mode='full')
            g_a = g_a[n - 1:2 * n - 1] * 2.0  # factor of 2 from the two terms

            # Mirror descent update (multiplicative weights)
            log_update = -eta * g_a
            log_update -= log_update.max()  # numerical stability
            a_new = a * np.exp(log_update)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            # Nesterov update for y
            y_new = (1.0 - alpha) * y + alpha * a_new

            # Evaluate
            score_new = eval_local(a_new)

            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                last_improve_step = step_count

            # Backtracking: reduce eta if no improvement for a while
            if step_count - last_improve_step > patience:
                eta *= 0.7
                patience = int(patience * 1.3)
                last_improve_step = step_count

            # Slowly increase eta if improving well
            if score_new < best_score * (1 - 1e-6):
                eta = min(eta * 1.01, 0.5)

            a = a_new
            y = y_new
            step_count += 1

            if (step + 1) % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    MirrorDescent stage={stage_idx} T={T:.3f} step={step+1} ({elapsed:.0f}s): "
                      f"best={best_score:.8f} eta={eta:.4f}")

    return best_a, best_score


def structured_init(n, rng, mode=None):
    """Create structured initializations. Returns normalized non-negative vector."""
    if mode is None:
        mode = rng.integers(0, 10)
    x = np.linspace(0, 1, n)

    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = np.cos(np.pi * x / 2) ** 2
    elif mode == 2:
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        a = np.exp(-4 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif mode == 5:
        # Gaussian mixture
        a = np.exp(-20 * (x - 0.3) ** 2) + 0.7 * np.exp(-20 * (x - 0.7) ** 2) + 0.1
    elif mode == 6:
        # Multi-frequency
        a = 0.5 + 0.2 * np.cos(2 * np.pi * 5 * x) + 0.15 * np.cos(2 * np.pi * 11 * x + 0.7)
        a = np.maximum(a, 0)
    elif mode == 7:
        # Random
        a = rng.random(n)
    elif mode == 8:
        # Flat with tapered edges
        edge = 0.1 * n
        a = np.ones(n)
        t = np.arange(edge)
        a[:edge] = 0.5 * (1 - np.cos(np.pi * t / edge))
        a[-edge:] = a[:edge][::-1]
    else:
        # Piecewise linear
        a = np.abs(np.sin(2 * np.pi * x * rng.integers(2, 5))) + 0.2
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_seq(a, rng, strength=0.02):
    """Perturb a sequence with structured noise."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 12)
    phase = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq * x + phase) + 0.5 * np.sin(2 * freq * x + phase * 1.3))
    a_pert = a + pert
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def residual_guided_perturb(a, rng, strength=0.03):
    """Perturb guided by convolution residual (target peaks)."""
    n = len(a)
    b = fast_convolve(a)
    b_norm = (b - b.min()) / (b.max() - b.min() + 1e-300)
    # Map residual back to a-space via correlation
    g = np.correlate(b_norm[:2*n-1], a, mode='full')[n-1:2*n-1]
    g_norm = (g - g.min()) / (g.max() - g.min() + 1e-300)
    # Reduce at high-residual positions, increase at low-residual
    pert = strength * (0.5 - g_norm)
    # Add some random component
    pert += 0.01 * (rng.random(n) - 0.5)
    a_pert = a + pert
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

            p_score = eval_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = eval_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Multi-start exploration at n=1200 (or parent n) ===
    print(f"\n=== Phase 1: Multi-start n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_starts = 5
    t_per_start = min(rem * 0.12 / n_starts, 60.0)
    for si in range(n_starts):
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        if si == 0:
            a_init = best_seq.copy()
        else:
            a_init = structured_init(n_best, rng, mode=si)
            # Blend with current best for informed start
            a_init = 0.3 * a_init + 0.7 * best_seq
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()

        t_sub = min(t_per_start, rem * 0.2)
        a_try, sc_try = mirror_descent_nesterov(a_init, t_sub,
                                                 [5.0, 2.0, 1.0, 0.5, 0.2, 0.1],
                                                 lr_init=0.05, top_k_frac=0.2)
        update_best(a_try, f"start_{si}")

    # === Phase 2: Main LP Alternating (primary method) ===
    print(f"\n=== Phase 2: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp1 = min(rem * 0.30, 500.0)
    if t_lp1 > 60:
        print(f"  Budget: {t_lp1:.0f}s")
        a_lp1, sc_lp1 = nesterov_alternating_lp(best_seq, t_lp1, solver, max_iters=500000)
        update_best(a_lp1, "lp_main")

    # === Phase 3: Mirror descent with full temperature schedule ===
    print(f"\n=== Phase 3: Mirror Descent Annealing (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_md = min(rem * 0.20, 300.0)
    if t_md > 30:
        print(f"  Budget: {t_md:.0f}s")
        T_sched = [3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003]
        a_md, sc_md = mirror_descent_nesterov(best_seq, t_md, T_sched,
                                               lr_init=0.04, top_k_frac=0.12)
        update_best(a_md, "md_anneal")

    # === Phase 4: Second LP round ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.15, 250.0)
    if t_lp2 > 40:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, sc_lp2 = nesterov_alternating_lp(best_seq, t_lp2, solver, max_iters=300000)
        update_best(a_lp2, "lp_2")

    # === Phase 5: Residual-guided restarts ===
    print(f"\n=== Phase 5: Residual Restarts (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_restart_total = min(rem * 0.15, 200.0)
        n_restarts = 4
        t_per_restart = t_restart_total / n_restarts
        for ri in range(n_restarts):
            rem_r = deadline - time.monotonic()
            if rem_r < 20:
                break
            strength = 0.01 + 0.01 * ri
            a_pert = residual_guided_perturb(best_seq, rng, strength=strength)
            t_sub = min(t_per_restart, rem_r * 0.3)
            # Quick LP + MD combo
            t_lp_r = t_sub * 0.5
            a_r, sc_r = nesterov_alternating_lp(a_pert, t_lp_r, solver, max_iters=50000)
            update_best(a_r, f"restart_lp_{ri}")
            rem_r = deadline - time.monotonic()
            if rem_r > 10:
                t_md_r = min(t_sub * 0.4, rem_r * 0.3)
                a_r2, sc_r2 = mirror_descent_nesterov(a_r, t_md_r,
                                                       [1.0, 0.5, 0.3, 0.1, 0.05, 0.02],
                                                       lr_init=0.03, top_k_frac=0.1)
                update_best(a_r2, f"restart_md_{ri}")

    # === Phase 6: Try n=1500 ===
    print(f"\n=== Phase 6: n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t_n1500 = min(rem * 0.18, 250.0)
        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_seq(best_seq, 1500)

        t_md_1500 = t_n1500 * 0.35
        a_1500a, sc_1500a = mirror_descent_nesterov(init_1500, t_md_1500,
                                                     [2.0, 1.0, 0.5, 0.3, 0.1, 0.05, 0.02],
                                                     lr_init=0.04, top_k_frac=0.15)
        update_best(a_1500a, "md_n1500")

        rem = deadline - time.monotonic()
        if rem > 40:
            t_lp_1500 = min(rem * 0.5, 120.0)
            a_1500b, sc_1500b = nesterov_alternating_lp(
                best_seq if len(best_seq) == 1500 else a_1500a,
                t_lp_1500, solver, max_iters=100000)
            update_best(a_1500b, "lp_n1500")

    # === Phase 7: Try n=2000 ===
    print(f"\n=== Phase 7: n=2000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_n2000 = min(rem * 0.12, 150.0)
        print(f"  Budget: {t_n2000:.0f}s")
        init_2000 = interpolate_seq(best_seq, 2000)

        t_md_2000 = t_n2000 * 0.5
        a_2000a, sc_2000a = mirror_descent_nesterov(init_2000, t_md_2000,
                                                     [3.0, 2.0, 1.0, 0.5, 0.3, 0.1, 0.05],
                                                     lr_init=0.04, top_k_frac=0.15)
        update_best(a_2000a, "md_n2000")

        rem = deadline - time.monotonic()
        if rem > 30:
            t_lp_2000 = min(rem * 0.5, 80.0)
            a_2000b, sc_2000b = nesterov_alternating_lp(
                a_2000a, t_lp_2000, solver, max_iters=50000)
            update_best(a_2000b, "lp_n2000")

    # === Phase 8: Fine polish on the best ===
    print(f"\n=== Phase 8: Fine Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_polish = min(rem * 0.35, 60.0)
        a_pol, sc_pol = fine_polish(best_seq, t_polish, solver)
        update_best(a_pol, "polish")

    # === Final verification ===
    if best_seq is not None:
        fs = eval_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
