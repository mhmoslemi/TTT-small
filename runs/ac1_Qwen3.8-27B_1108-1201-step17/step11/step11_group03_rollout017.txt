def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    return T


def solve_lp_fast(a_fixed):
    """Solve LP: min M s.t. conv(a_fixed, b)_k <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    
    if result.success:
        b_val = np.maximum(result.x[:n], 0)
        s = b_val.sum()
        if s > 1e-15:
            b_val /= s
            return b_val, float(max(0.0, result.x[n]))
        return b_val / n, None
    return None, None


def build_q_sparse(k, n):
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_qcqp_cvxpy(a_init, active_k, solver_names):
    """Solve min t s.t. a^T Q_k a <= t for k in active_k, sum(a)=1, a>=0."""
    n = len(a_init)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_k:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    for s in solver_names:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum > 1e-15:
                    a_val /= s_sum
                    return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def get_cvxpy_solvers():
    """Get list of working cvxpy solvers for QCQP."""
    working = []
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.quad_form(x, cp.eye(3))), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                working.append(s)
        except Exception:
            continue
    return working


def nesterov_lp(a0, time_limit, use_fast_lp=True):
    """Nesterov-accelerated LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a /= s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(42)
    last_timing = time.monotonic()
    iter_count = 0
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        if use_fast_lp:
            b_new, M_b = solve_lp_fast(y)
        else:
            b_new, M_b = None, None
        
        if b_new is None:
            break
        
        iter_count += 1
        alpha = 2.0 / (it + 2)
        
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y /= sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a /= sa
        
        # Compute scores
        b_a = fast_convolve(a)
        score_a = 2.0 * n * b_a.max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Also check y
        b_y = fast_convolve(y)
        score_y = 2.0 * n * b_y.max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Stall detection
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 10:
                # Perturb y
                noise = rng.normal(0, 0.02, n)
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y /= sy2
                stall_count = 0
            
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NL(it={it+1}, {elapsed:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score, iter_count


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    return fi


def eval_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def qcqp_polish(a, solvers, rel_tol_list, max_active_list):
    """Try multiple QCQP configurations."""
    best_a = a.copy()
    best_score = eval_local(a)
    n = len(a)
    
    for rel_tol in rel_tol_list:
        for max_active in max_active_list:
            b = fast_convolve(best_a)
            max_b = b.max()
            threshold = max_b * (1.0 - rel_tol)
            active = np.where(b >= threshold)[0]
            
            if len(active) > max_active:
                active = active[np.argsort(b[active])[-max_active:]]
            
            if len(active) == 0:
                active = np.array([np.argmax(b)])
            
            a_new, t_val = solve_qcqp_cvxpy(best_a, active, solvers)
            if a_new is not None:
                sc = eval_local(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new
                    print(f"    QCQP(tol={rel_tol:.0e}, active={len(active)}): {sc:.10f}")
    
    return best_a, best_score
