        max_b = b.max()

        # Active set: top few peaks

        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute gradient (average of gradients at active positions)
        g = np.zeros(n)
        count = 0
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
            count += 1
        g /= count

        # Multiplicative update
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve >= 20:
            eta *= 0.85
            no_improve = 0
            if eta < 0.001:
                break

    return best_a, best_score

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():

def solve_active_qcqp(a, solver, rel_tol=1e-5, max_active=25, time_limit=60.0):
    """
    Solve active-set QCQP with iterative constraint repair.
    Repeatedly: solve QP on active set, check all constraints, add violated ones.
    """
    n = len(a)
    t0 = time.monotonic()

    b = fast_convolve(a)
    max_b = b.max()

    threshold = max_b * (1.0 - rel_tol)
    active_list = list(np.where(b >= threshold)[0])

    if len(active_list) > max_active:
        b_vals = b[active_list]
        top_idx = np.argsort(b_vals)[-max_active:]
        active_list = [active_list[i] for i in top_idx]

    if not active_list:
        active_list = [int(np.argmax(b))]

    best_a = a.copy()
    best_t = fast_convolve(a).max()

    for round_idx in range(15):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break

        if len(active_list) > max_active:
            b_cur = fast_convolve(best_a)
            active_list.sort(key=lambda k: -b_cur[k])
            active_list = active_list[:max_active]

        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in sorted(active_list):
            Q = build_q_sparse(int(k), n)
            try:
                constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
            except Exception:
                constraints.append(cp.quad_form(a_var, Q) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        solved = False
        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
        for s in solvers_to_try:
            if s is None:
                continue
            try:
                if time.monotonic() - t0 > time_limit - 2:
                    break
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_val = np.asarray(a_var.value, dtype=np.float64)
                    a_val = np.maximum(a_val, 0)
                    s_sum = a_val.sum()
                    if s_sum < 1e-15:
                        continue
                    a_val = a_val / s_sum
                    t_val = float(max(0, t_var.value))
                    b_new = fast_convolve(a_val)
                    new_max = b_new.max()

                    if new_max < best_t:
                        best_t = new_max
                        best_a = a_val.copy()
                        b = b_new

                    max_violation = new_max - t_val
                    if max_violation < 1e-12:
                        solved = True
                        break

                    viol_indices = np.where(b_new > t_val * (1 + 1e-10) + 1e-14)[0]
                    if len(viol_indices) > 0:
                        for vi in viol_indices[:3]:
                            vi_int = int(vi)
                            if vi_int not in active_list:

                    else:
                        k_argmax = int(np.argmax(b_new))
                        if k_argmax not in active_list:
                            active_list.append(k_argmax)

                    solved = True
                    break
            except Exception:
                continue

        if solved and max_violation < 1e-10:
            break

    return best_a, float(2.0 * n * best_t)

    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    no_improve = 0
    eta = 0.04

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            g[idx] += a[k_int - idx]
        g /= len(active)

        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 25:
            eta *= 0.82
            no_improve = 0
            if eta < 0.0005:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses extended LP alternating, active-set QCQP with constraint repair,
    multiplicative weights, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.12f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.12f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.12f} (n={len(a)})")

    # === Phase 1: LP alternating (main workhorse, ~55% budget) ===
    print(f"\n=== Phase 1: LP Alternating n={n_best} (score={best_score:.12f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.55, 900.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=2000000)

    # === Phase 2: Multiplicative weights quick refine ===
