def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def spectral_flatten(a, n_iters=100, blend=0.4, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)
    
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break
        
        stages_left = n_stages - stage_i
        avail_time = remaining
        iters = max(20, min(iters_per_stage * 3, int(avail_time * 30 / max(1, stages_left))))
        
        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj
        
        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def peak_shave(a, time_limit, n_steps=100000):
    """Peak shaving with multiplicative updates and adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.03
    scale = 1.0
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Use top peaks with exponential weighting
        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:max(1, int(0.02 * n))]
        # Only include peaks within 1% of max
        active = sorted_idx[b >= max_b * 0.99]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Exponential weighting based on peak height
        w = np.exp((b[active] - max_b) / (max_b * 0.01))
        w = w / w.sum()
        
        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[i] * a[k - idx]
        
        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()

    a = rng.rand(n_test)
    a = a / a.sum()
    T_mat = build_conv_matrix(a)
    results = {}
    solver_list = [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI]
    for solver_name in solver_list:
        if solver_name is None:
            continue
        try:
            b_var = cp.Variable(n_test, nonneg=True)
            M_var = cp.Variable()
            constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            t1 = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate'):
                results[solver_name] = t1
        except Exception:
            continue
    if not results:
        return cp.ECOS
    print(f"  Solver benchmarks: {[(k, round(v, 4)) for k, v in results.items()]}")
    return min(results, key=results.get)


def solve_lp(a_fixed, solver):
    """Solve the best-response LP: min max_k (a*a_fixed)_k s.t. b on simplex."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [solver]
    if solver != 'ECOS':
        solvers.append('ECOS')
    if solver != 'SCS':
        solvers.append('SCS')
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None
                return b_val / s_val
        except Exception:
            continue
    return None


def spectral_flatten(a, n_iters=100, blend=0.4, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1.0 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    return a


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lse_func(z, n, T_temp, deadline):
    """Objective for L-BFGS (top-level)."""
    if time.monotonic() > deadline:
        return 1e15
    return lse_obj_grad(z, n, T_temp)[0]


def lse_jac(z, n, T_temp, deadline):
    """Gradient for L-BFGS (top-level)."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    return lse_obj_grad(z, n, T_temp)[1]


def lbgfs_refine(a0, time_limit, T_stages, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    if deadline is None:
        deadline = time.monotonic() + time_limit
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break
        stages_left = max(1, len(T_stages) - stage_i)
        iters = max(10, int(remaining * 15.0 / stages_left))
        try:
            res = minimize(fun=lse_func, x0=z, jac=lse_jac, method='L-BFGS-B',
                           args=(n, stage_T, deadline),
                           options={'maxiter': iters, 'ftol': 1e-28,
                                    'gtol': 1e-22, 'maxfun': max(100, iters * 80),
                                    'disp': False})
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def peak_shave(a, time_limit, n_steps=200000):
    """Peak shaving with exponential weighting and adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.025
    scale = 1.0
    no_improve = 0
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Exponential weighting over all peaks
        w = np.exp((b - max_b) / (max_b * 0.005))
        active_mask = w > 1e-6
        if not active_mask.any():
            active_mask[np.argmax(b)] = True
        w[~active_mask] = 0
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
            w_sum = 1.0
        else:
            w = w / w_sum
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[k] * a[k - idx]
        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 40:
            scale *= 0.75
            no_improve = 0
            if scale < 0.003:
                break
        if step % 30000 == 0 and step > 0:
            print(f"    PeakShave {time.monotonic()-t0:.0f}s: score={best_score:.8f}")
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    lp_times = []
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        lp_t0 = time.monotonic()
        b_new = solve_lp(y, solver)
        lp_times.append(time.monotonic() - lp_t0)
        if b_new is None:
            continue
        if len(lp_times) >= 30:
            avg_lp = np.mean(lp_times[-50:])
            if avg_lp > 5.0:
                print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping")
                break
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 10:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                noise = 0.04 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0
            if (it + 1) % 2000 == 0:
                print(f"    NesterovLP it={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    return best_a, best_score


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 12)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def structured_init(n, rng):
    """Structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / max(1, center)
        a = np.sinc(t)
        a = np.maximum(a, 0)
    else:
        a = 1.0 - np.abs(2 * x - 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multiplicative_subgrad(a, time_limit, n_steps=200000):
    """Mirror descent subgradient method with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        # Compute gradient at lookahead point y
        b = fast_convolve(y)
        max_b = b.max()
        # Use top peaks
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        # Uniform weights on active peaks
        w = np.ones(len(active)) / len(active)
        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[i] * y[k - idx]
        # Nesterov mirror descent update
        alpha = 2.0 / (it + 2)
        # Update y (lookahead)
        eta = 0.05
        log_update = -eta * g
        log_update -= log_update.max()
        y_new = y * np.exp(log_update)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        # Update a (weighted average)
        a = ((it + 1) * a + alpha * (it + 2) * y_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y_new.copy()
        else:
            a = a / sa
        y = y_new
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
    return best_a, best_score

        if best_seq is not None and len(best_seq) == nn and rem > 20:
            T_st2 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_lb_n = lbgfs_refine(best_seq, min(rem, 40.0), T_st2,
                                  deadline=time.monotonic() + min(rem, 40.0))
            local_update(a_lb_n, f"lbgfs_n{nn}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # L-BFGS final with very fine temps
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            T_fin = [3.0, 1.5, 0.8, 0.5, 0.3, 0.2, 0.1, 0.06,
                     0.04, 0.025, 0.015, 0.01, 0.006, 0.004, 0.002, 0.001]
            a_fin = lbgfs_refine(best_seq, t_fin, T_fin,
                                 deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.3, 25.0)
        if t_ps > 5:
            a_ps, _ = peak_shave(best_seq, t_ps, n_steps=100000)
            local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_flp = min(rem * 0.4, 20.0)
        if t_flp > 5:
            a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, n_max_iters=20000)
            local_update(a_flp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.10f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
