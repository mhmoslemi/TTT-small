```python
def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (css - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (css[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
from scipy.optimize import linprog

def solve_lp_scipy(y):
    n = len(y)
    size = 2 * n - 1
    
    # Variables: b_0, ..., b_{n-1}, M
    # Total: n + 1 variables
    
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: (y*b)_k <= M for all k
    # i.e., sum_j y_{k-j} * b_j - M <= 0
    A_ub = np.zeros((size, n + 1))
    for k in range(size):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = y[idx]
        A_ub[k, -1] = -1.0
    
    # Equality constraint: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_j >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=np.zeros(size),
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        b_val = result.x[:n]
        M_val = result.x[-1]
        return b_val, M_val
    return None, None
```
