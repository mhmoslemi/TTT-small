   In terms of FFT: `R_{aw} = ifft(fft(a) * conj(fft(w)))`
   
   But we need to be careful about padding and the exact definition.
   
   Let me use a different approach: note that `sum_j a[j] * w[j+m]` for `j = 0, ..., n-1` and `w` defined on `0, ..., 2n-2`.
   
   Let's define `a_pad` of length L (power of 2, >= 2n-1) with `a_pad[j] = a[j]` for j < n, 0 otherwise.
   And `w_pad` of length L with `w_pad[k] = w[k]` for k <= 2n-2, 0 otherwise.
   
   Then the auto-convolution-like sum `sum_j a[j] * w[j+m]` is:
   `= sum_j a_pad[j] * w_pad[j+m]`
   
   This is the cross-correlation at lag m. In FFT terms:
   `crosscorr[m] = ifft(fft(a_pad) * conj(fft(w_pad)))[m]`
   
   So the gradient is `4n * crosscorr[m]` for m = 0, ..., n-1.

   Let me verify with a small example. `a = [1, 2]`, n=2.
   `b = conv([1,2], [1,2]) = [1, 4, 4]` (length 3)
   `b[0] = 1, b[1] = 4, b[2] = 4`
   
   `db[0]/da[0] = 2*a[0] = 2` (since k=0, m=0, k-m=0, a[0]=1, so 2*1=2) ✓
   `db[0]/da[1] = 0` (k=0, m=1, k-m=-1, out of range) ✓
   
   `db[1]/da[0] = 2*a[1] = 4` (k=1, m=0, k-m=1, a[1]=2, so 2*2=4) ✓
   `db[1]/da[1] = 2*a[0] = 2` (k=1, m=1, k-m=0, a[0]=1, so 2*1=2) ✓
   
   `db[2]/da[0] = 2*a[2]` → k=2, m=0, k-m=2, out of range (n=2), so 0.
   
   Wait, but `b[2] = a[0]*a[2] + a[1]*a[1] + a[2]*a[0]`. Since a[2] doesn't exist (n=2), `b[2] = a[1]*a[1] = 4`.
   
   `db[2]/da[0]`: k=2, m=0, k-m=2, out of range → 0 ✓
   `db[2]/da[1]`: k=2, m=1, k-m=1, a[1]=2, so 2*2=4 ✓
   
   Now let's verify with the formula: `d(obj)/da[m] = 4n * sum_j a[j] * w[m+j]`
   
   With `w = softmax(b/T)`. For T→∞, w ≈ uniform = [1/3, 1/3, 1/3].
   
   `d(obj)/da[0] = 8 * (a[0]*w[0] + a[1]*w[1]) = 8 * (1/3 + 2/3) = 8`
   
   Direct: `d(obj)/da[0] = 2n * sum_k w[k] * db[k]/da[0] = 4 * (w[0]*2 + w[1]*4 + w[2]*0) = 4*(2/3 + 4/3) = 4*2 = 8` ✓
   
   `d(obj)/da[1] = 8 * (a[0]*w[1] + a[1]*w[2]) = 8 * (1/3 + 2/3) = 8`
   
   Direct: `d(obj)/da[1] = 4 * (w[0]*0 + w[1]*2 + w[2]*4) = 4*(2/3 + 4/3) = 8` ✓
   
   Great, the formula works!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Efficient LSE gradient via cross-correlation (FFT)
2. Adaptive temperature based on effective constraint count
3. Log-softmax parameterization for non-negativity
4. Multi-phase: parametric init → LSE descent → subgradient polish → restarts
5. Length exploration and final fine-grained search
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_obj_grad_a(a, T):
    """Compute LSE-smoothed objective and gradient w.r.t. a (NOT z).
    Assumes a >= 0 (not necessarily normalized).
    Returns (obj, grad_a).
    
    obj = 2n * LSE(b, T) where b = conv(a,a)
    grad_a[m] = 4n * sum_j a[j] * w[m+j] where w = softmax(b/T)
    """
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    w = exp_b / exp_b.sum()
    lse = b_max + T * np.log(exp_b.sum())
    obj = 2.0 * n * lse
    
    # Gradient: grad_a[m] = 4n * sum_j a[j] * w[m+j]
    # This is cross-correlation of a and w
    # Using FFT: crosscorr[m] = ifft(fft(a) * conj(fft(w)))
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    crosscorr = irfft(A * np.conj(W), fft_size)[:n]
    
    grad_a = 4.0 * n * crosscorr
    return obj, grad_a


def lse_obj_grad_z(z, T):
    """Compute LSE objective and gradient w.r.t. z (log-parameters).
    a = softmax(z), so sum(a) = 1, a >= 0.
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    obj, grad_a = lse_obj_grad_a(a, T)
    
    # Transform to z-space: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def evaluate_local(a):
    """Evaluate the true (non-smoothed) objective locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    # Normalize to sum=1 for consistent evaluation
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_gradient_descent(z0, time_limit, T_stages, n_iter_per_stage=500):
    """LSE gradient descent with temperature annealing."""
    z = z0.copy()
    t0 = time.monotonic()
    n = len(z)
    best_obj = lse_obj_grad_z(z, T_stages[0])[0] if len(T_stages) > 0 else float('inf')
    
    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        lr = 1.0 / (4.0 * n * T + 1e-10)  # Initial learning rate
        lr = min(lr, 0.5)
        
        for it in range(n_iter_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            obj, grad_z = lse_obj_grad_z(z, T)
            
            if not np.isfinite(obj) or not np.all(np.isfinite(grad_z)):
                lr *= 0.5
                continue
            
            # Backtracking line search
            alpha = 1.0
            for _ in range(20):
                z_new = z - lr * alpha * grad_z
                obj_new, _ = lse_obj_grad_z(z_new, T)
                if obj_new < obj:
                    break
                alpha *= 0.5
            else:
                break  # No improvement, try smaller lr
            
            z = z_new
            if obj_new < best_obj:
                best_obj = obj_new
            
        # Update lr for next stage
        lr = min(1.0 / (4.0 * n * T * 0.5 + 1e-10), 1.0)
    
    return z


def subgradient_polish(a0, time_limit, n_outer=10000, n_inner=50):
    """Polish using direct subgradient of the max."""
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    n = len(a)
    
    lr = 0.01
    for outer in range(n_outer):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_max = b.max()
        
        # Find active set: positions where b is within 1% of max
        threshold = b_max * 0.99
        active = np.where(b >= threshold)[0]
        
        # Subgradient: average over active positions
        # grad[m] = (1/|S|) * sum_{k in S} 2*a[k-m] * 1(0<=k-m<n)
        grad = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                grad[j] += a[k - j]
        grad /= len(active)
        
        # Gradient descent step (we want to minimize, so subtract)
        for scale in [1.0, 0.5, 0.25, 0.1]:
            a_new = a - lr * scale * grad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            break
        
        if outer % 100 == 0:
            lr *= 0.8
    
    return best_a, best_score


def lbgfs_lse(z0, time_limit, T_stages, maxiter_per_stage=1000):
    """L-BFGS with LSE smoothing and temperature annealing."""
    from scipy.optimize import minimize
    
    z = z0.copy()
    n = len(z)
    t0 = time.monotonic()
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, T)
            return obj
        
        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, T)
            return g
        
        iters = min(maxiter_per_stage, max(10, int(remaining * 20)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-18, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    x = np.linspace(0, 1, n)
    candidates = []
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Cosine families
    for k in [1, 2, 3, 4, 5, 6, 8, 10]:
        for alpha in [0.3, 0.5, 0.7, 0.9]:
            a = 1.0 + alpha * np.cos(2 * np.pi * k * x)
            a = np.maximum(a, 0)
            if a.sum() > 1e-10:
                candidates.append(a / a.sum())
    
    # Sine families
    for k in [1, 2, 3, 4]:
        for p in [0.5, 1, 2, 3]:
            a = np.abs(np.sin(np.pi * k * x)) ** p
            a = np.maximum(a, 1e-10)
            candidates.append(a / a.sum())
    
    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3, 0.4]:
            a = np.exp(-0.5 * ((x - center) / width) ** 2)
            candidates.append(a / a.sum())
    
    # Edge-peaked
    for w in [0.1, 0.2, 0.3]:
        a = np.exp(-(x / w) ** 2) + np.exp(-((1-x) / w) ** 2)
        candidates.append(a / a.sum())
    
    # Random smooth
    for _ in range(5):
        coeffs = rng.normal(0, 1, 8)
        phases = rng.uniform(0, 2*np.pi, 8)
        a = np.ones(n) * 0.5
        for i in range(8):
            a += 0.1 * coeffs[i] * np.cos(2 * np.pi * (i+1) * x + phases[i])
        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            candidates.append(a / a.sum())
    
    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturb_sequence(a, rng, strength=0.01):
    """Perturb a sequence with smooth noise."""
    n = len(a)
    # Add smooth random perturbation
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2 * np.pi, n)
    noise = strength * np.cos(freq * x + phase)
    a_new = a + noise
    return normalize(a_new)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Efficient LSE gradient via FFT cross-correlation
2. Adaptive temperature with effective-constraint-count based scheduling
3. L-BFGS optimization with aggressive low-T polishing
4. Multi-restart diversification with smooth perturbations
5. Length exploration and spectral-informed initialization
Key: correct gradient formula grad_a[m] = 4n * sum_j a[j]*w[m+j]

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_obj_grad_a(a, T):
    """Compute LSE-smoothed objective and gradient w.r.t. a.
    obj = 2n * LSE(b, T), b = conv(a,a)
    grad_a[m] = 4n * sum_j a[j] * w[m+j] where w = softmax(b/T)
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    w = exp_b / exp_b.sum()
    lse_val = b_max + T * np.log(exp_b.sum())
    obj = 2.0 * n * lse_val

    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    crosscorr = irfft(A * np.conj(W), fft_size)[:n]
    grad_a = 4.0 * n * crosscorr
    return obj, grad_a


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in log-parameterization. a = softmax(z)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    obj, grad_a = lse_obj_grad_a(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def evaluate_local(a):
    """Evaluate true objective (assumes a normalized to sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lbgfs_lse(z0, time_limit, T_stages, maxiter_per_stage=2000):
    """L-BFGS with LSE smoothing and temperature annealing."""
    z = z0.copy()
    t0 = time.monotonic()
    n = len(z)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, T)
            return obj

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, T)
            return g

        iters = min(maxiter_per_stage, max(20, int(remaining * 15)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 200,
                         'maxls': 50, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def lse_gradient_descent(z0, time_limit, T_stages, maxiter_per_stage=2000):
    """Plain gradient descent with LSE and adaptive step size."""
    z = z0.copy()
    t0 = time.monotonic()
    n = len(z)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        obj, grad_z = lse_obj_grad_z(z, T)
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            continue

        # Adaptive LR based on gradient magnitude
        lr = 1.0 / (grad_norm * np.sqrt(n) + 1e-10)
        lr = min(lr, 2.0 / max(1.0, grad_norm))

        for it in range(maxiter_per_stage):
            if time.monotonic() - t0 > time_limit:
                break

            obj, grad_z = lse_obj_grad_z(z, T)
            if not np.isfinite(obj):
                lr *= 0.5
                continue

            # Backtracking line search
            alpha = 1.0
            improved = False
            for _ in range(30):
                z_new = z - lr * alpha * grad_z
                obj_new, _ = lse_obj_grad_z(z_new, T)
                if obj_new < obj:
                    z = z_new
                    obj = obj_new
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                lr *= 0.5
                if lr < 1e-15:
                    break
            elif it % 100 == 99:
                lr = min(lr * 1.1, 2.0 / (np.linalg.norm(grad_z) + 1e-10))

    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    """Linear interpolation to change length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def smooth_perturb(a, rng, strength=0.01):

        phase = rng.uniform(0, 2 * np.pi)

    return normalize(a + noise)

        local_update(a_sub, "subgrad_fft")

    # === Phase 3: Second L-BFGS round (very low T) ===
    print(f"\n=== Phase 3: L-BFGS Low-T Refine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        T_low = [
            0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03,
            0.025, 0.02, 0.017, 0.015, 0.012, 0.01, 0.008, 0.007, 0.005,
            0.004, 0.003, 0.0025, 0.002, 0.0015, 0.001, 0.0007, 0.0005,
            0.0003, 0.0002, 0.0001
        ]
        z0 = z_from_a(best_seq)
        a_res2 = lbgfs_lse(z0, t_alloc, T_low, maxiter_per_stage=2500)
        local_update(a_res2, "lbgfs_lowT")

    # === Phase 4: Multi-restart from perturbed best ===
    print(f"\n=== Phase 4: Multi-Restart (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_restarts = min(6, max(2, int(rem * 0.5 / 50)))
        T_restart = [3, 2, 1, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003]
        for r in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            strength = 0.005 + 0.005 * r
            a_pert = smooth_perturb(best_seq, rng, strength=strength)
            z_pert = z_from_a(a_pert)
            t_sub = min(rem * 0.3, 80.0)
            a_r = lbgfs_lse(z_pert, t_sub, T_restart, maxiter_per_stage=1500)
            local_update(a_r, f"restart_{r}")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [n_best + 300, n_best - 200, int(n_best * 1.5), max(400, n_best // 2)]
    for n_try in lengths_to_try:
        if n_try < 200 or n_try > 4000 or n_try == n_best:
            continue
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        init = interpolate_sequence(best_seq, n_try)
        T_med = [5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]
        z_init = z_from_a(init)
        t_sub = min(rem * 0.2, 60.0)
        a_try = lbgfs_lse(z_init, t_sub, T_med, maxiter_per_stage=1200)
        local_update(a_try, f"length_{n_try}")

    # === Phase 6: Final ultra-low-T polishing ===
    print(f"\n=== Phase 6: Final Ultra-Low-T Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        T_final = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006, 0.005, 0.004,
                   0.003, 0.0025, 0.002, 0.0015, 0.001, 0.0008, 0.0006, 0.0005,
                   0.0004, 0.0003, 0.0002, 0.0001]
        z0 = z_from_a(best_seq)
        a_fin = lbgfs_lse(z0, min(rem * 0.7, 150.0), T_final, maxiter_per_stage=3000)
        local_update(a_fin, "final_ultra_lowT")

    # === Phase 7: One more subgradient pass ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_s2, _ = subgradient_polish_fft(best_seq, min(rem * 0.5, 60.0))
        local_update(a_s2, "final_subgrad")

    # Final check
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
