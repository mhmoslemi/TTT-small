"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP alternating with Nesterov acceleration (main workhorse)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_rows(a, k_indices):
    """Build rows of the convolution matrix for specific k indices.
    Returns a sparse matrix T where T @ b = [(a*b)[k] for k in k_indices]
    """
    n = len(a)
    T = lil_matrix((len(k_indices), n))
    for i, k in enumerate(k_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[i, j] = a[k - j]
    return csc_matrix(T)


def build_full_conv_matrix(a):
    """Build full (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    results = {}
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                results[solver_name] = time.monotonic() - t0
        except Exception:
            continue
    if results:
        return min(results, key=results.get)
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, full_matrix=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    if full_matrix is None:
        full_matrix = build_full_conv_matrix(a_fixed)
    
    constraints = [full_matrix @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None, full_matrix
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value)), full_matrix
        except Exception:
            continue
    
    return None, None, full_matrix


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_active_qcqp(a_init, solver, rel_tol=1e-5, max_active=30, time_budget=30.0):
    """
    Solve the active-set QCQP with dynamic constraint management.
    min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0
    """
    n = len(a_init)
    t0 = time.monotonic()
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    b = fast_convolve(a)
    max_b = b.max()
    
    # Start with top active constraints
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]
    if len(active) > max_active:
        active = active[np.argsort(b[active])[-max_active:]]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    # Active set iteration
    for epoch in range(50):
        if time.monotonic() - t0 > time_budget:
            break
        
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]
        
        for k in active:
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        solved = False
        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
        for s in solvers_to_try:
            if s is None:
                continue
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_val = np.asarray(a_var.value, dtype=np.float64)
                    a_val = np.maximum(a_val, 0)
                    s_sum = a_val.sum()
                    if s_sum < 1e-15:
                        break
                    a_val = a_val / s_sum
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            break
        
        a = a_val
        
        # Check if any constraint is violated
        b_new = fast_convolve(a)
        t_val = b_new.max()
        
        # Find violated constraints (not in active set)
        violated = np.where((b_new > t_val * (1 + 1e-8)) & (np.arange(len(b_new))[:, None] != active[None, :])).flatten()
        
        # Actually, let's check all constraints against the optimal t
        all_violated = np.where(b_new > t_val + 1e-10)[0]
        new_violated = np.setdiff1d(all_violated, active)
        
        if len(new_violated) == 0:
            # Check if we can remove any active constraints
            # (a constraint is not active if b[k] < t_val - epsilon)
            epsilon = 1e-8
            not_binding = np.where(b_new[active] < t_val - epsilon)[0]
            if len(not_binding) > 0:
                active = np.delete(active, not_binding)
            
            break
        
        # Add the most violated constraint
        # Limit active set size
        active = np.union1d(active, new_violated)
        if len(active) > max_active:
            # Keep the most constraining ones
            b_active = b_new[active]
            keep = active[np.argsort(b_active)[-max_active:]]
            active = keep
    
    return a, float(fast_convolve(a).max())


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    # Pre-build the convolution matrix (it changes with y, so we can't really pre-build)
    # Actually, the matrix T depends on the FIXED sequence (y), not on the variable (b)
    # So we need to rebuild it each iteration. This is the bottleneck.
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    
    # Use a simpler approach: build the matrix each time
    # For n=1200, building the (2n-1) x n matrix takes O(n^2) which is about 1.4M operations
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP best response for fixed y
        T = build_full_conv_matrix(y)
        b_new, M_b, _ = solve_lp_best_response(y, solver, full_matrix=T)
        
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 5:
                # Perturb
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.05 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0
            
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")
        
        # Stop if LP is too slow
        if (it + 1) % 100 == 0 and (it + 1) >= 100:
            elapsed = time.monotonic() - t0
            avg_time = elapsed / (it + 1)
            if avg_time * 1000 > 50:  # 50ms avg per iteration
                print(f"    LP too slow ({avg_time*1000:.1f}ms/iter avg), stopping at iter {it+1}")
                break
    
    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lse_func(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_temp)
    return obj


def lse_jac(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_temp)
    return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    from scipy.optimize import minimize
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac,
                args=(n, stage_T, deadline),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def multiplicative_subgradient(a0, time_limit, n_steps):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta_base = 0.03
    no_improve = 0
    scale = 1.0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 30:
            scale *= 0.8
            no_improve = 0
            if scale < 0.005:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def projected_gradient_step(a, grad, eta):
    """Take a gradient step and project onto the simplex."""
    v = a - eta * grad
    # Project v onto {x >= 0, sum(x) = 1}
    # Binary search for threshold tau such that sum(max(v - tau, 0)) = 1
    lo = v.min() - 1
    hi = v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s_val = np.maximum(v - mid, 0).sum()
        if s_val > 1:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - (lo + hi) / 2, 0)


def ogd_refine(a0, time_limit, n_steps=100000):
    """Optimistic gradient descent with projection onto simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.1
    momentum = 0.9
    v = np.zeros(n)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Subgradient: average over active constraints
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)
        
        # Adjust for the constraint sum(a) = 1
        # The gradient of the constraint is (1, 1, ..., 1)
        # We project, so we don't need to adjust explicitly
        
        # Nesterov momentum
        v = momentum * v + (1 - momentum) * grad
        a_new = a - eta * v
        a_new = projected_gradient_step(a, v, eta)
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 0.5)
        else:
            eta = max(eta * 0.95, 0.001)
    
    return best_a, best_score
