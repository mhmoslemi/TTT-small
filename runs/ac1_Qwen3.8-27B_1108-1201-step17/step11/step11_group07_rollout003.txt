    # In the circular case with N large enough:
    # FFT(cross_va)[k] = ? 
    # cross_va[m] = sum_j a[j] * v[m+j]
    # This is the cross-correlation of v and a with v "shifted":
    # = (v ⊛ a)[m] where ⊛ is defined as (v ⊛ a)[m] = sum_j v[j+m] * a[j]
    # Standard: (x ⊛ y)[m] = sum_n x[n+m] * conj(y[n]) = IFFT(FFT(x) * conj(FFT(y)))
    # So (v ⊛ a)[m] = IFFT(W * conj(A))[m]
    
    R = irfft(W * np.conj(A), N)
    cross = R[:n]  # cross[m] = sum_j w[m+j] * a[j] for m=0..n-1
    
    grad = 2.0 * cross
    return obj, grad

def lse_obj_and_grad_z(z, n, T):
    """LSE objective and gradient in log-parameterization (softmax)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    obj, grad_a = lse_obj_and_grad(a, T)
    
    # Chain rule: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto the simplex {x : x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Scale to make sum = target_sum after projection to sum=1
    v = v / target_sum if target_sum != 1.0 else v.copy()
    # Project onto {x >= 0, sum(x) = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(v - theta, 0)
    return proj * target_sum

def subgradient_step(a, eta, n_active=None):
    """One subgradient step for the max convolution objective.
    Returns new a and the subgradient used.
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    
    # Find active set (peaks within threshold of max)
    threshold = c_max * (1 - 1e-6)
    if n_active is not None:
        # Use top n_active peaks
        idx = np.argsort(c)[::-1][:n_active]
    else:
        idx = np.where(c >= threshold)[0]
    
    if len(idx) == 0:
        idx = np.array([np.argmax(c)])
    
    # Subgradient: average of subgradients at each active peak
    # For peak at position k: grad_k[m] = 2*a[k-m] (if 0 <= k-m <= n-1)
    g = np.zeros(n)
    for k in idx:
        # grad_k[m] = 2*a[k-m] for m such that 0 <= k-m <= n-1
        # i.e., m in [k-(n-1), k] intersect [0, n-1]
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_end >= m_start:
            g[m_start:m_end+1] += 2.0 * a[k - np.arange(m_end + 1 - m_start)][::-1]
            # Wait, let me be more careful
            # For m in [m_start, m_end]: grad_k[m] = 2 * a[k-m]
            # So g[m] += 2 * a[k-m]
            for m in range(m_start, m_end + 1):
                g[m] += 2.0 * a[k - m]
    
    g /= len(idx)
    
    # Projected gradient step
    a_new = a - eta * g
    a_new = project_simplex(a_new, 1.0)
    
    return a_new, g, c_max

def subgradient_optimize(a0, time_limit, eta0=0.01, lb=1.0):
    """Subgradient method with Polyak step size."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_new, g, c_max = subgradient_step(a, eta)
        sc = evaluate_local(a_new)
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            # Polyak step: eta = (f - f*) / ||g||^2
            g_norm_sq = np.dot(g, g)
            if g_norm_sq > 1e-20:
                eta_polyak = (sc - lb) / g_norm_sq
                eta = min(eta, max(eta_polyak, eta * 0.9))
        else:
            eta *= 0.8
            if eta < 1e-10:
                break
        
        if step % 1000 == 0:
            print(f"    SubGrad step={step}: score={best_score:.8f}, eta={eta:.6f}")
    
    return best_a, best_score

def mirror_descent_optimize(a0, time_limit, eta0=0.05, n_stages=None):
    """Mirror descent (exponential weights) with LSE smoothing."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    if n_stages is None:
        n_stages = 30
    
    T_schedule = np.exp(np.linspace(np.log(50), np.log(0.001), n_stages))
    eta = eta0
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Determine iterations for this stage
        iter_budget = int(remaining * 0.15)
        iter_budget = max(iter_budget, 50)
        
        for it in range(iter_budget):
            if time.monotonic() - t0 > time_limit:
                break
            
            _, grad = lse_obj_and_grad(a, T)
            
            # Mirror descent step: a_new = a * exp(-eta * grad) / sum
            log_factor = -eta * grad
            log_factor -= log_factor.max()
            a_new = a * np.exp(log_factor)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                eta *= 0.5
                continue
            a_new = a_new / s
            
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                eta = min(eta * 1.05, 0.5)  # slight increase if improving
            else:
                eta *= 0.7
        
    return best_a, best_score

def lse_lbfgs_optimize(a0, time_limit, T_stages=None):
    """L-BFGS optimization with LSE smoothing."""
    if T_stages is None:
        T_stages = [50, 20, 10, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0007, 0.0005]
    
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z_in):
            obj, _ = lse_obj_and_grad_z(z_in, n, stage_T)
            return obj
        
        def jac(z_in):
            _, g = lse_obj_and_grad_z(z_in, n, stage_T)
            return g
        
        # Scale iterations based on remaining time
        iters = min(5000, max(20, int(remaining * 0.2)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-18, 'maxfun': iters * 500, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS error at T={stage_T}: {e}")
    
    return softmax_from_z(z)

def simulated_annealing(a0, time_limit, T_init=0.1, T_final=0.001, n_iter=None):
    """Simulated annealing for escaping local minima."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(789)
    
    if n_iter is None:
        n_iter = 50000
    
    T = T_init
    decay = (T_final / T_init) ** (1.0 / max(n_iter, 1))
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Random perturbation
        strength = 0.01 * T / T_init
        perturbation = rng.normal(0, strength, n)
        a_perturbed = a + perturbation
        a_perturbed = project_simplex(a_perturbed, 1.0)
        
        sc = evaluate_local(a_perturbed)
        current_score = evaluate_local(a)
        
        if sc < best_score:
            best_score = sc
            best_a = a_perturbed.copy()
            a = a_perturbed.copy()
        else:
            # Metropolis acceptance
            delta = sc - current_score
            if rng.uniform() < np.exp(-delta / (T + 1e-10)):
                a = a_perturbed.copy()
        
        T *= decay
    
    return best_a, best_score

def spectral_flatten(a0, time_limit, n_iters=5000):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    lr = 0.01
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_target = P.mean()
        
        # Direction to flatten P
        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-12
        H[valid] = -(P[valid] - P_target) * np.conj(A[valid]) / (P[valid] + 1e-10)
        H[0] = 0.0
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h /= h_norm
        
        # Line search
        improved = False
        for scale in [lr, lr * 0.5, lr * 0.1, lr * 0.01]:
            a_new = a + scale * h
            a_new = project_simplex(a_new, 1.0)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break
        if not improved:
            lr *= 0.7
            if lr < 1e-8:
                break
        elif it % 500 == 0:
            lr = min(lr * 1.1, 0.05)
    
    return best_a, best_score

def fourier_init(n, rng, order=8):
    """Initialize using Fourier series: a[i] = (1/n)(1 + sum c_k cos(...))"""
    x = np.arange(n) / n
    a = np.ones(n)
    for k in range(1, order + 1):
        c = rng.normal(0, 0.3 / k)
        phase = rng.uniform(0, 2 * np.pi)
        a += c * np.cos(2 * np.pi * k * x + phase)
    a = np.maximum(a, 0.01)
    return normalize(a)

def diverse_initializations(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Gaussian
    a = np.exp(-5 * (x - 0.5) ** 2)
    candidates.append(normalize(a))
    
    # Cosine
    a = 0.5 + 0.4 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.01)
    candidates.append(normalize(a))
    
    # Sine squared
    a = np.sin(np.pi * x) ** 2
    candidates.append(normalize(a))
    
    # Two-bump
    a = np.exp(-15 * (x - 0.3) ** 2) + np.exp(-15 * (x - 0.7) ** 2)
    candidates.append(normalize(a))
    
    # Edge-peaked
    a = np.exp(-8 * x) + np.exp(-8 * (1 - x))
    candidates.append(normalize(a))
    
    # Fourier-based
    for _ in range(3):
        candidates.append(fourier_init(n, rng, order=rng.integers(4, 12)))
    
    # Random smooth
    freqs = rng.integers(1, 10, 6)
    coeffs = rng.normal(0, 0.5, 6)
    phases = rng.uniform(0, 2 * np.pi, 6)
    a = np.ones(n)
    for f, c, p in zip(freqs, coeffs, phases):
        a += c * np.cos(2 * np.pi * f * x + p)
    a = np.maximum(a, 0.01)
    candidates.append(normalize(a))
    
    return candidates

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.001)
    return normalize(f_interp)

def block_refine(a, time_limit, block_size=30, n_rounds=10000):
    """Refine by optimizing blocks of consecutive elements."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(321)
    
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Pick a block
        start = rng.integers(0, n - block_size + 1)
        end = start + block_size
        
        # Get gradient for the block
        _, grad = lse_obj_and_grad(a, T=0.01)
        g_block = grad[start:end]
        
        # Try to reduce the gradient in this block
        # Simple: push block elements down where gradient is high
        lr = 0.001 / (np.max(np.abs(g_block)) + 1e-10)
        for scale in [1.0, 0.5, 0.1]:
            a_new = a.copy()
            a_new[start:end] -= lr * scale * g_block
            a_new = project_simplex(a_new, 1.0)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                break
    
    return best_a, best_score
