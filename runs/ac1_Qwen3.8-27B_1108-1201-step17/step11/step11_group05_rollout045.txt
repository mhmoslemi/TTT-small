Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP alternating minimization with warm-started dual variables

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Return a function that computes T @ b where T is the conv matrix."""
    n = len(a)
    def matvec(b):
        return np.convolve(a, b)
    return matvec


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_primal(a, solver=None, y_warm=None):
    """
    Given fixed a (non-negative, normalized to sum=1), solve:
    minimize M
    subject to: conv(a, b)[k] <= M for all k
                sum(b) = 1
                b >= 0
    Returns (b, M, y_dual) where y_dual is the dual variable on the conv constraints.
    """
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Use the conv matrix
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    y_var = cp.Variable(size, nonneg=True)
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    # Try solvers
    solvers = [solver, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    if solvers[0] is None:
        solvers = solvers[1:]
    
    last_err = None
    for s in solvers:
        try:

def spectral_equalize(a, alpha=0.1):
    """Flatten the magnitude spectrum to reduce autocorrelation peak."""
    n = len(a)
    fft_size = 1 << (2*n - 1).bit_length()
    A = rfft(a, fft_size)
    mag = np.abs(A)
    # Smooth the magnitude spectrum
    phase = np.angle(A)
    # Flatten: new_mag = mag^(1-alpha) * mag_max^alpha (push towards uniform)
    mag_max = mag.max()
    if mag_max < 1e-15:
        return a.copy()
    new_mag = mag ** (1 - alpha) * mag_max ** alpha
    new_A = new_mag * np.exp(1j * phase)
    a_new = irfft(new_A, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

                a_new = a.copy()

Hmm, the coordinate descent is going to be too slow because each iteration requires an FFT convolution. For n=1200, each FFT is about O(n log n) ≈ 12000 operations, and we need n=1200 coordinates per pass. So each pass is about 1200 * 12000 = 1.4e7 operations, plus the overhead of numpy. That's maybe 1-2 seconds per pass. With 100 passes, that's 100-200 seconds. But we also need to evaluate the objective at each step, which adds more FFTs.

Actually for coordinate descent, I can avoid the full FFT by noting that changing a[i] only affects the convolution at positions k where k-i is in [0, n-1], i.e., k in [i, i+n-1]. That's n out of 2n-1 positions. And the change is just 2*d*a[k-i] (plus d^2 at k=2i). So I can update the convolution in O(n) time without FFT!
