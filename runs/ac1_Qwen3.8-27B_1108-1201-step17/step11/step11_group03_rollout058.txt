"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP alternating (primary convergence driver)
2. KKT-based Newton refinement (exploits active set structure for high precision)

4. Multi-length exploration (n = 1198-1202)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def Qk_a(k, a, n):
    """Compute (Q_k @ a) efficiently without forming Q_k."""
    result = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        l = k - j
        if j == l:
            result[j] = a[j]
        else:
            result[j] = 0.5 * a[l]
    return result


def build_A_dense(lam, active_set, n):
    """Build A = 2 * sum_i lam_i * Q_{k_i} as dense matrix."""
    A = np.zeros((n, n))
    for i, k in enumerate(active_set):
        if abs(lam[i]) < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        c = 2.0 * lam[i]
        for j in range(js, je + 1):
            l = k - j
            if j == l:
                A[j, j] += c
            else:
                A[j, l] += 0.5 * c
                A[l, j] += 0.5 * c
    return A


def get_solver():
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def solve_lp(a_fixed, solver):
    """Solve min_b max_k (a_fixed * b)[k] s.t. sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build constraint matrix: convolve(a_fixed, b)[:2n-1] <= M
    # Use cvxpy's convolve or build manually
    # For efficiency, build the Toeplitz matrix
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for s in [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None
                return b_val / s_sum
        except Exception:
            continue
    return None


def nesterov_lp_refine(a0, time_limit, solver, max_iters=500000):

    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Nesterov extrapolation
        a_new = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a_new / sa
        
        score = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

                stall = 0
            else:
                stall += 1
            prev_best = best_score

                noise = 0.02 * np.random.randn(n)

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score


def kkt_newton_solve(a0, active_set, n, max_iter=100, tol=1e-14):
    """Solve KKT system using Newton's method."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, None
    a = a / s
    
    s_active = len(active_set)
    if s_active == 0:
        return a, None
    
    lam = np.ones(s_active) / s_active
    mu = 0.0
    conv = fast_convolve(a)
    t = float(conv[max(active_set)])
    
    for it in range(max_iter):
        # Compute Q_k @ a for each active k
        Qa = np.array([Qk_a(k, a, n) for k in active_set])  # shape (s, n)
        
        # Compute a^T Q_k a for each active k
        conv = fast_convolve(a)
        active_vals = conv[list(active_set)]
        
        # Residuals
        F0 = lam.sum() - 1.0
        F1 = 2.0 * (Qa.T @ lam) + mu  # n-vector
        F2 = active_vals - t  # s-vector
        F3 = a.sum() - 1.0
        
        resid = max(abs(F0), float(np.max(np.abs(F1))), float(np.max(np.abs(F2))), abs(F3))
        if resid < tol:
            break
        
        # Build Jacobian
        N = n + s_active + 2
        J = np.zeros((N, N))
        
        # Row 0: d(sum(lam)-1)/d(lam) = 1
        J[0, n:n+s_active] = 1.0
        
        # Rows 1..n: dF1
        A_mat = build_A_dense(lam, active_set, n)
        J[1:1+n, 0:n] = A_mat
        J[1:1+n, n:n+s_active] = 2.0 * Qa.T  # dF1/dlam_i = 2*(Q_{k_i} a)_j
        J[1:1+n, n+s_active] = 1.0  # dF1/dmu = 1
        
        # Rows n+1..n+s: dF2
        J[1+n:1+n+s_active, 0:n] = 2.0 * Qa  # dF2_i/da_j = 2*(Q_{k_i} a)_j
        J[1+n:1+n+s_active, n+s_active+1:n+s_active+2] = -np.eye(s_active)  # dF2/dt = -I
        
        # Row n+s+1: dF3
        J[N-1, 0:n] = 1.0
        
        # RHS
        rhs = np.zeros(N)
        rhs[0] = -F0
        rhs[1:1+n] = -F1
        rhs[1+n:1+n+s_active] = -F2
        rhs[N-1] = -F3
        
        # Solve
        try:
            delta = np.linalg.solve(J, rhs)
        except np.linalg.LinAlgError:
            try:
                delta = np.linalg.lstsq(J, rhs, rcond=None)[0]
            except Exception:
                break
        
        # Line search for a >= 0
        step = 1.0
        for _ in range(30):
            a_trial = a + step * delta[0:n]
            if a_trial.min() >= 0:
                break
            step *= 0.5
        
        a_new = np.maximum(a + step * delta[0:n], 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        lam_new = np.maximum(lam + step * delta[n:n+s_active], 0)
        mu_new = mu + step * delta[n+s_active]
        t_new = t + step * delta[n+s_active+1]
        
        # Check if lambda sum is approximately 1 (it should be from the KKT)
        # Renormalize lambda
        lam_sum = lam_new.sum()
        if lam_sum > 1e-15:
            lam_new = lam_new / lam_sum
        
        a = a_new
        lam = lam_new
        mu = mu_new
        t = max(t_new, 0)
        
        if it % 10 == 0:
            score = 2.0 * n * fast_convolve(a).max()
            print(f"    KKT iter={it}: score={score:.12f}, resid={resid:.2e}")
    
    return a, lam


def active_set_refine(a0, n, time_limit=60.0):
    """Iteratively refine using KKT with active set management."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    for outer in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Identify active set: positions at 99.9% of max
        threshold = max_val * (1 - 1e-6)
        active = np.where(conv >= threshold)[0]
        
        # Limit active set size
        if len(active) > 10:
            # Keep top 10
            active = active[np.argsort(conv[active])[-10:]]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Solve KKT
        a_new, lam = kkt_newton_solve(a, active, n, max_iter=50, tol=1e-13)
        
        if a_new is not None:
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
        
        # Check KKT multipliers: remove constraints with lam < 0
        if lam is not None:
            # Remove any constraint with negative multiplier (shouldn't happen with projection)
            # Check if any non-active constraint is violated
            conv_new = fast_convolve(a)
            non_active = np.setdiff1d(np.arange(len(conv_new)), active)
            if len(non_active) > 0:
                max_violation = conv_new[non_active].max()
                if max_violation > conv_new[active].min() * (1 + 1e-8):
                    # Add the most violated constraint
                    k_new = non_active[np.argmax(conv_new[non_active])]
                    active = np.append(active, k_new)
                    # Re-solve
                    a, lam = kkt_newton_solve(a, active, n, max_iter=50, tol=1e-13)
                    if a is not None:
                        score = 2.0 * n * fast_convolve(a).max()
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
        
        # Check convergence
        conv = fast_convolve(a)
        max_b = conv.max()
        second_max = np.sort(conv)[-2] if len(conv) > 1 else 0
        if max_b - second_max < 1e-10 * max_b:
            # Many constraints are active, should be close to optimal
            pass
    
    return best_a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fn = np.interp(new_x, old_x, f_n)
    fn = np.maximum(fn, 0)
    ss = fn.sum()
    if ss > 1e-15:
        fn = fn / ss
    return fn


def eval_local(a):

    s = a.sum()

    return float(2.0 * n * fast_convolve(a).max() / (s * s))
