import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import math

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    if n != m:
        raise ValueError("Must be same length")
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def get_lp_solver():
    """Find a working LP solver, preferring exact ones."""
    for solver_name in [cp.GLPK, cp.CBC, cp.SCS, cp.PDLP, cp.CLARABEL, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.GLPK, cp.CBC, cp.SCS, cp.PDLP, cp.CLARABEL]
    result = None
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                result = (np.asarray(b_var.value, dtype=np.float64), float(M_var.value))
                break
        except Exception:
            continue

    if result is None:
        return None, None

    b_val, M_val = result
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, max(0.0, M_val)

def edge_concentrated_init(n, rng, alpha=2.0):
    """Create U-shaped initialization that concentrates mass at edges."""
    x = np.linspace(0, 1, n)
    # U-shaped: high at edges, low in middle
    a = 0.5 * (x**alpha + (1-x)**alpha) + 0.05 * np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def inf_lp_alternate(a0, time_limit, solver, n_max_iters=1000000):
    """
    Infimal projection LP alternating with Nesterov dual averaging.
    More careful implementation with exact solver.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    
    # Use a slower, more robust step size
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP best response for y
        b_new, M_val = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Nesterov x update
        a_new_numerator = (it + 1) * a + alpha * (it + 2) * b_new
        a_new_denom = (it + 1) + alpha * (it + 2)
        a = np.maximum(a_new_numerator / a_new_denom, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate both a and y
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 30:
                # Perturb to escape
                noise = rng.normal(0, 0.02, n)
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LPA iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M={M_val:.8f}")

    return best_a, best_score

def active_set_qp(a0, time_limit, n_max_iters=5000):
    """
    Active-set method: identify active constraints, solve QP.
    minimize a^T Q a subject to a >= 0, sum(a) = 1
    where Q is built from active constraints.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Identify active set (constraints within epsilon of max)
        eps = max(1e-6, 0.01 * max_b / n)
        active = np.where(b >= max_b - eps)[0]

        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Build Q matrix for active constraints
        # Q = sum_{k in active} (1/|active|) * T_k^T T_k
        # where T_k is the k-th row of convolution matrix
        # T_k[i] = a[k-i] if 0 <= k-i < n, else 0
        # So Q[i,j] = sum_k (1/|active|) * a[k-i] * a[k-j] (with bounds)
        # This is (1/|active|) * sum_k a[k-i]*a[k-j] over active k
        
        # Actually, the QP is: min a^T Q a where Q[i,j] = (1/|S|) * sum_{k in S} T_k[i] * T_k[j]
        # T_k[i] = a[k-i] if valid
        # Hmm, but a is what we're optimizing, so this is circular.
        # 
        # Let me reconsider. The objective is max_k sum_i sum_j a_i a_j T_k[i,j_div]
        # Actually conv(a,a)_k = sum_i a_i * a[k-i] (with bounds 0<=i<n, 0<=k-i<n)
        # = sum_i a_i^2 * [k-i in valid range... no]
        # conv(a,a)_k = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a_i * a_{k-i}
        # This is quadratic in a: a^T M_k a where (M_k)[i,j] = 1 if i+j=k, else 0
        # Wait no. conv(a,a)_k = sum_i a_i * a_{k-i}. This is NOT a^T M_k a in general.
        # It IS a^T M_k a where M_k is the matrix with (M_k)[i,j] = 1 if i+j = k.
        # Because a^T M_k a = sum_{i,j} a_i M_k[i,j] a_j = sum_i a_i * a_{k-i} (when i+j=k)
        # Yes! So conv(a,a)_k = a^T M_k a where M_k[i,j] = 1 if i+j=k else 0.
        # M_k is a permutation-like matrix (anti-diagonal strip).
        
        # So the QP is: min (1/|S|) * a^T (sum_{k in S} M_k) a s.t. a>=0, sum(a)=1
        # Let Q = (1/|S|) * sum_{k in S} M_k
        # Q[i,j] = (number of k in S with i+j=k) / |S|
        # = (1 if i+j in S) / |S|   [since i+j is unique for given i,j]
        
        # So Q[i,j] = 1/|S| if i+j in active set, else 0.
        # This is a very sparse matrix!
        
        # Build Q implicitly and use a QP solver
        Q = np.zeros((n, n))
        active_set = set(active.tolist())
        for i in range(n):
            for j in range(n):
                if (i + j) in active_set:
                    Q[i, j] = 1.0 / len(active)
        
        # Solve QP: min a^T Q a s.t. a >= 0, sum(a) = 1
        # Use cvxpy
        a_var = cp.Variable(n, nonneg=True)
        obj = cp.quad_form(a_var, Q, assume_PSD=True)
        constraints = [cp.sum(a_var) == 1]
        prob = cp.Problem(cp.Minimize(obj), constraints)
        
        solved = False
        for s in [solver, cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate') and a_var.value is not None:
                    a_new = np.asarray(a_var.value, dtype=np.float64)
                    a_new = np.maximum(a_new, 0)
                    sn = a_new.sum()
                    if sn > 1e-15:
                        a_new = a_new / sn
                    score = evaluate_local(a_new)
                    if score < best_score:
                        best_score = score
                        best_a = a_new.copy()
                        a = a_new
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            # Fallback: subgradient step
            g = np.zeros(n)
            for k in active:

def nesterov_proximal(a0, time_limit, n_steps=1000000):
    """
    Nesterov-accelerated proximal gradient for the constrained problem.
    min f(a) = max_k a^T M_k a  subject to a >= 0, sum(a) = 1
    
    Uses LogSumExp surrogate with adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    # LogSumExp approach
    T = 5.0  # Initial temperature
    T_min = 0.001
    T_decay = 0.9995
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Compute gradient of LogSumExp surrogate
        # LSE = max_b + T * log(sum(exp((b_k - max_b)/T)))
        # dLSE/dT_k = (1/T) * (b_k - max_b) * exp((b_k-max_b)/T) / sum(exp((b_k-max_b)/T))
        # = (b_k - max_b) * w_k / T
        # where w_k = exp((b_k-max_b)/T) / sum(exp(...))
        
        shifted = (b - max_b) / T
        # Clip to avoid overflow
        shifted = np.maximum(shifted, -50)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()
        
        # Gradient of LSE w.r.t. a:
        # d(a^T M_k a)/da = 2 * M_k @ a (since M_k is symmetric)
        # M_k @ a has element i: sum_j M_k[i,j] a_j = a[k-i] (if valid)
        # So d(a^T M_k a)/da_i = 2 * a[k-i] (with bounds)
        # Total gradient: sum_k w_k * 2 * (M_k @ a)
        # = 2 * sum_k w_k * a[k-i] for valid k-i
        
        # This is 2 * correlate(w, a) at appropriate lags
        # Specifically: grad_a[i] = 2 * sum_k w[k] * a[k-i] (where 0 <= k-i < n)
        
        # Use np.correlate
        # np.correlate(w, a, 'full')[n-1:2n-1] gives sum_k w[k+i] * a[k] 
        # We want sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] = correlate(w, a) at lag i
        
        g_a = np.correlate(w, a, mode='full')[n-1:2*n-1] * 2.0
        
        # Project gradient onto the tangent space of the simplex
        # (remove component along the all-ones vector)
        g_a = g_a - g_a.mean()
        
        # Adaptive step size
        eta = 0.1 / (np.linalg.norm(g_a) + 1e-8)
        eta = min(eta, 0.5)
        
        # Multiplicative update (mirror descent on simplex)
        a_new = a * np.exp(-eta * g_a)
        # Better: use additive with projection
        a_new = a - eta * g_a
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn > 1e-15:
            a_new = a_new / sn
        
        score_new = evaluate_local(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new
        
        # Decay temperature
        if step > 100:
            T *= T_decay
            T = max(T, T_min)
        
        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovProx step={step+1} ({elapsed:.0f}s): best={best_score:.8f} T={T:.4f}")
        
        if T <= T_min * 1.01 and step > 5000:
            # Try to restart with perturbation
            if best_score > 1.5 and rng.random() < 0.1:
                a = a + rng.normal(0, 0.01, n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                T = 1.0
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. LP alternating with exact solver (GLPK) + Nesterov dual averaging
3. Active-set QP refinement for exact local optimization

5. Multi-length exploration (n=1200, 1500, 1800)

Key improvements over previous:
- Exact LP solver for better best-response accuracy

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    """Find a working LP solver, preferring exact ones."""
    for solver_name in [cp.GLPK, cp.CBC, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.GLPK, cp.CLARABEL, cp.CBC, cp.PDLP, cp.SCS]
    result = None
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                result = (np.asarray(b_var.value, dtype=np.float64),
                          float(M_var.value) if M_var.value is not None else 0.0)
                break
        except Exception:
            continue

    if result is None:
        return None, None

    b_val, M_val = result
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, max(0.0, M_val)


def edge_concentrated_init(n, rng, sharpness=3.0):
    """Create U-shaped initialization concentrating mass at edges."""
    x = np.linspace(0, 1, n)
    a = x**sharpness + (1 - x)**sharpness + 0.05
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def lp_alternate_smoothed(a0, time_limit, solver, n_max_iters=1000000):
    """
    LP alternating with running average smoothing and Nesterov dual averaging.
    Uses both the running average and Nesterov x for best solution tracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()  # Dual variable
    avg = a.copy()  # Running average
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(54321)
    slow_stall = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP best response for y
        b_new, M_val = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Nesterov x update
        a_num = (it + 1) * a + alpha * (it + 2) * b_new
        a_den = (it + 1) + alpha * (it + 2)
        a = np.maximum(a_num / a_den, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Running average (weighted)
        w = 1.0 / (it + 1)
        avg = (1.0 - w) * avg + w * b_new
        avg = np.maximum(avg, 0)
        s_avg = avg.sum()
        if s_avg < 1e-15:
            avg = a.copy()
        else:
            avg = avg / s_avg

        # Evaluate all three candidates
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        score_avg = evaluate_local(avg)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg.copy()

        # Stall detection
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
                slow_stall = 0
            else:
                stall_count += 1
                slow_stall += 1
            prev_best = best_score

            if stall_count >= 20 and slow_stall >= 60:
                # Perturb to escape
                strength = 0.01 + 0.005 * (slow_stall / 100)
                noise = rng.normal(0, strength, n)
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
                slow_stall = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LPA iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M={M_val:.6f}")

    return best_a, best_score


def active_set_qp_refine(a0, time_limit, solver, n_max_iters=2000):
    """
    Active-set QP refinement.
    Identify active constraints, solve the corresponding QP exactly.
    Iterate until convergence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Identify active set
        eps = max(1e-8, 0.005 * max_b)
        active = np.where(b >= max_b - eps)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Build Q = (1/|S|) * sum_{k in S} M_k
        # M_k[i,j] = 1 if i+j = k, else 0
        # Q[i,j] = 1/|S| if i+j in active set, else 0
        active_set = set(active.tolist())
        Q = np.zeros((n, n))
        for i in range(n):
            j_start = max(0, active[0] - i)
            j_end = min(n - 1, active[-1] - i)
            if j_start <= j_end:
                for j in range(j_start, j_end + 1):
                    if (i + j) in active_set:
                        Q[i, j] = 1.0 / len(active)

        # Solve QP: min a^T Q a s.t. a >= 0, sum(a) = 1
        a_var = cp.Variable(n, nonneg=True)
        obj = cp.quad_form(a_var, Q, assume_PSD=True)
        constraints = [cp.sum(a_var) == 1]
        prob = cp.Problem(cp.Minimize(obj), constraints)

        solved = False
        q_solvers = [cp.CLARABEL, solver, cp.SCS, cp.ECOS]
        for qs in q_solvers:
            try:
                prob.solve(solver=qs)
                if prob.status in ('optimal', 'optimal_inaccurate') and a_var.value is not None:
                    a_new = np.asarray(a_var.value, dtype=np.float64)
                    a_new = np.maximum(a_new, 0)
                    sn = a_new.sum()
                    if sn > 1e-15:
                        a_new = a_new / sn
                    score = evaluate_local(a_new)
                    if score < best_score:
                        best_score = score
                        best_a = a_new.copy()
                    a = a_new
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            # Fallback: multiplicative weights step
            w = np.zeros(2 * n - 1)
            w[active] = 1.0 / len(active)
            g = np.correlate(w, a, mode='full')[n - 1:2 * n - 1]
            a_new = a * np.exp(-0.01 * g)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
                score = evaluate_local(a_new)
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                a = a_new

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ActQP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} |S|={len(active)}")

        # Check convergence
        if it > 0 and (it + 1) % 50 == 0:
            b_check = fast_convolve(a)
            b_max = b_check.max()
            b_min_active = b_check[active].min()
            if np.abs(b_max - b_min_active) < 1e-10 * b_max:
                print(f"    ActQP converged at iter {it+1}")
                break

    return best_a, best_score


def multiplicative_weights(a0, time_limit, n_steps=300000):
    """
    Multiplicative weights (mirror descent) with adaptive active set and step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(888)

    eta = 0.03
    patience = 0
    max_patience = 300

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Adaptive threshold
        rel_thresh = max(1e-5, 0.05 * np.exp(-step / 3000))
        threshold = max_b * (1.0 - rel_thresh)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute gradient via FFT
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        g_full = np.correlate(w, a, mode='full')
        g = g_full[n - 1:2 * n - 1]

        # Mirror descent update
        a_new = a * np.exp(-eta * g)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score = evaluate_local(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            eta *= 0.7
            patience = 0
            if eta < 0.001:
                a = a + rng.normal(0, 0.01, n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                eta = 0.03

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW step={step+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.5f} |S|={len(active)}")

    return best_a, best_score


def nesterov_subgradient(a0, time_limit, n_steps=200000):
    """
    Nesterov-accelerated subgradient with adaptive active set.
    Uses the smoothed max (LogSumExp) for better gradient information.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    # Nesterov momentum
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    T = 3.0
    T_min = 0.01
    T_decay = 0.9997

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = y.copy()

        # LogSumExp gradient
        if T > T_min * 2:
            shifted = (b - max_b) / T
            shifted = np.maximum(shifted, -40)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()
        else:
            # Hard active set
            thresh = max_b * (1 - 1e-4)
            active = np.where(b >= thresh)[0]
            if len(active) == 0:
                active = np.array([np.argmax(b)])
            w = np.zeros(2 * n - 1)
            w[active] = 1.0 / len(active)

        # Gradient: g[i] = 2 * sum_k w[k] * a[k-i]
        g = np.correlate(w, y, mode='full')[n - 1:2 * n - 1] * 2.0

        # Project onto tangent space
        g = g - g.mean()

        # Nesterov step size
        t_k = 2.0 / (step + 2)

        # Multiplicative update with step size
        eta = 0.5 / (np.linalg.norm(g) + 1e-8)
        eta = min(eta, 0.3)

        y_new = y * np.exp(-eta * g)
        sn = y_new.sum()
        if sn < 1e-15:
            y_new = y.copy()
            sn = 1.0
        y_new = y_new / sn

        # Nesterov interpolation
        a = y_new
        y = a + t_k * (a - y)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        # Decay temperature
        T = max(T * T_decay, T_min)

        if (step + 1) % 30000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG step={step+1} ({elapsed:.0f}s): best={best_score:.8f} T={T:.4f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a_qp, _ = active_set_qp_refine(best_seq, t_alloc, solver)
        local_update(a_qp, "active_qp")

    # === Phase 3: Second LP round ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_alternate_smoothed(best_seq, t_alloc, solver)
        local_update(a_lp2, "lp_round2")

    # === Phase 4: Nesterov subgradient ===
    print(f"\n=== Phase 4: Nesterov Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ns, _ = nesterov_subgradient(best_seq, t_alloc)
        local_update(a_ns, "nesterov_sg")

    # === Phase 5: Active-set QP again ===
    print(f"\n=== Phase 5: QP Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_qp2, _ = active_set_qp_refine(best_seq, t_alloc, solver)
        local_update(a_qp2, "qp_polish")

    # === Phase 6: Try n=1500 ===
    print(f"\n=== Phase 6: n=1500 exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_1500 = min(rem * 0.15, 150.0)
        print(f"  Budget: {t_1500:.0f}s")

        # Interpolate best to 1500
        init_1500 = interpolate_sequence(best_seq, 1500)
        # Also try edge-concentrated at 1500
        init_1500e = edge_concentrated_init(1500, rng, sharpness=3.0)

        # Quick MW on both
        t_sub = t_1500 * 0.3
        a_1500a, _ = multiplicative_weights(init_1500, t_sub, n_steps=5000)
        local_update(a_1500a, "mw_n1500")

        rem = deadline - time.monotonic()
        if rem > 40:
            start_1500 = init_1500 if evaluate_local(init_1500) < evaluate_local(init_1500e) else init_1500e
            t_lp1500 = min(rem * 0.5, 100.0)
            a_1500b, _ = lp_alternate_smoothed(start_1500, t_lp1500, solver)
            local_update(a_1500b, "lp_n1500")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 60.0)
        a_fin, _ = lp_alternate_smoothed(best_seq, t_fin, solver)
        local_update(a_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_qp = min(rem * 0.4, 30.0)
        a_fqp, _ = active_set_qp_refine(best_seq, t_qp, solver)
        local_update(a_fqp, "final_qp")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_fm, _ = multiplicative_weights(best_seq, min(rem * 0.6, 20.0), n_steps=10000)
        local_update(a_fm, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        clipped = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(clipped)
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
