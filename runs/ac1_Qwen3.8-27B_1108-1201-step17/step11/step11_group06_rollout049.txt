import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    else:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, n, T):
    """LogSumExp objective and gradient w.r.t. z (softmax parameterization)."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse_val = max_b + T * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse_val
    w = exp_terms / (exp_terms.sum() + 1e-300)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def nesterov_mw(a0, time_limit, n_steps, eta0=0.1):
    """Nesterov-accelerated multiplicative weights with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    x = a.copy()  # Nesterov extrapolation point
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(x)
        max_b = b.max()
        # Subgradient of max at x
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / (b[active].sum() + 1e-300)
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * x[k - idx]

        # Mirror descent step (multiplicative weights)
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = x * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        a_new = a_new / s_new if s_new > 1e-15 else a.copy()

        # Nesterov update
        theta = 2.0 / (step + 2 + 2.0)
        x = (1.0 - theta) * a_new + theta * a
        x = np.maximum(x, 0)
        sx = x.sum()
        x = x / sx if sx > 1e-15 else a_new.copy()

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new
        # Adaptive eta
        if score_new < best_score + 1e-12:
            eta = min(eta * 1.005, 0.5)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break
    return best_a, best_score

def lse_lbgfs(a0, time_limit, T_list, maxiter=400):
    """L-BFGS on LogSumExp with temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            o, _ = lse_obj_grad(z_, n, T)
            return o

        def jac(z_):
            if time.monotonic() - t0 > time_limit:
                return np.zeros(len(z_))
            _, g = lse_obj_grad(z_, n, T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-30,
                                    'gtol': 1e-20, 'disp': False})
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization. Key improvements over previous:
    - Asymmetric (ramp) initializations exploiting the fact that optimal is NOT palindromic
    - Nesterov-accelerated MW
    - Multiple LP step-size strategies
    - Adaptive LSE temperature
    - Better time management
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = ramp_init(n_init, rng, strength=0.3)
        best_score = evaluate_local(best_seq)
        print(f"Init (ramp): n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick diverse initialization (5% time) ===
    print(f"\n=== Phase 1: Diverse init (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    n_try = n_best
    best_init = best_seq.copy()
    inits = [best_seq]
    inits.append(ramp_init(n_try, rng, 0.2))
    inits.append(ramp_init(n_try, rng, 0.5))
    inits.append(geometric_init(n_try, 0.9995))
    inits.append(structured_init(n_try, rng))
    inits.append(structured_init(n_try, rng))
    inits.append(1.0 + 0.3 * np.linspace(0, 1, n_try))
    inits[-1] = np.maximum(inits[-1], 0.01)
    inits[-1] = inits[-1] / inits[-1].sum()

    t_per_init = t1 / (len(inits) + 1)
    for i, init in enumerate(inits[1:], 1):
        rem = deadline - time.monotonic()
        if rem < 5:
            break
        sc = evaluate_local(init)
        print(f"  Init {i}: score={sc:.8f}")
        # Quick MW on this init
        a_mw, _ = nesterov_mw(init, min(t_per_init, rem * 0.3), n_steps=50000)
        try_update(a_mw, f"init_mw_{i}")
        if evaluate_local(a_mw) < evaluate_local(best_init):
            best_init = a_mw.copy()

    try_update(best_init, "best_init")

    # === Phase 2: Nesterov MW on best (8% time) ===
    print(f"\n=== Phase 2: Nesterov MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.08, 120.0)
    if t2 > 10:
        a_mw, _ = nesterov_mw(best_seq, t2, n_steps=500000, eta0=0.15)
        try_update(a_mw, "mw_nest")

    # === Phase 3: LP alternating with sqrt schedule (30% time) ===
    print(f"\n=== Phase 3: LP sqrt schedule (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.30, 500.0)
    if t3 > 60:
        print(f"  Budget: {t3:.0f}s")
        a_lp, _ = lp_alternate(best_seq, t3, solver, n_max_iters=500000, alpha_mode='sqrt')
        try_update(a_lp, "lp_sqrt")

    # === Phase 4: LP alternating with Nesterov schedule (15% time) ===
    print(f"\n=== Phase 4: LP nesterov schedule (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    if t4 > 30:
        a_lp2, _ = lp_alternate(best_seq, t4, solver, n_max_iters=300000, alpha_mode='nestеров')
        try_update(a_lp2, "lp_nest")

    # === Phase 5: LSE L-BFGS with adaptive temperature (20% time) ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.20, 350.0)
        # Adaptive temperature schedule: start based on current "sharpness"
        b_curr = fast_convolve(best_seq)
        curr_max = b_curr.max()
        curr_mean = b_curr.mean()
        sharpness = curr_max / (curr_mean + 1e-30)
        T_start = max(1.0, sharpness * 0.1)
        T_stages = np.geomspace(T_start, 0.001, num=12)
        print(f"  T range: [{T_stages[0]:.3f}, {T_stages[-1]:.3f}], budget={t5:.0f}s")
        a_lse = lse_lbgfs(best_seq, t5, T_stages.tolist(), maxiter=300)
        try_update(a_lse, "lse")

    # === Phase 6: Multi-n exploration (10% time) ===
    print(f"\n=== Phase 6: Multi-n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for ln in [1000, 1400, 1600]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.2, 180.0)
        if t_ln < 30:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln, _ = nesterov_mw(init_ln, t_ln * 0.3, n_steps=50000, eta0=0.1)
        try_update(a_ln, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.6)
        if t_lp > 20:
            a_ln_lp, _ = lp_alternate(a_ln, t_lp, solver, n_max_iters=100000, alpha_mode='sqrt')
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 7: Asymmetric refinement (10% time) ===
    print(f"\n=== Phase 7: Asymmetric refine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t7 = min(rem * 0.10, 150.0)
        n_cur = len(best_seq)
        # Try different ramp strengths on current best
        for eps in [0.1, -0.1, 0.2, -0.2]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            x = np.linspace(0, 1, n_cur)
            a_asym = best_seq * (1.0 + eps * (2 * x - 1))
            a_asym = np.maximum(a_asym, 0)
            s = a_asym.sum()
            a_asym = a_asym / s if s > 1e-15 else best_seq.copy()
            t_sub = min(rem * 0.3, 40.0)
            a_opt, _ = nesterov_mw(a_asym, t_sub, n_steps=50000, eta0=0.05)
            try_update(a_opt, f"asym_{eps}")

    # === Phase 8: Final LP polish (10% time) ===
    print(f"\n=== Phase 8: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t8 = min(rem * 0.10, 180.0)
        a_fin, _ = lp_alternate(best_seq, t8, solver, n_max_iters=200000, alpha_mode='sqrt')
        try_update(a_fin, "final_lp")

    # === Phase 9: Final LSE polish ===
    print(f"\n=== Phase 9: Final LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.4, 200.0)
        T_fine = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_fin_lse = lse_lbgfs(best_seq, t9, T_fine, maxiter=500)
        try_update(a_fin_lse, "final_lse")

    # === Phase 10: Ultra-fine MW ===
    print(f"\n=== Phase 10: Ultra-fine MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t10 = min(rem * 0.5, 100.0)
        a_uf, _ = nesterov_mw(best_seq, t10, n_steps=200000, eta0=0.01)
        try_update(a_uf, "ultra_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
