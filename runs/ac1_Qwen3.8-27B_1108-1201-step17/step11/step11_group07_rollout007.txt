"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_grad_fft(a, T):
    """Compute LSE objective and gradient for normalized a (sum=1) using FFT.
    Returns (obj, grad) where grad is the gradient of 2n*LSE(conv(a,a), T).
    """
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    c_max = c.max()
    shifted = (c - c_max) / T
    shifted_max = shifted.max()
    exp_c = np.exp(shifted - shifted_max)
    exp_sum = exp_c.sum()
    lse_val = c_max + T * (shifted_max + np.log(exp_sum))
    obj = 2.0 * n * lse_val
    
    w = exp_c / exp_sum  # length 2n-1, sums to 1
    
    # Compute H[m] = sum_j w[j+m] * a[j] for m=0..n-1
    # Using FFT: H = ifft(fft(w, L) * conj(fft(a, L)))[:n]
    L = 1 << (2 * n).bit_length()  # L >= 2n-1 + n-1 = 3n-2, so L >= 4n is safe
    W = rfft(w, L)
    A = rfft(a, L)
    H_full = irfft(W * np.conj(A), L)
    H = H_full[:n]
    
    # grad_m = 4*n*T*H_m (before projection)
    grad = 4.0 * n * T * H
    
    return obj, grad


def lse_grad_softmax(z, T):
    """Compute LSE objective and gradient in log-parameterization (softmax)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    obj, grad_a = lse_grad_fft(a, T)
    
    # Project to simplex tangent space: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lp_solve(a, M, time_limit=10.0):
    """Solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns g (unnormalized) or None if failed.
    """
    n = len(a)
    t_start = time.monotonic()
    
    # Build constraint matrix A_ub: (2n-1) x n
    # A_ub[k, m] = a[k-m] if 0 <= k-m < n, else 0
    # b_ub = M * ones(2n-1)
    
    # For efficiency, build A_ub using the structure (it's a convolution matrix)
    # A_ub is a "convolution matrix" (Toeplitz-like)
    
    A_ub = np.zeros((2 * n - 1, n))
    for m in range(n):
        # Column m: A_ub[k, m] = a[k-m] for k-m in [0, n-1]
        # So k ranges from m to m+n-1
        A_ub[m:m+n, m] = a
    
    b_ub = np.full(2 * n - 1, M)
    c_obj = np.ones(n)  # minimize -sum(g) => c = -ones(n)
    
    # Use linprog with HiGHS
    res = linprog(
        c=-c_obj,  # minimize -sum(g)
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method='highs',
        options={'time_limit': time_limit}
    )
    
    if res.status == 0:
        return res.x
    elif res.status == 3:  # Time limit reached, try to get incumbent
        if res.x is not None:
            return res.x
    return None


def lp_optimize(a_init, time_limit, n_lp_iters=50):
    """LP-based optimization following the SOTA approach."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    print(f"    LP start: n={n}, score={best_score:.8f}")
    
    for it in range(n_lp_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        M = c.max()
        
        # Solve LP
        t_lp_start = time.monotonic()
        g = lp_solve(a, M, time_limit=min(30.0, (time_limit - (t_lp_start - t0)) * 0.5))
        
        if g is None:
            break
        
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum  # normalize to sum=1
        
        # Try different mixing parameters
        found = False
        for t in [0.1, 0.05, 0.2, 0.3, 0.15, 0.5]:
            a_new = (1 - t) * a + t * g
            a_new = normalize(a_new)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                found = True
                break
        
        if not found:
            # Try smaller steps
            for t in [0.01, 0.02, 0.03]:
                a_new = (1 - t) * a + t * g
                a_new = normalize(a_new)
                sc = evaluate_local(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    found = True
                    break
        
        if it % 10 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, M={M:.6f}, g_sum_raw={g_sum:.4f}")
    
    return best_a, best_score


def adam_lse_optimize(a_init, time_limit, T_stages=None, lr=0.01, beta1=0.9, beta2=0.999):
    """Adam optimizer on LSE objective with temperature annealing."""
    n = len(a_init)
    if T_stages is None:
        T_stages = [20, 10, 5, 3, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    z = np.log(np.maximum(a_init, 1e-300))
    best_a = normalize(a_init.copy())
    best_score = evaluate_local(best_a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t_count = 0
    
    t0 = time.monotonic()
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Run Adam for a fixed number of steps or until time runs out
        stage_steps = max(10, int(remaining * 100))  # ~100 steps per second
        for step in range(stage_steps):
            if time.monotonic() - t0 > time_limit:
                break
            if step > 0 and step % max(1, stage_steps // 10) == 0:
                if time.monotonic() - t0 > time_limit - 5:
                    break
            
            obj, grad_z = lse_grad_softmax(z, stage_T)
            
            t_count += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_count)
            v_hat = v / (1 - beta2 ** t_count)
            
            z -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            
            # Evaluate every 50 steps
            if step % 50 == 0 or step == stage_steps - 1:
                a = np.exp(z - z.max())
                a = a / a.sum()
                sc = evaluate_local(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        # Reset Adam moments at stage boundaries
        m = np.zeros(n)
        v = np.zeros(n)
        t_count = 0
    
    return best_a, best_score


def peak_shave(a_init, time_limit):
    """Direct peak-shaving: reduce elements contributing to autocorrelation peaks."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(789)
    
    no_improve = 0
    step_size = 0.1
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        c_max = c.max()
        
        # Find peak positions (within 5% of max)
        threshold = c_max * 0.95
        peaks = np.where(c >= threshold)[0]
        
        if len(peaks) == 0:
            peaks = np.array([np.argmax(c)])
        
        # For each peak, find the elements that contribute most
        contributions = np.zeros(n)
        for k in peaks:
            for m in range(max(0, k - n + 1), min(n, k + 1)):
                contributions[m] += a[k - m]
        
        contributions /= len(peaks)
        
        # Decrease high-contribution elements, increase low-contribution ones
        # Use a smooth version
        step = step_size * (contributions - contributions.mean())
        step = np.clip(step, -0.5, 0.5) * a  # Proportional decrease
        
        a_new = a - step
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 10:
                    step_size *= 0.5
                    no_improve = 0
                    if step_size < 1e-6:
                        break
    
    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)
    
    candidates.append(normalize(np.exp(-8 * (x - 0.5) ** 2)))
    candidates.append(normalize(0.5 * (1 + np.cos(2 * np.pi * x))))
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    candidates.append(normalize(np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)))
    candidates.append(np.ones(n) / n)
    candidates.append(normalize(np.exp(-5 * x) + np.exp(-5 * (1 - x))))
    
    # Random smooth
    a = np.zeros(n)
    for i in range(8):
        a += rng.normal(0, 1) * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    candidates.append(normalize(np.maximum(a + 0.5, 0)))
    
    # Power law
    candidates.append(normalize(x * (1 - x)))
    
    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def sobol_like_init(n, rng):
    """Create a sequence based on a smoothed version with structured noise."""
    x = np.linspace(0, 1, n)
    # Start with a smooth base
    base = np.exp(-4 * (x - 0.5) ** 2)
    # Add structured low-frequency noise
    for k in range(1, 6):
        base += 0.1 * np.cos(2 * np.pi * k * x + rng.uniform(0, 2 * np.pi))
    return normalize(np.maximum(base, 0))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Linear Programming (LP)-based optimization (SOTA linear programming approach)
2. Adam-optimized LSE gradient descent with aggressive temperature annealing
3. Direct peak-shaving heuristic with adaptive step sizes
4. Multi-length exploration with interpolation
5. Random nudge exploration for escaping flat local minima
6. Final low-temperature polish
Key innovations: LP-based direction finding, Adam optimizer with
momentum for LSE landscape, and structured peak-shaving.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to nonnegative, sum=1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_grad_fft(a, T):
    """Compute LSE objective and gradient for normalized a (sum=1).
    Returns (obj, grad) where grad = d(2n * LSE(conv(a,a), T))/da.
    """
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    c_max = c.max()
    shifted = (c - c_max) / T
    shifted_max = shifted.max()
    exp_c = np.exp(shifted - shifted_max)
    exp_sum = exp_c.sum()
    lse_val = c_max + T * (shifted_max + np.log(exp_sum))
    obj = 2.0 * n * lse_val

    w = exp_c / exp_sum  # length 2n-1, sums to 1

    # Compute H[m] = sum_j w[j+m] * a[j] for m=0..n-1 via FFT
    L = 1 << (3 * n).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    H_full = irfft(W * np.conj(A), L)
    H = H_full[:n]

    grad = 4.0 * n * T * H
    return obj, grad


def lse_grad_softmax(z, T):
    """Compute LSE objective and gradient in log-parameterization (softmax)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    obj, grad_a = lse_grad_fft(a, T)

    # Project to simplex tangent: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lp_solve_single(a, M, time_limit=10.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns g (nonnegative, possibly unnormalized) or None.
    """
    n = len(a)

    # Build constraint matrix: A_ub[k, m] = a[k-m] if 0 <= k-m < n
    A_ub = np.zeros((2 * n - 1, n))
    for m in range(n):
        A_ub[m:m + n, m] = a

    b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12))
    c_obj = -np.ones(n)  # minimize -sum(g)

    try:
        res = linprog(
            c=c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': time_limit, 'presolve': True}
        )
        if res.status == 0:
            return np.maximum(res.x, 0)
        elif res.status == 3 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_optimize(a_init, time_limit, max_iters=200):

        lp_time = min(20.0, remaining * 0.3)

        if g is None:
            break

def adam_lse_optimize(a_init, time_limit, T_stages=None, lr=0.05,

    best_a = normalize(a_init.copy())
    best_score = evaluate_local(best_a)

    t0 = time.monotonic()

    for stage_idx, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Adaptive steps per stage: more steps for earlier (higher T) stages
        frac = (len(T_stages) - stage_idx) / len(T_stages)
        n_steps = max(20, int(remaining * 50 * (0.5 + 0.5 * frac)))
        n_steps = min(n_steps, 50000)

        m = np.zeros(n)
        v = np.zeros(n)
        t_count = 0
        stage_lr = lr * (1.0 + 0.5 * frac)  # Higher LR early on

        for step in range(n_steps):
            if time.monotonic() - t0 > time_limit - 2:
                break

            obj, grad_z = lse_grad_softmax(z, stage_T)

            t_count += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_count)
            v_hat = v / (1 - beta2 ** t_count)

            z -= stage_lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            if (step + 1) % eval_interval == 0:
                a = np.exp(z - z.max())
                a = a / a.sum()
                sc = evaluate_local(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        # Evaluate at end of stage
        a = np.exp(z - z.max())
        a = a / a.sum()
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if stage_idx % 5 == 0:
            print(f"    Adam LSE stage {stage_idx}/{len(T_stages)}: "

    return best_a, best_score


def peak_shave(a_init, time_limit, rng=None):
    """Direct peak-shaving: reduce elements most contributing to autocorrelation peaks."""
    if rng is None:
        rng = np.random.default_rng(789)

    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    step_size = 0.05
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        c_max = c.max()

        # Find peak positions (within 10% of max)
        threshold = c_max * 0.90
        peaks = np.where(c >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(c)])

        # Compute contribution of each element to the peaks
        contributions = np.zeros(n)
        for k in peaks[:50]:  # Limit to 50 peaks for speed
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for m in range(js, je + 1):
                contributions[m] += a[k - m]
        contributions /= min(len(peaks), 50)

        # Direction: decrease high-contribution, increase low-contribution
        direction = -(contributions - contributions.mean())
        # Scale by current value to avoid going negative too easily
        direction = direction * (a + 0.01)

        # Try a few step sizes
        found = False
        for scale in [step_size, step_size * 0.5, step_size * 0.1, step_size * 2.0]:
            a_new = a + scale * direction
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                found = True
                no_improve = 0
                break

        if not found:
            no_improve += 1
            if no_improve >= 8:
                step_size *= 0.6
                no_improve = 0
                if step_size < 1e-7:
                    break

            # Occasionally try random perturbation
            if no_improve >= 4:
                noise = rng.normal(0, 0.02, n)
                a_new = a + noise
                a_new = normalize(a_new)
                sc = evaluate_local(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    step_size = 0.03
                    no_improve = 0

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def random_nudge_search(a_init, time_limit, rng, n_tries=5000):

def lbgfs_lse_optimize(a_init, time_limit, T_stages=None):
