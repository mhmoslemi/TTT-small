import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # Each column j has nonzeros at rows j..j+n-1 with values a[0], a[1], ..., a[n-1]
    # But only valid rows: max(0, j) to min(size-1, j+n-1)
    total_nonzero = n * n  # upper bound
    rows = np.zeros(total_nonzero, dtype=np.int32)
    cols = np.zeros(total_nonzero, dtype=np.int32)
    vals = np.zeros(total_nonzero, dtype=np.float64)
    count = 0
    for j in range(n):
        r_start = j
        r_end = min(j + n, size)
        for r in range(r_start, r_end):
            rows[count] = r
            cols[count] = j
            vals[count] = a[r - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a, find b minimizing max_k conv(a,b)[k] s.t. b>=0, sum(b)=1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_and_grad(z, T_temp, time_limit_check=None):
    """Compute LogSumExp objective and gradient w.r.t. z for softmax parameterization."""
    if time_limit_check is not None and time.monotonic() > time_limit_check:
        return None, None
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    # Gradient w.r.t. a
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wk = w[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += wk * a[k - idx]
    grad_a *= 2.0
    # Gradient w.r.t. z: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


class LSEFuncObj:
    def __init__(self, T, deadline):
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = lse_obj_and_grad(z, self.T)
        if obj is None:
            return 1e15
        return obj


class LSEJacObj:
    def __init__(self, T, deadline):
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = lse_obj_and_grad(z, self.T)
        if g is None:
            return np.zeros(len(z))
        return g


def nesterov_lp_refine(a0, time_limit, solver, T_mat_init=None, n_max_iters=500000):
    """Nesterov-accelerated LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    lp_times = []
    T_mat = T_mat_init if T_mat_init is not None else build_conv_matrix_sparse(y)
    x_prev = a.copy()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_times.append(time.monotonic() - lp_t0)
        if b_new is None:
            continue
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 10.0:
                break

        # Nesterov momentum: beta = (sqrt(1+4/t^2)-1)/2
        t_cur = it + 1.0
        beta = (np.sqrt(1.0 + 4.0 / (t_cur * t_cur)) - 1.0) / 2.0
        alpha = 2.0 / (t_cur + 2.0)

        # Update y with Nesterov extrapolation
        x_new = b_new
        y = x_new + beta * (x_new - x_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x_new.copy()
            sy = x_new.sum()
        y = y / sy
        x_prev = x_new.copy()

        # Track average iterate
        a = (it * a + alpha * (it + 2) * b_new) / (it + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        T_mat = build_conv_matrix_sparse(y)

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:

                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)

                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0

def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):

    n = len(a0)

def multiplicative_weights_fast(a0, time_limit, n_steps=200000, eta0=0.05):

def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):

    s = a.sum()
    if s < 0.01:

    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        a = 0.3 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 7 * x + 1.0)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-2 * (x - 0.5) ** 2 / 0.05) + 0.3
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_scale_warm_start(target_n, time_limit, solver, rng):
    """Multi-scale optimization: start small, grow to target_n."""
    scales = []
    s = 100
    while s < target_n:
        scales.append(s)
        s *= 2
    scales.append(target_n)
    # Ensure we don't skip too much
    if target_n not in scales:
        scales = [x for x in scales if x <= target_n]
        scales.append(target_n)
    # Deduplicate and sort
    scales = sorted(set(scales))

    a = structured_init(scales[0], rng)
    t0 = time.monotonic()

    for i, n_scale in enumerate(scales):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        if n_scale == len(a):
            t_sub = remaining * 0.4
        else:
            a = interpolate_sequence(a, n_scale)
            t_sub = remaining * 0.3

        # Quick MW
        t_mw = min(t_sub * 0.3, 30.0)
        if t_mw > 3:
            a, _ = multiplicative_weights_fast(a, t_mw, n_steps=50000, eta0=0.05)

        # LP refine
        t_lp = t_sub - t_mw - 2
        if t_lp > 5:
            T_mat = build_conv_matrix_sparse(a)
            a, _ = nesterov_lp_refine(a, t_lp, solver, T_mat_init=T_mat, n_max_iters=100000)
            score = evaluate_local(a)
            print(f"    Multi-scale n={n_scale}: score={score:.8f} ({time.monotonic()-t0:.0f}s)")

    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses multi-scale warm start, Nesterov LP alternating, fine LSE L-BFGS,
    multi-length exploration, and structured perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        best_seq = structured_init(1200, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 0: Quick MW on parent ===
    print(f"\n=== Phase 0: Quick MW n={n_best} ===")
    rem = deadline - time.monotonic()
    t0_ph = min(rem * 0.03, 40.0)
    if t0_ph > 5:
        a_mw, _ = multiplicative_weights_fast(best_seq, t0_ph, n_steps=100000, eta0=0.05)
        try_update(a_mw, "mw_0")

    # === Phase 1: LP alternating (main) ===
    print(f"\n=== Phase 1: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        T_mat = build_conv_matrix_sparse(best_seq)
        a_lp, _ = nesterov_lp_refine(best_seq, t1, solver, T_mat_init=T_mat, n_max_iters=500000)
        try_update(a_lp, "lp_1")

    # === Phase 2: LSE L-BFGS fine polish ===
    print(f"\n=== Phase 2: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t2 = min(rem * 0.15, 250.0)
        T_stages = [20.0, 10.0, 5.0, 2.5, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_lse = lbgfs_refine(best_seq, t2, T_stages, maxiter_per_stage=600)
        try_update(a_lse, "lse_2")

    # === Phase 3: Palindromic LP ===
    print(f"\n=== Phase 3: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.10, 150.0)
        a_pal = make_palindromic(best_seq)
        T_mat_pal = build_conv_matrix_sparse(a_pal)
        a_pal_lp, _ = nesterov_lp_refine(a_pal, t3, solver, T_mat_init=T_mat_pal, n_max_iters=100000)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    candidate_lengths = [1000, 1100, 1300, 1400, 1500]
    for ln in candidate_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        if abs(ln - len(best_seq)) < 50:
            continue
        t_ln = min(rem * 0.12, 120.0)
        if t_ln < 20:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        t_mw_sub = min(t_ln * 0.2, 20.0)
        a_ln_mw, _ = multiplicative_weights_fast(init_ln, t_mw_sub, n_steps=50000, eta0=0.05)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP refine
        rem = deadline - time.monotonic()
        t_lp_sub = min(rem * 0.4, t_ln * 0.7)
        if t_lp_sub > 15:
            T_mat_ln = build_conv_matrix_sparse(a_ln_mw)
            a_ln_lp, _ = nesterov_lp_refine(a_ln_mw, t_lp_sub, solver, T_mat_init=T_mat_ln, n_max_iters=100000)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 5: Multi-scale from scratch ===
    print(f"\n=== Phase 5: Multi-scale (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t5 = min(rem * 0.15, 200.0)
        a_ms = multi_scale_warm_start(1200, t5, solver, rng)
        try_update(a_ms, "multi_scale")

    # === Phase 6: Perturbation escape ===
    print(f"\n=== Phase 6: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for trial in range(5):
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        strength = 0.01 + 0.008 * trial
        a_pert = perturb_sequence(best_seq, rng, strength=strength)
        t_sub = min(rem * 0.20, 50.0)
        if t_sub < 10:
            break
        T_mat_pert = build_conv_matrix_sparse(a_pert)
        a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, T_mat_init=T_mat_pert, n_max_iters=50000)
        try_update(a_opt, f"pert{trial}")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.35, 100.0)
        T_mat_fin = build_conv_matrix_sparse(best_seq)
        a_fin, _ = nesterov_lp_refine(best_seq, t7, solver, T_mat_init=T_mat_fin, n_max_iters=200000)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final LSE polish ===
    print(f"\n=== Phase 8: Final LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.4, 60.0)
        T_stages_final = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_fin_lse = lbgfs_refine(best_seq, t8, T_stages_final, maxiter_per_stage=400)
        try_update(a_fin_lse, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    # Return as list of positive values, clipped to [0, 1000]
    # Scale so sum is reasonable (the evaluator clips to [0,1000] and handles sum)
    # Keep normalized (sum=1) since score is scale-invariant
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
