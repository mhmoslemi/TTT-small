import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def lse_grad(a, T):
    """Compute LSE objective and gradient with respect to a (unconstrained)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    return obj, grad_a

def adam_lse_optimize(a0, time_limit, T_start=5.0, T_end=0.001, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on LSE-smoothed objective with simplex projection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    total_iters = 500000  # Estimate
    # We'll anneal T exponentially
    
    no_improve = 0
    lr_scale = 1.0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Anneal temperature
        progress = min(1.0, t / 200000)
        T = T_start * (T_end / T_start) ** progress
        T = max(T, T_end)
        
        obj, grad = lse_grad(a, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * lr_scale * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 100:
            lr_scale *= 0.8
            no_improve = 0
            if lr_scale < 0.001:
                break
    
    return best_a, best_score

def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Spectral flattening via Gerchberg-Saxton."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a

def peak_shave(a, time_limit, n_steps=100000):
    """Peak shaving with multiplicative updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.03
    scale = 1.0
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Target top peaks
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Weighted subgradient
        weights = np.ones(len(active)) / len(active)
        
        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[i] * a[k - idx]
        
        # Multiplicative update (mirror descent)
        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break
    
    return best_a, best_score

def build_conv_matrix_sparse_info(a):
    """Return the structure for the convolution matrix (for LP)."""
    n = len(a)
    size = 2 * n - 1
    # T @ b = convolve(a, b)
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    return n, size

def solve_lp_best_response(a_fixed, time_limit_per_lp=2.0):
    """Solve the LP: min M s.t. conv(a_fixed, b) <= M, sum(b) = 1, b >= 0."""
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build constraint matrix A_ub for linprog
    # Variables: b_0, ..., b_{n-1}, M
    # Constraints: sum_j a_fixed[k-j] * b_j <= M for each k
    #              sum b_j = 1
    #              b_j >= 0, M free
    
    # For linprog: min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
    # x = (b_0, ..., b_{n-1}, M)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # Minimize M
    
    # A_eq: sum b_j = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # A_ub: For each k, sum_j T[k,j] b_j - M <= 0
    # T[k,j] = a_fixed[k-j] if 0 <= k-j < n
    A_ub = np.zeros((size, n + 1))
    for k in range(size):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a_fixed[idx]
        A_ub[k, -1] = -1.0  # -M term
    b_ub = np.zeros(size)
    
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    b_val = result.x[:n]
    M_val = result.x[-1]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_val))

def nesterov_lp_alt(a0, time_limit, n_max_iters=1000000):
    """Nesterov-accelerated alternating LP method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    lp_count = 0
    lp_fail_count = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y)
        
        if b_new is None:
            lp_fail_count += 1
            if lp_fail_count > 20:
                break
            continue
        
        lp_count += 1
        lp_fail_count = 0
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = elapsed / max(1, lp_count)
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}, LP avg={avg_lp:.3f}s")
    
    return best_a, best_score

def structured_init(n, rng):
    """Create structured initial sequences."""
    choice = rng.integers(0, 10)

    if choice == 0:

    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 - np.abs(2 * x - 1)
    elif choice == 8:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        # Hann window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def perturb_sequence(a, rng, strength=0.03):
    """Create perturbed sequence."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
