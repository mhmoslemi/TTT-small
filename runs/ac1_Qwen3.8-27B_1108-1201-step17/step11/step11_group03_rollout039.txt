    def __init__(self, n):

        self.size = 2 * n - 1
        self.nnz_T = n * n
        self.total_nnz = self.nnz_T + self.size

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast LP alternating 
with Nesterov acceleration (scipy HiGHS), active-set QCQP polish, 
and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csc_matrix
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Compute 2*n*max_b/(sum_a)^2 locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


class FastLPAlternator:
    """Precomputes LP structure and solves best-response LPs quickly."""
    
    def __init__(self, n):
        self.n = n
        self.size = 2 * n - 1
        nnz_t = n * n
        total_nnz = nnz_t + self.size
        
        # Precompute sparsity pattern
        rows = np.empty(total_nnz, dtype=np.int32)
        cols = np.empty(total_nnz, dtype=np.int32)
        
        idx = 0
        for j in range(n):
            for i in range(n):
                rows[idx] = j + i
                cols[idx] = j
                idx += 1
        rows[nnz_t:] = np.arange(self.size)
        cols[nnz_t:] = n  # M column index
        
        self.rows = rows
        self.cols = cols
        self.nnz_t = nnz_t
        self.total_nnz = total_nnz
        
        # Precompute value indices (which element of 'a' goes into each T entry)
        self.val_idx = np.empty(nnz_t, dtype=np.int32)
        for idx in range(nnz_t):
            self.val_idx[idx] = rows[idx] - cols[idx]
        
        # Fixed arrays
        self.data = np.empty(total_nnz)
        self.bub = np.zeros(self.size)
        
        aeq = np.zeros((1, n + 1))
        aeq[0, :n] = 1.0
        self.aeq = aeq
        self.beq = np.array([1.0])
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        self.c_obj = c_obj
        
        self.bounds = [(0.0, None)] * (n + 1)
    
    def solve(self, a):
        """Solve min M s.t. T_a @ b <= M, sum(b)=1, b>=0, M>=0."""
        # Update data array
        self.data[:self.nnz_t] = a[self.val_idx]
        self.data[self.nnz_t:] = -1.0
        
        A_ub = csc_matrix((self.data, (self.rows, self.cols)),
                          shape=(self.size, self.n + 1))
        
        res = linprog(self.c_obj, A_ub=A_ub, b_ub=self.bub,
                      A_eq=self.aeq, b_eq=self.beq,
                      bounds=self.bounds, method='highs',
                      options={'presolve': True})
        
        if res.status != 0:
            return None, None
        
        b_val = np.maximum(res.x[:self.n], 0.0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        return b_val, float(max(0.0, res.x[self.n]))

def build_q_sparse(k, n):
    """Build sparse Q for conv(a,a)[k] = a^T Q a."""
    from scipy.sparse import lil_matrix
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_active_qcqp(a, rel_tol=1e-5, max_active=20, time_limit=60):
    """Solve active-set QCQP for polishing."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]
    if len(active) > max_active:
        active = active[np.argsort(b[active])[::-1][:max_active]]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    t0 = time.monotonic()
    for solver in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s = a_val.sum()
                if s < 1e-15:
                    return None, None
                return a_val / s, float(max(0, t_var.value))
        except Exception:
            continue
    
    return None, None


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
