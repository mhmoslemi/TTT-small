def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():

def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lse_func(z, n, T_temp, deadline):
    """Top-level LSE objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_temp)
    return obj


def lse_jac(z, n, T_temp, deadline):
    """Top-level LSE gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_temp)
    return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac,
                args=(n, stage_T, deadline),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def multiplicative_subgradient(a0, time_limit, n_steps):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.03
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.8
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))
