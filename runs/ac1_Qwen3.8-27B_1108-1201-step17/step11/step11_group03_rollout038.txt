import numpy as np

# ===== Helper Functions =====

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Compute the objective value."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def nesterov_lp(a0, T_precomputed, solver, time_limit, print_interval=500):

            break

        if b_new is None:
            break

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = normalize(y)

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = normalize(a)

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:

        if score_y < best_score:

            best_a = y.copy()

            else:

                y = y + noise
                y = normalize(y)
                stall_count = 0

            print(f"    NLP it={it+1} ({elapsed:.0f}s) score={best_score:.10f} stall={stall_count}")

    return best_a, best_score

def multiplicative_weights_polish(a0, time_limit, n_steps=50000):
    """Multiplicative weights (mirror descent) for the actual problem."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.02
    no_improve = 0
    min_eta = 1e-5

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: positions within 1% of max
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient: average of gradients over active set
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Multiplicative update
        log_scale = -eta * g
        log_scale -= log_scale.max()  # numerical stability
        a_new = a * np.exp(log_scale)
        a_new = normalize(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            eta *= 0.9
            no_improve = 0
            if eta < min_eta:
                break

    return best_a, best_score

def logsumexp_smoothed_obj(a, beta):
    """Compute LogSumExp-smoothed objective and gradient."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / beta
    exp_shifted = np.exp(shifted)
    lse = max_b + beta * np.log(exp_shifted.sum())
    obj = 2.0 * n * lse

    # Gradient
    w = exp_shifted / exp_shifted.sum()
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]

    return obj, grad

def lse_lbgfs(a0, time_limit, betas=[1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]):
    """L-BFGS on LogSumExp approximation with annealing."""
    n = len(a0)
    a = normalize(a0.copy())
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

        # Use scipy minimize with log-sum-exp objective
        # Parameterize in log-space for non-negativity
        z = np.log(np.maximum(a, 1e-300))

        def obj_func(z_val):
            a_val = softmax(z_val)
            o, _ = logsumexp_smoothed_obj(a_val, beta)
            return o

        def grad_func(z_val):
            a_val = softmax(z_val)
            _, g_a = logsumexp_smoothed_obj(a_val, beta)
            # Chain rule: d/dz = diag(a) @ (I - 11^T) @ g_a
            S = np.dot(g_a, a_val)
            g_z = a_val * (g_a - S)
            return g_z

        try:
            res = minimize(fun=obj_func, x0=z, jac=grad_func,
                          method='L-BFGS-B',
                          options={'maxiter': 200, 'ftol': 1e-20, 'gtol': 1e-15})
            a = softmax(res.x)
        except Exception:
            pass

    return a

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP alternating with Nesterov acceleration (main workhorse)
2. Multiplicative weights subgradient for the actual problem
3. Low-temperature LogSumExp L-BFGS for final precision
4. Broad n-sweep (1195-1205) to find optimal length
5. Coordinate-descent and final MW polish for last-mile improvement

Key improvements over previous: broader n-sweep, better step schedules,
coordinate descent polish, more efficient time allocation.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Compute the objective value locally."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize_vec(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_best_solver():

            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

def solve_best_response_lp(a_fixed, T_precomputed, solver):
    """Given fixed a, solve: min M s.t. T @ b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_precomputed @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for s in solvers:
        if s is None:
            continue
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                return normalize_vec(b_val)
        except Exception:
            continue
    return None

def multiplicative_weights_optimize(a0, time_limit, n_steps=80000):
    """Mirror descent (multiplicative weights) for the actual optimization problem."""
    n = len(a0)
    a = normalize_vec(a0.copy())

    t0 = time.monotonic()

    no_improve = 0
    min_eta = 5e-6

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Adaptive active set width
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Subgradient: average gradient over active positions
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Multiplicative update with step size adaptation
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)

            no_improve += 1

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def logsumexp_obj_and_grad(a, beta):
    """Compute LogSumExp-smoothed objective and gradient w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / beta
    exp_shifted = np.exp(shifted)
    lse = max_b + beta * np.log(exp_shifted.sum())
    obj = 2.0 * n * lse

    w = exp_shifted / exp_shifted.sum()
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-16:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]

    return obj, grad


def lse_lbgfs_optimize(a0, time_limit):
    """L-BFGS on LogSumExp approximation with temperature annealing."""
    n = len(a0)
    a = normalize_vec(a0.copy())
    t0 = time.monotonic()
    deadline = t0 + time_limit
    betas = [2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]

    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        z = np.log(np.maximum(a, 1e-300))
        max_iter = max(30, int(remaining * 3))

        def obj_z(z_val):
            a_val = softmax_fn(z_val)
            o, _ = logsumexp_obj_and_grad(a_val, beta)
            return o

        def grad_z(z_val):
            a_val = softmax_fn(z_val)
            _, g_a = logsumexp_obj_and_grad(a_val, beta)
            S = np.dot(g_a, a_val)
            return a_val * (g_a - S)

        try:
            res = minimize(fun=obj_z, x0=z, jac=grad_z,
                          method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-22,
                                   'gtol': 1e-16, 'maxfun': max_iter * 200})
            a = softmax_fn(res.x)
        except Exception:
            pass

    return a

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize_vec(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize_vec(f_interp)

        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
