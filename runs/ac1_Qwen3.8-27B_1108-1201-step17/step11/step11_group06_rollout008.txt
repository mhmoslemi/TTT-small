def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def evaluate_local(a):
    """Fast local evaluation assuming a >= 0, sum > 0."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()

def build_conv_matrix(y):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(y, b)."""
    n = len(y)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows.append(k)
            cols.append(j)
            vals.append(y[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def update_conv_matrix_values(T, y):
    """Update values of existing sparse matrix T to match new y."""
    n = len(y)
    size = 2 * n - 1
    T_coo = T.tocoo()
    new_vals = np.empty(len(T_coo.row), dtype=np.float64)
    for idx in range(len(T_coo.row)):
        k = T_coo.row[idx]
        j = T_coo.col[idx]
        new_vals[idx] = y[k - j]
    T.data = new_vals
    return T

def solve_lp_best_response(y, T_mat, solver):
    """Given fixed y (non-negative, sum=1), find b minimizing max_k conv(y,b)[k]."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / (exp_terms.sum() + 1e-300)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lse_func_factory(n, T, deadline):
    """Create LSE objective function."""
    def func(z):
        if time.monotonic() > deadline:
            return 1e15

        return obj
    return func

    return jac

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    # === Phase 1: Quick MW warmup ===
    print(f"\n=== Phase 1: MW warmup n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights_adaptive(best_seq, t1, n_steps=200000, eta0=0.08)
        try_update(a_mw, "mw_warm")

    # === Phase 2: Main LP alternating ===
    print(f"\n=== Phase 2: LP main n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 540.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver, n_max_iters=1000000)
        try_update(a_lp, "lp_main")

    # === Phase 3: Subgradient Nesterov (fast, no LP) ===
    print(f"\n=== Phase 3: SubGrad Nesterov (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.05, 90.0)
        a_sg, _ = subgradient_nesterov(best_seq, t3, n_steps=500000)
        try_update(a_sg, "subgrad")

    # === Phase 4: Palindromic LP ===
    print(f"\n=== Phase 4: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t4 = min(rem * 0.10, 150.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = nesterov_lp_refine(a_pal, t4, solver, n_max_iters=200000)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [900, 1000, 1400, 1500, 1800]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.12, 180.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln_mw, _ = multiplicative_weights_adaptive(init_ln, t_ln * 0.2, n_steps=50000, eta0=0.06)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.6)
        if t_lp > 30:
            start_lp = a_ln_mw if a_ln_mw is not None else init_ln
            a_ln_lp, _ = nesterov_lp_refine(start_lp, t_lp, solver, n_max_iters=200000)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 6: LSE L-BFGS fine polish ===
    print(f"\n=== Phase 6: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t6 = min(rem * 0.18, 300.0)
        T_stages = [8.0, 4.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        print(f"  {len(T_stages)} stages, budget={t6:.0f}s")
        a_lse = lbgfs_refine(best_seq, t6, T_stages, maxiter_per_stage=600)
        try_update(a_lse, "lse_fine")

    # === Phase 7: Spectral perturbation + re-optimize ===
    print(f"\n=== Phase 7: Spectral perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            a_pert = spectral_perturb(best_seq, rng, strength=0.01 + 0.01 * trial)
            t_sub = min(rem * 0.2, 60.0)
            if t_sub < 20:
                break
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=100000)
            try_update(a_opt, f"spec{trial}")

    # === Phase 8: More diverse restarts ===
    print(f"\n=== Phase 8: Diverse restarts (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            # Mix with a structured init
            n_cur = len(best_seq)
            a_struct = structured_init(n_cur, rng)
            a_mix = 0.7 * best_seq + 0.3 * a_struct
            a_mix = np.maximum(a_mix, 0)
            a_mix = a_mix / a_mix.sum()
            t_sub = min(rem * 0.25, 80.0)
            if t_sub < 20:
                break
            a_opt, _ = nesterov_lp_refine(a_mix, t_sub, solver, n_max_iters=100000)
            try_update(a_opt, f"mix{trial}")

    # === Phase 9: Final LP polish ===
    print(f"\n=== Phase 9: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.35, 180.0)
        a_fin_lp, _ = nesterov_lp_refine(best_seq, t9, solver, n_max_iters=200000)
        try_update(a_fin_lp, "final_lp")

    # === Phase 10: Final MW + LSE ===
    print(f"\n=== Phase 10: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t10_mw = min(rem * 0.3, 60.0)
        a_fin_mw, _ = multiplicative_weights_adaptive(best_seq, t10_mw, n_steps=100000, eta0=0.03)
        try_update(a_fin_mw, "final_mw")
        rem = deadline - time.monotonic()
        if rem > 30:
            t10_lse = min(rem * 0.5, 120.0)
            T_final = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_fin_lse = lbgfs_refine(best_seq, t10_lse, T_final, maxiter_per_stage=400)
            try_update(a_fin_lse, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
