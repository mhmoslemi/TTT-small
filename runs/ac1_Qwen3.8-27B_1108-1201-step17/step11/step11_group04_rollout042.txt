"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-n exploration with spectral initialization
2. Nesterov-accelerated mirror descent (multiplicative updates) with top-K targeting
3. L-BFGS with Lp-norm smoothing (numerically stable, p up to 4096)
4. Active-set refinement for final polish
5. Adaptive time allocation with verification

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_flatten(a, n_iters=200, blend=0.3, deadline=None):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    return a


def compute_gradient_lp(a, p):
    """
    Compute gradient of 2n * Lp_norm(c) where c = conv(a,a), sum(a)=1.
    Uses FFT-based cross-correlation for efficiency.
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    if max_c < 1e-300:
        return 2.0 * n * max_c, np.zeros(n)
    # Normalized: c_norm[k] = c[k] / max_c
    c_norm = c / max_c
    # Lp norm: max_c * (sum c_norm^p)^(1/p)
    c_norm_p = c_norm ** p
    S = c_norm_p.sum()
    obj = 2.0 * n * max_c * S ** (1.0 / p)
    if S < 1e-300:
        return obj, np.zeros(n)
    # Weights: w[k] = c_norm[k]^(p-1)
    w = c_norm ** (p - 1)
    # grad_a[j] = 2n * S^(1/p - 1) * max_c * (1/max_b) * 2 * sum_k w[k] * a[k-j]
    # = 2n * S^(1/p - 1) * 2 * sum_k w[k] * a[k-j]
    # The sum is cross-correlation of w with a
    # Compute via FFT
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = fft(w, fft_size)
    A = fft(np.zeros(fft_size), fft_size)  # placeholder
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    A = fft(a_padded, fft_size)
    # Cross-correlation: R[j] = sum_k w[k] * a[k-j]
    # R = ifft(fft(w) * conj(fft(a_padded)))
    R = ifft(W * np.conj(A), fft_size).real[:2 * n - 1]
    # grad_a[j] = 4n * S^(1/p - 1) * R[j] for j = 0, ..., n-1
    factor = 4.0 * n * S ** (1.0 / p - 1.0)
    grad_a = factor * R[:n]
    # Project to tangent space of simplex (sum a = 1, a > 0):
    # For the constrained problem, we use the gradient as-is in the multiplicative update.
    # But for L-BFGS in log-domain, we need the log-gradient.
    return obj, grad_a


def lp_logdomain_obj(z, n, p, deadline):
    """Lp objective in log-domain for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    obj, grad_a = compute_gradient_lp(a, p)
    # Convert to log-domain gradient: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lp_logdomain_func(z, n, p, deadline):
    obj, _ = lp_logdomain_obj(z, n, p, deadline)
    return obj


def lp_logdomain_jac(z, n, p, deadline):
    _, g = lp_logdomain_obj(z, n, p, deadline)
    return g


def nesterov_multiplicative(a0, time_limit, K=5, deadline=None):
    """
    Nesterov-accelerated mirror descent (multiplicative weights) targeting top-K peaks.
    """
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit

    momentum = 0.0
    base_momentum = 0.9
    eta = 0.05
    no_improve = 0
    scale = 1.0

    for step in range(500000):
        if time.monotonic() > deadline:
            break
        if step % 500 == 0:
            momentum = min(base_momentum, 0.5 + 0.4 * step / 2000.0)

        # Nesterov extrapolation
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Compute subgradient from top-K peaks
        c = fast_convolve(y)
        max_c = c.max()
        if max_c < 1e-15:
            break
        # Find top-K peaks
        K_eff = min(K, len(c))
        top_k = np.argpartition(c, -K_eff)[-K_eff:]
        threshold = c[top_k].min()
        active = np.where(c >= threshold * 0.999)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Weighted subgradient
        weights = c[active] - c[active].min()
        if weights.sum() < 1e-15:
            weights = np.ones(len(active))
        weights = weights / weights.sum()

        g = np.zeros(n)
        for i, k in enumerate(active):
            w_i = weights[i]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w_i * y[k - idx]

        # Center the gradient
        g_shifted = g - g.mean()

        # Multiplicative update on a (not y, for stability)
        eta_eff = eta * scale
        log_scale = -eta_eff * g_shifted
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            a_prev = a

        if no_improve >= 50:
            scale *= 0.8
            no_improve = 0
            if scale < 0.01:
                break

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovMult step={step} ({elapsed:.0f}s): best={best_score:.10f} scale={scale:.4f}")

    return best_a, best_score


def lbgfs_lp_refine(a0, time_limit, p_stages=None, deadline=None):
    """L-BFGS with Lp-norm smoothing, increasing p for better max approximation."""
    n = len(a0)
    if p_stages is None:
        p_stages = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096]
    a = normalize(np.asarray(a0, dtype=np.float64))
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(p_stages)

    for stage_i, p in enumerate(p_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        stages_left = n_stages - stage_i
        iters = max(30, min(500, int(remaining * 80 / stages_left)))
        print(f"    Lp-LBFGS stage {stage_i+1}/{n_stages}: p={p}, iters~{iters}, remaining={remaining:.0f}s")

        def func(z_in):
            return lp_logdomain_func(z_in, n, p, deadline)

        def jac(z_in):
            return lp_logdomain_jac(z_in, n, p, deadline)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
            a = np.exp(z - z.max())
            a = a / a.sum()
            sc = 2.0 * n * fast_convolve(a).max()
            print(f"      Stage {stage_i+1} done: p={p}, score={sc:.10f}")
        except Exception as e:
            print(f"      Stage {stage_i+1} error: {e}")

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    return a_final


def active_set_refine(a0, time_limit, deadline=None):
    """
    Refine using tight active set identification.
    Iteratively identify peaks at the max and apply targeted multiplicative updates.
    """
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit

    eta = 0.02
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        c = fast_convolve(a)
        max_c = c.max()

        # Tight active set: peaks within 1e-8 of max
        threshold = max_c * (1.0 - 1e-8)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Compute average gradient from active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Center
        g_shifted = g - g.mean()

        # Adaptive step size
        g_norm = np.sqrt(np.sum(g_shifted ** 2))
        if g_norm < 1e-15:
            break
        c_range = c.max() - c.min()
        eta_eff = eta * (c_range / max(g_norm, 1e-15))
        eta_eff = min(eta_eff, 0.5)

        log_scale = -eta_eff * g_shifted
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.9

        if it % 50000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    ActiveSet {elapsed:.0f}s: score={best_score:.10f} |active|={len(active)}")

    return best_a, best_score


def build_conv_matrix_sparse(a_fixed):
    """Build the convolution operator as a linear map (for LP)."""
    n = len(a_fixed)
    size = 2 * n - 1
    # Represent as: T @ b = conv(a_fixed, b)
    # T has shape (size, n), T[k, j] = a_fixed[k-j] if 0 <= k-j < n
    T = np.zeros((size, n))
    for j in range(n):
        ks = np.arange(j, j + n)
        T[ks, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix_sparse(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    return None
                return b_val / sv
        except Exception:
            continue
    return None


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=200000):
    """Nesterov dual averaging LP alternating."""
    n = len(a0)

    rng = np.random.default_rng(99999)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:

            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()

            best_a = a.copy()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:

            if (it + 1) % 2000 == 0:

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    else:
        a = 1.0 - np.abs(2 * x - 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_parent = len(best_seq)
    print(f"Starting: n={n_parent}, score={best_score:.10f}, budget={deadline - time.monotonic():.0f}s")

    # Determine which n values to try, prioritizing the parent's n
    n_candidates = sorted(set([n_parent, 800, 1000, 1200, 1500, 1800, 2000]))
    # Prioritize parent n and nearby values
    n_candidates = [n_parent] + [n for n in n_candidates if n != n_parent]

    # Phase 1: Intensive optimization on parent n
    print(f"\n=== Phase 1: Intensive optimization on n={n_parent} ===")
    t_phase1 = min((deadline - time.monotonic()) * 0.5, 800.0)
    a_opt, sc_opt = optimize_single_n(best_seq, n_parent, t_phase1, solver, deadline)
    if sc_opt < best_score:
        best_score = sc_opt
        best_seq = a_opt.copy()
    print(f"  Phase 1 done: best score = {best_score:.10f} (n={len(best_seq)})")

    # Phase 2: Try other n values with remaining time
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 2: Multi-n exploration (remaining={rem:.0f}s) ===")
    n_tried = {n_parent}
    for n_try in n_candidates[1:]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        if n_try in n_tried:
            continue
        # Allocate time based on how far this n is from parent
        dist_factor = 1.0 - 0.1 * abs(n_try - n_parent) / max(n_parent, 1)
        t_try = min(rem * 0.35 * dist_factor, 200.0)
        if t_try < 30:
            t_try = 30.0
        if rem < t_try + 30:
            break
        print(f"\n  Trying n={n_try} (budget={t_try:.0f}s)")
        init_try = interpolate_sequence(best_seq, n_try)
        a_try, sc_try = optimize_single_n(init_try, n_try, t_try, solver, deadline)
        n_tried.add(n_try)
        if sc_try < best_score:
            best_score = sc_try
            best_seq = a_try.copy()
            print(f"  NEW GLOBAL BEST: {best_score:.10f} (n={n_try})")
        else:
            print(f"  n={n_try}: score={sc_try:.10f} (not better than {best_score:.10f})")

    # Phase 3: Final intensive polish on the best
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 3: Final polish (remaining={rem:.0f}s) ===")
    if rem > 30:
        n_best = len(best_seq)
        # L-BFGS with very high p
        t_final_lbfgs = min(rem * 0.5, 120.0)
        if t_final_lbfgs > 10:
            p_fin = [1024, 2048, 4096, 8192, 16384, 32768]
            a_fin = lbgfs_lp_refine(best_seq, t_final_lbfgs, p_stages=p_fin,
                                    deadline=time.monotonic() + t_final_lbfgs)
            sc_fin = 2.0 * n_best * fast_convolve(a_fin).max()
            if sc_fin < best_score:
                best_score = sc_fin
                best_seq = a_fin.copy()
                print(f"  Final L-BFGS: {best_score:.10f}")

        # Active set
        rem = deadline - time.monotonic()
        if rem > 15:
            t_as = min(rem * 0.5, 60.0)
            a_as, _ = active_set_refine(best_seq, t_as, deadline=time.monotonic() + t_as)
            sc_as = 2.0 * n_best * fast_convolve(a_as).max()
            if sc_as < best_score:
                best_score = sc_as
                best_seq = a_as.copy()
                print(f"  Final active set: {best_score:.10f}")

        # Nesterov multiplicative
        rem = deadline - time.monotonic()
        if rem > 10:
            t_nm = min(rem * 0.5, 30.0)
            a_nm, _ = nesterov_multiplicative(best_seq, t_nm, K=3,
                                              deadline=time.monotonic() + t_nm)
            sc_nm = 2.0 * n_best * fast_convolve(a_nm).max()
            if sc_nm < best_score:
                best_score = sc_nm
                best_seq = a_nm.copy()
                print(f"  Final NesterovMult: {best_score:.10f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}, time={elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.10f}")
            if eval_score < best_score:
                best_score = eval_score
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
