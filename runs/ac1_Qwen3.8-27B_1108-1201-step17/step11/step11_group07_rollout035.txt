"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. FFT-accelerated subgradient with Nesterov momentum for fast bulk optimization
2. LP-based linearized refinement (HiGHS) for precise convergence near optimum
3. Spectral flattening to escape local minima
4. Multi-length exploration and diverse restarts
Key insight: the subgradient of max(conv(a,a)) is a cross-correlation of the
active-set indicator with a, computable in O(n log n) via FFT.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient_fft(a, active_lags):
    """Compute subgradient of max(conv(a,a)) using FFT.
    g[m] = 2 * sum_{k in active} a[k-m] * I(0<=k-m<n)
    This is 2 * cross_correlation(active_indicator, a).
    """
    n = len(a)
    v = np.zeros(2 * n - 1)
    v[active_lags] = 1.0
    # Normalize by number of active lags for stability
    v /= len(active_lags)

    # Cross-correlation: cross[m] = sum_k v[k] * a[k-m]
    # Using FFT: cross = ifft(fft(v) * conj(fft(a)))
    N = 1 << (3 * n).bit_length()
    V = fft(v, N)
    W = fft(a, N)
    cross = np.real(ifft(V * np.conj(W), N))[:n]
    return 2.0 * cross


def project_subgradient(g, a):
    """Project gradient onto the feasible direction space (a>=0, sum=1).
    The feasible directions are d with sum(d)=0 and d[m]>=0 where a[m]=0.
    We use a simple projection: remove the mean, and zero out components where a is near 0.
    """
    n = len(a)
    # Project onto sum(d)=0
    g = g - g.mean()
    # For components where a is very small, the constraint a>=0 is active
    # We set gradient to 0 there (projected gradient)
    mask = a < 1e-10
    if mask.any():
        g[mask] = 0.0
        # Re-project onto sum=0
        g = g - g.mean()
    return g


def subgradient_optimize(a0, time_limit, eta0=0.01, momentum_beta=0.9,
                         active_eps=1e-4, adapt_rate=0.999):
    """FFT-based subgradient optimization with Nesterov momentum."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    eta = eta0
    v_mom = np.zeros(n)  # momentum
    y = a.copy()  # Nesterov extrapolation
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        t_curr = it + 1
        beta_curr = (1 + np.sqrt(1 + 4 * t_curr * t_curr)) / 2
        y_new = y + ((t_curr - 1) / t_curr / beta_curr - 1) * (y - a)
        y_new = np.maximum(y_new, 0)
        
        # Compute autocorrelation of y (or a for simpler version)
        c = fast_convolve(a)
        c_max = c.max()

        # Active set: lags within eps of max
        threshold = c_max * (1.0 - active_eps)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Compute subgradient
        g = compute_subgradient_fft(a, active)

        # Project
        g = project_subgradient(g, a)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Nesterov momentum update
        v_mom = momentum_beta * v_mom + g / g_norm
        v_mom = project_subgradient(v_mom, a)
        v_mom_norm = np.linalg.norm(v_mom)
        if v_mom_norm < 1e-15:
            v_mom = g / g_norm
            v_mom_norm = 1.0

        # Adaptive step size
        # Optimal for subgradient: eta_t = c / sqrt(sum of past gradient norms)
        eta = eta0 / np.sqrt(1 + it * 0.01)

        # Try step
        a_try = a - eta * v_mom / v_mom_norm
        a_try = normalize(a_try)
        sc = evaluate_local(a_try)

        if sc < best_score:
            best_score = sc
            best_a = a_try.copy()
            a = a_try.copy()
            no_improve = 0
        else:
            # Try smaller step
            a_try = a - 0.5 * eta * v_mom / v_mom_norm
            a_try = normalize(a_try)
            sc = evaluate_local(a_try)
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()
                a = a_try.copy()
                no_improve = 0
            else:
                no_improve += 1

        y = y_new
        it += 1

        if no_improve > 200:
            eta *= 0.9
            momentum_beta = min(0.99, momentum_beta + 0.001)
            no_improve = 0
            if eta < 1e-8:
                break

    return best_a, best_score


def lp_refine(a0, time_limit, delta0=0.1, n_top_lags=30):
    """LP-based linearized refinement.
    Solves: min t s.t. c[k] + 2*conv(a,da)[k] <= t for top lags,
            a+da >= 0, sum(da)=0, |da|<=delta
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    delta = delta0
    t0 = time.monotonic()
    no_improve = 0

    # Precompute LP structure
    # Group 1: autocorrelation constraints (top lags only, rebuilt each iter)
    # Group 2: non-negativity: -da[m] <= a[m]
    # Group 3: step size: |da[m]| <= delta

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        c_max = c.max()

        # Select top lags for the LP
        sorted_lags = np.argsort(c)[::-1][:n_top_lags]

        # Build sparse constraint matrix
        rows = []
        cols = []
        data = []
        b_ub = []

        # Group 1: c[k] + 2*conv(a,da)[k] - t <= 0
        for k in sorted_lags:
            row_idx = len(b_ub)
            b_ub.append(-c[k])
            # -t coefficient
            rows.append(row_idx)
            cols.append(n)
            data.append(-1.0)
            # da[m] coefficients: 2*a[k-m] for valid m
            for m in range(n):
                km = k - m
                if 0 <= km < n:
                    rows.append(row_idx)
                    cols.append(m)
                    data.append(2.0 * a[km])

        # Group 2: -da[m] <= a[m]
        for m in range(n):
            row_idx = len(b_ub)
            rows.append(row_idx)
            cols.append(m)
            data.append(-1.0)
            b_ub.append(a[m])

        # Group 3: da[m] <= delta, -da[m] <= delta
        for m in range(n):
            row_idx = len(b_ub)
            rows.append(row_idx)
            cols.append(m)
            data.append(1.0)
            b_ub.append(delta)
            row_idx = len(b_ub)
            rows.append(row_idx)
            cols.append(m)
            data.append(-1.0)
            b_ub.append(delta)

        n_rows = len(b_ub)
        A_ub = csr_matrix((data, (rows, cols)), shape=(n_rows, n + 1))
        b_ub = np.array(b_ub)

        # Objective: minimize t
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        # Equality: sum(da) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        bounds = [(None, None)] * (n + 1)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 5})
            if res.success:
                da = res.x[:n]
                t_pred = res.x[n]

                # Take the step with line search
                improved = False
                for alpha in [1.0, 0.8, 0.5, 0.3, 0.1]:
                    a_try = a + alpha * da
                    a_try = normalize(a_try)
                    sc = evaluate_local(a_try)
                    if sc < best_score * (1 - 1e-10):
                        best_score = sc
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        no_improve = 0
                        break

                if not improved:
                    no_improve += 1
                    delta *= 0.8
                    if delta < 1e-8:
                        break
            else:
                no_improve += 1
                delta *= 0.9
        except Exception:
            no_improve += 1

        if no_improve > 50:
            break

    return best_a, best_score


def spectral_shaping(a0, time_limit, n_iters=5000):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n).bit_length()
    t0 = time.monotonic()
    lr = 0.003

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = fft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        # Direction to flatten P
        r = (P - P_mean) / (P_mean + 1e-15)
        H = np.zeros(fft_size, dtype=np.complex128)
        valid = P > 1e-12
        H[valid] = -r[valid] * np.conj(A[valid]) / (P[valid] + 1e-10)
        H[0] = 0.0  # Don't change DC
        h = np.real(ifft(H, fft_size))[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        improved = False
        for scale in [lr, lr * 0.3, lr * 0.1]:
            a_new = a + scale * h
            a_new = normalize(a_new)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if it % 500 == 0 and it > 0:
            lr *= 0.8

    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []

    candidates.append(normalize(a))

    candidates.append(normalize(a))

    # Random smooth

    a = np.zeros(n)
    for i in range(12):
        a += coeffs[i] * np.cos(2 * np.pi * (i + 1) * x + phases[i])
    a = np.maximum(a + 0.5, 0)
    candidates.append(normalize(a))

    # Uniform
    candidates.append(np.ones(n) / n)

    # PRNG-like ( random but smooth)
    a = np.abs(rng.normal(0, 1, n))
    # Smooth it
    kernel = np.ones(20) / 20
    a_smooth = np.convolve(a, kernel, mode='same')
    candidates.append(normalize(a_smooth))

    # Two-tone
    a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x + rng.uniform(0, 2 * np.pi))
    candidates.append(normalize(np.maximum(a, 0)))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)


def hybrid_optimize(a0, time_limit):

    n = len(a0)
    a = normalize(a0.copy())

    t0 = time.monotonic()

        ("subgrad", 0.30),
        ("spectral", 0.15),
        ("lp", 0.35),

        ("lp2", 0.10),
    ]

    for phase_name, frac in phases:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        t_alloc = min(rem * frac, rem * 0.5)
        if t_alloc < 3:
            continue

        if phase_name == "subgrad" or phase_name == "subgrad2":

        elif phase_name == "spectral":
            a_res, sc = spectral_shaping(a, t_alloc)
        elif phase_name == "lp" or phase_name == "lp2":
            delta = 0.05 if phase_name == "lp2" else 0.1
            a_res, sc = lp_refine(a, t_alloc, delta0=delta, n_top_lags=40)

        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
