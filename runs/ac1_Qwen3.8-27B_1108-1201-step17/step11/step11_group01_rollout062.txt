def fast_convolve(a):

    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve(a):

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix_rows(a_fixed, indices):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            a_r2, _ = peak_subgradient(a_r, min(5, max(1, t6 * 0.2)), n_steps=10000, eta_start=0.1, eta_end=0.002)
            try_update(a_r2, f"restart_{restart}_sg")
    print(f"  Phase 6 done: {best_score:.8f}")

    # Phase 7: Final LSE fine-tune (remaining)
    print(f"\n=== Phase 7: Final LSE (score={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 8:
        t7 = min(remaining * 0.7, 60.0)
        if t7 > 5:
            a_final = lse_optimize(best_seq, t7, T_start=20.0, T_end=0.1, n_stages=8)
            try_update(a_final, "final_lse")
    print(f"  Phase 7 done: {best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
