        # (conv(a + delta, a + delta))_p = max_b

**Faster peak equalization using FFT:**

The key equation is: `(conv(delta, a))_p = c_p` for peak positions p.

In the frequency domain: `FFT(delta) * FFT(a) = FFT(c_p_vector)` (where c_p_vector is zero except at peak positions).

def fourier_peak_equalize(a, iterations=1000):
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    fft_size = 2 * n - 1
    fft_size = 1 << (fft_size - 1).bit_length()  # next power of 2
    
    A_fft = rfft(a, fft_size)
    
    for it in range(iterations):
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find peak positions
        num_peaks = min(20, n)  # use top 20 peaks
        peak_indices = np.argsort(b)[-num_peaks:]
        
        # Build target: we want conv(a_new, a_new) to be flat at peaks
        # Linearized: conv(a, a)_p + conv(delta, a)_p + conv(a, delta)_p ≈ max_b
        # Since conv is symmetric: 2 * conv(delta, a)_p ≈ max_b - b_p
        
        target = np.zeros(fft_size)
        for p in peak_indices:
            target[p] = (max_b - b[p]) / 2
        
        target_fft = rfft(target, fft_size)
        
        # delta_fft = target_fft / A_fft
        # Need to handle division by zero
        with np.errstate(divide='ignore', invalid='ignore'):
            delta_fft = np.where(np.abs(A_fft) > 1e-15, target_fft / A_fft, 0)
        
        delta = irfft(delta_fft, fft_size)[:n]
        
        # Small step
        step_size = 0.1
        a_new = a + step_size * delta
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        score_new = 2 * n * fast_convolve(a_new).max()
        score_old = 2 * n * max_b
        
        if score_new < score_old:
            a = a_new
        else:
            step_size *= 0.5
        
    return a

    n = len(a_fixed)
    b = np.ones(n) / n  # initial guess
    active_constraints = set()
    
    for round in range(max_rounds):

        T_full = build_conv_matrix(a_fixed)  # (2n-1) x n
        # Current violation
        conv_vals = T_full @ b

            b = np.maximum(b, 0)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_constrained(a_fixed, solver, T_mat, constraint_indices):
    """Solve LP with only specified constraints (cutting plane)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_sub = T_mat[constraint_indices, :]
    constraints = [T_sub @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def solve_lp_full(a_fixed, solver, T_mat):
    """Solve the full LP with all constraints."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def lp_best_response_cutting_plane(a_fixed, solver, T_mat, max_rounds=30, top_k=40):
    """LP best response using constraint generation (cutting plane)."""
    n = len(a_fixed)
    size = 2 * n - 1
    b = np.ones(n) / n
    active_set = set()
    
    for rnd in range(max_rounds):
        conv_vals = np.asarray(T_mat @ b).flatten()
        max_val = conv_vals.max()
        # Add top constraints
        k = min(top_k, size)
        top_idx = np.argsort(conv_vals)[-k:]
        new_set = set(top_idx.tolist())
        
        if new_set == active_set and rnd > 0:
            break
        active_set = new_set
        sorted_idx = np.array(sorted(active_set), dtype=np.int32)
        
        b_new, M_val = solve_lp_constrained(a_fixed, solver, T_mat, sorted_idx)
        if b_new is None:
            b_new, M_val = solve_lp_full(a_fixed, solver, T_mat)
            if b_new is None:
                return None, None
            return b_new, M_val
        
        # Check for violated constraints
        conv_new = np.asarray(T_mat @ b_new).flatten()
        if conv_new.max() > M_val * (1 + 1e-6) + 1e-12:
            # Add violated constraints
            violated = np.where(conv_new > M_val + 1e-10)[0]
            active_set.update(violated.tolist())
            b = b_new
            continue
        
        b = b_new
    
    return b, M_val


def multiplicative_weights_adaptive(a0, time_limit, n_steps, eta0=0.05):
    """Fast multiplicative weights with adaptive backtracking step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.4)
            stall = 0
        else:
            eta *= 0.85
            stall += 1
            if stall > 50:
                break
    return best_a, best_score


def fourier_peak_equalize(a0, time_limit, max_iters=50000, n_peaks=None):
    """
    Fourier-domain peak equalization.
    Linearizes the convolution peak equalization condition and solves
    via FFT division. Fast O(n log n) per iteration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if n_peaks is None:
        n_peaks = min(30, n)
    
    fft_size = 2 * n - 1
    fft_size = 1 << (fft_size - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    step_size = 0.3
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Select peak positions
        k = min(n_peaks, len(b))
        peak_indices = np.argsort(b)[-k:]
        
        # Build target in Fourier domain
        target = np.zeros(fft_size, dtype=np.float64)
        for p in peak_indices:
            target[p] = 0.5 * (max_b - b[p])
        
        target_fft = rfft(target, fft_size)
        A_fft = rfft(a, fft_size)
        
        # Solve: FFT(delta) = target_fft / (2 * A_fft) with Tikhonov
        denom = 2.0 * np.abs(A_fft) ** 2
        tikhonov = np.max(denom) * 1e-10
        delta_fft = (target_fft * np.conj(A_fft)) / (denom + tikhonov)
        
        delta = irfft(delta_fft, fft_size)[:n]
        
        # Line search
        best_local = a.copy()
        best_local_score = best_score
        for trial_step in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1]:
            a_try = a + trial_step * delta
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            score_try = 2.0 * n * fast_convolve(a_try).max()
            if score_try < best_local_score:
                best_local_score = score_try
                best_local = a_try.copy()
        
        if best_local_score < best_score - 1e-14:
            a = best_local
            best_score = best_local_score
            best_a = a.copy()
            step_size = min(step_size * 1.05, 1.0)
            stall = 0
        else:
            step_size *= 0.7
            stall += 1
            if stall > 100:
                break
    
    return best_a, best_score

            best_a = a.copy()
        if score_y < best_score:

            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.02 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def logsumexp_obj_grad(a, n, T_temp):
    """Compute LSE objective and gradient w.r.t. a (normalized, sum=1)."""
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    return obj, 2.0 * n * grad_a


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    """L-BFGS with LogSumExp smoothing and temperature annealing (direct a-space)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit

    def func(z):
        if time.monotonic() > deadline:
            return 1e15
        aa = np.exp(z - z.max())
        aa = aa / aa.sum()
        obj, _ = logsumexp_obj_grad(aa, n, T_stages[0] if isinstance(T_stages, list) else T_stages)
        return obj

    for stage_idx, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        cur_T = stage_T

        def f_stage(z):
            if time.monotonic() > deadline:
                return 1e15
            aa = np.exp(z - z.max())
            aa = aa / aa.sum()
            obj, _ = logsumexp_obj_grad(aa, n, cur_T)
            return obj

        def jf_stage(z):
            if time.monotonic() > deadline:
                return np.zeros(len(z))
            aa = np.exp(z - z.max())
            aa = aa / aa.sum()
            obj, ga = logsumexp_obj_grad(aa, n, cur_T)
            S = np.dot(ga, aa)
            return 2.0 * n * aa * (ga - S)

        try:
            res = minimize(
                fun=f_stage, x0=z, jac=jf_stage, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': maxiter_per_stage * 1000,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        if (stage_idx + 1) % 5 == 0:
            aa = np.exp(z - z.max())
            aa = aa / aa.sum()
            sc = 2.0 * n * fast_convolve(aa).max()
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {stage_idx+1}/{len(T_stages)} T={stage_T}: score={sc:.8f} ({elapsed:.0f}s)")

    z_final = z
    a_final = np.exp(z_final - z_final.max())
    a_final = a_final / a_final.sum()
    return a_final


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        # Blackman-like
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 9:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 10:
        # Gaussian centered
        a = np.exp(-8 * (x - 0.5) ** 2)
    else:
        # Flat top
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def fista_simplex(a0, time_limit, T=0.1, max_iters=100000):
    """FISTA on the simplex for the LSE objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Estimate Lipschitz constant
    L = 8.0 * n  # rough upper bound
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Gradient of LSE at y
        b = fast_convolve(y)
        max_b = b.max()
        scaled = (b - max_b) / T
        scaled_shifted = scaled - scaled.max()
        exp_terms = np.exp(scaled_shifted)
        w = exp_terms / exp_terms.sum()
        
        # Gradient w.r.t. a
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w[k] * y[k - idx]
        grad = 2.0 * n * grad
        
        # Gradient step
        z = y - grad / L
        
        # Project onto simplex
        a_new = simplex_project(z)
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        a = a_new
        t_k = t_new
        
        # Track best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating with cutting plane, Fourier peak equalization,
    fine LSE L-BFGS, FISTA, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_best = len(best_seq)

    # === Phase 1: Quick MW warm start ===
    print(f"\n=== Phase 1: MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.03, 40.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights_adaptive(best_seq, t1, n_steps=300000, eta0=0.05)
        try_update(a_mw, "mw_1")

    # === Phase 2: LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 650.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver, n_max_iters=2000000)
        try_update(a_lp, "lp_1")

    # === Phase 3: Fourier peak equalization (NEW) ===
    print(f"\n=== Phase 3: Fourier Peak Equalization (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.08, 120.0)
    if t3 > 20:
        print(f"  Budget: {t3:.0f}s")
        a_fp, _ = fourier_peak_equalize(best_seq, t3, max_iters=200000, n_peaks=40)
        try_update(a_fp, "fourier_peq")
        # Run again with more peaks
        rem = deadline - time.monotonic()
        if rem > 30:
            a_fp2, _ = fourier_peak_equalize(best_seq, min(rem * 0.5, 60.0), max_iters=100000, n_peaks=80)
            try_update(a_fp2, "fourier_peq80")

    # === Phase 4: LSE L-BFGS with fine temperature ===
    print(f"\n=== Phase 4: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t4 = min(rem * 0.22, 400.0)
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        print(f"  {len(T_stages)} stages, budget={t4:.0f}s")
        a_lse = lbgfs_refine(best_seq, t4, T_stages, maxiter_per_stage=400)
        try_update(a_lse, "lse_fine")

    # === Phase 5: FISTA on simplex ===
    print(f"\n=== Phase 5: FISTA (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t5 = min(rem * 0.08, 100.0)
        a_fista, _ = fista_simplex(best_seq, t5, T=0.05, max_iters=500000)
        try_update(a_fista, "fista")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [1000, 1500, 2000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.12, 180.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln_mw, _ = multiplicative_weights_adaptive(init_ln, t_ln * 0.15, n_steps=100000, eta0=0.05)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.6)
        if t_lp > 30:
            a_ln_lp, _ = nesterov_lp_refine(a_ln_mw, t_lp, solver, n_max_iters=200000)
            try_update(a_ln_lp, f"lp_n{ln}")
        # Fourier
        rem = deadline - time.monotonic()
        if rem > 20 and a_ln_lp is not None:
            a_ln_f, _ = fourier_peak_equalize(a_ln_lp, min(rem * 0.3, 40.0), max_iters=50000, n_peaks=40)
            try_update(a_ln_f, f"fourier_n{ln}")

    # === Phase 7: Palindromic refinement ===
    print(f"\n=== Phase 7: Palindromic (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t7 = min(rem * 0.1, 100.0)
        a_pal = make_palindromic(best_seq)
        a_pal_fp, _ = fourier_peak_equalize(a_pal, t7 * 0.5, max_iters=50000, n_peaks=40)
        try_update(a_pal_fp, "pal_fourier")
        rem = deadline - time.monotonic()
        if rem > 30:
            a_pal_lp, _ = nesterov_lp_refine(a_pal, min(rem * 0.5, 50.0), solver, n_max_iters=100000)
            try_update(a_pal_lp, "pal_lp")

    # === Phase 8: Perturbation escape ===
    print(f"\n=== Phase 8: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            strength = 0.01 + 0.008 * trial
            a_pert = perturb_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem * 0.2, 50.0)
            a_opt, _ = fourier_peak_equalize(a_pert, t_sub * 0.6, max_iters=20000, n_peaks=30)
            try_update(a_opt, f"pert_f{trial}")
            rem = deadline - time.monotonic()
            if rem > 15:
                a_opt2, _ = nesterov_lp_refine(a_opt, t_sub * 0.4, solver, n_max_iters=50000)
                try_update(a_opt2, f"pert_lp{trial}")

    # === Phase 9: Final LP polish ===
    print(f"\n=== Phase 9: Final LP polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.4, 80.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t9, solver, n_max_iters=200000)
        try_update(a_fin, "final_lp")

    # === Phase 10: Final Fourier + MW ===
    print(f"\n=== Phase 10: Final Fourier+MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t10a = min(rem * 0.5, 40.0)
        a_ff, _ = fourier_peak_equalize(best_seq, t10a, max_iters=100000, n_peaks=50)
        try_update(a_ff, "final_fourier")
        rem = deadline - time.monotonic()
        if rem > 5:
            a_fm, _ = multiplicative_weights_adaptive(best_seq, min(rem * 0.5, 20.0), n_steps=100000, eta0=0.02)
            try_update(a_fm, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")
    else:
        print("\nWARNING: No valid sequence found, using uniform")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
