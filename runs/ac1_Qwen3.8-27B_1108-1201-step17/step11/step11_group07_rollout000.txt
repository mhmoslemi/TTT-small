import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_and_grad(a, T):
    """Compute LogSumExp-smoothed objective and gradient.
    a is normalized (sum=1). Returns (obj, grad) where
    obj = 2n * LSE(c, T), grad = d(obj)/da (unconstrained).
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse_val = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse_val

    w = exp_c / exp_c.sum()

    # Gradient: for each m, d(c_k)/da_m = a_{k-m} (if valid)
    # d(LSE)/da_m = sum_k w_k * a_{k-m} (summed over valid k)
    # = cross-correlation of w and a
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    grad = 2.0 * n * R[:n]

    return obj, grad


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in log-parametrization."""
    a = softmax_from_z(z)
    obj, grad_a = lse_obj_and_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

    c_mean = c.mean()

    # Start hot, end cold
    T_start = spread * 2.0
    T_end = spread * 0.01
    if T_end < 1e-8:
        T_end = 1e-8
    return np.geomspace(T_start, T_end, n_stages)


def lbfgs_with_adaptive_T(a0, time_limit, max_stages=25):
    """L-BFGS with adaptive temperature schedule."""
    n = len(a0)
    z = z_from_a(a0.copy())
    t0 = time.monotonic()
    best_z = z.copy()
    best_score = evaluate_local(a0)

    # Compute initial autocorrelation for temperature
    a_init = normalize(a0)
    c_init = fast_convolve(a_init)
    T_stages = adaptive_temperature(c_init, max_stages)

    for stage_i, stage_T in enumerate(T_stages):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 3:
            break

        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        iters = max(20, int(remaining * 50 / (max_stages - stage_i)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-18, 'maxfun': iters * 100,
                         'maxcor': 20, 'disp': False}
            )
            z = res.x
            a_new = softmax_from_z(z)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
        except Exception:
            pass

    return softmax_from_z(best_z), best_score


def peak_directed_search(a0, time_limit, n_steps=50000):
    """Targeted local search that identifies autocorrelation peaks
    and transfers mass from high-contributing to low-contributing elements.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(789)

    step_size = 0.002

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # Find top peaks (within 99.5% of max)
        threshold = c_max * 0.995
        peak_indices = np.where(c >= threshold)[0]

        if len(peak_indices) == 0:
            peak_indices = np.array([n - 1])

        # For each peak, identify the top contributing pairs
        improved = False
        for k in peak_indices[:5]:  # Focus on top 5 peaks
            # Contributing pairs: (j, k-j) for valid j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js >= je:
                continue

            # Contribution of each j to c_k
            contribs = np.zeros(je - js + 1)
            for idx, j in enumerate(range(js, je + 1)):
                contribs[idx] = a[j] * a[k - j]

            # Sort by contribution (descending)
            order = np.argsort(contribs)[::-1]
            top_j = range(js + order[0], je + 1)

            # Try transferring from high contributor to a low position
            for trial in range(3):
                if time.monotonic() - t0 > time_limit:
                    break

                j_high = js + order[trial % len(order)]
                # Pick a random low position (far from the peak's contribution)
                j_low = rng.integers(0, n)
                if j_low == j_high:
                    continue

                # Transfer amount
                amt = step_size * a[j_high]
                if amt < 1e-12:
                    continue

                a_new = a.copy()
                a_new[j_high] -= amt
                a_new[j_low] += amt
                if a_new[j_high] < 0:
                    a_new[j_high] = 0
                    a_new[j_low] += amt  # keep sum

                a_new = normalize(a_new)
                sc = evaluate_local(a_new)

                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True
                    break

        if not improved:
            # Try a broader random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                amt = step_size * 0.5 * a.max()
                a_new = a.copy()
                a_new[i] -= amt
                a_new[j] += amt
                a_new = np.maximum(a_new, 0)
                a_new = normalize(a_new)
                sc = evaluate_local(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True

        # Adapt step size
        if step % 100 == 0:
            if improved:
                step_size = min(0.01, step_size * 1.1)
            else:
                step_size *= 0.85

        if step % 5000 == 0 and step > 0:
            print(f"    PeakSearch step={step}: score={best_score:.8f}, step_size={step_size:.5f}")

    return best_a, best_score


def pairwise_coordinate_descent(a0, time_limit, n_rounds=200):
    """Coordinate-style descent: for each position, find optimal value
    by 1D search, while adjusting one other position to maintain sum=1.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(321)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick a random position to optimize
        j = rng.integers(0, n)
        # Pick a partner to adjust
        partner = rng.integers(0, n)
        if partner == j:
            partner = (j + 1) % n

        # Try different values for a[j]
        current_val = a[j]
        best_local_val = current_val
        best_local_score = best_score

        # Grid search around current value
        for frac in [0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 0.3, 0.1, 2.0, 0.0]:
            new_val_j = max(0, current_val * frac)
            # Adjust partner to maintain sum
            diff = current_val - new_val_j
            new_val_partner = a[partner] + diff
            if new_val_partner < 0:
                continue

            a_new = a.copy()
            a_new[j] = new_val_j
            a_new[partner] = new_val_partner
            a_new = normalize(a_new)
            sc = evaluate_local(a_new)

            if sc < best_local_score:
                best_local_score = sc
                best_local_val = new_val_j

        if best_local_score < best_score:
            # Reconstruct the best
            current_val = a[j]
            diff = current_val - best_local_val
            a[partner] = max(0, a[partner] + diff)
            a[j] = best_local_val
            a = normalize(a)
            best_score = best_local_score
            best_a = a.copy()

    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)

    # Gaussian bump
    candidates.append(normalize(np.exp(-8 * (x - 0.5) ** 2)))
    # Raised cosine
    candidates.append(normalize(0.5 * (1 + np.cos(2 * np.pi * x))))
    # Sine squared
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    # Uniform
    candidates.append(np.ones(n) / n)
    # Two bumps
    candidates.append(normalize(np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)))
    # Power law
    candidates.append(normalize((x * (1 - x)) ** 0.3))
    # Edge peaked
    candidates.append(normalize(np.exp(-3 * x) + np.exp(-3 * (1 - x))))
    # Random smooth
    coeffs = rng.normal(0, 1, 8)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 0.5, 0)
    candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def coarse_to_fine(a_best, time_limit, small_n_list=[200, 400, 600]):
    """Solve smaller problems, interpolate to larger, refine."""
    best_a = a_best.copy()
    best_score = evaluate_local(a_best)
    n_target = len(a_best)
    t0 = time.monotonic()

    for small_n in small_n_list:
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break

        # Interpolate best to small_n
        a_small = interpolate_sequence(best_a, small_n)
        t_sub = min(rem * 0.3, 40.0)

        # Optimize at small_n
        a_opt_small, sc_small = lbfgs_with_adaptive_T(a_small, t_sub, max_stages=10)
        print(f"    Coarse n={small_n}: score={sc_small:.8f}")

        # Interpolate back to target
        a_interp = interpolate_sequence(a_opt_small, n_target)
        sc_interp = evaluate_local(a_interp)
        if sc_interp < best_score:
            best_score = sc_interp
            best_a = a_interp.copy()

    return best_a, best_score


def symmetry_breaking(a0, time_limit, n_tries=100):
    """Try to break symmetries in the autocorrelation by
    applying controlled asymmetric perturbations.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    for trial in range(n_tries):
        if time.monotonic() - t0 > time_limit:
            break

        # Try different types of asymmetry
        method = trial % 4
        if method == 0:
            # Linear tilt
            x = np.linspace(0, 1, n)
            tilt = rng.uniform(-0.1, 0.1)
            a_pert = a * (1 + tilt * (x - 0.5))
        elif method == 1:
            # Quadratic asymmetry
            x = np.linspace(0, 1, n)
            quad = rng.uniform(-0.2, 0.2)
            a_pert = a * (1 + quad * (x - 0.5) ** 2 * 4)
        elif method == 2:
            # Random low-freq modulation
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.linspace(0, 1, n)
            mod = 1 + 0.05 * np.cos(2 * np.pi * freq * x + phase)
            a_pert = a * mod
        else:
            # Random perturbation
            a_pert = a + rng.normal(0, 0.01, n)

        a_pert = np.maximum(a_pert, 0)
        a_pert = normalize(a_pert)
        sc = evaluate_local(a_pert)

        if sc < best_score:
            best_score = sc
            best_a = a_pert.copy()
            a = a_pert.copy()
            print(f"    SymBreak trial={trial}: score={best_score:.8f}")

    return best_a, best_score


def multiplicative_update(a0, time_limit, n_steps=30000, eta0=0.05):
    """Multiplicative weights with adaptive step size and active set."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # Active set: lags near the max
        threshold = c_max * (1.0 - 1e-4)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([n - 1])

        # Gradient: weighted sum of contributions from active lags
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for m in range(js, je + 1):
                g[m] += a[k - m]
        g /= len(active)

        # Center the gradient (to preserve sum)
        g_centered = g - np.dot(g, a)  # tangent to the simplex

        # Multiplicative update
        for scale in [1.0, 0.5, 0.25, 0.1]:
            log_update = -eta * scale * g_centered
            log_update -= log_update.max()
            a_new = a * np.exp(log_update)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                no_improve = 0
            break

        no_improve += 1
        if no_improve >= 30:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-7:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines adaptive LSE + L-BFGS, peak-directed local search,
    multiplicative updates, coarse-to-fine, and symmetry breaking.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # --- Load parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    print(f"Starting: n={len(best_seq)}, score={best_score:.8f}")

    # === Phase 1: Quick L-BFGS warm-up ===
    print(f"\n=== Phase 1: L-BFGS warm-up ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.08, 100.0)
    if t1 > 15:
        a1, _ = lbfgs_with_adaptive_T(best_seq, t1, max_stages=12)
        update_best(a1, "lbfgs_warmup")

    # === Phase 2: Symmetry breaking + refine ===
    print(f"\n=== Phase 2: Symmetry breaking ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.05, 60.0)
    if t2 > 10:
        a2, _ = symmetry_breaking(best_seq, t2, n_tries=200)
        update_best(a2, "symbreak")
        if best_score < 1.51:
            a2b, _ = lbfgs_with_adaptive_T(best_seq, min(rem * 0.05, 60.0), max_stages=10)
            update_best(a2b, "lbfgs_post_sym")

    # === Phase 3: Peak-directed local search ===
    print(f"\n=== Phase 3: Peak-directed search ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        a3, _ = peak_directed_search(best_seq, t3, n_steps=80000)
        update_best(a3, "peaksearch_1")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    n_tries = [n_best]
    if n_best > 1500:
        n_tries.append(int(n_best * 0.75))
    n_tries.extend([int(n_best * 1.25), int(n_best * 1.5), int(n_best * 2.0)])
    # Filter valid lengths
    n_tries = [n for n in n_tries if 400 <= n <= 5000]
    # Remove duplicates
    n_tries = list(dict.fromkeys(sorted(set(n_tries))))

    for n_try in n_tries:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        if n_try == n_best:
            continue  # Already optimized

        t_sub = min(rem * 0.15, 120.0)
        if t_sub < 20:
            t_sub = min(rem * 0.2, 80.0)

        print(f"  Try n={n_try} (budget={t_sub:.0f}s)")
        a_init = interpolate_sequence(best_seq, n_try)
        a_try, sc_try = lbfgs_with_adaptive_T(a_init, t_sub, max_stages=15)
        update_best(a_try, f"length_{n_try}")

    # === Phase 5: Main L-BFGS on best ===
    print(f"\n=== Phase 5: Main L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.25, 350.0)
    if t5 > 30:
        a5, _ = lbfgs_with_adaptive_T(best_seq, t5, max_stages=30)
        update_best(a5, "lbfgs_main")

    # === Phase 6: Multiplicative update ===
    print(f"\n=== Phase 6: Multiplicative update (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 120.0)
    if t6 > 15:
        a6, _ = multiplicative_update(best_seq, t6, n_steps=20000, eta0=0.03)
        update_best(a6, "mult_update")

    # === Phase 7: Peak-directed search round 2 ===
    print(f"\n=== Phase 7: Peak-directed search 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.15, 180.0)
    if t7 > 20:
        a7, _ = peak_directed_search(best_seq, t7, n_steps=100000)
        update_best(a7, "peaksearch_2")

    # === Phase 8: Pairwise coordinate descent ===
    print(f"\n=== Phase 8: Pairwise CD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.08, 80.0)
    if t8 > 10:
        a8, _ = pairwise_coordinate_descent(best_seq, t8, n_rounds=5000)
        update_best(a8, "pairwise_cd")

    # === Phase 9: Coarse-to-fine ===
    print(f"\n=== Phase 9: Coarse-to-fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t9 = min(rem * 0.10, 120.0)
    if t9 > 30:
        a9, _ = coarse_to_fine(best_seq, t9, small_n_list=[200, 400])
        update_best(a9, "coarse_fine")

    # === Phase 10: Final polish ===
    print(f"\n=== Phase 10: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Short L-BFGS with very low T
        a10, _ = lbfgs_with_adaptive_T(best_seq, min(rem * 0.4, 80.0), max_stages=15)
        update_best(a10, "final_lbfgs")

        # Very short peak search
        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            a11, _ = peak_directed_search(best_seq, min(rem2 * 0.5, 40.0), n_steps=20000)
            update_best(a11, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
