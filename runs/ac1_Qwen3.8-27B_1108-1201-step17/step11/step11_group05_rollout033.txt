def compute_smooth_max_and_grad(a, temperature, top_k_frac=0.15):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Uses LogSumExp with top-k focusing for efficiency.
    Returns (smooth_max_value, gradient_vector).
    """
    n = len(a)
    conv = fast_convolve(a)
    L = len(conv)  # 2n-1

    # Determine top-k
    k = max(10, int(top_k_frac * L))
    if k < L:
        part_idx = np.argpartition(conv, -k)[-k:]
        top_vals = conv[part_idx]
        max_top = top_vals.max()
        exp_top = np.exp((top_vals - max_top) / temperature)
        w_top = exp_top / exp_top.sum()
        smooth_max = max_top + temperature * math.log(exp_top.sum())
        # Construct full weight vector
        w = np.zeros(L)
        w[part_idx] = w_top
    else:
        max_conv = conv.max()

        w = exp_all / exp_all.sum()

    grad = 2.0 * np.correlate(w, a, mode='full')[0:n]

    return smooth_max, grad


def compute_max_conv_grad(a, active_fraction=0.05):
    """
    Compute gradient using only the active set (peaks near max).
    Returns (max_value, gradient_vector).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()

    # Active set: positions where conv is within active_fraction of max
    threshold = max_val * (1.0 - active_fraction)
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])

    # Uniform weights on active set
    w = np.zeros(len(conv))
    w[active] = 1.0 / len(active)

    # Gradient
    grad = 2.0 * np.correlate(w, a, mode='full')[0:n]

    return max_val, grad


def score_sequence(a):
    """Compute the objective value for sequence a."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_best_response(a_fixed, solver_name, timeout=30.0):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]
    subject to b >= 0, sum(b) = 1.
    Returns (b_optimal, M_optimal) or (None, None) on failure.
    """
    if not HAS_CVXPY:
        return None, None

    a = np.asarray(a_fixed, dtype=np.float64)
    n = len(a)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    T = build_conv_matrix_sparse(a)

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver_name]
    if solver_name != cp.PDLP:
        solvers_to_try.append(cp.PDLP)
    if solver_name != cp.CLARABEL:
        solvers_to_try.append(cp.CLARABEL)
    solvers_to_try.extend([cp.SCS, cp.ECOS])

    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                s_val = b_val.sum()
                if s_val < 1e-15:
                    continue
                b_val = b_val / s_val
                return b_val, float(max(0.0, M_var.value))
        except Exception:
            continue

    return None, None


def find_lp_solver():
    """Find a working LP solver."""
    if not HAS_CVXPY:
        return None
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return None


def adam_logsumexp(a0, time_limit, T_stages, lr_init=0.005, lr_decay=0.9,
                   top_k_frac=0.15, verbose=True):
    """
    Adam optimization in log-space with temperature annealing.
    a0: initial sequence (will be normalized)
    time_limit: seconds to run
    T_stages: list of temperature values to sweep through
    Returns: (best_a, best_score)
    """
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-30))  # log-space parameter

    t0 = time.monotonic()
    deadline = t0 + time_limit

    best_a = a.copy()
    best_score = score_sequence(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0

    total_stages = len(T_stages)
    steps_per_stage = max(1000, int(time_limit * 30000 / total_stages))

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1.0:
            break

        for s in range(steps_per_stage):
            if time.monotonic() > deadline:
                break

            # Convert z to a
            z_shift = z - z.max()
            exp_z = np.exp(z_shift)
            a = exp_z / exp_z.sum()

            # Compute objective and gradient
            if T_stage < 0.01:
                max_val, grad_a = compute_max_conv_grad(a, active_fraction=0.02)
                obj = max_val
            else:
                smooth_val, grad_a = compute_smooth_max_and_grad(a, T_stage, top_k_frac)
                obj = smooth_val

            score = 2.0 * n * obj
            step += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Gradient in z-space: grad_z = a * (grad_a - a^T grad_a)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            # Adam update
            lr = lr_init * (lr_decay ** stage_idx) * (1.0 / (1.0 + 0.05 * s / steps_per_stage))
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** (step + 1))
            v_hat = v / (1.0 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Center z (doesn't change softmax)
            z -= z.mean()

            # Progress
            if verbose and (step + 1) % 10000 == 0:
                elapsed = time.monotonic() - t0
                cur_score = 2.0 * n * fast_convolve(normalize(np.exp(z - z.max()) / np.exp(z - z.max()).sum())).max()
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {step+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver_name, verbose=True):
    """
    Nesterov dual averaging LP alternating.
    a0: initial sequence
    time_limit: seconds
    solver_name: cvxpy solver to use
    Returns: (best_a, best_score)
    """
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_sequence(a)
    t0 = time.monotonic()

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # LP step: find best response b to current y
        b_new, M_b = lp_best_response(y, solver_name)
        if b_new is None:
            break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = normalize(y)

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = normalize(a)

        # Track best
        score_a = score_sequence(a)
        score_y = score_sequence(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if verbose and (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it+1} ({elapsed:.0f}s): best={best_score:.8f}")

        # Stall detection
        if (it + 1) % 100 == 0:
            # Try a perturbation if stuck
            if it > 200 and (it % 200 == 0):
                rng = np.random.default_rng(it)
                noise = 0.02 * np.cos(rng.uniform(1, 8) * np.linspace(0, 2 * np.pi, n) + rng.uniform(0, 2 * np.pi))
                y_pert = y + noise
                y = normalize(y_pert)

    return best_a, best_score


def multiplicative_weights(a0, time_limit, verbose=True):
    """
    Mirror descent (multiplicative weights) on active set.
    Returns: (best_a, best_score)
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_sequence(a)
    t0 = time.monotonic()

    eta = 0.05
    no_improve = 0

    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        max_b = conv.max()

        # Active set
        threshold = max_b * (1.0 - max(0.001, 0.05 / (1 + step / 500)))
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Gradient
        w = np.zeros(len(conv))
        w[active] = 1.0 / len(active)
        g = 2.0 * np.correlate(w, a, mode='full')[0:n]

        # Multiplicative update
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = normalize(a_new)

        score = score_sequence(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            eta *= 0.7
            no_improve = 0
            if eta < 0.001:
                break

        if verbose and step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MW step {step} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.4f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-6 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.35 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 2 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.6 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.4 + 0.3 * np.cos(2 * np.pi * 7 * x) + 0.2 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + np.exp(-8 * (x - 0.5)))
    elif choice == 8:
        # Fourier-based
        a = 1.0
        for k in range(1, 5):
            a += 0.3 / k * np.cos(2 * np.pi * k * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
    else:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x + 0.5) * np.cos(2 * np.pi * 1.5 * x)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s


def perturbed_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 12)
    freq2 = rng.integers(1, 20)
    phase1 = rng.uniform(0, 2 * np.pi)
    phase2 = rng.uniform(0, 2 * np.pi)
    perturbation = strength * (np.cos(freq1 * x + phase1) + 0.5 * np.sin(freq2 * x + phase2))
    a_pert = a + perturbation
    return normalize(a_pert)

    def local_update(a, label):

            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)

        if s < 0.01:
            return
        a = a / s
        sc = score_sequence(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc

    print(f"\n=== Phase 1: Multi-n Exploration ===")
    n_candidates = [1200, 1400, 1600]
    if parent is not None:
        n_candidates.insert(0, len(parent))
    # Remove duplicates while preserving order

    for nc in n_candidates:
        if nc not in seen and 500 <= nc <= 2000:
            seen.add(nc)
            n_candidates_dedup.append(nc)
    n_candidates = n_candidates_dedup

    print(f"  Trying n values: {n_candidates}")
    n_results = {}

        # Quick Adam
