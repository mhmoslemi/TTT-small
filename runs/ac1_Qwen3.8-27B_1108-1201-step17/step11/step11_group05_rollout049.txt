```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve_ab(a, b):
    """Compute convolve(a, b) where both have length n."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Equivalently, column j of T is a shifted version of a
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T
```

```python
def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n

    rng = np.random.default_rng(12345)

    # Polyak averaging buffer
    polyak_window = 200
    polyak_buf = []

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP
        lp_start = time.monotonic()
        b_star, t_val = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - lp_start
        lp_times.append(lp_time)

        if b_star is None:
            b_star, t_val = solve_lp_best_response(a, solver)
            if b_star is None:
                break

        # Check if LP is too slow
        if len(lp_times) > 30:
            avg_lp = np.mean(lp_times[-30:])
            if avg_lp > 10.0:
                break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_star
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_star) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Polyak averaging (average recent iterates)
        polyak_buf.append(b_star.copy())
        if len(polyak_buf) > polyak_window:
            polyak_buf.pop(0)

        # Evaluate scores
        score_a = normalized_score(a)
        score_y = normalized_score(y)

        # Also evaluate Polyak average
        if len(polyak_buf) >= 50:
            a_polyak = np.mean(polyak_buf, axis=0)
            a_polyak = np.maximum(a_polyak, 0)
            sp = a_polyak.sum()
            if sp > 1e-15:
                a_polyak = a_polyak / sp
                score_polyak = normalized_score(a_polyak)
                if score_polyak < best_score:
                    best_score = score_polyak
                    best_a = a_polyak.copy()
            else:
                score_polyak = float('inf')
        else:
            score_polyak = float('inf')

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection and restart
        if it % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 30:
                # Perturb to escape
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                strength = 0.02 * min(stall_count / 30.0, 3.0)
                y = y + strength * np.cos(freq * x + phase)
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                a = a + 0.5 * strength * np.cos(freq * x + phase)
                a = np.maximum(a, 0)
                sa2 = a.sum()
                if sa2 > 1e-15:
                    a = a / sa2
                stall_count = 0
                polyak_buf = []

            if it % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      LP-alt iter={it} ({elapsed:.0f}s): best={best_score:.8f} avg_lp={np.mean(lp_times[-30:])*1000:.1f}ms")

    return best_a, best_score


def subgradient_polish(a_init, time_limit, n_steps=500000):
    """
    Subgradient descent on the non-smooth objective max_k conv(a,a)[k]
    with sum(a)=1, a>=0 constraint via projection.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = normalized_score(a)

    eta = 0.01
    no_improve = 0
    min_eta = 1e-5

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find active set (peaks near maximum)
        threshold = max_b * (1.0 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Subgradient: g[i] = (2/|S|) * sum_{k in S} a[k-i] (valid indices only)
        g = np.zeros(n)
        for k in active:
            # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end >= i_start:
                idx = np.arange(i_start, i_end + 1)
                g[idx] += a[k - idx]
        g = g * (2.0 / len(active))

        # Gradient step + projection onto simplex {a >= 0, sum(a) = 1}
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = normalized_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            eta *= 0.9
            no_improve = 0
            if eta < min_eta:
                break

    return best_a, best_score


def adam_logspace(a_init, time_limit, solver=None):
    """
    Adam optimization in log-space with LP-derived gradient.
    a[i] = exp(z[i]) / sum(exp(z))
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-30).copy()
    s = a.sum()
    a = a / s
    z = np.log(a)
    z -= z.mean()  # center

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = normalized_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    T_temp = 5.0
    T_min = 0.01

    while time.monotonic() - t0 < time_limit:
        step += 1
        a = np.exp(z - z.max())
        a = a / a.sum()

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Temperature annealing
        progress = min(1.0, step / 10000)
        T_temp = max(T_min, 5.0 * (0.999 ** step))

        # Compute weighted active set
        if T_temp < 0.05:
            threshold = max_b * (1.0 - 1e-4)
            active = np.where(b >= threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(b)])
            w = np.zeros(2 * n - 1)
            w[active] = 1.0 / len(active)
        else:
            max_idx = np.argmax(b)
            exp_terms = np.exp((b - b[max_idx]) / T_temp)
            w = exp_terms / exp_terms.sum()

        # Gradient in a-space: g_a[i] = 2 * sum_k w[k] * a[k-i]
        # This is 2 * correlate(w, a) at the right offset
        g_a = np.correlate(w, a, mode='full')[n - 1:2 * n - 1] * 2.0

        # Convert to z-space gradient
        S = np.dot(g_a, a)
        grad_z = a * (g_a - S)

        # Adam update
        lr = 0.005 / (1.0 + 0.05 * step / 1000)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()

        # Periodic logging
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam step={step} ({elapsed:.0f}s): best={best_score:.8f} T={T_temp:.4f}")

    return best_a, best_score


def get_parent_sequence():
    """Try to access the parent sequence."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP best-response alternating with Nesterov acceleration,
    Adam log-space optimization, subgradient polish, and multi-start.
    """
    t_global_start = time.monotonic()
    deadline = t_global_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    solver = find_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s:.0f}s, deadline in {deadline - time.monotonic():.0f}s")

    # Get parent
    parent = get_parent_sequence()
    best_score = float('inf')
    best_a = None
    best_n = 0

    def try_update(a, label):
        nonlocal best_score, best_a, best_n
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            return
        a = a / s
        sc = normalized_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_n = len(a)
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: score={sc:.8f} n={len(a)} (t={elapsed:.0f}s) ***")

    # Determine initial sequences to try
    configs = []

    if parent is not None:
        n_par = len(parent)
        print(f"Parent: n={n_par}, score={normalized_score(parent):.8f}")
        configs.append(('parent', parent))
        # Also try interpolated versions
        for n_try in [1500, 2000]:
            if n_try != n_par:
                configs.append((f'parent_interp_n{n_try}', interpolate_sequence(parent, n_try)))
    else:
        print("No parent available, using structured inits")
        for n_try in [1200, 1500, 2000]:
            configs.append((f'uniform_n{n_try}', init_uniform(n_try)))

    # Add structured inits for the primary length
    primary_n = len(configs[0][1]) if configs else 1200
    structured_inits = [
        ('half_cos', init_half_cosine(primary_n)),
        ('raised_cos', init_raised_cosine(primary_n)),
        ('dual_peak', init_dual_peak(primary_n)),
        ('ripple', init_ripple(primary_n, 3, 0.3)),
        ('smoothed_barker', init_smoothed_barker(primary_n)),
    ]
    configs.extend(structured_inits)

    print(f"Total configs to try: {len(configs)}")

    # Phase 1: Quick scoping - run short LP alternating on all configs
    print(f"\n=== Phase 1: Quick Scoping ===")
    scope_time = min(30.0, (deadline - time.monotonic()) * 0.1)
    scores_by_config = []
    for label, a_init in configs:
        rem = deadline - time.monotonic()
        if rem < 5:
            break
        t_alloc = min(scope_time / len(configs) + 5, rem * 0.5)
        a_opt, sc = lp_alternating_nesterov(a_init, t_alloc, solver)
        scores_by_config.append((sc, label, a_opt, a_init))
        print(f"    {label}: score={sc:.8f}")

    # Sort by score
    scores_by_config.sort(key=lambda x: x[0])
    print(f"  Best from scoping: {scores_by_config[0][1]} -> {scores_by_config[0][0]:.8f}")

    # Phase 2: Deep optimization on top configs
    print(f"\n=== Phase 2: Deep LP Alternating ===")
    top_configs = scores_by_config[: min(5, len(scores_by_config))]
    rem = deadline - time.monotonic()
    deep_time = rem * 0.45
    per_config_time = deep_time / len(top_configs)

    for i, (sc, label, a_opt_scoped, a_init) in enumerate(top_configs):
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        t_alloc = min(per_config_time, rem * 0.9)
        print(f"  Deep optimization [{label}] (budget={t_alloc:.0f}s, start_score={sc:.8f})")
        a_deep, sc_deep = lp_alternating_nesterov(a_opt_scoped, t_alloc, solver)
        try_update(a_deep, f"deep_{label}")

    # Phase 3: Adam log-space on current best
    print(f"\n=== Phase 3: Adam Log-Space ===")
    rem = deadline - time.monotonic()
    if rem > 60 and best_a is not None:
        t_adam = min(rem * 0.25, 300.0)
        print(f"  Adam budget: {t_adam:.0f}s, starting from score={best_score:.8f}")
        a_adam, sc_adam = adam_logspace(best_a, t_adam, solver)
        try_update(a_adam, "adam_logspace")

    # Phase 4: Subgradient polish
    print(f"\n=== Phase 4: Subgradient Polish ===")
    rem = deadline - time.monotonic()
    if rem > 30 and best_a is not None:
        t_sub = min(rem * 0.15, 120.0)
        print(f"  SubGrad budget: {t_sub:.0f}s")
        a_sg, sc_sg = subgradient_polish(best_a, t_sub)
        try_update(a_sg, "subgrad_polish")

    # Phase 5: Multi-length exploration
    print(f"\n=== Phase 5: Multi-Length Exploration ===")
    rem = deadline - time.monotonic()
    if rem > 60 and best_a is not None:
        for n_try in [1800, 2500, 3000]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            if n_try == best_n:
                continue
            a_interp = interpolate_sequence(best_a, n_try)
            t_alloc = min(rem * 0.3, 150.0)
            print(f"  Try n={n_try} (budget={t_alloc:.0f}s)")
            a_opt_n, sc_n = lp_alternating_nesterov(a_interp, t_alloc, solver)
            try_update(a_opt_n, f"n{n_try}_lp")

    # Phase 6: Final combined polish
    print(f"\n=== Phase 6: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20 and best_a is not None:
        # Quick LP refinement
        t_lp_final = min(rem * 0.5, 60.0)
        if t_lp_final > 5:
            a_final, sc_final = lp_alternating_nesterov(best_a, t_lp_final, solver)
            try_update(a_final, "final_lp")

        # Subgradient on the result
        rem = deadline - time.monotonic()
        if rem > 10:
            a_sg2, _ = subgradient_polish(best_a, min(rem * 0.7, 30.0))
            try_update(a_sg2, "final_subgrad")

    # Final verification
    if best_a is not None:
        final_score_local = normalized_score(best_a)
        # Convert to the format evaluate_sequence expects
        # Scale so that the local score matches: we need 2*n*max_b/S^2
        # With sum(a)=1, this is just 2*n*max_b = final_score_local
        print(f"\nFINAL: score={final_score_local:.8f}, n={len(best_a)}")

        # Verify with evaluate_sequence
        seq_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
        eval_score = evaluate_sequence(seq_list)
        print(f"  evaluate_sequence score: {eval_score:.8f}")

        if abs(eval_score - final_score_local) > 0.01:
            print(f"  WARNING: local={final_score_local:.8f} vs eval={eval_score:.8f}")

    if best_a is None:
        best_a = init_uniform(1200)

    return [float(max(0.0, min(1000.0, x))) for x in best_a]
```
