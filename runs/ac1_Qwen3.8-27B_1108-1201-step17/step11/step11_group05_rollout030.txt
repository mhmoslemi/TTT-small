    hi = x.max()  # t can be at most this high

        s = np.maximum(x - mid, 0).sum()
        if s > target_sum:

        else:
            hi = mid

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = rfft(a.astype(np.float64), size)
    return irfft(A * A, size)[:2*n-1]

def fast_convolve_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def simplex_project(x, target_sum=1.0):
    """Euclidean projection onto {a >= 0, sum(a) = target_sum}."""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    # Sort in decreasing order
    sorted_x = np.sort(x)[::-1]
    cv = np.cumsum(sorted_x)
    rho = np.arange(1, n+1)
    cond = (sorted_x - (cv - target_sum) / rho) > 0
    rho_star = np.where(cond)[0]
    if len(rho_star) == 0:
        return np.full(n, target_sum / n)
    rho_star = rho_star[-1] + 1
    tau = (cv[rho_star - 1] - target_sum) / rho_star
    return np.maximum(x - tau, 0)

def compute_grad_logspace(a, b, T, top_k=None):
    """
    Compute gradient of F_T in log-space.
    F_T(a) = 2n * T * log(sum exp(b[k]/T))
    ∇_a F_T[j] = 4n * sum_k w_k * a[k-j]
    ∇_z F_T = diag(a) * (∇_a F_T - mean(a * ∇_a F_T) * 1)
    """
    n = len(a)
    size_b = len(b)
    
    if top_k is not None and top_k < size_b:
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[idx]
        max_top = b_top.max()
        exp_top = np.exp((b_top - max_top) / T)
        w_top = exp_top / exp_top.sum()
        w = np.zeros(size_b)
        w[idx] = w_top
    else:
        max_b = b.max()
        exp_all = np.exp((b - max_b) / T)
        w = exp_all / exp_all.sum()
    
    # Compute ∇_a F_T[j] = 4n * sum_k w_k * a[k-j]
    grad_a = np.zeros(n)
    active = np.where(w > 0)[0]
    for k in active:
        wk = w[k]
        if wk < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad_a[j_start:j_end+1] += wk * a[k - np.arange(j_start, j_end + 1)]
    grad_a *= 4.0 * n
    
    # Convert to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z

    """
    Nesterov accelerated gradient in log-space with temperature annealing.
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    z -= z.mean()
    
    best_a = a0.copy()
    best_score = fast_convolve_score(a0)
    t0 = time.monotonic()
    
    total_stages = len(T_stages)
    steps_per_stage = max(2000, int(time_limit * 30000 / total_stages))
    
    # Nesterov state
    z_prev = z.copy()
    mu = momentum_init  # momentum parameter
    
    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        n_steps_this_stage = min(steps_per_stage, int(remaining * 20000))
        top_k = max(10, int(top_k_frac * (2 * n - 1)))
        
        mu = momentum_init + (1.0 - momentum_init) * stage_idx / max(1, total_stages - 1)
        
        for step in range(n_steps_this_stage):
            if time.monotonic() - t0 > time_limit:
                break
            if (step + 1) % 5000 == 0:
                if time.monotonic() - t0 > time_limit:
                    break
            
            # Nesterov lookahead
            z_y = z + mu * (z - z_prev)
            a = softmax_from_z(z_y)
            b = fast_convolve(a)
            score = 2.0 * n * b.max()  # sum(a) = 1 by construction
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Compute gradient
            grad_z = compute_grad_logspace(a, b, T, top_k=top_k)
            
            # Adam-style adaptive step 
            # Actually, let's use a fixed step with momentum
            lr = lr_init * (0.9 ** stage_idx) / (1.0 + step * 0.001)
            
            z_prev = z.copy()
            z = z - lr * grad_z
            z -= z.mean()  # centering (doesn't change softmax)
            
            # Bound z to prevent overflow
            z = np.clip(z, -50, 50)
        
        print(f"    Nesterov stage {stage_idx} (T={T:.3f}): best={best_score:.8f}")
    
    return best_a, best_score

    """Linear auto-convolution via FFT."""

def fast_convolve_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 quickly."""

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)

    """
    Compute grad_a[j] = 4*n * sum_k w[k] * a[k-j] using FFT.
    This is the gradient of F_T(a) = 2n*T*log(sum exp(b[k]/T)) w.r.t. a,
    where b = conv(a,a).

def simplex_project(x, target_sum=1.0):
    """Euclidean projection onto {a >= 0, sum(a) = target_sum}."""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    sorted_x = np.sort(x)[::-1]
    cv = np.cumsum(sorted_x)
    rho = np.arange(1, n + 1)
    cond = (sorted_x - (cv - target_sum) / rho) > 0
    rho_star = np.where(cond)[0]
    if len(rho_star) == 0:
        return np.full(n, target_sum / n)
    rho_star = rho_star[-1] + 1
    tau = (cv[rho_star - 1] - target_sum) / rho_star
    return np.maximum(x - tau, 0)

    """Compute softmax weights on b with temperature T and optional top-k."""
    size_b = len(b)
    if top_k is not None and top_k < size_b and top_k > 0:
        idx = np.argpartition(b, -top_k)[-top_k:]

        w[idx] = w_sel

    w = np.zeros(len(b))
    w[active] = 1.0 / len(active)
    return w

def adam_logspace(a0, T_stages, time_limit, lr_init=0.01, top_k_frac=0.1, verbose=True):
    """
    Adam optimization in log-space with temperature annealing.
    Optimizes F_T(a) = 2n * T * log(sum exp(b[k]/T)) where b = conv(a,a).
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    z -= z.mean()

    best_a = a0.copy()
    best_score = fast_convolve_score(a0)
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step_count = 0

    total_stages = len(T_stages)

    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        top_k = max(5, int(top_k_frac * (2 * n - 1)))
        target_iters = int(remaining * 40000)

        while target_iters > 0 and time.monotonic() - t0 < time_limit:
            a = softmax_from_z(z)
            b = fast_convolve(a)
            score = 2.0 * n * b.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute weights
            if T < 0.01:
                w = get_active_set_weights(b, threshold_frac=1e-5)
            else:
                w = get_softmax_weights(b, T, top_k=top_k)

            # Compute gradient (use direct for sparse, FFT for dense)
            active = np.where(w > 1e-15)[0]
            if len(active) < 50:
                grad_a = compute_grad_a_direct(a, w, active)
            else:
                grad_a = compute_grad_a_fft(a, w)

            # Convert to z-space gradient
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            # Adam update
            step_count += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1.0 - beta1 ** step_count)
            v_hat = v / (1.0 - beta2 ** step_count)

            # Decay learning rate
            lr = lr_init * (0.92 ** stage_idx)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()
            z = np.clip(z, -50.0, 50.0)

            target_iters -= 1

            if verbose and step_count % 50000 == 0:
                elapsed = time.monotonic() - t0
                print(f"      Adam stage {stage_idx} (T={T:.3f}) step {step_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def nesterov_projected(a0, time_limit, lr_base=0.001, top_k=50, verbose=True):
    """
    Nesterov accelerated projected gradient in original space.
    Uses active-set subgradient for the max function.
    """
    n = len(a0)
    a_old = np.maximum(a0, 0).copy()
    s = a_old.sum()
    if s < 1e-15:
        a_old = np.ones(n) / n
        s = 1.0
    a_current = a_old / s

    best_a = a_current.copy()
    best_score = fast_convolve_score(a_current)
    t0 = time.monotonic()

    t_iter = 1

    max_iters = int(time_limit * 30000)
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if (it + 1) % 10000 == 0 and time.monotonic() - t0 > time_limit:
            break

        # Nesterov lookahead
        alpha = (1.0 + math.sqrt(1.0 + 4.0 * t_iter * t_iter)) / 2.0
        theta = (t_iter - 1 + alpha) / (t_iter + alpha)
        y = a_current + theta * (a_current - a_old)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        # Compute gradient at y
        b = fast_convolve(y)
        score_y = 2.0 * n * b.max()

        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Active set
        max_b = b.max()
        if top_k < len(b):
            idx = np.argpartition(b, -top_k)[-top_k:]
            # Use softmax weights on active set for smoothness
            b_active = b[idx]
            T_local = max(0.01, 0.1 * max_b / (len(b) * 0.01))
            exp_vals = np.exp((b_active - b_active.max()) / T_local)
            w_active = exp_vals / exp_vals.sum()
            w = np.zeros(len(b))
            w[idx] = w_active
        else:
            w = get_active_set_weights(b, 1e-4)

        active = np.where(w > 1e-15)[0]
        if len(active) < 100:
            grad = compute_grad_a_direct(y, w, active)
        else:
            grad = compute_grad_a_fft(y, w)

        # Gradient step with projection
        lr = lr_base / (t_iter ** 0.5)
        a_new = y - lr * grad
        a_new = simplex_project(a_new)

        a_old = a_current
        a_current = a_new
        t_iter += 1

        if verbose and (it + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Nesterov iter {it+1} ({elapsed:.0f}s): best={best_score:.8f} cur={fast_convolve_score(a_current):.8f}")

    return best_a, best_score

def lp_best_response(a_fixed, time_limit_single=5.0):
    """
    Solve: min_b max_k (conv(a_fixed, b)) subject to b >= 0, sum(b) = 1.
    Uses simple multiplicative weights / projected gradient (no cvxpy dependency).
    """
    n = len(a_fixed)
    b = np.ones(n) / n
    t0 = time.monotonic()

    for it in range(500):
        if time.monotonic() - t0 > time_limit_single:
            break
        # Compute conv(a_fixed, b)
        size = 1 << (2 * n - 1).bit_length()
        A = rfft(a_fixed, size)
        B = rfft(b, size)
        conv_ab = irfft(A * B, size)[:2 * n - 1]

        max_val = conv_ab.max()
        # Gradient w.r.t. b: sum_k w_k * a_fixed[k-j] (similar to before)
        w = get_active_set_weights(conv_ab, 1e-3)
        active = np.where(w > 1e-15)[0]
        grad_b = np.zeros(n)
        for k in active:
            wk = w[k]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                grad_b[j_start:j_end + 1] += wk * a_fixed[k - np.arange(j_start, j_end + 1)]

        # Multiplicative update
        eta = 0.1 / (it + 10)
        b_new = b * np.exp(-eta * grad_b)
        s = b_new.sum()
        if s > 1e-15:
            b_new = b_new / s
        b = b_new

    return b

def lp_alternating(a0, time_limit, n_iters=1000, verbose=True):
    """
    Alternating minimization: fix a, optimize b (and vice versa).
    Uses multiplicative weights for each subproblem.
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = fast_convolve_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if (it + 1) % 100 == 0 and time.monotonic() - t0 > time_limit:
            break

        # Fix a, optimize b
        b = lp_best_response(a, time_limit_single=2.0)

        # Fix b, optimize a (symmetric)
        a_new = lp_best_response(b, time_limit_single=2.0)

        # Check scores
        score_b = fast_convolve_score(b)
        score_a = fast_convolve_score(a_new)

        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
        if score_b < best_score:
            best_score = score_b
            best_a = b.copy()

        # Update with averaging
        a = 0.5 * a + 0.5 * a_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa

        if verbose and (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-alt iter {it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, seed=42):
    """Create a structured initial sequence."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.6 * np.exp(-2 * (x - 0.7) ** 2) + 0.05
    else:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * x + 0.5) + 0.2 * np.cos(2 * np.pi * 11 * x + 1.0)
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def perturb_sequence(a, seed, strength=0.02):
    """Perturb a sequence with a smooth random modulation."""
    rng = np.random.default_rng(seed)
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 15)
    phase = rng.uniform(0, 2 * np.pi)
    pert = strength * np.cos(freq * x + phase)
    a_pert = a + pert
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FFT-accelerated gradients, Adam in log-space with temperature annealing,
    Nesterov projected gradient, LP alternating, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = fast_convolve_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, seed)
        best_score = fast_convolve_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_convolve_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    n_best = len(best_seq)
    print(f"Starting optimization with n={n_best}, initial score={best_score:.8f}")

    # === Phase 1: Adam in log-space, coarse temperature ===
    print(f"\n=== Phase 1: Adam coarse (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        T_coarse = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7]
        a1, _ = adam_logspace(best_seq, T_coarse, t1, lr_init=0.015, top_k_frac=0.2)
        local_update(a1, "adam_coarse")

    # === Phase 2: Adam in log-space, fine temperature ===
    print(f"\n=== Phase 2: Adam fine (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if t2 > 30:
        print(f"  Time budget: {t2:.0f}s")
        T_fine = [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]
        a2, _ = adam_logspace(best_seq, T_fine, t2, lr_init=0.008, top_k_frac=0.1)
        local_update(a2, "adam_fine")

    # === Phase 3: Active-set subgradient polish ===
    print(f"\n=== Phase 3: Active-set polish (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 250.0)
    if t3 > 20:
        print(f"  Time budget: {t3:.0f}s")
        T_very_fine = [0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        a3, _ = adam_logspace(best_seq, T_very_fine, t3, lr_init=0.004, top_k_frac=0.05)
        local_update(a3, "active_set")

    # === Phase 4: Nesterov projected gradient ===
    print(f"\n=== Phase 4: Nesterov projected (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    if t4 > 20:
        print(f"  Time budget: {t4:.0f}s")
        a4, _ = nesterov_projected(best_seq, t4, lr_base=0.002, top_k=100)
        local_update(a4, "nesterov_proj")

    # === Phase 5: LP alternating ===
    print(f"\n=== Phase 5: LP alternating (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 200.0)
    if t5 > 30:
        print(f"  Time budget: {t5:.0f}s")
        a5, _ = lp_alternating(best_seq, t5, n_iters=2000)
        local_update(a5, "lp_alt")

    # === Phase 6: Second round of fine Adam from improved solution ===
    print(f"\n=== Phase 6: Fine Adam round 2 (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 150.0)
    if t6 > 20:
        print(f"  Time budget: {t6:.0f}s")
        T_r2 = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002]
        a6, _ = adam_logspace(best_seq, T_r2, t6, lr_init=0.005, top_k_frac=0.08)
        local_update(a6, "adam_r2")

    # === Phase 7: Try n=1500 ===
    print(f"\n=== Phase 7: Try n=1500 ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        n_try = 1500
        t7 = min(rem * 0.20, 200.0)
        print(f"  Time budget: {t7:.0f}s")
        init_1500 = interpolate_sequence(best_seq, n_try)
        T_1500 = [5.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
        t7a = t7 * 0.6
        a7, _ = adam_logspace(init_1500, T_1500, t7a, lr_init=0.008, top_k_frac=0.1)
        local_update(a7, "n1500_adam")

        rem = deadline - time.monotonic()
        if rem > 40:
            t7b = min(rem * 0.3, 100.0)
            a7b, _ = nesterov_projected(a7, t7b, lr_base=0.001, top_k=100)
            local_update(a7b, "n1500_nesterov")

    # === Phase 8: Perturbation and restart ===
    print(f"\n=== Phase 8: Perturbation restarts ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            strength = 0.01 + 0.015 * trial
            a_pert = perturb_sequence(best_seq, seed=1000 + trial, strength=strength)
            t_sub = min(rem_sub * 0.25, 60.0)
            T_pert = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            a_p, _ = adam_logspace(a_pert, T_pert, t_sub, lr_init=0.006, top_k_frac=0.1)
            local_update(a_p, f"pert_{trial}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.5, 60.0)
        T_final = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        a9, _ = adam_logspace(best_seq, T_final, t9, lr_init=0.003, top_k_frac=0.05)
        local_update(a9, "final_adam")

    rem = deadline - time.monotonic()
    if rem > 15:
        a9b, _ = nesterov_projected(best_seq, min(rem * 0.5, 30.0), lr_base=0.001, top_k=50)
        local_update(a9b, "final_nesterov")

    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        s = best_seq.sum()
        if s > 1e-15:
            best_seq = best_seq / s
        fs = fast_convolve_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
