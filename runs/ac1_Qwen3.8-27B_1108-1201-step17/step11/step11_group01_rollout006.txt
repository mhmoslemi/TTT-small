def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a_fixed):

    return T

def spectral_shaping(a, alpha=0.5, n_shapes=1):
    """Apply spectral magnitude flattening to reduce peak autocorrelation."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    a = a / s
    
    # Use FFT size that's power of 2 and >= 2n-1
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = fft(a, fft_size)
    P = np.abs(A) ** 2
    phase = np.angle(A)
    
    # Flatten: compress the dynamic range of the power spectrum
    P_max = P.max()
    P_min = P.min()
    if P_max < 1e-30:
        return a.copy()
    
    # Normalize P to [0, 1] range, then apply power compression
    P_norm = P / P_max
    P_new = P_norm ** alpha * P_max
    
    # Reconstruct
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new)[:n])
    
    # Project onto non-negative simplex
    a_new = project_simplex(a_new)
    
    return a_new

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
