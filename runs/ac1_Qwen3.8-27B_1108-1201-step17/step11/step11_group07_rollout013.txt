"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally: 2*n*max_b/(sum_a)^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj_grad_z(z, T):
    """Compute LSE-smoothed objective and gradient in log-parameterization."""
    n = len(z)
    a = softmax(z)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    w = exp_c / exp_c.sum()
    N = 1 << (2 * n).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)[:n]
    grad_a = 2.0 * R
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_optimize(z0, T_stages, iters_per_stage, time_limit, lr=0.005,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with multi-stage temperature annealing."""
    n = len(z0)
    z = z0.copy()
    t0 = time.monotonic()
    best_z = z.copy()
    best_score = evaluate_local(softmax(z))

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    for stage_i, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0

        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                return best_z, best_score
            if it % 50 == 0:
                rem = time_limit - (time.monotonic() - t0)
                if rem < 1:
                    return best_z, best_score

            t += 1
            obj, g = lse_obj_grad_z(z, T)
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = softmax(z_new)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_z = z_new.copy()
            z = z_new

    return best_z, best_score

def multiplicative_weights_refine(a0, time_limit, n_steps=50000, eta0=0.1):

        threshold = c_max * 0.99

        w = np.zeros(2 * n - 1)

        a_scale = np.exp(-eta * grad)
        a_scale -= a_scale.min()
        a_scale = np.maximum(a_scale, 1e-15)
        a_new = a * a_scale
        a_new = a_new / a_new.sum()

        sc = evaluate_local(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            eta = min(eta * 1.05, 3.0)
        else:
            # Try smaller step
            found = False
            for mult in [0.5, 0.25, 0.1]:
                a_try = a * np.exp(-eta * mult * grad)
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-15:

                a_try = a_try / s

def spectral_flatten(a0, time_limit, n_iters=3000, lr=0.003):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 100 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 1:
                break

        t += 1
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        # Gradient of variance of P w.r.t. a
        # dP[k]/da = 2*Re(A[k] * conj(FFT(delta_k))) = 2*Re(A[k] * conj(exp(-2pi i k j / N)))
        # For each j: dP[k]/da[j] = 2*Re(A[k] * exp(2pi i k j / N))
        # We want to minimize sum_k (P[k] - P_mean)^2
        # d/d a[j] = sum_k 2*(P[k]-P_mean) * 2*Re(A[k]*exp(2pi i k j / N))
        # = 4 * Re[sum_k (P[k]-P_mean)*A[k]*exp(2pi i k j / N)]
        # = 4 * Re[IFFT((P-P_mean)*A)[j]]
        H = np.zeros(fft_size // 2 + 1, dtype=np.float64)
        # Compute in frequency domain
        S = (P - P_mean) * A  # complex
        g_freq = irfft(S, fft_size)[:n]
        g = 4.0 * g_freq

        # Adam step
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        grad = np.zeros(n)
        for m in range(js, je + 1):
            grad[m] = 2.0 * a[k_star - m]

        # Multiplicative update
        log_scale = -eta * grad
        log_scale -= log_scale.min()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        sc = evaluate_local(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            found = False
            for mult in [0.5, 0.2]:
                log_scale = -eta * mult * grad
                log_scale -= log_scale.min()
                a_try = a * np.exp(log_scale)
                a_try = np.maximum(a_try, 0)
                s2 = a_try.sum()
                if s2 < 1e-15:
                    continue
                a_try = a_try / s2
                sc2 = evaluate_local(a_try)
                if sc2 < best_score - 1e-12:
                    best_score = sc2
                    best_a = a_try.copy()
                    a = a_try.copy()
                    found = True
                    break
            if not found:
                eta *= 0.8
                if eta < 1e-10:
                    break

    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)
    candidates.append(normalize(np.exp(-8 * (x - 0.5) ** 2)))
    candidates.append(normalize(0.5 * (1 + np.cos(2 * np.pi * x))))
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    candidates.append(normalize((x * (1 - x)) ** 0.5))
    candidates.append(normalize(np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)))
    a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
    candidates.append(normalize(np.maximum(a, 0)))
    coeffs = rng.normal(0, 1, 10)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    candidates.append(normalize(np.maximum(a + 0.5, 0)))
    candidates.append(np.ones(n) / n)
    candidates.append(normalize(np.exp(-5 * x) + np.exp(-5 * (1 - x))))
    # Random with low-freq structure
    freqs = rng.choice(range(1, 20), size=5, replace=False)
    phases = rng.uniform(0, 2 * np.pi, 5)
    a = np.ones(n)
    for f, p in zip(freqs, phases):
        a += 0.3 * np.cos(2 * np.pi * f * x + p)
    candidates.append(normalize(a))
    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(np.maximum(f_interp, 0))


def perturb_sequence(a, rng, method='gaussian', strength=0.02):
    """Perturb a sequence."""
    n = len(a)
    if method == 'gaussian':
        a_new = a + rng.normal(0, strength, n)
    elif method == 'low_freq':
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        a_new = a + strength * np.cos(freq * x + phase)
    elif method == 'block_swap':
        a_new = a.copy()
        bs = rng.integers(10, min(50, n // 4))
        i = rng.integers(0, n - bs)
        j = rng.integers(0, n - bs)
        if abs(i - j) > bs:
            a_new[i:i + bs], a_new[j:j + bs] = a_new[j:j + bs].copy(), a_new[i:i + bs].copy()
    elif method == 'edge_push':
        a_new = a.copy()
        edge = rng.integers(0, 2)
        w = rng.uniform(0.01, 0.05)
        k = min(20, n // 5)
        if edge == 0:
            a_new[:k] += w
            a_new[k:] -= w * k / (n - k)
        else:
            a_new[-k:] += w
            a_new[:-k] -= w * k / (n - k)
    elif method == 'random_sparse':
        a_new = a.copy()
        idx = rng.choice(n, size=n // 10, replace=False)
        a_new[idx] += rng.normal(0, strength * 3, len(idx))
    else:
        a_new = a + rng.normal(0, strength, n)
    return normalize(np.maximum(a_new, 0))


def run_full_pipeline(a_init, time_limit, rng):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    def check_time():
        return time.monotonic() - t0 > time_limit

    # Stage 1: Spectral pre-conditioning
    rem = time_limit - (time.monotonic() - t0)
    if rem > 30:
        t_alloc = min(rem * 0.08, 60.0)
        a_spec, sc = spectral_flatten(a, t_alloc, n_iters=2000, lr=0.002)
        if sc < best_score:
            best_score = sc
            best_a = a_spec.copy()
            a = a_spec.copy()

    # Stage 2: Adam with LSE annealing (main workhorse)
    if not check_time():
        T_stages = [20, 10, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3, 0.2, 0.15,
                    0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002]
        iters_per_stage = max(300, int((time_limit - (time.monotonic() - t0)) / len(T_stages) / 0.005))
        iters_per_stage = min(iters_per_stage, 5000)
        z = np.log(np.maximum(a, 1e-300))
        z_best, sc = adam_optimize(z, T_stages, iters_per_stage,
                                   time_limit - (time.monotonic() - t0), lr=0.003)
        a_candidate = softmax(z_best)
        sc = evaluate_local(a_candidate)
        if sc < best_score:
            best_score = sc
            best_a = a_candidate.copy()
            a = a_candidate.copy()

    # Stage 3: Multiplicative weights peak-shaving
    if not check_time():
        rem = time_limit - (time.monotonic() - t0)
        t_alloc = min(rem * 0.15, 120.0)
        a_mw, sc = multiplicative_weights_refine(a, t_alloc, n_steps=30000, eta0=0.1)
        if sc < best_score:
            best_score = sc
            best_a = a_mw.copy()
            a = a_mw.copy()

    # Stage 4: Second Adam pass (finer)
    if not check_time():
        rem = time_limit - (time.monotonic() - t0)
        if rem > 30:
            T_stages2 = [1, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
            iters2 = max(200, int(rem / len(T_stages2) / 0.005))
            iters2 = min(iters2, 3000)
            z = np.log(np.maximum(a, 1e-300))
            z_best, sc = adam_optimize(z, T_stages2, iters2,
                                       rem - (time.monotonic() - t0) - 10, lr=0.002)
            a_candidate = softmax(z_best)
            sc = evaluate_local(a_candidate)
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()
                a = a_candidate.copy()

    # Stage 5: Exact subgradient polish
    if not check_time():
        rem = time_limit - (time.monotonic() - t0)
        if rem > 15:
            a_ex, sc = exact_subgradient_step(a, rem, n_steps=20000, eta0=0.03)
            if sc < best_score:
                best_score = sc
                best_a = a_ex.copy()
                a = a_ex.copy()

    # Stage 6: One more multiplicative pass
    if not check_time():
        rem = time_limit - (time.monotonic() - t0)
        if rem > 10:
            a_mw2, sc = multiplicative_weights_refine(a, rem, n_steps=20000, eta0=0.05)
            if sc < best_score:
                best_score = sc
                best_a = a_mw2.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Adam with LSE annealing, spectral flattening, multiplicative
    weights peak-shaving, and diverse multi-restart.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = 1200
    if parent is None:
        candidates = diverse_init(n_base, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    print(f"Starting: n={len(best_seq)}, score={best_score:.8f}")

    # === Phase 1: Main pipeline on parent ===
    print(f"\n=== Phase 1: Main Pipeline ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        a1, s1 = run_full_pipeline(best_seq, min(rem * 0.35, 500.0), rng)
        update_best(a1, "main_pipeline")

    # === Phase 2: Multi-restart with perturbations ===
    print(f"\n=== Phase 2: Multi-Restart (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_restarts = 5
    if rem > n_restarts * 30:
        t_per_restart = min(rem / n_restarts - 2, 250.0)
        methods = ['gaussian', 'low_freq', 'random_sparse', 'edge_push', 'block_swap']
        for i in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            method = methods[i % len(methods)]
            strength = 0.015 + 0.01 * (i % 3)
            a_pert = perturb_sequence(best_seq, rng, method, strength)
            t_alloc = min(t_per_restart, rem - 5)
            if t_alloc < 15:
                t_alloc = rem * 0.3
            a_r, s_r = run_full_pipeline(a_pert, t_alloc, rng)
            update_best(a_r, f"restart_{i}_{method}")

    # === Phase 3: Try different lengths ===
    print(f"\n=== Phase 3: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    length_tries = [n_best + 100, n_best - 100, n_best + 200, n_best + 50]
    for n_try in length_tries:
        if n_try < 200 or n_try > 3000:
            continue
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        init = interpolate_sequence(best_seq, n_try)
        t_alloc = min(rem * 0.25, 200.0)
        a_l, s_l = run_full_pipeline(init, t_alloc, rng)
        update_best(a_l, f"length_{n_try}")

    # === Phase 4: Intensive final polish on best ===
    print(f"\n=== Phase 4: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Do a focused MW + exact subgradient pass
        a_mw, sc = multiplicative_weights_refine(best_seq, min(rem * 0.3, 60.0),
                                                  n_steps=50000, eta0=0.03)
        update_best(a_mw, "final_mw")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_ex, sc = exact_subgradient_step(best_seq, min(rem * 0.5, 40.0),
                                               n_steps=30000, eta0=0.02)
            update_best(a_ex, "final_exact")
        rem = deadline - time.monotonic()
        if rem > 5:
            # Very fine Adam
            T_final = [0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            z = np.log(np.maximum(best_seq, 1e-300))
            iters = max(100, int(rem / 0.003))
            iters = min(iters, 2000)
            z_best, sc = adam_optimize(z, T_final, iters,
                                       max(1.0, rem - 5), lr=0.001)
            update_best(softmax(z_best), "final_adam")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
