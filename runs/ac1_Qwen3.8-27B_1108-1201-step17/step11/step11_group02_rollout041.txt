    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    
    # Mirror descent update
    log_a = np.log(np.maximum(a, 1e-300))
    log_a -= eta * g
    log_a -= log_a.max()  # numerical stability
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new, max_b

    A = rfft(a_pad)
    corr_full = irfft(W * A.conj(), fft_size)
    # corr_full[j] = sum_i w_pad[i] * a_pad[i-j] (circular)
    # We want: result[j] = sum_i w[j+i] * a[i]
    # This is cross-correlation, which is: ifft(conj(fft(w)) * fft(a))
    # Let me just compute it directly for small n, FFT for large n
    if n <= 500:
        result = np.zeros(n)
        for j in range(n):
            max_i = min(n, len(w) - j)
            if max_i > 0:
                result[j] = np.dot(w[j:j + max_i], a[:max_i])
        return result
    # For larger n, use the FFT approach
    # corr[j] = sum_i w[j+i] * a[i] for i=0..min(n-1, len(w)-1-j)
    # = convolution of w (reversed) with a, evaluated at specific positions
    # Actually, let's use: corr = ifft(fft(w, 2n) * conj(fft(a, 2n)))[:n]
    # NO that's not right either. Let me just use the direct FFT:
    # Define: c[j] = sum_i w[j+i] * a[i]
    # This is the "cross-correlation" of w and a with a specific convention
    # c = ifft(fft(w, N) * conj(fft(a, N))) where N >= n + len(w) - 1
    N = 1 << (2 * n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    c = irfft(np.conj(W) * A, N)
    # c[k] = sum_i w[i] * a[(k-i) mod N]
    # We want: c[j] = sum_i w[j+i] * a[i]
    # Let's substitute: let m = j+i, so i = m-j
    # c[j] = sum_m w[m] * a[m-j] for m=j..j+n-1 (and m < len(w), m-j >= 0)
    # This is: sum_m w[m] * a[m-j] which is the cross-correlation evaluated at lag j
    # ifft(conj(fft(w)) * fft(a))[k] = sum_m w[m] * a[m-k] (circular)
    # So c[k] = sum_m w[m] * a[m-k] = our desired value at j=k
    return c[:n]


def compute_lse_gradient(a, T):
    """Compute gradient of (2n/T) * LSE_T(conv(a,a)) with respect to a."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    w = exp_b / exp_b.sum()
    # Gradient[j] = (2n/T) * 2 * sum_i w[j+i] * a[i]
    # = (4n/T) * cross_correlate(w, a)[j]
    corr = fast_cross_correlate(w, a)
    grad = (4.0 * n / T) * corr
    return grad, max_b


def evaluate_local(a):
    """Evaluate sequence using local computation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def fista_logsumexp(a0, time_limit, T_stages=None, maxiter_total=200000):
    """
    FISTA with LogSumExp smoothing and adaptive temperature.
    Works on the simplex {a >= 0, sum(a) = 1}.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [30.0, 20.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.3, 0.2, 0.12, 0.08]

    iters_per_stage = max(50, maxiter_total // len(T_stages))
    global_iter = 0

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break

        # FISTA state
        y = a.copy()
        t_fista = 1.0
        # Lipshitz estimate: use backtracking
        L = 10.0
        stage_iter = 0

        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit - 2:
                break
            stage_iter += 1
            global_iter += 1

            # Compute gradient at y
            grad_y, _ = compute_lse_gradient(y, T)

            # Mirror descent step (in log-space)
            # a_new[i] ∝ a[i] * exp(-grad[i] / L)
            log_ratio = -grad_y / L
            log_ratio -= log_ratio.max()  # stability
            a_new = a * np.exp(log_ratio)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
                s_new = 1.0
            a_new = a_new / s_new

            # Check improvement, else backtrack
            score_new = evaluate_local(a_new)
            while score_new > evaluate_local(a) + 1e-12 and L < 1e6:
                L *= 2.0
                log_ratio = -grad_y / L
                log_ratio -= log_ratio.max()
                a_new = a * np.exp(log_ratio)
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    a_new = a.copy()
                    s_new = 1.0
                a_new = a_new / s_new
                score_new = evaluate_local(a_new)

            if score_new <= evaluate_local(a) + 1e-12:
                a = a_new
                L = max(1.0, L * 0.8)  # slightly decrease L for next step

            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
            beta = (t_fista - 1.0) / t_new
            y = a + beta * (a - y)
            y = np.maximum(y, 0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = a.copy()
            else:
                y = y / s_y
            t_fista = t_new

            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        elapsed = time.monotonic() - t0
        sc = evaluate_local(best_a)
        print(f"    FISTA T={T:.2f} ({elapsed:.0f}s): best_score={sc:.8f}")

    return best_a, best_score


def solve_cutting_plane_lp(a_fixed, active_indices, M_current, solver='highs'):
    """
    Solve LP: minimize t subject to conv(a_fixed, b)[k] <= t for k in active_indices,
    sum(b) = 1, b >= 0.
    
    Uses scipy.optimize.linprog with HiGHS.
    Returns (b_optimal, t_optimal) or (None, None) on failure.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build constraint matrix for active constraints only
    # For each active index k: sum_j a_fixed[j] * b[k-j] <= t
    # i.e., sum_j a_fixed[j] * b[k-j] - t <= 0
    # Variables: [b_0, ..., b_{n-1}, t]
    # A_ub has |active_indices| rows, n+1 columns
    
    m = len(active_indices)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for row, k in enumerate(active_indices):
        for j in range(n):
            idx = k - j
            if 0 <= idx < size:
                A_ub[row, j] = a_fixed[j]
        A_ub[row, n] = -1.0  # coefficient of t
        b_ub[row] = 0.0
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: b >= 0, t free (but effectively >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    b_val = np.maximum(result.x[:n], 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    t_val = float(result.x[n])
    
    # Verify
    conv_ab = np.convolve(a_fixed, b_val)
    t_actual = float(max(conv_ab[k] for k in active_indices))
    
    return b_val, t_actual


def cutting_plane_lp_refine(a0, time_limit, max_iters=5000):
    """
    Cutting-plane LP refinement.
    Maintain a set of active constraints (peak positions).
    Solve LP with only active constraints, then add violated ones.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    y = a.copy()
    stall = 0
    rng = np.random.default_rng(99)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit - 5:
            break

        # Compute current autocorrelation
        b_y = fast_convolve(y)
        max_b = b_y.max()

        # Identify active peaks (top 10% or those within 1% of max)
        threshold = max_b * 0.995
        active = np.where(b_y >= threshold)[0]
        # Limit to top 50 for LP efficiency
        if len(active) > 50:
            top_idx = np.argsort(b_y[active])[::-1][:50]
            active = active[top_idx]
        
        # Solve LP with active constraints
        b_new, M_new = solve_cutting_plane_lp(y, active, max_b, 'highs')
        if b_new is None:
            # Fallback: simple projection
            b_new = y.copy()
            M_new = max_b
            stall += 1
        else:
            # Check if solution violates other constraints
            conv_check = np.convolve(y, b_new)
            max_check = conv_check.max()
            if max_check > max_b * 1.01:
                # Add the most violated constraint
                new_active = int(np.argmax(conv_check))
                if new_active not in active:
                    active = np.append(active, [new_active])
                    if len(active) > 50:
                        active = active[np.argsort(conv_check[active])[::-1][:50]]
                    # Re-solve
                    b_new, M_new = solve_cutting_plane_lp(y, active, max_b, 'highs')
                    if b_new is None:
                        b_new = y.copy()
                        M_new = max_b

        # Nesterov-style update
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Weighted average for a
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        # Exploration on stall
        if stall >= 15:
            noise = rng.standard_normal(n) * 0.01
            y = y + noise
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            a = y.copy()
            stall = 0

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CP-LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}, active={len(active)}")

    return best_a, best_score


def peak_balancing(a0, time_limit, n_iters=5000):
    """
    Explicit peak-balancing: find directions that reduce all top peaks simultaneously.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit - 3:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Find top peaks
        n_peaks = min(5, max(2, len(b) // 100))
        peak_indices = np.argsort(b)[::-1][:n_peaks]
        
        # Target: reduce all peaks to below the 2nd highest
        target = b[peak_indices[1]] if len(peak_indices) > 1 else max_b * 0.999
        
        # Compute balancing direction
        # For each peak k, the contribution vector is c_k[i] = a[i] * a[k-i]
        # We want to decrease r_k = sum c_k[i] for all peak k
        # Direction: delta[i] -= sum_k w_k * c_k[i] where w_k = max(0, r_k - target)
        
        weights = np.zeros(n_peaks)
        for pi, k in enumerate(peak_indices):
            weights[pi] = max(0.0, b[k] - target)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue
        weights = weights / w_sum

        # Build direction
        delta = np.zeros(n)
        for pi, k in enumerate(peak_indices):
            w_k = weights[pi]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            delta[idx] -= w_k * contribs
        
        # Make sum(delta) = 0 (redistribute mass)
        delta -= delta.mean()
        
        # Line search for step size
        best_alpha = 0.0
        base_score = best_score
        for alpha_trial in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_trial = a + alpha_trial * delta
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            sc = evaluate_local(a_trial)
            if sc < best_score - 1e-14:
                best_alpha = alpha_trial
                a = a_trial
                best_score = sc
                best_a = a.copy()
                stall = 0
                break

        if best_alpha == 0:
            stall += 1
            if stall > 200:
                # Small random perturbation
                rng = np.random.default_rng(it * 7 + 3)
                noise = rng.standard_normal(n) * 0.005
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakBal iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_final(a0, time_limit, n_steps=200000):
    """
    Adaptive coordinate descent focusing on peak-contributing elements.
    Uses line search for optimal step size per coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit - 3:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        contribs_norm = contribs / max(contribs.sum(), 1e-300)
        
        # Select top 10 contributing positions
        n_select = min(10, len(idx))
        top_pos = idx[np.argsort(contribs)[::-1][:n_select]]

        improved = False
        for pos in top_pos:
            if a[pos] < 1e-12:
                continue
            
            # Try reducing this coordinate
            for factor in [0.999, 0.998, 0.995, 0.99, 0.98, 0.97, 0.95]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-15:
                    continue
                # Redistribute to other coordinates proportionally
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try increasing a non-contributing coordinate
            non_contrib = np.setdiff1d(np.arange(n), top_pos)
            if len(non_contrib) > 0:
                rng = np.random.default_rng(step * 13 + 1)
                candidates = non_contrib[rng.integers(0, len(non_contrib), size=min(5, len(non_contrib)))]
                for pos in candidates:
                    a_trial = a.copy()
                    a_trial[pos] *= 1.005
                    a_trial[pos + 1 if pos + 1 < n else 0] *= 0.999
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            stall += 1
            if stall > 500:
                # Simultaneous reduction of top contributors
                a_trial = a.copy()
                for pos in top_pos[:4]:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.999
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall > 2000:
                    break
        else:
            stall = 0

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.15)
    elif choice == 5:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        # Fejer-like kernel
        a = np.sinc(4 * (x - 0.5)) ** 2
        a = np.maximum(a, 0)
    else:
        # Multi-scale Gaussian
        a = np.exp(-20 * (x - 0.2) ** 2) + 0.7 * np.exp(-20 * (x - 0.5) ** 2) + 0.5 * np.exp(-20 * (x - 0.8) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def fista_with_restarts(a0, time_limit, n_restarts=5):
    """FISTA with warm restarts on stall detection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    T_schedule = [20, 12, 8, 5, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.3, 0.2, 0.12, 0.08, 0.05]
    T_idx = 0
    stall_count = 0
    prev_best = best_score

    total_iters = 0
    max_total = 500000

    while total_iters < max_total:
        if time.monotonic() - t0 > time_limit - 5:
            break
        if T_idx >= len(T_schedule):
            break

        T = T_schedule[T_idx]
        iters_this_stage = max(100, (max_total - total_iters) // (len(T_schedule) - T_idx) // 3)

        # FISTA within this stage
        y = a.copy()
        t_f = 1.0
        L = 5.0

        for it in range(iters_this_stage):
            if time.monotonic() - t0 > time_limit - 3:
                break
            total_iters += 1

            grad_y, _ = compute_lse_gradient(y, T)

            # Mirror descent step
            log_ratio = -grad_y / L
            log_ratio -= log_ratio.max()
            a_new = a * np.exp(log_ratio)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            # Backtracking line search (simplified: check score)
            sc_new = evaluate_local(a_new)
            retries = 0
            while sc_new > best_score + 0.01 and retries < 5:
                L *= 2.0
                log_ratio = -grad_y / L
                log_ratio -= log_ratio.max()
                a_new = a * np.exp(log_ratio)
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                sc_new = evaluate_local(a_new)
                retries += 1

            if sc_new < best_score:
                a = a_new
                best_score = sc_new
                best_a = a.copy()
                L = max(1.0, L * 0.85)
            else:
                a = a_new  # still update for momentum
                L = L * 1.2

            # FISTA momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_f * t_f)) / 2.0
            beta = (t_f - 1.0) / t_new
            y = a + beta * (a - y)
            y = np.maximum(y, 0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            t_f = t_new

        # Check progress
        elapsed = time.monotonic() - t0
        if best_score < prev_best - 1e-6:
            stall_count = 0
            prev_best = best_score
        else:
            stall_count += 1

        T_idx += 1
        if T_idx < len(T_schedule):
            print(f"    FISTA-R T={T_schedule[T_idx]:.1f} ({elapsed:.0f}s): score={best_score:.8f} (stall={stall_count})")

        # Warm restart on stall
        if stall_count >= 2:
            noise = rng.standard_normal(n) * 0.02
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0
            T_idx = max(0, T_idx - 3)  # go back a few stages

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. FISTA with adaptive LogSumExp (fast global optimization)
    2. Cutting-plane LP precision refinement
    3. Peak-balancing directional optimization
    4. Adaptive coordinate descent polish
    5. Multi-resolution diversification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_best = len(best_seq)

    # === Phase 1: FISTA with restarts (fast, ~400s) ===
    print(f"\n=== Phase 1: FISTA w/ restarts (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_f, _ = fista_with_restarts(best_seq, t_alloc)
        try_update(a_f, "FISTA_R")

    # === Phase 2: Cutting-plane LP (~500s) ===
    print(f"\n=== Phase 2: Cutting-Plane LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cp, _ = cutting_plane_lp_refine(best_seq, t_alloc)
        try_update(a_cp, "CP_LP")

    # === Phase 3: Peak balancing (~300s) ===
    print(f"\n=== Phase 3: Peak Balancing (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.17, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pb, _ = peak_balancing(best_seq, t_alloc, n_iters=10000)
        try_update(a_pb, "PeakBal")

    # === Phase 4: Coordinate descent (~350s) ===
    print(f"\n=== Phase 4: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.19, 350.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_final(best_seq, t_alloc, n_steps=300000)
        try_update(a_cd, "CoordDesc")

    # === Phase 5: Multi-resolution exploration (~200s) ===
    print(f"\n=== Phase 5: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 30:
        n_cur = len(best_seq)
        candidate_ns = []
        for delta in [-200, -100, 100, 200, 300]:
            nt = n_cur + delta
            if nt >= 200:
                candidate_ns.append(nt)
        # Also try half and double
        if n_cur // 2 >= 200:
            candidate_ns.append(n_cur // 2)
        candidate_ns = sorted(set(candidate_ns))
        
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}, {per_n:.0f}s each")
        
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 10:
                break
            if n_try == n_cur:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 5)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            a_sub, _ = fista_with_restarts(a_interp, t_sub)
            try_update(a_sub, f"multi_n_{n_try}")

    # === Phase 6: Final refinement (~150s) ===
    print(f"\n=== Phase 6: Final Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 150.0)
    if t_alloc > 15:
        # Quick FISTA + coord descent combo
        a_q1, _ = fista_with_restarts(best_seq, t_alloc * 0.5)
        try_update(a_q1, "final_fista")
        
        rem2 = deadline - time.monotonic()
        if rem2 > 15:
            a_q2, _ = coordinate_descent_final(best_seq, rem2 - 5, n_steps=100000)
            try_update(a_q2, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
