2. Compute `b = conv(a,a)` via FFT.

        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        
        # Subgradient
        g = conv_gradient(a, w)
        
        # Adaptive eta
        g_max = np.abs(g).max()
        if g_max > 1e-10:
            eta_eff = eta / g_max  # normalize gradient
        else:
            eta_eff = eta
        
        # Multiplicative update
        log_ratio = -eta_eff * g
        log_ratio -= log_ratio.max()
        a = a * np.exp(log_ratio)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s
        
        # Adapt parameters
        if no_improve > 100:
            eta *= 0.95
            no_improve = 0
            if eta < 1e-5:
                break
        
        # Adaptively narrow active set
        if step % 500 == 0:
            eps = max(0.001, eps * 0.95)
        
        step += 1
        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW step {step} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}, eps={eps:.4f}, |S|={len(active)}")
    
    return best_a, best_score

def get_lp_solver():
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build constraint: conv(a_fixed, b)[k] <= M for all k
    # conv(a_fixed, b)[k] = sum_j a_fixed[j] * b[k-j]
    # This is T @ b where T[k,j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    # For efficiency, build T as a banded matrix
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
                return b_val, float(max(0, M_var.value))
    except Exception:
        pass
    return None, None

def nesterov_lp(a_init, time_limit, solver, max_iters=100000):
    """Nesterov dual averaging with LP best response."""
    n = len(a_init)
    y = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = y.sum()
    if s < 1e-15:
        y = np.ones(n) / n
    else:
        y = y / s
    a = y.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_val = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        a = ((it + 1) * a + (it + 2) * b_new) / (2 * (it + 1.0))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter {it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-20 * (x - 0.5) ** 2) + 0.3
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
