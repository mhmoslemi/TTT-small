def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def logsumexp_grad_fft(a, T):
    """
    Compute LogSumExp smoothed objective and gradient using FFT.
    Objective: LogSumExp(conv(a,a)/T)
    Gradient: (2/T) * cross_correlation(w, a) approximately
    
    where w[k] = softmax(b/T), b = conv(a,a)
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()
    
    # Cross-correlation: c[j] = sum_k w[k] * a[k-j]
    # This is ifft(fft(w) * conj(fft(a))) truncated to n elements
    fft_size = next_fast_len(2 * n - 1)
    W = fft(w, fft_size)
    A = fft(a, fft_size)
    c = np.real(ifft(W * np.conj(A), fft_size))[:n]
    
    # The gradient is approximately (2/T) * c[j]
    # More precisely, for k=2j terms, the factor should be 1 not 2
    # But this correction is O(1/n) relative, so we use the approximation
    grad = (2.0 / T) * c
    
    # Objective value
    obj = max_b / T + np.log(exp_b.sum())
    
    return obj, grad

def spectral_shaping(a0, time_limit, n_trials=20, beta_range=(0.1, 10.0)):
    """
    Novel exploration via spectral power compression.
    Soft-caps the power spectrum of a to flatten the convolution,
    then projects back to non-negative.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    fft_size = next_fast_len(2 * n - 1)
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        # Try different beta values (soft cap strength)
        beta = rng.uniform(beta_range[0], beta_range[1])
        
        # Power spectrum
        A = fft(a, fft_size)
        P = np.abs(A)**2
        phase = np.angle(A)
        
        # Soft cap: P_capped = P / (1 + beta * P / P_mean)
        P_mean = P.mean()
        P_capped = P / (1.0 + beta * P / P_mean)
        
        # Reconstruct with modified amplitude
        A_new = np.sqrt(P_capped) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fft_size))[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"  Spectral trial {trial} beta={beta:.3f}: {best_score:.8f}")
        
        # Also try blending with original
        for alpha in [0.3, 0.5, 0.7]:
            a_blend = (1 - alpha) * a + alpha * a_new
            a_blend = np.maximum(a_blend, 0)
            sb = a_blend.sum()
            if sb > 1e-10:
                a_blend = a_blend / sb
                sc_b = evaluate_local(a_blend)
                if sc_b < best_score:
                    best_score = sc_b
                    best_a = a_blend.copy()
        
        # Update working point
        a = best_a.copy()
    
    return best_a, best_score

def nesterov_subgradient(a0, time_limit, n_max_iters=1000000):
    """
    Nesterov-accelerated mirror descent (subgradient) method.
    No LP solving - uses subgradient of max(conv(a,a)) directly.
    Very fast per iteration (O(n log n) via FFT).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    # Nesterov parameters
    k = 1.0
    prev_y = y.copy()
    
    # Working point for Nesterov
    x = a.copy()
    
    no_improve = 0
    patience = 5000
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution and find peak
        b_y = fast_convolve(y)
        max_b = b_y.max()
        peak_idx = int(np.argmax(b_y))
        
        # Subgradient at y: g[j] = sum_k w[k] * (2*a[k-j] approx)
        # For the max, the subgradient is: g[j] = y[peak_idx - j] (for valid j)
        # More precisely, use the "conic combination" of all near-max peaks
        threshold = max_b * 0.9995
        active_peaks = np.where(b_y >= threshold)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for pk in active_peaks[:5]:  # Limit to top 5 peaks for speed
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            g[idx] += y[pk - idx]
        g /= max(1, len(active_peaks))
        
        # Nesterov update (mirror descent / exponentiated gradient)
        # x_new = x * exp(-eta * g / max(g))
        g_max = g.max()
        if g_max < 1e-30:
            break
        g_norm = g / g_max
        
        # Adaptive step size
        eta = 2.0 / (np.sqrt(it + 1) + 1.0)
        eta = np.clip(eta, 0.01, 1.0)
        
        # Exponentiated gradient update
        log_x = np.log(np.maximum(x, 1e-300))
        log_x_new = log_x - eta * g_norm
        log_x_new -= log_x_new.max()
        x_new = np.exp(log_x_new)
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-300:
            x_new = x.copy()
            sx = 1.0
        x_new = x_new / sx
        
        # Nesterov momentum
        k_new = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        y_new = x_new + (k / k_new - 1.0) * (x_new - prev_y)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-300:
            y_new = x_new.copy()
            sy = 1.0
        y_new = y_new / sy
        
        prev_y = y
        y = y_new
        x = x_new
        k = k_new
        
        # Update best
        sc_y = evaluate_local(y)
        sc_x = evaluate_local(x)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        elif sc_x < best_score:
            best_score = sc_x
            best_a = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart on long stall
        if no_improve > patience:
            # Weighted geometric mean restart
            w = 0.5
            x = np.maximum(x**w, 1e-100)
            sx = x.sum()
            x = x / sx
            y = x.copy()
            k = 1.0
            prev_y = y.copy()
            no_improve = 0
            patience = max(2000, patience // 2)
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSubGrad iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit):
    """
    Coordinate descent focusing on peak-contributing elements.
    Uses line search for optimal step.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 1:
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]
        
        improved = False
        for peak_k in peaks[:5]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(12, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                # Try multiple reduction factors
                for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95]:
                    a_trial = a.copy()
                    reduction = a_trial[pos] * (1 - factor)
                    a_trial[pos] *= factor
                    # Redistribute to other elements proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += reduction * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    
                    sc = evaluate_local(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 100:
                # Simultaneous reduction of top contributors
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:6]]
                a_trial = a.copy()
                total_red = 0
                for pos in top_positions:
                    if a_trial[pos] > 1e-12:
                        red = a_trial[pos] * 0.002
                        a_trial[pos] -= red
                        total_red += red
                if total_red > 0:
                    mask = np.ones(n, dtype=bool)
                    for pos in top_positions:
                        mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum > 1e-15:
                        a_trial[mask] += total_red * a_trial[mask] / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = evaluate_local(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
                if stall > 500:
                    break
        else:
            stall = 0
        
        if (stall + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): score={best_score:.8f}, stall={stall}")
    
    return best_a, best_score


def build_sparse_conv_matrix(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    row_ind = []
    col_ind = []
    data = []
    for i in range(n):
        row_ind.extend(range(i, i + n))
        col_ind.extend([i] * n)
        data.extend(a)
    return csr_matrix((data, (np.array(row_ind), np.array(col_ind))), shape=(size, n))

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b)."""
    n = len(a_fixed)
    T_sparse = build_sparse_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    size = 2 * n - 1
    constraints = [T_sparse @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                return None, None
    except Exception:
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    T_sparse2 = build_sparse_conv_matrix(a_fixed)
    conv_vals = T_sparse2 @ b_val
    M_val = float(np.max(conv_vals))

def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=50000):
    """Nesterov LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(777)
    
    # Track time per LP solve to adapt frequency
    lp_times = []
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP best response
        t_lp_start = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - t_lp_start
        lp_times.append(lp_time)
        if len(lp_times) > 20:
            lp_times.pop(0)
        
        if b_new is None:
            stall += 1
            if stall > 20:
                break
            continue
        
        # Nesterov update
        alpha = 2.0 / (np.sqrt(float(it + 1)) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Weighted average for a
        weight_a = float(it + 1)
        weight_b = alpha * (it + 2)
        a = (weight_a * a + weight_b * b_new) / (weight_a + weight_b)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa
        
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        if (it + 1) % 10 == 0:
            if best_score < prev_best - 1e-12:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            
            # Exploration on stall
            if stall >= 6:
                b_cur = fast_convolve(y)
                peak_idx = np.argmax(b_cur)
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                idx = np.arange(js, je + 1)
                contribs = y[idx] * y[peak_idx - idx]
                n_reduce = max(5, len(idx) // 8)
                top_pos = idx[np.argsort(contribs)[::-1][:n_reduce]]
                for pos in top_pos:
                    if y[pos] > 1e-10:
                        y[pos] *= (1.0 - 0.03 * min(3.0, np.sqrt(float(stall) / 6.0)))
                noise = rng.standard_normal(n) * 0.005
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = y.copy()
                stall = 0
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times) if lp_times else 0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}, LP_time={avg_lp:.3f}s")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.2)
    elif choice == 5:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        # Power-law decay from center
        xc = x - 0.5
        a = np.exp(-5.0 * np.abs(xc) ** 1.5)
    else:
        # Flat top with smooth edges
        a = np.where(np.abs(x - 0.5) < 0.3, 1.0,
                    np.exp(-20 * (np.abs(x - 0.5) - 0.3)**2 / 0.01))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining:
    - FFT-accelerated subgradient (fast, no LP)
    - Nesterov LP (precise but slower)
    - Spectral shaping (novel exploration)
    - LogSumExp L-BFGS (polishing)
    - Coordinate descent (micro-optimization)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Fast Nesterov Subgradient (no LP) ===
    print(f"\n=== Phase 1: Nesterov Subgradient (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = nesterov_subgradient(best_seq, t_alloc)
        try_update(a_sg, "NSubGrad")

    # === Phase 2: Spectral Shaping (novel exploration) ===
    print(f"\n=== Phase 2: Spectral Shaping (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_spec, _ = spectral_shaping_iterative(best_seq, t_alloc, n_iters=2000)
        try_update(a_spec, "Spectral")
        a_spec2, _ = spectral_shaping(best_seq, t_alloc * 0.5, n_trials=50, beta_range=(0.05, 20.0))
        try_update(a_spec2, "Spectral2")

    # === Phase 3: Nesterov LP (precise) ===
    print(f"\n=== Phase 3: Nesterov LP (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_lp, "LP_p3")

    # === Phase 4: FFT-accelerated LogSumExp L-BFGS ===
    print(f"\n=== Phase 4: LSE L-BFGS FFT (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        T_stages = [50.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb, _ = logsumexp_lbgfs_fft(best_seq, t_alloc, T_stages=T_stages)
        try_update(a_lb, "LSE_LB")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        candidate_ns = [max(500, n_best - 300), max(700, n_best + 200), n_best * 2 if n_best < 2000 else max(1000, n_best // 2)]
        candidate_ns = sorted(set(candidate_ns))
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 10:
                break
            if n_try == n_best:
                continue
            if n_try < 100 or n_try > 5000:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 5)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            # Use subgradient for speed at different n
            a_sub, _ = nesterov_subgradient(a_interp, t_sub * 0.5)
            try_update(a_sub, f"multi_n_{n_try}_sg")
            # Then LP if time allows
            rem_sub = t_sub * 0.5 - (time.monotonic() - (deadline - t_sub))
            if rem_sub > 10:
                a_sub2, _ = nesterov_lp_refine(a_sub, min(rem_sub, 60), solver=solver, n_max_iters=5000)
                try_update(a_sub2, f"multi_n_{n_try}_lp")

    # === Phase 6: Deep LP Refinement ===
    print(f"\n=== Phase 6: Deep LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_deep, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_deep, "deep_lp")

    # === Phase 7: Final LogSumExp with very small T ===
    print(f"\n=== Phase 7: Final LSE Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 80.0)
    if t_alloc > 10:
        T_final = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
        a_fin, _ = logsumexp_lbgfs_fft(best_seq, t_alloc, T_stages=T_final)
        try_update(a_fin, "final_lse")

    # === Phase 8: Coordinate Descent Polish ===
    print(f"\n=== Phase 8: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
