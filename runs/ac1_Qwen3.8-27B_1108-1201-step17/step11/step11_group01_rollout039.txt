def kkt_fixed_point(a0, time_limit, max_iters=10000, tol=1e-8, damping=1.0):
    """
    KKT fixed-point iteration for min max_k (conv(a,a))_k s.t. sum(a)=1, a>=0.
    
    At optimality (interior), there exist weights w on the active set such that:
    T_w @ a = (f(a)/n) * 1
    
    We iterate: given a, find w (via QP), then solve T_w @ a_new = const * 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Step 1: Compute autocorrelation and find active set
        b = fast_convolve(a)
        max_b = b.max()
        # Active set: constraints within 99.5% of max
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        # Limit active set size for computational efficiency
        if len(active) > 100:
            # Keep the top 100
            top_vals = np.sort(b)[::-1][:100]
            active = np.argsort(b)[::-1][:100]
        
        # Step 2: Find weights w on active set
        # Solve: min ||M @ w - c*1||^2 s.t. sum(w)=1, w>=0
        # where M[p,k] = a[k-p] (if 0<=k-p<n), and c = max_b/n
        m = len(active)
        M_mat = np.zeros((n, m))
        for i, k in enumerate(active):
            for p in range(n):
                j = k - p
                if 0 <= j < n:
                    M_mat[p, i] = a[j]
        
        c_target = max_b / n
        target = np.full(n, c_target)
        
        # Solve the QP: min ||M @ w - target||^2 s.t. sum(w)=1, w>=0
        # This is equivalent to: min w' M'M w - 2 target' M w + const
        # subject to sum(w) = 1, w >= 0
        A_qp = M_mat.T @ M_mat  # m x m
        b_qp = M_mat.T @ target  # m x 1
        
        # Solve using cvxpy
        w_var = cp.Variable(m, nonneg=True)
        qp_prob = cp.Problem(
            cp.Minimize(cp.quad_form(w_var, cp.psd_wrap(A_qp)) - 2 * b_qp @ w_var),
            [cp.sum(w_var) == 1]
        )
        try:
            qp_prob.solve(solver=get_lp_solver())
        except Exception:
            w_val = np.ones(m) / m
        else:
            if qp_prob.status in ('optimal', 'optimal_inaccurate'):
                w_val = np.asarray(w_var.value, dtype=np.float64)
                w_val = np.maximum(w_val, 0)
                ws = w_val.sum()
                if ws < 1e-15:
                    w_val = np.ones(m) / m
                else:
                    w_val = w_val / ws
            else:
                w_val = np.ones(m) / m
        
        # Step 3: Solve T_w @ a_new = c * 1
        # T_w @ x has p-th component: sum_i w_val[i] * a[active[i] - p] (if valid)
        # We need: sum_i w_val[i] * x[active[i] - p] = c_target for all p
        # This is: M_new @ x = c_target * 1
        # where M_new[p, j] = sum_i w_val[i] * (1 if active[i]-p == j else 0)
        #           = sum_i w_val[i] * (1 if j == active[i]-p else 0)
        
        # Build the matrix T_w (n x n)
        T_w = np.zeros((n, n))
        for i, k in enumerate(active):
            if w_val[i] < 1e-15:
                continue
            for p in range(n):
                j = k - p
                if 0 <= j < n:
                    T_w[p, j] += w_val[i]
        
        # Solve T_w @ a_new = c_target * 1
        try:
            a_candidate = np.linalg.solve(T_w, np.full(n, c_target))
        except np.linalg.LinAlgError:
            a_candidate = np.linalg.lstsq(T_w, np.full(n, c_target), rcond=None)[0]
        
        # Project to simplex
        a_candidate = np.maximum(a_candidate, 0)
        sc = a_candidate.sum()
        if sc < 1e-15:
            a_candidate = a.copy()
        else:
            a_candidate = a_candidate / sc
        
        # Damped update
        a_new = (1 - damping) * a + damping * a_candidate
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        # Check convergence
        if it > 0:
            diff = np.max(np.abs(a_new - a))
            if diff < tol:
                break
        
        a = a_new
    
    return best_a, best_score

"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project onto the unit simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_from_z(z):
    """Compute softmax of z vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert amplitude to log-space."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build the full convolution constraint matrix T where (T@b)_k = sum_j a_fixed[k-j]*b_j."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        js = max(0, j)
        je = min(size, j + n)
        T[j:je, j] = a_fixed[0:je - j]
    return T


_lp_solver = None


def get_lp_solver():
    """Detect available CVXPY solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve: min M s.t. T @ b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000):
    """
    Nesterov-accelerated LP best-response with adaptive restart.
    Solves the bilinear relaxation min_a max_b max_k (a*b)_k.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    restart_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0

        # Adaptive restart: if the new point is worse, reset momentum
        sc_check = score_normalized(b_new)
        if sc_check > best_score + 1e-6 and it > 20:
            t_k = 1.0
            restart_count += 1

        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a_prev = a_old
        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): best_score={best_score:.8f} M_lp={M_lp:.6f} restarts={restart_count}")
                last_print = elapsed

    return best_a, best_score


def Tw_matvec(w_vals, active_k, x):
    """Compute T_w @ x where T_w = sum_i w[i] * T_{active_k[i]}.
    (T_k @ x)_p = x[k-p] if 0 <= k-p < n, else 0.
    """
    n = len(x)
    result = np.zeros(n)
    for i, k in enumerate(active_k):
        wi = w_vals[i]
        if wi < 1e-15:
            continue
        p_start = max(0, k - n + 1)
        p_end = min(k, n - 1)
        if p_start <= p_end:
            # For p in [p_start, p_end]: x[k-p]
            # When p = p_start: x[k - p_start]
            # When p = p_end: x[k - p_end]
            # So we need x[k-p_end : k-p_start+1] reversed
            idx = k - np.arange(p_start, p_end + 1)
            result[p_start:p_end + 1] += wi * x[idx]
    return result


def kkt_fixed_point(a0, time_limit, max_iters=2000, tol=1e-10, max_active=80):
    """
    KKT fixed-point iteration:
    At optimality (interior), there exist weights w on active set A such that
    T_w @ a = (f(a)/n) * 1, where T_w = sum_{k in A} w_k * T_k.
    
    Iterate: given a, find w (QP), solve T_w @ a_new = c*1, project.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Compute autocorrelation and find active set
        b = fast_convolve(a)
        max_b = b.max()

        # Active set: within 99% of max
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > max_active:
            active = np.argsort(b)[::-1][:max_active]

        m = len(active)

        # Step 2: Find weights w via QP
        # Minimize ||M @ w - c*1||^2 s.t. sum(w)=1, w>=0
        # where M[p, i] = a[active[i] - p] (if valid), c = max_b / n
        c_target = max_b / n

        # Build M matrix (n x m)
        M_mat = np.zeros((n, m))
        for i, k in enumerate(active):
            p_start = max(0, k - n + 1)
            p_end = min(k, n - 1)
            if p_start <= p_end:
                p_idx = np.arange(p_start, p_end + 1)
                j_idx = k - p_idx
                M_mat[p_idx, i] = a[j_idx]

        target = np.full(n, c_target)

        # Solve QP using cvxpy
        w_var = cp.Variable(m, nonneg=True)
        A_qp = M_mat.T @ M_mat
        b_qp = M_mat.T @ target
        obj = cp.quad_form(w_var, cp.psd_wrap(A_qp)) - 2.0 * b_qp @ w_var
        qp_prob = cp.Problem(cp.Minimize(obj), [cp.sum(w_var) == 1])
        try:
            qp_prob.solve(solver=get_lp_solver())
        except Exception:
            w_val = np.ones(m) / m
        if qp_prob.status in ('optimal', 'optimal_inaccurate'):
            w_val = np.asarray(w_var.value, dtype=np.float64)
            w_val = np.maximum(w_val, 0)
            ws = w_val.sum()
            if ws < 1e-15:
                w_val = np.ones(m) / m
            else:
                w_val = w_val / ws
        else:
            w_val = np.ones(m) / m

        # Step 3: Solve T_w @ a_new = c_target * 1 using iterative method
        # Use fixed-point: a_new = diag(T_w)^{-1} @ (c*1 - (T_w - diag(T_w)) @ a_cur)
        # Precompute diagonal of T_w: D[p] = sum_i w[i] * (1 if 0 <= active[i]-p < n)
        D = np.zeros(n)
        for i, k in enumerate(active):
            if w_val[i] < 1e-15:
                continue
            p_start = max(0, k - n + 1)
            p_end = min(k, n - 1)
            if p_start <= p_end:
                D[p_start:p_end + 1] += w_val[i]

        # Avoid division by zero
        D_safe = np.maximum(D, 1e-10)

        # Iterate the linear solve
        rhs = np.full(n, c_target)
        a_lin = a.copy()  # warm start
        for inner in range(50):
            # Compute (T_w - D) @ a_lin
            Tw_al = Tw_matvec(w_val, active, a_lin)
            residual = rhs - Tw_al
            # Jacobi step
            a_lin_new = residual / D_safe
            # Damped update
            a_lin = 0.5 * a_lin + 0.5 * a_lin_new
            if inner > 5:
                diff = np.max(np.abs(a_lin_new - a_lin))
                if diff < 1e-12 * c_target:
                    break

        # Project to simplex
        a_new = np.maximum(a_lin, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        # Check score
        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        # Convergence check
        if it > 0:
            prev_best = best_score
            if abs(prev_best - best_score) < tol * best_score:
                pass  # Could break, but let's continue a bit

        a = a_new

        if (it + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 5:
                print(f"    KKT iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n, T_mat=None):
    """
    Compute LogSumExp objective and gradient in z-space.
    a = softmax(z), objective = (2n/T) * log(sum exp(T * (Tmat @ a)))
    where Tmat @ a gives the autocorrelation values.
    """
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 500:
        max_Tb = 500
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient: d/dz [f(a)] = a * (grad_a f - a' grad_a f)
    # grad_a f = (2n/T) * sum_k w_k * d(conv(a,a))_k / da
    # d(conv(a,a))_k / da_p = 2 * a[k-p] (if 0 <= k-p < n)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        p_start = max(0, k - n + 1)
        p_end = min(k, n - 1)
        if p_start <= p_end:
            p_idx = np.arange(p_start, p_end + 1)
            j_idx = k - p_idx
            grad_a[p_idx] += 2.0 * w[k] * a[j_idx]

    grad_a *= (2.0 * n / T_val)

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=20):
    """Multi-stage LogSumExp L-BFGS optimization."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        stage_time = remaining * 0.4
        max_iter = max(10, int(stage_time * 50))

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': max_iter * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.2f} ({elapsed:.0f}s): score={sc:.8f}")
        except Exception:
            pass

    return softmax_from_z(z)
