from scipy.fft import rfft, irfft

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

**Key idea for local polishing: "Peak shaving"**

At each step:
1. Find the position k* where conv(a,a) is maximum
2. The value at k* is sum_j a_j * a_{k*-j}
3. To reduce this, we can:
   - Decrease a_j for j where a_{k*-j} is large (and vice versa)
   - This is like a local "anti-correlation" adjustment

More specifically, the gradient of conv(a,a)_{k*} with respect to a_i is a_{k*-i}. So to decrease conv(a,a)_{k*}, we should decrease a_i in proportion to a_{k*-i}.

But we also need to maintain sum(a) = 1, so we should increase some other positions.

**Refined coordinate descent:**
1. Find k* = argmax conv(a,a)
2. For each i, compute the "cost" of decreasing a_i by δ: this reduces conv(a,a)_{k*} by approximately δ * a_{k*-i}
3. The "benefit" of increasing a_j by δ: this increases conv(a,a)_{k*} by approximately δ * a_{k*-j}
4. So we want to move mass from positions i with large a_{k*-i} to positions j with small a_{k*-j}

This is a linear assignment problem! We want to find the best (i, j) pair to swap.

But doing this naively is O(n²) per step. For n=1200, that's 1.44M operations per step, which is fine.

Actually, we don't need to do the full assignment. A simple approach:
- i* = argmax_i a_i * a_{k*-i} (the position contributing most to the peak)
- j* = argmin_j a_j * a_{k*-j} (the position contributing least)
- Move a small amount of mass from i* to j*

This is O(n) per step and should be effective.

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):

        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

_lp_solver = None

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):

                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        a = a_new
        y = y_new
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed
    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        # numerical overflow protection
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n) * log_sum_exp / T_val
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z
