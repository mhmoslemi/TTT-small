import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def adam_peak_focused(a0, time_limit, T_stages, lr_init, top_k_schedule):
    """Adam optimization with aggressive peak focusing and adaptive top-k.
    
    top_k_schedule: list of (fraction, min_k) pairs for each T_stage
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step_count = 0

    total_stages = len(T_stages)
    steps_per_stage = max(2000, int(time_limit * 30000 / total_stages))

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break

        # Determine top-k for this stage
        if stage_idx < len(top_k_schedule):
            frac, min_k = top_k_schedule[stage_idx]
        else:
            frac, min_k = top_k_schedule[-1]
        top_k = max(min_k, int(frac * (2 * n - 1)))

        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break

            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute gradient with focused weights
            if T_stage < 0.005:
                # Pure subgradient: focus on the single worst peak
                peak_idx = np.argmax(b)
                w = np.zeros(len(b))
                w[peak_idx] = 1.0
            else:
                # Log-sum-exp with top-k
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    max_b_all = b.max()
                    exp_all = np.exp((b - max_b_all) / T_stage)
                    w = exp_all / exp_all.sum()

            # Gradient in a-space
            g_a_raw = np.correlate(w, a, mode='full')
            g_a = g_a_raw[n - 1:2 * n - 1] * 2.0

            # Gradient in z-space
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)

            # Adam update with cosine learning rate decay within stage
            stage_progress = step / max(1, steps_per_stage)
            lr = lr_init * (0.5 + 0.5 * np.cos(np.pi * stage_progress))
            lr *= (0.7 ** stage_idx)

            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Centering for numerical stability
            z -= z.mean()

            # Progress
            if (step + 1) % 15000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam stage {stage_idx} (T={T_stage:.4f}) step {step+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_rounds=50000):
    """Coordinate descent on the worst peak using small clashes."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_pos = int(np.argmax(b))
        max_b = b[peak_pos]

        # Find which a[i] contribute to this peak
        # conv(a,a)[k] = sum_i a[i]*a[k-i]
        i_start = max(0, peak_pos - n + 1)
        i_end = min(n - 1, peak_pos)

        if i_start > i_end:
            break

        # Try reducing each contributing element
        best_delta = 0
        best_i = -1
        delta = max_b * 0.005  # small perturbation

        # Sample a few indices for speed
        candidates = np.arange(i_start, i_end + 1)
        if len(candidates) > 20:
            candidates = rng.choice(candidates, size=20, replace=False)

        for i in candidates:
            j = peak_pos - i
            if j < 0 or j >= n:
                continue
            # Try reducing a[i]
            a_trial = a.copy()
            reduction = a[i] * 0.01
            if reduction < 1e-10:
                continue
            a_trial[i] = a[i] - reduction
            # Redistribute to j
            a_trial[j] = a[j] + reduction
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            b_t = fast_convolve(a_trial)
            if b_t.max() < max_b - 1e-10:
                best_delta = -1
                best_i = i
                best_j = j
                best_red = reduction
                break

        if best_i >= 0:
            a[best_i] = a[best_i] - best_red
            a[best_j] = a[best_j] + best_red
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        else:
            # Try a random small perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            a_trial = a.copy()
            transfer = a[i] * 0.005
            a_trial[i] -= transfer
            a_trial[j] += transfer
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                score_t = 2.0 * n * fast_convolve(a_trial).max()
                if score_t < best_score - 1e-10:
                    a = a_trial
                    best_score = score_t
                    best_a = a.copy()

        if rnd % 5000 == 0 and rnd > 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc rnd={rnd} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def dual_weighted_subgradient(a0, time_limit, solver, n_rounds=2000):
    """Use LP dual information to get optimal subgradient weights."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.02
    no_improve = 0

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit - 5:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Get LP-based active set and weights
        b_lp, M_lp, active = solve_lp_best_response(a, solver)

        if active is not None and len(active) > 0:
            # Use the active set with uniform weights (or LP-based)
            w = np.zeros(len(b))
            w[active] = 1.0 / len(active)
        else:
            # Fallback: top-5 peaks
            idx = np.argpartition(b, -5)[-5:]
            w = np.zeros(len(b))
            w[idx] = 1.0 / 5

        # Subgradient
        g = np.zeros(n)
        g_raw = np.correlate(w, a, mode='full')
        g = g_raw[n - 1:2 * n - 1] * 2.0

        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            eta *= 0.7
            no_improve = 0
            if eta < 0.001:
                break

        if rnd % 100 == 0 and rnd > 0:
            elapsed = time.monotonic() - t0
            print(f"    DualSG rnd={rnd} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.5f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {budget_s:.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: LP Alternating (main workhorse, 45% time) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 700.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=1000000)
        local_update(a_lp, "lp_phase1")

    # === Phase 2: Adam with aggressive peak focusing (30% time) ===
    print(f"\n=== Phase 2: Adam Peak-Focused (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15,
                    0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
        n = len(best_seq)
        top_k_sched = [
            (0.5, 100), (0.4, 80), (0.3, 60), (0.25, 50), (0.2, 40),
            (0.15, 30), (0.12, 25), (0.1, 20), (0.08, 15), (0.06, 12),
            (0.05, 10), (0.04, 8), (0.03, 6), (0.02, 5), (0.02, 4),
            (0.01, 3), (0.01, 2), (1, 2), (1, 1)
        ]
        a_adam, _ = adam_peak_focused(best_seq, t_alloc, T_stages,
                                       lr_init=0.004, top_k_schedule=top_k_sched)
        local_update(a_adam, "adam_phase2")

    # === Phase 3: Dual-weighted subgradient (10% time) ===
    print(f"\n=== Phase 3: Dual-Weighted Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_dsg, _ = dual_weighted_subgradient(best_seq, t_alloc, solver, n_rounds=500)
        local_update(a_dsg, "dual_sg")

    # === Phase 4: Second LP round (10% time) ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=200000)
        local_update(a_lp2, "lp_phase4")

    # === Phase 5: Fine Adam polish (5% time) ===
    print(f"\n=== Phase 5: Fine Adam Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 80.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        T_fine = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
        top_k_fine = [(0.03, 5), (0.02, 4), (0.02, 3), (0.01, 3),
                      (0.01, 2), (1, 2), (1, 1), (1, 1)]
        a_fine, _ = adam_peak_focused(best_seq, t_alloc, T_fine,
                                       lr_init=0.001, top_k_schedule=top_k_fine)
        local_update(a_fine, "fine_adam")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coord Descent Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_polish(best_seq, t_alloc, n_rounds=30000)
        local_update(a_cd, "coord_desc")

    # === Phase 7: Try different n values ===
    print(f"\n=== Phase 7: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_n_explore = min(rem * 0.25, 200.0)
        print(f"  Budget: {t_n_explore:.0f}s")
        for n_try in [n_best - 100, n_best + 100, n_best + 200]:
            if n_try < 500 or n_try > 3000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            t_sub = min(rem_sub * 0.35, 60.0)
            init_try = interpolate_sequence(best_seq, n_try)
            T_sub = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            top_k_sub = [(0.1, 10), (0.05, 8), (0.03, 6), (0.02, 5),
                         (0.02, 4), (0.01, 3), (1, 2), (1, 1)]
            a_try, _ = adam_peak_focused(init_try, t_sub, T_sub,
                                          lr_init=0.003, top_k_schedule=top_k_sub)
            local_update(a_try, f"n={n_try}")

    # === Phase 8: Perturbation restarts ===
    print(f"\n=== Phase 8: Perturbation Restarts (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_pert = min(rem * 0.3, 150.0)
        print(f"  Budget: {t_pert:.0f}s")
        n_cur = len(best_seq)
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            strength = 0.008 + 0.008 * trial
            x = np.linspace(0, 2 * np.pi, n_cur)
            freq = rng.integers(1, 15)
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = best_seq + strength * np.cos(freq * x + phase)
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert / s_p
            t_sub = min(rem_sub * 0.3, 40.0)
            T_sub = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
            top_k_sub = [(0.05, 8), (0.03, 6), (0.02, 5), (0.02, 4),
                         (0.01, 3), (0.01, 2), (1, 1), (1, 1)]
            a_opt, _ = adam_peak_focused(a_pert, t_sub, T_sub,
                                          lr_init=0.002, top_k_schedule=top_k_sub)
            local_update(a_opt, f"pert_t{trial}")

    # === Phase 9: Final LP + Adam squeeze ===
    print(f"\n=== Phase 9: Final Squeeze (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_final_lp, _ = nesterov_lp_refine(best_seq, min(rem * 0.4, 30.0), solver, n_max_iters=5000)
        local_update(a_final_lp, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 10:
        T_final = [0.05, 0.03, 0.01, 0.005, 0.003, 0.001]
        top_k_final = [(0.02, 3), (0.01, 3), (1, 2), (1, 1), (1, 1), (1, 1)]
        a_final, _ = adam_peak_focused(best_seq, min(rem * 0.5, 20.0), T_final,
                                        lr_init=0.0005, top_k_schedule=top_k_final)
        local_update(a_final, "final_adam")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        elapsed = time.monotonic() - t_start
        print(f"\nElapsed: {elapsed:.0f}s")
        print(f"FINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
