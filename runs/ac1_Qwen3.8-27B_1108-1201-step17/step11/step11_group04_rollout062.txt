For the LogSumExp: the objective is:
f_T(a) = 2n * T * log(sum_k exp(conv(a,a)[k]/T))

As T -> 0, this approaches 2n * max_k conv(a,a)[k].

The gradient is:
grad f_T = 2n * sum_k w_k * grad conv(a,a)[k]
where w_k = exp(conv(a,a)[k]/T) / sum_j exp(conv(a,a)[j]/

For very small T, the weights w_k are concentrated on the argmax, and the gradient is essentially the subgradient of the max function.

The key challenge is numerical stability: for T = 0.001 and conv values around 0.0006, conv/T is around 600, and exp(600) overflows. We need to use the log-sum-exp trick.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive LogSumExp L-BFGS with ultra-low temperatures
2. New iterative convolution-targeting spectral method
3. Active-set subgradient with backtracking
4. Multi-n exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def logsumexp_obj_grad(z, n, T, deadline=None):
    """Compute LogSumExp objective and gradient w.r.t. z (log-domain)."""
    if deadline is not None and time.monotonic() > deadline:
        return 1e15, np.zeros_like(z)
    
    # z -> a via softmax
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    max_b = b.max()
    
    # Stable logsumexp
    scaled = (b - max_b) / T
    exp_scaled = np.exp(scaled)
    lse = max_b + T * np.log(exp_scaled.sum())
    obj = 2.0 * n * lse
    
    # Weights for gradient
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient of max_k conv(a,a)[k] w.r.t. a (weighted)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    
    # Chain rule: d/da -> dz via softmax
    # a = softmax(z), so da/dz = a*(I - a*a^T)
    # grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, 2.0 * n * grad_z

def lbgfs_refine(a0, time_limit, T_stages, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = np.size(a0)
    a0 = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
    a0 = a0 / s
    
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    
    best_z = z.copy()
    best_obj = 2.0 * n * fast_convolve(a0).max()
    
    for stage_T in T_stages:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break
        
        # Estimate iterations: each iter is ~n*2n/fft_size operations for the conv
        # For n=1200, fft_size=4096, each conv is ~4096*64 = 262K ops
        # L-BFGS does ~1-3 function evals per iter, each needing conv
        # So ~10-20K evals per second roughly
        estimated_evals_per_sec = 5000  # conservative
        iters = max(10, int(remaining * estimated_evals_per_sec / max(1, len(T_stages) - T_stages.index(stage_T))))
        iters = min(iters, 50000)
        
        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            o, _ = logsumexp_obj_grad(z_in, n, stage_T, deadline)
            return o
        
        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T, deadline)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 200,
                         'maxls': 50, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        # Check improvement
        a_new = softmax_from_z(z)
        sc = evaluate_local(a_new)
        if sc < best_obj:
            best_obj = sc
            best_z = z.copy()
    
    return softmax_from_z(best_z)

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def active_set_subgradient(a0, time_limit, n_iter=100000, tol=1e-6):
    """
    Subgradient method focusing on active peaks.
    Uses adaptive step size and backtracking.
    """
    n = np.size(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.01
    stall = 0
    window = 100  # peaks within this fraction of max
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Active set: peaks within window of max
        threshold = M * (1.0 - window)
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute weighted subgradient
        # Weight peaks by how close they are to M
        weights = (b[active] - threshold) / max(1e-30, M - threshold)
        weights = weights / weights.sum()
        
        g = np.zeros(n)
        for i, k in enumerate(active):
            if weights[i] < 1e-12:
                continue
            g += weights[i] * _conv_grad_component(a, k)
        
        # Subgradient step with projection onto simplex
        a_new = a - eta * g
        a_new = _project_simplex(a_new)
        
        # Check improvement
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # still update (momentum-like)
            stall += 1
        
        if stall >= 30:
            stall = 0
            if it < 50:
                eta *= 0.5
            else:
                # Try a different step or reset
                eta = max(0.001, eta * 0.7)
        
        if stall >= 200:
            break
        
        if (it + 1) % 5000 == 0:
            print(f"    SubGrad iter={it+1}: score={best_score:.8f}, eta={eta:.6f}, n_active={len(active)}")
    
    return best_a, best_score

def _conv_grad_component(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a."""
    n = np.size(a)
    g = np.zeros(n)
    # d conv(a,a)[k] / da[j] = a[k-j] + a[j-k] (with bounds)
    for j in range(n):
        val = 0.0
        if 0 <= k - j < n:
            val += a[k - j]
        if 0 <= j - k < n:
            val += a[j - k]
        g[j] = val
    return g

```python
def conv_flatten(a, target, n_iters=1000):
    """Iteratively flatten the convolution toward a target level."""
    n = np.size(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    
    fft_size = 1 << (2*n - 2).bit_length()
    
    for it in range(n_iters):
        # Compute current convolution
        A = np.fft.fft(a, fft_size)
        b = np.fft.ifft(A * A, fft_size).real[:2*n-1]
        b = np.maximum(b, 0)
        
        # Flatten: cap at target
        b_new = np.minimum(b, target)
        s_b = b_new.sum()
        if s_b < 1e-15:
            break
        b_new = b_new / s_b  # normalize so sum = 1
        
        # Go to frequency domain
        B = np.fft.fft(b_new, fft_size)
        
        # Take square root (magnitude and phase)
        B_mag = np.abs(B)
        B_phase = np.angle(B)
        A_mag = np.sqrt(B_mag)
        A_phase = B_phase / 2.0
        A_new = A_mag * np.exp(1j * A_phase)
        
        # Ensure A_new[0] is real and positive (DC component)
        A_new[0] = abs(A_new[0])
        
        # Conjugate symmetry (for real a)
        # A_new[L-k] should be conj(A_new[k])
        # This is automatically satisfied if B has conjugate symmetry and we take sqrt properly
        # But numerical errors might break it, so enforce:
        for k in range(1, fft_size // 2 + 1):
            A_new[fft_size - k] = np.conj(A_new[k])
        
        a_new = np.fft.ifft(A_new, fft_size).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a
```
