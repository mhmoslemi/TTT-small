import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence locally (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_lp_matrix(a):
    """Build (2n-1) x n matrix T where T[k,j] = a[k-j] if valid."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    data = []
    for j in range(n):
        # a[k-j] is non-zero when 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_start = j
        k_end = min(size - 1, j + n - 1)
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * (k_end - k_start + 1))
        data.extend(a)
    return csr_matrix((data, (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))), shape=(size, n))

    for name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

            prob.solve(solver=name)

                _lp_solver = name
                return name
        except Exception:

    _lp_solver = cp.ECOS
    return _lp_solver


def solve_best_response(a, solver):
    """Solve LP: min t s.t. conv(a,b) <= t, sum(b)=1, b>=0."""
    n = len(a)
    size = 2 * n - 1
    T = build_lp_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        return None
    if b_var.value is None:
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp(a0, time_limit, solver, max_iters=500000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_best_response(y, solver)
        if b_new is None:
            break

        # Nesterov step with decaying momentum
        mu = 2.0 / (np.sqrt(float(it + 1)) + 2.0)
        y = (1.0 - mu) * y + mu * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # FTL-average for a
        a = ((float(it + 1)) * a + mu * float(it + 2) * b_new) / (float(it + 1) + mu * float(it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        a = a / a.sum()

        # Check improvement
        score_y = 2.0 * n * fast_convolve(y).max()
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Restart on prolonged stall
        if stall > 30:
            # Perturb toward a random direction
            noise = rng.standard_normal(n) * 0.01
            y = y + noise
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            a = y.copy()
            stall = 0
            mu = 1.0  # Reset momentum

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def logsumexp_obj_grad(a, T):
    """Compute LogSumExp smoothed objective and gradient."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = T * (b - max_b)
    exp_b = np.exp(shifted)
    w = exp_b / exp_b.sum()

    # grad_i = 4n * sum_j w[i+j] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        valid = min(n, 2 * n - 1 - i)
        grad[i] = np.dot(w[i:i + valid], a[:valid])
    grad *= 4.0 * n

    logsumexp_val = T * max_b + np.log(exp_b.sum())
    obj = (2.0 * n / T) * logsumexp_val

    return obj, grad


def lse_lbgfs(a0, time_limit, T_stages=None, maxiter=2000):
    """L-BFGS on LogSumExp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015]

    best_a = a.copy()
    best_score = evaluate_local(a)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 3:
            break

        def func(z):
            if time.monotonic() - t0 > time_limit - 1:
                return 1e15
            a_t = np.maximum(z, 0)
            st = a_t.sum()
            if st < 1e-15:
                return 1e15
            a_t = a_t / st
            obj, _ = logsumexp_obj_grad(a_t, T)
            return obj

        def jac(z):
            if time.monotonic() - t0 > time_limit - 1:
                return np.zeros(len(z))
            a_t = np.maximum(z, 0)
            st = a_t.sum()
            if st < 1e-15:
                return np.zeros(len(z))
            a_t = a_t / st
            _, g = logsumexp_obj_grad(a_t, T)
            return g

        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-16,
                                    'gtol': 1e-13, 'maxfun': maxiter * 500,
                                    'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception:
            pass

        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        elapsed = time.monotonic() - t0
        print(f"    LSE T={T:.3f} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coord_descent(a0, time_limit, max_steps=500000):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Find active peaks
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        improved = False
        for pk in peaks[:5]:
            if improved:
                break
            # Find top contributors to this peak
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_n = min(10, len(idx))
            top_pos = idx[np.argsort(contribs)[::-1][:top_n]]

            for pos in top_pos:
                if a[pos] < 1e-14:
                    continue
                # Try reducing pos and redistributing
                for factor in [0.999, 0.995, 0.99, 0.98, 0.97, 0.95]:
                    a_t = a.copy()
                    a_t[pos] *= factor
                    diff = a[pos] - a_t[pos]
                    if diff < 1e-15:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_t[mask]
                    os_sum = others.sum()
                    if os_sum < 1e-15:
                        break
                    a_t[mask] = others + diff * others / os_sum
                    a_t = a_t / a_t.sum()
                    sc = evaluate_local(a_t)
                    if sc < best_score - 1e-13:
                        a = a_t
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 200:
                # Try pairwise: reduce top contributor, increase a random other
                pk = int(np.argmax(b))
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[pk - idx]
                top_pos = idx[np.argsort(contribs)[::-1][:2]]
                # Find low-contributing positions
                low_pos = idx[np.argsort(contribs)[:5]]

                for tp in top_pos:
                    for lp in low_pos:
                        if tp == lp:
                            continue
                        for ratio in [0.99, 0.98, 0.97]:
                            a_t = a.copy()
                            amount = a_t[tp] * (1 - ratio)
                            a_t[tp] -= amount
                            a_t[lp] += amount
                            a_t = np.maximum(a_t, 0)
                            a_t = a_t / a_t.sum()
                            sc = evaluate_local(a_t)
                            if sc < best_score - 1e-13:
                                a = a_t
                                best_score = sc
                                best_a = a.copy()
                                stall = 0
                                break
                        if stall == 0:
                            break
                    if stall == 0:
                        break
            if stall > 2000:
                break
        else:
            stall = 0

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_n)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def mirror_descent(a0, time_limit, eta0=0.08):
    """Mirror descent (exponential family) on the max-conv objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    no_improve = 0
    eta = eta0
    rng = np.random.default_rng(777)

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        thresh = max_b * 0.9999
        active = np.where(b >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))

        # Mirror descent step (exponential family)
        log_a = np.log(np.maximum(a, 1e-300))
        g_shift = g - g.min()
        log_a_new = log_a - eta * g_shift
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        a = a_new

        sc = evaluate_local(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive eta
        if no_improve > 100:
            eta *= 0.95
            no_improve = 0
        if no_improve > 500:
            noise = rng.standard_normal(n) * 0.005
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            eta = eta0

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
