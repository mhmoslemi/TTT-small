Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LogSumExp L-BFGS with multi-stage temperature annealing (primary method)
2. Improved Nesterov LP with adaptive step sizes and warm-starting
3. Direct active-set subgradient descent with projected line search
4. Coordinate descent final polish
5. Multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

import time
from scipy.fft import fft, ifft, rfft, irfft

def fast_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log of a (for softmax parametrization)."""
    return np.log(np.maximum(a, 1e-300))


def lse_objective_grad(z_in, T, n):
    """
    Compute LSE objective and gradient w.r.t. z.
    f_T(z) = (2n/T) * log(sum_k exp(T * c_k(softmax(z))))
    Returns (obj, grad_z) where a = softmax(z).
    """
    a = softmax_from_z(z_in)
    c = fast_convolve(a)  # length 2n-1
    
    # Stable log-sum-exp
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    S = exp_Tc.sum()
    w = exp_Tc / S  # normalized weights, length 2n-1
    wc = float(np.dot(w, c))  # weighted average of c
    obj = 2.0 * n * (Tc_max + np.log(S)) / T
    
    # Compute g_l = sum_k w[k] * a[k-l] for l = 0,...,n-1
    # g[l] = (a_rev * w)[n-1+l] where a_rev = a[::-1]
    a_rev = a[::-1]
    N = 1 << (3 * n - 2).bit_length()
    A_rev = fft(a_rev, N)
    W = fft(w, N)
    conv_result = ifft(A_rev * W).real
    g = conv_result[n - 1:2 * n - 1].copy()
    
    # Gradient: 4n * a * (g - wc)
    grad_z = 4.0 * n * a * (g - wc)
    
    return obj, grad_z


def lse_lbgfs(a0, time_limit, T_start=50.0, T_end=300.0, n_stages=20):
    """
    Multi-stage LSE L-BFGS optimization.
    Start with smooth (low T) and progress to sharp (high T ≈ max).
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 80
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad
        
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0 or si == n_stages - 1:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.1f}: score={sc:.8f} ({elapsed:.0f}s)")
        except Exception as e:
            print(f"    LSE stage {si+1} error: {e}")
            break
    
    return softmax_from_z(z)


def get_lp_solver():

def build_conv_matrix_rows(a_fixed, rows):
    """Build only the specified rows of the convolution matrix T."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((len(rows), n))
    for ii, k in enumerate(rows):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                T[ii, j] = a_fixed[idx]
    return T


def solve_lp_fast(a_fixed, active_rows, M_ub, solver):
    """
    Solve reduced LP: min M s.t. T_active @ b <= M, sum(b)=1, b>=0.
    Only uses active constraints (rows near the max).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_rows(a_fixed, active_rows)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def subgradient_polish(a0, time_limit, n_steps=200000):
    """
    Active-set subgradient descent with projected line search.
    Works directly on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Adaptive step size
    eta = 0.1
    patience = 0
    max_patience = 50
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step % 1000 == 0 and patience > max_patience:
            eta *= 0.5
            patience = 0
            if eta < 1e-6:
                break
        
        b = fast_convolve(a)
        b_max = b.max()
        # Active set: k values near max
        active = np.where(b >= b_max * 0.98)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute subgradient of max(c_k) w.r.t. a
        # grad_a[m] = (2/|active|) * sum_{k in active} a[k-m]
        g = np.zeros(n)
        for k in active:
            # a[k-m] is valid when 0 <= k-m < n, i.e., max(0,k-n+1) <= m <= min(n-1, k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                g[js:je + 1] += a[k - je:k - js + 1][::-1]
        g *= 2.0 / len(active)
        
        # Project gradient onto tangent space of simplex (sum=1): subtract mean
        g_tangent = g - g.mean()
        
        # In z-space: grad_z = a * (g_tangent - a @ g_tangent)
        # But a @ g_tangent = 0 since g_tangent has zero mean and a sums to 1... 
        # Actually not quite. Let's use: grad_z = a * g_tangent (approximately, since sum(a*g_tangent) ≈ 0)
        S = float(np.dot(a, g_tangent))
        grad_z = a * (g_tangent - S)
        
        # Update
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        
        sc_new = score_normalized(a_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            z = z_new
            a = a_new
            patience = 0
        else:
            patience += 1
    
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_passes=100):
    """
    Coordinate descent on the simplex.
    For each pair (i,j), find optimal mass transfer.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for pas in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Try a subset of random pairs
        n_trials = min(500, n * 2)
        for _ in range(n_trials):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            
            # 1D optimization: transfer delta from j to i
            # a_i + delta, a_j - delta
            # Constraint: 0 <= delta <= a_j
            
            def obj_1d(delta):
                a_trial = a.copy()
                a_trial[i] += delta
                a_trial[j] -
