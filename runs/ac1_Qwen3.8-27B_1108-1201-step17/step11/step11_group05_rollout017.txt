"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated projected subgradient with adaptive LogSumExp temperature
2. LP-based best response for periodic escape from local minima
3. Multi-start exploration with structured initializations
4. Simplex projection (O(n log n)) for constraint handling

Key approach: Fast O(n) per-step subgradient (90% of time) with periodic LP calls
(10% of time) for large-scale moves. Uses Polyak step size with theoretical lower bound.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    mu = np.zeros(n)
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)
    rho = np.nonzero(sorted_v * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    mu = np.maximum(v - theta, 0)
    return mu


def compute_subgradient(a, b, T):
    """Compute subgradient of soft-max objective w.r.t. a.
    Returns gradient g where g_i = 2 * sum_k w_k * a_{k-i}
    """
    n = len(a)
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / T)
    w = exp_b / exp_b.sum()
    # g_i = 2 * sum_j w[i+j] * a_j for j=0..n-1 (w defined on [0, 2n-2])
    # This is cross-correlation
    g = 2.0 * np.correlate(w, a, mode='full')[n - 1:2 * n - 1]
    return g


def score_local(a):
    """Compute objective score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def nesterov_subgradient(a0, time_limit, T_start=10.0, T_end=0.01,
                         step_init=0.05, momentum_init=0.85, momentum_end=0.98,
                         check_interval=2000, verbose=True):
    """Nesterov-accelerated projected subgradient with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Y position (Nesterov)
    y = a.copy()
    v = np.zeros(n)  # momentum accumulator
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    deadline = t0 + time_limit
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    step_count = 0
    no_improve = 0
    patience = 50000

    total_steps_est = int(time_limit * 8000)  # rough estimate of steps possible
    for step in range(10**9):
        if time.monotonic() > deadline:
            break

        # Annealing parameters
        frac = min(1.0, step / max(1, total_steps_est))
        T = T_start * (T_end / T_start) ** frac
        momentum = momentum_init + (momentum_end - momentum_init) * frac
        step_size = step_init * (1.0 + 0.5 * frac) ** (-1)

        # Evaluate at y
        b = fast_convolve(y)
        max_b = b.max()
        obj_val = 2.0 * n * max_b

        # Compute subgradient
        g = compute_subgradient(y, b, T)

        # Polyak-style step size adjustment
        gap = obj_val - lower_bound
        if gap > 1e-10:
            g_norm_sq = np.dot(g, g)
            if g_norm_sq > 1e-15:
                polyak_step = 0.5 * gap / g_norm_sq
                step_size = min(step_size, polyak_step * 5.0)
                step_size = max(step_size, 1e-7)

        # Nesterov update
        v = momentum * v + g
        y_new = y - step_size * v

        # Project to simplex
        a_new = simplex_project(y_new)

        # Ensure non-negative
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        # Check improvement
        step_count += 1
        if step_count % 100 == 0:
            b_check = fast_convolve(a_new)
            score_check = 2.0 * n * b_check.max()
            if score_check < best_score - 1e-12:
                best_score = score_check
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 100

            # Reduce step size if stuck
            if no_improve > patience:
                step_size *= 0.7
                no_improve = 0
                if step_size < 1e-9:
                    break

            if verbose and step_count % check_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"      NS step={step_count} T={T:.4f} best={best_score:.8f} cur={obj_val:.8f} ({elapsed:.0f}s)")

        y = y_new
        a = a_new

    return best_a, best_score


def build_sparse_conv_matrix(a):
    """Build the convolution operator as a function (avoids storing full matrix)."""
    n = len(a)

    def apply(b_vec):
        """Apply T @ b_vec where T[k,j] = a[j] if 0<=k-j<n."""
        return fast_convolve(b_vec)  # This computes conv(a, b_vec) incorrectly
        # Actually we need conv of a and b_vec
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        B = rfft(b_vec, fft_size)
        return irfft(A * B, fft_size)[:size]

    return apply


def lp_best_response(a_fixed, n_iters_lp=50, verbose=False):
    """Solve LP: max sum(b) s.t. conv(a_fixed, b) <= M, b >= 0, sum(b)=1.
    Uses projected subgradient as a fallback if cvxpy is too slow.
    Returns (b_opt, M_opt).
    """
    try:
        import cvxpy as cp

        n = len(a_fixed)
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        # Build constraint matrix (sparse)
        size = 2 * n - 1
        # Use dense for simplicity (n up to ~2000 is OK)
        T = np.zeros((size, n))
        for j in range(n):
            T[j:j + n, j] = a_fixed

        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        # Try multiple solvers
        solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
        result = None
        for solver in solvers:
            try:
                prob.solve(solver=solver)
                if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                    result = (b_var.value, M_var.value)
                    break
            except Exception:
                continue

        if result is None:
            return None, None

        b_val = np.asarray(result[0], dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        b_val[np.isnan(b_val)] = 0
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        return b_val, float(max(0.0, result[1]))

    except Exception:
        return None, None


def lp_alternating(a0, time_limit, max_iterations=200, verbose=True):
    """LP alternating with Nesterov dual averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    deadline = t0 + time_limit
    lp_times = []

    for it in range(max_iterations):
        if time.monotonic() > deadline:
            break

        # Solve LP for best response
        lp_t0 = time.monotonic()
        b_new, M_val = lp_best_response(y)
        lp_time = time.monotonic() - lp_t0
        lp_times.append(lp_time)

        if b_new is None:
            continue

        # Nesterov dual averaging update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if verbose and (it + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times[-10:])
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={avg_lp*1000:.0f}ms")

        # Stop if LP is too slow
        if len(lp_times) > 10 and np.mean(lp_times[-10:]) > 10.0:
            if verbose:
                print(f"      LP too slow ({np.mean(lp_times[-10:]):.1f}s/iter), stopping.")
            break

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)

        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * x) + 0.15 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0.01)
    elif choice == 7:
        a = 0.5 + 0.2 * np.cos(2 * np.pi * 5 * x) * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0.01)
    elif choice == 8:
        # Multi-scale cosine
        a = np.ones(n)
        for freq in [2, 5, 11]:
            a += 0.1 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def freq_domain_refine(a0, time_limit, n_freq_iters=10000, verbose=True):
    """Refine in frequency domain: alternate between time (non-negativity + sum=1)
    and frequency (flatten |A|^2) domains."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    T_anneal_start = 5.0
    T_anneal_end = 0.1

    for it in range(n_freq_iters):
        if time.monotonic() > deadline:
            break

        # Forward FFT
        A = rfft(a, fft_size)

        # Target: make |A|^2 as flat as possible (but A^2 should give flat auto-conv)
        # The auto-convolution is IDFT(A^2). We want this to be flat.
        # So we want A^2 to be "as delta-like at frequency 0 as possible"
        # Actually, for a flat auto-conv (value 1/(2n-1) everywhere),
        # A^2 should have DFT = (2n-1)/(2n-1) at freq 0 only... no that gives a constant auto-conv.
        # IDFT(A^2) = flat means A^2 = DFT(flat) = c at freq 0, 0 elsewhere.
        # So A should be non-zero only at freq 0, meaning a is constant.
        # That's the uniform sequence! So this approach just converges to uniform.
        #
        # Instead, let's minimize the peak of IDFT(A^2) by adjusting |A|.
        # Keep phase of A, adjust magnitude to reduce peak auto-conv.

        current_auto_conv = irfft(A * A, fft_size)[:2 * n - 1]
        max_ac = current_auto_conv.max()

        if max_ac < best_score / (2.0 * n):
            best_score = 2.0 * n * max_ac
            best_a = a.copy()

        # Adjust magnitude: reduce components that contribute to peaks
        phase_A = np.angle(A)
        mag_A = np.abs(A)

        # Add small regularization toward flat magnitude
        T_anneal = T_anneal_start * (T_anneal_end / T_anneal_start) ** (it / n_freq_iters)
        target_mag = np.mean(mag_A)
        mag_new = mag_A + 0.01 * (target_mag - mag_A)  # slight pull toward flat
        mag_new = np.maximum(mag_new, 0)

        A_new = mag_new * np.exp(1j * phase_A)
        a_new = irfft(A_new, fft_size)[:n]

        # Enforce non-negativity and normalize
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new

        if verbose and (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            cur_score = 2.0 * n * fast_convolve(a).max()
            print(f"      FreqRefine it={it+1} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f}")

    return best_a, best_score


def coordinate_polish(a0, n_rounds=5, verbose=True):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n * fast_convolve(a).max()
    a = a / s

    best_score = 2.0 * n * fast_convolve(a).max()
    total_ops = n * n_rounds
    done = 0
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        for i in range(n):
            done += 1
            if time.monotonic() - t0 > 60:  # cap at 60s
                break

            # Binary search for optimal a[i]
            lo = 0.0
            hi = a.sum() - (n - 1) * 1e-15
            if hi <= lo:
                hi = 1.0

            for _ in range(40):
                mid = (lo + hi) / 2
                a_trial = a.copy()
                a_trial[i] = mid
                sa = a_trial.sum()
                if sa < 1e-15:
                    break
                a_trial = a_trial / sa
                b_trial = fast_convolve(a_trial)
                score_trial = 2.0 * n * b_trial.max()

                a_plus = a.copy()
                a_plus[i] = mid + 0.001
                sp = a_plus.sum()
                if sp > 1e-15:
                    a_plus = a_plus / sp
                    score_plus = 2.0 * n * fast_convolve(a_plus).max()
                    if score_plus < score_trial:
                        lo = mid
                    else:
                        hi = mid
                else:
                    hi = mid

                # Also try decreasing
                a_minus = a.copy()
                a_minus[i] = max(0, mid - 0.001)
                sm = a_minus.sum()
                if sm > 1e-15:
                    a_minus = a_minus / sm
                    score_minus = 2.0 * n * fast_convolve(a_minus).max()
                    if score_minus < score_trial:
                        hi = mid
                    else:
                        lo = mid

            a[i] = (lo + hi) / 2
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa

        if verbose and (rnd + 1) % 2 == 0:
            cur_score = 2.0 * n * fast_convolve(a).max()
            if cur_score < best_score:
                best_score = cur_score
            print(f"      CoordPolish round={rnd+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    a = np.maximum(a, 0)
    sa = a.sum()
    if sa > 1e-15:
        a = a / sa

    return a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Nesterov subgradient (fast, 50% time)
    Phase 2: LP alternating (accurate, 25% time)
    Phase 3: Fine subgradient (20% time)
    Phase 4: Exploration & polish (5% time)
    """
    rng = np.random.default_rng(seed)
    global_start = time.monotonic()
    deadline = global_start + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    print(f"Budget: {budget_s:.0f}s, deadline in {deadline - global_start:.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = score_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: Nesterov Subgradient (main driver, 50% time) ===
    rem = deadline - time.monotonic()
    t_p1 = min(rem * 0.50, 800.0)
    if t_p1 > 30:
        print(f"\n=== Phase 1: Nesterov Subgradient n={n_best} (score={best_score:.8f}, {t_p1:.0f}s) ===")
        a_p1, _ = nesterov_subgradient(
            best_seq, t_p1,
            T_start=20.0, T_end=0.05,
            step_init=0.03, momentum_init=0.80, momentum_end=0.97,
            check_interval=5000
        )
        update_best(a_p1, "nesterov_p1")

    # === Phase 2: LP Alternating (25% time) ===
    rem = deadline - time.monotonic()
    t_p2 = min(rem * 0.25, 400.0)
    if t_p2 > 30:
        print(f"\n=== Phase 2: LP Alternating n={len(best_seq)} (score={best_score:.8f}, {t_p2:.0f}s) ===")
        a_p2, _ = lp_alternating(best_seq, t_p2, max_iterations=500, verbose=True)
        update_best(a_p2, "lp_p2")

    # === Phase 3: Fine Nesterov Subgradient (20% time) ===
    rem = deadline - time.monotonic()
    t_p3 = min(rem * 0.20, 300.0)
    if t_p3 > 20:
        print(f"\n=== Phase 3: Fine Nesterov (score={best_score:.8f}, {t_p3:.0f}s) ===")
        a_p3, _ = nesterov_subgradient(
            best_seq, t_p3,
            T_start=1.0, T_end=0.005,
            step_init=0.01, momentum_init=0.90, momentum_end=0.99,
            check_interval=5000
        )
        update_best(a_p3, "nesterov_p3")

    # === Phase 4: Try different lengths (10% time) ===
    rem = deadline - time.monotonic()
    if rem > 100:
        print(f"\n=== Phase 4: Multi-length exploration (score={best_score:.8f}, {rem:.0f}s) ===")
        t_per_len = rem * 0.3
        n_current = len(best_seq)
        candidate_ns = [n for n in [800, 1000, 1500, 2000] if n != n_current]

        for n_trial in candidate_ns[:2]:  # Try up to 2 new lengths
            rem_sub = deadline - time.monotonic()
            if rem_sub < 50:
                break
            t_sub = min(rem_sub * 0.45, 200.0)
            print(f"  Trying n={n_trial} ({t_sub:.0f}s)")
            init_trial = interpolate_sequence(best_seq, n_trial)
            # Quick subgradient
            a_trial, _ = nesterov_subgradient(
                init_trial, t_sub,
                T_start=5.0, T_end=0.05,
                step_init=0.02, momentum_init=0.85, momentum_end=0.97,
                check_interval=10000
            )
            update_best(a_trial, f"n{n_trial}")

    # === Phase 5: Frequency domain refinement ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 5: Freq Domain Refine (score={best_score:.8f}, {rem:.0f}s) ===")
        t_p5 = min(rem * 0.4, 100.0)
        a_p5, _ = freq_domain_refine(best_seq, t_p5, n_freq_iters=5000, verbose=True)
        update_best(a_p5, "freq_refine")

    # === Phase 6: Final LP polish ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 6: Final LP (score={best_score:.8f}, {rem:.0f}s) ===")
        t_p6 = min(rem * 0.5, 60.0)
        a_p6, _ = lp_alternating(best_seq, t_p6, max_iterations=100, verbose=True)
        update_best(a_p6, "final_lp")

    # === Phase 7: Final subgradient polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}, {rem:.0f}s) ===")
        t_p7 = min(rem * 0.6, 40.0)
        a_p7, _ = nesterov_subgradient(
            best_seq, t_p7,
            T_start=0.5, T_end=0.001,
            step_init=0.005, momentum_init=0.95, momentum_end=0.995,
            check_interval=10000
        )
        update_best(a_p7, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = score_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")
        elapsed_total = time.monotonic() - global_start
        print(f"Total time: {elapsed_total:.0f}s")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
