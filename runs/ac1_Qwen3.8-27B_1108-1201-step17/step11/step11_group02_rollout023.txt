    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def p_norm_obj_and_grad_z(z, p):

    a = softmax(z)
    Objective: ||a*a||_p^p
    """
    n = len(z)

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)

    b_pos = np.maximum(b, 0)
    Q = np.sum(b_pos ** p)
    
    # Gradient of Q wrt a: grad_a[i] = 2*p * sum_m b[m+i]^(p-1) * a[m]
    # This is 2*p * cross_correlation(b^(p-1), a)[i]
    w = b_pos ** (p - 1)

    # We need: corr[i] = sum_{m=0}^{n-1} a[m] * w[m+i] for i = 0, ..., n-1
    # This is the "full correlation" of a and w, at lags 0 to n-1
    
    # Method: zero-pad both to length >= n + len(w) - 1 = n + (2n-1) - 1 = 3n - 2
    fft_size = 1 << (3 * n - 1).bit_length()
