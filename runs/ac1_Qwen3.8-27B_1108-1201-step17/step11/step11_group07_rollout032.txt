import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def triple_convolve(a):
    """Compute conv(conv(a,a), a) via FFT. Length 3n-2."""
    n = len(a)
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    C = A * A  # This is FFT of conv(a,a)
    D = C * A  # This is FFT of conv(conv(a,a), a)
    return irfft(D, fft_size)[:size]

def evaluate_local(a):
    """Evaluate score locally (fast version)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

# --- Adam optimizer ---
class Adam:
    def __init__(self, params, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.params = [p.copy() for p in params]
        self.m = [np.zeros_like(p) for p in params]
        self.v = [np.zeros_like(p) for p in params]
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.t = 0
    
    def step(self, gradients):
        self.t += 1
        for i in range(len(self.params)):
            self.m[i] = self.beta1 * self.m[i] + (1 - self.beta1) * gradients[i]
            self.v[i] = self.beta2 * self.v[i] + (1 - self.beta2) * gradients[i]**2
            m_hat = self.m[i] / (1 - self.beta1**self.t)
            v_hat = self.v[i] / (1 - self.beta2**self.t)
            self.params[i] -= self.lr * m_hat / (np.sqrt(v_hat) + self.eps)
    
    def get_params(self):
        return [p.copy() for p in self.params]

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

def triple_convolve(a):
    """Compute conv(conv(a,a), a) via FFT. Result length 3n-2."""
    n = len(a)
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    D = A * A * A
    return irfft(D, fft_size)[:size]


def cross_corr(w, a):

    n = len(a)
    m_len = len(w)
    N = 1 << (m_len + n - 1).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    return np.fft.ifft(W * np.conj(A)).real[:n]

def normalize(a):

    s = a.sum()
    if s < 1e-15:

def softmax_fn(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)

def make_adam(n_params, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Create Adam optimizer state as a dict."""
    return {
        't': 0,
        'lr': lr,
        'beta1': beta1,
        'beta2': beta2,
        'eps': eps,
        'm': np.zeros(n_params),
        'v': np.zeros(n_params),
    }


def adam_step(state, grad):
    """Perform one Adam step, returns updated parameters offset."""
    state['t'] += 1
    t = state['t']
    state['m'] = state['beta1'] * state['m'] + (1 - state['beta1']) * grad
    state['v'] = state['beta2'] * state['v'] + (1 - state['beta2']) * grad * grad
    m_hat = state['m'] / (1 - state['beta1'] ** t)
    v_hat = state['v'] / (1 - state['beta2'] ** t)
    return state['lr'] * m_hat / (np.sqrt(v_hat) + state['eps'])


def spectral_flatness_adam(a0, time_limit, lr_init=0.05, n_iters=None):
    """Minimize sum(conv(a,a)^2) i.e. 4th-order spectral moment, using Adam in softmax space."""
    n = len(a0)
    a = normalize(a0.copy())
    z = np.log(a + 1e-300)
    best_a = a.copy()
    best_score = evaluate_local(a)
    state = make_adam(n, lr_init)
    t0 = time.monotonic()
    it = 0

    if n_iters is None:
        n_iters = 1000000

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a = softmax_fn(z)
        # Gradient of sum(c^2) w.r.t. a is 2*(a*a*a)[:n]
        tc = triple_convolve(a)
        grad_a = 2.0 * tc[:n]

        # Transform to z-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        # Adam step
        step = adam_step(state, grad_z)
        z = z - step

        # Periodic evaluation
        if it % 20 == 0:
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Adaptive LR decay
        if it % 2000 == 0 and it > 0:
            state['lr'] *= 0.8

    a = softmax_fn(z)
    sc = evaluate_local(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def lse_adam_optimize(a0, time_limit, T_stages=None, lr_init=0.02):
    """Minimize LSE-smoothed max using Adam in softmax space with temperature annealing."""
    n = len(a0)
    a = normalize(a0.copy())
    z = np.log(a + 1e-300)
    best_a = a.copy()
    best_score = evaluate_local(a)

    if T_stages is None:
        T_stages = [100, 70, 50, 35, 25, 18, 12, 8, 6, 4, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]

    t0 = time.monotonic()

    for stage_idx, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Allocate iterations proportionally
        n_stages_remaining = len(T_stages) - stage_idx
        iters = max(20, int(remaining * 25 / n_stages_remaining))

        # Learning rate: smaller for smaller T (sharper landscape)
        lr = lr_init * min(1.0, 5.0 / (stage_T + 0.1))
        state = make_adam(n, lr)

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            c = fast_convolve(a)
            c_max = c.max()

            # Softmax weights (numerically stable)
            w = np.exp((c - c_max) / stage_T)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros_like(c)
                w[np.argmax(c)] = 1.0
            else:
                w /= w_sum

            # Gradient: (2n/T) * cross_corr(w, a)
            R = cross_corr(w, a)
            grad_a = (2.0 * n / stage_T) * R

            # Transform to z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            step = adam_step(state, grad_z)
            z = z - step

            if it % 30 == 0:
                sc = evaluate_local(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        z = z.copy()
        a = softmax_fn(z)
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if stage_idx % 5 == 0:
            print(f"    LSE-Adam stage {stage_idx} (T={stage_T}): score={best_score:.8f}")

    return best_a, best_score


def hybrid_objective_adam(a0, time_limit, T=5.0, lam=0.1, lr_init=0.01, n_iters=50000):
    """Minimize a hybrid of LSE-max and spectral flatness."""
    n = len(a0)
    a = normalize(a0.copy())
    z = np.log(a + 1e-300)
    best_a = a.copy()
    best_score = evaluate_local(a)
    state = make_adam(n, lr_init)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a = softmax_fn(z)
        c = fast_convolve(a)
        c_max = c.max()

        # LSE part
        w = np.exp((c - c_max) / T)
        w_sum = w.sum()
        if w_sum > 1e-300:
            w /= w_sum
        else:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0

        R = cross_corr(w, a)
        grad_lse = (2.0 * n / T) * R

        # Spectral flatness part
        tc = triple_convolve(a)
        grad_flat = (lam / (2 * n - 1)) * 2.0 * tc[:n]

        grad_a = grad_lse + grad_flat

        # Transform to z-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        step = adam_step(state, grad_z)
        z = z - step

        if it % 30 == 0:
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if it % 5000 == 0 and it > 0:
            state['lr'] *= 0.85

    a = softmax_fn(z)
    sc = evaluate_local(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_iters=200000):
    """Subgradient method focusing on active set (peaks of convolution)."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = 0.01
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # Active set: positions within 5% of max
        threshold = c_max * 0.95
        active = np.where(c >= threshold)[0]

        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Subgradient: average of gradients at active peaks
        # grad[m] = sum_{k in active} (a[k-m] / len(active))
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je+1] += a[k - js:k - js + (je - js + 1)]
        g /= len(active)

        # Project onto simplex: a_new = a - eta * g, then projection
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            eta *= 0.5
            continue
        a_new /= s

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            # Slightly increase eta on improvement
            eta = min(eta * 1.05, 0.05)
        else:
            no_improve += 1
            if no_improve >= 10:
                eta *= 0.7
                no_improve = 0
                if eta < 1e-8:
                    break

    return best_a, best_score


def local_search(a0, time_limit, rng, n_trials=50000):
    """Local combinatorial search: small random perturbations, pairwise swaps."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    trial = 0

    while time.monotonic() - t0 < time_limit and trial < n_trials:
        trial += 1
        method = rng.integers(0, 4)

        if method == 0:
            # Small Gaussian perturbation
            strength = 0.01 + 0.02 * rng.random()
            a_new = a + rng.normal(0, strength, n)
        elif method == 1:
            # Pairwise transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            amt = rng.uniform(0.001, 0.02) * a.sum()
            a_new = a.copy()
            a_new[i] -= amt
            a_new[j] += amt
        elif method == 2:
            # Block perturbation
            blen = int(rng.integers(10, 50))
            start = int(rng.integers(0, n - blen))
            a_new = a.copy()
            a_new[start:start+blen] += rng.normal(0, 0.01, blen)
        else:
            # Multiply edge
            edge = rng.integers(0, 2)
            a_new = a.copy()
            w = rng.uniform(0.95, 1.05)
            if edge == 0:
                a_new[:20] *= w
            else:
                a_new[-20:] *= w

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new /= s

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)

    # Edge-concentrated (key insight: edges reduce middle convolution)
    a = np.exp(-8 * x) + np.exp(-8 * (1 - x))
    candidates.append(normalize(a))

    # Strong edge
    a = np.exp(-15 * x) + np.exp(-15 * (1 - x))
    candidates.append(normalize(a))

    # Ramp (asymmetric)
    a = np.linspace(0.1, 1.0, n)
    candidates.append(normalize(a))

    # Inverted ramp
    a = np.linspace(1.0, 0.1, n)
    candidates.append(normalize(a))

    # Bimodal
    a = np.exp(-20 * (x - 0.15) ** 2) + np.exp(-20 * (x - 0.85) ** 2)
    candidates.append(normalize(a))

    # Sine modulated
    a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
    a = np.maximum(a, 0.01)
    candidates.append(normalize(a))

    # Random smooth
    coeffs = rng.normal(0, 1, 8)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 0.5, 0.01)
    candidates.append(normalize(a))

    # Uniform
    candidates.append(np.ones(n) / n)

    # Quadratic edge
    a = x ** 2 + (1 - x) ** 2
    candidates.append(normalize(a))

    # Cosine window (Hann)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses spectral flatness (triple-convolution), LSE-Adam, hybrid objective,
    subgradient polish, and local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Spectral Flatness (triple-convolution + Adam) ===
    print(f"\n=== Phase 1: Spectral Flatness (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sf, _ = spectral_flatness_adam(best_seq, t_alloc, lr_init=0.05)
        local_update(a_sf, "spectral_flat")

    # === Phase 2: LSE-Adam (main optimization) ===
    print(f"\n=== Phase 2: LSE-Adam (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        T_stages = [100, 70, 50, 35, 25, 18, 12, 8, 6, 4, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
        print(f"  Budget: {t_alloc:.0f}s, {len(T_stages)} stages")
        a_lse, _ = lse_adam_optimize(best_seq, t_alloc, T_stages=T_stages, lr_init=0.02)
        local_update(a_lse, "lse_adam")

    # === Phase 3: Hybrid Objective ===
    print(f"\n=== Phase 3: Hybrid (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_hyb, _ = hybrid_objective_adam(best_seq, t_alloc, T=5.0, lam=0.05, lr_init=0.01,
                                         n_iters=int(t_alloc * 50))
        local_update(a_hyb, "hybrid")

    # === Phase 4: Fine LSE-Adam (second pass with lower temperatures) ===
    print(f"\n=== Phase 4: Fine LSE-Adam (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        T_fine = [3, 2, 1.5, 1, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]
        print(f"  Budget: {t_alloc:.0f}s")
        a_fine, _ = lse_adam_optimize(best_seq, t_alloc, T_stages=T_fine, lr_init=0.005)
        local_update(a_fine, "lse_fine")

    # === Phase 5: Subgradient Polish ===
    print(f"\n=== Phase 5: Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_sub, _ = subgradient_polish(best_seq, t_alloc, n_iters=int(t_alloc * 200))
        local_update(a_sub, "subgrad")

    # === Phase 6: Local Search ===
    print(f"\n=== Phase 6: Local Search (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_loc, _ = local_search(best_seq, t_alloc, rng, n_trials=int(t_alloc * 100))
        local_update(a_loc, "local")

    # === Phase 7: Spectral Flatness Round 2 (on best found so far) ===
    print(f"\n=== Phase 7: Spectral Flatness 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_sf2, _ = spectral_flatness_adam(best_seq, t_alloc, lr_init=0.02)
        local_update(a_sf2, "spectral_flat2")

    # === Phase 8: Multi-length exploration ===
    print(f"\n=== Phase 8: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best + 100, n_best - 100, n_best + 200, n_best - 200]:
            if n_try < 200 or n_try > 3000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 40:
                break
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.25, 80.0)
            T_short = [20, 10, 5, 3, 2, 1, 0.5, 0.3, 0.15, 0.1, 0.05, 0.02, 0.01]
            a_try, _ = lse_adam_optimize(init, t_sub, T_stages=T_short, lr_init=0.01)
            local_update(a_try, f"length_{n_try}")

    # === Phase 9: Final combined polish ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_sub = min(rem * 0.4, 60.0)
        T_final = [1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01]
        a_fin, _ = lse_adam_optimize(best_seq, t_sub, T_stages=T_final, lr_init=0.003)
        local_update(a_fin, "final_lse")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_fin2, _ = spectral_flatness_adam(best_seq, min(rem * 0.5, 30.0), lr_init=0.005)
        local_update(a_fin2, "final_flat")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            final_verified = fs
        else:
            final_verified = best_score
        # Also verify with the actual evaluator
        try:
            ev_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            if np.isfinite(ev_score):
                final_verified = ev_score
        except Exception:
            pass
        print(f"\nFINAL SCORE: {final_verified:.8f} (n={len(best_seq)})")

    if best_seq is None:
        # Fallback: return uniform
        best_seq = np.ones(1200) / 1200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
