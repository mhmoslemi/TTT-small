"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated gradient computation with LogSumExp smoothing
2. Adam optimizer with adaptive learning rate
3. Active-set-based peak targeting
4. Spectral domain flattening
5. Multi-scale random restarts
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autoconv(a):
    """Compute linear auto-convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr_fft(w, a, n_out):
    """Compute cross-correlation (w ⋆ a)[m] = sum_l w[l+m]*a[l] for m = 0..n_out-1."""
    n_w = len(w)
    n_a = len(a)
    fft_size = 1 << (n_w + n_a - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    result = irfft(W * np.conj(A), fft_size)
    return result[:n_out]

def lse_gradient(a, T):
    """Compute LSE-smoothed objective and gradient.
    Returns (obj, grad) where obj = 2n * LSE(conv(a,a), T).
    a should be non-negative, sum(a) = 1.
    """
    n = len(a)
    c = fast_autoconv(a)
    c_max = c.max()
    shifted = (c - c_max) / T
    exp_c = np.exp(shifted)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    
    w = exp_c / exp_c.sum()
    
    # Cross-correlation
    cc = cross_corr_fft(w, a, n)
    
    # Gradient
    g = 2.0 * cc
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]
    
    # Scale by 2*n
    g *= 2.0 * n
    
    return obj, g

def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autoconv(a)
    return float(2.0 * n * b.max() / (s * s))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def adam_optimize(a0, time_limit, T_schedule=None, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer for the LSE-smoothed objective with annealing."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    if T_schedule is None:
        T_schedule = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Use log-parameterization: a = softmax(z)
    z = np.log(a + 1e-300)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    for stage_idx, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Phase for this temperature
        phase_budget = min(120.0, (time_limit - (time.monotonic() - t0)) / max(1, len(T_schedule) - stage_idx))
        phase_end = time.monotonic() + phase_budget
        
        while time.monotonic() < phase_end:
            t += 1
            
            # Compute a from z
            z_shift = z - z.max()
            exp_z = np.exp(z_shift)
            a = exp_z / exp_z.sum()
            
            # Compute gradient
            obj, g = lse_gradient(a, T)
            
            # Transform to z-space
            S = np.dot(g, a)
            gz = a * (g - S)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * gz
            v = beta2 * v + (1 - beta2) * gz * gz
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Check score periodically
            if t % 50 == 0:
                a_check = np.exp(z - z.max())
                a_check = a_check / a_check.sum()
                sc = evaluate_local(a_check)
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
    
    return best_a, best_score

def spectral_flatten(a, n_iters=1000, lr=0.01):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > max(10, (n_iters // 10)):
            break
        
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P[1:].mean()  # Exclude DC
        
        # Target: reduce variance of P (excluding DC)
        # Move P toward P_mean
        # h = IFFT(-(P-1) * conj(A) / (P + eps)) in frequency domain
        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        H[0] = 0.0  # Don't change DC
        valid = P[1:] > 1e-12
        H[1:][valid] = -(P[1:][valid] - P_mean) * np.conj(A[1:][valid]) / (P[1:][valid] + 1e-10)
        
        h = irfft(H, fft_size)[:n]
        h -= h.mean()  # Preserve normalization
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm
        
        # Line search
        found = False
        for scale in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_new = a + scale * h
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                found = True
                break
        
        if not found:
            lr *= 0.5
            if lr < 1e-8:
                break
        
        if it % 200 == 0 and it > 0:
            lr *= 0.8
    
    return best_a, best_score

def peak_targeting(a, time_limit, n_peaks=3, step_size=0.01):
    """Directly target the highest peaks of the convolution."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)

                best_a = a_new.copy()
                a = a_new.copy()
                found = True

                break
        
        # Adaptive: reduce step size if stuck

    return best_a, best_score

def fast_autoconv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_obj_grad(a, T):
    n = len(a)
    c = fast_autoconv(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    w = exp_c / exp_c.sum()
    
    # Gradient via cross-correlation
    fft_size = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cc = irfft(W * np.conj(A), fft_size)[:n]
    
    g = 2.0 * cc
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]
    g *= 2.0 * n
    
    return obj, g
