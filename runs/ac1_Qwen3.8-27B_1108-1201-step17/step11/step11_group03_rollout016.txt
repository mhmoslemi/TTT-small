import numpy as np
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.GUROBI, cp.MOSEK, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_active_qcqp_iterative(a0, solver, max_rounds=20, max_active=50):
    """
    Iterative active-set QCQP:
    1. Start with top-m active constraints
    2. Solve QCQP
    3. Check all constraints for violations
    4. Add violated constraints
    5. Repeat
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = fast_convolve(a)
    max_b = b.max()

    # Initial active set: top constraints
    threshold = max_b * (1.0 - 1e-3)
    active = set(np.where(b >= threshold)[0].tolist())

    for round_idx in range(max_rounds):
        active_list = sorted(active)
        if len(active_list) > max_active:
            # Keep the most violated ones
            b_cur = fast_convolve(a)
            active_list = sorted(active_list, key=lambda k: -b_cur[k])[:max_active]
            active = set(active_list)

        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in active:
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        solved = False
        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
        for s_solver in solvers_to_try:
            if s_solver is None:
                continue
            try:
                prob.solve(solver=s_solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_val = np.asarray(a_var.value, dtype=np.float64)
                    a_val = np.maximum(a_val, 0)
                    s_sum = a_val.sum()
                    if s_sum < 1e-15:
                        return a, None
                    a_val = a_val / s_sum
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            return a, None

        a = a_val
        b = fast_convolve(a)
        max_b = b.max()

        # Check for violations
        tol = 1e-9
        violated = np.where(b > max_b + tol)[0]
        # Also add constraints that are close to the max
        near_max = np.where(b >= max_b * (1.0 - 1e-4))[0]

        new_violated = set(violated.tolist()) - active
        if not new_violated:
            # Check if any near-max constraint is not in active set
            new_near = set(near_max.tolist()) - active
            if not new_near:
                break
            active.update(new_near)
        else:
            active.update(new_violated)

    return a, float(max_b)


def solve_full_qcqp(a0, solver, timeout=300):
    """Try to solve the full QCQP with all 2n-1 constraints."""
    n = len(a0)
    num_constraints = 2 * n - 1

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in range(num_constraints):
        Q = build_q_sparse(k, n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    t0 = time.monotonic()
    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver]
    for s_solver in solvers_to_try:
        if s_solver is None:
            continue
        if time.monotonic() - t0 > timeout:
            break
        try:
            prob.solve(solver=s_solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum > 1e-15:
                    return a_val / s_sum, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        try:
            res = minimize(
                fun=lambda z, n=n, T=stage_T: lse_obj_grad(z, n, T)[0],
                x0=z,
                jac=lambda z, n=n, T=stage_T: lse_obj_grad(z, n, T)[1],
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def multiplicative_subgradient(a0, time_limit, n_steps):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.05
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Use top peaks with adaptive threshold
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 20:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.GUROBI, cp.MOSEK, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:

                b_val = b_val / s_sum

        except Exception:

    return None, None


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_active_qcqp_iterative(a0, solver, max_rounds=20, max_active=50):
    """
    Iterative active-set QCQP:
    1. Start with top-m active constraints
    2. Solve QCQP
    3. Check all constraints for violations
    4. Add violated constraints
    5. Repeat until no violations or max_rounds
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    max_b = b.max()

    # Initial active set: top constraints near the max
    threshold = max_b * (1.0 - 1e-3)
    active = set(np.where(b >= threshold)[0].tolist())

    for round_idx in range(max_rounds):
        active_list = sorted(active)
        if len(active_list) > max_active:
            b_cur = fast_convolve(a)
            active_list = sorted(active_list, key=lambda k: -b_cur[k])[:max_active]
            active = set(active_list)

        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in active:
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        solved = False
        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
        for s_solver in solvers_to_try:
            if s_solver is None:
                continue
            try:
                prob.solve(solver=s_solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_val = np.asarray(a_var.value, dtype=np.float64)
                    a_val = np.maximum(a_val, 0)
                    s_sum = a_val.sum()
                    if s_sum < 1e-15:
                        return a, None
                    a_val = a_val / s_sum
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            return a, None

        a = a_val
        b = fast_convolve(a)
        max_b = b.max()

        # Check for violated constraints (b[k] > t_val means violation)
        t_val = float(max(0, max_b))
        tol = 1e-8
        violated = np.where(b > t_val + tol)[0]

            active.update(new_violated)
        else:

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        # Nesterov acceleration
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise_amp = 0.02 * rng.uniform(0.5, 1.5)
                noise = noise_amp * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

        if len(lp_times) > 50 and np.mean(lp_times[-50:]) > 8.0:
            print(f"    LP too slow, stopping at iter {it+1}")
            break

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_func(z, n_param, T_param):
    """LSE objective for scipy minimize."""
    return lse_obj_grad(z, n_param, T_param)[0]


def lse_jac_func(z, n_param, T_param):
    """LSE gradient for scipy minimize."""
    return lse_obj_grad(z, n_param, T_param)[1]


def lse_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        try:
            res = minimize(
                fun=lse_obj_func, x0=z, jac=lse_jac_func,
                args=(n, stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def multiplicative_subgradient(a0, time_limit, n_steps):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.05
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 20:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def try_update_best(best_score, best_seq, a_new, label):
    """Try to update the best sequence. Returns (new_best_score, new_best_seq)."""
    if a_new is None:
        return best_score, best_seq
    a_new = np.maximum(np.asarray(a_new, dtype=np.float64), 0)
    s = a_new.sum()
    if s < 0.01:
        return best_score, best_seq
    a_new = a_new / s
    sc = evaluate_local(a_new)
    if np.isfinite(sc) and sc < best_score:
        print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a_new)})")
        return sc, a_new.copy()
    return best_score, best_seq


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating with Nesterov acceleration, iterative active-set QCQP,
    full QCQP attempt, low-temp L-BFGS, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    # === Phase 1: LP alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        best_score, best_seq = try_update_best(best_score, best_seq, a_lp, "lp_phase1")

    # === Phase 2: Iterative active-set QCQP ===
    print(f"\n=== Phase 2: Iterative Active QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_qcqp = min(rem * 0.15, 200.0)
    if t_qcqp > 15:
        q_start = time.monotonic()
        a_qcqp, t_val = solve_active_qcqp_iterative(best_seq, solver, max_rounds=20, max_active=40)
        best_score, best_seq = try_update_best(best_score, best_seq, a_qcqp, "qcqp_iter")

        # Try again from the improved solution
        if time.monotonic() - q_start < t_qcqp - 10:
            a_qcqp2, _ = solve_active_qcqp_iterative(best_seq, solver, max_rounds=15, max_active=25)
            best_score, best_seq = try_update_best(best_score, best_seq, a_qcqp2, "qcqp_iter2")

        print(f"  QCQP time: {time.monotonic() - q_start:.1f}s")

    # === Phase 3: Try full QCQP ===
    print(f"\n=== Phase 3: Full QCQP Attempt (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60 and n_best <= 1210:
        t_full = min(rem * 0.10, 120.0)
        print(f"  Attempting full QCQP with {2*n_best-1} constraints...")
        a_full, t_full_val = solve_full_qcqp(best_seq, solver, timeout=t_full)
        best_score, best_seq = try_update_best(best_score, best_seq, a_full, "full_qcqp")

    # === Phase 4: LP alternating round 2 ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=300000)
        best_score, best_seq = try_update_best(best_score, best_seq, a_lp2, "lp_phase4")

    # === Phase 5: Low-temp L-BFGS ===
    print(f"\n=== Phase 5: L-BFGS Low-Temp (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lbgfs = min(rem * 0.10, 120.0)
    if t_lbgfs > 20:
        T_stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_lb = lbgfs_refine(best_seq, t_lbgfs, T_stages, maxiter_per_stage=6000)
        best_score, best_seq = try_update_best(best_score, best_seq, a_lb, "lbgfs_lowtemp")

    # === Phase 6: Final QCQP polish ===
    print(f"\n=== Phase 6: Final QCQP Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_qcqp3, _ = solve_active_qcqp_iterative(best_seq, solver, max_rounds=10, max_active=20)
        best_score, best_seq = try_update_best(best_score, best_seq, a_qcqp3, "qcqp_final")

    # === Phase 7: Multi-n exploration ===
    for n_try in [1199, 1201, 1198, 1202, 1203]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        if n_try == len(best_seq):
            continue
        t_budget = min(rem * 0.08, 80.0)
        if t_budget < 20:
            continue
        print(f"\n=== Phase 7: Try n={n_try} (score={best_score:.10f}) ===")
        print(f"  Budget: {t_budget:.0f}s")
        init_n = interpolate_sequence(best_seq, n_try)
        a_n, _ = nesterov_lp_refine(init_n, t_budget * 0.6, solver, n_max_iters=80000)
        best_score, best_seq = try_update_best(best_score, best_seq, a_n, f"lp_n{n_try}")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == n_try and rem > 15:
            a_q, _ = solve_active_qcqp_iterative(best_seq, solver, max_rounds=8, max_active=20)
            best_score, best_seq = try_update_best(best_score, best_seq, a_q, f"qcqp_n{n_try}")

    # === Phase 8: Final multiplicative subgradient + QCQP ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_ms, _ = multiplicative_subgradient(best_seq, min(rem * 0.25, 20.0), n_steps=30000)
        best_score, best_seq = try_update_best(best_score, best_seq, a_ms, "final_mult")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_qf, _ = solve_active_qcqp_iterative(best_seq, solver, max_rounds=5, max_active=10)
            best_score, best_seq = try_update_best(best_score, best_seq, a_qf, "final_qcqp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
