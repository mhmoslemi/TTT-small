    Compute LSE approximation of 2n*max(conv(a,a)) and its gradient w.r.t. z.
    Uses FFT for O(n log n) gradient computation.
    a = softmax(z), and we compute f_T = (2n/T)*log(sum_k exp(T*(a*a)[k]))
    """
    a = softmax_fn(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    # grad_a[j] = 2 * (conv(w, a_flip))[n-1+j]
    a_flip = a[::-1]
    conv_w_af = fast_convolve_xy(w, a_flip)
    grad_a = 2.0 * conv_w_af[n - 1:2 * n - 1]

    # Gradient w.r.t. z (softmax parameterization)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lse_lbgfs_optimize(a0, time_limit, T_start=30.0, T_end=500.0, n_stages=20, iters_per_stage=50):
    """
    Multi-stage annealed L-BFGS on the LSE approximation.
    Very fast due to O(n log n) gradient via FFT.
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-300)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
    a0 = a0 / s
    z = np.log(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        def func_staged(z_val):
            obj, _ = lse_obj_grad_fft(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_obj_grad_fft(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_check = softmax_fn(z)
        sc = score_normalized(a_check)
        if sc < best_score:
            best_score = sc
            best_a = a_check.copy()

        if (si + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si+1}/{n_stages} T={stage_T:.1f} ({elapsed:.1f}s): score={sc:.8f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver is None:
        solvers_to_try = [cp.PDLP, cp.ECOS, cp.SCS]
    else:
        solvers_to_try = [solver, cp.PDLP, cp.ECOS, cp.SCS]

    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_val = b_val.sum()
    if s_val < 1e-15:
        return None, None
    b_val = b_val / s_val
    return b_val, float(M_var.value)
