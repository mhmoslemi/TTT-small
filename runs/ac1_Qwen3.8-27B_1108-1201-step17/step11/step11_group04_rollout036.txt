        deadline = t0 + time_limit

    for stage_T in T_stages:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = lse_logsumexp_obj_grad(z_in, n, stage_T)
            return g

        iters = max(20, min(maxiter_per_stage, int(remaining * 200)))

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def interpolate_sequence(f, new_n):
    """Interpolate to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def two_player_game(a0, b0, time_limit, n_mw_steps=80, eta=0.05, n_max_iters=500000):
    """
    Explicit two-player zero-sum game with Nesterov on both sides.
    Player A minimizes max_k conv(a,b)[k], Player B maximizes.
    The symmetric Nash equilibrium gives the optimal sequence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    sa = a.sum()
    if sa < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / sa

    b = np.maximum(np.asarray(b0, dtype=np.float64), 0)
    sb = b.sum()
    if sb < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / sb

    ya = a.copy()
    yb = b.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Player A's best response to yb (minimize max)
        alpha = 2.0 / (it + 2)
        a_best = fast_best_response(yb, n_mw_steps=n_mw_steps, eta=eta)

        # Player B's best response to ya (maximize max)
        # B wants to maximize max_k conv(a,b)[k]
        # This is equivalent to: B puts weight on the peak of conv(ya, .)
        c = convolve_ab(ya, b)
        max_idx = np.argmax(c)
        # Subgradient for B: row of T_a at max_idx
        # For multiplicative weights maximizing: b[j] *= exp(+eta * gradient[j])
        g_b = np.zeros(n)
        k = max_idx
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g_b[idx] = ya[k - idx]

        log_b = np.log(b + 1e-300) + eta * g_b
        log_b -= log_b.max()
        b_new = np.exp(log_b)
        b_new = np.maximum(b_new, 0)
        sb_new = b_new.sum()
        if sb_new < 1e-15:
            b_new = b.copy()
        else:
            b_new = b_new / sb_new

        # Nesterov updates
        ya = (1.0 - alpha) * ya + alpha * a_best
        ya = np.maximum(ya, 0)
        sya = ya.sum()
        if sya > 1e-15:
            ya = ya / sya

        yb = (1.0 - alpha) * yb + alpha * b_new
        yb = np.maximum(yb, 0)
        syb = yb.sum()
        if syb > 1e-15:
            yb = yb / syb

        a = ((it + 1) * a + alpha * (it + 2) * a_best) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa_new = a.sum()
        if sa_new > 1e-15:
            a = a / sa_new

        b = ((it + 1) * b + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        b = np.maximum(b, 0)
        sb2 = b.sum()
        if sb2 > 1e-15:
            b = b / sb2

        # Average of players as the sequence (by symmetry, this should converge to optimal)
        avg = 0.5 * (a + b)
        avg = np.maximum(avg, 0)
        savg = avg.sum()
        if savg > 1e-15:
            avg = avg / savg

        score_avg = evaluate_local(avg)
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg.copy()

        score_a = evaluate_local(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TwoPlayer iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Spectral flattening for good initialization
    Phase 2: Fast Nesterov game (multiplicative weights, no LP)
    Phase 3: Two-player game variant
    Phase 4: Spec-MSGD direct optimization
    Phase 5: L-BFGS LogSumExp polishing
    Phase 6: Multi-n exploration
    Phase 7: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        parent = spectral_flatten(np.ones(n_init) / n_init, n_iters=200, blend=0.3)
        best_seq = parent.copy()
        best_score = evaluate_local(parent)
        print(f"Generated init: n={len(parent)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"\nBudget: {rem:.0f}s, starting: {best_score:.8f}")

    # === Phase 1: Spectral flattening improvement (20s) ===
    print(f"\n=== Phase 1: Spectral Flattening ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.02, 20.0)
    if t1 > 5:
        a_sf = spectral_flatten(best_seq, n_iters=300, blend=0.25,
                                deadline=time.monotonic() + t1)
        local_update(a_sf, "spectral_v1")
        a_sf2 = spectral_flatten(best_seq, n_iters=200, blend=0.5,
                                 deadline=time.monotonic() + min(t1, 10))
        local_update(a_sf2, "spectral_v2")

    # === Phase 2: Nesterov game (main workhorse, 40% of time) ===
    print(f"=== Phase 2: Nesterov Game (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 450.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_g1, _ = nesterov_game(best_seq, t2, n_mw_steps=80, eta_mw=0.05,
                                n_max_iters=1000000, print_interval=3000)
        local_update(a_g1, "nesterov_game_1")

    # === Phase 3: Adaptive MW game (25% of time) ===
    print(f"=== Phase 3: Adaptive MW Game ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 300.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        a_g2, _ = game_with_adaptive_mw(best_seq, t3, n_max_iters=500000)
        local_update(a_g2, "adaptive_game")

    # === Phase 4: Spec-MSGD (10% of time) ===
    print(f"=== Phase 4: Spec-MSGD ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if t4 > 15:
        a_msgd, _ = spec_msgd(best_seq, t4, n_steps=300000, eta_base=0.04)
        local_update(a_msgd, "spec_msgd")

    # === Phase 5: L-BFGS LogSumExp polishing (20% of time) ===
    print(f"=== Phase 5: L-BFGS Polishing ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 250.0)
    if t5 > 20:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05,
                    0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        print(f"  Budget: {t5:.0f}s, {len(T_stages)} stages")
        a_lb = lbgfs_refine(best_seq, t5, T_stages, maxiter_per_stage=10000,
                            deadline=time.monotonic() + t5)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 6: Multi-n exploration ===
    print(f"=== Phase 6: Multi-n Exploration ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best + 200, n_best - 200, n_best + 500]:
            if n_try < 500:
                continue
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_sub = min(rem * 0.3, 80.0)
            if t_sub < 15:
                continue
            print(f"  Trying n={n_try}, budget={t_sub:.0f}s")
            init = interpolate_sequence(best_seq, n_try)
            init_sf = spectral_flatten(init, n_iters=50, blend=0.3)
            a_sub, _ = nesterov_game(init_sf, t_sub * 0.7, n_mw_steps=60,
                                     eta_mw=0.05, n_max_iters=200000,
                                     print_interval=5000)
            local_update(a_sub, f"game_n{n_try}")

    # === Phase 7: Final polish ===
    print(f"=== Phase 7: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.5, 60.0)
        T_fin = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
        a_fin = lbgfs_refine(best_seq, t_fin, T_fin, maxiter_per_stage=5000,
                             deadline=time.monotonic() + t_fin)
        local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        a_ps, _ = spec_msgd(best_seq, min(rem * 0.5, 30.0), n_steps=100000, eta_base=0.02)
        local_update(a_ps, "final_msgd")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_gf, _ = nesterov_game(best_seq, min(rem * 0.6, 30.0), n_mw_steps=100,
                                eta_mw=0.03, n_max_iters=100000, print_interval=10000)
        local_update(a_gf, "final_game")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified: {eval_score:.8f}")
        except Exception as e:
            print(f"Verify error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
