    if z.sum() < 1e-30:
        return np.ones(n) / n
    # Efficient Euclidean projection onto simplex
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last(u * np.arange(1, n + 1) > (cssv - 1), n - 1)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(z - theta, 0)
    s = proj.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return proj / s


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter=300):
    """
    L-BFGS refinement with LogSumExp smoothing and temperature annealing.
    Uses projection onto simplex.
    """
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [30.0, 20.0, 14.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.3, 0.15, 0.08]

    for stage_idx, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit - 3:
            break

        remaining_stages = len(T_stages) - stage_idx
        alloc = max(2, int((time_limit - (time.monotonic() - t0)) / remaining_stages))
        maxiter_cur = min(maxiter, max(20, alloc * 3))

        def func(z):
            a_t = project_simplex(z)
            obj, _ = lse_obj_grad(a_t, T)
            return obj

        def jac(z):
            a_t = project_simplex(z)
            _, g = lse_obj_grad(a_t, T)
            return g

        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter_cur, 'ftol': 1e-15,
                                    'gtol': 1e-12, 'maxfun': maxiter_cur * 200,
                                    'disp': False})
            a = project_simplex(res.x)
        except Exception:
            pass

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        elapsed = time.monotonic() - t0
        print(f"    LASF T={T:.3f} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def coord_descent_refine(a0, time_limit, n_max_steps=500000):
    """
    Peak-aware coordinate descent on the exact objective.
    Focuses on reducing contributions to the maximum peaks.
    """
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(99)

    for step in range(n_max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Find active peaks (within 0.1% of max)
        threshold = max_b * 0.998
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        improved = False
        # For each peak, try reducing top contributors
        for pk in peaks[:4]:
            if improved:
                break
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contribs = a[idx] * a[pk - idx]
            top_m = min(12, len(idx))
            top_pos = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_pos:
                if a[pos] < 1e-15:
                    continue
                # Try reducing this coordinate and redistributing to non-contributing coords
                for factor in [0.9995, 0.999, 0.998, 0.997, 0.995, 0.993, 0.99]:
                    a_trial = a.copy()
                    old_val = a_trial[pos]
                    a_trial[pos] = old_val * factor
                    diff = old_val - a_trial[pos]
                    if diff < 1e-18:
                        continue
                    
                    # Redistribute to coordinates that minimize impact on peaks
                    # Prefer coordinates far from peak contributions
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    # Score each coordinate by its contribution to active peaks
                    peak_contrib = np.zeros(n)
                    for pk2 in peaks[:3]:
                        js2 = max(0, pk2 - n + 1)
                        je2 = min(n - 1, pk2)
                        idx2 = np.arange(js2, je2 + 1)
                        peak_contrib[idx2] += a[idx2] * a[pk2 - idx2]
                    
                    # Redistribute to low-peak-contrib coordinates
                    low_mask = mask & (peak_contrib < np.median(peak_contrib[mask]) if mask.sum() > 0 else mask)
                    targets = np.where(low_mask)[0]
                    if len(targets) == 0:
                        targets = np.where(mask)[0]
                    
                    target_weights = a_trial[targets]
                    tw_sum = target_weights.sum()
                    if tw_sum < 1e-18:
                        # Redistribute uniformly
                        a_trial[targets] += diff / len(targets)
                    else:
                        a_trial[targets] += diff * target_weights / tw_sum
                    
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-18:
                        break
                    a_trial /= st
                    
                    score = compute_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall == 100:
                # Multi-coordinate simultaneous reduction
                pk = int(np.argmax(b))
                js = max(0, pk - n + 1)
                je = min(n - 1, pk)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[pk - idx]
                top_pos = idx[np.argsort(contribs)[::-1][:6]]
                a_trial = a.copy()
                for pos in top_pos:
                    if a_trial[pos] > 1e-15:
                        a_trial[pos] *= 0.998
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    stall = 101  # stays >= 100
            elif stall > 500:
                # Random perturbation
                n_perturb = max(3, n // 50)
                positions = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for pos in positions:
                    a_trial[pos] *= (1.0 + 0.02 * rng.standard_normal())
                a_trial = project_simplex(a_trial)
                score = compute_score(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    stall = 501
            if stall > 3000:
                break
        else:
            stall = 0

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def riesz_init(n, levels=5):
    """
    Initialize using a Riesz-product-inspired construction.
    The idea: the optimal sequence has a self-similar structure.
    We build it iteratively by "folding" operations.
    """
    # Start with a base pattern
    a = np.ones(n) / n
    
    # Apply iterative smoothing/shaping
    for level in range(levels):
        # Compute current auto-convolution
        c = fast_convolve(a)
        max_c = c.max()
        
        # Create a "target" that's more uniform
        target = np.ones(len(c)) / len(c)
        
        # Gradient-like step: reduce contributions to peaks
        peaks = np.where(c >= max_c * 0.95)[0]
        reduction = np.zeros(n)
        for pk in peaks:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            reduction[idx] += contribs
        
        # Update: reduce where reduction is high
        a_new = a * (1.0 - 0.1 * reduction / (reduction.max() + 1e-30))
        a_new = np.maximum(a_new, 1e-10)
        a = a_new / a_new.sum()
    
    return a


def structured_initializations(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine tapered
    a = np.cos(np.pi * x / 2) ** 2
    inits.append(a / a.sum())
    
    # Sine
    a = np.sin(np.pi * x) ** 2
    inits.append(a / a.sum())
    
    # Gaussian centered
    a = np.exp(-4 * (x - 0.5) ** 2 / 0.15)
    inits.append(a / a.sum())
    
    # Bimodal
    a = np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)
    inits.append(a / a.sum())
    
    # Riesz-inspired
    inits.append(riegsz_init(n, levels=8))
    
    # Quadratic
    a = 1.0 - (2 * x - 1) ** 2
    a = np.maximum(a, 0)
    if a.sum() > 0:
        inits.append(a / a.sum())
    
    # Random smooth (low-pass filtered noise)
    noise = rng.standard_normal(n)
    # Simple moving average as low-pass
    kernel_size = max(3, n // 100)
    kernel = np.ones(kernel_size) / kernel_size
    smoothed = np.convolve(noise, kernel, mode='same')
    a = np.exp(smoothed - smoothed.max())
    inits.append(a / a.sum())
    
    return inits


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving non-negativity and normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def gradient_variance_check(a, n_peaks=10):
    """
    Check optimality: at the optimum, the effective gradient
    (weighted by peak distribution) should be constant.
    Returns the variance as a measure of suboptimality.
    """
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    
    # Use top peaks with softmax weighting
    T = max_c * 0.01
    exp_c = np.exp(c / T - max_c / T)
    w = exp_c / exp_c.sum()
    
    # Effective gradient: grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # Compute via FFT
    fft_size = 1 << (3 * n - 3 + 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    conv_wa = irfft(W * A, fft_size)[:n]
    
    a_rev = a[::-1]
    AR = rfft(a_rev, fft_size)
    conv_war = irfft(W * AR, fft_size)
    term1 = conv_war[n-1:2*n-1]
    
    eff_grad = term1 + conv_wa
    
    # Variance of effective gradient
    return float(np.var(eff_grad))


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FFT-accelerated gradient methods with multi-resolution exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Load parent sequence ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = riesz_init(n_init, levels=10)
        best_score = compute_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_parent = len(best_seq)
    print(f"Starting score: {best_score:.10f}, n={n_parent}")

    # === Phase 1: Multi-n exploration with L-BFGS ===
    print(f"\n=== Phase 1: Multi-n L-BFGS Exploration ===")
    rem = deadline - time.monotonic()
    
    # Try several n values
    candidate_ns = sorted(set([
        max(400, n_parent - 300),
        max(500, n_parent - 100),
        n_parent,
        n_parent + 200,
        n_parent + 500,
        max(800, n_parent + 1000),
    ]))
    
    # Filter to reasonable range
    candidate_ns = [n for n in candidate_ns if 300 <= n <= 4000]
    
    t_per_n = max(15, min(60, rem * 0.35 / max(1, len(candidate_ns))))
    print(f"  n values: {candidate_ns}, ~{t_per_n:.0f}s each")
    
    results_by_n = {}
    for n_try in candidate_ns:
        if time.monotonic() > deadline - 60:
            break
        t_sub = min(t_per_n, deadline - time.monotonic() - 30)
        if t_sub < 10:
            continue
            
        a_interp = interpolate_sequence(best_seq, n_try)
        # Try a couple of initializations
        best_for_n = a_interp
        best_score_n = compute_score(a_interp)
        
        t_each = t_sub / 3
        inits = [a_interp, riesz_init(n_try, levels=6)]
        
        for init_fn in inits:
            if time.monotonic() > deadline - 30:
                break
            t_sub2 = min(t_each, deadline - time.monotonic() - 15)
            if t_sub2 < 8:
                continue
            T_stages = [20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.2]
            a_res, sc = lbgfs_refine(init_fn, t_sub2, T_stages=T_stages, maxiter=200)
            if sc < best_score_n:
                best_score_n = sc
                best_for_n = a_res
        
        results_by_n[n_try] = (best_for_n, best_score_n)
        try_update(best_for_n, f"phase1_n{n_try}")

    # === Phase 2: Deep L-BFGS refinement on best ===
    print(f"\n=== Phase 2: Deep L-BFGS Refinement (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages_deep = [8.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.3, 0.15, 0.08, 0.04]
        a_deep, sc_deep = lbgfs_refine(best_seq, t_alloc, T_stages=T_stages_deep, maxiter=500)
        try_update(a_deep, "deep_lbgfs")

    # === Phase 3: Coordinate Descent Polish ===
    print(f"\n=== Phase 3: Coordinate Descent Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = coord_descent_refine(best_seq, t_alloc, n_max_steps=1000000)
        try_update(a_cd, "coord_desc")

    # === Phase 4: Very tight L-BFGS with tiny T ===

    rem = deadline - time.monotonic()

    if t_alloc > 10:

        a_fin, sc_fin = coord_descent_refine(best_seq, t_alloc, n_max_steps=1000000)

    # === Phase 6: Try slightly different n from best ===

    rem = deadline - time.monotonic()
    if rem > 30:
        n_best = len(best_seq)
        for dn in [-50, 50, -100, 100]:
            if time.monotonic() > deadline - 10:
                break
            n_try = n_best + dn
            if n_try < 200:
                continue
            t_sub = min(20.0, deadline - time.monotonic() - 5)
            if t_sub < 5:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            T_stages_q = [2.0, 1.0, 0.5, 0.2, 0.1]
            a_q, sc_q = lbgfs_refine(a_interp, t_sub, T_stages=T_stages_q, maxiter=300)
            try_update(a_q, f"n{n_try}_quick")

    # === Final verification ===
    if best_seq is not None:
        fs = compute_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        # Gradient variance check
        gv = gradient_variance_check(best_seq)
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)}, grad_var={gv:.2e})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
