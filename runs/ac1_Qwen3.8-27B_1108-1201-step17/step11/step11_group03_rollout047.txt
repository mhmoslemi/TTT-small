import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_q_dense(k, n):
    """Build dense Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = np.zeros((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return Q

def solve_qp_active_set(a, active_set, solver, n):
    """
    Solve: min t s.t. a^T Q_k a <= t for k in active_set, sum(a)=1, a>=0
    """
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_set:
        Q = build_q_dense(k, n)
        constraints.append(cp.quad_form(a_var, cp.psd_wrap(Q)) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    for s in [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an active-set convex optimization method.
The problem is convex (minimax of PSD quadratics over the simplex), so the active-set
method converges to the global optimum. Key improvements over the LP-alternating approach:
1. Direct QP solves on the active set give exact (machine-precision) solutions.
2. Adaptive active-set management with proper convergence criteria.
3. Symmetry exploitation: only constrain the first half of the palindromic convolution.
4. Multi-n exploration to find the optimal sequence length.
5. Dense Q matrices for small active sets (faster than sparse for n~1200 with few constraints).
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_q_dense(k, n):
    """Build dense Q matrix (n x n) for conv(a,a)[k] = a^T Q a."""
    Q = np.zeros((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return Q


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def evaluate_local(a):
    """Evaluate score locally: 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_best_solver():
    """Find the best available QP/SOCP solver."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum_squares(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_active_qp(a_init, active_set, solver, n, max_iter_inner=50):
    """
    Solve the active-set QP:
        min t
        s.t. a^T Q_k a <= t for k in active_set
             sum(a) = 1
             a >= 0
    Returns (a_opt, t_opt) or (None, None) on failure.
    """
    if len(active_set) == 0:
        return a_init, 0.0

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    # Build quadratic form constraints
    for k in active_set:
        Q = build_q_dense(int(k), n)
        constraints.append(cp.quad_form(a_var, cp.psd_wrap(Q)) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                t_val = float(max(0, t_var.value))
                return a_val, t_val
        except Exception:
            continue

    return None, None


def active_set_optimize(a_init, solver, n, time_limit, tol=1e-11, max_active=80, verbose=True):
    """
    Active-set method to solve:
        min max_k (conv(a,a))[k]  s.t. sum(a)=1, a>=0
    Returns (a_opt, max_conv_value).
    """
    t_start = time.monotonic()
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Initial active set: top M positions in first half of convolution
    b = fast_convolve(a)
    b_half = b[:n]  # By symmetry, this covers all
    max_b = b_half.max()

    # Start with top 20 constraints
    n_init_active = min(20, n)
    top_idx = np.argsort(b_half)[-n_init_active:][::-1]
    active_list = list(set(int(x) for x in top_idx))

    if verbose:
        print(f"    Initial active set size: {len(active_list)}, max_b={max_b:.15f}")

    best_a = a.copy()
    best_max_b = max_b
    prev_max_b = max_b
    stall = 0

    for it in range(200):
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit:
            break

        # Solve the active-set QP
        a_new, t_qp = solve_active_qp(a, active_list, solver, n)
        if a_new is None:
            # Try with fewer constraints (drop the least active)
            if len(active_list) > 3:
                b_vals = b_half[active_list]
                min_k = active_list[np.argmin(b_vals)]
                active_list.remove(min_k)
                continue
            else:
                break

        a = a_new
        b = fast_convolve(a)
        b_half = b[:n]
        max_b = b_half.max()

        if max_b < best_max_b - 1e-15:
            best_max_b = max_b
            best_a = a.copy()

        # Check convergence: is the QP solution feasible for ALL constraints?
        if max_b <= t_qp + tol * max(1.0, max_b):
            if verbose:
                elapsed = time.monotonic() - t_start
                print(f"    CONVERGED at iter {it}: max_b={max_b:.15f}, t_qp={t_qp:.15f}, time={elapsed:.1f}s")
            return a, max_b

        # Find most violated constraint
        violations = b_half - t_qp
        k_violated = int(np.argmax(violations))

        if violations[k_violated] > tol * max(1.0, max_b):
            if k_violated not in active_list:
                active_list.append(k_violated)
                if verbose and it < 30:
                    print(f"    Iter {it}: added k={k_violated}, |active|={len(active_list)}, max_b={max_b:.12f}")

                # If active set is too large, prune
                if len(active_list) > max_active:
                    b_at_active = b_half[active_list]
                    # Keep the ones closest to t_qp (most "active")
                    # Remove the ones furthest below max
                    sorted_by_val = np.argsort(b_at_active)
                    # Remove the smallest values (least binding)
                    for idx in sorted_by_val:
                        if len(active_list) <= max_active:
                            break
                        k_to_remove = active_list[idx]
                        active_list.remove(k_to_remove)
                        active_list = list(active_list)
                        # Re-index
                        break
                    # More robust pruning
                    active_list = [int(x) for x in active_list]
                    if len(active_list) > max_active:
                        b_at_active = b_half[active_list]
                        keep = np.argsort(b_at_active)[-max_active:]
                        active_list = [active_list[i] for i in sorted(keep)]
        else:
            # No significantly violated constraint; we've converged
            if verbose:
                print(f"    CONVERGED (no violation) at iter {it}: max_b={max_b:.15f}, time={time.monotonic()-t_start:.1f}s")
            return a, max_b

        # Stall detection
        if abs(prev_max_b - max_b) < 1e-14:
            stall += 1
        else:
            stall = 0
        prev_max_b = max_b

        if stall > 20:
            if verbose:
                print(f"    Stalled at iter {it}, max_b={max_b:.15f}")
            break

    return best_a, best_max_b


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def cosine_init(n):
    """Cosine bump initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x / 2) ** 2
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def quick_mirror_descent(a, n_steps, n, eta=0.05):
    """Quick multiplicative weights / mirror descent for initialization."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    for step in range(n_steps):
        b = fast_convolve(a)
        b_half = b[:n]
        max_b = b_half.max()

        # Active set: top 10
        top_k = min(10, n)
        active = np.argsort(b_half)[-top_k:][::-1]

        # Gradient of max (approximate by averaging over active)
        g = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Mirror descent update
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new

    return a

        update_best(a_opt, f"active_n{n_parent}")

    # === Phase 3: Try nearby n values ===
    print(f"\n=== Phase 3: Nearby n values (best so far: {best_score:.12f}) ===")
    # Determine which n values to try
    n_candidates = []
    for dn in [-3, -2, -1, 1, 2, 3, -5, 5, -10, 10]:
        nn = n_parent + dn
        if 500 <= nn <= 3000 and nn not in n_candidates:
            n_candidates.append(nn)

    # Sort by |dn| to prioritize nearby values
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_for_this = min(rem * 0.12, 120.0)
        if t_for_this < 20:
            break

        print(f"\n  --- n={n_try} (budget {t_for_this:.0f}s) ---")
        # Interpolate
        a_interp = interpolate_sequence(best_seq, n_try)
        # Quick mirror descent
        a_interp = quick_mirror_descent(a_interp, 200, n_try, eta=0.02)
        sc_interp = evaluate_local(a_interp)
        print(f"    Interp+MD: score={sc_interp:.12f}")

        # Active-set optimization
        t_qp = max(10.0, t_for_this - 30.0)
        a_opt_n, _ = active_set_optimize(
            a_interp, solver, n_try, t_qp,
            tol=1e-11, max_active=50, verbose=False
        )
        update_best(a_opt_n, f"active_n{n_try}")

    # === Phase 4: Final polish on the best ===
    print(f"\n=== Phase 4: Final Polish (best: {best_score:.12f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        n_best = len(best_seq)
        t_polish = min(rem * 0.5, 200.0)
        print(f"  Polish budget: {t_polish:.0f}s")
        # Run active-set again with tighter tolerance
        a_final, _ = active_set_optimize(
            best_seq, solver, n_best, t_polish,
            tol=1e-13, max_active=80, verbose=True
        )
        update_best(a_final, "final_polish")

    # === Phase 5: Try a small number of random perturbations + re-optimize ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 5: Perturbation search (best: {best_score:.12f}) ===")
        n_best = len(best_seq)
        t_remaining = rem - 20
        n_trials = max(1, int(t_remaining // 40))
        for trial in range(n_trials):
            rem = deadline - time.monotonic()
            if rem < 35:
                break

            # Perturb
            noise_scale = 0.05
            a_pert = best_seq + noise_scale * rng.normal(0, 1, n_best)
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert = a_pert / sp

            # Quick optimize
            t_trial = min(rem - 20, 35.0)
            a_res, _ = active_set_optimize(
                a_pert, solver, n_best, t_trial,
                tol=1e-11, max_active=40, verbose=False
            )
            update_best(a_res, f"perturb_{trial}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*60}")
        print(f"FINAL SCORE: {best_score:.12f} (n={len(best_seq)})")
        print(f"Total time: {time.monotonic() - t_global_start:.1f}s")
        print(f"{'='*60}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
