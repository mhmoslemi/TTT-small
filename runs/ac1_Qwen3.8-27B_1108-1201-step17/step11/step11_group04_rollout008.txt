def lp_direction(a, active_lags, solver):
    """Solve LP to find direction that reduces max autocorrelation."""
    n = len(a)
    b = fast_convolve(a)
    M_current = b.max()
    
    # Build constraint matrix: for each active lag k, grad_k · delta <= M_target - b[k]
    # We want: b[k] + grad_k · delta <= M for all active k
    # And: sum(delta) = 0 (maintain normalization)
    # And: a + delta >= 0
    # Minimize: M
    # Actually, let's fix M = M_current * (1 - eps) and check feasibility
    # Or minimize M directly.
    
    delta = cp.Variable(n)
    M = cp.Variable()
    
    constraints = [cp.sum(delta) == 0]
    for k in active_lags:
        # grad_k[i] = a[i+k] + a[i-k] (with zero padding)
        grad_k = np.zeros(n)
        for i in range(n):
            if i + k < n:
                grad_k[i] += a[i + k]
            if i - k >= 0:
                grad_k[i] += a[i - k]
        constraints.append(grad_k @ delta <= M - b[k])
    
    # Non-negativity: a + delta >= 0 => delta >= -a
    constraints.append(delta >= -a)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=solver)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    return np.asarray(delta.value), float(M.value)

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.CVXOPT, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, timeout=30.0):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = conv_matrix_sparse(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.CVXOPT, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    t0 = time.monotonic()
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            if time.monotonic() - t0 > timeout:
                break
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    continue
                b_val = b_val / s_sum
                return b_val
        except Exception:
            continue
    return None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj(z, n, T):
    """LogSumExp objective value."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    return 2.0 * n * lse


def lse_grad(z, n, T):
    """LogSumExp gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    w = exp_terms / exp_terms.sum()

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            return lse_obj(z_in, n, stage_T)

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            return lse_grad(z_in, n, stage_T)

        stages_left = len(T_stages) - stage_i
        iters = max(30, min(maxiter_per_stage, int((deadline - time.monotonic()) * 30 / max(1, stages_left))))

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:

def spectral_flatten(a, n_iters=50, blend=0.3, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a

def peak_shave_momentum(a0, time_limit, n_steps=200000):
    """Peak shaving with momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.9
    eta = 0.03
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        momentum = beta * momentum + (1 - beta) * g
        eta_eff = eta * scale
        log_scale = -eta_eff * momentum
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.75
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def single_point_lp_refine(a0, time_limit, solver, n_steps=5000):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 0.005)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        active = active[:50]  # Limit to 50 constraints for speed

        delta = cp.Variable(n)
        M = cp.Variable()

        constraints = [cp.sum(delta) == 0, delta >= -a * 0.99]
        for k in active:
            grad_k = np.zeros(n)
            for i in range(n):
                if i + k < n:
                    grad_k[i] += a[i + k]
                if i - k >= 0:
                    grad_k[i] += a[i - k]
            constraints.append(grad_k @ delta <= M - b[k])

        prob = cp.Problem(cp.Minimize(M), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue
            d = np.asarray(delta.value, dtype=np.float64)
        except Exception:
            continue

        # Backtracking line search
        for lr in [0.5, 0.3, 0.1, 0.05, 0.02, 0.01]:
            a_try = a + lr * d
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 0.01:
                continue
            a_try = a_try / st
            sc = 2.0 * n * fast_convolve(a_try).max()
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()
                a = a_try
                break

        if (step + 1) % 500 == 0:
            print(f"    SLP step={step+1}: score={best_score:.8f}")

    return best_a, best_score


def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
