Wait, this gradient computation is O(n * num_active_lags) which could be slow. Let me use FFT instead.

Actually, for the subgradient, I can use a simpler approach. The gradient of `conv(a,a)[k]` w.r.t. `a` is `2 * flip_shift(a, k)` where `flip_shift(a, k)[m] = a[k-m]` if `0 <= k-m < n` else `0`.

This is just `a` reversed and shifted. I can compute this efficiently.

But for multiple active lags, I need a weighted sum. Let me use the FFT approach:
- The gradient w.r.t. `a` of `sum_k w_k * conv(a,a)[k]` can be computed as a cross-correlation.

Actually, let me just use a different approach. Instead of computing exact gradients, I'll use a **direct optimization** approach:

**Algorithm: Adaptive Peak Reduction**

1. Compute `b = conv(a,a)`
2. Find the top `K` peaks (largest values)
3. For each peak at lag `k*`:
   - The contributing pairs are `(j, k*-j)` for valid `j`
   - Find the pair with the largest product `a[j]*a[k*-j]`
   - Move a small amount of mass from the larger of `a[j]`, `a[k*-j]` to a position that doesn't contribute to any top peak
4. Use binary search to find the optimal amount to move
5. Repeat

This is similar to the previous peak shaving but with:
- Multiple peaks considered simultaneously
- Binary search for optimal step size
- Smarter selection of destination positions

Let me also think about a **momentum-based approach**:
- Keep a running record of the direction of recent improvements
- Bias future moves in that direction
- This helps escape shallow local minima

And a **temperature-based acceptance**:
- Occasionally accept moves that slightly worsen the score (simulated annealing)
- This helps escape local minima

Let me now write the complete solution:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

4. Multi-length exploration
Key: sophisticated local search with multi-peak awareness and momentum.
"""

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg >= 10 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def peak_reduction_search(a0, time_limit, seed=42):
    """
    Sophisticated peak reduction with line search and momentum.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    momentum_dir = np.zeros(n)
    momentum_alpha = 0.1
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b
        
        # Find top peaks
        sorted_idx = np.argsort(b)[::-1]
        top_peaks = sorted_idx[:min(8, len(sorted_idx))]
        
        improved = False
        
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the highest contributor
            src_local = np.argmax(contrib)
            src = idx[src_local]
            partner = k - src
            
            # Determine which of src, partner to reduce
            if a[src] >= a[partner]:
                reduce_idx = src
            else:
                reduce_idx = partner
            
            # Find good destination: not contributing to any top peak
            peak_areas = set()
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            
            candidates = [j for j in range(n) if j not in peak_areas and j != reduce_idx]
            if not candidates:
                candidates = [j for j in range(n) if j != reduce_idx]
            
            # Pick a few best candidates (low value, far from peaks)
            candidates_can = np.array(candidates)
            cand_values = a[candidates_can]
            # Sort by value (lower is better for destination)
            order = np.argsort(cand_values)
            top_candidates = candidates_can[order[:max(1, len(order) // 4)]]
            
            dst = top_candidates[rng.integers(len(top_candidates))]
            
            # Line search: try different fractions
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[reduce_idx] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[reduce_idx] -= amt
                a_trial[dst] += amt
                if a_trial[reduce_idx] < 0:
                    continue
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    # Update momentum
                    move = np.zeros(n)
                    move[reduce_idx] = -amt
                    move[dst] = amt
                    momentum_dir = momentum_alpha * momentum_dir + (1 - momentum_alpha) * move
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        if improved:
            continue
        
        # Try momentum-guided move
        if np.abs(momentum_dir).sum() > 1e-15:
            # Normalize momentum to have sum zero
            mom = momentum_dir - momentum_dir.mean()
            # Scale to a small perturbation
            scale = 0.01 / (np.abs(mom).max() + 1e-15)
            perturbation = mom * scale
            a_trial = a + perturbation
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial / s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    continue
        
        # Random exploration
        stall += 1
        for attempt in range(3):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or a[i] < 1e-12:
                continue
            frac = rng.uniform(0.002, 0.05)
            amt = a[i] * frac
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            if a_trial[i] < 0:
                continue
            s = a_trial.sum()
            a_trial = a_trial / s
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
                break
        # Temperature-based acceptance (escape local minima)
        if stall > 50 and rng.random() < 0.1:
            # Make a larger random perturbation
            perturbation = rng.normal(0, 0.01, n)
            perturbation -= perturbation.mean()
            a_trial = a + perturbation
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial = a_trial / s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score * 1.01:  # Allow slight worsening
                    a = a_trial
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
        
        if stall > 500:
            break
    
    return best_a, best_score


def lse_gradient_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=15):
    """LogSumExp-smoothed optimization with L-BFGS."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        iters = max(20, min(60, int(30 + 30 * (si / n_stages))))
        
        def func(z_in):
            a_s = softmax_fn(z_in)
            b = fast_autocorr(a_s)
            Tb = stage_T * b
            max_Tb = Tb.max()
            return (max_Tb + np.log(np.exp(Tb - max_Tb).sum())) / stage_T
        
        def jac(z_in):
            a_s = softmax_fn(z_in)
            b = fast_autocorr(a_s)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            # Gradient w.r.t. a: 2 * cross_corr(a, w)
            conv_len = 2 * n - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            A_rev = rfft(a_s[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]
            
            S = np.dot(grad_a, a_s)
            grad_z = a_s * (grad_a - S)
            return grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass
    
    return best_a, best_score


def interpolate_to(new_n, a):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
