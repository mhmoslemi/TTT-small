def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lse_obj_and_grad_a(a, n, T_temp):
    """Compute LogSumExp objective value and gradient w.r.t. a (not z)."""
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    
    return 2.0 * n * lse, 2.0 * n * grad_a


def lse_lbgfs(z, n, T_temp, deadline):

        return 1e15

    S = np.dot(grad_a, a)
    # grad w.r.t. z = a * (grad_a - S) but we just return obj for L-BFGS
    return obj

def nesterov_lp(a0, time_limit, solver, n_max_iters=1000000):

            y = y / sy

def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)
    
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break
        
        stages_left = n_stages - stage_i
        iters = max(10, min(iters_per_stage * 2, int(remaining * 30 / max(1, stages_left))))
        
        def func(z_in):
            return lse_lbgfs(z_in, n, stage_T, deadline)
        
        def jac(z_in):
            return lse_lbgfs_jac(z_in, n, stage_T, deadline)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def multiplicative_weights(a0, time_limit, n_steps=100000, eta_base=0.03):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    scale = 1.0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Use top K peaks
        K = min(n, max(1, int(0.01 * (2*n-1))))
        top_k = np.argsort(b)[-K:]
        
        g = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= K
        
        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 40:
            scale *= 0.75
            no_improve = 0
            if scale < 0.003:
                break
    
    return best_a, best_score


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        
        mags = np.abs(F)
        non_dc = mags[1:L//2+1]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):

    x = np.linspace(0, 2 * np.pi, n)

    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

    # Better: use the top K based on the current a's self-correlation
    c_self = fast_convolve(a_fixed)

    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(max(0, M_var.value))
    return b_val, M_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Cutting-plane LP alternating with Nesterov acceleration (faster than full LP)
- Multi-temperature L-BFGS with LogSumExp smoothing
- Multiplicative weights (mirror descent) for active peak reduction
- Multi-restart final polish with diverse perturbations
- Variable-length exploration (n = 1000, 1200, 1500)
- Adaptive time allocation

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CBC, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_cutting_plane(a_fixed, K_max, solver, max_cp_iters=15):
    """Solve LP using cutting plane: start with top constraints, add violated ones."""
    n = len(a_fixed)
    size = 2 * n - 1
    T_mat = build_conv_matrix(a_fixed)

    c_self = fast_convolve(a_fixed)
    init_K = min(K_max, size)
    active_set = set(np.argsort(c_self)[-init_K:].tolist())

    b_val = np.ones(n) / n
    M_val = float('inf')

    for _ in range(max_cp_iters):
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(b_var) == 1]
        for k in sorted(active_set):
            constraints.append(T_mat[k] @ b_var <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solved = False
        for s in [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if s is None:
                continue
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        b_new = np.asarray(b_var.value, dtype=np.float64)
        b_new = np.maximum(b_new, 0)
        s = b_new.sum()
        if s < 1e-15:
            return None, None
        b_new = b_new / s
        M_new = float(max(0, M_var.value))

        T_b = T_mat @ b_new
        tol = 1e-7
        violations = np.where(T_b > M_new + tol)[0]

        if len(violations) == 0:
            return b_new, M_new

        active_set.update(violations.tolist())
        if len(active_set) > K_max:
            active_sorted = sorted(active_set, key=lambda k: T_b[k], reverse=True)
            active_set = set(active_sorted[:K_max])

        b_val = b_new
        M_val = M_new

    return b_val, M_val


def solve_lp_full(a_fixed, solver):
    """Solve full LP best response."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solved = False
    for s in [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GUROBI, cp.MOSEK, cp.CBC]:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_and_grad_a(a, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. a."""
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]

    return 2.0 * n * lse, 2.0 * n * grad_a


def lse_lbgfs_func(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return 1e15
    a = softmax_from_z(z)
    obj, _ = lse_obj_and_grad_a(a, n, T_temp)
    return obj


def lse_lbgfs_jac_func(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    a = softmax_from_z(z)
    _, grad_a = lse_obj_and_grad_a(a, n, T_temp)
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def nesterov_lp_optimize(a0, time_limit, solver, K_max=300):
    """Nesterov dual averaging LP alternating with cutting plane."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_cutting_plane(y, K_max, solver, max_cp_iters=12)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 12:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 9)
                noise = 0.015 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

        stages_left = n_stages - stage_i
        iters = max(10, min(iters_per_stage * 2, int(remaining * 40 / max(1, stages_left))))

        def func_wrapper(z_in):
            return lse_lbgfs_func(z_in, n, stage_T, deadline)

        def jac_wrapper(z_in):
            return lse_lbgfs_jac_func(z_in, n, stage_T, deadline)

        try:
            res = minimize(
                fun=func_wrapper, x0=z, jac=jac_wrapper, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-22, 'gtol': 1e-16,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def multiplicative_weights_optimize(a0, time_limit, n_steps=100000, eta_base=0.025, K_frac=0.01):
    """Mirror descent targeting top-K active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    no_improve = 0
    scale = 1.0
    K = max(1, int(K_frac * (2 * n - 1)))

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        top_k = np.argsort(b)[-K:]

        g = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= K

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.75
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def spectral_flatten_opt(a, n_iters=100, blend=0.4, deadline=None):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:L // 2 + 1]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def evaluate_local(a):
    """Quick local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 12)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def perturb_random(a, rng, strength=0.02):
    """Add random Gaussian perturbation."""
    a_pert = a + strength * rng.standard_normal(len(a))
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

            print(f"  NEW BEST [mult_weights]: {sc_mw_check:.10f}")

    # === Phase 4: Spectral flattening ===
    print(f"=== Phase 4: Spectral Flattening (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        for blend in [0.3, 0.5, 0.7]:
            a_sf = spectral_flatten_opt(best_seq, n_iters=150, blend=blend,
                                        deadline=time.monotonic() + min(rem * 0.3, 30))
            sc_sf = evaluate_local(a_sf)
            if sc_sf < best_score:
                best_score = sc_sf
                best_seq = a_sf.copy()
                print(f"  NEW BEST [spectral blend={blend}]: {sc_sf:.10f}")

    # === Phase 5: Second Nesterov LP round ===
    print(f"=== Phase 5: Nesterov LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.15, 180.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_optimize(best_seq, t_lp2, solver, K_max=200)
        sc_lp2 = evaluate_local(a_lp2)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_seq = a_lp2.copy()
            print(f"  NEW BEST [nesterov_lp2]: {sc_lp2:.10f}")

    # === Phase 6: Multi-restart final polish ===
    print(f"=== Phase 6: Multi-restart Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_restarts = min(10, max(3, int(rem / 25)))
        print(f"  {n_restarts} restarts")
        for r in range(n_restarts):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break

            # Diverse perturbations
            if r % 3 == 0:
                a_pert = perturbed_sequence(best_seq, rng, strength=0.008 + 0.005 * r)
            elif r % 3 == 1:
                a_pert = perturb_random(best_seq, rng, strength=0.01 + 0.003 * r)
            else:
                a_pert = perturbed_sequence(best_seq, rng, strength=0.015)

            t_sub = min(rem_sub * 0.12, 40.0)
            if t_sub > 8:
                # Quick Nesterov LP
                t_sub_lp = t_sub * 0.5
                a_sub, _ = nesterov_lp_optimize(a_pert, t_sub_lp, solver, K_max=150)
                sc_sub = evaluate_local(a_sub)
                if sc_sub < best_score:
                    best_score = sc_sub
                    best_seq = a_sub.copy()
                    print(f"  NEW BEST [restart{r}_lp]: {sc_sub:.10f}")

                rem_sub = deadline - time.monotonic()
                if rem_sub > 8:
                    # Quick fine L-BFGS on current best
                    T_sub = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
                    a_sub2 = lbgfs_refine(best_seq, min(rem_sub * 0.4, 15.0), T_sub,
                                          maxiter_per_stage=3000,
                                          deadline=time.monotonic() + min(rem_sub * 0.4, 15.0))
                    sc_sub2 = evaluate_local(a_sub2)
                    if sc_sub2 < best_score:
                        best_score = sc_sub2
                        best_seq = a_sub2.copy()
                        print(f"  NEW BEST [restart{r}_lbgfs]: {sc_sub2:.10f}")

    # === Phase 7: Try different n ===
    print(f"=== Phase 7: Variable Length (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    for n_try in [1000, 1400, 1600, 1800]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        if n_try == len(best_seq):
            continue

        t_n = min(rem * 0.18, 100.0)
        if t_n < 25:
            continue

        print(f"  Trying n={n_try}, budget={t_n:.0f}s")
        init_n = interpolate_sequence(best_seq, n_try)
        init_n = spectral_flatten_opt(init_n, n_iters=30, blend=0.3)

        t_lp_n = t_n * 0.6
        a_n, _ = nesterov_lp_optimize(init_n, t_lp_n, solver, K_max=150)
        sc_n = evaluate_local(a_n)
        if sc_n < best_score:
            best_score = sc_n
            best_seq = a_n.copy()
            print(f"  NEW BEST [lp_n{n_try}]: {sc_n:.10f}")

        rem = deadline - time.monotonic()
        if rem > 15 and len(best_seq) == n_try:
            T_n = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_n2 = lbgfs_refine(best_seq, min(rem * 0.5, 30.0), T_n,
                                maxiter_per_stage=3000,
                                deadline=time.monotonic() + min(rem * 0.5, 30.0))
            sc_n2 = evaluate_local(a_n2)
            if sc_n2 < best_score:
                best_score = sc_n2
                best_seq = a_n2.copy()
                print(f"  NEW BEST [lbgfs_n{n_try}]: {sc_n2:.10f}")

    # === Phase 8: Final aggressive polish ===
    print(f"=== Phase 8: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        T_fin = [1.0, 0.5, 0.3, 0.2, 0.1, 0.06, 0.04, 0.03, 0.02, 0.015,
                 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        a_fin = lbgfs_refine(best_seq, min(rem * 0.35, 50.0), T_fin,
                             maxiter_per_stage=4000,
                             deadline=time.monotonic() + min(rem * 0.35, 50.0))
        sc_fin = evaluate_local(a_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"  NEW BEST [final_lbgfs]: {sc_fin:.10f}")

    rem = deadline - time.monotonic()
    if rem > 12:
        a_fin2, _ = multiplicative_weights_optimize(best_seq, min(rem * 0.4, 20.0),
                                                     n_steps=50000, eta_base=0.008, K_frac=0.003)
        sc_fin2 = evaluate_local(a_fin2)
        if sc_fin2 < best_score:
            best_score = sc_fin2
            best_seq = a_fin2.copy()
            print(f"  NEW BEST [final_mw]: {sc_fin2:.10f}")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin3, _ = nesterov_lp_optimize(best_seq, min(rem * 0.5, 15.0), solver, K_max=200)
        sc_fin3 = evaluate_local(a_fin3)
        if sc_fin3 < best_score:
            best_score = sc_fin3
            best_seq = a_fin3.copy()
            print(f"  NEW BEST [final_lp]: {sc_fin3:.10f}")

    # Final output
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)}) at {elapsed:.1f}s")

        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.10f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
