    old_n = len(f)
    if old_n == new_n:
        return f.copy()

    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)

def build_constraint_matrix(y, lags, n):
    K = len(lags)
    # Each row k: (y*g)[k] = sum_j y[max(0,k-n+1):min(n,k+1)] * g[...]
    # For lag k, valid j range: max(0, k-n+1) to min(n-1, k)
    # (y*g)[k] = sum_{j} y[j] * g[k-j] where 0<=j<n and 0<=k-j<n
    #            = sum_{j} g[k-j] * y[j]
    # Let i = k-j, so j = k-i: sum_i g[i] * y[k-i]
    # Valid i: 0<=i<n and 0<=k-i<n → max(0,k-n+1) <= i <= min(n-1,k)
    
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

np.convolve(a, b)[k] = sum_j a[j] * b[k-j]

For np.convolve(y, g)[k] = sum_j y[j] * g[k-j] where 0<=j<len(y) and 0<=k-j<len(g).

If y and g both have length n:
- j ranges from max(0, k-n+1) to min(n-1, k)
- (y*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} y[j] * g[k-j]

So for the LP constraint (y*g)[k] <= M:
- The coefficient of g[i] (where i = k-j, so j = k-i) is y[k-i]
- i ranges from max(0, k-n+1) to min(n-1, k)

So coeffs[k_row, i] = y[k - i] for i in [max(0,k-n+1), min(n-1,k)]
