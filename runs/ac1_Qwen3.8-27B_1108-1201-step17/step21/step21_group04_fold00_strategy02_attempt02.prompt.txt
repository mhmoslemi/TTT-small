import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def subgradient_fw(a0, time_limit, n_steps=500, verbose=True):
    """Subgradient Frank-Wolfe: use the peak lag to define a linearized direction."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    decay = 2.0 / (n_steps + 2.0)

    for t in range(1, n_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        # Find top few lags
        top_lags = np.argsort(ac)[::-1][:5]
        # Subgradient: average of the "active" gradient directions
        grad = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx = np.arange(i_s, i_e)
            grad[idx] += y[ki - idx]
        grad /= len(top_lags)

        # FW step: minimize grad · g over simplex => g = delta at argmin grad
        g_dir = np.zeros(n)
        g_dir[int(np.argmin(grad))] = 1.0

        alpha = 2.0 / (t + 2.0)
        y_new = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy

    return best_a, 2.0 * n * best_m

def peak_equalize(a0, time_limit, n_targets=15, verbose=True):
    """Direct peak deflation: move mass from high-contributing sources to low-impact destinations."""
    n = len(a0)

        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False

        for k in targets:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            top_srcs = idx[np.argsort(contrib)[::-1][:3]]
            for src in top_srcs:
                src = int(src)
                if a[src] < 1e-12:
                    continue

                # Score destinations by impact on top peaks
                scores = np.zeros(n)
                for k2 in targets[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10

                candidates = np.argsort(scores)[:25]
                found = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.001, 0.003, 0.007, 0.015, 0.03]:
                        amt = a[src] * frac

                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:

                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-14:

                            best_m = Mt
                            best_a = a.copy()

                    break

            stall += 1
            if stall > 100:
                # Random perturbation
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-10:

def diverse_init(n, parent_seq):
    """Generate diverse initial candidates."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Linear ramps (exploit asymmetry)
    for p in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
        candidates.append(normalize((i_arr + 1.0) ** p))
        candidates.append(normalize((n - i_arr) ** p))

    # Asymmetric Gaussians
    for c in [0.3, 0.4, 0.5, 0.6]:
        for sg in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))

    # Skewed shapes
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        shape = x ** alpha * (1 - x)
        shape = np.maximum(shape, 1e-10)
        candidates.append(normalize(shape))
        shape2 = x * (1 - x) ** alpha
        shape2 = np.maximum(shape2, 1e-10)
        candidates.append(normalize(shape2))

    # Sine-based
    for f in [1, 2, 3]:
        candidates.append(normalize(np.sin(np.pi * x * f) ** 2))

    # Blend with parent
    if parent_seq is not None:
        bi = normalize(parent_seq)
        for w in [0.3, 0.5, 0.7, 0.9]:
            candidates.append(normalize(w * bi + (1 - w) * candidates[0]))
            candidates.append(normalize(w * bi + (1 - w) * candidates[len(candidates) // 2]))

    best_a = normalize(parent_seq) if parent_seq is not None else normalize(np.sin(np.pi * x) ** 2)
    best_val = obj_val(best_a)

    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()

    return best_a, best_val
