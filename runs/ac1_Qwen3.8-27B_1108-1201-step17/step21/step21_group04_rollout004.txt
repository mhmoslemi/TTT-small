def fast_conv(a):
    """Compute np.convolve(a,a) via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def eval_safe(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def lp_fw(y0, time_limit, solver, init_K=400, max_K=4000, verbose=True):
    """LP-Frank-Wolfe with adaptive active sets."""
    n = len(y0)
    y = normalize(y0)
    best_a = y.copy()
    best_m = fast_conv(y).max()
    t0 = time.monotonic()

    ac = fast_conv(y)
    active_set = list(np.argsort(ac)[::-1][:init_K])
    no_improve = 0
    stall_limit = 1200

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0

    it = 0
    while time.monotonic() - t0 < time_limit and no_improve < stall_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt active set: keep top lags by value
        active_set = sorted(int(k) for k in np.argsort(ac)[::-1][:init_K])
        # Also include lags that are within 5% of max
        for k in range(size):
            if ac[k] > 0.95 * M_curr:
                if k not in active_set:
                    active_set.append(int(k))
        active_set = sorted(active_set)[:max_K]

        K = len(active_set)
        if K == 0:
            break

        coeffs = build_lp_matrix(y, active_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                if no_improve > 400:
                    break
                # Reduce active set and retry
                active_set = sorted(active_set[:max(100, K // 2)])
                continue
        except Exception:
            if no_improve > 300:
                break
            active_set = sorted(active_set[:max(100, K // 3)])
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            break
        g_dir = g_val / g_sum

        # Line search: golden section
        def trial_obj(alpha):
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                return 1e10
            trial = trial / ts
            return fast_conv(trial).max()

        lo, hi = 0.001, 1.0
        c1 = lo + inv_phi * (hi - lo)
        c2 = hi - inv_phi * (hi - lo)
        f1 = trial_obj(c1)
        f2 = trial_obj(c2)
        for _ in range(30):
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = lo + inv_phi * (hi - lo)
                f1 = trial_obj(c1)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = hi - inv_phi * (hi - lo)
                f2 = trial_obj(c2)
        best_alpha = (lo + hi) / 2.0

        # Additional refinement
        da = (hi - lo) * 0.3
        for _ in range(3):
            am = (lo + hi) / 2.0
            al = max(0.001, am - da)
            ar = min(0.999, am + da)
            fm = trial_obj(am)
            fl = trial_obj(al)
            fr = trial_obj(ar)
            denom = (ar - al) * (ar - am) * (al - am)
            if abs(denom) > 1e-18:
                a_v = am - 0.5 * (fl * (ar - am) / (al - am) - fr * (al - am) / (ar - am))
                if al < a_v < ar and trial_obj(a_v) < fm:
                    best_alpha = a_v
            da *= 0.4

        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy

        if verbose and it % 50 == 0:
            print("    LP-FW it=" + str(it) + " t=" + str(round(time.monotonic() - t0, 1)) +
                  "s best=" + repr(2 * n * best_m) + " |S|=" + str(K))

    if verbose:
        print("    LP-FW done it=" + str(it) + " t=" + str(round(time.monotonic() - t0, 1)) +
              "s best=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def lse_gradient_fw(y0, time_limit, beta=50.0, n_steps=200, verbose=True):
    """Log-sum-exp gradient Frank-Wolfe for smoothing."""
    n = len(y0)
    y = normalize(y0)
    best_a = y.copy()
    best_m = fast_conv(y).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    for t in range(1, n_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        # Compute softmax weights over autocorrelation
        ac_shifted = ac - ac.max()
        w = np.exp(beta * ac_shifted)
        w /= w.sum()

        # Gradient of LSE: grad[i] = sum_k w[k] * 2*y[k-i] (valid indices)
        grad = np.zeros(n)
        for k in range(size):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                grad[idx] += 2.0 * w[k] * y[k - idx]

        # FW step: find argmin grad on simplex
        g_dir = np.zeros(n)
        g_dir[int(np.argmin(grad))] = 1.0

        alpha = 2.0 / (t + 2.0)
        y_new = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy

        if verbose and t % 50 == 0:
            print("    LSE-FW t=" + str(t) + " M=" + repr(M_curr))

    return best_a, 2.0 * n * best_m

def coord_descent(a0, time_limit, n_peaks=20, verbose=True):
    """Direct coordinate descent: move mass to reduce peak autocorrelation."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit and stall < 5000:
        ac = fast_conv(a)
        peaks = np.argsort(ac)[::-1][:n_peaks]
        improved = False

        for k in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            # Find source with highest contribution
            src = idx[np.argmax(contrib)]
            src = int(src)
            if a[src] < 1e-14:
                continue

            # Score destinations by how much they'd reduce peaks
            scores = np.zeros(n)
            for k2 in peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]
            scores[src] = np.inf

            candidates = np.argsort(scores)[:40]
            found = False
            for dst in candidates:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-16:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_conv(a_t).max()
                    if Mt < best_m - 1e-15:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break

        if not improved:
            stall += 1
            if stall > 100:
                # Random perturbation
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i1] * frac
                    a[i1] -= amt
                    a[i2] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_conv(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 1000 == 0:
            print("    CD t=" + str(round(time.monotonic() - t0, 1)) + "s M=" +
                  repr(2 * n * best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Default init: n=" + str(n_init) + " score=" + repr(best_score))

    n_parent = len(best_seq)

    # === Phase 1: Quick diverse initialization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.02, 20.0)
    print("\n=== P1: Diverse Init ===")
    if t1 > 5:
        candidates = []
        x = np.linspace(0, 1, n_parent)
        i_arr = np.arange(n_parent, dtype=np.float64)
        for p_pow in [0.3, 0.5, 0.7, 1.0, 1.5]:
            candidates.append(normalize((i_arr + 1.0) ** p_pow))
            candidates.append(normalize((n_parent - i_arr) ** p_pow))
        for freq in [1, 2, 3, 4]:
            candidates.append(normalize(np.sin(np.pi * x * freq) ** 2))
        # Blend with parent
        bp = normalize(best_seq)
        for c in [candidates[0], candidates[len(candidates)//2]]:
            for w in [0.3, 0.5, 0.7]:
                candidates.append(normalize(w * bp + (1 - w) * c))
        for c in candidates:
            sc = eval_safe(c)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = c.copy()
                print("  Init improved: " + repr(best_score))

    # === Phase 2: Main LP-FW at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_main = min(len(best_seq), 10000)
    a_main = interpolate_seq(best_seq, n_main) if n_main != len(best_seq) else best_seq.copy()
    print("\n=== P2: Main LP-FW (n=" + str(n_main) + ") ===")
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 100:
        a_out, v_out = multi_start_lp(a_main, t2, solver, verbose=True)
        # Interpolate back to full length
        a_full = interpolate_seq(a_out, n_parent) if n_parent > n_main else a_out
        sc = eval_safe(a_full)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_full
            print("  P2 improved: " + repr(best_score))
        else:
            # Also try at working length
            sc2 = eval_safe(a_out)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_out
                n_parent = len(a_out)
                print("  P2 (working len) improved: " + repr(best_score))

    # === Phase 3: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cd = min(len(best_seq), 12000)
    a_cd = interpolate_seq(best_seq, n_cd) if n_cd != len(best_seq) else best_seq.copy()
    print("\n=== P3: Coord Descent (n=" + str(n_cd) + ") ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 30:
        a_out, v_out = coord_descent(a_cd, t3, n_peaks=15, verbose=True)
        a_full = interpolate_seq(a_out, len(best_seq)) if len(best_seq) > n_cd else a_out
        sc = eval_safe(a_full)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_full
            print("  P3 improved: " + repr(best_score))

    # === Phase 4: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: Multi-length ===")
    if remaining > 200:
        n_cur = len(best_seq)
        for n_try in [n_cur + 2000, n_cur - 1500, n_cur + 4000, n_cur - 3000]:
            if n_try < 4000 or n_try > 22000:
                continue
            if time.monotonic() - t_global > time_budget - 100:
                break
            a_try = interpolate_seq(best_seq, n_try)
            n_work = min(n_try, 8000)
            a_work = interpolate_seq(a_try, n_work) if n_work != n_try else a_try
            t_try = min(remaining * 0.18, 200.0)
            if t_try < 30:
                break
            a_out, v_out = multi_start_lp(a_work, t_try, solver, verbose=False)
            a_full = interpolate_seq(a_out, n_try) if n_try > n_work else a_out
            sc = eval_safe(a_full)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_full
                print("  n=" + str(n_try) + " improved: " + repr(best_score))

    # === Phase 5: LSE + LP kick at current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_kick = min(len(best_seq), 8000)
    a_kick = interpolate_seq(best_seq, n_kick) if n_kick != len(best_seq) else best_seq.copy()
    print("\n=== P5: LSE+LP Kick (n=" + str(n_kick) + ") ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 50:
        # LSE perturbation
        a_lse, _ = lse_gradient_fw(a_kick, min(t5 * 0.3, 40.0), beta=60.0, n_steps=100, verbose=False)
        # LP-FW from perturbed point
        a_out, v_out = lp_fw(a_lse, t5 * 0.6, solver, init_K=500, max_K=4000, verbose=True)
        a_full = interpolate_seq(a_out, len(best_seq)) if len(best_seq) > n_kick else a_out
        sc = eval_safe(a_full)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_full
            print("  P5 improved: " + repr(best_score))

    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_pol = min(len(best_seq), 8000)
    a_pol = interpolate_seq(best_seq, n_pol) if n_pol != len(best_seq) else best_seq.copy()
    print("\n=== P6: Final LP (n=" + str(n_pol) + ") ===")
    t6 = min(remaining * 0.30, 350.0)
    if t6 > 40:
        a_out, v_out = lp_fw(a_pol, t6, solver, init_K=700, max_K=5000, verbose=True)
        a_full = interpolate_seq(a_out, len(best_seq)) if len(best_seq) > n_pol else a_out
        sc = eval_safe(a_full)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_full
            print("  P6 improved: " + repr(best_score))

    # === Phase 7: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_fd = min(len(best_seq), 10000)
    a_fd = interpolate_seq(best_seq, n_fd) if n_fd != len(best_seq) else best_seq.copy()
    t7 = remaining - 15.0
    print("\n=== P7: Final CD (n=" + str(n_fd) + ") ===")
    if t7 > 20:
        a_out, v_out = coord_descent(a_fd, t7, n_peaks=12, verbose=False)
        a_full = interpolate_seq(a_out, len(best_seq)) if len(best_seq) > n_fd else a_out
        sc = eval_safe(a_full)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_full
            print("  P7 improved: " + repr(best_score))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score=" + repr(best_score) + " n=" + str(len(best_seq)) +
              " t=" + str(round(elapsed, 1)) + "s")
    else:
        print("\nERROR: fallback")
        n_fb = 10000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in np.asarray(best_seq, dtype=np.float64)]
    return result
