import numpy as np
import time

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative, finite."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n, dtype=np.float64) / n
    return a / s

def obj_val(a):
    """Compute objective 2*n*max_autocorr for normalized a."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def make_symmetric(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = (a + a[::-1].copy()) / 2.0
    return normalize(a_sym)

def mirror_descent(a0, time_limit, n_top=50, lr0=0.05, restart_interval=3000,
                   noise_scale=0.01, verbose=False, symmetric=False):
    """Mirror descent with adaptive top-k subgradient."""
    n = len(a0)
    a = normalize(a0).copy()
    if symmetric:
        a = make_symmetric(a)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr
        
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Select top-k lags
        k_sel = min(n_top, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[top_idx]
        
        # Compute subgradient (weight by lag value)
        s = np.zeros(n, dtype=np.float64)
        max_v = top_vals[0] if top_vals[0] > 0 else 1.0
        for ii in range(k_sel):
            ki = int(top_idx[ii])
            w = top_vals[ii] / max_v
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                indices = np.arange(j_start, j_end)
                s[indices] += 2.0 * w * a[ki - indices]
        
        # Adaptive learning rate
        lr = lr0 / (1.0 + 0.003 * iters)
        
        # Backtracking line search
        accepted = False
        for bt in range(6):
            log_a = np.log(a + 1e-30) - lr * s
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            if symmetric:
                a_new = (a_new + a_new[::-1].copy()) / 2.0
            a_new = np.maximum(a_new, 1e-25)
            a_new = a_new / a_new.sum()
            
            val_new = 2.0 * n * fast_autocorr(a_new).max()
            if val_new <= cur_val * 1.0001 + 1e-18:
                a = a_new
                accepted = True
                break
            lr *= 0.5
        
        if not accepted:
            a = a_new
        
        # Periodic restart with structured perturbation
        if stall > restart_interval:
            # Structured perturbation: shift mass between adjacent positions
            perturbation = np.zeros(n)
            for _ in range(n // 50):
                i = int(rng.integers(0, n - 1))
                amt = a[i] * rng.uniform(0.01, 0.05)

            a_r = np.maximum(a_r, 1e-12)

            a = a_r / a_r.sum()
            stall = 0
            lr = lr0
        
        if verbose and iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    MD t={:5d} ({:.1f}s) J={:.7f} stall={}".format(iters, elapsed, best_val, stall))

def lp_direction(f, M, n_vars=None):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (f*g)[k] <= M for all k.
    Returns the optimal g (normalized) or None if LP fails.
    """
    try:
        import cvxpy as cp
        n = len(f)
        n_v = n_vars if n_vars is not None else min(n, 2000)  # Limit LP size
        
        g = cp.Variable(n_v, nonneg=True)
        objective = cp.Maximize(cp.sum(g))
        
        # Build convolution constraint matrix (sparse)
        # (f*g)[k] = sum_i f[i]*g[k-i] for valid i, k-i
        # We need this <= M for k = 0, ..., n_v+n-2 (but we only need up to 2*n-2)
        # Actually for the LP, g has n_v elements, so f*g has at most (n + n_v - 1) elements
        num_constr = n + n_v - 1
        
        # Build sparse constraint matrix
        rows = []
        cols = []
        vals = []
        for k in range(num_constr):
            for i in range(n):
                j = k - i
                if 0 <= j < n_v:
                    rows.append(k)
                    cols.append(j)
                    vals.append(f[i])
        
        from scipy.sparse import csr_matrix
        A = csr_matrix((vals, (rows, cols)), shape=(num_constr, n_v))
        constraints = [A @ g <= M]
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CLARABEL)  # or PDLPO, CBC
        
        if prob.status == 'optimal' and g.value is not None:
            gv = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
            if gv.sum() > 1e-15:
                return gv / gv.sum()
    except Exception as e:
        print("    LP failed: {}".format(str(e)[:100]))
    return None

            pressure[i_src] = np.inf

                    a_t[dst] += amt
                    if symmetric:
                        a_t[n-1-i_src] -= amt
                        a_t[n-1-dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-16:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Random exploration
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    if symmetric:
                        a[n-1-i] -= amt
                        a[n-1-j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0
        else:
            stall = max(0, stall - 1)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse structural initializations with adaptive length selection
2. Mirror descent with adaptive top-k subgradient and backtracking
3. Periodic structured restarts (mass shuffling, not just noise)
4. Multi-pass coordinate descent with increasing granularity
5. Symmetric variant optimization
6. Final ultra-fine polishing

import numpy as np

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative, finite."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1e30)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def obj_val(a):
    """Compute objective 2*n*max_autocorr for normalized a."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def eval_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:

    except Exception:
        return float('inf')


def make_symmetric(a):
    """Make sequence symmetric and normalize."""
    a = (a + a[::-1].copy()) / 2.0

def mirror_descent(a0, time_limit, n_top=50, lr0=0.04, restart_every=2500,
                   noise_scale=0.008, verbose=False, symmetric=False):

    Mirror descent (multiplicative weights) with adaptive top-k subgradient.
    """
    n = len(a0)
    a = normalize(a0).copy()
    if symmetric:
        a = make_symmetric(a)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Select top-k lags with smooth weight decay
        k_sel = min(n_top, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[top_idx].copy()
        # Exponential decay of weights to emphasize top lag
        weights = (top_vals / (top_vals[0] + 1e-30)) ** 1.5

        # Compute subgradient
        s = np.zeros(n, dtype=np.float64)
        for ii in range(k_sel):
            ki = int(top_idx[ii])
            w = weights[ii]
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                indices = np.arange(j_start, j_end)
                s[indices] += 2.0 * w * a[ki - indices]

        # Adaptive learning rate
        lr = lr0 / (1.0 + 0.002 * iters)

        # Backtracking line search
        accepted = False
        for bt in range(7):
            log_a = np.log(a + 1e-30) - lr * s
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            if symmetric:
                a_new = (a_new + a_new[::-1].copy()) / 2.0
            a_new = np.maximum(a_new, 1e-25)
            a_new = a_new / a_new.sum()

            val_new = 2.0 * n * fast_autocorr(a_new).max()
            if val_new <= cur_val * 1.00005 + 1e-18:
                a = a_new
                accepted = True
                break
            lr *= 0.5

        if not accepted:
            a = a_new

                a_r = make_symmetric(a_r)
            a = a_r / a_r.sum()
            stall = 0
            lr = lr0

        if verbose and iters % 5000 == 0:

def coord_descent(a0, time_limit, n_targets=12, verbose=False, symmetric=False):

            best_m = cur_m
            best_a = a.copy()

        actual_targets = min(n_targets + pass_count // 500, 30)
        top_k = np.argsort(ac)[::-1][:actual_targets]
        improved = False

        for ki in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start >= j_end:
                continue

            indices = np.arange(j_start, j_end)
            contribs = np.array([a[jj] * a[ki - jj] for jj in indices])
            if contribs.max() < 1e-25:

            src_local = int(np.argmax(contribs))

def block_swap_descent(a0, time_limit, block_sizes=None, verbose=False):
    """
    Try swapping adjacent blocks of coefficients to reduce peak autocorrelation.
    """
    n = len(a0)
    a = normalize(a0).copy()

        ac = fast_autocorr(a)
        cur_m = ac.max()
        if cur_m < best_m - 1e-15:
            best_m = cur_m
            best_a = a.copy()

        improved = False
        bs = block_sizes[int(rng.integers(0, len(block_sizes)))]
        
        # Try swapping two random blocks
        for attempt in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            i1 = int(rng.integers(0, max(1, n - 2 * bs)))
            i2 = int(rng.integers(0, max(1, n - 2 * bs)))
            if abs(i1 - i2) < bs:
                continue
            imin = min(i1, i2)
            imax = max(i1, i2)
            if imax - imin < bs:
                continue

            a_t = a.copy()
            # Swap blocks
            a_t[imin:imin+bs], a_t[imax:imax+bs] = a_t[imax:imax+bs].copy(), a_t[imin:imin+bs].copy()
            # Also try mirrored swap
            a_t2 = a.copy()
            a_t2[imax:imax+bs] = a_t2[min(imin, max(0, n-1-imax)):min(imin, max(0, n-1-imax))+bs].copy()

            Mt = fast_autocorr(a_t).max()
            if Mt < best_m - 1e-16:
                a = a_t
                best_m = Mt
                best_a = a.copy()
                improved = True
                break

            Mt2 = fast_autocorr(a_t2).max()
            if Mt2 < best_m - 1e-16:
                a = a_t2
                best_m = Mt2
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Random single swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))

def gen_init(n, kind, rng):
    """Generate diverse initial sequences."""
    if kind == 0:  # uniform
        return np.ones(n) / n
    elif kind == 1:  # sine^2
        x = np.linspace(0, 1, n)
        return np.sin(np.pi * x) ** 2
    elif kind == 2:  # U-shaped
        x = np.linspace(0, 1, n)
        return (x - 0.5) ** 2 + 0.005
    elif kind == 3:  # double peak
        x = np.linspace(0, 1, n)
        return np.exp(-30 * (x - 0.25) ** 2) + np.exp(-30 * (x - 0.75) ** 2) + 0.005
    elif kind == 4:  # edge-weighted
        x = np.linspace(0, 1, n)
        d = np.minimum(x, 1 - x)
        return np.exp(-4 * d) + 0.005
    elif kind == 5:  # noisy uniform
        return np.maximum(np.ones(n) + 0.5 * rng.standard_normal(n), 0.01)
    elif kind == 6:  # triangular (peak at center)
        x = np.linspace(0, 1, n)
        return np.maximum(1 - 2 * np.abs(x - 0.5), 0.001)
    elif kind == 7:  # 1/(1+c*x^2) Cauchy-like
        x = np.linspace(0, 1, n)
        return 1.0 / (1.0 + 20.0 * x ** 2)
    return np.ones(n) / n


def propose_candidate(budget_s=1800.0):
    """
    Main search function: diverse initialization + multi-phase optimization.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = normalize(p)
            best_score = eval_safe(best_seq)
            print("Parent: n={} score={}".format(len(p), repr(best_score)))
    except Exception:
        best_seq = None

    rng_main = np.random.default_rng(2024)

    # Phase 1: Evaluate diverse initializations
    print("\n=== Phase 1: Diverse Initialization ===")
    candidates = []

    if best_seq is not None:
        candidates.append(('parent', best_seq.copy(), best_score))

    # Try different lengths and shapes
    lengths_to_try = [12000, 15000, 18000]
    if best_seq is not None:
        lengths_to_try.insert(0, len(best_seq))

    kinds_to_try = [0, 1, 2, 3, 4, 5, 6, 7]

    t_p1_end = t_global + 20  # 20 seconds for init evaluation
    for n_try in lengths_to_try[:3]:
        for kind in kinds_to_try:
            if time.monotonic() > t_p1_end:
                break
            a_try = normalize(gen_init(n_try, kind, rng_main))
            sc = eval_safe(a_try)
            if np.isfinite(sc):
                candidates.append(('n{}_k{}'.format(n_try, kind), a_try, sc))

    candidates.sort(key=lambda x: x[2])
    print("  Top 5 candidates:")
    for name, a, sc in candidates[:5]:
        print("    {} (n={}): score={}".format(name, len(a), repr(sc)))

    if not candidates:
        best_seq = normalize(np.ones(12000))
        best_score = eval_safe(best_seq)
        candidates = [('default', best_seq, best_score)]

    if candidates[0][2] < best_score:
        best_score = candidates[0][2]
        best_seq = candidates[0][1].copy()

    # Phase 2: Main Mirror Descent (multiple starts)
    print("\n=== Phase 2: Mirror Descent (top-3 starts) ===")
    t_p2_end = t_global + time_budget * 0.45

    for idx in range(min(3, len(candidates))):
        remaining = t_p2_end - time.monotonic()
        if remaining < 60:
            break
        n_starts_left = min(3, len(candidates)) - idx
        alloc = min(remaining / n_starts_left, 300.0)
        name, a_start, sc_start = candidates[idx]
        print("  MD [{}] n={} init={} alloc={:.0f}s".format(name, len(a_start), repr(sc_start), alloc))

        a_out, v_out = mirror_descent(a_start, alloc, n_top=50, lr0=0.04,
                                       restart_every=2000, noise_scale=0.008, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  -> improved: {}".format(repr(sc)))

    # Phase 3: Asymmetric MD on current best with different params
    print("\n=== Phase 3: MD Wide (n_top=100, low lr) ===")
    t_p3_end = t_global + time_budget * 0.60
    remaining = t_p3_end - time.monotonic()
    if remaining > 60:
        a_out, v_out = mirror_descent(normalize(best_seq), min(remaining, 250),
                                       n_top=100, lr0=0.015, restart_every=3000,
                                       noise_scale=0.005, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  -> improved: {}".format(repr(sc)))

    # Phase 4: Coordinate Descent
    print("\n=== Phase 4: Coordinate Descent ===")
    t_p4_end = t_global + time_budget * 0.75
    remaining = t_p4_end - time.monotonic()
    if remaining > 30:
        a_out, v_out = coord_descent(normalize(best_seq), min(remaining, 250),
                                      n_targets=15, verbose=True, symmetric=False)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  -> improved: {}".format(repr(sc)))

    # Phase 5: Symmetric variant
    print("\n=== Phase 5: Symmetric MD + CD ===")
    t_p5_end = t_global + time_budget * 0.85
    remaining = t_p5_end - time.monotonic()
    if remaining > 30:
        a_sym = make_symmetric(best_seq)
        sc_sym = eval_safe(a_sym)
        a_out, v_out = mirror_descent(a_sym, min(remaining * 0.6, 150),
                                       n_top=40, lr0=0.02, restart_every=2000,
                                       noise_scale=0.005, verbose=False, symmetric=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  -> sym improved: {}".format(repr(sc)))
        else:
            # CD on symmetric
            remaining = t_p5_end - time.monotonic()
            if remaining > 20:
                a_out, v_out = coord_descent(a_sym, min(remaining, 100),
                                              n_targets=12, verbose=False, symmetric=True)
                sc = eval_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out
                    print("  -> sym CD improved: {}".format(repr(sc)))

    # Phase 6: Final ultra-fine polish
    print("\n=== Phase 6: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        a_out, v_out = mirror_descent(normalize(best_seq), min(remaining, 80),
                                       n_top=15, lr0=0.003, restart_every=8000,
                                       noise_scale=0.002, verbose=True)
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  -> final improved: {}".format(repr(sc)))
        else:
            # Tiny CD
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_out, v_out = coord_descent(normalize(best_seq), min(remaining, 60),
                                              n_targets=20, verbose=False, symmetric=False)
                sc = eval_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_out
                    print("  -> final CD improved: {}".format(repr(sc)))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} t={:.1f}s".format(repr(best_score), len(best_seq), elapsed))
    else:
        print("\nERROR: using fallback")
        n_fb = 12000
        x = np.linspace(0, 1, n_fb)
        best_seq = normalize(np.sin(np.pi * x) ** 2)
        best_score = eval_safe(best_seq)
        print("Fallback: score={} t={:.1f}s".format(repr(best_score), time.monotonic() - t_global))

    return [float(x) for x in best_seq]
