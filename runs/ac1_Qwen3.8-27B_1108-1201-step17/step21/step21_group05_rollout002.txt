import time
import scipy.optimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), n=fft_size)
    return np.fft.irfft(A * A, n=fft_size)[:size]

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_lp_direction(f, M, top_lags):
    """
    Solve: max sum(g) s.t. (f*g)[k_i] <= M for i in range(len(top_lags)), g >= 0
    Returns the optimal g (normalized to sum=1).
    """
    n = len(f)
    K = len(top_lags)
    
    # Build constraint matrix A (K x n)
    # A[i, j] = f[k_i - j] if 0 <= k_i - j < n, else 0
    A = np.zeros((K, n), dtype=np.float64)
    for i, ki in enumerate(top_lags):
        ki = int(ki)
        # g[j] contributes to (f*g)[ki] with coefficient f[ki-j]
        # Valid when 0 <= ki-j < n, i.e., ki-n+1 <= j <= ki
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start < j_end:
            A[i, j_start:j_end] = f[ki - j_end + 1:ki - j_start + 1][::-1]
    
    # Solve: max 1^T g s.t. A g <= M*1, g >= 0
    # Using linprog: min -1^T g s.t. A g <= M*1, g >= 0
    c = -np.ones(n)
    b = np.full(K, M)
    
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None
    
    g = res.x
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s

Let me also consider: should I use the LP to find g, or should I use a first-order method on the dual?

The dual of the LP is:
min M * sum(y_i) s.t. A^T y >= 1, y >= 0

where A^T is n x K. The constraint A^T y >= 1 means:
for each j in [0, n-1]: sum_i A[i,j] * y_i >= 1
i.e., sum_i f[k_i - j] * y_i >= 1

This is (f * y_extended)[j] >= 1 for all j, where y has K non-zero entries.

If I can solve this dual, I get y*. Then the primal g* is determined by complementary slackness: g*[j] > 0 only if (A^T y*)[j] = 1.

For the practical implementation, I'll just use linprog on the primal (which has n variables and K constraints).

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based SOTA iterations (solve LP to find optimal direction)
2. Mirror descent with adaptive subgradient
3. Coordinate descent for fine-tuning
4. Multi-start exploration
"""

import numpy as np
import time
import scipy.optimize

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), n=fft_size)
    return np.fft.irfft(A * A, n=fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    """Compute objective 2*n*max_autocorr for normalized a."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def lp_direction(f, M, top_lags, verbose=False):
    """
    Solve LP: max sum(g) s.t. (f*g)[k_i] <= M for i, g >= 0.
    Returns normalized g or None.
    """
    n = len(f)
    K = len(top_lags)
    
    if K == 0 or n == 0:
        return None
    
    # Build constraint matrix A (K x n)
    A = np.zeros((K, n), dtype=np.float64)
    for i in range(K):
        ki = int(top_lags[i])
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start < j_end:
            # A[i, j] = f[ki - j] for j in [j_start, j_end)
            # ki - j ranges from ki - j_start to ki - (j_end-1)
            # = [ki - j_start, ki - j_end + 1]
            A[i, j_start:j_end] = f[ki - j_end + 1 : ki - j_start + 1][::-1]
    
    # Solve: min -sum(g) s.t. A@G <= M*1, g >= 0
    c = -np.ones(n, dtype=np.float64)
    b = np.full(K, M, dtype=np.float64)
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, 
                                      bounds=[(0, None)] * n,
                                      method='highs',
                                      options={'presolve': True})
        if res.status != 0:
            if verbose:
                print(f"  LP status: {res.status}")
            return None
        g = res.x
    except Exception as e:
        if verbose:
            print(f"  LP error: {e}")
        return None
    
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s

def sota_iterate(a, time_limit, K=100, blend=0.3, verbose=True):
    """
    LP-based SOTA iterations.
    """
    n = len(a)
    a_work = normalize(a.copy())
    best_a = a_work.copy()
    best_val = obj_val(a_work)
    t0 = time.monotonic()
    
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a_work)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr
        
        if cur_val < best_val - 1e-15:
            best_val = cur_val
            best_a = a_work.copy()
        
        # Select top-K lags
        k_sel = min(K, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        # Use the actual values at those lags as M
        M_use = ac[top_idx].max()
        
        # Solve LP
        g = lp_direction(a_work, M_use, top_idx)
        if g is None:
            # Fallback: uniform
            g = np.ones(n) / n
        
        # Blend
        a_new = (1 - blend) * a_work + blend * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a_work
        else:
            a_new = a_new / s
        
        a_work = a_new
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SOTA it={it} ({elapsed:.1f}s) J={best_val:.7f} cur={2*n*fast_autocorr(a_work).max():.7f}")
    
    return best_a, best_val

def mirror_descent_verbose(a0, time_limit, n_top=50, lr0=0.05, verbose=True):
    """Mirror descent with multiplicative weights."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr
        
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        k_sel = min(n_top, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[top_idx]
        
        # Compute subgradient
        s_grad = np.zeros(n, dtype=np.float64)
        for ii in range(k_sel):
            ki = int(top_idx[ii])
            w = top_vals[ii] / (top_vals[0] + 1e-30)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                indices = np.arange(j_start, j_end)
                s_grad[indices] += 2.0 * w * a[ki - indices]
        
        # Mirror descent update
        lr = lr0 / (1.0 + 0.002 * t)
        log_a = np.log(a + 1e-30) - lr * s_grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-25)
        a_new = a_new / a_new.sum()
        
        val_new = 2.0 * n * fast_autocorr(a_new).max()
        if val_new > cur_val * 1.005 + 1e-15:
            lr *= 0.5
            log_a = np.log(a + 1e-30) - lr * s_grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-25)
            a_new = a_new / a_new.sum()
        
        a = a_new
        
        if stall > 3000 and stall % 500 == 0:

            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-12)
            a = a_r / a_r.sum()
            stall = 0
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0

def coord_descent_verbose(a0, time_limit, n_targets=10, verbose=True):
    """Coordinate descent: move mass from peak contributors to quiet positions."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    fracs = [0.01, 0.02, 0.05, 0.1, 0.2]
    
    while time.monotonic() - t0 < time_limit and stall < 3000:
        ac = fast_autocorr(a)
        cur_m = ac.max()
        if cur_m < best_m - 1e-16:
            best_m = cur_m
            best_a = a.copy()
        
        top_k = np.argsort(ac)[::-1][:n_targets]
        improved = False
        
        for ki in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start >= j_end:
                continue
            
            indices = np.arange(j_start, j_end)
            contribs = a[indices] * a[ki - indices]
            if contribs.max() < 1e-25:
                continue
            
            src_local = int(np.argmax(contribs))
            i_src = int(indices[src_local])
            if a[i_src] < 1e-15:
                continue
            
            # Find quiet positions
            pressure = np.zeros(n, dtype=np.float64)
            for kl in top_k[:5]:
                kll = int(kl)
                js = max(0, kll - n + 1)
                je = min(n, kll + 1)
                if js < je:
                    idx2 = np.arange(js, je)
                    pressure[idx2] += a[kll - idx2]
            pressure[i_src] = np.inf
            quiet = np.argsort(pressure)[:20]
            
            for dst in quiet:
                dst = int(dst)
                if dst == i_src:
                    continue
                for frac in fracs:
                    amt = a[i_src] * frac
                    if amt < 1e-20:
                        break
                    a_t = a.copy()
                    a_t[i_src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m - 1e-17:
                        a = a_t
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            
        if verbose and stall % 500 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"  CD ({elapsed:.1f}s) J={2*n*best_m:.7f} stall={stall}")
    
    return best_a, 2.0 * n * best_m
