def mirror_descent_smooth(a0, time_limit, t_start=10.0, t_end=5000.0, verbose=True):
    """Mirror descent with soft-max smoothing."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = score_fast(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    # Momentum
    x_prev = x.copy()
    
    t_smooth = t_start
    total_iters = 50000
    stall = 0
    max_stall = 3000
    
    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        t_smooth = t_start * (t_end / t_start) ** (it / total_iters)
        
        # Nesterov momentum
        beta = 0.5 * (it + 1) / (it + 1 + 10.0)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        w, grad = softmax_grad(y, t_smooth, top_k=min(200, 2*n-1))
        
        # Mirror descent step (multiplicative weights)
        eta = 3.0 / (np.sqrt(max(it, 1)) + 2.0)
        log_new = np.log(y) - eta * grad
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        val = score_fast(x)
        if val < best_val - 1e-13:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 1000 and stall % 1000 == 0:
            # Perturb
            noise = 0.01 * rng.standard_normal(n) * best_a
            x = best_a + noise
            x = np.maximum(x, 1e-12)
            x = x / x.sum()
            x_prev = x.copy()
            stall = 0
        
        if verbose and it % 500 == 0:
            print(f"    MD t={it} ({time.monotonic()-t0:.1f}s) J={best_val:.6f} t_smooth={t_smooth:.1f}")
    
    return best_a, best_val

        else:
            no_improve += 1

        if len(clist) > 6000:

        coeffs = np.zeros((K, n), dtype=np.float64)

            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1]
        constraints += [coeffs[jj] @ g_var <= M_var for jj in range(K)]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                lp_fail += 1
                continue
        except Exception:
            lp_fail += 1
            continue

        g_sum = g_val.sum()
        if g_sum < 1e-15:

        g_dir = g_val / g_sum

            if ts < 1e-15:

            trial /= ts
            f = score_fast(trial)
            if f < best_f:

                best_alpha = alpha

        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        if verbose and t % 100 == 0:
            print(f"    LP-FW t={t} ({time.monotonic()-t0:.1f}s) J={2*n*best_m:.6f} |S|={len(active_set)}")
    
    return best_a, 2.0 * n * best_m
