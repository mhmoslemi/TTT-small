import numpy as np
import time
import cvxpy as cp

# Helper functions

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    na = len(a)
    size = 2 * na - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        al = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(al))
    except Exception:
        return float('inf')

# Main optimization: LP-FW with adaptive active set
def lp_fw(a0, time_limit, solver, init_K=500, max_K=4000, verbose=True):
    n = len(a0)
    y = normalize(a0)
    best = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    ac = fast_autocorr(y)
    active = set(int(k) for k in np.argsort(ac)[::-1][:init_K])
    no_improve = 0
    stall_limit = 3000
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M = ac.max()
        if M < best_m - 1e-15:
            best_m = M
            best = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodically refresh active set
        if t % 100 == 0:
            top = np.argsort(ac)[::-1][:max_K]
            active = set(int(k) for k in top)
        
        clist = sorted(active)
        K = len(clist)
        if K == 0:
            break
        
        # Build LP: minimize M subject to (y*g)_k <= M for k in clist, sum(g)=1, g>=0
        # (y*g)_k = sum_j y_{k-j} * g_j (for valid j)
        # This is linear in g!
        coeffs = np.zeros((K, n))
        for ii, k in enumerate(clist):
            js = max(0, k - n + 1)
            je = min(n, k + 1)
            if js < je:
                idx = np.arange(js, je)
                coeffs[ii, idx] = y[k - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                # Reduce active set
                if len(active) > 100:
                    active = set(int(k) for k in np.argsort(ac)[::-1][:min(len(active)//2, 500)])
                continue
        except Exception:
            if len(active) > 100:
                active = set(int(k) for k in np.argsort(ac)[::-1][:500])
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        gs = g_val.sum()
        if gs < 1e-15:
            continue
        g_dir = g_val / gs
        
        # Add violated constraints to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        ygm = np.fft.irfft(Yf * Gf, fft_size)[:size]
        lp_M = float(M_var.value) if M_var.value is not None else M
        thresh = lp_M * 1.01
        viol = np.where(ygm > thresh)[0]
        if len(viol) > 0:
            top_v = viol[np.argsort(ygm[viol])[::-1][:100]]
            for k in top_v:
                active.add(int(k))
        
        # Also add top autocorrelation peaks
        top_peaks = np.argsort(ac)[::-1][:min(200, size)]
        for k in top_peaks[:100]:
            active.add(int(k))
        
        # Line search
        # Try a few step sizes
        best_alpha = 0.1
        best_f = fast_autocorr(normalize((1-0.1)*y + 0.1*g_dir)).max()
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            trial = np.maximum((1-alpha)*y + alpha*g_dir, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            f = fast_autocorr(trial).max()
            if f < best_f:
                best_f = f
                best_alpha = alpha
        
        y = np.maximum((1-best_alpha)*y + best_alpha*g_dir, 0)
        y = y / y.sum()
        
        if verbose and t % 100 == 0:
            print(f"  LP-FW t={t} ({time.monotonic()-t0:.0f}s) J={2*n*best_m:.6f} |A|={len(active)} a={best_alpha:.3f}")
    
    return best, 2.0 * n * best_m

def grad_flat(a0, time_limit, verbose=True):
    n = len(a0)
    x = normalize(a0)
    best = x.copy()
    best_val = obj(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    stall = 0
    eta = 0.5
    
    x_prev = x.copy()
    
    for t in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break
        
        # Nesterov momentum
        beta = 0.9
        y = np.maximum(x + beta*(x - x_prev), 1e-15)
        y = y / y.sum()
        
        ac = fast_autocorr(y)
        
        # Subgradient of max(autocorr): concentrate on top peaks
        k_max = min(50, size)
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        # Softmax weights
        temp = 5.0 / (1 + t/1000)
        w = np.exp(temp * (top_vals - top_vals[0]))
        w = w / w.sum()
        
        # Compute gradient of sum w_k * b_k
        # b_k = sum_i a_i * a_{k-i}
        # db_k/da_j = a_{k-j} + a_{k-j} (if j != k-j) or a_j (if j = k-j)
        # = 2*a_{k-j} (for the linear corr, accounting for both terms)
        grad = np.zeros(n)
        for wi, k in zip(w, idx):
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n, ki + 1)
            idx_arr = np.arange(js, je)
            # db_k/da_j = a_{k-j} (from i=j term) + a_{k-j} (from i=k-j term, if different)
            # Actually: b_k = sum_i a[i]*a[k-i]
            # db_k/da[j] = a[k-j] (from i=j) + a[k-j] (from i=k-j, if k-j in range and k-j != j)
            # More carefully: b_k = sum_i a[i]*a[k-i]
            # ∂/∂a[j] = a[k-j] (i=j term) + sum over i where k-i=j, i.e., i=k-j: a[i]*1 = a[k-j]
            # But if j = k-j, we only count it once: a[j]
            # So: ∂b_k/∂a[j] = 2*a[k-j] if j != k-j, a[j] if j = k-j
            # In vector form: ∂b_k/∂a = 2*a[k-n+1:k+1] (padded) with correction at center
            grad[idx_arr] += 2.0 * wi * y[ki - idx_arr]
        # Fix the center (if k is even, the center element is double-counted)
        # This is a minor correction, skip for simplicity
        
        # Gradient step on simplex
        eta_t = eta / (1.0 + 0.01 * t)
        y_new = y - eta_t * grad
        y_new = np.maximum(y_new, 0)
        y_new = y_new / y_new.sum()
        
        x_prev = x
        x = y_new
        
        val = obj(x)
        if val < best_val - 1e-14:
            best_val = val
            best = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Periodic perturbation
        if stall > 1000 and stall % 500 == 0:
            noise = 0.01 * rng.standard_normal(n) * best
            x_r = best + noise
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
