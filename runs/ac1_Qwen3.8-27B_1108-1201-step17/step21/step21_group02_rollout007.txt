                continue
            idx = np.arange(i_lo, i_hi + 1)
            contrib = p[idx] * p[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            srcs = idx[np.argsort(contrib)[::-1][:3]]
            for src in srcs:
                src = int(src)
                if p[src] < 1e-12:
                    continue
                
                # Find good destination
                dest_candidates = np.argsort(p)[:50]
                for dst in dest_candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02]:
                        amt = p[src] * frac
                        p_t = p.copy()
                        p_t[src] -= amt
                        p_t[dst] += amt
                        p_t = np.maximum(p_t, 0)
                        p_t = p_t / p_t.sum()
                        vt = fast_conv(p_t).max()
                        if 2.0 * n * vt < best_v - 1e-15:
                            p = p_t
                            best_v = 2.0 * n * vt
                            best_p = p.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and p[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    p[i] -= p[i] * frac
                    p[j] += p[i] * frac  # BUG: p[i] already modified
                    # Fix:
                    pass
        
        if verbose and stall % 500 == 0:
            print(f"  Coord ({time.monotonic()-t0:.1f}s) J={best_v:.8f} stall={stall}")
    
    return best_p, best_v

def diverse_init(nn, parent=None):
    x = np.linspace(0, 1, nn)
    ia = np.arange(nn, dtype=np.float64)
    cands = []
    
    # Various shapes
    for pw in [1, 2, 3, 5]:
        cands.append(normalize_p(np.sin(np.pi * x)**pw))
    
    for c in [0.1, 0.25, 0.5, 0.75, 0.9]:
        for sg in [0.05, 0.1, 0.2, 0.3]:
            cands.append(normalize_p(np.exp(-((x-c)**2)/(2*sg*sg))))
    
    for f in [1,2,3,5,7,11,13,17]:
        cands.append(normalize_p(np.maximum(1+np.cos(2*np.pi*f*x), 0.01)))
    
    for pw in [0.3, 0.7, 1.0, 2.0]:
        cands.append(normalize_p((ia+1)**pw * (nn-ia+1)**pw))
    
    # Random smooth
    rng = np.random.default_rng(123)
    for _ in range(5):
        r = rng.standard_normal(nn)
        # Smooth
        from numpy import convolve
        kern = np.ones(10)/10
        r = convolve(r, kern, 'same')
        r = np.maximum(r, 0)
        if r.sum() > 0:
            cands.append(r/r.sum())
    
    if parent is not None:
        pn = normalize_p(parent)
        cands.append(pn)
        for w in [0.5, 0.7, 0.9]:
            for c in cands[:4]:
                cands.append(normalize_p(w*pn + (1-w)*c))
    
    best_a = None
    best_v = float('inf')
    for c in cands:
        cv = obj(c)
        if cv < best_v:
            best_v = cv
            best_a = c.copy()
    
    if best_a is None:
        best_a = np.ones(nn)/nn
        best_v = obj(best_a)
    
    return best_a, best_v
