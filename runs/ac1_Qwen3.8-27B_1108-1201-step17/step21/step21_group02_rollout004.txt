def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(y, lags, n):
    """Build constraint matrix for LP: row k has y[k-i] at position i where valid."""
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    return coeffs

            lp_fail_count = 0
        else:

        clist = sorted(active_set)

        lp_fail_count = 0

        g_sum = g_val.sum()
        if g_sum < 1e-15:

        g_dir = g_val / g_sum

        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]

        # Add lags where (y*g) is close to the LP bound
        threshold = max(lp_M * 1.01, M_curr * 0.9) if lp_M is not None else M_curr * 0.95

def multiplicative_update(a0, time_limit, verbose=True):
    """Multiplicative/entropic updates with adaptive temperature for peak deflation."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Adaptive parameters
    temp = 50.0
    eta_base = 5.0
    top_k = 60

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(x)
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx]
        peak = top_vals[0]

        # Adaptive temperature: cool over time
        cur_temp = temp * max(0.3, 1.0 - t / 20000.0)

        # Weights emphasizes peaks
        w = np.exp(cur_temp * (top_vals - peak))
        w = w / w.sum()

        # Compute gradient-like quantity
        # For each j, grad_j = sum_k w_k * (a_{j-k} + a_{k-j})
        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            if i_s < i_e:
                idx = np.arange(i_s, i_e)
                grad[idx] += wi * x[ki - idx]
                grad[ki - idx] += wi * x[idx]

        # Adaptive step size
        eta_t = eta_base / (np.sqrt(max(t, 1)) + 2.0)

        # Multiplicative update (preserves positivity)
        update = np.exp(-eta_t * grad * peak)  # peak normalization for scale
        x_new = x * update
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()

        val = obj_val(x_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = x_new.copy()
        x = x_new

        # Periodic perturbation to escape local minima
        if t % 5000 == 0 and t > 1000:
            noise_scale = 0.003 + 0.005 * min(1.0, t / 10000.0)
            noise = noise_scale * rng.standard_normal(n)
            x_pert = x + noise * x
            x_pert = np.maximum(x_pert, 1e-15)
            x_pert = x_pert / x_pert.sum()
            if obj_val(x_pert) < best_val:
                x = x_pert

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    MultUpd t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val) + " peak=" + repr(peak))

    return best_a, best_val

def try_initials(n, parent=None):
    """Try various initial shapes and return the best one."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Sine family
    for p in [0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 4.0]:
        candidates.append(np.sin(np.pi * x) ** p)

    # Gaussian bumps
    for c in [0.15, 0.25, 0.35, 0.5, 0.65, 0.75, 0.85]:
        for sg in [0.08, 0.12, 0.18, 0.25]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sg * sg)))

    # Double bumps
    for c1 in [0.2, 0.3]:
        for c2 in [0.7, 0.8]:
            for sg in [0.08, 0.12, 0.18]:
                candidates.append(np.exp(-((x - c1) ** 2) / (2 * sg * sg)) +
                                  np.exp(-((x - c2) ** 2) / (2 * sg * sg)))

    # Power-law edges
    for p in [0.3, 0.5, 1.0, 2.0]:
        candidates.append((i_arr + 1.0) ** p * (n - i_arr) ** p)

    # Raised cosine / Tukey
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        cands = np.ones(n)
        m = max(1, int(alpha * n // 2))
        taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
        cands[:m] = taper
        cands[-m:] = taper[::-1]
        candidates.append(cands)

    # Blackman-like
    a0, a1, a2 = 0.42, 0.5, 0.08
    candidates.append(a0 - a1 * np.cos(2 * np.pi * x / n) + a2 * np.cos(4 * np.pi * x / n))

    # Hamming
    candidates.append(0.54 - 0.46 * np.cos(2 * np.pi * x / n))

    # Blend with parent
    if parent is not None:
        bp = normalize(parent)
        for w in [0.3, 0.5, 0.7, 0.9]:
            for c in candidates[:8]:
                candidates.append(w * bp + (1 - w) * c)

    best_a = None
    best_val = float('inf')
    for c in candidates:
        cn = normalize(c)
        v = obj_val(cn)
        if v < best_val:
            best_val = v
            best_a = cn

    if parent is not None:
        pv = obj_val(normalize(parent))
        if pv < best_val:
            best_val = pv
            best_a = normalize(parent)

    return best_a, best_val

        if t % 250 == 0 and len(active_set) > max_K:
            top_refresh = np.argsort(ac)[::-1][:max_K]
            active_set = set(int(k) for k in top_refresh)

        # Line search
        if t <= 4:
            candidates = [1.0, 0.8, 0.6, 0.4, 0.2, 0.05]
            best_alpha = candidates[0]
            best_f = fw_trial_max(candidates[0], y, g_dir, n, fft_size)
            for alpha in candidates[1:]:
                f_trial = fw_trial_max(alpha, y, g_dir, n, fft_size)
                if f_trial < best_f:
                    best_f = f_trial
                    best_alpha = alpha
        else:
            lo = 0.001
            hi = 1.0
            c1 = lo + inv_phi * (hi - lo)
            c2 = hi - inv_phi * (hi - lo)
            f1 = fw_trial_max(c1, y, g_dir, n, fft_size)
            f2 = fw_trial_max(c2, y, g_dir, n, fft_size)
            for _ in range(28):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_phi * (hi - lo)
                    f1 = fw_trial_max(c1, y, g_dir, n, fft_size)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = hi - inv_phi * (hi - lo)
                    f2 = fw_trial_max(c2, y, g_dir, n, fft_size)
            best_alpha = (lo + hi) * 0.5

        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()

        if verbose and t % 50 == 0:
            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" +
                  repr(2 * n * best_m) + " |S|=" + str(len(active_set)) +
                  " a=" + repr(round(best_alpha, 4)))

    if verbose:
        print("    LP-FW done (" + str(round(time.monotonic() - t0, 1)) +
              "s) J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m
