    # Budget for beta
    total_steps = 50000
    if beta_schedule is None:
        beta_schedule = 20.0
    
    stall = 0
    max_stall = 2000
    
    for t in range(1, total_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            break
        
        # Adaptive beta
        beta = min(200.0, beta_schedule * (1.0 + 0.1 * t))
        
        grad, weights, ac = smoothed_max_gradient(a, beta)
        
        # Project gradient to sum=0 subspace
        grad = grad - grad.mean()
        
        # Adam update
        t_step += 1
        lr = 0.3 / (1.0 + 0.01 * t_step)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project: non-negative and sum to 1
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        val = obj_val(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try smaller step
            a_half = a - 0.1 * lr * m_hat / (np.sqrt(v_hat) + eps)
            a_half = np.maximum(a_half, 0.0)
            sh = a_half.sum()
            if sh > 1e-15:
                a_half = a_half / sh
            val_h = obj_val(a_half)
            if val_h < obj_val(a):
                a = a_half
            stall += 1
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    GD t=%d (%.1fs) J=%.9f beta=%.1f avg_w_max=%.6f" % (t, elapsed, best_val, beta, weights.max()))
    
    return best_a, best_val

def water_filling_direction(a, active_lags, M, max_iter=200):
    """Compute FW direction via iterative water-filling (no LP solver)."""
    n = len(a)
    g = np.ones(n) / n  # start uniform
    
    for it in range(max_iter):
        # Compute (a * g) at active lags
        conv = fast_conv_slice(a, g, active_lags)
        violations = conv - M
        if violations.max() < 1e-12:
            break
        
        # For each violating lag k, reduce g at positions contributing to it
        for k in active_lags:
            vk = conv[int(k)] - M
            if vk <= 1e-12:
                continue
            # contrib[j] = a[j] * g[k-j] + a[k-j] * g[j] (approximately)
            # Simplest: reduce g at positions where a is large and k-a is in range
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * g[int(k) - idx] + a[int(k) - idx] * g[idx]
            total = contrib.sum()
            if total < 1e-15:
                continue
            # Reduce proportionally
            reduction = min(1.0, abs(vk) / (total * 2.0 + 1e-15))
            g[idx] -= reduction * contrib / (g[idx].max() + 1e-15) * reduction
            g = np.maximum(g, 0)
            g_sum = g.sum()
            if g_sum < 1e-15:
                return np.ones(n) / n
            g = g / g_sum
    
    return g

def fast_conv_slice(a, g, lags):
    """Compute convolution (a*g)[k] for specific lags k."""
    # Use FFT for efficiency
    size = len(a) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=fft_size)
    G = np.fft.rfft(g, n=fft_size)
    conv = np.fft.irfft(A * G, fft_size)[:size]
    return np.array([conv[int(k)] for k in lags])

def lp_fw_simple(a0, time_limit, init_K=300, max_K=600, verbose=True):
    """Simplified LP-FW using water-filling instead of cvxpy."""
    n = len(a0)
    y = normalize(a0).copy()
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    ac = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[-init_K:])
    no_improve = 0
    stall_limit = 500
    
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update active set
        top_k = np.argsort(ac)[-max_K:]
        active_set = set(int(k) for k in top_k)
        
        # Get direction via water-filling
        g_dir = water_filling_direction(y, sorted(active_set), M_curr, max_iter=50)
        
        # Line search (golden section)
        lo, hi = 0.0, 1.0
        c1 = lo + inv_phi * (hi - lo)
        c2 = hi - inv_phi * (hi - lo)
        
        def trial(alpha):
            trial_a = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial_a.sum()
            if ts < 1e-15:
                return 1e10
            trial_a = trial_a / ts
            return fast_autocorr(trial_a).max()
        
        f1, f2 = trial(c1), trial(c2)
        for _ in range(15):
            if f1 < f2:
                hi, c2, f2 = c2, c1, f1
                c1 = lo + inv_phi * (hi - lo)
                f1 = trial(c1)
            else:
                lo, c1, f1 = c1, c2, f2
                c2 = hi - inv_phi * (hi - lo)
                f2 = trial(c2)
        
        best_alpha = (lo + hi) / 2.0
        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print("    FW t=%d (%.1fs) J=%.9f" % (t, elapsed, 2 * n * best_m))
    
    return best_a, 2.0 * n * best_m

def peak_local_opt(a0, time_limit, n_targets=15, verbose=True):
    """Target specific peak lags with local moves."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for rnd in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False
        
        for k in targets:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            top_srcs = idx[np.argsort(contrib)[::-1][:3]]
            for src in top_srcs:
                src = int(src)
                if a[src] < 1e-12:
                    continue
                
                # Find best destination
                scores = np.zeros(n)
                for k2 in targets[:8]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                scores[src] = 1e10
                
                candidates = np.argsort(scores)[:30]
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02]:
                        amt = a[src] * frac
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Random kick
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            improved = True
                            break
        
        if verbose and rnd % 200 == 0:
            print("    PeakOpt rnd=%d J=%.9f" % (rnd, 2 * n * best_m))
    
    return best_a, 2.0 * n * best_m

def diverse_init(n, parent_seq=None):
    """Generate diverse initial candidates."""
    x = np.linspace(0.0, 1.0, n)
    
    candidates = []
    # Sine windows with different powers
    for p in [1, 2, 3, 4, 5]:
        candidates.append(normalize(np.sin(np.pi * x) ** (2 * p)))
    # Cosine windows
    for a in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(normalize(np.cos(np.pi * (x - 0.5)) ** (2 * a)))
    # Gaussian bumps
    for c in [0.25, 0.5, 0.75]:
        for sg in [0.1, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))
    # Tapered uniform
    for m_frac in [0.1, 0.2, 0.3]:
        cands = np.ones(n)
        m = max(1, int(m_frac * n))
        taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
        cands[:m] = taper
        cands[-m:] = taper[::-1]
        candidates.append(normalize(cands))
    
    if parent_seq is not None:
        bi = normalize(parent_seq)
        for c in candidates[:8]:
            for w in [0.7, 0.9]:
                candidates.append(normalize(w * bi + (1 - w) * c))
    
    best_a = candidates[0]
    best_val = obj_val(best_a)
    for c in candidates:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()
    
    return best_a, best_val
