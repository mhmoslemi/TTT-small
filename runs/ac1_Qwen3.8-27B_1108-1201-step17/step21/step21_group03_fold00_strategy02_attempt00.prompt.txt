import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_corr(a, g):
    """Compute sum_i a[k-i]*g[i] for all k (= cross-correlation)."""
    n_a = len(a)

    A = np.fft.rfft(a, n=fft_size)
    G = np.fft.rfft(g, n=fft_size)
    return np.fft.irfft(A * np.conj(G), fft_size)[:size]

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def lp_fw_optimize(a0, time_limit, solver, init_K=400, max_K=800, add_per=30, verbose=True):
    """LP-based Frank-Wolfe with golden section line search."""
    n = len(a0)
    y = normalize(a0).copy()
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()

    ac = fast_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[-init_K:])
    no_improve = 0
    stall_limit = 800

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        clist = sorted(active_set)
        if len(clist) > max_K:
            margins = np.array([M_curr - ac[k] for k in clist])
            keep = np.argsort(margins)[:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = sorted(active_set)

        K = len(clist)
        if K == 0:
            break

        g_dir = lp_direction_cvxpy(y, active_set, solver)
        if g_dir is None:
            # Fallback: use the uniform direction on active set complement
            if no_improve > stall_limit // 3:
                break
            # Try with smaller set
            if len(clist) > 100:
                active_set = set(int(clist[i]) for i in np.argsort(ac)[-200:])
                clist = sorted(active_set)
                continue
            break

        # Check violations and add to active set
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        violations = np.where(conv_ys > M_curr + 1e-10)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per]]
            for k in top_v:
                active_set.add(int(k))

        # Golden section line search
        lo = 0.0
        hi = 1.0
        c1 = lo + inv_phi * (hi - lo)
        c2 = hi - inv_phi * (hi - lo)

        def trial_obj(alpha):
            trial = np.maximum((1.0 - alpha) * y + alpha * g_dir, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                return 1e10
            trial = trial / ts
            return fast_autocorr(trial).max()

        f1 = trial_obj(c1)
        f2 = trial_obj(c2)
        for _ in range(18):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = lo + inv_phi * (hi - lo)
                f1 = trial_obj(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = hi - inv_phi * (hi - lo)
                f2 = trial_obj(c2)

        best_alpha = (lo + hi) / 2.0
        y_new = np.maximum((1.0 - best_alpha) * y + best_alpha * g_dir, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g_dir.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print("    LP-FW t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " |S|=" + str(len(active_set)))

    if verbose:
        print("    LP-FW done (" + str(round(time.monotonic() - t0, 1)) + "s) J=" + repr(2 * n * best_m))
    return best_a, 2.0 * n * best_m

def smoothed_max_grad(a0, time_limit, beta_start=5.0, beta_end=100.0, lr_base=0.5, momentum=0.85, verbose=True):
    """Gradient descent on log-sum-exp smoothed max of autocorrelation."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    m = np.zeros(n, dtype=np.float64)
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 3000:
            # Random restart
            rng = np.random.default_rng(t)
            noise = 0.02 * rng.standard_normal(n) * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            m = np.zeros(n)
            stall = 0

        b = fast_autocorr(a)
        b_max = b.max()
        b_shift = b - b_max
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / 3000.0)
        weights = np.exp(beta * b_shift)
        weights /= weights.sum()

        # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] (where 0<=k-j<n)
        # Use FFT for the convolution-like operation
        # c[j] = sum_k w[k] * a[k-j] = (w_flip * a)[2n-2-j] in full conv
        # Equivalently: cross-correlation
        W_f = np.fft.rfft(weights, n=fft_size)
        A_f = np.fft.rfft(a, n=fft_size)
        # First moment: sum_k w[k] (this is 1) -> contributes a[j] * 1 to grad
        # Second: sum_k w[k] * a[k-j] for valid k-j
        # cross_corr = ifft(fft(w) * conj(fft(a))) gives (w * a_rev)[shift]
        # Let me compute: c[j] = sum_k w[k] * a[k-j] = (w ⊛ a)[j]
        # Using: ifft(fft(w) * conj(fft(a))) = sum_k w[k+l] * a[k] ... hmm
        # Let me just do it with a direct FFT convolution:
        # c[j] = sum_{i=0}^{n-1} a[i] * w[j+i] (where j+i < 2n-1)
        # Define w_ext of length 2n-1 (the full autocorr has this length)
        # c[j] = sum_i a[i] * w[j+i] = (a * w_flip)[j + 2n - 2] where w_flip[i] = w[2n-2-i]
        w_flip = weights[::-1]
        ac_f = np.fft.rfft(a, n=fft_size)
        wf_f = np.fft.rfft(w_flip, n=fft_size)
        c_full = np.fft.irfft(ac_f * wf_f, fft_size)
        # c[j] = c_full[j + n - 1] (shifted)
        # Actually: (a * w_flip)[m] = sum_i a[i] * w_flip[m-i] = sum_i a[i] * w[2n-2-(m-i)]
        # We want sum_i a[i] * w[j+i], so 2n-2-(m-i) = j+i -> m = 2n-2-j
        # So c[j] = c_full[2n-2-j]
        c = c_full[2*n-2 - np.arange(n)]
        c = np.maximum(c, 0)
        
        grad = 2.0 * (c + a)  # a[j] from the "identity" part + c[j] from cross term
        
        # Project gradient onto sum=0 subspace (normalization constraint)
        grad = grad - grad.mean()
        
        # Momentum update
        lr = lr_base / (1.0 + 0.05 * np.sqrt(t))
        m = momentum * m + lr * grad
        a_new = a - m
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        val = obj_val(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Half step
            a_half = a - 0.3 * m
            a_half = np.maximum(a_half, 1e-15)
            a_half = a_half / a_half.sum()
            val_half = obj_val(a_half)
            if val_half < val:
                a = a_half
            stall += 1

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print("  SmMax t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val) + " beta=" + repr(round(beta, 1)))

    return best_a, best_val

def peak_deflation(a0, time_limit, n_targets=8, verbose=True):
    """Target specific peak lags and reduce them."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for rnd in range(300):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False

        for k in targets:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find low-activity destinations
            scores = np.zeros(n)
            for k2 in targets:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
            scores[src] = 1e10
            scores += 0.01 * a  # prefer low-mass destinations

            candidates = np.argsort(scores)[:40]
            best_m_trial = best_m
            best_frac = None
            best_dst = None

            for dst in candidates:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in [0.003, 0.008, 0.015, 0.03]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    Mt = fast_autocorr(a_t).max()
                    if Mt < best_m_trial - 1e-15:
                        best_m_trial = Mt
                        best_frac = frac
                        best_dst = dst

            if best_frac is not None:
                amt = a[src] * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                    best_m = fast_autocorr(a).max()
                    best_a = a.copy()
                    improved = True

        if not improved:
            for _ in range(8):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            improved = True
                            break

        if verbose and rnd % 100 == 0:
            print("    Deflate rnd=" + str(rnd) + " J=" + repr(2 * n * best_m))

    return best_a, 2.0 * n * best_m

def diverse_init(n, parent_seq=None):
    """Generate diverse initial candidates and return the best."""
    x = np.linspace(0.0, 1.0, n)
    best_a = normalize(np.sin(np.pi * x) ** 2)
    best_val = obj_val(best_a)

    candidates = [best_a]
    for f in [1, 2, 3]:
        candidates.append(normalize(np.sin(np.pi * x) ** (2 * f)))
    for c in [0.3, 0.5, 0.7]:
        for sg in [0.15, 0.25, 0.35]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))
    for alpha in [0.1, 0.2, 0.4]:
        cands = np.ones(n)
        m = max(1, int(alpha * n))
        taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
        cands[:m] = taper
