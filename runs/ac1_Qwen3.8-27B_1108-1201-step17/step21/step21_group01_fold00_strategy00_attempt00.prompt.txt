import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def build_lp_matrix(a, lags, n):

        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]

def batch_line_search(y, g_dir, alphas):

        trial_ac = np.fft.irfft(trial * trial, fft_size)[:size]

def spectral_equalize(a0, time_limit, beta_start=0.5, beta_end=0.05, verbose=True):
    """Frequency-domain peak reduction via spectral shaping."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    total_iters = 2000

    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive beta: start aggressive, become gentle
        progress = it / max(total_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress

        A = np.fft.fft(a)
        S = np.abs(A) ** 2
        S_sum = S.sum()
        if S_sum < 1e-15:
            break

        # Auto-correlation (circular approximation)
        r_circ = np.fft.ifft(S).real
        r_circ = np.roll(r_circ, n // 2)  # center the zero lag

        # Find top peaks
        peak_idx = np.argsort(r_circ)[::-1][:min(5, n)]
        peak_val = r_circ[peak_idx[0]]
        if peak_val < 1e-15:
            break

        # Build suppression mask
        freqs = np.fft.fftfreq(n)
        mask = np.ones(n)
        for pk in peak_idx[:3]:
            pk_int = int(pk)
            # Cosine of the peak lag at each frequency
            cos_term = np.cos(2.0 * np.pi * freqs * pk_int)
            # Suppress where cosine is positive (contributes to peak)
            mask *= (1.0 + beta * np.maximum(cos_term, 0.0))

        S_new = S / mask
        # Preserve total power (sum constraint)
        S_new_sum = S_new.sum()
        if S_new_sum > 1e-15:
            S_new = S_new * (S_sum / S_new_sum)

        # Update A preserving phase
        phase = np.angle(A)
        mag_new = np.sqrt(np.maximum(S_new, 0.0))
        A_new = mag_new * np.exp(1j * phase)

        # Project to non-negative
        a_new = np.fft.ifft(A_new).real
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum

        m_new = fast_autocorr(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller beta
            if beta > 0.01:
                beta *= 0.5
                mask = np.ones(n)
                pk = int(peak_idx[0])
                cos_term = np.cos(2.0 * np.pi * freqs * pk)
                mask *= (1.0 + beta * np.maximum(cos_term, 0.0))
                S_new = S / mask
                S_new_sum = S_new.sum()
                if S_new_sum > 1e-15:
                    S_new = S_new * (S_sum / S_new_sum)
                mag_new = np.sqrt(np.maximum(S_new, 0.0))
                A_new = mag_new * np.exp(1j * phase)
                a_new = np.fft.ifft(A_new).real
                a_new = np.maximum(a_new, 0.0)
                a_sum = a_new.sum()
                if a_sum > 1e-15:
                    a_new = a_new / a_sum
                    m_new = fast_autocorr(a_new).max()
                    if m_new < best_m:
                        best_m = m_new
                        best_a = a_new.copy()
                        a = a_new

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    Spectral it=" + str(it) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " beta=" + str(round(beta, 4)))

    return best_a, 2.0 * n * best_m

def grad_mult_nesterov(a0, time_limit, top_k=80, temp=50.0, c0=3.0, verbose=True):
    """Nesterov-accelerated multiplicative gradient descent."""
    n = len(a0)
    x = normalize(a0)
    best_a = x.copy()
    best_val = obj_val(x)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    x_prev = x.copy()
    v = np.zeros(n, dtype=np.float64)  # momentum
    moment = 0.9
    stall = 0
    cur_temp = temp

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 20000:
            break

        # Nesterov lookahead
        mu = (1.0 + np.sqrt(1.0 + 4.0 * (t - 1) ** 2)) / 2.0
        beta_t = (1.0 - 1.0 / mu)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[idx]
        cur_temp = temp * max(0.2, 1.0 - t / 80000.0)
        w = np.exp(cur_temp * (top_vals - top_vals[0]))
        w = w / w.sum()

        grad = np.zeros(n, dtype=np.float64)
        for wi, k in zip(w, idx):
            ki = int(k)
            i_s = max(0, ki - n + 1)
            i_e = min(n, ki + 1)
            idx_arr = np.arange(i_s, i_e)
            grad[idx_arr] += 2.0 * wi * y[ki - idx_arr]

        # Update momentum
        v = moment * v + grad
        eta_t = c0 / (np.sqrt(max(t, 1)) + 1.0)
        log_a = np.log(y) - eta_t * v
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x = a_new
        val = obj_val(x)
        if val < best_val - 1e-12:
            best_val = val
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 8000 and stall % 8000 == 0:
            noise_scale = 0.005 + 0.02 * (stall / 30000.0)
            noise = noise_scale * rng.standard_normal(n) * best_a
            x_r = best_a + noise
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            v = np.zeros(n, dtype=np.float64)
            stall = 0

        x_prev = x.copy()

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    GradNest t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" + repr(best_val) + " T=" + str(round(cur_temp, 1)))

    return best_a, best_val

def coord_polish(a0, time_limit, seed=42, verbose=True):

    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 3000
    tabu = {}
    tabu_size = 50

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        improved = False

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            top_srcs = idx[np.argsort(contrib)[::-1][:3]]
            for src in top_srcs:
                src = int(src)
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                scores = np.zeros(n)
                for k2 in top_k[:6]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                        scores[k2i - idx2] += a[idx2]
                scores[src] = 1e10

                candidates = np.argsort(scores)[:25]
                found = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src:
                        continue
                    key = (src, dst)
                    if key in tabu:
                        continue
                    for frac in [0.003, 0.007, 0.015, 0.03]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = fast_autocorr(a_t).max()
                        if Mt < best_m - 1e-15:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            if len(tabu) > tabu_size:
                                tabu.clear()
                            tabu[(dst, src)] = True
                            found = True
                            break
                    if found:
                        break

        if not improved:
            stall += 1
            if stall > 100:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = fast_autocorr(a).max()
                        if Mn < best_m:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 1000 == 0:
            elapsed = time.monotonic() - t0
            print("    Coord (" + str(round(elapsed, 1)) + "s) J=" + repr(2 * n * best_m) + " stall=" + str(stall))

    return best_a, 2.0 * n * best_m

def diverse_init(n, parent_seq=None):
    """Generate diverse initial candidates and return the best."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)

    candidates = []

    # Raised cosine
    candidates.append(normalize(0.5 * (1.0 + np.cos(2.0 * np.pi * x))))

    # Sin squared
    candidates.append(normalize(np.sin(np.pi * x) ** 2))

    # Gaussian centered
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sg in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(normalize(np.exp(-((x - c) ** 2) / (2 * sg * sg))))

    # Product of cosines (multi-lobe)
    for f1 in [1, 2, 3, 4, 6]:
        for f2 in [1, 2, 3]:
            cands = np.maximum(np.cos(2.0 * np.pi * f1 * x) * np.cos(2.0 * np.pi * f2 * x), 0.0)
            if cands.sum() > 1e-10:
                candidates.append(normalize(cands))

    # Half-wave rectified sine
    for f in [1, 2, 3, 4]:
        cands = np.maximum(np.sin(2.0 * np.pi * f * x), 0.0)
        if cands.sum() > 1e-10:
            candidates.append(normalize(cands))

    # Double Gaussian
    for c1 in [0.25, 0.35]:
        for c2 in [0.65, 0.75]:
            for sg in [0.08, 0.12, 0.18]:
                cands = np.exp(-((x - c1) ** 2) / (2 * sg * sg)) + np.exp(-((x - c2) ** 2) / (2 * sg * sg))
                candidates.append(normalize(cands))

    # Tukey windows
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        cands = np.ones(n)
        m = int(alpha * n)
        if m > 0 and m < n // 2:
            taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
            cands[:m] = taper
            cands[-m:] = taper[::-1]
        candidates.append(normalize(cands))

    # Power law tapered
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        cands = (x + 0.01) ** p * (1.0 - x + 0.01) ** p
        cands = np.maximum(cands, 1e-10)
