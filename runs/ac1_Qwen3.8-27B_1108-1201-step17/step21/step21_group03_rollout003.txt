    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT
        A = np.fft.rfft(a, n=fft_size)
        P = np.abs(A)**2

        P_mean = P[P > 0].mean() if (P > 0).any() else 1.0
        # Soft flattening: compress the dynamic range
        alpha = alpha_init * (alpha_end / alpha_init) ** (t / n_iters_max)

        excess = np.maximum(P - P_mean, 0) / (P_mean + 1e-15)

        # Reconstruct A with new magnitudes, same phases
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))

def fast_lp_direction(a, active_set, n_iters=50):

    n = len(a)
    clist = sorted(active_set)
    K = len(clist)
    if K == 0:
        return None
    
    # Start with uniform weights
    lam = np.ones(K) / K
    
    for _ in range(n_iters):
        # Compute h(i) = sum_k lam_k * a[k-i] for valid indices
        h = np.zeros(n)
        for jj, k in enumerate(clist):
            hi = int(k)
            i_start = max(0, hi - n + 1)
            i_end = min(n, hi + 1)
            if i_start < i_end:
                h[i_start:i_end] += lam[jj] * a[hi - np.arange(i_start, i_end)]
        
        # Find the minimum
        mu = h.min()
        # Active set for g: where h is close to mu
        threshold = mu + 0.01 * (h.max() - mu + 1e-15)
        support = h <= threshold
        
        # Update lam: increase weights for k that have high a[k-i] at low-h positions
        low_positions = h <= (mu + 0.05 * (h.max() - mu + 1e-15))
        gain = np.zeros(K)
        for jj, k in enumerate(clist):
            hi = int(k)
            i_start = max(0, hi - n + 1)
            i_end = min(n, hi + 1)
            if i_start < i_end:
                gain[jj] = a[hi - np.arange(i_start, i_end)][low_positions[i_start:i_end]].sum()
        
        # Multiplicative weight update
        lam = lam * (1.0 + 0.1 * gain / (gain.max() + 1e-15))
        lam = lam / lam.sum()
    
    # g is uniform over the support
    g = np.zeros(n)
    g[support] = 1.0
    gs = g.sum()
    if gs < 1e-10:
        return None
    return g / gs
