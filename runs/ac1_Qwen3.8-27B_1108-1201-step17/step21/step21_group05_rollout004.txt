def fast_autocorr(a):
    """Compute linear autocorrelation (convolution of a with a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def upsample_sequence(a, factor):
    """Upsample a sequence by factor using interpolation."""
    n = len(a)
    new_n = n * factor
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    return normalize(a_interp)

def fast_autocorr(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def obj_val(a):
    """Compute objective 2*n*max_autocorr for sum-normalized a."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def make_initial(n, window_type='hann'):
    """Create initial sequence with given window function."""
    x = np.linspace(0, 1, n)
    if window_type == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif window_type == 'hann':
        a = np.hann(n)
    elif window_type == 'flattop':
        a = (0.2156 - 0.4160 * np.cos(2 * np.pi * x)
             + 0.2782 * np.cos(4 * np.pi * x)
             - 0.0833 * np.cos(6 * np.pi * x)
             + 0.0069 * np.cos(8 * np.pi * x))
    elif window_type == 'gauss':
        sigma = 0.28
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif window_type == 'poly4':
        a = (4.0 * x * (1.0 - x)) ** 2
    else:
        a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-30:
        a /= s
    return a


def compute_subgradient(a, top_idx, top_vals, weights):
    """Compute weighted subgradient of max_k (a*a)[k]."""
    n = len(a)
    s = np.zeros(n, dtype=np.float64)
    k_sel = len(top_idx)
    for ii in range(k_sel):
        ki = int(top_idx[ii])
        w = weights[ii]
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start < j_end:
            indices = np.arange(j_start, j_end)
            s[indices] += 2.0 * w * a[ki - indices]
    return s


def mirror_descent(a0, time_limit, rng=None, n_top=60, lr0=0.04,
                   restart_interval=2500, noise_scale=0.004,
                   exp_weight=True, verbose=True, label=""):
    """
    Mirror descent (multiplicative weights) with:
    - Exponential peak weighting
    - Cosine-annealed learning rate
    - Backtracking line search
    - Periodic noisy restarts
    """
    if rng is None:
        rng = np.random.default_rng(123)
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    stall = 0
    total_iters = 50000

    for t in range(1, total_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Select top-k lags
        k_sel = min(n_top, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[top_idx]

        # Exponential weighting
        if exp_weight and top_vals[0] > 1e-20:
            spread = max(top_vals[0] * 0.15, 1e-20)
            weights = np.exp((top_vals - top_vals[0]) / spread)
            weights /= weights.sum()
        else:
            weights = np.ones(k_sel) / k_sel

        # Subgradient
        s = compute_subgradient(a, top_idx, top_vals, weights)

        # Cosine-annealed learning rate
        progress = min(t / total_iters, 1.0)
        lr_cos = 0.5 * (1.0 + np.cos(np.pi * progress))
        cur_lr = lr0 * lr_cos

        # Backtracking
        accepted = False
        max_bt = 10
        for bt in range(max_bt):
            log_a = np.log(a + 1e-30) - cur_lr * s
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-25)
            a_new = a_new / a_new.sum()
            val_new = 2.0 * n * fast_autocorr(a_new).max()
            if val_new <= cur_val * (1.0 + 1e-10) + 1e-18:
                a = a_new
                accepted = True
                break
            cur_lr *= 0.5

        if not accepted:
            a = a_new

        # Periodic restart
        if stall > restart_interval:
            noise = noise_scale * rng.standard_normal(n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-12)
            a = a_r / a_r.sum()
            stall = 0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    {} MD t={:5d} ({:.1f}s) J={:.8f} stall={}".format(
                label, t, elapsed, best_val, stall))

    return best_a, best_val

def upsample_sequence(a, factor):
    """Upsample via cubic-like interpolation and renormalize."""
    n = len(a)
    new_n = n * factor
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use spline-like interpolation for smoothness
    a_interp = np.interp(x_new, x_old, a, kind='cubic') if hasattr(np, 'interpolate') else None
    # Fallback: simple linear then smooth
    a_interp = np.interp(x_new, x_old, a)
    # Smooth with small box filter
    k = min(3, new_n // 10)
    if k > 1:
        kernel = np.ones(k) / k
        a_interp = np.convolve(a_interp, kernel, mode='same')
    return normalize(a_interp)


def dual_refine(a0, time_limit, verbose=True, label=""):
    """
    Dual-style refinement: for fixed a, compute the "best response" g
    that would reduce the peak. Inspired by the LP approach:
    At each step, identify the active lags, compute a direction that
    reduces the max, and take a step.
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    stall = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find active lags (within 1% of max)
        threshold = M_curr * 0.99
        active_idx = np.where(ac >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(ac))])
        k_sel = min(40, len(active_idx))
        top_idx = np.array(active_idx[:k_sel])

        # Compute the "dual" direction: for each active lag k,
        # the gradient of a^T Q_k a is 2 * Q_k * a (shifted a)
        # We want to move in a direction that reduces all active correlations
        # Direction: negative sum of shifted-a vectors (weighted by excess)
        direction = np.zeros(n, dtype=np.float64)
        for ki in top_idx:
            ki = int(ki)
            excess = max(0, ac[ki] - M_curr * 0.95)
            w = 1.0 + excess / M_curr
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                indices = np.arange(j_start, j_end)
                direction[indices] += w * a[ki - indices]

        # Normalize direction
        d_norm = np.linalg.norm(direction)
        if d_norm < 1e-20:
            direction = rng.standard_normal(n)
            direction = np.maximum(direction, 0)
            direction /= direction.sum() + 1e-30
        else:
            direction /= d_norm

        # Take a step: a_new = a - lr * direction, projected
        lr = 0.02 / (1.0 + 0.001 * t)
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / (a_new.sum() + 1e-30)

        val_new = obj_val(a_new)
        if val_new > cur_val * 1.01:
            lr *= 0.1
            a_new = a - lr * direction
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / (a_new.sum() + 1e-30)
            val_new = obj_val(a_new)

        a = a_new

        if stall > 3000 and stall % 500 == 0:
            noise = 0.003 * rng.standard_normal(n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-12)
            a = a_r / a_r.sum()
            stall = 0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print("    {} Dual t={:5d} ({:.1f}s) J={:.8f} stall={}".format(
                label, t, elapsed, best_val, stall))

    return best_a, best_val

    n0 = len(best_seq)

    # P1: Mirror descent on parent
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P1: Mirror Descent (n={}) ===".format(n0))
    t1 = min(remaining * 0.20, 340.0)
    if t1 > 30:
        a_out, v_out = mirror_descent(normalize(best_seq), t1, rng,
                                      n_top=60, lr0=0.04,
                                      restart_interval=2000, noise_scale=0.004,
                                      exp_weight=True, verbose=True, label="P1")
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P1 improved: {}".format(repr(best_score)))

    # P2: Coordinate refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P2: Coord Refine (n={}) ===".format(len(best_seq)))
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_out, v_out = coord_refine(normalize(best_seq), t2, n_targets=10,
                                     verbose=True, label="P2")
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P2 improved: {}".format(repr(best_score)))

    # P3: Dual refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P3: Dual Refine (n={}) ===".format(len(best_seq)))
    t3 = min(remaining * 0.10, 170.0)
    if t3 > 20:
        a_out, v_out = dual_refine(normalize(best_seq), t3, verbose=True, label="P3")
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P3 improved: {}".format(repr(best_score)))

    # P4: Upsampled + MD
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P4: Upsampled + MD (n={}) ===".format(len(best_seq)))
    t4 = min(remaining * 0.25, 425.0)
    if t4 > 60:
        n_up = len(best_seq) * 2
        if n_up <= 30000:
            a_up = upsample_sequence(best_seq, 2)
            print("  Upsampling to n={}".format(n_up))
            a_out, v_out = mirror_descent(a_up, t4, rng,
                                          n_top=80, lr0=0.03,
                                          restart_interval=2500, noise_scale=0.003,
                                          exp_weight=True, verbose=True, label="P4")
            sc = eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P4 improved: {} (n={})".format(repr(best_score), n_up))
            else:
                print("  P4 no improvement: {} (n={})".format(repr(sc), n_up))

    # P5: Fresh different-length inits
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P5: Fresh inits, different lengths ===".format(len(best_seq)))
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 40:
        # Try a few candidate lengths and windows
        candidates = []
        n_parent = len(best_seq)
        for n_try in [n_parent - 1000, n_parent + 1000, n_parent + 2000]:
            if n_try < 5000 or n_try > 30000:
                continue
            for wt in ['hann', 'sin2', 'flattop']:
                a_init = make_initial(n_try, wt)
                candidates.append(a_init)

        rng_sub = np.random.default_rng(999)
        n_cand = min(4, len(candidates))
        t_sub = t5 / n_cand
        for idx in range(n_cand):
            if time.monotonic() - t_global > time_budget - 60:
                break
            a_init = candidates[idx]
            a_out, v_out = mirror_descent(a_init, t_sub, rng_sub,
                                          n_top=50, lr0=0.05,
                                          restart_interval=1500, noise_scale=0.008,
                                          exp_weight=True, verbose=False,
                                          label="P5")
            sc = eval_safe(a_out)
            print("  P5 cand{} (n={}): {}".format(idx, len(a_out), repr(sc)))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print("  P5 improved: {}".format(repr(best_score)))

    # P6: Final polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== P6: Final polish (n={}) ===".format(len(best_seq)))
    t6 = max(remaining - 30.0, 0.0)
    if t6 > 20:
        a_out, v_out = mirror_descent(normalize(best_seq), t6, rng,
                                      n_top=40, lr0=0.012,
                                      restart_interval=5000, noise_scale=0.002,
                                      exp_weight=True, verbose=True, label="P6")
        sc = eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print("  P6 improved: {}".format(repr(best_score)))

    # Final validation
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print("\nFINAL: score={} n={} t={:.1f}s".format(
            repr(best_score), len(best_seq), elapsed))
    else:
        best_seq = make_initial(12000, 'hann')
        best_score = eval_safe(best_seq)
        print("FALLBACK: {}".format(repr(best_score)))

    # Ensure all values are valid (non-negative, finite)
    best_seq = np.maximum(np.asarray(best_seq, dtype=np.float64), 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    return [float(x) for x in best_seq]
