def fast_conv(a):
    """Compute full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Compute full linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

        return f.copy()
    s = f.sum()

    old_x = np.linspace(0.0, 1.0, old_n)

def peak_shave(a0, time_limit, n_peaks=50, eta_init=0.5, verbose=True):
    """
    Iterative peak shaving using spectral gradient computation.
    a0: initial normalized sequence (sum=1, non-negative)
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    eta = eta_init
    no_improve = 0
    stall_limit = 200
    
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        c = fast_conv(a)
        curr_val = c.max()
        
        if curr_val < best_val - 1e-14:
            best_val = curr_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top peaks
        K = min(n_peaks, len(c))
        peak_idx = np.argpartition(c, -K)[-K:]
        peak_val = c.max()
        
        # Weight by excess (exponential weighting)
        threshold = peak_val * 0.85
        excess = np.maximum(c - threshold, 0)
        total_excess = excess.sum()
        if total_excess < 1e-30:
            # All peaks are similar; use uniform weighting on top K
            weights = np.zeros(len(c))
            weights[peak_idx] = 1.0 / K
        else:
            weights = excess / total_excess
            weights[peak_idx] = np.maximum(weights[peak_idx], 1.0 / (K * 10))
            weights = weights / weights.sum()
        
        # Compute gradient via FFT
        grad = compute_grad_fft(a, weights)
        
        # Check for numerical issues
        if not np.all(np.isfinite(grad)):
            break
        
        # Adaptive step size
        max_grad = grad.max()
        if max_grad < 1e-30:
            break
        
        # Multiplicative update
        # Scale eta so that the maximum change is reasonable
        scale = peak_val / max_grad
        eta_eff = eta * scale / n  # normalized step
        
        # Apply update with damping
        log_a = np.log(np.maximum(a, 1e-30)) - eta_eff * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Damping: blend with old
        damping = 0.3
        a = (1 - damping) * a + damping * a_new
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        # Reduce eta over time
        eta = max(eta * 0.999, 0.01)
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print("  PeakShave it=%d (%.1fs) J=%.6f peaks=%d" % (it, elapsed, 2*n*best_val, K))
    
    return best_a, 2.0 * n * best_val
