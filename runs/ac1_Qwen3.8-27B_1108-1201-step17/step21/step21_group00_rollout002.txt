Uses subgradient descent with multiplicative updates, parametric initializations,
and peak-targeted local search.

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s

def obj_val_norm(a):

def subgradient_mult(a, top_idx, weights, n):
    """Compute weighted subgradient of max(conv) for top lags."""
    grad = np.zeros(n, dtype=np.float64)
    for ki, w in zip(top_idx, weights):
        ki = int(ki)
        j_min = max(0, ki - n + 1)
        j_max = min(ki, n - 1)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            ks = ki - js
            grad[js] += w * 2.0 * a[ks]
    return grad

def multiplicative_descent(a0, time_budget, lr_init=0.05, n_top=30, weight_temp=5.0):
    """Multiplicative gradient descent to minimize 2n*max(conv) for sum=1 sequences."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_max = fast_conv(a).max()
    t0 = time.monotonic()
    no_improve = 0

    mom_rate = 0.85
    
    for step in range(500000):
        if time.monotonic() - t0 > time_budget:
            break
        if no_improve > 5000:
            break
        
        conv = fast_conv(a)
        cur_max = conv.max()
        
        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top lags
        # Use argpartition for efficiency
        if n_top >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -n_top)[-n_top:]
        
        top_vals = conv[top_idx]
        max_val = top_vals.max()
        if max_val < 1e-15:
            break
        
        # Exponential weighting
        weights = np.exp(-weight_temp * (max_val - top_vals) / max_val)
        weights /= weights.sum()
        
        # Gradient
        grad = subgradient_mult(a, top_idx, weights, n)
        
        # Momentum
        momentum = mom_rate * momentum + (1 - mom_rate) * grad
        
        # Adaptive learning rate
        lr = lr_init / (1.0 + step * 0.005)

        a = a / a.sum()
