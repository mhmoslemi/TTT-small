    n = len(a)

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)

def normalize(a):

    s = a.sum()
    if s < 1e-15:

def interp(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    an = a / s
    if old_n == 1:
        return np.full(new_n, an[0])

    ai = np.maximum(ai, 0.0)
    ss = ai.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return ai / ss
