    old_n = len(f)
    if old_n == new_n:
        return f.copy()

    f_interp = np.interp(new_x, old_x, f_norm)

    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def lp_direction(a, active_set, solver, max_K=600):
    n = len(a)
    clist = sorted(active_set)
    K = len(clist)
    if K == 0:
        return None
    if K > max_K:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        margins = np.array([M_curr - ac[k] for k in clist])
        keep = np.argsort(margins)[:max_K]
        clist = [clist[i] for i in keep]

            gs = g_val.sum()

    except Exception:
        pass
    return None
