def normalize_sequence(seq):
    """Normalize the sequence: clip to non-negative and ensure sum >=0.01."""
    s = sum(seq)
    if s < 0.01:
        seq = [max(0, x) for x in seq]
        s = sum(seq)
        if s < 0.01:
            seq = [0.01/len(seq)] * len(seq)  # Make a uniform sequence if not enough mass
        else:
            seq = [x / s for x in seq]  # Normalize to sum==1
    else:
        seq = [max(0, x) for x in seq]
        s = sum(seq)
        seq = [x / s for x in seq]
    return seq, s

def make_geometric(n, ratio=0.97):
    """Generate a geometric sequence of count `n`."""
    a = np.power(ratio, np.arange(n))
    s = np.sum(a)
    a = a.tolist()
    return a, s

def make_bimodal(n, pos1=0.3, pos2=0.7, width=0.08):
    """Make a bimodal sequence around two positions."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) / width)**2) + np.exp(-((x - pos2) / width)**2)
    s = np.sum(a)
    a = a.tolist()
    return a, s

def make_exp_asym(n, alpha=0.5):
    """Make an exponential asymmetric sequence."""
    x = np.linspace(0, 1, n)
    a = (np.exp(-alpha*((x-0.2)**2) + np.exp(-alpha*((x-0.8)**2))
    return a.tolist(), np.sum(a)

def make_random_sparse(n, density=0.3, seed=42):
    """Make a random sparse-ish sequence."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    mask = rng.random(n) < density
    a[~mask] *= 0.1
    s = np.sum(a)
    return a.tolist(), s
