Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed mirror descent (soft max approximation)
2. Periodic Frank-Wolfe (conditional gradient) steps
3. Coordinate descent with pair swaps for fine-tuning
4. Multi-start with diverse initializations
5. Optional sequence length expansion

import numpy as np
import time

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    """Compute objective 2*n*max_autocorr (for normalized a)."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def logsumexp_obj(a, beta):
    """Compute log-sum-exp of autocorrelation scaled by beta."""

            g = compute_pressure(a, ac, beta)
        else:
            # For large beta, use top-k for efficiency
            k_sel = min(n_top_fallback, ac_len)
            top_idx = np.argsort(ac)[::-1][:k_sel]
            max_ac = ac[top_idx[0]]
            logits = beta * (ac[top_idx] - max_ac)
            logits = np.clip(logits, -50, 50)
            p_top = np.exp(logits)
            p_top_sum = p_top.sum()
            if p_top_sum < 1e-30:
                p_top = np.ones(k_sel) / k_sel
            else:
                p_top = p_top / p_top_sum
            
            g = np.zeros(n)
            for ii in range(k_sel):
                ki = int(top_idx[ii])
                w = p_top[ii] / ac_len
                j_start = max(0, ki - n + 1)
                j_end = min(n, ki + 1)
                if j_start < j_end:
                    indices = np.arange(j_start, j_end)
                    g[indices] += 2.0 * w * a[ki - indices]
        
        # Mirror descent update
        lr = lr0 / (1.0 + 0.002 * t)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-25)
        a_new = a_new / a_new.sum()
        
        # Check improvement (backtrack if needed)
        val_new = 2.0 * n * fast_autocorr(a_new).max()
        if val_new > cur_val * 1.001 + 1e-18:
            lr *= 0.5
            log_a = np.log(a + 1e-30) - lr * g
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-25)
            a_new = a_new / a_new.sum()
        
        a = a_new
        
        # Periodic restart
        if stall > restart_interval and stall % (restart_interval // 5) == 0:
            noise = noise_scale * rng.standard_normal(n) * best_a
            a_r = best_a + noise
            a_r = np.maximum(a_r, 1e-12)
            a = a_r / a_r.sum()
            stall = 0
            lr = lr0
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    MD t={:5d} ({:.1f}s) J={:.7f} stall={} beta={:.0f} lr={:.5f}".format(
                t, elapsed, best_val, stall, beta, lr))
    
    return best_a, best_val
