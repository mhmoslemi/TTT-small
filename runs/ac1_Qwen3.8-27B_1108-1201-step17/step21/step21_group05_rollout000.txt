"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parametric family optimization (cosine, Gaussian, superellipse, harmonic)
2. Symmetric enforcement
3. Remez-like peak equalization
4. Projected gradient descent with Nesterov momentum in log-domain
5. Multi-start with different initial shapes
"""

import numpy as np
import time
import math

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def parametric_cosine(n, p):
    """a[i] = cos^p(pi*i/(n-1)) clipped to be non-negative."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x / 2.0) ** p
    return a  # don't normalize yet

def parametric_gaussian(n, sigma):
    """a[i] = exp(-(i-center)^2/(2*sigma^2))."""
    center = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-(x - center) ** 2 / (2.0 * sigma ** 2))
    return a

def parametric_superellipse(n, p):
    """a[i] = (1 - t^2)^p where t = 2*i/(n-1) - 1."""
    x = np.linspace(0, 1, n)
    t = 2.0 * x - 1.0
    a = np.maximum(1.0 - t * t, 0.0) ** p
    return a

def parametric_harmonic(n, p1, p2=0.0, p3=0.0):
    """a[i] = 1 + p1*cos(2*pi*i/(n-1)) + p2*cos(4*pi*i/(n-1)) + ..."""
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    if p1 != 0:
        a += p1 * np.cos(2.0 * np.pi * x / (n - 1))
    if p2 != 0:
        a += p2 * np.cos(4.0 * np.pi * x / (n - 1))
    if p3 != 0:
        a += p3 * np.cos(6.0 * np.pi * x / (n - 1))
    return np.maximum(a, 0.0)

def optimize_parametric(n, param_func, init_params, bounds, time_limit=30.0, verbose=True):
    """
    Optimize parametric family parameters to minimize objective.
    param_func(n, *params) -> a (unnormalized)
    """
    from scipy.optimize import minimize
    
    t0 = time.monotonic()
    
    def wrapper(params):
        a = param_func(n, *params)
        a = normalize(a)
        return obj_val(a)
    
    result = minimize(wrapper, init_params, method='Nelder-Mead',
                      options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12,
                               'adaptive': True})
    
    best_params = result.x
    a = normalize(param_func(n, *best_params))
    val = obj_val(a)
    
    elapsed = time.monotonic() - t0
    if verbose:
        print("    Parametric: val={:.7f} params={} t={:.1f}s".format(val, best_params, elapsed))
    
    return a, val, best_params

def nesterov_md(a0, time_limit, lr0=0.05, momentum=0.9, n_top=50, verbose=True):
    """
    Nesterov-accelerated mirror descent in log-domain.
    """
    n = len(a0)
    a = normalize(a0).copy()
    v = np.zeros(n, dtype=np.float64)  # velocity for Nesterov
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    stall = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov look-ahead
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 1e-25)
        a_look = a_look / a_look.sum()
        
        ac = fast_autocorr(a_look)
        M_curr = ac.max()
        cur_val = 2.0 * n * M_curr
        
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Subgradient at top lags
        k_sel = min(n_top, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_sel]
        top_vals = ac[top_idx]
        
        s = np.zeros(n, dtype=np.float64)
        max_v = top_vals[0] if top_vals[0] > 0 else 1.0
        for ii in range(k_sel):
            ki = int(top_idx[ii])
            w = top_vals[ii] / max_v
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                indices = np.arange(j_start, j_end)
                s[indices] += 2.0 * w * a_look[ki - indices]
        
        # Log-domain mirror descent step
        lr = lr0 / (1.0 + 0.002 * t)
        log_a = np.log(a + 1e-30) - lr * s
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-25)
        a_new = a_new / a_new.sum()
        
        # Nesterov velocity update
        v = momentum * v + (a_new - a)
        a = a_new
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print("    NM t={:5d} ({:.1f}s) J={:.7f} stall={}".format(t, elapsed, best_val, stall))
    
    return best_a, best_val

def remez_equate(a0, time_limit, n_active=20, verbose=True):
    """
    Remez-like iteration: identify peaks, then solve to equalize them.
    """
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    for it in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        cur_val = 2.0 * n * M
        
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
        
        # Find active peaks (within 1% of max)
        threshold = M * 0.99
        active_lags = np.where(ac >= threshold)[0]
        
        if len(active_lags) == 0:
            active_lags = np.argsort(ac)[::-1][:n_active]
        
        # For each active lag k, the constraint is:
        # sum_i a[i]*a[k-i] = M (for valid i)
        # We want to perturb a to equalize these.
        
        # Compute gradient of max knot
        s = np.zeros(n, dtype=np.float64)
        for ki in active_lags:
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                indices = np.arange(j_start, j_end)
                s[indices] += a[ki - indices]
        
        s /= len(active_lags)
        
        # Projected gradient step in log-domain
        lr = 0.01 / (1.0 + 0.01 * it)
        log_a = np.log(a + 1e-30) - lr * s
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-25)
        a_new = a_new / a_new.sum()
        
        # Also try moving mass from peak contributors to quiet spots
        pressure = np.zeros(n, dtype=np.float64)
        for ki in active_lags[:10]:
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                indices = np.arange(j_start, j_end)
                pressure[indices] += a[ki - indices]
        
        # Add small noise to explore
        a_new += 0.001 * rng.standard_normal(n) * best_a
        a_new = np.maximum(a_new, 1e-25)
        a_new = a_new / a_new.sum()
        
        a = a_new
    
    return best_a, best_val

def symmetry_refine(a0, time_limit, verbose=True):
    """Enforce symmetry and optimize."""
    n = len(a0)
    a = normalize(a0).copy()
    a_sym = (a + a[::-1]) / 2.0
    a_sym = normalize(a_sym)
    
    val_orig = obj_val(a)
    val_sym = obj_val(a_sym)
    
    if verbose:
        print("    Symmetry: orig={:.7f} sym={:.7f}".format(val_orig, val_sym))
    
    base = a_sym if val_sym < val_orig else a
    return base, min(val_orig, val_sym)

def coarse_to_fine(a0, time_limit, verbose=True):
    """
    Multi-resolution: downsample, optimize, upsample, refine.
    """
    n = len(a0)
    a = normalize(a0).copy()
    
    # Downsample by factor 8
    factor = 8
    n_coarse = n // factor
    if n_coarse < 100:
        factor = max(1, n // 200)
        n_coarse = n // factor
    
    # Downsample by averaging
    a_padded = np.zeros(n_coarse * factor)
    a_padded[:n] = a
    a_coarse = a_padded.reshape(n_coarse, factor).mean(axis=1)
    a_coarse = normalize(a_coarse)
    
    # Optimize coarse
    if verbose:
        print("    Coarse n={} val={:.7f}".format(n_coarse, obj_val(a_coarse)))
    
    a_c_out, v_c = nesterov_md(a_coarse, time_limit * 0.3, lr0=0.05, n_top=30, verbose=False)
    
    # Upsample
    a_up = np.repeat(a_c_out, factor)
    if len(a_up) < n:
        a_up = np.concatenate([a_up, a_c_out[-(n - len(a_up)):]])
    a_up = a_up[:n]
    a_up = normalize(a_up)
    
    if verbose:
        print("    Upsample val={:.7f}".format(obj_val(a_up)))
    
    # Refine full
    a_out, v_out = nesterov_md(a_up, time_limit * 0.7, lr0=0.03, n_top=50, verbose=verbose)
    
    return a_out, v_out
