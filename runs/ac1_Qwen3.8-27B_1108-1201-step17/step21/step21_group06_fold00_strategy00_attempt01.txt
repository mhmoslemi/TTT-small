            best_sequence = enhanced_candidate.copy()
        
        # Global resampling strategy after 3 steps
        if cycles % 5 == 0:
            new_base = generate_random_sequence(len(best_sequence))
            new_base_score = full_evaluation(new_base)
            if new_base_score < init_score:
                init_score = new_base_score
                init_sequence = new_base.copy()
        
        cycles += 1
    
    # Report the best found
    return [float(x) if abs(x)<1000 else 1.0 for x in best_sequence]
