import numpy as np

import cvxpy as cp

_SOLVER_CACHE = None

def _get_solver():

def _fft_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fs)
    return np.fft.irfft(A * A, fs)[:size]


def _fft_crosscorr(a, b):
    """Compute np.convolve(a, b) via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fs)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), n=fs)
    return np.fft.irfft(A * B, fs)[:size]


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _obj(a):
    """Objective value assuming sum(a)=1: 2*n*max(autocorr)."""
    n = len(a)
    return 2.0 * n * _fft_autocorr(a).max()


def _eval_safe(a):
    try:
        a_list = [float(x) for x in a]

def _interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    fn = f / s
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    out = np.interp(nx, ox, fn)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return out / ss


def _build_lp_rows(a, lags, n):
    """Build sparse-ish rows for the LP: row k gives (a conv g)[k] coefficients on g."""
    K = len(lags)
    rows = []
    for k in lags:
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo > hi:
            rows.append(np.zeros(n))
        else:
            row = np.zeros(n)
            idx = np.arange(lo, hi + 1)
            row[idx] = a[ki - idx]
            rows.append(row)
    return np.array(rows, dtype=np.float64)


def _lp_direction(a, active_lags, solver, max_K=300):
    """Solve LP: min M s.t. (a*g)[k] <= M for k in active, sum(g)=1, g>=0."""
    n = len(a)
    clist = list(active_lags)
    if len(clist) == 0:
        return None
    if len(clist) > max_K:
        # Keep top-constraint violations
        ac = _fft_autocorr(a)
        margins = np.array([ac[k] for k in clist])
        order = np.argsort(margins)[::-1][:max_K]
        clist = [clist[i] for i in order]

    K = len(clist)
    coeffs = _build_lp_rows(a, clist, n)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            gv = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
            gs = gv.sum()
            if gs > 1e-12:
                return gv / gs
    except Exception:
        pass
    return None

    n = len(y)
    size = 2 * n - 1
    candidates = [0.0, 1.0, 0.5]

    # Add minimizers of top-quadratics
    denom = ac_g  # ac_g[k] > 0 for all k (autocorrelation)
    denom = np.maximum(denom, 1e-30)
    t_mins = (ac_y - xcorr_yg) / denom
    t_mins = np.clip(t_mins, 0.0, 1.0)

    # Pick top 20 by ac_y[k] (likely to be on the upper envelope)
    top_idx = np.argsort(ac_y)[::-1][:30]
    for k in top_idx:
        candidates.append(t_mins[k])

    candidates = np.array(candidates)
    candidates = np.unique(np.round(candidates, 6))

    # Evaluate f(t) at each candidate
    best_t = 0.0
    best_f = float('inf')
    for t in candidates:
        tq = ac_g * t * t + 2.0 * (xcorr_yg - ac_y) * t + ac_y
        fq = tq.max()
        if fq < best_f:
            best_f = fq
            best_t = t

    # Refine: add neighbors of best_t
    for dt in [0.01, 0.02, 0.05, 0.1]:
        for tt in [best_t - dt, best_t + dt]:
            tt = np.clip(tt, 0.0, 1.0)
            tq = ac_g * tt * tt + 2.0 * (xcorr_yg - ac_y) * tt + ac_y
            fq = tq.max()
            if fq < best_f:
                best_f = fq
                best_t = tt

    return float(best_t)


def _lp_fw(a0, time_limit, solver, init_K=200, max_K=400, add_per=20, verbose=True):
    """LP-based Frank-Wolfe with exact convex line search."""
    n = len(a0)
    y = _normalize(a0).copy()
    best_a = y.copy()
    best_m = _fft_autocorr(y).max()
    t0 = time.monotonic()

    ac = _fft_autocorr(y)
    active_set = set(int(k) for k in np.argsort(ac)[-init_K:])
    no_improve = 0
    stall_limit = 500
    iter_count = 0

    for t in range(1, 5000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        if no_improve > stall_limit:
            break

        ac = _fft_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-13:
            best_m = M_curr
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        clist = list(active_set)
        if len(clist) > max_K:
            margins = np.array([ac[k] for k in clist])
            keep = np.argsort(margins)[::-1][:max_K]
            active_set = set(int(clist[i]) for i in keep)
            clist = list(active_set)

        if len(clist) == 0:
            break

        g = _lp_direction(y, active_set, solver, max_K=max_K)
        if g is None:
            if len(clist) > 100:
                active_set = set(int(k) for k in np.argsort(_fft_autocorr(y))[-150:])
                continue
            if no_improve > stall_limit // 2:
                break
            continue

        # Compute cross-correlation y*g for line search
        xcorr_yg = _fft_crosscorr(y, g)
        ac_g = _fft_autocorr(g)

        # Exact convex line search
        best_alpha = _exact_fw_linesearch(y, g, ac, ac_g, xcorr_yg)

        y_new = (1.0 - best_alpha) * y + best_alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy > 1e-15:
            y = y_new / sy
        else:
            y = g.copy()
            y = y / y.sum()

        iter_count += 1

        # Update active set: add violated constraints
        if t % 3 == 0:
            conv_ys = _fft_crosscorr(y, g)  # Just for checking; quick approximation
            # Better: check actual autocorr of y
            ac_check = _fft_autocorr(y)
            M_new = ac_check.max()
            # Add top violations
            top_viol = np.argsort(ac_check)[::-1][:init_K]
            for k in top_viol:
                if ac_check[k] > M_new - 0.01 * M_new:
                    active_set.add(int(k))

        if verbose and t % 100 == 0:
            print("    LP-FW iter=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" +
                  repr(2 * n * best_m) + " |S|=" + str(len(active_set)) +
                  " alpha=" + repr(round(best_alpha, 4)))

    if verbose:
        print("    LP-FW done (" + str(round(time.monotonic() - t0, 1)) + "s) J=" +
              repr(2 * n * best_m) + " iters=" + str(iter_count))
    return best_a, 2.0 * n * best_m


def _adam_smoothed_max(a0, time_limit, beta0=10.0, beta1=200.0, lr=0.3,
                       beta_adam1=0.9, beta_adam2=0.999, eps=1e-8,
                       verbose=True):
    """Adam-optimized gradient descent on log-sum-exp smoothed max."""
    n = len(a0)
    a = _normalize(a0).copy()
    best_a = a.copy()
    best_val = _obj(a)
    t0 = time.monotonic()
    fs = 1 << (2 * n - 1 - 1).bit_length()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    stall = 0
    t_iter = 0

    for t in range(1, 10000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        if stall > 5000:
            # Perturb and reset
            rng = np.random.default_rng(t)
            noise = 0.01 * rng.standard_normal(n) * best_a
            a = best_a + noise
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            stall = 0

        ac = _fft_autocorr(a)
        ac_max = ac.max()
        shift = ac - ac_max
        beta = beta0 + (beta1 - beta0) * min(1.0, t / 2000.0)
        weights = np.exp(beta * shift)
        weights /= weights.sum()

        # Gradient: d/da_j sum_k w_k * ac_k
        # = 2 * sum_k w_k * a[k-j] (with appropriate bounds)
        # Use FFT: cross-correlation of weights and a
        w_flipped = weights[::-1]
        A_f = np.fft.rfft(a, n=fs)
        Wf = np.fft.rfft(w_flipped, n=fs)
        c_full = np.fft.irfft(A_f * Wf, fs)
        # c[j] = c_full[2n-2-j]
        c = c_full[2 * n - 2 - np.arange(n)]
        c = np.maximum(c, 0.0)
        grad = 2.0 * (c + a * weights.sum())
        # Actually: dac_k/da_j contributes a[k-j] to the gradient
        # The log-sum-exp gradient is: sum_k (w_k / 1) * dac_k/da_j
        # dac_k/da_j = a[k-j] + a[j-k] (when valid)
        # This equals 2 * (a conv w_reversed)[shift]... let me just use the FFT result
        # grad_j = 2 * sum_k w_k * a[k-j] for valid k-j, which is the cross-corr

        # Project onto sum=0 (normalization)
        grad = grad - grad.mean()

        t_iter += 1
        m_adam = beta_adam1 * m_adam + (1 - beta_adam1) * grad
        v_adam = beta_adam2 * v_adam + (1 - beta_adam2) * grad * grad
        m_hat = m_adam / (1 - beta_adam1 ** t_iter)
        v_hat = v_adam / (1 - beta_adam2 ** t_iter)
        lr_t = lr / (1.0 + 0.01 * np.sqrt(t_iter))
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        val = _obj(a_new)
        if val < best_val - 1e-13:
            best_val = val
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try smaller step
            a_half = a - 0.3 * lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a_half = np.maximum(a_half, 1e-15)
            a_half = a_half / a_half.sum()
            val_h = _obj(a_half)
            if val_h < _obj(a):
                a = a_half
            stall += 1

        if verbose and t % 500 == 0:
            print("  Adam-SmMax t=" + str(t) + " (" + str(round(elapsed, 1)) + "s) J=" +
                  repr(best_val) + " beta=" + repr(round(beta, 1)))

    return best_a, best_val


def _peak_deflation_quadratic(a0, time_limit, n_targets=5, verbose=True):
    """Move mass using quadratic approximation to reduce top peaks."""
    n = len(a0)
    a = _normalize(a0).copy()
    best_a = a.copy()
    best_m = _fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)

    for rnd in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        ac = _fft_autocorr(a)
        targets = np.argsort(ac)[::-1][:n_targets]
        improved = False

        for k in targets:
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            contrib = a[idx] * a[ki - idx]
            top_srcs = idx[np.argsort(contrib)[::-1][:3]]

            for src in top_srcs:
                src = int(src)
                if a[src] < 1e-10:
                    continue
                sv = a[src]

                # Score destinations: minimize contribution to all target peaks
                scores = np.zeros(n)
                for k2 in targets:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                scores[src] = 1e15
                scores += 0.005 * a

                cands = np.argsort(scores)[:50]
                found = False
                for dst in cands:
                    dst = int(dst)
                    if dst == src:
                        continue
                    # Quadratic: try several fractions
                    best_gain = 0
                    best_frac = None
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                        amt = sv * frac
                        if amt < 1e-14:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = _fft_autocorr(a_t).max()
                        gain = best_m - Mt
                        if gain > best_gain:
                            best_gain = gain
                            best_frac = frac
                            best_dst = dst

                    if best_frac is not None and best_gain > 1e-14:
                        amt = a[src] * best_frac
                        a[src] -= amt
                        a[best_dst] += amt
                        a = np.maximum(a, 0.0)
                        st = a.sum()
                        if st > 1e-15:
                            a = a / st
                            best_m = _fft_autocorr(a).max()
                            best_a = a.copy()
                            improved = True
                            found = True
                            break
                if found:
                    break

        if not improved:
            # Random perturbation
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.008)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = _fft_autocorr(a).max()
                        if Mn < best_m - 1e-14:
                            best_m = Mn
                            best_a = a.copy()
                            improved = True
                            break

        if verbose and rnd % 50 == 0:
            print("    Deflate rnd=" + str(rnd) + " J=" + repr(2 * n * best_m))

    return best_a, 2.0 * n * best_m


def _coord_descent_polish(a0, time_limit, seed=42, verbose=False):
    """Coordinate descent: try moving small amounts between pairs."""
    n = len(a0)
    a = _normalize(a0).copy()
    best_a = a.copy()
    best_m = _fft_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = _fft_autocorr(a)
        top_k = np.argsort(ac)[::-1][:15]
        improved = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            contrib = a[idx] * a[ki - idx]
            top_srcs = idx[np.argsort(contrib)[::-1][:2]]

            for src in top_srcs:
                src = int(src)
                if a[src] < 1e-10:
                    continue
                # Find best destination
                scores = np.zeros(n)
                for k2 in top_k[:5]:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        scores[idx2] += a[k2i - idx2]
                scores[src] = 1e15
                cands = np.argsort(scores)[:20]

                for dst in cands:
                    dst = int(dst)
                    if dst == src:
                        continue
                    found = False
                    for frac in [0.002, 0.005, 0.01, 0.02]:
                        amt = a[src] * frac
                        if amt < 1e-14:
                            continue
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        st = a_t.sum()
                        if st < 1e-15:
                            continue
                        a_t = a_t / st
                        Mt = _fft_autocorr(a_t).max()
                        if Mt < best_m - 1e-14:
                            a = a_t
                            best_m = Mt
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break

        if not improved:
            stall += 1
            if stall > 50:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.006)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    st = a.sum()
                    if st > 1e-15:
                        a = a / st
                        Mn = _fft_autocorr(a).max()
                        if Mn < best_m - 1e-14:
                            best_m = Mn
                            best_a = a.copy()
                            stall = 0

    return best_a, 2.0 * n * best_m


def _diverse_init(n, parent=None):
    """Generate diverse initial candidates."""
    x = np.linspace(0, 1, n)
    candidates = []

    # Sinusoidal
    for power in [1, 2, 3, 4]:
        candidates.append(np.sin(np.pi * x) ** power)
    # Gaussian bumps
    for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for sg in [0.1, 0.15, 0.2, 0.3]:
            candidates.append(np.exp(-((x - c) ** 2) / (2 * sg * sg)))
    # Cosine tapers
    for alpha in [0.1, 0.2, 0.3]:
        arr = np.ones(n)
        m = max(1, int(alpha * n))
        taper = 0.5 * (1 + np.cos(np.pi * np.arange(m) / m))
        arr[:m] = taper
        arr[-m:] = taper[::-1]
        candidates.append(arr)
    # Sinc-like
    candidates.append(np.sinc(x - 0.5) ** 2)
    candidates.append(np.maximum(np.sinc(x - 0.5), 0))

    best = None
    best_v = float('inf')
    for c in candidates:
        cn = _normalize(c)
        v = _obj(cn)
        if v < best_v:
            best_v = v
            best = cn

    if parent is not None:
        pn = _normalize(parent)
        pv = _obj(pn)
        if pv < best_v:
            best = pn
            best_v = pv
        # Blends
        for w in [0.7, 0.85, 0.95]:
            for c in candidates[:5]:
                blend = w * pn + (1 - w) * _normalize(c)
                bn = _normalize(blend)
                v = _obj(bn)
                if v < best_v:
                    best_v = v
                    best = bn

    return best, best_v
