    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def obj_val(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def lse_grad_step(a, beta, lr, momentum_vec, mu=0.9):
    """One step of momentum-based gradient descent on LSE(beta * autocorr)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Compute autocorrelation
    A = np.fft.rfft(a, n=fft_size)
    b = np.fft.irfft(A * A, fft_size)[:size]
    
    # LSE weights
    b_max = b.max()
    w = np.exp(beta * (b - b_max))
    w /= w.sum()
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
    # c[j] = sum_i a[i] * w[j+i] (where j+i < 2n-1)
    w_flip = w[::-1]
    wf_f = np.fft.rfft(w_flip, n=fft_size)
    ac_f = np.fft.rfft(a, n=fft_size)
    c_full = np.fft.irfft(ac_f * wf_f, fft_size)
    c = c_full[2*n-2 - np.arange(n)]
    
    grad = 2.0 * (c + a)
    grad = grad - grad.mean()  # Project onto sum=0
    
    # Momentum update
    momentum_vec[:] = mu * momentum_vec + lr * grad
    a_new = a - momentum_vec
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    
    return a_new

def lse_gradient_descent(a0, time_limit, beta_start=5.0, beta_end=80.0, lr_base=0.3, mu=0.9, verbose=False):
    """LSE gradient descent with momentum."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    m = np.zeros(n, dtype=np.float64)
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > 5000:
            break
        
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / 2000.0)
        lr = lr_base / (1.0 + 0.1 * np.sqrt(t))
        a_new = lse_grad_step(a, beta, lr, m, mu)
        val = obj_val(a_new)
        
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
        
        if verbose and t % 2000 == 0:
            print("  LSE t=" + str(t) + " (" + str(round(time.monotonic()-t0,1)) + "s) J=" + repr(best_val))
    
    return best_a, best_val
