def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def build_lp_matrix(a, lags, n):
    K = len(lags)
    coeffs = np.zeros((K, n), dtype=np.float64)
    for jj, k in enumerate(lags):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = a[k_int - idx]
    return coeffs

def lp_direction(a, active_lags, solver):
    """Solve LP to find FW direction: min t s.t. C@g <= t, sum(g)=1, g>=0"""
    n = len(a)
    K = len(active_lags)
    if K == 0:
        return None
    coeffs = build_lp_matrix(a, active_lags, n)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
            gs = g_val.sum()
            if gs > 1e-15:
                return g_val / gs
    except Exception:
        pass
    return None

def smoothed_grad_optimize(a0, time_limit, beta_start=10.0, beta_end=300.0, lr_base=0.3, verbose=True):
    """
    Gradient descent on log-sum-exp smoothed max of autocorrelation.
    Uses Nesterov momentum and adaptive step size.
    """
    n = len(a0)
    a = normalize(a0).copy()
    a_nest = a.copy()  # Nesterov extrapolation
    v = np.zeros(n, dtype=np.float64)  # momentum
    best_a = a.copy()
    best_val = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2*n - 2).bit_length()
    mu = 0.85  # momentum
    
    n_total = 10000  # target iterations
    total_beta_steps = max(1, int(time_limit / 0.05))  # rough
    
    for t in range(1, n_total + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        a_temp = a + mu * v
        
        b = fast_autocorr(a_temp)
        b_max = b.max()
        
        # Smoothed gradient
        beta = beta_start + (beta_end - beta_start) * min(1.0, (t - 1) / (n_total * 0.8))
        b_shift = b - b_max
        weights = np.exp(beta * b_shift)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.zeros_like(b)
            weights[b.argmax()] = 1.0
        else:
            weights /= wsum
        
        # Gradient of smooth max for a[j]:
        # grad[j] = 2 * sum_k weights[k] * (a_temp[k-j] if 0<=k-j<n else 0)
        # This is a cross-correlation: (weights_flip * a_temp)[shift]
        # c[j] = sum_k weights[k] * a_temp[k-j] for valid indices
        # Using FFT: c = ifft(fft(a_temp) * fft(weights_flip))
        # where weights_flip[i] = weights[2n-2-i]
        w_flip = weights[2*n-2::-1]  # reverse
        A_f = np.fft.rfft(a_temp, n=fft_size)
        Wf_f = np.fft.rfft(w_flip, n=fft_size)
        c_full = np.fft.irfft(A_f * Wf_f, fft_size)
        # c[j] = c_full[2n-2-j]
        c = c_full[2*n-2 - np.arange(n)]
        c = np.maximum(c, 0.0)
        
        grad = 2.0 * (c + a_temp)  # +a_temp accounts for the "identity" part
        grad = grad - grad.mean()  # project to sum=0
        
        # Adaptive step size (diminishing)
        lr = lr_base / (1.0 + 0.1 * np.sqrt(t))
        
        # Momentum update
        v = mu * v - lr * grad
        a_new = np.maximum(a + v, 1e-15)
        a_new = a_new / a_new.sum()
        
        val = 2.0 * n * fast_autocorr(a_new).max()
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
        a = a_new
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print("    SmGrad t=" + str(t) + " (" + str(round(elapsed,1)) + "s) obj=" + repr(best_val) + " beta=" + repr(round(beta,1)))
    
    return best_a, best_val

def coord_descent(a0, time_limit, seed=42, verbose=True):

def diverse_init(n, parent_seq=None):
    """Generate diverse initial candidates."""
    x = np.linspace(0.0, 1.0, n)
    candidates = []
    
    # Various window shapes
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    candidates.append(normalize(np.sin(np.pi * x) ** 4))
    candidates.append(normalize(np.sin(np.pi * x) ** 0.5))
    
    # Asymmetric shapes
    for alpha in [0.3, 0.4, 0.5, 0.6, 0.7]:
        candidates.append(normalize(np.exp(-((x - alpha) ** 2) / (0.05))))
    
    # Power laws
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(normalize(x ** p))
        candidates.append(normalize((1 - x) ** p))
    
    # Sine products
    for f1, f2 in [(1,2),(1,3),(2,3)]:
        candidates.append(normalize(np.abs(np.sin(np.pi*f1*x) * np.sin(np.pi*f2*x))))
    
    if parent_seq is not None:
        bi = normalize(parent_seq)
        candidates.append(bi)
        for w in [0.7, 0.85, 0.95]:
            for c in candidates[:4]:
                candidates.append(normalize(w * bi + (1-w) * c))
    
    best_a = candidates[0]
    best_val = obj_val(best_a)
    for c in candidates[1:]:
        v = obj_val(c)
        if v < best_val:
            best_val = v
            best_a = c.copy()
    
    return best_a, best_val
