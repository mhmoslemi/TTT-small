Minimize  2*n*max(conv(a,a)) / (sum(a))^2  for a non-negative step function a.

Strategy (different, more LP/convex-first than prior art):
  * View the full autocorrelation C = a * a (length 2n-1) as a linear map L(a).
  * The score is scale-invariant; fix ||a||_1 = 1 and minimize max_k C_k, i.e.
    minimize t s.t. a>=0, sum a = 1, L(a) <= t.  This is the exact convex form.
  * Drive this with an accelerated Frank-Wolfe loop where the linear-minimisation
    subproblem keeps ALL (or the most binding) active convolution constraints,
    alternated with a mirror (log-domain) multiplicative update that weaves in
    strong asymmetry (palindrome-breaking), plus a frequency-domain "spectrum
    flatten" polish and a mass-transport coordinate polish.
  * Asymmetric / ramp-style initialisations are the key new ingredient: a flat
    sequence is a saddle of max-autocorrelation (score=2), but deliberately

def _eval_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float("inf")


def _nonneg(a):
    return np.maximum(np.asarray(a, dtype=np.float64), 0.0)


def _norm1(a):
    a = _nonneg(a)
    s = a.sum()
    if s < 1e-18:
        return np.ones_like(a) / max(1, len(a))
    return a / s


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _row_of_L(y, k, n):
    # row k of the convolution linear map L: (L a)[k] = sum_i a[i] a[k-i]
    # for a fixed y the constraint is row_k . a <= t  with row_k[i]= y[k-i] if valid
    js = max(0, k - (n - 1))
    je = min(n - 1, k)
    row = np.zeros(n)
    if js <= je:
        row[js:je + 1] = y[k - np.arange(js, je + 1)]
    return row


def _interp_unit(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = _norm1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    g = np.interp(new_x, old_x, f)
    return _norm1(g)

def _init_ramp(n, slope=1.0, mirror=False, floor=0.02):
    x = np.arange(n, dtype=np.float64)
    a = slope * x + floor
    if mirror:
        a = a * (1.0 - x / n) + floor
    return _norm1(a)


def _init_exp(n, rate=0.8, mirror=False, floor=0.0):
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = np.exp(-rate * x) + floor
    if mirror:

def _amd(a0, time_limit, top_k=60, beta=4.0, c0=4.0, restart_every=5000, seed=42):
    """Accelerated mirror (log-domain) descent with a soft multi-peak subgradient."""
    n = len(a0)
    a = _norm1(a0)
    best = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.time()
    rng = np.random.default_rng(seed)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_k, cur_b, gam = top_k, beta, c0

    while True:
        if time.time() - t0 > time_limit:
            break
        beta_t = (stall * 1e-3) * 0.0  # placeholder to keep numpy import light
        y = x + 0.0 * (x - x_prev)
        sy = y.sum()
        if sy < 1e-18:
            y = x.copy()
        else:
            y = y / sy
        ac = _fast_autocorr(y)
        M = ac.max()
        if M < 1e-18:
            break
        # subgradient: push down the top lag(s) via their mirror row
        kmax = min(cur_k, len(ac))
        top = np.argsort(ac)[::-1][:kmax]
        w = (np.maximum(ac[top], 1e-30) ** cur_b)
        w = w / w.sum()
        g = np.zeros(n)
        for wi, k in zip(w, top):
            ki = int(k)
            js = max(0, ki - (n - 1))
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            g[idx] += wi * 2.0 * y[ki - idx]
        eta = gam / (np.sqrt(stall) + 1.0)
        loga = np.log(np.maximum(y, 1e-15)) - eta * g
        loga -= loga.max()
        a_new = np.exp(loga)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        x_prev = x
        x = a_new
        Mn = _fast_autocorr(x).max()
        if Mn < best_m - 1e-15:
            best_m = Mn
            best = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > restart_every:
            pert = rng.normal(0, 0.05, n) * best
            xr = _norm1(best + pert)
            x = xr.copy()
            x_prev = xr.copy()
            stall = 0
            cur_k = min(cur_k + 10, 140)
            cur_b = max(cur_b - 0.3, 2.0)
    return best, best_m


def _lp_fw(a0, time_limit, solver, topK=300, force_asym=True):
    """Accelerated Frank-Wolfe minimising max L(a) over the simplex a>=0,sum=1.
    The LMO keeps all current binding convolution constraints -> exact solution
    of the convex subproblem.  A palindromic perturbation is injected when the
    iterate becomes too symmetric (the score-2 saddle)."""
    n = len(a0)
    a = _norm1(a0)
    best = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.time()
    y = a.copy()
    x = a.copy()
    prev = a.copy()

    for t in range(1, 200000):
        if time.time() - t0 > time_limit:
            break
        ac = _fast_autocorr(y)
        M = ac.max()
        if M < 1e-16:
            break
        # anti-symmetry kick (escape the flat/palindromic saddle at score=2)
        if force_asym:
            pal = np.maximum(M - ac[:n].max(), 0.0)  # central lag vs left-half
            if M > 0.6 * best_m and np.abs(np.corrcoef(y, y[::-1])[0, 1]) > 0.999:
                y = _norm1(y * (1.0 + 0.03 * np.random.default_rng(t).uniform(-1, 1, n)))
                ac = _fast_autocorr(y)
                M = ac.max()

        margins = M - ac
        K = min(topK, 2 * n - 1)
        sel = np.argsort(margins)[:K]

        import cvxpy as cp
        b = cp.Variable(n, nonneg=True)
        cons = [np.sum(b) == 1]
        for k in sel:
            cons.append(_row_of_L(y, int(k), n) @ b <= M)
        prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ("optimal", "optimal_inaccurate"):
            continue
        v = _nonneg(np.asarray(b.value, dtype=np.float64))
        sv = v.sum()
        if sv < 1e-15:
            continue
        g_dir = v / sv

        alpha = 2.0 / (t + 2.0)
        y_new = _norm1((1.0 - alpha) * y + alpha * g_dir)
        if t == 1:
            x_new = g_dir.copy()
        else:
            bt = (t - 1.0) / (t + 2.0)
            x_new = _norm1(np.maximum(y_new + bt * (y_new - prev), 0.0))
        prev = y
        y = y_new
        x = x_new

        my = _fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best = y.copy()
        mx = _fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best = x.copy()
    return best, best_m


def _freq_polish(a0, time_limit, iters=2000, eta0=0.6):
    n = len(a0)
    a = _norm1(a0)
    best = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.time()
    fft_size = 1 << (2 * n - 1).bit_length()
    for it in range(iters):
        if time.time() - t0 > time_limit:
            break
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        Pm = P.mean()
        if Pm < 1e-30:
            break
        eta = eta0 * (1.0 + 0.5 * it / max(1, iters))
        Pt = Pm * (P / Pm) ** (1.0 - eta)
        Pt = np.maximum(Pt, 1e-30)
        an = np.fft.irfft(np.sqrt(Pt) * np.exp(1j * np.angle(A)), fft_size)[:n]
        an = _norm1(an)
        Mn = _fast_autocorr(an).max()
        if Mn < best_m - 1e-15:
            best_m = Mn
            best = an.copy()
            a = an
        else:
            a = _norm1(0.5 * a + 0.5 * an)
    return best, best_m

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()
            print("*** NEW BEST [%s] = %.6f  n=%d  t=%.0fs" % (label, sc, len(a), time.time() - t_global))

    # parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p = p / p.sum()
            upd(p, "parent")
    except Exception:
        pass

    if best is None:
        upd(_init_cos_asym(4200, mode=1, skew=0.6), "init")

    n0 = len(best)
    print("start n=%d score=%.6f" % (n0, best_score))

    # ---- Phase 1: diverse asymmetric AMD short passes ----
    rem = T - (time.time() - t_global)
    t1 = min(rem * 0.12, 180.0)
    if t1 > 30:
        cands = [
            _init_ramp(n0, 1.0, False),
            _init_ramp(n0, 1.0, True),
            _init_exp(n0, 0.8, False),
            _init_exp(n0, 0.8, True),
            _init_cos_asym(n0, 1, 0.6),
            _init_cos_asym(n0, 2, 0.4),
            _init_bell_asym(n0, 0.4, 0.18),
            _init_bell_asym(n0, 0.55, 0.12),
            _init_sinc_asym(n0, 6.0),
            _init_random_smooth(n0, 7, 8),
            _norm1(best),
        ]
        tp = max(8.0, t1 / len(cands))
        for idx, c in enumerate(cands):
            if time.time() - t_global > T - 60:
                break
            ta = min(tp, T - 60 - (time.time() - t_global))
            if ta < 6:
                break
            out, _ = _amd(c, ta, top_k=60, beta=4.0, c0=4.0, restart_every=4000, seed=100 + idx)
            upd(out, "amd%d" % idx)

    # ---- Phase 2: BIG LP-FW on current best (workhorse) ----
    rem = T - (time.time() - t_global)
    t2 = min(rem * 0.40, 600.0)
    if t2 > 30 and len(best) <= 5000:
        out, _ = _lp_fw(_norm1(best), t2, solver, topK=400, force_asym=True)
        upd(out, "lp_big")

    # ---- Phase 3: freq polish + AMD refit ----
    rem = T - (time.time() - t_global)
    if rem > 40:
        out, _ = _freq_polish(_norm1(best), min(rem * 0.12, 120.0), iters=1500, eta0=0.6)
        upd(out, "freq")
        rem = T - (time.time() - t_global)
        if rem > 20:
            out, _ = _amd(_norm1(best), min(rem * 0.10, 100.0), top_k=80, beta=3.5, c0=3.5, restart_every=5000, seed=9)
            upd(out, "amd_polish")

    # ---- Phase 4: larger LP-FW with more constraints ----
    rem = T - (time.time() - t_global)
    if rem > 30 and len(best) <= 5000:
        out, _ = _lp_fw(_norm1(best), min(rem * 0.30, 400.0), solver, topK=600, force_asym=True)
        upd(out, "lp_more")

    # ---- Phase 5: transport polish ----
    rem = T - (time.time() - t_global)
    if rem > 20:
        out, _ = _transport(_norm1(best), min(rem * 0.20, 150.0), seed=777)
        upd(out, "transport")

    # ---- Phase 6: try a longer length (asymmetry dividend grows with n) ----
    rem = T - (time.time() - t_global)
    if rem > 60 and len(best) < 6000:
        nl = min(len(best) + 800, 6000)
        a_long = _interp_unit(best, nl)
        upd(a_long, "interp_long")
        out, _ = _amd(a_long, min(rem * 0.25, 150.0), top_k=60, beta=4.0, c0=3.5, restart_every=5000, seed=4321)
        upd(out, "amd_long")
        rem = T - (time.time() - t_global)
        if rem > 30 and nl <= 5000:
            out, _ = _lp_fw(_norm1(best if len(best) <= 5000 else a_long), min(rem * 0.5, 200.0), solver, topK=500, force_asym=True)
            upd(out, "lp_long")

    # ---- Phase 7: final tight LP ----
    rem = T - (time.time() - t_global)
    if rem > 20 and len(best) <= 5000:
        out, _ = _lp_fw(_norm1(best), min(rem * 0.8, 150.0), solver, topK=800, force_asym=True)
        upd(out, "lp_final")

    # final validation
    if best is None:
        best = _init_cos_asym(4200, 1, 0.6)
    fs = _eval_safe(best)
    if np.isfinite(fs):
        best_score = fs
    best = np.clip(best, 0.0, 1000.0)
    print("FINAL score=%.6f n=%d time=%.0fs" % (best_score, len(best), time.time() - t_global))
    return [float(v) for v in best]
