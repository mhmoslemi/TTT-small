def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)

    w_sum = w.sum()
    if w_sum < 1e-30:

        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * y[j_idx]

        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_sorted_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        if use_backtrack:
            eta = gamma / (np.sqrt(t) + 1.0)
            base_m = fast_autocorr(y).max()
            accepted = False
            for _ in range(8):
                log_a = np.log(y) - eta * g
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    break
                a_new = a_new / s_new
                m_trial = fast_autocorr(a_new).max()
                if m_trial < base_m - 1e-18:
                    x_prev = x
                    x = a_new
                    accepted = True
                    break
                else:
                    eta *= 0.4
            if not accepted:
                log_a = np.log(y) - eta * g
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    a_new = y.copy()
                else:
                    a_new = a_new / s_new
                x_prev = x
                x = a_new
        else:
            eta = gamma / (np.sqrt(t) + 1.0)
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 100)
            cur_beta = max(cur_beta - 0.5, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=150,
                   margin_frac=1e-3, use_min_formulation=True):
    """LP Frank-Wolfe with adaptive constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    it = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select active constraints (peaks with small margin)
        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = min(len(active_idx), top_K)
        active_idx = active_idx[:K]

        # Build constraint matrix: T_sel[k, i] = y[active_idx[k] - i] for valid i
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)

        if use_min_formulation:
            # Minimize the weighted peak autocorrelation with new b
            t_var = cp.Variable()
            constraints = [cp.sum(b_var) == 1]
            for row in range(K):
                constraints.append(T_sel[row] @ b_var <= t_var + M_curr * 0.01)
            # Use a soft formulation: minimize t + regularization
            obj = cp.Minimize(t_var + 1e-4 * cp.sum_squares(b_var))
        else:
            # Maximize sum(b) subject to peak constraints (original formulation)
            constraints = [T_sel @ b_var <= M_curr]
            obj = cp.Maximize(cp.sum(b_var))

        prob = cp.Problem(obj, constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Frank-Wolfe step with Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            it += 1
            margin_frac = max(margin_frac * 0.92, 1e-6)

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            it += 1

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}, improvements={it}")

    return best_a, best_m


def spectral_peak_crush(a0, time_limit, verbose=True, rng_seed=77):
    """Use spectral analysis to identify and crush autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (2 * n - 1).bit_length()

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        # Identify second peak
        ac_no_peak = ac.copy()
        ac_no_peak[peak_k] = 0
        second_k = int(np.argmax(ac_no_peak))

        # Try moving mass from high-contribution positions
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 1
            continue

        # Find top contributing positions
        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

            # Try multiple fractions and random destinations
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                n_trials = 40
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpectralCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    SpectralCrush final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def freq_perturb(a, strength=0.01, rng=None):
    """Perturb in frequency domain for diversity."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def multi_start_explore(base_seq, time_limit, n_starts=4):
    """Try multiple parametric perturbations of the base sequence."""
    n = len(base_seq)
    best_a = base_seq.copy()
    best_m = fast_autocorr(base_seq).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for s in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break

        # Create diverse starting points
        if s == 0:
            start = make_parametric(n, 'sine')
        elif s == 1:
            start = make_parametric(n, 'cosine')
        elif s == 2:
            start = make_parametric(n, 'triangular')
        else:
            start = freq_perturb(base_seq, strength=0.05, rng=rng)

        # Blend with base
        alpha = 0.3 + 0.2 * s
        mixed = (1 - alpha) * norm_sum1(base_seq) + alpha * norm_sum1(start)
        mixed = np.maximum(mixed, 0)
        mixed = mixed / mixed.sum()

        # Quick AMD
        a_out, m_out = amd_optimize(mixed, time_limit / n_starts, top_k=40,
                                     beta=5.0, c0=3.0, restart_every=2000,
                                     verbose=False, use_backtrack=True)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: AMD warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD warm-up (n={n0}) ===")
    t1 = min(remaining * 0.12, 200.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t1, top_k=50, beta=4.0, c0=3.5,
                                restart_every=3000, verbose=True, use_backtrack=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd_warm")

    # === Phase 2: LP-FW main ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW main (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=200,
                                   margin_frac=2e-3, use_min_formulation=False)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 3: Spectral peak crushing ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral peak-crush (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_peak_crush(a_in, t3, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "spectral1")

    # === Phase 4: Second LP with tighter constraints ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW tight (n={n_best}) ===")
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K=300,
                                   margin_frac=5e-4, use_min_formulation=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 5: AMD refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: AMD refine (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t5, top_k=80, beta=3.0, c0=2.5,
                                restart_every=2000, verbose=False, use_backtrack=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd_refine")

    # === Phase 6: Multi-start exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-start (n={len(best_seq)}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multi_start_explore(a_in, t6, n_starts=3)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "multistart")

    # === Phase 7: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = sorted(set([n_best + 200, n_best + 500, n_best + 1000]))
    longer_lengths = [l for l in longer_lengths if l <= 5500 and remaining > 150]

    for new_n in longer_lengths[:1]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n=== Phase 7: Extend to n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.30, 200.0)
        if t7a > 30:
            a_out, _ = amd_optimize(a_long, t7a, top_k=60, beta=4.0, c0=3.0,
                                     restart_every=3000, verbose=False, use_backtrack=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 4500:
            t7b = min(remaining * 0.25, 150.0)
            if t7b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=True, top_K=200,
                                           margin_frac=2e-3, use_min_formulation=False)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

    # === Phase 8: Final spectral crush ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final spectral crush (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.25, 150.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = spectral_peak_crush(a_in, t8, verbose=True, rng_seed=999)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "spectral_final")

    # === Phase 9: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final LP polish (n={n_best}) ===")
    t9 = min(remaining * 0.20, 100.0)
    if t9 > 20 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t9, solver, verbose=True, top_K=400,
                                   margin_frac=1e-4, use_min_formulation=False)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_polish")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Safety margin
    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining")

    return [float(x) for x in best_seq]
