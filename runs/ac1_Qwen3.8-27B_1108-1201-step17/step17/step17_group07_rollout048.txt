```python
def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * (np.arange(1, n+1)) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

import numpy as np
import cvxpy as cp
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def grad_lag(a, k):
    """Compute df_k/da for a given lag k. O(n)."""
    n = len(a)
    g = np.zeros(n)
    k = int(k)
    # Term 1: a[k-j] for j where 0 <= k-j < n
    j1_s = max(0, k - n + 1)
    j1_e = min(n, k + 1)
    if j1_s < j1_e:
        idx = np.arange(j1_s, j1_e)
        g[idx] += a[k - idx]
    # Term 2: a[j-k] for j where 0 <= j-k < n
    j2_s = max(0, k)
    j2_e = min(n, k + n)
    if j2_s < j2_e:
        idx = np.arange(j2_s, j2_e)
        g[idx] += a[idx - k]
    return g

def compute_J_subgrad(a, active_lags, M):
    """Compute subgradient of J = 2n*M/S^2 at point a."""
    n = len(a)
    S = a.sum()
    g = np.zeros(n)
    for k in active_lags:
        g += grad_lag(a, k)
    g /= len(active_lags)
    # dJ/da[j] = 2n * (g[j]/S^2 - 2M*a[j]/S^3)
    g = 2.0 * n * (g / (S * S) - 2.0 * M * a / (S ** 3))
    return g

def nesterov_optimize(a0, time_limit=600.0, verbose=False, top_lags=10, c0=2.0):
    """Nesterov accelerated subgradient method with simplex projection."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s
    
    best_a = a.copy()
    best_b = fast_conv(a)
    best_M = best_b.max()
    best_J = 2.0 * n * best_M
    
    x = a.copy()
    x_prev = a.copy()
    t0 = time.monotonic()
    c = c0
    stall = 0
    max_stall = 2000
    rng = np.random.default_rng(42 + n)
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-30)
        y = y / y.sum()
        
        # Compute convolution and find active lags
        b_y = fast_conv(y)
        M_y = b_y.max()
        # Get top lags
        n_lags = len(b_y)
        k_top = min(top_lags, n_lags)
        active_lags = np.argsort(b_y)[::-1][:k_top]
        
        # Compute subgradient
        g = compute_J_subgrad(y, active_lags, M_y)
        
        # Step size
        eta = c / (np.sqrt(t) + 1.0)
        
        # Gradient step
        a_new = y - eta * g
        a_new = simplex_project(a_new)
        
        x_prev = x
        x = a_new
        
        # Check improvement (every 10 iterations to save time)
        if t % 10 == 0:
            b_new = fast_conv(x)
            M_new = b_new.max()
            J_new = 2.0 * n * M_new
            if J_new < best_J - 1e-15:
                best_J = J_new
                best_a = x.copy()
                stall = 0
            else:
                stall += 1
        
        if stall > max_stall:
            stall = 0
            max_stall = max(200, int(max_stall * 0.7))
            # Restart with perturbation
            strategy = int(rng.integers(3))
            if strategy == 0:
                x = best_a * (1 + rng.normal(0, 0.01, n))
            elif strategy == 1:
                x = best_a.copy()
                idx = rng.choice(n, size=min(n//5, 300), replace=False)
                x[idx] *= rng.uniform(0.6, 1.4, len(idx))
            else:
                x = best_a * (1 + rng.normal(0, 0.005, n))
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            c = max(0.3, c * 0.7)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov t={t} ({elapsed:.1f}s): J={best_J:.8f}, stall={stall}, c={c:.3f}")
    
    return best_a
