def fast_conv(a):
    """FFT-based full convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

def compute_subgrad_full(y, ac, top_k):
    """Compute subgradient using top-k lags of autocorrelation."""
    n = len(y)
    k_max = min(top_k, len(ac))
    if k_max == len(ac):
        top_idx = np.arange(len(ac))
    else:
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
    top_idx = np.sort(top_idx)
    
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= k_max
    return g

def lp_remez_single(a, max_cons=50, r=0.1, solver=None):
    """Single LP Remez step with active constraint selection."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Select active constraints: lags where b is close to max
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    # Trust region: constrain movement from current point
    if r < 1.0:
        constraints.append(a_var >= a * (1.0 - r))
        constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy(), 0.0
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), 0.0
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), 0.0
    return a_new / s, M_var.value if M_var.value is not None else 0.0

            break
        
        # For the peak lag, b[k] = sum_i a[i]*a[k-i]
        # To reduce it, we can decrease elements that have high product with their partner
        products = np.zeros(n)

            continue
        
        # Try to move mass from max_contrib_idx to a random other position
        if a[max_contrib_idx] < 1e-12:
            continue
        
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
            amt = a[max_contrib_idx] * frac
            dest = int(rng.integers(n))
            a_trial = a.copy()

def interpolate_sequence(f, new_n):
    """Resample sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    """Generate initial sequence of given type and length."""
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sech':
        a = 1.0 / np.cosh(4.0 * (x - 0.5))
    elif kind == 'sqrt_bell':
        a = np.sqrt(np.maximum(1.0 - (2.0 * x - 1.0) ** 2, 0.0))
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def spectral_init(n, seed=0):
    """Initialize with near-flat DFT magnitude (low PAPR sequence idea)."""
    rng = np.random.default_rng(seed)
    # Start with random phases, take inverse FFT, rectify
    for attempt in range(20):
        phases = rng.uniform(0, 2 * np.pi, n)
        F = np.exp(1j * phases)
        a = np.fft.irfft(F, n)
        a = np.real(np.fft.irfft(F, n * 2)[:n])
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            # Check if this is better than flat
            m = fast_conv(a).max()
            m_flat = 1.0 / n  # flat gives max conv = 1/n when sum=1
            if m < m_flat * 1.1:

    return make_init(n, 'flat')

        a_out, _ = remez_lp_method(norm_sum1(best_seq), t6a, solver=solver, r_init=0.04, verbose=False, seed=77)
        try_update(a_out, f"Remez_polish_{len(best_seq)}")
        
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        t6b = min(t6b, remaining - 5)
        if t6b > 10:
            a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t6b, top_k=10, c0=1.0, verbose=False, seed=777)
            try_update(a_out, f"NMD_polish_{len(best_seq)}")
    
    # === Phase 7: Final CD + peak-targeted (7% time) ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t7 = min(remaining * 0.07, 120.0)
    if t7 > 15:
        print(f"\n=== Phase 7: Final peak+CD (budget={t7:.0f}s) ===")
        t7a = t7 * 0.5
        t7b = t7 * 0.5
        a_out, _ = peak_targeted_search(norm_sum1(best_seq), t7a, verbose=False, seed=333)
        try_update(a_out, f"Peak_final_{len(best_seq)}")
        
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        t7b = min(t7b, remaining - 5)
        if t7b > 10:
            a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t7b, verbose=False, seed=444, group_size=2)
            try_update(a_out, f"CD_final_{len(best_seq)}")
    
    # === Phase 8: Try spectral init at best length (remaining time) ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining > 40:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Spectral exploration n={n_best} (budget={remaining:.0f}s) ===")
        a_spec = spectral_init(n_best, seed=1)
        # Quick NMD from spectral init
        t8 = min(remaining * 0.6, 100.0)
        if t8 > 15:
            a_out, _ = nesterov_mirror_descent(a_spec, t8, top_k=20, c0=3.0, verbose=False, seed=888)
            try_update(a_out, f"Spectral_n{n_best}")
        
        # Also try a different length with spectral init
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        if remaining > 30:
            n_alt = n_best + 150
            a_spec2 = spectral_init(n_alt, seed=2)
            t8b = min(remaining * 0.5, 60.0)
            if t8b > 10:
                a_out, _ = nesterov_mirror_descent(a_spec2, t8b, top_k=20, c0=3.0, verbose=False, seed=999)
                try_update(a_out, f"Spectral_n{n_alt}")
    
    # === Final: One more Remez squeeze on absolute best ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining > 10 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== Final squeeze: Remez n={n_best} (budget={remaining-5:.0f}s) ===")
        a_out, _ = remez_lp_method(norm_sum1(best_seq), remaining - 5, solver=solver, r_init=0.02, verbose=False, seed=1234)
        try_update(a_out, f"Remez_last_{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
