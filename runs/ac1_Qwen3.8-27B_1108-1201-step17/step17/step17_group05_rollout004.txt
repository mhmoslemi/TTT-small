import numpy as np

def fast_conv(a, b=None):
    """Compute full linear convolution using FFT."""
    if b is None:
        b = a
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

        # Restart if stalled too long
        if stall > restart_every:
            stall = 0
            # Perturb around best
            scale = max(0.01, 0.03 * (1.0 - float(stall) / (2.0 * restart_every)))
            pert = rng.normal(0, scale, n) * best
            x_r = best + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r /= x_r.sum()
            x = x_r
            x_prev = x_r.copy()

    return best, best_m


def peak_shave(a0, time_limit, seed=0):
    """
    Targeted peak shaving: directly reduce contributions to peak lags
    by moving mass from high-product pairs to low-sensitivity positions.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    improvements = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()
        if M < best_m:
            best_m = M
            best = a.copy()

        # Find top peak lags
        top_lags = np.argsort(ac)[::-1][:8]
        found_improve = False

        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            if found_improve:
                break

            # Find contributing pairs
            js_valid = np.arange(max(0, ki - n + 1), min(n - 1, ki) + 1)
            if len(js_valid) == 0:
                continue
            contrib = a[js_valid] * a[ki - js_valid]
            order = np.argsort(contrib)[::-1]

            # Try to reduce the top contributing pairs
            for oi in order[:6]:
                if time.monotonic() - t0 > time_limit:
                    break
                j = js_valid[oi]
                ji = int(j)
                partner = ki - ji

                src_val = max(a[ji], a[partner])
                if src_val < 1e-10:
                    continue

                # Determine which to reduce (reduce the larger one)
                if a[ji] >= a[partner]:
                    src = ji
                else:
                    src = int(partner)

                if a[src] < 1e-10:
                    continue

                # Find best destination: position with low "sensitivity"
                # Sensitivity of position d = sum over peak lags of a[lag - d]
                sens = np.zeros(n)
                for k2 in top_lags[:4]:
                    k2i = int(k2)
                    for d in range(max(0, k2i - n + 1), min(n - 1, k2i) + 1):
                        sens[d] += a[k2i - d]

                # Best dest: low sensitivity, far from src, has some mass to absorb
                mask = (sens < np.median(sens)) & (np.abs(np.arange(n) - src) > n // 10)
                candidates = np.where(mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(sens < np.median(sens))[0]
                if len(candidates) == 0:
                    candidates = np.arange(n)

                # Score: prefer low sensitivity and far distance
                dist = np.abs(candidates.astype(float) - src)
                dist[dist < 1] = 1.0
                scores = sens[candidates] / (dist / n)
                best_dst = int(candidates[np.argmin(scores)])

                if best_dst == src:
                    continue

                # Try several transfer amounts
                for frac in [0.01, 0.02, 0.005, 0.03, 0.015, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-14 or a[src] - amt < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial /= s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best = a.copy()
                        found_improve = True
                        improvements += 1
                        break

        if not found_improve:
            # Random perturbation to escape
            i = int(rng.integers(n))
            j2 = int(rng.integers(n))
            if i != j2 and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j2] += amt
                a = np.maximum(a, 0.0)
                a /= a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best = a.copy()
                    improvements += 1
            # If no improvement for a while, do a larger random jump
            elif improvements < 100 and rng.random() < 0.1:
                # Blend with a random smooth function
                x = np.linspace(0, 1, n)
                rand_f = 1.0 + 0.2 * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 6.28))
                a_new = (1 - 0.1) * a + 0.1 * rand_f / rand_f.sum()
                a = norm_sum1(a_new)

    return best, best_m


def coordinate_polish(a0, time_limit, seed=0):
    """
    Focused coordinate descent: cycle through positions and try pairwise swaps.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        # Identify which positions are "dangerous" (contribute to peaks)
        ac = fast_autocorr(a)
        top_lags = np.argsort(ac)[::-1][:5]

        # Compute sensitivity for each position
        danger = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            for m in range(max(0, ki - n + 1), min(n - 1, ki) + 1):
                danger[m] += a[ki - m]

        # Sort positions by danger (most dangerous first)
        order = np.argsort(danger)[::-1]

        improved = False
        for pos_idx in order[:min(n, 200)]:
            if time.monotonic() - t0 > time_limit or improved:
                break
            src = int(pos_idx)
            if a[src] < 1e-10:
                continue

            # Try moving a small amount to a random low-danger position
            safe_positions = np.argsort(danger)[:n // 3]
            dst = int(safe_positions[int(rng.integers(len(safe_positions)))])
            if dst == src:
                continue

            for frac in [0.01, 0.02, 0.005]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial /= a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1

    return best, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = norm_sum1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_smooth_init(n, seed=0):
    """Create a smooth non-negative initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in [1, 2, 3, 4, 5]:
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-phase optimization combining mirror descent,
    peak shaving, coordinate polish, and length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_smooth_init(n_base, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Mirror descent (main optimization) ===
    print(f"\n=== Phase 1: Mirror Descent (High Beta) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 600.0)

    if t_phase1 > 60:
        # Run from parent with high beta
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t_phase1 * 0.6, beta_init=80.0, beta_end=20.0,
                                  c0=5.0, seed=42, frac=0.10)
        try_update(a_out, "md_main")

    # === Phase 2: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Peak Shaving ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shave(a_in, t2, seed=7)
        try_update(a_out, "peak_shave")

    # === Phase 3: Second MD round (different params) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: MD Round 2 (Low Beta, More Exploration) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t3, beta_init=20.0, beta_end=8.0,
                                  c0=3.0, seed=123, frac=0.25, use_nesterov=True)
        try_update(a_out, "md_r2")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coordinate Polish ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, seed=99)
        try_update(a_out, "coord")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length Exploration (base n={n_best}) ===")
    candidate_lengths = [n_best - 200, n_best + 200, n_best - 400, n_best + 400]
    candidate_lengths = [l for l in candidate_lengths if 1000 <= l <= 5000]
    # Remove the base length
    candidate_lengths = [l for l in candidate_lengths if l != n_best]
    candidate_lengths = candidate_lengths[:3]

    for new_n in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t_md = min(remaining * 0.3, 150.0)
        if t_md > 20:
            a_out, _ = mirror_descent(a_long, t_md, beta_init=60.0, beta_end=15.0,
                                      c0=4.0, seed=555, frac=0.15)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_ps = min(remaining * 0.2, 80.0)
            if t_ps > 15:
                a_out, _ = peak_shave(norm_sum1(best_seq), t_ps, seed=77)
                # Interpolate back to new_n and polish
                a_interp = interpolate_sequence(a_out, new_n)
                a_out2, _ = peak_shave(a_interp, t_ps, seed=88)
                try_update(a_out2, f"ps_n{new_n}")

    # === Phase 6: Final combined polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Polish (n={n_best}) ===")
    if remaining > 40:
        a_in = norm_sum1(best_seq)

        # Quick MD
        t_md = min(remaining * 0.3, 120.0)
        if t_md > 15:
            a_out, _ = mirror_descent(a_in, t_md, beta_init=100.0, beta_end=30.0,
                                      c0=2.0, seed=2024, frac=0.05)
            try_update(a_out, "md_final")

        # Peak shave
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_ps = min(remaining * 0.4, 100.0)
            if t_ps > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = peak_shave(a_in, t_ps, seed=314)
                try_update(a_out, "ps_final")

        # Coordinate
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_cd = min(remaining * 0.4, 80.0)
            if t_cd > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t_cd, seed=271)
                try_update(a_out, "cd_final")

    # === Phase 7: Aggressive restart from scratch ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        print(f"\n=== Phase 7: Fresh Restart ===")
        # Try a different initialization structure
        a_fresh = make_smooth_init(n0, seed=999)
        # Apply mirroring for symmetry
        a_fresh = 0.5 * (a_fresh + a_fresh[::-1])
        a_fresh = norm_sum1(a_fresh)

        t_fresh = min(remaining * 0.5, 200.0)
        a_out, _ = mirror_descent(a_fresh, t_fresh, beta_init=40.0, beta_end=10.0,
                                  c0=4.0, seed=7777, frac=0.20)
        try_update(a_out, "fresh")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_smooth_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
