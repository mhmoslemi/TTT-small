def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n, parent, parent_score):
    """Try multiple parametric families, blend with parent, return best."""
    t0 = time.monotonic()
    i = np.arange(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    best_a = parent.copy()
    best_val = parent_score

    def try_update(a, label):
        nonlocal best_a, best_val
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a = a / s
        ac = fast_autocorr(a)
        val = 2.0 * n * ac.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"    Init [{label}]: score={val:.6f}")

    # Geometric sequences (asymmetric - key insight)
    for r in [0.3, 0.4, 0.5, 0.6, 0.65, 0.7, 0.75, 0.8, 0.85]:
        a = r ** i
        try_update(a, f"geom_{r}")
        a = r ** (n - 1 - i)
        try_update(a, f"geomr_{r}")

    # Power law decay
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (n - i) ** p
        try_update(a, f"pow_{p}")
        a = (i + 1) ** (-p)
        try_update(a, f"invpow_{p}")

    # Exponential decay
    for lam in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = np.exp(-lam * x)
        try_update(a, f"exp_{lam}")
        a = np.exp(lam * x)
        try_update(a, f"expR_{lam}")

    # Blend parent with geometric
    if parent_score < 2.0:
        parent_norm = parent / parent.sum()
        for r in [0.5, 0.6, 0.7, 0.8]:
            for w in [0.3, 0.5, 0.7]:
                geom = r ** i
                geom = geom / geom.sum()
                a = w * parent_norm + (1 - w) * geom
                try_update(a, f"blend_w{w}_r{r}")

    # Smoothstep / sigmoid shapes
    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20]:
            a = 1.0 / (1.0 + np.exp(-s * (c - x)))
            try_update(a, f"sig_c{c}_s{s}")

    elapsed = time.monotonic() - t0
    print(f"  Parametric init done ({elapsed:.0f}s): best score={best_val:.6f}")
    return best_a, best_val


def build_conv_matrix(y, k_list, n):
    """Build sparse constraint matrix for LP. Row k: (y*g)_k = sum_j y_{k-j} g_j"""
    K = len(k_list)
    rows = []
    cols = []
    data = []
    for ii, k in enumerate(k_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            vals = y[k_int - idx]
            mask = vals != 0
            rows.extend([ii] * mask.sum())
            cols.extend(idx[mask].tolist())
            data.extend(vals[mask].tolist())
    from scipy.sparse import csr_matrix
    mat = csr_matrix((data, (rows, cols)), shape=(K, n))
    return mat


def lp_frank_wolfe(a0, time_limit, solver, initial_K=400, max_K=3000,
                    add_per_iter=40, verbose=True):
    """Improved LP-based Frank-Wolfe with cutting planes."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    ac_y = fast_autocorr(y)
    init_idx = np.argsort(ac_y)[::-1][:initial_K]
    constraint_set = list(init_idx)
    constraint_set_set = set(int(k) for k in constraint_set)

    iter_count = 0
    lp_fail_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()

        # Manage constraint set size
        if len(constraint_set) > max_K:
            margins = np.array([M_curr - ac[int(k)] for k in constraint_set])
            keep = np.argsort(margins)[:max_K]
            constraint_set = [constraint_set[i] for i in keep]
            constraint_set_set = set(int(k) for k in constraint_set)

        K = len(constraint_set)
        if K == 0:
            break

        # Build sparse matrix
        mat = build_conv_matrix(y, constraint_set, n)

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, mat @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            if lp_fail_count > 5:
                # Reduce constraints and retry
                constraint_set = constraint_set[:initial_K]
                constraint_set_set = set(int(k) for k in constraint_set)
                lp_fail_count = 0
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check violations across all lags
        size_full = 2 * n - 1
        fft_size = 1 << (size_full - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size_full]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                k_int = int(k)
                if k_int not in constraint_set_set:
                    constraint_set.append(k_int)
                    constraint_set_set.add(k_int)

        # Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): score={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): score={J:.6f}, iters={iter_count}")
    return best_a, best_m


def compute_subgradient(a, ac, top_k, temperature):
    """Compute smooth approximation of subgradient of max(ac)."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    ac_max = ac.max()
    # Softmax weights
    w = np.exp((ac[top_idx] - ac_max) / (temperature + 1e-15))
    w = w / w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad

        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, cur_top_k, cur_temp)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 25, 150)
            cur_temp = max(cur_temp * 0.7, 0.001)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    PG restart t={t} ({elapsed:.0f}s): score={J:.6f}, tk={cur_top_k}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, rng_seed, verbose):
    """Aggressive coordinate descent: move mass between positions to reduce max conv."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 600
    iter_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iter_count += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()

        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        # For each top lag, try moving mass from high-contribution positions
        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score candidate destinations: low contribution to all top lags
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]

            # Exclude positions that are in top lags
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Try top few candidates
            cand_scores = scores[candidates]
            order = np.argsort(cand_scores)[:5]
            found = False
            for dst_idx in order[:3]:
                dst = candidates[dst_idx]
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.15]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                while j == i:
                    j = int(rng.integers(n))
                if a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): score={J:.6f}, stall={stall}, iter={iter_count}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        i = np.arange(n_base, dtype=np.float64)
        best_seq = 0.7 ** i
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric init ===
    print(f"\n=== Phase 1: Parametric init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 60.0)
    if t1 > 10:
        a_param, val_param = parametric_init(n0, best_seq, 2.0 * n0 * fast_autocorr(best_seq).max())
        check_and_update(a_param, "parametric")

    # === Phase 2: Main LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.40, 600.0)
    if t2 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, initial_K=400, max_K=3500,
                                   add_per_iter=40, verbose=True)
        check_and_update(a_out, f"lp_main_n{n_best}")

    # === Phase 3: Projected gradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Projected gradient (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = projected_gradient(a_in, t3, top_k=80, c0=4.0, verbose=False)
        check_and_update(a_out, f"pg_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate descent (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t4, rng_seed=42, verbose=True)
        check_and_update(a_out, f"coord_n{n_best}")

    # === Phase 5: LP-FW again on polished result ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW round 2 (n={n_best}) ===")
    if remaining > 60 and n_best <= 6000:
        t5 = min(remaining * 0.30, 350.0)
        if t5 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t5, solver, initial_K=500, max_K=4000,
                                       add_per_iter=50, verbose=True)
            check_and_update(a_out, f"lp_r2_n{n_best}")

    # === Phase 6: Try slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 120:
        for delta in [300, -300, 600]:
            new_n = n_best + delta
            if new_n < 100 or new_n > 6000:
                continue
            if remaining < 80:
                break
            print(f"\n=== Phase 6: Try n={new_n} ===")
            a_long = interpolate_sequence(best_seq, new_n)

            t6a = min(remaining * 0.30, 150.0)
            if t6a > 20:
                a_out, _ = projected_gradient(a_long, t6a, top_k=70, c0=3.5, verbose=False)
                check_and_update(a_out, f"pg_n{new_n}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 60 and new_n <= 5500:
                t6b = min(remaining * 0.30, 150.0)
                if t6b > 20:
                    a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t6b, solver,
                                               initial_K=300, max_K=2500,
                                               add_per_iter=30, verbose=False)
                    check_and_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coordinate polish (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.50, 120.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent(a_in, t7, rng_seed=999, verbose=True)
            check_and_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        i = np.arange(n_fb, dtype=np.float64)
        best_seq = 0.7 ** i
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
