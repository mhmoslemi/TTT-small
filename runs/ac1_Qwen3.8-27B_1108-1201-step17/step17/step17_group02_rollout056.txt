- For a[k-i] = 0: no constraint (the value is just conv_old[k], which must be <= M)

And: delta >= -old_val

From the quadratic: delta^2 + 2*a[i]*delta + (conv_old[2i] - M) <= 0
The roots are: delta = -a[i] ± sqrt(a[i]^2 - conv_old[2i] + M)
For this to have real roots: M >= conv_old[2i] - a[i]^2 (always true since conv_old[2i] >= a[i]^2)
So: -a[i] - sqrt(a[i]^2 - conv_old[2i] + M) <= delta <= -a[i] + sqrt(a[i]^2 - conv_old[2i] + M)

Let lower_quad(M) = -a[i] - sqrt(a[i]^2 - conv_old[2i] + M)
Let upper_quad(M) = -a[i] + sqrt(a[i]^2 - conv_old[2i] + M)

Feasibility condition: max(-old_val, lower_quad(M)) <= min(upper(M), upper_quad(M))

We binary search on M to find the minimum feasible M, then pick delta = max(-old_val, lower_quad(M)).

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with logsumexp continuation and Nesterov momentum
2. Exact 1D coordinate descent for polishing (solves minimax per coordinate)
3. Multi-start with diverse initializations
4. Length exploration
"""

import numpy as np
import time
import sys

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)

def compute_smooth_grad(a, p):
    """Compute grad[i] = 2 * sum_j a[j] * p[j+i]"""
    n = len(a)
    m = len(p)
    N = 1 << (n + m - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    r = np.fft.irfft(np.conj(A) * P, N)[:n]
    return 2.0 * r

def smooth_optimize(a0, time_limit, verbose=True, seed=42):
    """Smooth gradient descent with continuation and Nesterov momentum."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_conv_max = fast_conv(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Continuation phases
    beta_schedule = [5, 10, 20, 40, 80, 150, 300, 500, 800, 1200, 1800, 2500, 3500, 5000]
    
    for phase, beta in enumerate(beta_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov setup
        a_curr = a.copy()
        a_prev = a.copy()
        momentum = 0.0
        step = 1.0 / (2.0 * beta)
        
        n_iters = 3000
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov extrapolation
            if it > 0:
                momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
                y = a_curr + ((momentum - 1.0) / (momentum + 1.0)) * (a_curr - a_prev)
                y = np.maximum(y, 1e-15)
                sy = y.sum()
                if sy < 1e-15:
                    y = a_curr.copy()
                else:
                    y = y / sy
            else:
                y = a_curr.copy()
            
            # Compute smooth objective and gradient at y
            z = fast_conv(y)
            z_max = z.max()
            exp_z = np.exp(beta * (z - z_max))
            Z = exp_z.sum()
            p = exp_z / Z
            f_smooth = (z_max + np.log(Z)) / beta
            
            grad = compute_smooth_grad(y, p)
            
            # Backtracking line search
            a_try = project_simplex(y - step * grad)
            z_try = fast_conv(a_try)
            z_max_try = z_max  # approximate
            
            # Check if step is good (using smooth objective)
            exp_try = np.exp(beta * (z_try - z_max))
            f_try = (z_max + np.log(exp_try.sum())) / beta
            
            backtrack = 0
            while f_try > f_smooth + 0.5 * step * np.dot(grad, a_try - y) and backtrack < 20:
                step *= 0.5
                a_try = project_simplex(y - step * grad)
                z_try = fast_conv(a_try)
                exp_try = np.exp(beta * (z_try - z_max))
                f_try = (z_max + np.log(exp_try.sum())) / beta
                backtrack += 1
            
            a_prev = a_curr
            a_curr = a_try
            
            # Check true objective
            true_max = z_try.max() if backtrack <= 0 else fast_conv(a_curr).max()
            if it % 50 == 0 or true_max < best_conv_max:
                true_max = fast_conv(a_curr).max()
            if true_max < best_conv_max:
                best_conv_max = true_max
                best_a = a_curr.copy()
        
        a = a_curr.copy()
        
        if verbose:

            print(f"  Smooth phase {phase}: beta={beta}, J={J:.6f}, t={elapsed:.0f}s")
    
    return best_a, best_conv_max

    best_conv_max = fast_conv(a).max()

        conv = fast_conv(a)
        M = conv.max()
        stall_this_round = 0
        
        # Greedy: pick coordinates in order of potential improvement
        # Or cyclic
        order = np.arange(n)
        rng.shuffle(order)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if stall >= max_stall:
                break
            
            # Solve 1D minimax for coordinate i
            old_val = a[i]
            if old_val < 1e-15:
                # Try increasing from 0 (take from another coordinate)
                continue
            
            # conv_new[k] = conv[k] + 2*delta*a[k-i] + delta^2 * I(k=2i)
            # Minimize max_k conv_new[k] over delta >= -old_val
            
            # Binary search on M_new
            lo = -old_val  # minimum delta
            # Upper bound on M: current max
            M_lo = best_conv_max * 0.9  # we want to beat current best
            M_hi = M  # current max (definitely feasible with delta=0)
            
            # Actually, let's find the optimal delta for this coordinate
            # Feasibility: exists delta >= -old_val such that:
            #   conv[k] + 2*delta*a[k-i] <= M_new for all k (k-i valid, k != 2i)
            #   conv[2i] + 2*delta*a[i] + delta^2 <= M_new
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with logsumexp continuation (main optimizer)
2. LP-based Frank-Wolfe for refinement
3. Exact 1D coordinate descent for final polishing
4. Multi-start with diverse initializations
"""

import numpy as np
import time

# Global variables to track best
_global_best_seq = None
_global_best_score = float('inf')
_global_t0 = 0

def fast_conv(a):
    """Compute full linear autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def project_simplex(v):
    """Project vector v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    r = rho_indices[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(a, p):
    """
    Compute gradient of smooth objective.
    grad[i] = 2 * sum_j a[j] * p[j+i]
    where p has length 2n-1 (the softmax weights).
    """
    n = len(a)
    m = len(p)
    # Need N >= n + m - 1 for linear cross-correlation
    N = 1 << (n + m - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    r = np.fft.irfft(np.conj(A) * P, N)[:n]
    return 2.0 * r

def smooth_optimize(a0, time_limit, verbose=True, seed=42):
    """
    Smooth gradient descent with logsumexp continuation and Nesterov momentum.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_conv_max = fast_conv(a).max()
    
    t0 = time.monotonic()
    
    # Beta schedule: gradually increase smoothness parameter
    betas = [5, 10, 20, 40, 80, 150, 300, 500, 800, 1200, 2000, 3000, 5000, 8000]
    
    for phase, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum state
        a_curr = a.copy()
        a_prev = a.copy()
        mu = 0.0  # momentum parameter
        step = 1.0 / (2.0 * beta)
        
        n_iters = 2000
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov extrapolation
            if it > 0:
                mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
                beta_nest = (mu_new - 1.0) / (mu_new + 1.0)
                y = a_curr + beta_nest * (a_curr - a_prev)
                y = np.maximum(y, 1e-15)
                sy = y.sum()
                if sy < 1e-15:
                    y = a_curr.copy()
                else:
                    y = y / sy
                mu = mu_new
            else:
                y = a_curr.copy()

            z = fast_conv(y)
            z_max = z.max()
            exp_z = np.exp(beta * (z - z_max))

            f_smooth = (z_max + np.log(Z)) / beta

            a_try = project_simplex(y - step * grad)
