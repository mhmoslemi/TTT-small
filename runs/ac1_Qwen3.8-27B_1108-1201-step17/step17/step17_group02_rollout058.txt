def simplex_project(z):
    """Project z onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(z)
    # Sort in decreasing order
    u = np.sort(z)[::-1]
    # Find rho
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u > (cssv - 1) / rho
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1) / rho_val
    return np.maximum(z - theta, 0)

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_project(z):
    """Project z onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(z)
    if n == 1:
        return np.array([max(0.0, z[0])])
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(z - theta, 0.0)


def compute_grad(a, p):
    """
    Compute gradient of sum_k p[k] * conv(a,a)[k] w.r.t. a.
    grad[j] = 2 * sum_i p[j+i] * a[i]
    Uses FFT for efficiency.
    """
    n = len(a)
    L = 2 * n - 1
    p_flip = p[::-1]

    conv_len = n + L - 1  # 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_flip, fft_size)
    conv_result = np.fft.irfft(A * P, fft_size)

    grad_half = conv_result[n - 1:2 * n - 1][::-1]
    return 2.0 * grad_half


def stable_softmax(x, tau):
    """Compute softmax(tau * x) numerically stably."""
    z = tau * x
    z_max = z.max()
    if z_max > 700:
        z = z - z_max
    exp_z = np.exp(z)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return exp_z / s


def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def init_u_shape(n, sharpness=2.0):

def init_cosh_shape(n, width=1.0):

def init_spectral(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.2 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, width=0.15, center_sep=0.7):
    x = np.linspace(0, 1, n)
    c1 = (1.0 - center_sep) / 2.0
    c2 = (1.0 + center_sep) / 2.0
    a = np.exp(-((x - c1) / width) ** 2) + np.exp(-((x - c2) / width) ** 2) + 0.005
    return normalize(a)


def init_structured(n, k=3, phase_shifts=None):
    """
    Create a sequence where sum of reflections at specific lags is approximately constant.
    This is inspired by the KKT condition.
    """
    rng = np.random.default_rng(12345)
    if phase_shifts is None:
        # Use lags spread across the range
        phase_shifts = [n // k * j for j in range(min(k, 5))]

    # Start with a base shape
    a = init_u_shape(n, 1.5)

    # Add oscillatory components that help equalize reflections
    for k_val in phase_shifts[:3]:
        x = np.linspace(0, 1, n)
        freq = k_val / n
        a += 0.1 * np.cos(2 * np.pi * freq * n * x + rng.uniform(0, 2 * np.pi))

    return normalize(a)


def gradient_optimize(a0, time_limit, tau_init=10.0, tau_rate=1.003, eta_init=0.5,
                      momentum=0.9, verbose=False, label=""):
    """
    Nesterov-accelerated projected gradient with smooth max approximation.
    Minimizes max_k conv(a,a)[k] subject to a>=0, sum(a)=1.
    """
    global _global_start_time
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()

    v = np.zeros(n)
    tau = tau_init
    eta = eta_init
    stall = 0
    max_stall = 500
    a_curr = a.copy()

    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1

        ac = fast_autocorr(a_curr)
        p = stable_softmax(ac, tau)
        g = compute_grad(a_curr, p)

        # Nesterov lookahead
        a_look = a_curr + momentum * v
        a_look = np.maximum(a_look, 0.0)
        sl = a_look.sum()
        if sl > 1e-15:
            a_look = a_look / sl

        ac_look = fast_autocorr(a_look)
        p_look = stable_softmax(ac_look, tau)
        g_look = compute_grad(a_look, p_look)

        v = momentum * v + (1.0 - momentum) * g_look

        # Backtracking line search
        a_trial = a_curr - eta * v
        a_trial = simplex_project(a_trial)
        val_trial = fast_autocorr(a_trial).max()

        retries = 0
        while val_trial > best_val * (1.0 + 1e-10) and retries < 10:
            eta *= 0.5
            a_trial = a_curr - eta * v
            a_trial = simplex_project(a_trial)
            val_trial = fast_autocorr(a_trial).max()
            retries += 1

        if val_trial < best_val:
            best_val = val_trial
            best_a = a_trial.copy()
            stall = 0
        else:
            stall += 1

        a_curr = a_trial

        # Adapt tau: increase slowly to sharpen focus on peaks
        tau = min(tau * tau_rate, 200.0)

        # Adapt eta: slowly increase if things are going well
        if stall == 0:
            eta = min(eta * 1.02, 5.0)
        else:
            eta = max(eta * 0.9, 0.01)

        # Periodic restart from best with small perturbation
        if stall > max_stall:
            stall = 0
            rng = np.random.default_rng(iter_count)
            scale = min(0.1 * np.sqrt(stall / 10.0), 0.3)
            pert = rng.normal(0, scale, n) * best_a
            a_r = best_a + pert
            a_r = simplex_project(a_r)
            a_curr = a_r
            v = np.zeros(n)
            eta = eta_init
            tau = max(tau_init, tau * 0.5)

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradOpt [{label}] iter={iter_count} ({elapsed:.0f}s): "
                  f"val={best_val:.8f}, tau={tau:.1f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_val


def basin_hopping(a0, time_limit, inner_time, n_restarts=None, verbose=False):
    """
    Basin hopping: repeatedly optimize from perturbed starting points.
    """
    global _global_start_time
    n = len(a0)
    a_curr = normalize(a0.copy())
    best_a = a_curr.copy()
    best_val = fast_autocorr(a_curr).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    n_attempt = 0
    temperature = 0.1
    while time.monotonic() - t0 < time_limit:
        n_attempt += 1

        # Generate perturbed starting point
        scale = temperature * np.sqrt(n) / n
        pert = rng.normal(0, scale, n) * a_curr
        a_start = a_curr + pert
        a_start = simplex_project(a_start)

        # Inner optimization
        inner_tl = min(inner_time, time_limit - (time.monotonic() - t0))
        if inner_tl < 5:
            break
        a_new, val_new = gradient_optimize(a_start, inner_tl,
                                           tau_init=5.0, tau_rate=1.005,
                                           eta_init=0.3, momentum=0.85,
                                           verbose=False, label=f"bh{n_attempt}")

        # Acceptance criterion (simulated annealing style)
        delta = val_new - best_val
        if delta < 0:
            best_a = a_new.copy()
            best_val = val_new
            a_curr = a_new.copy()
        else:
            accept_prob = math.exp(-delta / max(temperature, 1e-10))
            if rng.random() < accept_prob:
                a_curr = a_new.copy()
            else:
                a_curr = best_a.copy()

        # Cool down
        temperature *= 0.95
        if verbose and n_attempt % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BasinHop [{n_attempt}] ({elapsed:.0f}s): best={best_val:.8f}, T={temperature:.4f}")

    return best_a, best_val


def coordinate_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each peak, find the most contributing pair and
    redistribute mass to reduce it.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_val = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    stall = 0
    max_stall = 300
    n_top = 20

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find destination: minimize increase in other peaks
            # Use a quick heuristic: pick random position not in top peak regions
            in_peak = np.zeros(n, dtype=bool)
            for kk in top_k[:5]:
                s2 = max(0, int(kk) - n + 1)
                e2 = min(n - 1, int(kk))
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            # Try a few random destinations
            n_tries = min(5, len(candidates))
            chosen = rng.choice(candidates, size=n_tries, replace=False)

            found = False
            for dst in chosen:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.005]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_val - 1e-16:
                        a = a_trial
                        best_val = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

            if not found:
                # Try random kick
                if stall > 50:
                    j = int(rng.integers(n))
                    if j != src:
                        frac = rng.uniform(0.005, 0.03)
                        amt = src_val * frac
                        a[src] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_val:
                            best_val = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_val:
                    best_val = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordPolish ({elapsed:.0f}s): val={best_val:.8f}, stall={stall}")

    return best_a, best_val


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

            try_update(a_out, "deep_fast")

    # === Phase 4: Basin hopping ===
    print(f"\n=== Phase 4: Basin hopping (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - _global_start_time)
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 60:
        a_in = normalize(best_seq)
        inner_t = min(30.0, t4 / 8.0)
        a_out, _ = basin_hopping(a_in, t4, inner_t, verbose=True)
        try_update(a_out, "basinhop")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length exploration ===")
    remaining = time_budget - (time.monotonic() - _global_start_time)
    n_best = len(best_seq)

    trial_lengths = []
    for delta in [200, 500, -200, 800, -500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - _global_start_time)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_to_length(best_seq, new_n)

        t5a = min(remaining * 0.3, 200.0)
        if t5a > 30:
            a_out, _ = gradient_optimize(a_long, t5a, tau_init=5.0, tau_rate=1.004,
                                         eta_init=0.4, momentum=0.9, verbose=True,
                                         label=f"n{new_n}")
            try_update(a_out, f"grad_n{new_n}")

    # === Phase 6: Coordinate polish ===
    print(f"\n=== Phase 6: Coordinate polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - _global_start_time)
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t6, verbose=True)
        try_update(a_out, "coord")

    # === Phase 7: Final gradient pass with high tau ===
    print(f"\n=== Phase 7: Final gradient pass ===")
    remaining = time_budget - (time.monotonic() - _global_start_time)
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 30:
        a_in = normalize(best_seq)
        a_out, _ = gradient_optimize(a_in, t7, tau_init=50.0, tau_rate=1.002,
                                     eta_init=0.1, momentum=0.95, verbose=True,
                                     label="final_high_tau")
        try_update(a_out, "final_high_tau")

    # === Phase 8: Very fine coordinate polish ===
    print(f"\n=== Phase 8: Fine coordinate polish ===")
    remaining = time_budget - (time.monotonic() - _global_start_time)
    t8 = min(remaining * 0.10, 100.0)
    if t8 > 15:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t8, verbose=False)
        try_update(a_out, "coord_fine")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-_global_start_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = init_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
