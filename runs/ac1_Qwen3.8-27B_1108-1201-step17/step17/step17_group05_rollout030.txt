```python
1. LP-based direction finding with surrogate objective

import numpy as np

def fast_conv(a, b=None):
    if b is None:
        b = a
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS
```

```python
def optimize_sequence(a0, time_limit, verbose=True):
    """
    Main optimization loop combining:
    1. Sharp subgradient descent (focusing on top-k lags)
    2. LP direction finding
    3. Line search
    4. Coordinate descent polish
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Nesterov momentum
    x = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient at top lags
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Find top lags (within tolerance of max)
        threshold = M * 0.999
        top_idx = np.where(ac >= threshold)[0]
        if len(top_idx) == 0:
            top_idx = np.array([np.argmax(ac)])
        
        # Equal weights for top lags
        w = np.ones(len(top_idx)) / len(top_idx)
        
        # Compute subgradient
        grad = np.zeros(n)
        for k, wi in zip(top_idx, w):
            k_int = int(k)
            for j in range(n):
                idx = k_int - j
                if 0 <= idx < n:
                    grad[j] += wi * 2.0 * y[idx]
        
        # Step size with line search
        # Start with eta = c / sqrt(t)
        eta = 5.0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent step (multiplicative)
        log_y = np.log(np.maximum(y, 1e-30))
        log_new = log_y - eta * grad
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        # Line search: try to improve
        M_new = fast_autocorr(a_new).max()
        
        # If no improvement, try smaller steps
        if M_new >= M:
            for shrink in [0.5, 0.25, 0.1, 0.05]:
                log_new = log_y - eta * shrink * grad
                log_new -= log_new.max()
                a_try = np.exp(log_new)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                M_try = fast_autocorr(a_try).max()
                if M_try < M:
                    a_new = a_try
                    M_new = M_try
                    break
        
        x_prev = x
        x = a_new
        
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m
```

For a 0-1 sequence (a_i ∈ {0,1}), the autocorrelation is:

This counts the number of pairs at distance k. Minimizing the maximum is the Golomb ruler problem (minimize the maximum number of pairs at any distance).

**Better LP formulation:**

Given current `a` (sum=1) with `max conv(a,a) = M`:
We want to find `b` on the simplex that minimizes the "new peak" after moving towards b.

If we move as a_new = (1-t)*a + t*b for small t, then:
conv(a_new, a_new)[k] = (1-t)^2 conv(a,a)[k] + 2t(1-t) conv(a,b)[k] + t^2 conv(b,b)[k]

For small t, the change is approximately:
Δconv[k] ≈ 2t * (conv(a,b)[k] - conv(a,a)[k])

We want to minimize the maximum of conv(a_new, a_new), which for small t is approximately:
minimize max_k [conv(a,a)[k] + 2t*(conv(a,b)[k] - conv(a,a)[k])]
= minimize max_k [M + 2t*(conv(a,b)[k] - conv(a,a)[k])] (for k where conv(a,a)[k] = M)

For the peak lags k*: we want conv(a,b)[k*] to be as small as possible.
For non-peak lags: we don't want conv(a,b)[k] to be too large (or conv(b,b)[k]).

A good LP approximation:
Minimize: sum_k* conv(a,b)[k*] (sum over peak lags)
Subject to: b >= 0, sum(b) = 1
            conv(a,b)[k] <= M + delta for all k (don't increase any lag too much)

This is a linear program! The objective is linear in b, and the constraints are linear in b.

The solution gives the best direction to reduce the peak while not increasing other lags.

```python
def lp_direction(a, M, regularize=0.01):
    """
    Find the best direction b on the simplex to reduce the peak autocorrelation.
    LP: minimize sum_k* conv(a,b)[k*] + regularize * sum(b^2) [approximated]
    Subject to: b >= 0, sum(b) = 1
                 conv(a,b)[k] <= M for all k
    """
    n = len(a)
    ac = fast_autocorr(a)
    
    # Find peak lags
    threshold = M * 0.99
    peak_lags = np.where(ac >= threshold)[0]
    
    # Build constraint matrix
    # conv(a,b)[k] = sum_j a[j]*b[k-j] (valid j, k-j)
    # For each k, this is a linear function of b
    
    n_constraints = 2 * n - 1
    T = np.zeros((n_constraints, n))
    for j in range(n):
        # a[j] * b[k-j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # i.e., T[k, j] = a[k-j] for valid k-j
        for k in range(j, j + n):
            T[k, j] = a[k - j]
    
    # Objective: minimize sum over peak lags of conv(a,b)[k]
    obj_vec = np.zeros(n)
    for k in peak_lags:
        ki = int(k)
        for j in range(max(0, ki-n+1), min(n, ki+1)):
            obj_vec[j] += a[ki - j]
    
    # Solve LP
    b_var = cp.Variable(n, nonneg=True)
    constraints = [
        T @ b_var <= M * np.ones(n_constraints),
```

```python
def _fast_autocorr(a):
    """Compute conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def _compute_subgrad(a, peak_lags, weights=None):
    """
    Compute subgradient of max_k conv(a,a)[k] at given peak lags.
    grad[j] = sum_k w_k * 2*a[k-j] (for valid k-j in [0,n))
    Uses direct summation (O(n * num_lags), fast for small num_lags).
    """
    n = len(a)
    if weights is None:
        weights = np.ones(len(peak_lags)) / len(peak_lags)
    grad = np.zeros(n)
    for k, wk in zip(peak_lags, weights):
        ki = int(k)
        wk = float(wk)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        js = np.arange(j_start, j_end)
        grad[js] += wk * 2.0 * a[ki - js]
    return grad

def _nesterov_mirror_descent(a0, time_limit, verbose=True, top_k=5,
                              c0=5.0, cooldown_rate=0.0005,
                              restart_every=3000, seed=42):

    Uses backtracking line search and periodic restarts.
    """
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    ac = _fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute autocorrelation and find peak lags
        ac = _fast_autocorr(y)
        M = ac.max()

        # Adaptive top-k: start with few, increase as we converge
        k_eff = min(top_k + t // 2000, len(ac))
        # Use a soft threshold: lags within 0.5% of max
        threshold = M * (1.0 - 0.005 - 0.002 * (1.0 - min(1.0, t / 5000.0)))
        peak_mask = ac >= threshold
        num_peaks = peak_mask.sum()
        if num_peaks == 0:
            peak_lags = np.array([np.argmax(ac)])
            weights = np.array([1.0])
        elif num_peaks <= k_eff:
            peak_lags = np.where(peak_mask)[0]
            # Weight by excess over threshold (KKT-inspired)
            excess = ac[peak_lags] - threshold
            if excess.max() > 1e-15:
                weights = excess / excess.sum()
            else:
                weights = np.ones(num_peaks) / num_peaks
        else:
            # Take top k_eff
            peak_lags = np.argsort(ac)[::-1][:k_eff]
            peak_vals = ac[peak_lags]
            w_raw = peak_vals - peak_vals.min()
            if w_raw.max() > 1e-15:
                weights = w_raw / w_raw.sum()
            else:
                weights = np.ones(k_eff) / k_eff

        # Compute subgradient
        grad = _compute_subgrad(y, peak_lags, weights)

        # Mirror descent step with backtracking
        eta = c0 / (np.sqrt(t) + 1.0)
        improved = False

        for shrink in [1.0, 0.7, 0.5, 0.3, 0.1]:
            eta_eff = eta * shrink
            ratio = np.exp(-eta_eff * grad / (y + 1e-30))
            a_new = y * ratio
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new

            M_new = _fast_autocorr(a_new).max()
            if M_new < M - 1e-14:
                x_prev = x
                x = a_new
                improved = True
                if M_new < best_m - 1e-12:
                    best_m = M_new
                    best_a = x.copy()
                    stall = 0
                else:
                    stall += 1
                break

        if not improved:
            x_prev = x
            x = y.copy()
            stall += 1

        # Periodic restart
        if stall > restart_every:
            stall = 0
            # Perturb around best
            scale = 0.05 + 0.05 * rng.random()
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert

def _lp_direction(a, M, solver, num_peaks=10, verbose=False):
    """
    LP: Find b on simplex minimizing sum of peak-lag conv(a,b)[k],
    subject to conv(a,b)[k] <= M for peak lags, b >= 0, sum(b) = 1.
    """
    n = len(a)
    ac = _fast_autocorr(a)

    # Get top lags
    k_eff = min(num_peaks, len(ac))
    peak_lags = np.argsort(ac)[::-1][:k_eff]
    peak_vals = ac[peak_lags]
    # Only use lags close to max
    thresh = peak_vals.min() * 0.95
    active = peak_lags[peak_vals >= thresh]
    if len(active) == 0:
        active = peak_lags[:3]

    # Build constraint matrix for active peaks only
    # conv(a,b)[k] = sum_j a[j]*b[k-j] for valid indices
    # As a linear function of b: T[k, j] = a[j] if 0 <= k-j < n... 
    # Actually: conv(a,b)[k] = sum_i a[i]*b[k-i] where valid
    # = sum_j b[j] * a[k-j] where valid (0 <= k-j < n)
    # So coefficient of b[j] in conv[k] is a[k-j] if 0 <= k-j < n

    T = np.zeros((len(active), n))
    for row, k in enumerate(active):
        ki = int(k)
        for j in range(max(0, ki - n + 1), min(n, ki + 1)):
            T[row, j] = a[ki - j]

    # Objective: minimize sum of conv(a,b) over active peaks
    # = (sum of T rows) @ b
    obj_vec = T.sum(axis=0)

    b_var = cp.Variable(n, nonneg=True)
    constraints = [
        T @ b_var <= M * np.ones(len(active)),
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(obj_vec @ b_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

            if stall % 5 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    M_new = _fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and (stall % 10 == 0 or found_improve):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def _interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def _smooth_init(n, seed=0):
    """Create a smooth non-negative initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for f in [1, 2, 3, 4, 5]:
        coeff = rng.normal(0, 0.1)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return _norm_sum1(a)

def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase: subgradient descent -> LP directions -> coordinate search -> final polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = _eval_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            parent = _norm_sum1(parent)
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = _smooth_init(n_base, seed=1)
        best_score = _eval_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # Phase 1: Nesterov mirror descent (40% of budget)
    print(f"\n=== Phase 1: Mirror Descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 700.0)
    if t1 > 60:
        a_in = best_seq.copy()
        a_out, _ = _nesterov_mirror_descent(
            a_in, t1, verbose=True, top_k=4, c0=6.0,
            cooldown_rate=0.0005, restart_every=2500, seed=42
        )
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** MD improved: {sc:.6f}")

        # Try with different parameters
        remaining = time_budget - (time.monotonic() - t_global)
        t1b = min(remaining * 0.15, 250.0)
        if t1b > 30:
            a_in = best_seq.copy()
            a_out, _ = _nesterov_mirror_descent(
                a_in, t1b, verbose=False, top_k=8, c0=4.0,
                cooldown_rate=0.001, restart_every=4000, seed=123
            )
            sc = _eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** MD2 improved: {sc:.6f}")

    # Phase 2: LP direction finding (20% of budget)
    print(f"\n=== Phase 2: LP Directions (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        n_best = len(best_seq)
        # LP direction + line search loop
        a_curr = best_seq.copy()
        ac = _fast_autocorr(a_curr)
        M_curr = ac.max()
        t2_start = time.monotonic()

        while time.monotonic() - t2_start < t2:
            direction = _lp_direction(a_curr, M_curr, solver, num_peaks=8)
            if direction is None:
                break

            # Line search: a_new = (1-t)*a_curr + t*direction
            best_t = 0.0
            best_M = M_curr
            for t_frac in [0.3, 0.2, 0.1, 0.05, 0.5, 0.15]:
                a_try = (1.0 - t_frac) * a_curr + t_frac * direction
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                M_try = _fast_autocorr(a_try).max()
                if M_try < best_M:
                    best_M = M_try
                    best_t = t_frac
                    a_new = a_try.copy()

            if best_t > 0:
                a_curr = a_new
                M_curr = best_M
                sc = _eval_safe(a_curr)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_curr.copy()
                    print(f"    LP+LS improved: {sc:.6f} (t={best_t:.2f})")
            else:
                # No improvement, try smaller step or break
                break

    # Phase 3: Coordinate search (20% of budget)
    print(f"\n=== Phase 3: Coordinate Search (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)

            best_seq = a_out.copy()
            print(f"  *** Coord improved: {sc:.6f}")

    # Phase 4: Final mirror descent polish (10% of budget)
    print(f"\n=== Phase 4: Final MD Polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)

            cooldown_rate=0.002, restart_every=5000, seed=999
        )
        sc = _eval_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Final MD improved: {sc:.6f}")

    # Phase 5: Try different lengths
    print(f"\n=== Phase 5: Length Exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    for new_n in [n_best - 200, n_best + 200]:

        if new_n < 1000 or new_n > 5000:

        print(f"  Trying n={new_n}...")
        a_long = _interpolate_seq(best_seq, new_n)
        t5 = min(remaining * 0.3, 100.0)
        if t5 > 30:
            a_out, _ = _nesterov_mirror_descent(
                a_long, t5, verbose=False, top_k=4, c0=5.0,
                cooldown_rate=0.001, restart_every=3000, seed=777
            )
            sc = _eval_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"    *** n={new_n} improved: {sc:.6f}")
        remaining = time_budget - (time.monotonic() - t_global)

    # Phase 6: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 6: Final Coord Polish ===")
        t6 = min(remaining * 0.5, 120.0)

                print(f"  *** Final coord improved: {sc:.6f}")

    if best_seq is not None:

            best_score = fs

    else:

        best_seq = _smooth_init(3200, seed=1)
        best_score = _eval_safe(best_seq)

    return [float(x) for x in best_seq]
```
