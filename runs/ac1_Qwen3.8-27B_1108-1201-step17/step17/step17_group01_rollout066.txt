"""

def _get_lp_solver():

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def fft_size_for(n):
    return 1 << (2 * n - 1).bit_length()

def record_best(a, label, t0, best_score):
    """Top-level recorder. Returns (a_norm, score) if a improves best, else None."""
    if a is None:
        return None
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if not np.isfinite(s) or s < 0.01:
        return None

    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:

        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.0fs)" % (label, sc, len(a_norm), el))

def spectral_flatten_step(a, alpha, fft_size):
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    if len(mag) > 1:
        ac_mean = mag[1:].mean()
        mag_new = alpha * ac_mean + (1.0 - alpha) * mag[1:]
        mag_new[0] = mag[0]
        mag = mag_new
    A_new = mag * np.exp(1j * phase)
    return np.fft.irfft(A_new, fft_size)[:len(a)]


def ac_equalize_step(a, alpha, fft_size):
    """Flatten the autocorrelation side band (ac[1:]) directly, keep phase of a."""
    n = len(a)
    ac = fast_autocorr(a)
    ac[0] = 0.0
    ac_mean = ac[1:].mean()
    ac_t = alpha * ac_mean + (1.0 - alpha) * ac
    ac_t = np.maximum(ac_t, 0.0)
    # ac_t must be a valid autocorrelation: take its log-magnitude as power spectrum
    # Use magnitude spectrum of (1+ac_t padded) approx cheaply: just blend spectra.
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    target_mag = np.sqrt(np.maximum(fast_autocorr_target(ac_t, n, fft_size), 0.0))
    if len(target_mag) == len(mag):
        mag_new = alpha * target_mag + (1.0 - alpha) * mag
        mag_new[0] = mag[0]
        mag = mag_new
    phase = np.angle(A)
    return np.fft.irfft(mag * np.exp(1j * phase), fft_size)[:n]


def fast_autocorr_target(ac_t, n, fft_size):
    """Power spectrum producing (desired) autocorrelation ac_t of length 2n-1."""
    pad = np.zeros(fft_size)
    m = len(ac_t)
    pad[:m] = ac_t[:min(m, fft_size)]
    P = np.fft.rfft(pad)
    return np.abs(P) ** 2


def compute_subgrad(a, ac, top_k, beta):
    n = len(a)
    k_max = min(top_k, len(ac))
    if k_max < len(ac):
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
    else:
        top_idx = np.arange(len(ac))
    top_vals = np.maximum(ac[top_idx], 1e-30)
    log_w = beta * (top_vals - top_vals.max())
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n, k_int + 1)
        if lo < hi:
            idx = np.arange(lo, hi)
            grad[lo:hi] += wi * a[k_int - idx]
        lo2 = max(0, k_int)
        hi2 = min(n, k_int + n)
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[lo2:hi2] += wi * a[idx2 - k_int]
    return grad

        else:
            y = y / sy

        eta_t = eta / (1.0 + 0.05 * t / n)

def lp_frank_wolfe(a0, time_limit, top_K=150, verbose=False):

        if M_curr < 1e-16:

        margins = M_curr - ac
        binding_idx = np.argsort(margins)[:K_sel]
        T_sel = np.zeros((K_sel, n))
        for row, k_val in enumerate(binding_idx):
            k_int = int(k_val)
            lo = max(0, k_int - n + 1)
            hi = min(n, k_int + 1)
            if lo < hi:
                T_sel[row, lo:hi] = y[k_int - np.arange(lo, hi)]
            lo2 = max(0, k_int)
            hi2 = min(n, k_int + n)
            if lo2 < hi2:
                T_sel[row, lo2:hi2] += y[np.arange(lo2, hi2) - k_int]
        M_sel = M_curr * np.ones(K_sel)
        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_sel @ b_var <= M_sel])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        alpha_fw = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_fw) * y + alpha_fw * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y.copy() if sy < 1e-15 else y_new / sy
        beta_t = (t - 1.0) / (t + 2.0)
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        x_new = y_new.copy() if sx < 1e-15 else x_new / sx

        x_mom = x_new
        for cand in (y, x_mom):
            mc = fast_autocorr(cand).max()
            if mc < best_m:
                best_m = mc
                best_a = cand.copy()
        if verbose and t % 5 == 0:
            el = time.monotonic() - t0
            print("    LP-FW iter=%d (%.0fs): J=%.6f" % (t, el, 2.0 * n * best_m))
    return best_a, best_m


def make_soft_binary(n, seed, density, ks):
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    n_ones = max(1, int(n * density))
    positions = rng.choice(n, size=n_ones, replace=False)
    a[positions] = 1.0
    kernel = np.ones(ks) / float(ks)
    a = np.convolve(a, kernel, mode='same')
    return norm_sum1(a)


def make_ramp(n, direction):
    if direction == 'up':
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, -1, dtype=np.float64)
    return norm_sum1(a)


def make_sinusoidal(n, freq, phase):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * freq * x + phase) ** 2
    a = np.maximum(a, 0.01)
    return norm_sum1(a)


def make_cosine(n, mode):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    return norm_sum1(np.maximum(a, 0.0))


def make_tilted_gaussian(n, sigma, tilt, center=0.5):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2) * np.exp(tilt * (x - center))
    return norm_sum1(np.maximum(a, 0.0))


def make_gaussian(n, sigma, center):
    x = np.linspace(0, 1, n)
    return norm_sum1(np.exp(-0.5 * ((x - center) / sigma) ** 2))


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    return (f_interp / ss) if ss > 1e-15 else (np.ones(new_n) / new_n)

    # ---- Phase 4: core polish + LP at current best length ----
    rem = time_budget - (time.monotonic() - t0)
    nb = len(best_seq)
    print("\n=== Phase 4: polish n=%d ===" % nb)
    t4 = min(rem * 0.13, 220.0)
    if t4 > 20:
        a_out, _ = optimize_core(norm_sum1(best_seq), t4, fft_size_for(nb), verbose=False)
        rec = record_best(a_out, "polish_n%d" % nb, t0, best_score)
        if rec is not None:
            best_seq, best_score = rec
    rem = time_budget - (time.monotonic() - t0)
    nb = len(best_seq)
    if rem > 30 and nb <= 5200:
        t4b = min(rem * 0.10, 150.0)
        if t4b > 15:
            a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t4b, top_K=180, verbose=False)
            rec = record_best(a_out, "lpFW2_n%d" % nb, t0, best_score)
            if rec is not None:
                best_seq, best_score = rec

    # ---- Phase 5: longer/shorter length exploration with shape transfer ----
    rem = time_budget - (time.monotonic() - t0)
    nb = len(best_seq)
    print("\n=== Phase 5: length transfer ===")
    cand_lengths = []
    for delta in (400, 800, -600):
        nn = nb + delta
        if 2000 <= nn <= 6000:
            cand_lengths.append(nn)
    for nn in cand_lengths[:2]:
        rem = time_budget - (time.monotonic() - t0)
        if rem < 60:
            break
        print("  length n=%d" % nn)
        a_long = interpolate_sequence(best_seq, nn)
        rec = record_best(a_long, "interp_n%d" % nn, t0, best_score)
        if rec is not None:
            best_seq, best_score = rec
        ta = min(rem * 0.40, 220.0)
        if ta > 15:
            a_out, _ = optimize_core(interpolate_sequence(best_seq, nn), ta, fft_size_for(nn), verbose=False)
            rec = record_best(a_out, "core_n%d" % nn, t0, best_score)
            if rec is not None:
                best_seq, best_score = rec
        rem = time_budget - (time.monotonic() - t0)
        if rem > 40 and nn <= 5200:
            tb = min(rem * 0.25, 120.0)
            if tb > 15:
                a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), tb, top_K=150, verbose=False)
                rec = record_best(a_out, "lpFW_n%d" % nn, t0, best_score)
                if rec is not None:
                    best_seq, best_score = rec

    # ---- Final validation + safe return ----
    if best_seq is None:
        n_fb = 4200
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    s = best_seq.sum()
    if not np.isfinite(s) or s < 0.01:
        best_seq = np.ones(n_fb) / n_fb
    else:
        best_seq = best_seq / best_seq.sum()
    fs = evaluate_safe(best_seq)
    if not np.isfinite(fs):
        best_seq = np.clip(best_seq, 0.02, 1000.0)
    print("\nFINAL: score=%.6f n=%d time=%.0fs" % (fs, len(best_seq), time.monotonic() - t0))
    return [float(v) for v in best_seq]
