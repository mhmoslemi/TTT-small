def lp_remez_improved(a, max_cons=60, r=0.15, solver=None):

    M_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    # Symmetric trust region
    constraints.append(a_var <= a * (1.0 + r))
    constraints.append(a_var >= a * (1.0 - r))
    prob = cp.Problem(cp.Minimize(M_var), constraints)

        else:
            prob.solve(verbose=False)

        return a.copy()
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

            x_new = x_new / x_new.sum()
            a_new = np.zeros(n)
            a_new[top_idx_sorted] = x_new
            a_new = a_new / a_new.sum()
            m_new = fast_conv(a_new).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a_new.copy()
                a = a_new.copy()
    
    return best_a, best_m

def perturbation_search(a0, time_limit, verbose=False):
    """Try random perturbations to escape local minima."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    
    while time.monotonic() - t0 < time_limit:
        # Generate perturbation
        strategy = int(rng.integers(5))
        if strategy == 0:
            # Gaussian noise
            a_try = a * (1 + rng.normal(0, 0.01, n))
        elif strategy == 1:
            # Random subset scaling
            a_try = a.copy()
            idx = rng.choice(n, size=n//10, replace=False)
            a_try[idx] *= rng.uniform(0.5, 1.5, len(idx))
        elif strategy == 2:
            # Asymmetric shift
            shift = rng.uniform(0.01, 0.05) * rng.choice([-1, 1])
            a_try = a * (1 + shift * (np.arange(n) / n - 0.5))
        elif strategy == 3:
            # Sine perturbation
            freq = rng.uniform(1, 5)
            a_try = a * (1 + 0.02 * np.sin(2 * np.pi * freq * np.arange(n) / n))
        else:
            # Endpoint boost
            a_try = a.copy()
            boost = rng.uniform(0.01, 0.05)
            side = rng.integers(n // 2)
            a_try[:side] *= (1 + boost)
            a_try[n-side:] *= (1 - boost * 0.5)
        
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        m_try = fast_conv(a_try).max()
        
        if m_try < best_m:
            best_m = m_try
            best_a = a_try.copy()
            a = a_try.copy()
        elif time.monotonic() - t0 > time_limit * 0.5 and rng.random() < 0.1:
            # Accept worse solutions sometimes (simulated annealing-like)
            if rng.random() < 0.05:
                a = a_try.copy()
    
    return best_a, best_m

def make_init(n, kind='skewed'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'linear_skew':
        a = 0.5 + 0.5 * x
    elif kind == 'quadratic_skew':
        a = 0.3 + 0.7 * x ** 1.5
    elif kind == 'cubic_skew':
        a = 0.2 + 0.8 * x ** 2.0
    elif kind == 'exp_skew':
        a = np.exp(1.5 * (x - 0.5))
    elif kind == 'beta_asym':
        # Beta(2, 3) distribution shape
        a = x * (1 - x) ** 2
    elif kind == 'beta_asym2':
        # Beta(3, 2) distribution shape  
        a = x ** 2 * (1 - x)
    elif kind == 'flat_skew':
        a = 1.0 + 0.3 * (x - 0.5)
    elif kind == 'skewed':
        # Mix of symmetric and asymmetric
        a = 0.6 * (1.0 + np.cos(2.0 * np.pi * x)) / 2.0 + 0.4 * x
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
