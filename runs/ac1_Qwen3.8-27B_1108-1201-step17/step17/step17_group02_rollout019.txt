Specifically, `conv(a,a)[k] = sum_i a[i]*a[k-i]`, so `d/da[j] conv(a,a)[k] = a[k-j] + a[j]` (for valid indices). This is `2*a[k-j]` when k-j != j, and `2*a[j]` when k-j=j.

The matrix `T_full[k,j] = y[k-j]` (for valid k-j) represents half the gradient. So the LP constraint `T_sel @ b <= M` is approximating the condition that moving in direction b won't increase the tracked peaks beyond M.

This is a linearization-based approach, which is standard for Frank-Wolfe on non-smooth objectives.

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lbfgs_smooth_optimize(a0, time_limit, top_k=80, beta_start=50.0, beta_end=200.0,
                          verbose=True, rng_seed=42):
    """L-BFGS-B optimization using logsumexp approximation of max autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    def objective_and_grad(x_flat):
        x = np.maximum(x_flat, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            return 2.0 * n, np.zeros(n)
        x = x / sx
        ac = fast_autocorr(x)
        # Use top_k peaks for logsumexp
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        vals = ac[top_idx] * beta
        max_val = vals.max()
        if max_val > 500:
            vals = np.full_like(vals, 500.0)
            max_val = 500.0
        exp_vals = np.exp(vals - max_val)
        w_sum = exp_vals.sum()
        if w_sum < 1e-30:
            w_sum = 1e-30
        w = exp_vals / w_sum
        obj = 2.0 * n * (max_val + np.log(w_sum)) / beta

        # Gradient
        grad = np.zeros(n)
        for idx, k in enumerate(top_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            i_vals = np.arange(i_start, i_end + 1)
            partner = ki - i_vals
            contrib = a[partner] + a[i_vals]
            # Actually we need x[partner] + x[i_vals]
            contrib = x[partner] + x[i_vals]
            grad[i_vals] += w[idx] * beta * contrib

        # Rescale gradient for the normalization constraint
        # The mapping x -> x/sum(x) has Jacobian (I - x*x^T)/sum(x)
        # But for L-BFGS we just use the gradient in the unnormalized space
        # and let the bounds/projector handle non-negativity

        return obj, grad

    # Multi-scale beta optimization
    betas = [beta_start]
    cur_beta = beta_start
    while cur_beta < beta_end:
        cur_beta = min(cur_beta * 1.8, beta_end)
        betas.append(cur_beta)

    for bi, beta in enumerate(betas):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 30:
            break
        t_avail = time_limit - elapsed - 10
        if t_avail < 20:
            break

        if verbose:
            J_best = 2.0 * n * best_m
            print(f"    L-BFGS beta={beta:.1f} (t_avail={t_avail:.0f}s), current J={J_best:.6f}")

        # Define objective/gradient with current beta
        def obj_grad(x_flat, b=beta):
            x = np.maximum(x_flat, 0.0)
            sx = x.sum()
            if sx < 1e-15:
                return 2.0 * n, np.zeros(n)
            x = x / sx
            ac = fast_autocorr(x)
            k_max = min(top_k, len(ac))
            top_idx = np.argsort(ac)[::-1][:k_max]
            vals = ac[top_idx] * b
            max_val = vals.max()
            if max_val > 500:
                vals = np.full_like(vals, 500.0)
                max_val = 500.0
            exp_vals = np.exp(vals - max_val)
            w_sum = exp_vals.sum()
            if w_sum < 1e-30:
                w_sum = 1e-30
            w = exp_vals / w_sum
            obj = 2.0 * n * (max_val + np.log(w_sum)) / b

            grad = np.zeros(n)
            for idx, k in enumerate(top_idx):
                ki = int(k)
                i_start = max(0, ki - n + 1)
                i_end = min(n - 1, ki)
                if i_start > i_end:
                    continue
                i_vals = np.arange(i_start, i_end + 1)
                partner = ki - i_vals
                contrib = x[partner] + x[i_vals]
                grad[i_vals] += w[idx] * b * contrib

            # Project gradient: account for normalization
            # d(x/sum(x))/dx[i] = e_i/sum(x) - x/sum(x)^2
            # So gradient in normalized space: (grad - sum(grad)*x)/sum(x) ... 
            # Actually for L-BFGS we work in the unnormalized space with bounds
            return obj, grad

        # Run L-BFGS-B
        best_a_beta = a.copy()
        result = minimize(
            obj_grad,
            a.copy(),
            method='L-BFGS-B',
            bounds=[(0.0, None)] * n,
            options={'maxiter': 200, 'maxfun': 5000, 'ftol': 1e-10, 'gtol': 1e-8,
                     'disp': False}
        )

        if result.success or hasattr(result, 'x'):
            a_candidate = norm_sum1(result.x)
            m_candidate = fast_autocorr(a_candidate).max()
            if m_candidate < best_m:
                best_m = m_candidate
                best_a = a_candidate.copy()

        # Also check the raw result
        a_candidate = norm_sum1(result.x)
        m_candidate = fast_autocorr(a_candidate).max()
        if m_candidate < best_m:
            best_m = m_candidate
            best_a = a_candidate.copy()

        # Update a for next beta
        a = best_a

        if verbose:
            J_now = 2.0 * n * best_m
            print(f"    L-BFGS beta={beta:.1f} done: J={J_now:.6f}, M={best_m:.2e}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=None, use_all=False):
    """LP-based Frank-Wolfe with adaptive constraint selection using actual values."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    if use_all:
        K_sel = 2 * n - 1
    elif top_K is not None:
        K_sel = min(top_K, 2 * n - 1)
    else:
        K_sel = min(300, 2 * n - 1)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            # Select constraints closest to M (most binding)
            margins = M_curr - ac
            binding_idx = np.argsort(margins)[:K_sel]

        # Build constraint matrix
        T_sel = np.zeros((len(binding_idx), n))
        for row, k in enumerate(binding_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                i_vals = np.arange(i_start, i_end + 1)
                partner = ki - i_vals
                T_sel[row, i_vals] = 2.0 * y[partner]

        # Use actual autocorrelation values as bounds (tighter for active constraints)
        M_sel = ac[binding_idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel, cp.sum(b_var) >= 1e-10]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.2e}")

        if not use_all and t % 15 == 0:
            K_sel = min(K_sel + 30, 2 * n - 1)

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=123, n_top=20):
    """Aggressive coordinate descent with line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity map
        sens = np.zeros(n)
        in_peak_mask = np.zeros(n, dtype=bool)
        for k in top_peaks[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak_mask[idx] = True
            partner = ki - idx
            sens[idx] += a[partner]

        improved = False

        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-25:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-14:
                continue

            # Find best destination
            candidates = np.where(~in_peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            # Try several destinations
            top_dst = candidates[np.argsort(cand_sens)[:5]]
            top_dst = [d for d in top_dst if d != src]

            found = False
            for dst in top_dst:
                if time.monotonic() - t0 > time_limit:
                    break
                # Try multiple fractions with best-first
                for frac in [0.01, 0.005, 0.02, 0.002, 0.03, 0.05, 0.001, 0.08]:
                    amt = src_val * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-18:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

            if not found and stall > 40:
                # Random kick
                j = int(rng.integers(n))
                if j != src:
                    frac = rng.uniform(0.003, 0.015)
                    amt = src_val * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Multi-point random perturbation
            n_pert = rng.integers(2, 6)
            srcs = rng.integers(0, n, n_pert)
            dsts = rng.integers(0, n, n_pert)
            a_trial = a.copy()
            for s_i, d_i in zip(srcs, dsts):
                if s_i != d_i and a_trial[s_i] > 1e-12:
                    amt = a_trial[s_i] * rng.uniform(0.002, 0.01)
                    a_trial[s_i] -= amt
                    a_trial[d_i] += amt
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() > 1e-15:
                a_trial = a_trial / a_trial.sum()
                M_new = fast_autocorr(a_trial).max()
                if M_new < best_m:
                    a = a_trial
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_init(n, n_modes=20, seed=0):
    """Generate a smooth non-negative sequence via Fourier synthesis."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):
    """U-shaped distribution."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coordinate descent (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.10, 180.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t3, verbose=True, rng_seed=123, n_top=20)
        try_update(a_out, "coord_p3")

    # === Phase 4: L-BFGS with different parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: L-BFGS round 2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = lbfgs_smooth_optimize(a_in, t4, top_k=100, beta_start=50.0,
                                         beta_end=400.0, verbose=True, rng_seed=100)
        try_update(a_out, "lbfgs_p4")

    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP-FW round 2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2500)
        top_K = min(3000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True, top_K=top_K, use_all=use_all)
        try_update(a_out, "lp_p5")

    # === Phase 6: Coordinate descent round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coord round 2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t6, verbose=True, rng_seed=456, n_top=25)
        try_update(a_out, "coord_p6")

    # === Phase 7: Multi-restart with perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Multi-restart (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t7 = min(remaining * 0.18, 300.0)
    if t7 > 60:
        rng = np.random.default_rng(777)
        n_restarts = min(5, int(t7 // 60))
        t_per_restart = t7 / max(n_restarts, 1)

        for r in range(n_restarts):
            elapsed = time.monotonic() - t_global
            if elapsed > time_budget - 60:
                break
            t_avail = t_per_restart - 5
            if t_avail < 30:
                break

            # Different perturbation strategies
            a_base = norm_sum1(best_seq)
            if r == 0:
                # Small Gaussian perturbation
                pert = rng.normal(0, 0.02, n_best) * a_base
                a_init = a_base + pert
            elif r == 1:
                # Medium perturbation
                pert = rng.normal(0, 0.05, n_best) * a_base
                a_init = a_base + pert
            elif r == 2:
                # Blend with U-shape
                a_init = 0.7 * a_base + 0.3 * make_u_shape(n_best, 2.0)
            elif r == 3:
                # Blend with spectral
                a_init = 0.6 * a_base + 0.4 * spectral_init(n_best, n_modes=15, seed=500 + r)
            else:
                # Sparse perturbation
                a_init = a_base.copy()
                n_sparse = n_best // 10
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_init[idx] *= rng.uniform(0.5, 1.5, n_sparse)

            a_init = norm_sum1(a_init)

            if verbose := True:
                sc_init = evaluate_safe(a_init)
                print(f"    Restart {r}: init score={sc_init:.6f}")

            # Optimize with L-BFGS then coord
            t_avail = min(t_avail, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail > 20:
                t_lbfgs = t_avail * 0.5
                a_out, _ = lbfgs_smooth_optimize(a_init, t_lbfgs, top_k=60,
                                                 beta_start=40.0, beta_end=150.0,
                                                 verbose=False, rng_seed=777 + r)
                try_update(a_out, f"restart_{r}_lbfgs")

                elapsed = time.monotonic() - t_global
                t_remain = time_budget - 60 - elapsed
                if t_remain > 20:
                    t_coord = min(t_avail * 0.4, t_remain, 80.0)
                    a_in = norm_sum1(best_seq)
                    a_out, _ = coordinate_descent(a_in, t_coord, verbose=False,
                                                  rng_seed=2000 + r, n_top=15)
                    try_update(a_out, f"restart_{r}_coord")

    # === Phase 8: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Length exploration (base n={n_best}) ===")

    trial_lengths = []
    for delta in [300, -300, 600, -600, 900]:
        new_n = n_best + delta
        if 1000 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 80]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick L-BFGS
        t8a = min(remaining * 0.30, 150.0)
        if t8a > 30:
            a_out, _ = lbfgs_smooth_optimize(a_long, t8a, top_k=60,
                                             beta_start=30.0, beta_end=150.0,
                                             verbose=False, rng_seed=new_n % 10000)
            try_update(a_out, f"lbfgs_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 3500:
            t8b = min(remaining * 0.25, 100.0)
            if t8b > 20:
                a_in = norm_sum1(best_seq)
                use_all = (new_n <= 2000)
                top_K = min(1500, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t8b, solver, verbose=False,
                                          top_K=top_K, use_all=use_all)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 9: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
    if remaining > 60:
        a_in = norm_sum1(best_seq)

        # L-BFGS final
        t9a = min(remaining * 0.30, 150.0)
        if t9a > 30:
            a_out, _ = lbfgs_smooth_optimize(a_in, t9a, top_k=100,
                                             beta_start=80.0, beta_end=300.0,
                                             verbose=False, rng_seed=9999)
            try_update(a_out, "lbfgs_final")

        # LP final
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and n_best <= 5000:
            t9b = min(remaining * 0.30, 150.0)
            if t9b > 20:
                a_in = norm_sum1(best_seq)
                use_all = (n_best <= 2500)
                top_K = min(2 * n_best - 1, 4000)
                a_out, _ = lp_frank_wolfe(a_in, t9b, solver, verbose=True,
                                          top_K=top_K, use_all=use_all)
                try_update(a_out, "lp_final")

        # Coord final
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t9c = min(remaining * 0.40, 100.0)
            if t9c > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_descent(a_in, t9c, verbose=False,
                                              rng_seed=5555, n_top=30)
                try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
