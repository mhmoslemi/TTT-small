        # Let m = n-1-j, j = n-1-m: = sum_m a[m]*p[k-n+1+m]
        # For k = i+n-1: = sum_m a[m]*p[i+m] ✓
        
        # So grad = convolve(a[::-1], p, mode='full')[n-1:n-1+n]
        
        a_flip = a[::-1]
        conv_flip_p = np.convolve(a_flip, p)  # length n + (2n-1) - 1 = 3n-2
        grad = conv_flip_p[n-1:n-1+n]  # length n
        grad = 2 * beta * grad
        
        # Gradient descent with projection
        lr_t = lr / (1 + 0.01 * t)
        a_new = a - lr_t * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Check improvement
        b_new = np.convolve(a_new, a_new)
        max_new = b_new.max()
        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_max

def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def spectral_flatten(a, gamma=0.5, n_iters=100, rng=None):
    """Flatten the power spectrum |FFT(a)|^2 using iterative normalization."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = norm_sum1(a)
    fft_size = 1 << (n - 1).bit_length()
    
    for _ in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        # Target: flat power spectrum
        # Adjust: multiply by (target / power)^gamma
        mean_power = power[1:].mean()  # exclude DC
        target = np.ones_like(power)
        target[1:] = mean_power  # keep DC as is, flatten the rest
        # Actually, let's flatten everything including DC
        target = np.ones_like(power) * power.mean()
        
        # Blend to avoid instability
        scale = (target / (power + 1e-30)) ** gamma
        A_new = A * scale
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    
    return a


def smoothed_gradient_optimize(a0, time_limit, verbose=True, seed=42):
    """
    Minimize max(conv(a,a)) using log-sum-exp smoothing with adaptive beta.
    Uses Adam optimizer with projection onto the probability simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    # Adaptive parameters
    base_beta = 5.0  # smoothing parameter
    max_beta = 100.0
    base_lr = 0.05
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Adaptive beta: increases over time
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta_t = base_beta + (max_beta - base_beta) * progress ** 2
        
        # Learning rate decay
        lr = base_lr / (1.0 + 0.005 * t)
        
        # Compute self-convolution
        b = fast_autocorr(a)
        
        # Compute softmax weights
        beta_b = beta_t * b
        max_bb = beta_b.max()
        exp_bb = np.exp(beta_b - max_bb)
        p = exp_bb / exp_bb.sum()  # length 2n-1
        
        # Compute gradient: grad[i] = 2*beta * sum_j a[j] * p[i+j]
        # Using FFT: grad = 2*beta * (ifft(fft(a) * conj(fft(p)))[proper slice])
        # Actually, let's use the convolution formula:
        # grad[i] = 2*beta * sum_{j=0}^{n-1} a[j] * p[i+j]
        # = 2*beta * (a_flip * p)[i + n - 1] where a_flip = a[::-1]
        a_flip = a[::-1]
        fft_size = 1 << (n + (2*n-1) - 1 - 1).bit_length()
        Af = np.fft.rfft(a_flip, fft_size)
        Pf = np.fft.rfft(p, fft_size)
        conv_result = np.fft.irfft(Af * Pf, fft_size)
        grad = 2.0 * beta_t * conv_result[n-1:n-1+n]
        
        # Adam update
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t)
        v_hat = v_state / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Track best
        b_new = fast_autocorr(a_new)
        m_new = b_new.max()
        if m_new < best_m * (1 - 1e-10):
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Periodic restart to escape local minima
        if t % 20000 == 0:
            # Small random perturbation
            pert = rng.normal(0, 0.01, n) * a
            a = a + pert
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    GradOpt t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta_t:.1f}")
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GradOpt t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta_t:.1f}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200, margin_frac=1e-3):
    """LP-based Frank-Wolfe method for minimizing max(conv(a,a))."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        # Select active constraints (peaks close to the maximum)
        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()
        
        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]
        
        K = len(active_idx)
        
        # Build constraint matrix: for each active lag k, the constraint is
        # sum_i y[i] * b_var[k-i] <= M_curr (where b_var is the LP variable)
        # b_var[k-i] is valid when 0 <= k-i < n
        # So: sum_{i: 0<=k-i<n} y[i] * b_var[k-i] <= M_curr
        # Let j = k-i: sum_{j: 0<=k-j<n} y[k-j] * b_var[j] <= M_curr
        # For each active k: T[k, j] = y[k-j] if 0 <= k-j < n, else 0
        
        T = np.zeros((K, n))
        for row, k in enumerate(active_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            for j in range(j_start, j_end):
                T[row, j] = y[k_int - j]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            # Tighten margin when improving
            margin_frac = max(margin_frac * 0.85, 1e-5)
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")
    
    return best_a, best_m


def peak_crush(a0, time_limit, verbose=True, seed=123):
    """
    Targeted peak-crushing: identify the maximum peak in the self-convolution
