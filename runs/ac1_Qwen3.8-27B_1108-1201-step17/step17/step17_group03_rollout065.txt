"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LP Frank-Wolfe
with adaptive constraint selection, heavy-ball extrapolation, peak crushing,
and multi-length exploration.
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(8, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print("  Using LP solver: " + s)
                return _global_solver
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  Using LP solver (fallback): SCS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64).ravel(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def build_T(a, active_idx, n):
    K = len(active_idx)
    T = np.zeros((K, n), dtype=np.float64)
    for r in range(K):
        k = int(active_idx[r])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start > j_end:
            continue
        T[r, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return T


def interp_seq(f, new_n):
    f = np.asarray(f, dtype=np.float64).ravel()
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, fn)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_new / ss


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'beta_3_3':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 3 * (1 - x) ** 3)
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def lp_fw(a0, time_limit, solver, verbose=True,

            continue

        if sx < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / sx

        mx = fast_autocorr(x_new).max()
        if mx < best_m:
            best_m = mx
            best_a = x_new.copy()

        x_prev = x
        x = x_new

        if verbose and (t % 400 == 0 or t == 1):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print("    LP-FW t=%d (%ds): J=%.6f, K=%d, margin=%.1e"

    return best_a, best_m
