This is exactly the "soft subgradient" approach used in the previous code. The key parameter is epsilon (or its reciprocal, beta).

    """Compute conv(a, a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

Now let me think about the coordinate descent approach more carefully. The key is to make it fast and effective.

For a 2-point move (decrease at i, increase at j by delta):
- The change in conv(a,a)_k is approximately:
  delta * (a[k-j] - a[k-i]) + delta * (a[j-k] - a[i-k])
  = delta * (a[k-j] + a[j-k] - a[k-i] - a[i-k])
  
  where we use a[m] = 0 for m outside [0, n-1].

Wait, let me be more careful. If a_new = a + delta*(e_j - e_i), then:
conv(a_new, a_new)_k = conv(a,a)_k + delta*(conv(e_j, a)_k + conv(a, e_j)_k - conv(e_i, a)_k - conv(a, e_i)_k) + O(delta^2)

conv(e_j, a)_k = a[k-j] (if 0 <= k-j < n)
conv(a, e_j)_k = a[k-j] (if 0 <= k-j < n) [since conv is commutative for the same sequences]

Wait, conv(a, e_j)_k = sum_l a[l] * e_j[k-l] = a[k-j] (since e_j[m] = 1 if m=j, else 0, so we need k-l = j, i.e., l = k-j).

So conv(e_j, a)_k + conv(a, e_j)_k = 2*a[k-j] (if 0 <= k-j < n), else 0.

Similarly for e_i.

So the first-order change at lag k is:
2*delta * (a[k-j] - a[k-i]) (with a[m] = 0 for m outside [0,n-1])

This is very useful! We can use this to quickly estimate the effect of a move without computing the full convolution.

def coordinate_descent_fast(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Precompute: for each position, the "profile" of how it affects each lag
    # profile[j][k] = a[k-j] + a[j-k] (the gradient contribution)
    # But this is n x (2n-1) which is too large. Instead, compute on the fly.
    
    peels = 0
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Find the peak lag(s)
        peak_lag = np.argmax(ac)
        
        # For the peak lag, find the best 2-point move
        k = int(peak_lag)
        js = np.arange(max(0, k - n + 1), min(n, k + 1))
        if len(js) == 0:
            stall += 1
            continue
        
        # Contribution at peak lag: a[j] * a[k-j]
        contrib = a[js] * a[k - js]
        
        # Source: position with max contribution
        src_local = np.argmax(contrib)
        src = js[src_local]
        partner = k - src
        
        if a[src] < 1e-12:
            stall += 1
            continue
        
        # Find best destination: minimize increase at peak lag and other top lags
        # For destination d, the change at peak lag k is: 2*delta*(a[k-d] - a[k-src])
        # We want this to be negative: a[k-d] < a[k-src]
        # Also want to not increase other top lags
        
        top_lags = np.argsort(ac)[::-1][:min(15, 2*n-1)]
        
        # Score each candidate destination
        # For destination d, the worst-case increase across top lags is:
        # max_k 2*delta*(a[k-d] - a[k-src])
        # We want this to be as negative as possible
        
        # Vectorized computation
        # For each top lag k2, the contribution of moving from src to d is:
        # 2*delta*(a[k2-d] - a[k2-src])
        # We want: a[k2-d] < a[k2-src] for as many k2 as possible
        
        # Simplified: choose d that minimizes a[k-d] (the value at the "mirror" position)
        # This directly reduces the peak lag contribution
        
        # Compute a[k-d] for all valid d
        # k-d is in [0, n-1] when d is in [k-n+1, k] (intersected with [0, n-1])
        
        best_dst = -1
        best_score = float('inf')
        
        # Sample a subset of candidates for speed
        if n > 1000:
            candidates = rng.choice(n, size=min(n, 200), replace=False)
        else:
            candidates = np.arange(n)
        
        for d in candidates:
            d = int(d)
            if d == src or d == partner:
                continue
            if a[d] > 0.5:  # Skip positions that are already large
                continue
            
            # Score: how much does this reduce the peak?
            # At peak lag k: change = 2*delta*(a[k-d] - a[k-src])
            # We want a[k-d] to be small
            val_peak = 0.0
            if 0 <= k - d < n:
                val_peak = a[k - d]
            val_src = 0.0
            if 0 <= k - src < n:
                val_src = a[k - src]
            
            # Penalty for other lags
            penalty = 0.0
            for k2 in top_lags[:8]:
                k2 = int(k2)
                v_d = a[k2 - d] if 0 <= k2 - d < n else 0.0
                v_src = a[k2 - src] if 0 <= k2 - src < n else 0.0
                penalty += max(0, v_d - v_src)
            
            score = val_peak + 0.3 * penalty
            if score < best_score:
                best_score = score
                best_dst = d
        
        if best_dst < 0:
            stall += 1
            continue
        
        # Try different amounts
        improved = False
        for frac in [0.02, 0.01, 0.005, 0.05, 0.003, 0.08, 0.015]:
            amt = a[src] * frac
            if amt < 1e-14:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[best_dst] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial /= a_trial.sum()
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                peels += 1
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random kick
            if stall % 20 == 0:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based active set optimization (targeted peak reduction)

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")
    return _global_solver


def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def lp_active_set(a0, time_limit, solver, verbose=True):
    """LP-based active set: minimize linearized max autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    iters = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        iters += 1
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Active lags
        tol = max(1e-14, M * 1e-6)
        active = np.where(ac >= M - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(ac))])
        if len(active) > 30:
            active = active[np.argsort(ac[active])[::-1]][:30]
        
        # Adaptive eps
        eps = max(0.001, 0.02 * M / (a.max() + 1e-30))
        eps = min(eps, 0.1)
        
        d_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(d_var) == 0, d_var <= eps, d_var >= -eps, a + d_var >= 0]
        
        for k in active:
            ki = int(k)
            js = np.arange(max(0, ki - n + 1), min(n, ki + 1))
            if len(js) == 0:
                continue
            g = 2.0 * a[ki - js]
            cons.append(g @ d_var[js] <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            stall += 1
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stall += 1
            continue
        
        d_val = np.asarray(d_var.value, dtype=np.float64)
        t_val = float(t_var.value)
        
        if not np.isfinite(t_val) or t_val > -1e-13:
            stall += 1
            continue
        
        a_new = a + d_val
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try half step
            a_half = a + 0.5 * d_val
            a_half = np.maximum(a_half, 0.0)
            a_half /= a_half.sum()
            M_half = fast_autocorr(a_half).max()
            if M_half < best_m - 1e-15:
                best_m = M_half
                best_a = a_half.copy()
                a = a_half
                stall = 0
            else:
                stall += 1
                a = a_half
        
        if verbose and iters % 200 == 0:
            el = time.monotonic() - t0
            print(f"    LP-AS it={iters} ({el:.0f}s) J={2*n*best_m:.6f} stall={stall}")
    
    return best_a, best_m


def fast_coord_descent(a0, time_limit, verbose=True, seed=42):
    """Fast coordinate descent using first-order sensitivity."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    peels = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 400:
        ac = fast_autocorr(a)
        M = ac.max()
        top_lags = np.argsort(ac)[::-1][:min(12, 2*n-1)]
        
        improved = False
        for k in top_lags[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = np.arange(max(0, ki - n + 1), min(n, ki + 1))
            if len(js) == 0:
                continue
            contrib = a[js] * a[ki - js]
            src = js[np.argmax(contrib)]
            partner = ki - src
            if a[src] < 1e-12:
                continue
            
            # Find best destination
            # Score: minimize a[ki-d] (reduces peak) + penalty from other top lags
            cand = rng.choice(n, size=min(n, 300), replace=False)
            best_d = -1
            best_score = 1e10
            for d in cand:
                d = int(d)
                if d == src or d == partner:
                    continue
                # Primary: reduce at peak lag
                v_peak = a[ki - d] if 0 <= ki - d < n else 0.0
                # Secondary: penalty from other lags
                pen = 0.0
                for k2 in top_lags[1:6]:
                    k2i = int(k2)
                    v = a[k2i - d] if 0 <= k2i - d < n else 0.0
                    v_s = a[k2i - src] if 0 <= k2i - src < n else 0.0
                    pen += max(0.0, v - v_s)
                score = v_peak + 0.5 * pen
                if score < best_score:
                    best_score = score
                    best_d = d
            
            if best_d < 0:
                continue
            
            for frac in [0.015, 0.008, 0.025, 0.005, 0.04, 0.01]:
                amt = a[src] * frac
                if amt < 1e-14:
                    continue
                at = a.copy()
                at[src] -= amt
                at[best_d] += amt
                at = np.maximum(at, 0.0)
                at /= at.sum()
                Mt = fast_autocorr(at).max()
                if Mt < best_m - 1e-14:
                    a = at
                    best_m = Mt
                    best_a = at.copy()
                    improved = True
                    peels += 1
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            if stall % 30 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    amt = a[i] * 0.004
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    Mn = fast_autocorr(a).max()
                    if Mn < best_m:
                        best_m = Mn
                        best_a = a.copy()
                        stall = 0
        
        if verbose and peels % 200 == 0:
            el = time.monotonic() - t0
            print(f"    CD ({el:.0f}s) J={2*n*best_m:.6f} peels={peels} stall={stall}")
    
    return best_a, best_m


def mirror_descent_soft(a0, time_limit, verbose=True, seed=0):
    """Soft-subgradient mirror descent with adaptive beta."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    x = a.copy()
    x_prev = a.copy()
    beta = 6.0
    c0 = 4.0
    
    for it in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        bt = (it - 1.0) / (it + 2.0)
        y = x + bt * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y /= y.sum()
        
        # Soft subgradient
        ac = fast_autocorr(y)
        M = ac.max()
        shifted = ac - M
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w /= ws
        w[w < 1e-10] = 0.0
        ws2 = w.sum()
        if ws2 < 1e-15:
            w = np.ones(len(ac)) / len(ac)
        else:
            w /= ws2
        
        # Gradient via FFT
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        a_rev = np.zeros(fft_size)
        a_rev[:n] = y[::-1]
        w_pad = np.zeros(fft_size)
        w_pad[:len(w)] = w
        conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
        grad = 2.0 * conv[n-1:2*n-1]
        
        eta = c0 / (np.sqrt(it) + 1.0)
        log_y = np.log(y)
        log_new = log_y - eta * grad
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        x_prev = x.copy()
        x = a_new
        
        Mx = fast_autocorr(x).max()
        if Mx < best_m - 1e-13:
            best_m = Mx
            best_a = x.copy()
        
        # Adaptive beta: increase focus on peaks
        if it % 500 == 0:
            beta = min(15.0, beta + 0.1)
        
        # Restart if stalled
        if it % 3000 == 0 and Mx > best_m * 1.01:
            x = best_a + rng.normal(0, 0.02, n) * best_a
            x = np.maximum(x, 1e-8)
            x /= x.sum()
            x_prev = x.copy()
        
        if verbose and it % 5000 == 0:
            el = time.monotonic() - t0
            print(f"    MD it={it} ({el:.0f}s) J={2*n*best_m:.6f} beta={beta:.1f}")
    
    return best_a, best_m


def make_init(n, seed, mode="symmetric"):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if mode == "symmetric":
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        for f in [2, 3, 4]:
            a += rng.normal(0, 0.05) * np.sin(2*np.pi*f*x + rng.uniform(0, 2*np.pi))
    elif mode == "smooth":
        a = np.ones(n) * 0.5
        for f in [1, 2, 3, 5, 7]:
            a += rng.normal(0, 0.1) * np.sin(2*np.pi*f*x + rng.uniform(0, 2*np.pi))
    elif mode == "bump":
        center = n // 2
        width = n // 4
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return norm_sum1(a)


def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_i = np.interp(new_x, old_x, f)
    f_i = np.maximum(f_i, 0.0)
    s = f_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_i / s

            break
        new_n = nb + delta
        if new_n < 1000 or new_n > 5000:
            continue
        print(f"\n=== Phase 6: Length n={new_n} (rem={rem:.0f}s) ===")
        a_interp = interpolate_seq(best_seq, new_n)
        t6 = min(rem * 0.3, 120.0)
        if t6 > 15:
            a_out, _ = lp_active_set(a_interp, t6, solver, verbose=False)
            try_update(a_out, f"lp_n{new_n}")
            rem2 = time_max - (time.monotonic() - t_global)
            if rem2 > 30:
                a_out2, _ = fast_coord_descent(a_out, min(rem2 * 0.3, 60.0), verbose=False, seed=99)
                try_update(a_out2, f"cd_n{new_n}")
    
    # === Phase 7: Final LP polish ===
    rem = time_max - (time.monotonic() - t_global)
    nb = len(best_seq)
    if rem > 30 and nb <= 4000:
        print(f"\n=== Phase 7: Final LP (n={nb}, rem={rem:.0f}s) ===")
        t7 = min(rem * 0.7, 150.0)
        if t7 > 15:
            a_out, _ = lp_active_set(best_seq, t7, solver, verbose=True)
            try_update(a_out, "lp_final")
    
    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        best_seq = make_init(3200, seed=1, mode="symmetric")
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
