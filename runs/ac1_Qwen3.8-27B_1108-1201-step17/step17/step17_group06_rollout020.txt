def smooth_grad_descent(a0, time_limit, tau_start=0.05, tau_end=0.001, 
                        eta_start=1.0, verbose=True):
    """GD on smooth approximation of max(C) with simplex projection."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)

    beta = 0.9
    eps = 1e-8
    
    # τ schedule parameters
    n_est_iterations = 50000  # estimate
    tau_log_start = np.log(tau_start)
    tau_log_end = np.log(tau_end)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Current τ
        frac = min(t / n_est_iterations, 1.0)
        tau = np.exp(tau_log_start + frac * (tau_log_end - tau_log_start))
        
        # Compute C and gradient
        C = fast_conv(a)
        g = compute_smooth_gradient(a, C, tau)
        
        # RMSProp update
        v = beta * v + (1 - beta) * g**2
        step = eta_start / (np.sqrt(v) + eps) * g
        
        # Line search (simplified: try a few step sizes)
        best_step = 0.0
        best_f = None
        for s in [1.0, 0.5, 0.25, 0.1]:

            a_try = np.maximum(a_try, 0)
            s_sum = a_try.sum()
            if s_sum < 1e-15:
                continue
            a_try /= s_sum
            # Evaluate smooth objective (cheap: just compute C and log-sum-exp)
            C_try = fast_conv(a_try)
            C_shift = C_try - C_try.max()
            f_try = tau * np.log(np.sum(np.exp(C_shift / tau)))
            if best_f is None or f_try < best_f:
                best_f = f_try
                best_step = s
        
        if best_step > 0:
            a = a - best_step * step
            a = np.maximum(a, 0)
            a /= a.sum()
        
        # Check actual objective periodically
        if t % 100 == 0 or t < 20:
            J = eval_J(a)
