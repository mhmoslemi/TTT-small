"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Theoretically-motivated initializations (inverse-sqrt ramps, binomial-coefficient shapes)
2. Nesterov-accelerated mirror descent with adaptive soft-max (log-sum-exp) subgradient
3. LP-based Frank-Wolfe with adaptive constraint activation
4. Adaptive coordinate descent with simulated annealing and peak-lookahead
5. Multi-resolution (coarse-to-fine) optimization across sequence lengths
6. Targeted second-peak suppression for fine-tuning
"""

import numpy as np
import time

_global_best = None
_global_best_score = float('inf')


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_softmax_subgradient(a, ac, temperature=1.0):
    n = len(a)
    ac_safe = np.maximum(ac, 1e-30)
    max_ac = ac_safe.max()
    shifted = (ac_safe - max_ac) / temperature
    max_shifted = shifted.max()
    weights = np.exp(shifted - max_shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones_like(ac) / len(ac)
    else:
        weights = weights / wsum
    grad = np.zeros(n)
    threshold = 1e-10
    active = np.where(weights > threshold)[0]
    for k in active:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * weights[k_int] * a[j_idx]
    return grad


def compute_topk_subgradient(a, ac, top_k=80, beta=4.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / wsum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, mode='topk', top_k=60, beta=4.0, c0=4.0,
                 temperature0=2.0, temp_decay=0.9995, restart_every=6000,
                 rng_seed=42, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    temp = temperature0
    stall = 0
    restart_count = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        if mode == 'softmax':
            g = compute_softmax_subgradient(y, ac, temperature=temp)
        else:
            g = compute_topk_subgradient(y, ac, top_k=top_k, beta=beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        temp = max(temp * temp_decay, 0.05)

        if stall > restart_every:
            stall = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.02 * restart_count)
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            temp = temperature0 * (0.3 + 0.1 * min(restart_count, 5))
            gamma = c0 * (1.0 + 0.15 * min(restart_count, 4))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_M
                print(f"    AMD restart#{restart_count} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_M


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=500, max_iters=5000):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    import cvxpy as cp

    y = a.copy()
    prev_y = a.copy()
    x_mom = a.copy()

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = np.zeros((K_sel, n))
        for ii, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                T_sel[ii, idx] = y

        M_sel = M_curr * np.ones(K_sel)
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta_t = (t - 1.0) / (t + 2.0)
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx

        prev_y = y
        y = y_new
        x_mom = x_new

        for candidate in [y, x_mom]:
            Mc = fast_autocorr(candidate).max()
            if Mc < best_M - 1e-15:
                best_M = Mc
                best_a = candidate.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_M


def coord_descent_adaptive(a0, time_limit, verbose=True, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()

    stall = 0
    max_stall = 300
    temp = 0.05

    for p in range(n):
        peak_order[p] = (n // 2) + (p % 2) * (1 if p % 4 < 2 else -1) * ((p + 1) // 2)
        peak_order[p] = max(0, min(2 * n - 2, peak_order[p]))

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)

            i_end = min(n - 1, k_int)

                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            top_src = idx[np.argsort(contrib)[::-1][:8]]

            low_positions = np.where(a < np.median(a))[0]

                        continue
                    for frac in [0.02, 0.05, 0.1, 0.15]:
                        amt = a[src] * frac

                        a_trial[src] -= amt

                            if rng.random() < 0.3:
                                a = a_trial
                                improved = True
                                stall = 0
                            break
                        else:
                            break

                        break
                if improved:
                    break
            if improved:
                break

        if not improved:

            i = int(rng.integers(n))

def make_inv_sqrt_ramp(n):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(x)

def make_power_law(n, power=1.0, center=False):
    x = np.arange(1, n + 1, dtype=np.float64)
    if center:
        x = np.abs(x - (n + 1) / 2.0) + 1.0
        a = x ** (-power)
    else:
        a = x ** power
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_binomial_shape(n, shape_param=1.5):
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = np.exp(-shape_param * x)
    return norm_sum1(a)


def make_bipolar_smooth(n, transition=0.3):
    x = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-20.0 * (x - transition)))

    return norm_sum1(a)


def make_random_smooth(n, seed=0, n_freq=10):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):

        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)

        labels = unique_labels

        t_per = max(8.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 6:
                break
            if idx % 2 == 0:
                a_out, _ = amd_optimize(init, t_avail, mode='topk', top_k=60,
                                         beta=4.0, c0=4.5, rng_seed=idx)
            else:
                a_out, _ = amd_optimize(init, t_avail, mode='softmax',
                                         temperature0=3.0, temp_decay=0.999,
                                         c0=4.0, rng_seed=idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP Frank-Wolfe (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.28, 450.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=400,
                                   max_iters=3000)
        try_update(a_out, f"lp_n{n_best}")

    # === Phase 3: Soft-max AMD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: SoftMax AMD (n={n_best}) ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t3, mode='softmax', temperature0=2.0,
                                 temp_decay=0.9995, c0=3.5, restart_every=5000,
                                 rng_seed=777)
        try_update(a_out, f"softmax_n{n_best}")

    # === Phase 4: Second-peak suppression ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Second-peak suppression (n={n_best}) ===")
    t4 = min(remaining * 0.10, 180.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = second_peak_suppress(a_in, t4, verbose=True, rng_seed=55)
        try_update(a_out, f"sup2_n{n_best}")

    # === Phase 5: Adaptive coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord Descent (n={n_best}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent_adaptive(a_in, t5, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 6: Try different n values ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-n search ===")

    n_candidates = set()
    for delta in [-500, -200, 0, 200, 500, 800, 1200]:
        nn = n_best + delta
        if 2000 <= nn <= 6500:
            n_candidates.add(nn)
    n_candidates = sorted(n_candidates)

    for new_n in n_candidates:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        if new_n == n_best:
            continue

        print(f"  Trying n={new_n}...")
        a_candidate = interpolate_sequence(best_seq, new_n)
        try_update(a_candidate, f"interp_n{new_n}")

        t_sub = min(remaining * 0.25, 100.0)
        if t_sub > 15:
            a_out, _ = amd_optimize(a_candidate, t_sub, mode='topk', top_k=50,
                                     beta=4.5, c0=4.0, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

    # === Phase 7: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 5000:
        t7 = min(remaining * 0.25, 250.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_K=600,
                                       max_iters=2000)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Coord Descent (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_adaptive(a_in, t8, verbose=True, rng_seed=888)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 9: Final short AMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 9: Final AMD (n={n_best}) ===")
        t9 = min(remaining * 0.7, 100.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t9, mode='topk', top_k=100,
                                     beta=3.0, c0=3.0, restart_every=10000,
                                     rng_seed=2024)
            try_update(a_out, f"amd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_inv_sqrt_ramp(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
