def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    next_p2 = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), next_p2)
    return np.fft.irfft(A * A, next_p2)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a)) assuming sum(a) = 1."""
    n = len(a)

def compute_softmax_gradient(a, alpha):
    """
    Compute gradient of 2*n*LSE(alpha * conv(a,a)) at point a (sum a = 1).
    Returns (gradient, convolution, softmax_probs).
    """
    n = len(a)
    # Compute convolution
    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:2 * n - 1]

    # Softmax
    scaled = alpha * c
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()

    # Gradient: g_i = 2*n * sum_j a_j * p_{i+j}
    # Using: g = (reverse(a) * p)[n-1 : 2n-1]
    a_rev = a[::-1]
    # Convolution of a_rev (len n) with p (len 2n-1) gives len 3n-2

    Ar = np.fft.rfft(a_rev, fft_size2)
    P = np.fft.rfft(p, fft_size2)
    h = np.fft.irfft(Ar * P, fft_size2)

    return g, c, p

def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, w1=0.2, w2=0.8, width=30.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - w1) ** 2) + np.exp(-width * (x - w2) ** 2) + 0.001
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length, preserving sum."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """
    Main optimization driver.
    Uses KL mirror descent with full softmax gradient, coordinate descent,
    pair-swap optimization, and length sweeping.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.7f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.7f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3200
        best_a = make_power_ramp(n0, 1.0)
        best_score = eval_J(best_a)
        print(f"Initial: n={len(best_a)}, J={best_score:.7f}")

    n0 = len(best_a)

    # === Phase 1: Mirror descent with softmax gradient on parent ===
    print(f"\n=== Phase 1: MD-Soft on parent (n={n0}) ===")
    elapsed = time.monotonic() - t_global
    t1 = min(time_budget * 0.25, elapsed * 0.25 + 400.0)
    t1 = max(t1, 100.0)
    print(f"  Time allocated: {t1:.0f}s")
    a_out, _ = mirror_descent_softmax(best_a.copy(), t1, alpha_start=2.0,
                                       alpha_end=80.0, c0=8.0, verbose=True)
    try_update(a_out, "md_soft_1")

    # === Phase 2: Coordinate descent refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coord Descent (n={len(best_a)}) [remaining={remaining:.0f}s] ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, _ = coordinate_descent_grad(best_a.copy(), t2, verbose=True, seed=77)
        try_update(a_out, "cd_1")

    # === Phase 3: Extend MD with different alpha range ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: MD-Soft extended (n={len(best_a)}) [remaining={remaining:.0f}s] ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_out, _ = mirror_descent_softmax(best_a.copy(), t3, alpha_start=5.0,
                                           alpha_end=150.0, c0=6.0, verbose=True)
        try_update(a_out, "md_soft_2")

    # === Phase 4: Pair swap optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: PairSwap (n={len(best_a)}) [remaining={remaining:.0f}s] ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_out, _ = pair_swap_optimize(best_a.copy(), t4, verbose=True)
        try_update(a_out, "pairs_1")

    # === Phase 5: Coordinate descent again ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coord Descent 2 (n={len(best_a)}) [remaining={remaining:.0f}s] ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 20:
        a_out, _ = coordinate_descent_grad(best_a.copy(), t5, verbose=True, seed=314)
        try_update(a_out, "cd_2")

    # === Phase 6: Multi-init exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-Init Exploration [remaining={remaining:.0f}s] ===")
    if remaining > 100:
        inits = []
        labels = []

        # Different ramps
        inits.append(make_power_ramp(n0, 0.7))
        labels.append("ramp0.7")
        inits.append(make_power_ramp(n0, 1.3))
        labels.append("ramp1.3")
        inits.append(make_power_ramp(n0, 2.0))
        labels.append("ramp2.0")
        inits.append(make_bimodal(n0, 0.15, 0.85, 40.0))
        labels.append("bimodal")
        inits.append(make_smooth_random(n0, seed=42))
        labels.append("smooth42")

        # Blends with parent
        blend = 0.5 * best_a + 0.5 * make_power_ramp(n0, 1.5)
        inits.append(blend / blend.sum())
        labels.append("blend1")
        blend2 = 0.7 * best_a + 0.3 * make_bimodal(n0, 0.25, 0.75, 50.0)
        inits.append(blend2 / blend2.sum())
        labels.append("blend2")

        t_per_init = max(15.0, (remaining - 60) * 0.5 / len(inits))
        for init, lbl in zip(inits, labels):
            elapsed = time.monotonic() - t_global
            if elapsed > time_budget - 30:
                break
            t_avail = min(t_per_init, time_budget - 30 - elapsed)
            if t_avail < 15:
                break
            print(f"  Multi-Init [{lbl}] ({t_avail:.0f}s)...")
            a_out, _ = mirror_descent_softmax(init, t_avail, alpha_start=3.0,
                                               alpha_end=100.0, c0=7.0)
            try_update(a_out, f"multi_{lbl}")

    # === Phase 7: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Length Sweep [remaining={remaining:.0f}s] ===")
    n_best = len(best_a)
    deltas = [-500, -300, -100, 100, 300, 500, 700, 1000]
    for d in deltas:
        nl = n_best + d
        if nl < 1000 or nl > 6000:
            continue
        if abs(nl - n_best) < 50:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Length n={nl} (delta={d})...")
        a_int = interpolate_seq(best_a, nl)
        J_int = eval_J(a_int)
        # Quick MD
        t_md = min(remaining * 0.2, 80.0)
        if t_md > 10:
            a_out, _ = mirror_descent_softmax(a_int, t_md, alpha_start=3.0,
                                               alpha_end=80.0, c0=7.0)
            try_update(a_out, f"len{nl}")
        # Quick CD
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_int2 = interpolate_seq(best_a, nl)
            t_cd = min(remaining * 0.1, 40.0)
            if t_cd > 10:
                a_out, _ = coordinate_descent_grad(a_int2, t_cd, seed=nl)
                try_update(a_out, f"len{nl}_cd")

    # === Phase 8: Final MD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 8: Final MD Polish (n={n_cur}) [remaining={remaining:.0f}s] ===")
    t8 = min(remaining * 0.3, 250.0)
    if t8 > 20:
        a_out, _ = mirror_descent_softmax(best_a.copy(), t8, alpha_start=10.0,
                                           alpha_end=200.0, c0=5.0,
                                           max_steps=800000, verbose=True)
        try_update(a_out, "final_md")

    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final CD (n={len(best_a)}) [remaining={remaining:.0f}s] ===")
    t9 = min(remaining * 0.35, 200.0)
    if t9 > 15:
        a_out, _ = coordinate_descent_grad(best_a.copy(), t9, verbose=True, seed=271)
        try_update(a_out, "final_cd")

    # === Phase 10: One more MD if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 10: Extra MD [remaining={remaining:.0f}s] ===")
        a_out, _ = mirror_descent_softmax(best_a.copy(), remaining - 10,
                                           alpha_start=15.0, alpha_end=300.0,
                                           c0=4.0, max_steps=1000000)
        try_update(a_out, "extra_md")

    # Final output
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.7f}, n={len(best_a)}, time={elapsed:.0f}s")
        result = [float(x) for x in best_a]
        return result
    else:
        n_fb = 3200
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.7f}")
        return [float(x) for x in fb]
