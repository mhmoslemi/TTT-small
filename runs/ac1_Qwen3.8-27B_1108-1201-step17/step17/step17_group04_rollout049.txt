            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J_normalized(a):
    """Compute 2*n*max(C) assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def subgradient_exact(a, ac):
    """Exact subgradient of max(C) at a."""
    n = len(a)
    k = int(np.argmax(ac))
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k - idx
    grad[idx] = 2.0 * a[j_idx]
    return grad


def subgradient_smoothed(a, ac, top_k=20, beta=3.0):
    """Smoothed subgradient using top-k peaks."""
    n = len(a)
    grad = np.zeros(n)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def adaptive_subgradient(a0, time_limit, verbose=False, seed=42,
                         top_k=20, beta=3.0, exact_frac=0.5,
                         restart_every=3000, c_polyak=3.0):
    """
    Nesterov-accelerated mirror descent with:
    - Correct Parseval lower bound for Polyak step
    - Backtracking line search
    - Adaptive exact/smoothed subgradient mixing
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    LB = 1.0 / (2 * n - 1)  # Parseval lower bound
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_c = c_polyak

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M = ac.max()

        # Adaptive subgradient
        if rng.random() < exact_frac:
            g = subgradient_exact(y, ac)
        else:
            g = subgradient_smoothed(y, ac, top_k=cur_top_k, beta=cur_beta)

        # Polyak step size with correct lower bound
        gap = M - LB
        if gap < 1e-15:
            break
        eta_polyak = cur_c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta_polyak, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)

        # Mirror descent step (log-domain)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        # Backtracking: if the step made things worse, try smaller step
        M_new = fast_autocorr(a_new).max()
        if M_new > M * 1.01 + 1e-15:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1, 0.01]:
                log_a_bt = np.log(y) - eta * factor * g
                max_log_bt = log_a_bt.max()
                a_bt = np.exp(log_a_bt - max_log_bt)
                a_bt = np.maximum(a_bt, 1e-30)
                sbt = a_bt.sum()
                if sbt < 1e-15:
                    continue
                a_bt = a_bt / sbt
                M_bt = fast_autocorr(a_bt).max()
                if M_bt <= M:
                    a_new = a_bt
                    break
            else:
                a_new = y.copy()  # No improvement, stay put

        x_prev = x
        x = a_new

        M_x = fast_autocorr(x).max()
        if M_x < best_M - 1e-15:
            best_M = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with adaptation
        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.005 * min(stall // restart_every, 5)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 5, 60)
            cur_beta = max(cur_beta - 0.2, 1.5)
            cur_c = min(cur_c + 0.3, 6.0)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SGD restart ({elapsed:.0f}s): M={best_M:.8f}, J={2*n*best_M:.6f}")

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.0f}s): M={best_M:.8f}, J={2*n*best_M:.6f}")

    return best_a, best_M


def peak_equalization(a0, time_limit, verbose=False, seed=100):
    """
    Peak equalization: explicitly try to reduce the spread among top peaks.
    At each step, identify the highest peak and the second-highest,
    and move mass to reduce the gap.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = min(10, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = ac[top_idx]
        
        # The spread among top peaks
        spread = top_vals[0] - top_vals[-1]
        
        if spread < 1e-12:
            stall += 1
            continue
        
        # Strategy: reduce the highest peak by moving mass from
        # its contributing positions to positions that boost lower peaks
        k_peak = int(top_idx[0])
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        peak_contributors = np.arange(i_start, i_end)
        
        # Find positions with high contribution to the peak
        contributions = np.zeros(n)
        for i in peak_contributors:
            j = k_peak - i
            contributions[i] = a[i] * a[j]
        
        # Sort contributors by contribution
        contrib_order = np.argsort(contributions)[::-1]
        
        # Try moving a small amount from the top contributor
        improved = False
        for src_pos in contrib_order[:5]:
            if a[src_pos] < 1e-8:
                continue
            
            # Find good destinations: positions that are NOT contributors to the peak
            # but are contributors to lower peaks
            peak_contribution_set = set(peak_contributors.tolist())
            
            # Compute which positions contribute to lower peaks
            lower_peak_contribs = np.zeros(n)
            for k_low in top_idx[2:]:
                k_low_int = int(k_low)
                i_s = max(0, k_low_int - n + 1)
                i_e = min(n, k_low_int + 1)
                for i in range(i_s, i_e):
                    j = k_low_int - i
                    lower_peak_contribs[i] += a[i] * a[j]
            
            # Good destinations: high contribution to lower peaks, low to main peak
            good_scores = lower_peak_contribs - 0.5 * contributions
            good_scores[src_pos] = -np.inf
            good_scores = np.maximum(good_scores, 0)
            
            if good_scores.sum() < 1e-15:
                continue
            
            # Top destination candidates
            n_cands = min(5, n)
            dst_cands = np.argsort(good_scores)[::-1][:n_cands]
            
            for dst in dst_cands:
                if dst == src_pos:
                    continue
                
                # Try different transfer amounts
                for frac in [0.001, 0.003, 0.01, 0.03, 0.05, 0.1]:
                    amt = a[src_pos] * frac
                    if amt < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[src_pos] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-15:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random escape
            if stall % 20 == 0:
                n_pert = max(3, n // 200)
                for _ in range(n_pert):
                    src = int(rng.integers(n))
                    dst = int(rng.integers(n))
                    if src != dst and a[src] > 1e-8:
                        amt = a[src] * rng.uniform(0.001, 0.02)
                        a[src] -= amt
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq ({elapsed:.0f}s): M={best_M:.8f}, J={2*n*best_M:.6f}, stall={stall}")

    return best_a, best_M


def lp_frank_wolfe(a0, time_limit, solver, verbose=False):
    """LP Frank-Wolfe with adaptive constraint selection and backtracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select constraints: all within 50% of max OR top 500
        margins = M_curr - ac
        threshold = M_curr * 0.5  # Include constraints within 50% of peak
        active_mask = margins <= M_curr - threshold  # i.e., ac >= threshold
        n_active = active_mask.sum()
        
        if n_active < 100:
            # Use top 500
            binding_idx = np.argsort(margins)[:500]
        elif n_active > 3000:
            # Use top 2000
            binding_idx = np.argsort(margins)[:2000]
        else:
            binding_idx = np.where(active_mask)[0]
        
        K = len(binding_idx)
        
        # Build constraint matrix
        T_sel = np.zeros((K, n))
        for j in range(n):
            # T_sel[k, j] = y[j] if row k corresponds to lag (j + something)
            # Actually T_sel[k, j] should be such that T_sel @ b gives (y*b)[k]
            # (y*b)[k] = sum_j y[j] * b[k-j]
            # So T_sel[k, j] = y[j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k
            row = binding_idx
            for jj in range(n):
                # For row k, column jj: T[k,jj] = y[jj] if (k - jj) is in [0, n-1]
                # i.e., k - (n-1) <= jj <= k
                pass
        
        # More efficient: build T_sel directly
        # T_sel has shape (K, n), element [k_idx, j] = y[j] if binding_idx[k_idx] - j is in [0, n-1]
        # i.e., binding_idx[k_idx] - (n-1) <= j <= binding_idx[k_idx]
        for k_idx in range(K):
            k = binding_idx[k_idx]
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            T_sel[k_idx, j_min:j_max+1] = y[j_min:j_max+1]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Backtracking line search
        best_alpha = 0
        best_M_try = M_curr
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
            y_try = (1.0 - alpha) * y + alpha * g_dir
            y_try = np.maximum(y_try, 0.0)
            sty = y_try.sum()
            if sty < 1e-15:
                continue
            y_try = y_try / sty
            M_try = fast_autocorr(y_try).max()
            if M_try < best_M_try:
                best_M_try = M_try
                best_alpha = alpha
        
        if best_alpha == 0:
            best_alpha = 2.0 / (t + 2.0)
        
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Nesterov
        if t == 1:
            x = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x = y_new.copy()
            else:
                x = x_new / sx
            y = y_new
        
        # Update best
        for candidate in [y, x]:
            Mc = fast_autocorr(candidate).max()
            if Mc < best_M:
                best_M = Mc
                best_a = candidate.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={2*n*best_M:.6f}, K={K}")

    return best_a, best_M


def symmetric_optimize(a0, time_limit, verbose=False, seed=200):
    """
    Optimize assuming symmetry: a[i] = a[n-1-i].
    This reduces the problem size by half and exploits structure.
    """
    n = len(a0)
    a = norm_sum1(a0)
    
    # Symmetrize
    a_sym = 0.5 * (a + a[::-1])
    a_sym = norm_sum1(a_sym)
    
    best_a = a_sym.copy()
    best_M = fast_autocorr(a_sym).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    n_half = n // 2
    
    # Optimize the first half; mirror to get the full sequence
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a_sym)
        M = ac.max()
        LB = 1.0 / (2 * n - 1)
        gap = M - LB
        
        # Subgradient on the full sequence, then project to symmetric
        g = subgradient_exact(a_sym, ac)
        # Symmetrize gradient
        g_sym = 0.5 * (g + g[::-1])
        
        eta = 3.0 * gap / (np.dot(g_sym, g_sym) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        
        # Mirror descent
        log_a = np.log(np.maximum(a_sym, 1e-15)) - eta * g_sym
        # Enforce symmetry in log domain
        log_a_sym = 0.5 * (log_a + log_a[::-1])
        max_log = log_a_sym.max()
        a_new_sym = np.exp(log_a_sym - max_log)
        a_new_sym = np.maximum(a_new_sym, 1e-30)
        
        # Enforce full symmetry
        a_new_full = np.zeros(n)
        a_new_full[:n_half] = a_new_sym[:n_half]
        a_new_full[n_half:] = a_new_sym[:n_half][::-1] if n % 2 == 0 else a_new_sym[:n_half+1][::-1]
        # Actually, let me just symmetrize
        a_new_full = 0.5 * (a_new_sym + a_new_sym[::-1])
        s_new = a_new_full.sum()
        if s_new < 1e-15:
            a_new_full = a_sym.copy()
        else:
            a_new_full = a_new_full / s_new
        
        M_new = fast_autocorr(a_new_full).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new_full.copy()
            a_sym = a_new_full.copy()
        else:
            # Accept anyway for momentum (like Nesterov)
            a_sym = a_new_full.copy()
    
    return best_a, best_M


def gradient_descent_simple(a0, time_limit, verbose=False, lr=0.001, seed=300):
    """Simple projected gradient descent with adaptive learning rate."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    cur_lr = lr
    stall = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Smoothed subgradient
        g = subgradient_smoothed(a, ac, top_k=30, beta=2.0)
        
        # Gradient step
        a_new = a - cur_lr * g
        
        # Project to simplex (a >= 0, sum = 1)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Decrease learning rate
            if stall % 100 == 0:
                cur_lr *= 0.9
        
        a = a_new
    
    return best_a, best_M


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search: adaptive subgradient with correct lower bound,
    peak equalization, LP Frank-Wolfe with backtracking, and
    symmetric optimization. Multi-phase with diverse restarts.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = np.maximum(np.cos(np.pi * x) ** 2 + 0.01, 0)
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Adaptive subgradient from parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60:
        print(f"\n=== Phase 1: Adaptive SGD from parent (t={t1:.0f}s) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = adaptive_subgradient(a_in, t1, verbose=True, seed=42,
                                        top_k=20, beta=3.0, exact_frac=0.5,
                                        restart_every=3000, c_polyak=3.0)
        try_update(a_out, "sgd_parent")

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J_norm(a):

def subgrad_exact(a, ac):
    n = len(a)
    k = int(np.argmax(ac))
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k - idx
    grad[idx] = 2.0 * a[j_idx]
    return grad


def subgrad_smoothed(a, ac, top_k=20, beta=3.0):
    n = len(a)
    grad = np.zeros(n)

        w = np.ones(k_max) / k_max

        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx

    return grad


def mirror_step(y, g, eta):
    """Log-domain mirror descent step with normalization."""
    log_a = np.log(np.maximum(y, 1e-15)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s
