Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Top-k subgradient cycling for smoother updates

import numpy as np
import time

def fast_autocorr(a):
    """Compute conv(a, a) using FFT. Returns full convolution of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def proj_simplex(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def exact_subgrad(a, top_k=10):
    """
    Compute a subgradient of max(conv(a,a)) using the top-k peak lags
    with uniform weighting. Returns (gradient, max_value).
    """
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    
    # Get top-k lags
    k_min = min(top_k, len(ac))
    # Get indices of top-k values
    top_idx = np.argpartition(ac, -k_min)[-k_min:]
    top_vals = ac[top_idx]
    
    # Weight: uniform over top-k (or could use softmax)
    weights = np.ones(k_min) / k_min
    
    # Compute gradient: sum over top lags of weight * 2*a[lag - j]
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        # g[j] += 2 * w * a[k-j] for valid j (where 0 <= k-j < n)
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            # a[k-j] for j in [j_start, j_end] is a[k_int - j_start : k_int - j_end + 1][::-1]
            idx = k_int - np.arange(j_start, j_end + 1)
            g[j_start:j_end+1] += 2.0 * w * a[idx]
    
    return g, M

def coord_descent(a0, time_limit, verbose=True, seed=123):

        M_curr = ac.max()
        
        # Find peak lags
        top_k = min(5, len(ac))
        top_lags = np.argpartition(ac, -top_k)[-top_k:]
        
        improvement_found = False
        
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue
            
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[k_int - idx]
            
            # Find the source (highest contribution) and try to move it
            src_pos = idx[np.argmax(contrib)]
            src_val = a[src_pos]
            if src_val < 1e-12:
                continue
            
            # Try moving to various destinations
            # Score destinations by how much they'd reduce the peak
            # A good destination is one that's "far" from the source in terms of
            # its contribution to peak lags
            best_dst = None
            best_trial_m = best_m
            
            # Try a random subset of destinations for speed
            n_try = min(50, n)
            candidates = rng.choice(n, size=n_try, replace=False)
            # Always include the farthest position
            farthest = (src_pos + n // 2) % n
            if farthest not in candidates:
                candidates[0] = farthest
            
            for dst in candidates:
                dst_int = int(dst)
                if dst_int == src_pos:
                    continue
                a_dst = a[dst_int]
                if a_dst + src_val < 1e-15:
                    continue
                
                # Try different fractions
                for frac in [0.1, 0.2, 0.05, 0.3]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src_pos] -= amt
                    a_trial[dst_int] += amt
                    a_trial = proj_simplex(a_trial)
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_dst = (dst_int, frac)
            
            if best_dst is not None and best_trial_m < best_m - 1e-15:
                dst_int, frac = best_dst
                amt = a[src_pos] * frac
                a[src_pos] -= amt
                a[dst_int] += amt
                a = proj_simplex(a)
                best_m = best_trial_m
                best_a = a.copy()
                improvement_found = True
                improvements += 1
                stall = 0
                break
        
        if not improvement_found:
            stall += 1
            # Random perturbation every few stalls

                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = proj_simplex(a)
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:

                        best_a = a.copy()

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def make_ramp_init(n, seed=0, direction=1):
    """Create a ramp initialization (non-uniform, asymmetric)."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if direction == 1:
        a = x.copy()  # Increasing
    else:
        a = (1 - x).copy()  # Decreasing
    # Add some noise
    a += rng.normal(0, 0.05, n)
    a = np.maximum(a, 0.0)
    return proj_simplex(a)

def make_dual_ramp_init(n, seed=0):
    """Create a dual-ramp (V or inverted-V) initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # V-shape: high at edges, low in middle

        a = 4.0 * np.abs(x - 0.5)
    a += rng.normal(0, 0.05, n)
    a = np.maximum(a, 0.0)
    return proj_simplex(a)

import numpy as np
import time


def fast_autocorr(a):
    """Compute conv(a, a) using FFT. Returns full convolution of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def proj_simplex(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1)
    cond = u * rho_arr > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_subgrad(a, top_k=10):
    """
    Compute a subgradient of max(conv(a,a)) using the top-k peak lags
    with uniform weighting. Returns (gradient_array, max_value).
    """
    n = len(a)
    ac = fast_autocorr(a)
    M = float(ac.max())

    k_min = min(top_k, len(ac))
    top_idx = np.argpartition(ac, -k_min)[-k_min:]
    weights = np.ones(k_min) / k_min

    g = np.zeros(n)
    a_arr = np.asarray(a, dtype=np.float64)
    for w, k in zip(weights, top_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = k_int - np.arange(j_start, j_end + 1)
            g[j_start:j_end + 1] += 2.0 * w * a_arr[idx]

    return g, M

def coord_descent(a0, time_limit, verbose=True, seed=123, n_dest=30):
    """
    Coordinate descent with pair moves for final polishing.
    """
    n = len(a0)
    a = proj_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_m = float(fast_autocorr(a).max())
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500
    improvements = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = float(ac.max())

        top_k = min(5, len(ac))
        top_lags = np.argpartition(ac, -top_k)[-top_k:]

        improvement_found = False

        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start > j_end:
                continue

            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[k_int - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue

            src_pos = idx[np.argmax(contrib)]
            src_val = a[src_pos]
            if src_val < 1e-12:
                continue

            best_trial_m = best_m
            best_move = None

            candidates = rng.choice(n, size=min(n_dest, n), replace=False)

            for dst in candidates:
                dst_int = int(dst)
                if dst_int == src_pos:
                    continue

                    amt = src_val * frac
                    if amt < 1e-15:
                        continue

                    a_trial[src_pos] -= amt
                    a_trial[dst_int] += amt
                    a_trial = proj_simplex(a_trial)
                    M_trial = float(fast_autocorr(a_trial).max())
                    if M_trial < best_trial_m - 1e-15:
                        best_trial_m = M_trial
                        best_move = (dst_int, frac)

            if best_move is not None and best_trial_m < best_m - 1e-15:
                dst_int, frac = best_move
                amt = a[src_pos] * frac
                a[src_pos] -= amt
                a[dst_int] += amt
                a = proj_simplex(a)
                best_m = best_trial_m
                best_a = a.copy()
                improvement_found = True
                improvements += 1
                stall = 0
                break

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

        return float('inf')


def make_ramp_init(n, seed=0, direction=1, noise_scale=0.08):
    """Create a ramp (asymmetric) initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if direction == 1:
        a = 0.3 + 0.7 * x
    else:
        a = 0.3 + 0.7 * (1.0 - x)
    a += rng.normal(0, noise_scale, n)
    a = np.maximum(a, 0.0)
    return proj_simplex(a)


def make_quadratic_init(n, seed=0):
    """Create a quadratic (asymmetric) initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    shape = rng.integers(4)
    if shape == 0:
        center = rng.uniform(0.2, 0.4)
        a = 1.0 + 2.0 * (x - center) ** 2
    elif shape == 1:
        center = rng.uniform(0.6, 0.8)
        a = 1.0 + 2.0 * (x - center) ** 2
    elif shape == 2:
        a = x ** 2 + 0.5
    else:
        a = (1.0 - x) ** 2 + 0.5
    a += rng.normal(0, 0.04, n)
    a = np.maximum(a, 0.0)
    return proj_simplex(a)


def make_spectral_init(n, seed=0):
    """
    Create initialization with approximately flat spectrum.
    Uses inverse FFT with random phases and takes the positive part.
    """
    rng = np.random.default_rng(seed)
    fft_n = 1 << (n - 1).bit_length()
    # Random phases with flat magnitude
    phases = rng.uniform(0, 2 * np.pi, fft_n // 2 + 1)
    A = np.exp(1j * phases)
    A[0] = 1.0 + 0j
    # Ensure Hermitian symmetry
    A_full = np.zeros(fft_n, dtype=np.complex128)
    A_full[:fft_n // 2 + 1] = A
    A_full[fft_n // 2 + 1:] = np.conj(A[1:fft_n // 2 + 1][::-1])
    sig = np.fft.irfft(A_full)[:n]
    # Take positive part
    a = np.maximum(sig, 0.0)
    s = a.sum()
    if s < 1e-10:
        return proj_simplex(make_ramp_init(n, seed=seed))
    return a / s


def make_blend_init(n, seed=0):
    """Blend of ramp and uniform."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    alpha = rng.uniform(0.3, 0.7)
    a = alpha * (0.2 + 0.8 * x) + (1 - alpha) * np.ones(n) * 0.5
    a += rng.normal(0, 0.03, n)
    a = np.maximum(a, 0.0)
    return proj_simplex(a)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = proj_simplex(parent)
            best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"  Warning: could not load parent: {e}")

    if best_seq is None:
        n_base = 3200
        best_seq = make_ramp_init(n_base, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a_arr = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a_arr.sum()
        if s < 0.01:
            return
        a_norm = a_arr / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: ASG from parent (most important) ===
    print(f"\n=== Phase 1: ASG from parent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 400.0)
    if t_phase1 > 30:
        a_in = proj_simplex(best_seq)
        a_out, _ = accelerated_subgrad(a_in, t_phase1, top_k=8,
                                       c_init=0.08, verbose=True, seed=42)
        try_update(a_out, f"asg_parent_n{n0}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t1b = min(remaining * 0.3, 150.0)
            a_out, _ = accelerated_subgrad(a_in, t1b, top_k=3,
                                           c_init=0.04, verbose=False, seed=142)
            try_update(a_out, f"asg_parent_tight_n{n0}")

    # === Phase 2: ASG from diverse inits ===
    print(f"\n=== Phase 2: ASG diverse inits ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase2 = min(remaining * 0.25, 400.0)

    if t_phase2 > 60:
        inits = []
        labels = []

        inits.append(make_ramp_init(n0, seed=10, direction=1))
        labels.append("ramp_up")
        inits.append(make_ramp_init(n0, seed=11, direction=-1))
        labels.append("ramp_down")
        inits.append(make_quadratic_init(n0, seed=12))
        labels.append("quad")
        inits.append(make_spectral_init(n0, seed=13))
        labels.append("spectral")
        inits.append(make_blend_init(n0, seed=14))
        labels.append("blend")

        # Blend with parent
        if parent is not None:
            p_norm = proj_simplex(parent)
            for seed_b in [15, 16]:
                other = make_ramp_init(n0, seed=seed_b, direction=(seed_b % 2) * 2 - 1)
                blend = 0.5 * p_norm + 0.5 * other
                inits.append(proj_simplex(blend))
                labels.append(f"blend_p_{seed_b}")

        # Remove duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 40)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(20.0, t_phase2 / n_inits)

        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  ASG '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = accelerated_subgrad(init, t_avail, top_k=10,
                                           c_init=0.1, verbose=False, seed=200 + idx)
            try_update(a_out, f"asg_{lbl}")

    # === Phase 3: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord Descent (n={n_best}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_in = proj_simplex(best_seq)
        a_out, _ = coord_descent(a_in, t3, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 4: Second ASG round on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: ASG Round 2 (n={n_best}) ===")
    t4 = min(remaining * 0.2, 250.0)
    if t4 > 30:
        a_in = proj_simplex(best_seq)
        a_out, _ = accelerated_subgrad(a_in, t4, top_k=5,
                                       c_init=0.05, verbose=False, seed=777)
        try_update(a_out, f"asg_r2_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t4b = min(remaining * 0.4, 100.0)
            a_out, _ = accelerated_subgrad(a_in, t4b, top_k=15,
                                           c_init=0.12, verbose=False, seed=888)
            try_update(a_out, f"asg_r2b_n{n_best}")

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length Exploration ===")
    candidate_lengths = sorted(set([
        n_best - 200, n_best - 100, n_best + 100, n_best + 200,
        n_best + 400, 3000, 3500, 4000, 4500
    ]))
    candidate_lengths = [l for l in candidate_lengths if 2000 <= l <= 5000]

    explored = 0
    for new_n in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80 or explored >= 3:
            break
        explored += 1
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.3, 120.0)
        if t5a > 20:
            a_out, _ = accelerated_subgrad(a_interp, t5a, top_k=10,
                                           c_init=0.1, verbose=False,
                                           seed=300 + new_n % 500)
            try_update(a_out, f"asg_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 25:
            t5b = min(remaining * 0.2, 60.0)
            if t5b > 10:
                a_out, _ = coord_descent(a_interp, t5b, verbose=False,
                                         seed=400 + new_n % 500)
                try_update(a_out, f"coord_n{new_n}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Polish (n={n_best}) ===")
    if remaining > 20:
        t6 = min(remaining * 0.5, 150.0)
        if t6 > 10:
            a_in = proj_simplex(best_seq)
            a_out, _ = coord_descent(a_in, t6, verbose=True, seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t6b = min(remaining * 0.5, 80.0)
            if t6b > 10:
                a_in = proj_simplex(best_seq)
                a_out, _ = accelerated_subgrad(a_in, t6b, top_k=3,
                                               c_init=0.02, verbose=False, seed=1234)
                try_update(a_out, f"asg_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_ramp_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
