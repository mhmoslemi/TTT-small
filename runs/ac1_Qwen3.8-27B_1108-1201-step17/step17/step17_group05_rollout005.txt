def _fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _normalize(a):

def _md_optimize(a0, time_limit, verbose=False, beta_init=10.0, beta_min=3.0,

    """

    x_prev = a.copy()
    stall = 0
    cur_beta = beta_init
    iters = 0

        iters += 1
        beta_t = (iters - 1.0) / (iters + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        else:

        w[w < 1e-12] = 0.0

def _lp_fw_optimize(a0, time_limit, solver, verbose=True, max_constraints=None):

    LP-based Frank-Wolfe with Nesterov acceleration and adaptive constraint selection.

            top_idx = np.argsort(ac)[::-1][:K_eff]

        for row, m in enumerate(top_idx):
            m_int = int(m)
            j_start = max(0, m_int - n + 1)
            j_end = min(n - 1, m_int)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            T[row, js] = y[m_int - js]

        # LP: maximize sum(b) s.t. T @ b <= M_curr, b >= 0, sum(b) = 1

            continue

            sx = x_ew.sum()

                x_ew = y_new.copy()

                x_ew = x_ew / sx

        y = y_new

        # Track best
