def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_subgradient_topk(a, ac, top_k=100, beta=3.0):
    """Compute weighted subgradient using top-K peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

            i_end = min(n, k_int + 1)

                T_sel[row, i] = y[k_int - i]

def pairwise_exchange(a0, time_limit, verbose=False, rng_seed=42):

        contrib = a[idx] * a[peak_k - idx]

        n_top = min(10, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]
        
        improved = False
        for li in top_local:
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Try moving mass from src to various destinations
            best_new_m = M_curr
            best_move = None

                amt = src_val * frac

                    continue

                    dst = int(rng.integers(n))
                    if dst == src:
                        continue

                    a_trial[dst] += amt

                    s_t = a_trial.sum()

                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new

                    break

                    best_a = a.copy()

def freq_perturb(a, strength=0.01, rng=None):
    """Perturb in frequency domain."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def polynomial_smooth(a, order=5):
    """Apply polynomial smoothing to reduce high-frequency content."""
    n = len(a)
    x = np.linspace(0, 1, n)
    # Fit a polynomial
    coeffs = np.polyfit(x, a, order)
    a_smooth = np.polyval(coeffs, x)
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s < 1e-15:
        return a.copy()
    return a_smooth / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            break

        progress = min(elapsed / time_limit, 1.0)
        K = int(K_start + (K_end - K_start) * progress)
        K = min(K, 2 * n)
        margin = margin_start * (margin_end / margin_start) ** progress

        margins = M_curr - ac
        K_use = min(K, len(margins))
        active_idx = np.argsort(margins)[:K_use]
        active_idx = [int(k) for k in active_idx]

        T_sel = np.zeros((K_use, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            fallback_count += 1
            if fallback_count > 20:
                break
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            fallback_count += 1
            if fallback_count > 20:
                break
            continue

        lp_count += 1
        fallback_count = 0

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        x_mom = momentum_decay * x_mom + (1.0 - momentum_decay) * y_new
        sx = x_mom.sum()
        if sx < 1e-15:
            x_mom = y_new.copy()
        else:
            x_mom = x_mom / sx

        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()

        if verbose and lp_count % 30 == 0:
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.0f}s, {lp_count} LPs): J={J:.6f}, K={K_use}, margin={margin:.2e}")

    return best_a, best_m


def spectral_subgradient(a0, time_limit, verbose=False, top_K=80, beta=4.0, c0=4.0):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient_topk(y, ac, top_k=top_K, beta=beta)

        eta = c0 / (np.sqrt(t) + 1.0)
        base_m = ac.max()
        accepted = False
        for _ in range(8):
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            else:
                eta *= 0.5
        if not accepted:
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpecGrad t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def pairwise_exchange(a0, time_limit, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 600

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top peaks and their contributors
        n_peaks = 3
        peak_k = int(np.argmax(ac))
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]

        n_top = min(12, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

            for frac in [0.003, 0.008, 0.015, 0.03]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for _ in range(25):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)
                if best_move is not None:
                    break

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.025)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def freq_perturb(a, strength=0.015, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def gaussian_perturb(a, strength=0.02, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    pert = rng.normal(0, strength, n) * a
    a_new = a + pert
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = norm_sum1(p)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_fb = 3600
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP-FW with growing K and tightening margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW main (n={n0}) ===")
    t1 = min(remaining * 0.28, 420.0)
    if t1 > 50:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t1, solver, verbose=True,
                                   K_start=60, K_end=250,
                                   margin_start=5e-3, margin_end=5e-5,
                                   momentum_decay=0.93)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 2: Spectral subgradient for intermediate refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Spectral subgrad (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_subgradient(a_in, t2, top_K=100, beta=3.5, c0=5.0)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "spec1")

    # === Phase 3: LP-FW with tighter margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP-FW tight (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.22, 330.0)
    if t3 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True,
                                   K_start=150, K_end=350,
                                   margin_start=2e-3, margin_end=3e-5,
                                   momentum_decay=0.95)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 4: Pairwise exchange for local refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Pairwise exchange (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_exchange(a_in, t4, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pair1")

    # === Phase 5: Multi-restart with diverse perturbations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-restart (n={len(best_seq)}) ===")
    rng = np.random.default_rng(777)
    t5_per = min(remaining * 0.07, 90.0)
    n_restarts = 3
    if t5_per > 40 and n_restarts > 0:
        for trial in range(n_restarts):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t5_per + 20:
                break
            if trial % 2 == 0:
                a_pert = freq_perturb(best_seq, strength=0.02, rng=rng)
            else:
                a_pert = gaussian_perturb(best_seq, strength=0.015, rng=rng)
            a_out, _ = lp_frank_wolfe(a_pert, t5_per, solver, verbose=False,
                                       K_start=100, K_end=250,
                                       margin_start=3e-3, margin_end=5e-5,
                                       momentum_decay=0.93)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"restart{trial}")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    candidates = sorted(set([n_best + 300, n_best + 600, n_best - 300]))
    candidates = [c for c in candidates if 1500 <= c <= 5500]
    for new_n in candidates[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t6 = min(remaining * 0.10, 120.0)
        if t6 > 25:
            a_out, _ = lp_frank_wolfe(a_long, t6, solver, verbose=False,
                                       K_start=100, K_end=200,
                                       margin_start=3e-3, margin_end=5e-5,
                                       momentum_decay=0.93)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"n{new_n}")

    # === Phase 7: Final LP with maximum constraints ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final LP (n={len(best_seq)}) ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True,
                                   K_start=200, K_end=400,
                                   margin_start=1e-3, margin_end=3e-5,
                                   momentum_decay=0.95)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 8: Final pairwise polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final pairwise (n={len(best_seq)}) ===")
    if remaining > 15:
        t8 = min(remaining * 0.10, 80.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = pairwise_exchange(a_in, t8, rng_seed=555)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
