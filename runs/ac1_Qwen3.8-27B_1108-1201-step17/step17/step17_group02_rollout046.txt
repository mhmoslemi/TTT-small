_global_solver = None

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def exact_subgradient(a, top_k=20):
    """Compute exact subgradient of max_k conv(a,a)[k] using top-k peaks."""
    n = len(a)
    ac = fast_autocorr(a)
    k_min = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_min]
    g = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        partner = ki - idx
        g[idx] += 2.0 * a[partner]
    g /= k_min
    return g, ac, top_idx


def soft_subgradient(a, top_k=30, beta=4.0):
    """Compute soft (exponentially weighted) subgradient."""
    n = len(a)
    ac = fast_autocorr(a)
    k_min = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_min]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_min) / k_min
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        partner = ki - idx
        g[idx] += 2.0 * wi * a[partner]
    return g, ac, top_idx, top_vals


def mirror_descent_opt(a0, time_limit, top_k=20, c0=5.0, restart_every=3000,
                       verbose=False, rng_seed=42, use_soft=False, beta=4.0):
    """Nesterov-accelerated mirror descent with exact/soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        if use_soft:
            g, ac, top_idx, top_vals = soft_subgradient(y, top_k=cur_top_k, beta=cur_beta)
        else:
            g, ac, top_idx = exact_subgradient(y, top_k=cur_top_k)

        M_curr = ac.max()

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive: increase top_k when stalled
        if stall_count > restart_every // 4:
            cur_top_k = min(cur_top_k + 3, 100)
            if use_soft:
                cur_beta = max(cur_beta * 0.95, 2.0)

        if stall_count > restart_every:
            stall_count = 0
            pert_scale = 0.02 + 0.03 * (stall_count // restart_every)
            pert = rng.normal(0, pert_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.2e}, stall={stall_count}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=None, use_all=False):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    if use_all:
        K_sel = 2 * n - 1
    elif top_K is not None:
        K_sel = min(top_K, 2 * n - 1)
    else:
        K_sel = min(300, 2 * n - 1)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        if use_all or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            constraint_vals = ac
            margins = M_curr - constraint_vals
            binding_idx = np.argsort(margins)[:K_sel]

        # Build constraint matrix T_sel
        T_sel = np.zeros((len(binding_idx), n))
        for row, k in enumerate(binding_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start < i_end:
                T_sel[row, i_start:i_end] = y[ki - (np.arange(i_start, i_end))]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

        if not use_all and t % 15 == 0:
            K_sel = min(K_sel + 30, 2 * n - 1)

    return best_a, best_m


def local_search(a0, time_limit, verbose=False, rng_seed=123, n_top=10, mode='pair'):
    """Local search with pair moves and simulated annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    T = 0.01
    T_min = 0.0001
    cooling = 0.9995
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:min(n_top, len(ac))]

        improved = False

        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            best_i_local = np.argmax(contrib)
            src = idx[best_i_local]
            src_partner = partner[best_i_local]

            if a[src] < 1e-12:
                continue

            # Sensitivity-based destination
            sens = np.zeros(n)
            for kk in top_k[:5]:
                kki = int(kk)
                ss = max(0, kki - n + 1)
                ee = min(n - 1, kki)
                ii = np.arange(ss, ee + 1)
                sens[ii] += a[kki - ii]

            exclude = {src, src_partner}
            candidates = np.array([d for d in range(n) if d not in exclude])
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            # Pick from low-sensitivity candidates with some randomness
            n_cand = len(candidates)
            if n_cand > 20:
                weights = 1.0 / (cand_sens + 1e-10)
                weights /= weights.sum()
                dst = candidates[rng.choice(n_cand, p=weights)]
            else:
                dst = candidates[np.argmin(cand_sens)]

            if dst == src:
                continue

            found = False
            for frac in [0.01, 0.005, 0.02, 0.002, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                delta = M_trial - best_m
                if delta < -1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
                elif rng.random() < np.exp(-delta / max(T, T_min)):
                    a = a_trial
                    stall = 0
                    found = True
                    break

            if found:
                break

        if not improved:
            stall += 1
            # Random perturbation
            n_pert = min(3, n)
            idx_pert = rng.integers(0, n, size=n_pert)
            for ip in idx_pert:
                if a[ip] > 1e-10:
                    dst = int(rng.integers(n))
                    if dst != ip:
                        frac = rng.uniform(0.005, 0.03)
                        amt = a[ip] * frac
                        a[ip] -= amt
                        a[dst] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        delta = M_new - best_m
                        if delta < 0:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break
                        elif rng.random() < np.exp(-delta / max(T, T_min)):
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break

        T = max(T * cooling, T_min)

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LS ({elapsed:.0f}s): J={J:.6f}, stall={stall}, T={T:.4f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    LS final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_sine_weighted(n, power=2.0):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** power
    return norm_sum1(a)


def active_set_refine(a0, time_limit, n_active=5, verbose=False, rng_seed=55):
    """Iteratively refine focusing on the most tied peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # Find peaks within 5% of max
        threshold = M_curr * 0.95
        active_peaks = np.where(ac >= threshold)[0]
        if len(active_peaks) > 50:
            active_peaks = np.argsort(ac)[::-1][:50]

        # Compute gradient from active peaks
        g = np.zeros(n)
        for k in active_peaks:
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            partner = ki - idx
            g[idx] += 2.0 * a[partner]
        g /= max(len(active_peaks), 1)

        #(projected mirror descent step with small step)
        eta = 0.1 / (np.sqrt(stall + 1) + 1.0)
        log_a = np.log(np.maximum(a, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 50:
            # Perturb and reset
            pert = rng.normal(0, 0.01, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ASR ({elapsed:.0f}s): J={J:.6f}, n_active={len(active_peaks)}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    ASR final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

        labels = unique_labels

        t_per = max(25.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 80:
                break
            t_avail = min(t_per, time_budget - 80 - (time.monotonic() - t_global))
            if t_avail < 20:
                break
            print(f"  MD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = mirror_descent_opt(init, t_avail, top_k=15, c0=5.0,
                                           restart_every=4000, verbose=False,
                                           rng_seed=100 + idx, use_soft=False)
            try_update(a_out, f"md_{lbl}")

            # Also try with soft subgradient
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 100:
                t_avail2 = min(t_per * 0.5, remaining - 80)
                if t_avail2 > 15:
                    a_out2, _ = mirror_descent_opt(init, t_avail2, top_k=30, c0=4.0,
                                                   restart_every=5000, verbose=False,
                                                   rng_seed=200 + idx, use_soft=True, beta=4.0)
                    try_update(a_out2, f"md_soft_{lbl}")

    # Phase 2: LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.18, 320.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K = min(1500, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=top_K, use_all=use_all)
        try_update(a_out, "lp_main")

    # Phase 3: Active set refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Active Set Refine (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = active_set_refine(a_in, t3, verbose=True, rng_seed=55)
        try_update(a_out, "active_set")

    # Phase 4: Mirror descent polish with higher top_k
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: MD Polish (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_opt(a_in, t4, top_k=40, c0=3.0,
                                       restart_every=6000, verbose=False, rng_seed=500,
                                       use_soft=False)
        try_update(a_out, "md_polish")

    # Phase 5: LP-FW round 2
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP-FW R2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 1500)
        top_K = min(2500, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True, top_K=top_K, use_all=use_all)
        try_update(a_out, "lp_r2")

    # Phase 6: Local search
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Local Search (n={len(best_seq)}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = local_search(a_in, t6, verbose=True, rng_seed=789, n_top=10)
        try_update(a_out, "local_search")

    # Phase 7: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length Exploration ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 80]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.30, 150.0)
        if t7a > 30:
            a_out, _ = mirror_descent_opt(a_long, t7a, top_k=20, c0=4.0,
                                           restart_every=5000, verbose=False,
                                           rng_seed=new_n % 1000, use_soft=False)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 3000:
            t7b = min(remaining * 0.30, 100.0)
            if t7b > 20:
                a_out, _ = lp_frank_wolfe(norm_sum1(best_seq), t7b, solver,
                                          verbose=False, top_K=min(1000, 2 * new_n - 1))
                # Interpolate LP result to new_n if needed
                try_update(a_out, f"lp_n{new_n}")

    # Phase 8: Final LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 30 and n_best <= 5000:
        t8 = min(remaining * 0.40, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            use_all = (n_best <= 1500)
            top_K = min(3000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=top_K, use_all=use_all)
            try_update(a_out, "lp_final")

    # Phase 9: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final Polish (n={n_best}) ===")
        t9 = min(remaining * 0.50, 60.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = local_search(a_in, t9 * 0.6, verbose=False, rng_seed=999, n_top=8)
            try_update(a_out, "ls_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t9b = min(remaining * 0.50, 30.0)
            if t9b > 5:
                a_in = norm_sum1(best_seq)
                a_out, _ = mirror_descent_opt(a_in, t9b, top_k=10, c0=2.0,
                                               restart_every=2000, verbose=False,
                                               rng_seed=111, use_soft=False)
                try_update(a_out, "md_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
