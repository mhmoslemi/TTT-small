import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def smooth_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*c[k])) where c = conv(a,a).
    
    Returns the gradient vector of length n.
    """
    n = len(a)
    c = fast_autocorr(a)
    # Softmax with numerical stability
    c_shifted = c - c.max()
    p = np.exp(tau * c_shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(c)) / len(c)
    else:
        p = p / p_sum
    
    # Compute grad[j] = sum_{j'=0}^{n-1} p[j+j'] * a[j']
    # Using FFT: grad = (flip(a) * p) evaluated at positions [n-1, ..., 2n-2]
    # (flip(a) * p) has length n + (2n-1) - 1 = 3n - 2
    # (flip(a) * p)[j + n - 1] = grad[j]
    
    r = a[::-1]  # flip(a), length n
    conv_size = n + (2*n-1) - 1  # = 3n - 2
    fft_size = 1 << (conv_size - 1).bit_length()
    
    R = np.fft.rfft(r, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(R * P, fft_size)[:conv_size]
    
    grad = conv_result[n-1 : n-1+n]  # positions n-1 to 2n-2
    return grad

def smooth_objective(a, tau):
    """Compute (1/tau)*log(sum_k exp(tau*c[k])) where c = conv(a,a)."""
    c = fast_autocorr(a)
    c_shifted = c - c.max()
    log_sum = np.log(np.exp(tau * c_shifted).sum()) + tau * c.max()
    return log_sum / tau

    best_exact_J = compute_J(a)
    t0 = time.monotonic()

"""

2. Dual-ascent guided peak targeting
3. Coordinate descent peak shaving for fine-tuning

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def smooth_gradient_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*c[k])) where c = conv(a,a).
    Uses FFT for O(n log n) computation.
    Returns gradient of length n.
    """
    n = len(a)
    c = fast_autocorr(a)
    # Numerically stable softmax
    c_max = c.max()
    p = np.exp(tau * (c - c_max))
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(c)) / len(c)
    else:
        p = p / p_sum

    # grad[j] = 2 * sum_{j'=0}^{n-1} p[j+j'] * a[j']
    # = 2 * (flip(a) * p)[j + n - 1]
    r = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    R = np.fft.rfft(r, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_result = np.fft.irfft(R * P, fft_size)[:conv_len]
    grad = 2.0 * conv_result[n - 1: n - 1 + n]
    return grad


def smooth_objective_value(a, tau):
    """Compute (1/tau)*log(sum_k exp(tau*c[k]))."""
    c = fast_autocorr(a)
    c_max = c.max()
    log_sum_exp = c_max * tau + np.log(np.exp(tau * (c - c_max)).sum())
    return log_sum_exp / tau


def nesterov_smooth_mirror(a0, time_limit, tau_start=5.0, tau_end=80.0,
                            verbose=False, seed=42, max_tau_growth=1.03):
    """
    Nesterov-accelerated mirror descent on the smooth surrogate.
    Adaptive tau: starts smooth, becomes sharper over time.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    tau = tau_start
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_every = 2000
    eta_base = 1.0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute smooth gradient
        grad = smooth_gradient_fft(y, tau)

        # Adaptive step size
        grad_norm = np.sqrt(np.dot(grad, grad))
        if grad_norm < 1e-15:
            eta = eta_base
        else:
            eta = eta_base / (grad_norm * 0.5)
        eta = min(eta, 2.0 / (np.sqrt(t + 1.0) * 0.1))
        eta = max(eta, 0.001)

        # Exponential (mirror) step
        log_w = -eta * grad
        log_w -= log_w.max()
        a_new = y * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Track best
        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adapt tau: increase slowly
        if t % 200 == 0:
            tau = min(tau * max_tau_growth, tau_end)

        # Restart on stall
        if stall > restart_every:
            stall = 0
            sigma = 0.05
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            tau = max(tau * 0.8, tau_start)
            eta_base = min(eta_base * 1.2, 3.0)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      Restart ({elapsed:.0f}s): J={best_J:.6f}, tau={tau:.1f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      t={t} ({elapsed:.0f}s): J={best_J:.6f}, tau={tau:.1f}")

    return best_a, best_J


def dual_ascent_smooth(a0, time_limit, tau=20.0, verbose=False, seed=100):
    """
    Alternating: update a via mirror descent, update dual (softmax weights) explicitly.
    This is equivalent to the smooth approach but with explicit dual tracking.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # Dual: compute softmax weights
        c = fast_autocorr(y)
        c_max = c.max()
        p = np.exp(tau * (c - c_max))
        p_sum = p.sum()
        p = p / p_sum

        # Primal step: gradient = 2 * (flip(y) * p)[n-1 : 2n-2]
        r = y[::-1]
        conv_len = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        R = np.fft.rfft(r, fft_size)
        P = np.fft.rfft(p, fft_size)
        conv_result = np.fft.irfft(R * P, fft_size)[:conv_len]
        grad = 2.0 * conv_result[n - 1: n - 1 + n]

        # Step size
        grad_inf = np.abs(grad).max()
        if grad_inf < 1e-15:
            eta = 0.5
        else:
            eta = 1.0 / grad_inf
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 0.005)

        # Mirror step
        log_w = -eta * grad
        log_w -= log_w.max()
        a_new = y * np.exp(log_w)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Dual t={t} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def peak_shaving_coordinate(a0, time_limit, verbose=False, seed=55):
    """
    Coordinate descent: iteratively find the peak lag, identify contributing pairs,
    and shift mass from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M = ac.max()
        peak_idx = np.argmax(ac)

        # Find contributing pairs for the peak
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n, peak_idx + 1)
        idx = np.arange(i_start, i_end)
        j_idx = peak_idx - idx
        contributions = a[idx] * a[j_idx]
        total_contrib = contributions.sum()

        if total_contrib < 1e-15:
            stall += 50
            continue

        # Normalize contribution weights
        w = contributions / total_contrib

        # Find positions with low total peak sensitivity
        sensitivity = np.zeros(n)
        n_check = min(8, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_check]
        for k in top_lags:
            ki = int(k)
            is_start = max(0, ki - n + 1)
            is_end = min(n, ki + 1)
            ii = np.arange(is_start, is_end)
            jj = ki - ii
            for ii_val, jj_val in zip(ii, jj):
                sensitivity[ii_val] += a[ii_val] * a[jj_val]
                sensitivity[jj_val] += a[ii_val] * a[jj_val]

        # Source: high sensitivity, high mass. Dest: low sensitivity.
        src_candidates = np.where((sensitivity > np.median(sensitivity)) & (a > 1e-6))[0]
        dst_candidates = np.where((sensitivity < np.median(sensitivity) * 0.5) | (sensitivity == 0))[0]

        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            stall += 1
            continue

        improved = False
        n_attempts = 30
        for _ in range(n_attempts):
            if time.monotonic() - t0 > time_limit:
                break
            src = src_candidates[rng.integers(len(src_candidates))]
            dst = dst_candidates[rng.integers(len(dst_candidates))]
            if src == dst:
                continue

            # Try multiple transfer amounts
            src_val = a[src]
            for frac in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] = a[src] - amt
                a_trial[dst] = a[dst] + amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                # Random perturbation to escape
                n_pert = max(3, n // 50)
                positions = rng.choice(n, size=n_pert, replace=False)
                for p in positions:
                    if a[p] > 1e-6:
                        dst = int(rng.integers(n))
                        amt = a[p] * rng.uniform(0.05, 0.3)
                        a[p] -= amt
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakShave ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def adaptive_subgradient_descent(a0, time_limit, verbose=False, seed=77,
                                  top_k_range=(5, 60), beta_range=(2.0, 5.0)):
    """
    Subgradient descent with adaptive top-k and beta, using Nesterov acceleration.
    Different from previous: uses proper line search and adaptive parameters.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    cur_top_k = 10
    cur_beta = 3.0
    stall = 0
    best_stall_curve = [best_J]

    for t in range(1, 3000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        ac = fast_autocorr(y)
        M = ac.max()

        # Compute subgradient (smoothed with top-k)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        # Step size with backtracking
        eta = 1.0
        found_step = False
        for _ in range(20):
            log_w = -eta * grad
            log_w -= log_w.max()
            a_new = y * np.exp(log_w)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            J_trial = compute_J(a_new)
            if J_trial <= M + 1e-10:  # sufficient decrease (approx)
                found_step = True
                break
            eta *= 0.5

        if not found_step:
            a_new = y.copy()

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adapt parameters
        if stall > 1500:
            stall = 0
            cur_top_k = min(cur_top_k + rng.integers(5, 20), top_k_range[1])
            cur_beta = max(cur_beta - rng.uniform(0.2, 0.5), beta_range[0])
            # Perturb
            sigma = 0.03
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): J={best_J:.6f}, tk={cur_top_k}, b={cur_beta:.1f}")

    return best_a, best_J


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_bimodal_init(n, edge_width=0.1, center_dip=0.3):
    """Bimodal initialization: high at edges, low in center."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.0) / edge_width) ** 2) + np.exp(-((x - 1.0) / edge_width) ** 2)
    a = a + center_dip * (1 - np.exp(-((x - 0.5) / 0.2) ** 2))
    a = np.maximum(a, 1e-6)
    return norm_sum1(a)


def make_cosine_init(n, power=2.0):
    """Cosine-based initialization."""
    x = np.linspace(0, np.pi, n)
    a = np.abs(np.cos(x)) ** power + 0.01
    return norm_sum1(a)


def make_random_sparse_init(n, n_dense, seed=0):
    """Initialize with random dense core and random perturbations."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) * 0.1
    # Add random bumps
    for _ in range(n_dense):
        center = rng.integers(n // 4, 3 * n // 4)
        width = rng.integers(n // 50, n // 20)
        x = np.arange(max(0, center - width), min(n, center + width))
        a[x] += np.exp(-((x - center) / (width / 3.0)) ** 2) * rng.uniform(0.5, 2.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining smooth mirror descent, subgradient,
    coordinate descent, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_bimodal_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Smooth mirror descent (main workhorse) ===
    print(f"\n=== Phase 1: Smooth Mirror Descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 500.0)

    if t_phase1 > 60:
        # Multiple inits
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_bimodal_init(n0, edge_width=0.08, center_dip=0.2))
        labels.append("bimodal")

        inits.append(make_cosine_init(n0, power=3.0))
        labels.append("cosine3")

        inits.append(make_cosine_init(n0, power=1.5))
        labels.append("cosine1.5")

        # Blend
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_bimodal_init(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove near-duplicates
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for u in unique_inits:
                if np.sum(np.abs(init - u)) < 0.03:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(15.0, t_phase1 / n_inits)

        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 30)
            if t_avail < 10:
                break
            print(f"  Smooth MD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = nesterov_smooth_mirror(init, t_avail, verbose=False,
                                               tau_start=5.0, tau_end=60.0,
                                               seed=42 + idx)
            try_update(a_out, f"smooth_{lbl}")

    # === Phase 2: Dual ascent smooth ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Dual Ascent Smooth (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = dual_ascent_smooth(a_in, t2, tau=30.0, verbose=False, seed=200)
        try_update(a_out, "dual_ascent")

    # === Phase 3: Adaptive subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Adaptive Subgradient (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = adaptive_subgradient_descent(a_in, t3, verbose=False, seed=111)
        try_update(a_out, "adap_subgrad")

    # === Phase 4: Peak shaving coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Peak Shaving (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving_coordinate(a_in, t4, verbose=True, seed=55)
        try_update(a_out, "peak_shave")

    # === Phase 5: Smooth MD with different tau schedule ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Smooth MD slow-tau (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smooth_mirror(a_in, t5, verbose=False,
                                           tau_start=3.0, tau_end=40.0,
                                           max_tau_growth=1.01, seed=333)
        try_update(a_out, "smooth_slow_tau")

        # Also try fast tau
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.5, 80.0)
            a_out, _ = nesterov_smooth_mirror(a_in, t5b, verbose=False,
                                               tau_start=20.0, tau_end=100.0,
                                               max_tau_growth=1.05, seed=444)
            try_update(a_out, "smooth_fast_tau")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length Exploration ===")
    lengths_to_try = []
    for delta in [200, 400, 600, -200, -400]:
        new_n = n_best + delta
        if 1000 <= new_n <= 5000:
            lengths_to_try.append(new_n)

    for new_n in lengths_to_try[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.20, 120.0)
        if t6a > 15:
            a_out, _ = nesterov_smooth_mirror(a_long, t6a, verbose=False,
                                               tau_start=5.0, tau_end=50.0,
                                               seed=555)
            try_update(a_out, f"smooth_n{new_n}")

            # Peak shave on new length
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                t6b = min(remaining * 0.15, 80.0)
                a_out2, _ = peak_shaving_coordinate(a_out, t6b, verbose=False, seed=666)
                try_update(a_out2, f"shave_n{new_n}")

    # === Phase 7: Final smooth polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Polish (n={n_best}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.4, 150.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smooth_mirror(a_in, t7, verbose=False,
                                           tau_start=10.0, tau_end=80.0,
                                           max_tau_growth=1.04, seed=777)
        try_update(a_out, "final_smooth")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_best = len(best_seq)
        t8 = min(remaining * 0.5, 100.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving_coordinate(a_in, t8, verbose=True, seed=888)
        try_update(a_out, "final_shave")

    # === Phase 9: One more subgradient round ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t9 = min(remaining * 0.6, 80.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = adaptive_subgradient_descent(a_in, t9, verbose=False, seed=999)
        try_update(a_out, "final_subgrad")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_bimodal_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
