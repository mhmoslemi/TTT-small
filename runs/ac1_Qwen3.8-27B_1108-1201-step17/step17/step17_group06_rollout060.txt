Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer in log-space with log-sum-exp smoothing of the max
2. Beta annealing from smooth to sharp

def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    """J = 2*n*max(conv) for normalized (sum=1) non-negative a."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def compute_grad_a(conv_grad, a, N):
    """Compute gradient of sum_k c[k]*(a*a)[k] w.r.t. a.
    Returns grad_a of length len(a).
    Uses: grad_a = 2 * ifft(fft(c, N) * conj(fft(a, N)), N)[:n]
    where N >= 3n-2 to avoid circular aliasing.
    """
    n = len(a)
    C = np.fft.rfft(conv_grad, N)
    A = np.fft.rfft(a, N)
    grad = np.fft.irfft(C * np.conj(A), N)[:n]
    return 2.0 * grad

    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a_init = a_init / a_init.sum()
    
    z = np.log(a_init)
    z -= z.mean()
    
    N = 3 * n  # >= 3n-2, round up for FFT efficiency
    
    best_a = a_init.copy()
    best_val = eval_J(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = beta_start
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        beta = min(beta * anneal_per_iter, beta_end)
        
        # Compute a from z
        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a = a / a.sum()
        
        # Compute objective and gradient
        conv = fast_conv(a)
        # LSE gradient: softmax(beta * conv)
        beta_conv = beta * conv
        beta_conv -= beta_conv.max()
        beta_conv = np.clip(beta_conv, -700, 700)
        e = np.exp(beta_conv)
        conv_grad = e / e.sum()
        
        # Gradient w.r.t. a
        grad_a = compute_grad_a(conv_grad, a, N)
        
        # Gradient w.r.t. z (log-space chain rule)
        # a = softmax(z), so grad_z_j = grad_a_j * a_j - a_j * sum_i(grad_a_i * a_i)
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Periodic check

            if val < best_val:
                best_val = val
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        
        # Restart
        if stall > restart_every // 50:
            stall = 0
            scale = rng.uniform(0.01, pert_scale)
            z += rng.normal(0, scale, n)
            z -= z.mean()
            if verbose:

    if val < best_val:
        best_val = val
        best_a = a.copy()
    
    return best_a, best_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer in log-space with log-sum-exp smoothing of the max
2. Beta annealing from smooth to sharp (exploration to exploitation)
3. Multi-strategy initialization with diverse parametric families
4. FFT-based gradient computation (cross-correlation)
5. Periodic perturbations and restarts to escape local minima
6. Length interpolation and multi-scale optimization
"""

import numpy as np
import time


def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """J = 2*n*max(conv) for normalized (sum=1) non-negative a."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def compute_grad_a(c, a, N):
    """Compute 2 * cross_correlation(c, a) truncated to len(a).
    c has length 2n-1 (the softmax weights on conv).
    Uses FFT: 2 * ifft(fft(c,N) * conj(fft(a,N)))[:n]
    N must be >= (2n-1) + n - 1 = 3n-2.
    """
    n = len(a)
    C = np.fft.rfft(c, N)
    A = np.fft.rfft(a, N)
    grad = np.fft.irfft(C * np.conj(A), N)[:n]
    return 2.0 * grad


def adam_lse(a0, time_limit, beta_start=5.0, beta_end=500.0,
             lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
             anneal_per_iter=1.002, restart_every=5000,
             pert_scale=0.05, seed=42, verbose=False,
             lr_decay=1.0):
    """Adam optimizer in log-space with LSE smoothing of max(conv)."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a_init = a_init / a_init.sum()

    z = np.log(a_init)
    z -= z.mean()

    # FFT size for gradient computation (>= 3n-2)
    N_fft = 1 << (3 * n - 1).bit_length()

    best_a = a_init.copy()
    best_val = eval_J(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = beta_start
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    lr_cur = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        beta = min(beta * anneal_per_iter, beta_end)
        lr_cur = lr * (lr_decay ** (t / 10000.0))

        # Compute a from z (softmax)
        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a_sum = a.sum()
        a = a / a_sum

        # Compute convolution
        conv = fast_conv(a)

        # LSE gradient: softmax(beta * conv)
        bc = beta * conv
        bc -= bc.max()
        bc = np.clip(bc, -700, 700)
        e = np.exp(bc)
        c = e / e.sum()  # normalized weights on conv lags

        # Gradient w.r.t. a via cross-correlation
        grad_a = compute_grad_a(c, a, N_fft)

        # Gradient w.r.t. z (log-space chain rule for softmax)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z -= lr_cur * m_hat / (np.sqrt(v_hat) + eps)

        # Periodic evaluation
        if t % 100 == 0:
            val = eval_J(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        # Restart with perturbation
        if stall > restart_every // 100:
            stall = 0
            z_best = np.log(np.maximum(best_a, 1e-15))
            z_best -= z_best.mean()
            scale = rng.uniform(0.01, pert_scale)
            z = z_best + rng.normal(0, scale, n)
            z -= z.mean()
            if verbose:
                el = time.monotonic() - t0
                print(f"    Restart t={t}, beta={beta:.1f}, best={best_val:.6f} ({el:.0f}s)")

        if verbose and t % 10000 == 0:
            el = time.monotonic() - t0
            val_now = eval_J(a)
            print(f"    Adam t={t}, beta={beta:.1f}, J={val_now:.6f}, best={best_val:.6f} ({el:.0f}s)")

    # Final evaluation
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    val = eval_J(a)
    if val < best_val:
        best_val = val
        best_a = a.copy()

    return best_a, best_val


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_inverse_ramp(n, p=1.0):
    """Inverse power-law ramp (high at start, low at end)."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** (-p)
    return a / a.sum()


def make_bimodal(n, width=50.0):
    """Bimodal Gaussian mixture."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.15) ** 2) + np.exp(-width * (x - 0.85) ** 2) + 0.01
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier series."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 20):
        coeff = rng.normal(0, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_cosine_window(n, alpha=0.5):
    """Cosine-power window."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** (2 * alpha)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def make_gaussian_centered(n, width=0.3):
    """Centered Gaussian."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) ** 2) / (2 * width ** 2))
    return a / a.sum()


def make_dual_ramp(n, p=1.0):
    """Symmetric triangular ramp."""
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = (1.0 - np.abs(x - mid) / (mid + 1e-10)) ** p
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def make_plateau(n, frac=0.6):
    """Plateau with smooth edges."""
    x = np.linspace(0, 1, n)
    edge = (1.0 - frac) / 2.0
    a = np.ones(n)
    for i in range(n):
        if x[i] < edge:
            a[i] = 0.5 * (1 + np.sin(np.pi * x[i] / edge - np.pi / 2))
        elif x[i] > 1 - edge:
            a[i] = 0.5 * (1 + np.sin(np.pi * (1 - x[i]) / edge - np.pi / 2))
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def interp_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 1e-10)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s
