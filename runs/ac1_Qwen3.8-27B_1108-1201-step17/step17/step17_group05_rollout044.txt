def fast_conv(a, b=None):
    """Compute full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lp_direction(a, M, top_K=500, solver=None):
    """
    Solve the LP to find the FW direction.
    max sum(b) s.t. conv(a,b)[k] <= M for selected k, b >= 0
    Returns the normalized direction b/sum(b).
    """
    n = len(a)
    ac = fast_conv(a, a)
    
    # Select constraints: those where ac is closest to M
    # Use top_K largest values
    K = min(top_K, len(ac))
    top_idx = np.argsort(ac)[::-1][:K]
    
    # Build constraint matrix (sparse)
    # Row m: T[m, j] = a[j] if 0 <= m-j < n
    rows = []
    cols = []
    vals = []
    
    for row_idx, m in enumerate(top_idx):
        m_int = int(m)
        j_start = max(0, m_int - n + 1)
        j_end = min(n - 1, m_int)
        for j in range(j_start, j_end + 1):
            rows.append(row_idx)
            cols.append(j)
            vals.append(a[j])
    
    T = sparse.csc_matrix((vals, (rows, cols)), shape=(K, n))
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(K)]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return a.copy()  # Fallback: stay in place
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return a.copy()
    
    return b_val / b_sum
