Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated LP-Frank-Wolfe

def _autocorr(a):
    """Compute a*a (full linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _eval(a):

def _build_T(a, top_k=None):
    """Build constraint matrix T where T[m,j] = a[m-j] if valid.
    If top_k is given, only include the top_k most binding rows."""
    n = len(a)
    ac = _autocorr(a)
    M = ac.max()
    
    if top_k is not None and top_k < 2*n - 1:
        # Select top_k lags
        idx = np.argsort(ac)[::-1][:top_k]
    else:
        idx = np.arange(2*n - 1)
    
    K = len(idx)
    T = np.zeros((K, n))
    for i, m in enumerate(idx):
        m_int = int(m)
        j_start = max(0, m_int - n + 1)
        j_end = min(n - 1, m_int)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            T[i, js] = a[m_int - js]
    return T, M

def _lp_fw_step(a, M, T, solver):
    """Solve the LP and return the Frank-Wolfe direction."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M * np.ones(T.shape[0]), cp.sum(b) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b.value, dtype=np.float64)
    val = np.maximum(val, 0.0)
    s = val.sum()
    if s < 1e-15:
        return None
    return val / s

def _subgrad_polish(a0, time_limit, verbose=True, beta_init=5.0, beta_max=30.0):
    """Subgradient polishing with adaptive softmax."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    ac = _autocorr(a)
    best_M = ac.max()
    J_best = 2.0 * n * best_M
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    
    for it in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta = min(beta_max, beta_init + 0.01 * it)
        ac_x = _autocorr(x)
        M_x = ac_x.max()
        
        # Softmax weights
        shifted = ac_x - M_x
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(ac_x) / len(ac_x)
        else:
            w = w / ws
        w[w < 1e-14] = 0
        w = w / w.sum()
        
        # Compute subgradient: g_k = 2 * sum_m w[m] * a[m-k]
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        a_rev = np.zeros(fft_size)
        a_rev[:n] = a[::-1]
        w_pad = np.zeros(fft_size)
        w_pad[:len(w)] = w
        conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
        g = 2.0 * conv[n-1:2*n-1]
        
        # Nesterov extrapolation
        beta_t = (it - 1.0) / (it + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Mirror (log) step
        gamma = 3.0 / (np.sqrt(it) + 1.0)
        log_y = np.log(y)
        log_new = log_y - gamma * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = _autocorr(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            J_best = 2.0 * n * best_M
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x = _normalize(best_a + pert)
            x_prev = x.copy()
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Subgrad iter={it} ({elapsed:.0f}s): J={J_best:.6f}, beta={beta:.1f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"  Subgrad done ({elapsed:.0f}s): J={J_best:.6f}")
    
    return best_a, best_M

def _coord_descent(a0, time_limit, verbose=True, seed=42):
    """Coordinate descent: move mass from high-contributing to low-sensitivity positions."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_M = _autocorr(a).max()
    J_best = 2.0 * n * best_M
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        ac = _autocorr(a)
        M_curr = ac.max()
        improved = False
        
        # Find top contributing lags
        top_n = min(20, len(ac))
        top_lags = np.argsort(ac)[::-1][:top_n]
        
        for k in top_lags[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find low-sensitivity destination
            sens = np.zeros(n)
            for k2 in top_lags[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    sens[d] += a[k2i - d]
            
            # Candidates: far from src, low sensitivity
            dist = np.abs(np.arange(n) - src).astype(float)
            candidates = np.where((sens < 0.5 * sens.mean()) & (dist > n * 0.1))[0]
            if len(candidates) == 0:
                candidates = np.where(sens < sens.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Score: low sens / distance
            dist_c = dist[candidates]
            dist_c[dist_c == 0] = 1
            scores = sens[candidates] / (1.0 + dist_c / n)
            best_dst = candidates[np.argmin(scores)]
            if best_dst == src:
                continue
            
            # Try fractions
            for frac in [0.01, 0.02, 0.005, 0.03, 0.002, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_t = _autocorr(a_trial).max()
                if M_t < best_M - 1e-14:
                    a = a_trial
                    best_M = M_t
                    best_a = a.copy()
                    J_best = 2.0 * n * best_M
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.02)
                    a[i] -= amt
                    a[j] += amt
                    a = _normalize(a)
                    M_new = _autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        J_best = 2.0 * n * best_M
                        stall = 0
            if stall > 300:
                break
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Coord ({elapsed:.0f}s): J={J_best:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"  Coord done ({elapsed:.0f}s): J={J_best:.6f}")
    
    return best_a, best_M

import numpy as np
import time
import cvxpy as cp

_global_solver = None


def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)

def _autocorr(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _normalize(a):
    """Clip to non-negative, normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _eval_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _build_T(y, active_idx):
    """Build constraint matrix rows for given active lags. T[i,j] = y[m_i - j]."""
    n = len(y)
    K = len(active_idx)
    T = np.zeros((K, n))
    for i, m in enumerate(active_idx):
        mi = int(m)
        js = max(0, mi - n + 1)
        je = min(n - 1, mi)
        if js <= je:
            idx = np.arange(js, je + 1)
            T[i, idx] = y[mi - idx]
    return T


def _lp_solve(y, M, T, solver):
    """Solve: max sum(b) s.t. T@b <= M, b >= 0. Returns normalized direction or None."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(T.shape[0])]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    if val is None:
        return None
    val = np.maximum(val, 0.0)
    s = val.sum()
    if s < 1e-15:
        return None
    return val / s

    a = _normalize(a0)
    best_a = a.copy()
    best_M = _autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    y = a.copy()
    x_prev = a.copy()
    stall = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        t_start = time.monotonic()

        ac_y = _autocorr(y)
        M_y = ac_y.max()

        # Constraint generation: select active lags

        alpha = 2.0 / (iteration + 2.0)

def _subgrad_optimize(a0, time_limit, verbose=True,
                      beta_init=5.0, beta_rate=0.005, beta_max=40.0):
    """Adaptive softmax subgradient with Nesterov acceleration on simplex (log barrier)."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_M = _autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(314)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        beta = min(beta_max, beta_init + beta_rate * it)
        ac_x = _autocorr(x)
        M_x = ac_x.max()

        # Softmax weights (focus on top lags)
        shifted = ac_x - M_x
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(ac_x) / len(ac_x)
        else:
            w = w / ws
        # Zero out tiny weights
        w[w < 1e-12] = 0.0
        if w.sum() < 1e-15:
            top_k = min(20, len(ac_x))
            idx = np.argsort(ac_x)[::-1][:top_k]
            w = np.zeros_like(ac_x)
            w[idx] = 1.0 / top_k
        else:
            w = w / w.sum()

        # Subgradient: g_k = 2 * sum_m w[m] * x[m-k] (via FFT)
        total_len = n + len(w) - 1
        fft_sz = 1 << (total_len - 1).bit_length()
        x_rev = np.zeros(fft_sz)
        x_rev[:n] = x[::-1]
        w_pad = np.zeros(fft_sz)
        w_pad[:len(w)] = w
        conv = np.fft.irfft(np.fft.rfft(x_rev) * np.fft.rfft(w_pad), fft_sz)
        g = 2.0 * conv[n - 1:2 * n - 1]

        # Nesterov extrapolation
        nt = (it - 1.0) / (it + 2.0)
        y = x + nt * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        # Mirror (exponential) step
        gamma = 4.0 / (np.sqrt(it) + 1.0)
        log_y = np.log(y)
        log_new = log_y - gamma * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = _autocorr(x).max()
        if M_new < best_M - 1e-16:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            stall = 0
            pert = rng.normal(0, 0.015, n) * best_a
            x = _normalize(best_a + pert)
            x_prev = x.copy()

        if verbose and it % 5000 == 0:
            el = time.monotonic() - t0
            print(f"  Subgrad iter={it} ({el:.0f}s): J={2*n*best_M:.6f}, beta={beta:.1f}")

    if verbose:
        el = time.monotonic() - t0
        print(f"  Subgrad done ({el:.0f}s): J={2*n*best_M:.6f}")
    return best_a, best_M


def _coord_descent(a0, time_limit, verbose=True, seed=42):
    """Coordinate descent: move mass from peak contributors to low-sensitivity spots."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_M = _autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    moves = 0

    while time.monotonic() - t0 < time_limit:
        ac = _autocorr(a)
        M_curr = ac.max()
        improved = False

        # Top lags to address
        n_top = min(15, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]

        for k in top_lags[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue
            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute sensitivity: for each position d, how much would placing mass there affect peaks?
            sens = np.zeros(n)
            for k2 in top_lags[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    sens[d] += a[k2i - d]

            # Find good destination: low sensitivity, far from source
            dist = np.abs(np.arange(n) - src).astype(float)
            mask = (sens < 0.4 * sens.mean()) & (dist > n * 0.08)
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                candidates = np.where(sens < sens.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            dist_c = dist[candidates]
            dist_c[dist_c == 0] = 1.0
            scores = sens[candidates] / (1.0 + dist_c / n)
            best_dst = int(candidates[np.argmin(scores)])
            if best_dst == src:
                continue

            # Try several transfer fractions
            for frac in [0.015, 0.01, 0.02, 0.005, 0.03, 0.008, 0.025, 0.012]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                Mt = _autocorr(a_trial).max()
                if Mt < best_M - 1e-15:
                    a = a_trial
                    best_M = Mt
                    best_a = a.copy()
                    improved = True
                    moves += 1
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random kick to escape
            if stall % 15 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.025)
                    a[i] -= amt
                    a[j] += amt
                    a = _normalize(a)
                    Mn = _autocorr(a).max()
                    if Mn < best_M:
                        best_M = Mn
                        best_a = a.copy()
                        stall = 0
            if stall > 400:
                break

        if verbose and (moves % 100 == 0 or stall % 100 == 0):
            el = time.monotonic() - t0
            print(f"  Coord ({el:.0f}s): J={2*n*best_M:.6f}, moves={moves}, stall={stall}")

    if verbose:
        el = time.monotonic() - t0
        print(f"  Coord done ({el:.0f}s): J={2*n*best_M:.6f}, moves={moves}")
    return best_a, best_M


def _interpolate(a, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_n = _normalize(a)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_n)
    return _normalize(a_interp)

    except Exception:
        pass

    if best_seq is None:
        n_fb = 3200
        best_seq = np.ones(n_fb) / n_fb
        best_score = _eval_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP-FW (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 800.0)
    print(f"\n{'='*60}")
    print(f"Phase 1: LP-FW with constraint generation (t={t1:.0f}s)")
    print(f"{'='*60}")
    if t1 > 30 and n0 <= 5000:
        a_out, _ = _lp_fw_optimize(best_seq, t1, solver, verbose=True,
                                    cg_threshold=0.93, max_constraints=300)
        try_update(a_out, "lpfw_p1")

    # === Phase 2: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 200.0)
    print(f"\n{'='*60}")
    print(f"Phase 2: Subgradient polish (t={t2:.0f}s)")
    print(f"{'='*60}")
    if t2 > 20:
        a_out, _ = _subgrad_optimize(best_seq, t2, verbose=True,
                                      beta_init=6.0, beta_rate=0.008, beta_max=35.0)
        try_update(a_out, "subgrad_p2")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 200.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Coordinate descent (t={t3:.0f}s)")
    print(f"{'='*60}")
    if t3 > 15:
        a_out, _ = _coord_descent(best_seq, t3, verbose=True, seed=42)
        try_update(a_out, "coord_p3")

    # === Phase 4: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 280.0)
    print(f"\n{'='*60}")
    print(f"Phase 4: LP-FW Round 2 (t={t4:.0f}s)")
    print(f"{'='*60}")
    if t4 > 30 and n_best <= 5000:
        a_out, _ = _lp_fw_optimize(best_seq, t4, solver, verbose=True,
                                    cg_threshold=0.90, max_constraints=500)
        try_update(a_out, "lpfw_p4")

    # === Phase 5: Final coordinate + subgrad ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.08, 120.0)
    print(f"\n{'='*60}")
    print(f"Phase 5a: Final coord (t={t5:.0f}s)")
    print(f"{'='*60}")
    if t5 > 10:
        a_out, _ = _coord_descent(best_seq, t5, verbose=True, seed=999)
        try_update(a_out, "coord_p5a")

    remaining = time_budget - (time.monotonic() - t_global)
    t5b = min(remaining * 0.07, 100.0)
    if t5b > 10:
        print(f"\nPhase 5b: Final subgrad (t={t5b:.0f}s)")
        a_out, _ = _subgrad_optimize(best_seq, t5b, verbose=True,
                                      beta_init=10.0, beta_rate=0.01, beta_max=50.0)
        try_update(a_out, "subgrad_p5b")

    # === Phase 6: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    offsets = [200, -200, 400]
    tried = set()
    for off in offsets:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        new_n = n_best + off
        if new_n in tried or new_n < 1000 or new_n > 6000:
            continue
        tried.add(new_n)
        print(f"\nPhase 6: Trying n={new_n} (t_remaining={remaining:.0f}s)")
        a_long = _interpolate(best_seq, new_n)
        t6 = min(remaining * 0.3, 150.0)
        if t6 > 20 and new_n <= 4000:
            a_out, _ = _lp_fw_optimize(a_long, t6, solver, verbose=False,
                                        cg_threshold=0.93, max_constraints=250)
            try_update(a_out, f"lpfw_n{new_n}")

    # === Final validation ===
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*60}")
        print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
        print(f"{'='*60}")
    else:
        print("ERROR: No valid sequence found!")
        best_seq = np.ones(3200) / 3200
        best_score = _eval_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
