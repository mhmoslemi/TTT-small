import numpy as np
import time
import cvxpy as cp
from scipy.fft import next_fast_len

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    _global_solver = cp.ECOS
    return _global_solver


def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_gradient(a, ac, top_k=20, beta=10.0):
    """Compute (soft) subgradient of max conv(a,a) at a."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = np.exp(beta * (top_vals - top_vals[0]))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        ki = int(k)
        s = max(0, ki - n + 1)
        e = min(n, ki + 1)
        idx = np.arange(s, e)
        partner = ki - idx
        grad[idx] += 2.0 * wi * a[partner]
    return grad


def projected_gradient_optimize(a0, time_limit, top_k=30, beta=10.0, verbose=False, rng_seed=42):
    """Nesterov-accelerated projected gradient with BB step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    g_prev = None
    s_prev = None
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        ac = fast_conv(y)
        g = compute_gradient(y, ac, top_k=top_k, beta=beta)
        
        # Barzilai-Borwein step size
        if g_prev is not None and s_prev is not None:
            ys = (x - x_prev) @ (g - g_prev)
            ss = (g - g_prev) @ (g - g_prev)
            if ss > 1e-20:
                eta = ys / ss if abs(ys) > 1e-15 else 1.0 / np.sqrt(ss)
                eta = np.clip(eta, 1e-8, 1.0)
            else:
                eta = 1.0 / (np.sqrt(t) + 1.0)
        else:
            eta = 1.0 / (np.sqrt(t) + 1.0)
        
        a_new = y - eta * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        s_prev = x - x_prev
        g_prev = g
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
        
        # Periodic restart
        if t % 5000 == 0 and M_new > best_m * 1.01:
            pert = rng.normal(0, 0.02, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            g_prev = None
            s_prev = None
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PG t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_K=3000):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    K_sel = min(200, 2 * n - 1)
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        # Select constraints
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            # Select top K by value (closest to M)
            binding_idx = np.argsort(ac)[::-1][:K_sel]
        
        # Build constraint matrix
        K = len(binding_idx)
        T = np.zeros((K, n))
        for i, k in enumerate(binding_idx):
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            T[i, s:e] = y[ki - (np.arange(s, e))]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y_new / sy
        
        beta_t = (t - 1.0) / (t + 2.0)
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")
        
        # Adapt K
        if t % 10 == 0:
            K_sel = min(K_sel + 50, max_K, 2 * n - 1)
    
    return best_a, best_m


def spectral_flatten(a0, time_limit, n_iter=200, verbose=False):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        fft_size = next_fast_len(2 * n - 1)
        A = np.fft.rfft(a, fft_size)
        
        # Flatten magnitude
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude (but keep DC component reasonable)
        target_mag = np.ones_like(mag)
        target_mag[0] = mag[0]  # Keep DC (sum) fixed
        
        # Interpolate between current and target
        blend = 0.3
        new_mag = (1 - blend) * mag + blend * target_mag * (mag[0] / target_mag[0])
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        M_new = fast_conv(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with sensitivity-based moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    n_top = 10
    
    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Sensitivity
        sens = np.zeros(n)
        for k in top_k[:5]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue
            
            # Find low-sensitivity destination
            low_sens = np.argsort(sens)[:20]
            for dst in low_sens:
                if dst == src:
                    continue
                for frac in [0.02, 0.01, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.005, 0.03)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init(n, kind='sine', seed=0):
    rng = np.random.default_rng(seed)
    if kind == 'sine':
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif kind == 'u_shape':
        x = np.linspace(-1, 1, n)
        a = 1.0 + 2.0 * x ** 2
    elif kind == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.2) / 0.15) ** 2) + np.exp(-((x - 0.8) / 0.15) ** 2) + 0.01
    elif kind == 'random_smooth':
        a = np.ones(n) * 0.5
        for freq in range(1, 20):
            coeff = rng.normal(0, 0.1 / np.sqrt(freq))
            phase = rng.uniform(0, 2 * np.pi)
            a += coeff * np.sin(2 * np.pi * freq * np.linspace(0, 1, n) + phase)
    else:
        a = np.ones(n)
    return norm_sum1(a)

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_gradient(a, ac, top_k=30, beta=10.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = np.exp(beta * (top_vals - top_vals[0]))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        ki = int(k)
        s = max(0, ki - n + 1)
        e = min(n, ki + 1)
        idx = np.arange(s, e)
        partner = ki - idx
        grad[idx] += 2.0 * wi * a[partner]
    return grad
