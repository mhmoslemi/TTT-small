import numpy as np
import time
import cvxpy as cp

def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_J(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()

def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def symmetrize(a):
    """Make a sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return norm_sum1(a_sym)

def make_u_shape(n, sharpness=2.0):
    """Create a U-shaped initialization."""
    x = np.linspace(0, 1, n)
    t = 2 * x - 1  # range [-1, 1]
    a = np.abs(t) ** sharpness + 0.01
    return norm_sum1(a)

def make_cos_u(n, power=2.0):
    """Cosine U-shape."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** power + 0.005
    a = np.abs(a)
    return norm_sum1(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using symmetry-enforced optimization,
subgradient descent with backtracking line search, exact quadratic peak-shaving
moves, and LP Frank-Wolfe polishing.
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_J(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()

def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def symmetrize(a):
    """Make a sequence symmetric and normalize."""
    a_sym = 0.5 * (a + a[::-1])
    return norm_sum1(a_sym)

def make_u_shape(n, gamma=1.0):
    """U-shaped initialization parameterized by gamma (higher = sharper U)."""
    x = np.linspace(0, 1, n)
    t = 2.0 * x - 1.0
    a = np.abs(t) ** (1.0 / gamma) + 0.005
    return norm_sum1(a)

def make_quadratic_u(n, c=0.1):
    """Quadratic U-shape: a[j] = (2j/(n-1) - 1)^2 + c."""
    x = np.linspace(0, 1, n)
    t = 2.0 * x - 1.0
    a = t**2 + c
    return norm_sum1(a)

def subgradient_bt(a0, time_limit, seed=42, top_k=20, beta=3.0, exact_frac=0.3):
    """
    Subgradient descent with backtracking line search.
    Uses exponential (mirror) descent on the simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    eta = 1.0 / n  # initial step size
    stall = 0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(a)
        M = ac.max()
        J = 2.0 * n * M
        
        # Compute subgradient
        use_exact = rng.random() < exact_frac
        if use_exact:
            k = int(np.argmax(ac))
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k - idx
            g = np.zeros(n)
            g[idx] = 2.0 * a[j_idx]
        else:
            k_max = min(top_k, len(ac))
            top_idx = np.argsort(ac)[::-1][:k_max]
            top_vals = np.maximum(ac[top_idx], 1e-30)
            w = top_vals ** beta
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(k_max) / k_max
            else:
                w = w / w_sum
            g = np.zeros(n)
            for wi, k in zip(w, top_idx):
                k_int = int(k)
                i_start = max(0, k_int - n + 1)
                i_end = min(n, k_int + 1)
                if i_start >= i_end:
                    continue
                idx = np.arange(i_start, i_end)
                j_idx = k_int - idx
                g[idx] += 2.0 * wi * a[j_idx]
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Backtracking line search
        a_old = a.copy()
        J_old = J
        found = False
        for _ in range(30):
            # Mirror descent step
            log_a = np.log(np.maximum(a_old, 1e-30)) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            a_new = a_new / s_new
            
            J_new = compute_J(a_new)
            if J_new < J_old - 1e-14:
                a = a_new
                found = True
                break
            else:
                eta = eta * 0.5
        
        if not found:
            # Try very small step
            eta = max(eta * 0.5, 1e-10)
            continue
        
        # If we made progress, try increasing eta a bit
        eta = min(eta * 1.1, 2.0 / n)
        
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > 2000:
            stall = 0
            sigma = 0.02
            pert = rng.normal(0, sigma, n) * best_a
            a_r = best_a + pert
            a = norm_sum1(a_r)
            eta = 1.0 / n
    
    return best_a, best_J

import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")

        except Exception:

    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")

def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_J(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()

def compute_J_and_conv(a):
    """Compute both objective and full convolution."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max(), c

def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def symmetrize(a):
    """Make a sequence symmetric and normalize."""
    a_sym = 0.5 * (np.asarray(a, dtype=np.float64) + np.asarray(a, dtype=np.float64)[::-1])
    return norm_sum1(a_sym)

def make_u_shape(n, gamma=1.5):
    """U-shaped initialization. gamma < 1 gives sharper U."""
    x = np.linspace(0, 1, n)
    t = 2.0 * x - 1.0
    a = np.abs(t) ** (1.0 / gamma) + 0.005
    return norm_sum1(a)

def make_quadratic_u(n, c=0.15):
    """Quadratic U-shape: a[j] = (2j/(n-1) - 1)^2 + c."""
    x = np.linspace(0, 1, n)
    t = 2.0 * x - 1.0
    a = t**2 + c
    return norm_sum1(a)

def make_cos_u(n, power=2.0):
    """Cosine U-shape."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** power + 0.003
    a = np.abs(a)
    return norm_sum1(a)

def get_subgradient(ac, a, exact=False, top_k=20, beta=3.0):
    """Compute subgradient of max(conv(a,a)) at a."""
    n = len(a)
    g = np.zeros(n)
    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        g[idx] = 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g

def subgradient_descent(a0, time_limit, seed=42, top_k=20, beta=3.0,
                        exact_frac=0.35, verbose=False):
    """
    Subgradient descent with adaptive step size on the simplex.
    Uses exponential (mirror) descent for non-negativity.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 1.5 / n
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_exact_frac = exact_frac

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(a)
        M = ac.max()
        J = 2.0 * n * M

        # Adaptive subgradient
        if rng.random() < cur_exact_frac:
            g = get_subgradient(ac, a, exact=True)
        else:
            g = get_subgradient(ac, a, exact=False, top_k=cur_top_k, beta=cur_beta)

        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break

        # Polyak-like step with cap
        lower_bound = 1.0 / (2 * n - 1)
        gap = M - lower_bound
        eta_polyak = 2.0 * gap / (g_norm_sq + 1e-20)
        eta = min(eta_polyak, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)

        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-30)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        J_new = compute_J(a)

        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive restart
        if stall > 3000:
            stall = 0
            sigma = 0.015
            pert = rng.normal(0, sigma, n) * best_a
            a_r = best_a + pert
            a = norm_sum1(a_r)
            eta = 1.5 / n
            cur_top_k = min(cur_top_k + 5, 60)
            cur_beta = max(cur_beta - 0.2, 1.5)
            cur_exact_frac = min(cur_exact_frac + 0.03, 0.7)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SGD restart ({elapsed:.0f}s): J={best_J:.6f}, tk={cur_top_k}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J

def peak_shaving(a0, time_limit, seed=77, verbose=False):
    """
    Targeted peak-equalization via exact quadratic transfer moves.
    Identifies peak contributors and finds optimal transfer amounts.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        J_curr = 2.0 * n * M_curr

        # Find top peaks
        n_peaks = min(10, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # For the highest peak, find contributing pairs
        k_star = int(top_lags[0])
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_star - idx
        products = a[idx] * a[j_idx]

        # Find high-contribution positions (from the peak lag)
        prod_order = np.argsort(products)[::-1]
        high_contrib = idx[prod_order[:min(20, len(prod_order))]]

        # Find "safe" positions: those with low sensitivity to ALL top peaks
        sensitivity = np.zeros(n)
        for k in top_lags[:5]:
            k_int = int(k)
            i_s = max(0, k_int - n + 1)
            i_e = min(n, k_int + 1)
            ii = np.arange(i_s, i_e)
            jj = k_int - ii
            sens_vals = a[ii] * a[jj]
            for pos, sv in zip(ii, sens_vals):
                sensitivity[pos] += sv
            for pos, sv in zip(jj, sens_vals):
                sensitivity[pos] += sv

        # Safe positions: low sensitivity and not in the high-contribution set
        safe_mask = (sensitivity < np.percentile(sensitivity, 30))
        safe_positions = np.where(safe_mask)[0]
        # Remove positions that are high contributors
        safe_set = set(safe_positions.tolist())
        for hc in high_contrib:
            safe_set.discard(int(hc))
        safe_positions = np.array(list(safe_set)) if safe_set else safe_positions

        if len(high_contrib) == 0 or len(safe_positions) == 0:
            stall += 1
            continue

        improved = False

        # Try transfers from top contributors to safe positions
        for _ in range(40):
            if time.monotonic() - t0 > time_limit:
                break

            src = int(high_contrib[rng.integers(len(high_contrib))])
            dst = int(safe_positions[rng.integers(len(safe_positions))])
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Find optimal transfer amount using 1D search
            # After transferring 'amt' from src to dst:
            # New a[src] = src_val - amt, new a[dst] = a[dst] + amt
            # We want to minimize max(conv) approximately
            # Focus on the peak lag: c[k_star] changes by
            #   delta = (a[dst+... ]*(src_val - amt) + ...) - (original)
            # Simplified: use a few trial amounts and pick best

            best_amt = 0
            best_trial_J = J_curr

            # Try logarithmically spaced amounts
            amts = src_val * np.array([0.001, 0.003, 0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5])
            for amt in amts:
                if amt >= src_val:
                    continue
                a_trial = a.copy()
                a_trial[src] = src_val - amt
                a_trial[dst] = a[dst] + amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                a_trial = a_trial / st
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_amt = amt

            if best_amt > 0 and best_trial_J < best_J - 1e-14:
                a[src] = a[src] - best_amt
                a[dst] = a[dst] + best_amt
                a = norm_sum1(a)
                best_J = best_trial_J
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            # Escape: random perturbation
            if stall > 50:
                n_pert = max(10, n // 50)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = norm_sum1(a)
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    PeakShave done ({elapsed:.0f}s): J={best_J:.6f}")
    return best_a, best_J

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300):
    """
    LP-based Frank-Wolfe with Nesterov acceleration.
    Inner LP: max sum(g) s.t. conv(a_current, g) <= M, g >= 0.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix for current y
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select binding constraints
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        y = y_new
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m

def simultaneous_peak_reduce(a0, time_limit, seed=55, verbose=False):
    """
    Simultaneous multi-pair peak reduction.
    Identifies all top contributors to the peak and finds a joint transfer plan.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_conv(a)
        M_curr = ac.max()
        J_curr = 2.0 * n * M_curr

        k_star = int(np.argmax(ac))
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_star - idx
        products = a[idx] * a[j_idx]

        # Top 5 contributors to the peak
        prod_order = np.argsort(products)[::-1]
        top_contribs = prod_order[:min(5, len(prod_order))]

        if len(top_contribs) == 0:
            stall += 1
            continue

        # For each top contributor pair, find a safe destination
        # Compute sensitivity across all top-3 peaks
        top_lags = np.argsort(ac)[::-1][:3]
        sensitivity = np.zeros(n)
        for k in top_lags:
            k_int = int(k)
            i_s = max(0, k_int - n + 1)
            i_e = min(n, k_int + 1)
            ii = np.arange(i_s, i_e)
            jj = k_int - ii
            for pos in ii:
                sensitivity[pos] += a[pos] * a[k_int - pos] if (0 <= k_int - pos < n) else 0
            for pos in jj:
                sensitivity[pos] += a[pos] * a[k_int - pos] if (0 <= k_int - pos < n) else 0

        # Safe destinations
        threshold = np.percentile(sensitivity, 20)
        safe_mask = sensitivity < threshold
        safe_positions = np.where(safe_mask)[0]

        if len(safe_positions) == 0:
            stall += 1
            continue

        improved = False
        # Try reducing each top contributor
        for ci in top_contribs:
            if time.monotonic() - t0 > time_limit:
                break

            src = int(idx[ci])
            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Try multiple destinations
            n_try = min(5, len(safe_positions))
            dst_indices = rng.choice(len(safe_positions), size=n_try, replace=False)

            best_improvement = 0
            best_move = None

            for di in dst_indices:
                dst = int(safe_positions[di])
                if dst == src:
                    continue
                # Try a range of transfer amounts
                for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    J_trial = compute_J(a_trial)
                    improvement = best_J - J_trial
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_move = (src, dst, amt)

            if best_move is not None and best_improvement > 1e-14:
                src, dst, amt = best_move
                a[src] -= amt
                a[dst] += amt
                a = norm_sum1(a)
                best_J = compute_J(a)
                best_a = a.copy()
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 30:
                # Escape move
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.02, 0.1)
                    a[i] -= amt
                    a[j] += amt
                    a = norm_sum1(a)
                    if compute_J(a) < best_J:
                        best_J = compute_J(a)
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SimPeak ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_u_shape(n_base, gamma=1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # Symmetrize the starting point
    best_sym = symmetrize(best_seq)
    sym_score = evaluate_safe(best_sym)
    if sym_score < best_score:
        print(f"Symmetrization improved: {best_score:.6f} -> {sym_score:.6f}")
        best_seq = best_sym
        best_score = sym_score

    # === Phase 1: Multi-init subgradient ===
    print(f"\n=== Phase 1: Subgradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.22, 380.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        # Parent (symmetrized)
        inits.append(best_sym)
        labels.append("parent_sym")

        # Different U-shapes
        inits.append(make_u_shape(n0, gamma=1.2))
        labels.append("u_gamma1.2")
        inits.append(make_u_shape(n0, gamma=2.0))
        labels.append("u_gamma2.0")
        inits.append(make_quadratic_u(n0, c=0.08))
        labels.append("quad_c0.08")
        inits.append(make_quadratic_u(n0, c=0.25))
        labels.append("quad_c0.25")
        inits.append(make_cos_u(n0, power=3.0))
        labels.append("cos3")

        # Blend parent with U-shape
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_u_shape(n0, gamma=1.5)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove near-duplicates
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for u in unique_inits:
                if np.sum(np.abs(init - u)) < 0.03:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(15.0, t_phase1 / n_inits)
        print(f"  {n_inits} initializations, ~{t_per:.0f}s each")

        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 30)
            if t_avail < 10:
                break
            print(f"  SGD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = subgradient_descent(init, t_avail, seed=100 + idx,
                                            top_k=20, beta=3.0,
                                            exact_frac=0.35, verbose=False)
            # Symmetrize output
            a_out_sym = symmetrize(a_out)
            try_update(a_out, f"sgd_{lbl}")
            try_update(a_out_sym, f"sgd_{lbl}_sym")

    # === Phase 2: Peak shaving ===
    print(f"\n=== Phase 2: Peak shaving (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 20:
        a_in = symmetrize(best_seq)
        a_out, _ = peak_shaving(a_in, t2, verbose=True, seed=77)
        try_update(a_out, f"peakshave_n{len(a_in)}")
        a_out_sym = symmetrize(a_out)
        try_update(a_out_sym, f"peakshave_sym_n{len(a_in)}")

    # === Phase 3: Simultaneous peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 15:
        a_in = symmetrize(best_seq)
        a_out, _ = simultaneous_peak_reduce(a_in, t3, verbose=True, seed=55)
        try_update(a_out, f"simpeak_n{len(a_in)}")
        a_out_sym = symmetrize(a_out)
        try_update(a_out_sym, f"simpeak_sym_n{len(a_in)}")

    # === Phase 4: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW (n={n_best}) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K=300)
        try_update(a_out, f"lpfw_n{n_best}")
        a_out_sym = symmetrize(a_out)
        try_update(a_out_sym, f"lpfw_sym_n{n_best}")

    # === Phase 5: SGD polish with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: SGD polish (n={n_best}) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_descent(a_in, t5, seed=999,
                                        top_k=40, beta=2.0,
                                        exact_frac=0.5, verbose=False)
        try_update(a_out, f"sgd_polish_n{n_best}")
        a_out_sym = symmetrize(a_out)
        try_update(a_out_sym, f"sgd_polish_sym_n{n_best}")

    # === Phase 6: Peak shaving again ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t6, verbose=True, seed=333)
        try_update(a_out, f"peakshave2_n{len(a_in)}")
        a_out_sym = symmetrize(a_out)
        try_update(a_out_sym, f"peakshave2_sym_n{len(a_in)}")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")
    test_lengths = []
    for delta in [-300, -150, 150, 300, 500]:
        new_n = n_best + delta
        if 1500 <= new_n <= 5000:
            test_lengths.append(new_n)

    for new_n in test_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        a_long = symmetrize(a_long)
        t7a = min(remaining * 0.2, 120.0)
        if t7a > 15:
            a_out, _ = subgradient_descent(a_long, t7a, seed=555,
                                            top_k=20, beta=3.0,
                                            exact_frac=0.4, verbose=False)
            try_update(a_out, f"sgd_n{new_n}")
            a_out_sym = symmetrize(a_out)
            try_update(a_out_sym, f"sgd_n{new_n}_sym")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_in2 = norm_sum1(best_seq)
            a_out2, _ = peak_shaving(a_in2, min(remaining * 0.15, 60.0), verbose=False, seed=333)
            try_update(a_out2, f"peakshave3_n{len(a_in2)}")

    # === Phase 8: Final LP if time permits ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 4000:
        t8 = min(remaining * 0.3, 200.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=400)
            try_update(a_out, f"lpfw_final_n{n_best}")
            a_out_sym = symmetrize(a_out)
            try_update(a_out_sym, f"lpfw_final_sym_n{n_best}")

    # === Phase 9: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final peak shave (n={n_best}) ===")
    if remaining > 20:
        t9 = min(remaining * 0.4, 150.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_shaving(a_in, t9, verbose=True, seed=1234)
            try_update(a_out, f"peakshave_final_n{n_best}")
            a_out_sym = symmetrize(a_out)
            try_update(a_out_sym, f"peakshave_final_sym_n{n_best}")

    # === Phase 10: Final SGD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 25:
        print(f"\n=== Phase 10: Final SGD (n={n_best}) ===")
        t10 = min(remaining * 0.5, 100.0)
        if t10 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = subgradient_descent(a_in, t10, seed=4321,
                                            top_k=30, beta=2.5,
                                            exact_frac=0.5, verbose=False)
            try_update(a_out, f"sgd_final_n{n_best}")
            a_out_sym = symmetrize(a_out)
            try_update(a_out_sym, f"sgd_final_sym_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_u_shape(n_fb, gamma=1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
