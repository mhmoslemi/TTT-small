If S = {top K lags by current autocorrelation}, the LP is a good approximation.

A better approach: use the "lazy constraint" method.
1. Start with S = {argmax of current autocorrelation} (just 1 constraint).
2. Solve the LP.
3. Check the solution g: compute (f*g)[k] for all k.
4. If there are violations (k where (f*g)[k] > M*), add them to S.
5. Repeat 2-4 until no violations (or S gets too large).

This is the standard cutting-plane approach for LPs.

- Use a warm start for the LP (keep the dual variables from the previous iteration).
- Limit the number of constraints to avoid solver slowdown.

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _MODULE_SOLVER

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _evaluate(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _parametric_init(n, time_limit, best_init, best_score):
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    best_a = best_init.copy()
    best_val = best_score
    i = np.arange(n, dtype=np.float64)

    def _try(a, label):
        nonlocal best_a, best_val
        a_n = _normalize(a)
        ac = _fast_autocorr(a_n)
        val = 2.0 * n * ac.max()
        if val < best_val:
            best_val = val
            best_a = a_n.copy()
            print(f"    Param [{label}]: J={val:.6f}")

    # Uniform
    _try(np.ones(n), "uniform")
    # Power functions
    for p in [0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        _try((i + 1.0) ** p, f"pow{p}")
    # Cosine-like
    for f in [1, 2, 3]:
        _try(np.cos(np.pi * f * (x - 0.5)) ** 2 + 0.01, f"cos{f}")
    # Sine
    for f in [1, 2, 3]:
        _try(np.sin(np.pi * f * x) ** 2 + 0.01, f"sin{f}")
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for s in [0.1, 0.2, 0.3]:
            _try(np.exp(-((x - c) ** 2) / (2 * s * s)), f"g{c}_{s}")
    # Bimodal
    _try(np.exp(-((x-0.25)**2)/0.05) + np.exp(-((x-0.75)**2)/0.05), "bimodal")
    # Exponential decay
    for r in [1, 3, 5, 10]:
        _try(np.exp(-r * x) + 0.01, f"er{r}")
        _try(np.exp(r * x) + 0.01, f"enr{r}")
    # Beta-like
    for a_p in [0.5, 1, 2, 5]:
        for b_p in [0.5, 1, 2, 5]:
            _try((i+1)**(a_p-1) * (n-i)**(b_p-1) + 0.01, f"b{a_p}_{b_p}")

    elapsed = time.monotonic() - t0
    print(f"  Parametric done ({elapsed:.0f}s): best J={best_val:.6f}")
    return best_a, best_val

def _lp_fw(a0, time_limit, solver, initial_K=200, max_K=3000, verbose=True):
    """LP-based Frank-Wolfe with lazy constraints and adaptive step size."""
    n = len(a0)
    a = _normalize(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()  # Current point
    x_ = a.copy()  # Nesterov point
    g_dir = a.copy()

    # Initialize constraint set with top K autocorrelation values
    ac_y = _fast_autocorr(y)
    init_idx = np.argsort(ac_y)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)

    # For lazy constraints: track all constraints ever added
    all_constraints = list(constraint_set)
    iter_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = _fast_autocorr(x_)
        if ac_x.max() < best_m - 1e-15:
            best_m = ac_x.max()
            best_a = x_.copy()

        # Manage constraint set size
        if len(constraint_set) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_set])
            keep = np.argsort(margins)[:max_K]
            constraint_list_tmp = sorted(constraint_set)
            constraint_set = set(int(constraint_list_tmp[i]) for i in keep)

        constraint_list = sorted(constraint_set)
        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        lp_cons = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), lp_cons)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(constraint_list) > initial_K:
                constraint_list = constraint_list[:initial_K]
                constraint_set = set(constraint_list)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Check for violated constraints (lazy constraint addition)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_

_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-6)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:40]]
            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov step with line search
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov acceleration
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x_ = x_new
        iter_count += 1

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m
