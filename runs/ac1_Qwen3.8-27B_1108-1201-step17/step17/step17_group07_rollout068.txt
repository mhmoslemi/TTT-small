def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def score_fast(a):
    """Scale-variant score computed locally (no evaluate_sequence call)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    mb = fast_conv(a).max()
    return float(2.0 * n * mb / (s * s))


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def symmetrize(a):
    a = np.asarray(a, dtype=np.float64)
    a = 0.5 * (a + a[::-1].copy())
    return norm_sum1(a)


def better_of_sym(a):
    """Return whichever of a and sym(a) has the lower score, renormalized."""
    a = np.asarray(a, dtype=np.float64)
    s0 = score_fast(a)
    sp = score_fast(symmetrize(a))
    if sp < s0:
        return symmetrize(a), sp
    return a / a.sum(), s0


def proj_simplex(v):
    """Euclidean projection onto {x >= 0, sum x = 1} (Duchi et al.)."""
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = len(v)
    if n <= 1:
        return np.ones(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0][-1]
    tau = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - tau, 0.0)
    s = out.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return out / s

        idx = np.arange(m)
    else:
        idx = np.argpartition(b, -K)[-K:]
    bw = b[idx]
    wm = np.exp(bw / mu)
    wm = wm / wm.sum()
    g = np.zeros(n)
    for k, w in zip(idx, wm):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            jidx = np.arange(j_start, j_end + 1)
            g[jidx] += w * (2.0 * y[ki - jidx])
    return g


def mirror_accel(a0, time_limit, mode="mirror", K=24, mu=0.5, c0=3.0,
                 seed=0, verbose=False):
    """Accelerated (Nesterov) subgradient method with adaptive top-K smoothing.
    mode='mirror'    -> multiplicative (KL) update.
    mode='euclid'    -> Euclidean proxy to simplex (subgradient step + project).
    """
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = score_fast(a)
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    c = c0
    c_base = c0
    stall = 0
    max_stall = 4000
    rng = np.random.default_rng(1000 + seed + n)
    upcount = 0

    for t in range(1, 40000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        b = fast_conv(y)
        M = b.max()
        near = int(np.sum(b >= M * 0.995))
        k_use = min(max(K, near + 4), 120)
        g = topk_smooth_subgrad(y, b, k_use, mu)

        eta = c / (np.sqrt(t) + 1.0)
        if mode == "mirror":
            log_a = np.log(y) - eta * g
            mlog = log_a.max()
            x_new = np.exp(log_a - mlog)
            x_new = np.maximum(x_new, 1e-30)
            x_new = x_new / x_new.sum()
        else:
            x_new = proj_simplex(y - eta * g)

        x_prev = x
        x = x_new

        J = 2.0 * n * fast_conv(x).max()  # sum==1 so J = 2n*max_b
        if J < best_m - 1e-18:
            best_m = J
            best_a = x.copy()
            stall = 0
            upcount += 1
            if upcount % 40 == 0:
                c = min(c_base * 1.25, c * 1.03)
        else:
            stall += 1
            if stall % 200 == 0:
                c = max(0.4, c * 0.92)
                upcount = 0

        if stall > max_stall:
            stall = 0
            max_stall = max(700, int(max_stall * 0.65))
            strat = int(rng.integers(4))
            if strat == 0:
                gbest = topk_smooth_subgrad(best_a, fast_conv(best_a), min(K, 2 * n - 1), mu)
                gbest = gbest / (np.abs(gbest).max() + 1e-15)
                xr = best_a * (1.0 - 0.6 * gbest)
            elif strat == 1:
                xr = best_a * (1.0 + rng.normal(0, 0.02, n))
            elif strat == 2:
                xr = best_a.copy()
                idx = rng.choice(n, size=min(n // 4, 800), replace=False)
                xr[idx] *= rng.uniform(0.4, 1.6, len(idx))
            else:
                xr = best_a.copy()
                xrr = xr[::-1].copy()
                xr = 0.5 * (xr + xrr) + rng.normal(0, 0.01, n)  # symmetric kick
            xr = np.maximum(xr, 1e-10)
            xr = xr / xr.sum()
            x = xr
            x_prev = xr
            c = max(0.5, c_base * 0.6)
            mu = max(0.02, mu * 0.85)  # anneal smoothing

        if verbose and t % 25000 == 0:
            el = time.monotonic() - t0
            print("      %s t=%d (%.1fs) J=%.8f stall=%d c=%.3f mu=%.4f"
                  % (mode, t, el, best_m, stall, c, mu))

    return best_a, best_m / (2.0 * n)  # return normalized-a and max_b


def lp_remez(a0, time_limit, solver=None, r_init=0.08, verbose=False):
    """LP Remez: minimize M s.t. linearized active-lag constraints + trust region."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_J = score_fast(a)
    t0 = time.monotonic()
    r = r_init
    noimp = 0
    rng = np.random.default_rng(7 + n)

    for t in range(1, 60000):
        if time.monotonic() - t0 > time_limit:
            break
        if not _HAVE_CP:
            break

        b = fast_conv(a)
        mb = b.max()
        thr = mb * 0.999
        active = np.where(b >= thr)[0]
        if len(active) > 44:
            active = np.argsort(b)[::-1][:44]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        try:
            av = cp.Variable(n, nonneg=True)
            Mv = cp.Variable()
            cons = [cp.sum(av) == 1, av <= a * (1.0 + r)]
            for k in active:
                ki = int(k)
                js = max(0, ki - n + 1)
                je = min(n - 1, ki)
                if js <= je:
                    idx = np.arange(js, je + 1)
                    coeffs = 2.0 * a[ki - idx]
                    cons.append(cp.sum(cp.multiply(coeffs, av[idx])) <= Mv)
            prob = cp.Problem(cp.Minimize(Mv), cons)
            if solver is not None:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
            cand = None
            if prob.status in ("optimal", "optimal_inaccurate") and av.value is not None:
                cand = np.maximum(np.asarray(av.value, dtype=np.float64), 0.0)
        except Exception:
            cand = None

        accepted = False
        if cand is not None and cand.sum() > 1e-12:
            cand = cand / cand.sum()
            for tf in [1.0, 0.7, 0.5, 0.3, 0.15]:
                trial = (1.0 - tf) * a + tf * cand
                trial = np.maximum(trial, 0.0)
                s = trial.sum()
                if s < 1e-15:
                    continue
                trial = trial / s
                Jt = score_fast(trial)
                if Jt < best_J - 1e-15:
                    a = trial
                    best_J = Jt
                    best_a = a.copy()
                    accepted = True
                    noimp = 0
                    r = min(0.35, r * 1.12)
                    break
        if not accepted:
            noimp += 1
            r = max(0.004, r * 0.7)
            if noimp > 25:
                idx = rng.choice(n, size=min(n // 5, 500), replace=False)
                a[idx] *= rng.uniform(0.6, 1.4, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                noimp = 0
                r = min(0.3, r * 1.5)

        if verbose and t % 15 == 0:
            print("      Remez t=%d (%.1fs) J=%.8f r=%.4f"
                  % (t, time.monotonic() - t0, best_J, r))

    return best_a, best_J


def pair_polish(a0, time_limit, seed=123, verbose=False):
    """Random 1-D pair transfers with ternary search on the transfer amount."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_J = score_fast(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = max(15, n // 150)

    def eval_transfer(i, j, amt):
        if amt <= 0.0 or i == j:
            return best_J
        at = a.copy()
        at[i] -= amt
        at[j] += amt
        if at[i] < -1e-15:
            return float("inf")
        at = np.maximum(at, 0.0)
        ss = at.sum()
        if ss < 1e-15:
            return float("inf")
        at = at / ss
        return score_fast(at)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        npairs = min(60, n // 4)
        for _ in range(npairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-13:
                continue
            lo, hi = 0.0, a[i]
            # ternary search over amount in (lo,hi)
            fl = best_J
            for _tt in range(14):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = eval_transfer(i, j, m1)
                f2 = eval_transfer(i, j, m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
                if min(f1, f2) < best_J - 1e-15:
                    improved = True
            amt = 0.5 * (lo + hi)
            Jb = eval_transfer(i, j, amt)
            if Jb < best_J - 1e-15:
                at = a.copy()
                at[i] -= amt
                at[j] += amt
                at = np.maximum(at, 0.0)
                at = at / at.sum()
                a = at
                best_J = Jb
                best_a = a.copy()
        if not improved:
            stall += 1

    return best_a, best_J


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def make_init(n, kind="triangle"):
    x = np.linspace(0.0, 1.0, n)
    if kind == "triangle":
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == "raised_cos":
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == "gaussian":
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.1)
    elif kind == "flat":
        a = np.ones(n)
    elif kind == "quartic":
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == "edge_bump":
        a = np.ones(n)
        w = max(2, n // 8)
        a[:w] += 2.0
        a[-w:] += 2.0
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _update(best_seq, best_score, a, label):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    a = a / s
    a, sc = better_of_sym(a)
    if np.isfinite(sc) and sc < best_score:
        return a.copy(), sc
    return best_seq, best_score


def _report(label, score, a, t0):
    print("  *** NEW BEST [%s]: %.8f  (n=%d, t=%.0fs)"
          % (label, score, len(a), time.monotonic() - t0))


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)
    solver = get_solver()
    print("Total budget=%.0fs  solver=%s  cvxpy=%s" %
          (time_budget, solver, _HAVE_CP))

    # ---- load parent ----
    best_seq = None
    best_score = float("inf")
    n_base = 2600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_seq, best_score = better_of_sym(best_seq)
            n_base = len(best_seq)
            _report("parent+sym", best_score, best_seq, t_global)
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_init(n_base, "triangle")
        best_seq, best_score = better_of_sym(best_seq)
        _report("default_init", best_score, best_seq, t_global)

    n0 = len(best_seq)

    # ---- Phase 1: accelerated MIRROR from parent (largest chunk) ----
    rem = time_budget - (time.monotonic() - t_global)
    t1 = min(rem * 0.34, 650.0)
    print("\n=== P1 mirror_accel n=%d (budget=%.0fs) ===" % (n0, t1))
    if t1 > 40:
        a_out, _ = mirror_accel(norm_sum1(best_seq), t1, mode="mirror",
                                K=26, mu=0.6, c0=4.0, seed=1, verbose=True)
        best_seq, old = _update(best_seq, best_score, a_out, "mirror_parent")
        if best_score < old:
            _report("mirror_parent", best_score, best_seq, t_global)

    # ---- Phase 2: accelerated EUCLIDEAN (different geometry) ----
    rem = time_budget - (time.monotonic() - t_global)
    t2 = min(rem * 0.20, 400.0)
    print("\n=== P2 euclid_accel n=%d (budget=%.0fs) ===" % (len(best_seq), t2))
    if t2 > 40:
        a_out, _ = mirror_accel(norm_sum1(best_seq), t2, mode="euclid",
                                K=20, mu=0.4, c0=2.5, seed=2, verbose=True)
        best_seq, old = _update(best_seq, best_score, a_out, "euclid_best")
        if best_score < old:
            _report("euclid_best", best_score, best_seq, t_global)

    # ---- Phase 3: LP Remez polish ----
    rem = time_budget - (time.monotonic() - t_global)
    t3 = min(rem * 0.16, 350.0)
    print("\n=== P3 LP Remez n=%d (budget=%.0fs) ===" % (len(best_seq), t3))
    if t3 > 30:
        a_out, _ = lp_remez(norm_sum1(best_seq), t3, solver=solver,
                            r_init=0.10, verbose=True)
        best_seq, old = _update(best_seq, best_score, a_out, "remez_main")
        if best_score < old:
            _report("remez_main", best_score, best_seq, t_global)

    # ---- Phase 4: multi-init exploration at base length ----
    rem = time_budget - (time.monotonic() - t_global)
    t4 = min(rem * 0.12, 260.0)
    inits = ["raised_cos", "gaussian", "edge_bump", "quartic"]
    print("\n=== P4 multi-init n=%d (budget=%.0fs) ===" % (n0, t4))
    if t4 > 40:
        per = t4 / len(inits)
        for kind in inits:
            rem = time_budget - (time.monotonic() - t_global)
            if rem < 20:
                break
            a_init = make_init(n0, kind)
            a_out, _ = mirror_accel(a_init, min(per, rem - 5), mode="mirror",
                                    K=20, mu=0.5, c0=3.0,
                                    seed=10 + n0, verbose=False)
            best_seq, old = _update(best_seq, best_score, a_out, "init_" + kind)
            if best_score < old:
                _report("init_" + kind, best_score, best_seq, t_global)

    # ---- Phase 5: pair polish on global best ----
    rem = time_budget - (time.monotonic() - t_global)
    t5 = min(rem * 0.06, 110.0)
    print("\n=== P5 pair_polish n=%d (budget=%.0fs) ===" % (len(best_seq), t5))
    if t5 > 15:
        a_out, _ = pair_polish(norm_sum1(best_seq), t5, seed=999, verbose=False)
        best_seq, old = _update(best_seq, best_score, a_out, "pairstep")
        if best_score < old:
            _report("pairstep", best_score, best_seq, t_global)

    # ---- Phase 6: alternate length ----
    rem = time_budget - (time.monotonic() - t_global)
    if rem > 120:
        n_alt = n0 + 200
        a_alt = interpolate_sequence(best_seq, n_alt)
        print("\n=== P6 alt-length n=%d (budget=%.0fs) ===" % (n_alt, rem))
        tA = min(rem * 0.45, 220.0)
        if tA > 30:
            a_out, _ = mirror_accel(a_alt, tA, mode="mirror", K=22, mu=0.5,
                                    c0=3.5, seed=41, verbose=False)
            best_seq, old = _update(best_seq, best_score, a_out, "altlen_mirror")
            if best_score < old:
                _report("altlen_mirror", best_score, best_seq, t_global)
        rem = time_budget - (time.monotonic() - t_global)
        if rem > 60 and solver is not None:
            a_out, _ = lp_remez(norm_sum1(best_seq), min(rem, 120.0),
                                solver=solver, r_init=0.05, verbose=False)
            best_seq, old = _update(best_seq, best_score, a_out, "altlen_remez")
            if best_score < old:
                _report("altlen_remez", best_score, best_seq, t_global)

    # ---- Phase 7: final tight polish (tight Remez + light accel + sym) ----
    rem = time_budget - (time.monotonic() - t_global)
    nb = len(best_seq)
    if rem > 60:
        print("\n=== P7 final push n=%d (budget=%.0fs) ===" % (nb, rem))
        tA = min(rem * 0.45, 140.0)
        if tA > 20:
            a_out, _ = mirror_accel(norm_sum1(best_seq), tA, mode="mirror",
                                    K=10, mu=0.05, c0=1.0, seed=7, verbose=False)
            best_seq, old = _update(best_seq, best_score, a_out, "final_mirror")
            if best_score < old:
                _report("final_mirror", best_score, best_seq, t_global)
        rem = time_budget - (time.monotonic() - t_global)
        if rem > 30 and solver is not None:
            a_out, _ = lp_remez(norm_sum1(best_seq), min(rem, 120.0),
                                solver=solver, r_init=0.02, verbose=False)
            best_seq, old = _update(best_seq, best_score, a_out, "final_remez")
            if best_score < old:
                _report("final_remez", best_score, best_seq, t_global)
        rem = time_budget - (time.monotonic() - t_global)
        if rem > 20:
            a_out, _ = pair_polish(norm_sum1(best_seq), min(rem, 60.0),
                                   seed=555, verbose=False)
            best_seq, old = _update(best_seq, best_score, a_out, "final_pair")
            if best_score < old:
                _report("final_pair", best_score, best_seq, t_global)

    # ---- final validation through the real evaluator ----
    if best_seq is None:
        print("ERROR: no valid sequence")
        return [1.0]
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    print("\nFINAL: score=%.8f  n=%d  time=%.0fs"
          % (best_score, len(best_seq), time.monotonic() - t_global))
    return [float(x) for x in best_seq]
