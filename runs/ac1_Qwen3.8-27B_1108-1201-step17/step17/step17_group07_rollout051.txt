def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def proj_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad_full(a, k_star):
    """Compute exact subgradient of max b at lag k_star. O(n)."""
    n = len(a)
    g = np.zeros(n)
    k = int(k_star)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        idx = np.arange(j_start, j_end + 1)
        g[idx] = 2.0 * a[k - idx]
    return g

def compute_subgrad_multi(a, active_lags):
    """Average subgradient over multiple active lags."""
    n = len(a)
    g = np.zeros(n)
    for k in active_lags:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * a[k - idx]
    g /= len(active_lags)
    return g

def remez_lp_full(a, max_cons=60, solver=None, verbose=False):
    """LP Remez without trust region. Minimizes M s.t. sum(a)=1, a>=0, b[k]<=M for active k."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Select active constraints: top max_cons lags
    if max_cons >= len(b):
        active = np.arange(len(b))
    else:
        active = np.argsort(b)[::-1][:max_cons]
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy(), max_b
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), max_b
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), max_b
    a_new = a_new / s
    b_new = fast_conv(a_new)
    return a_new, b_new.max()

def remez_method(a0, time_limit, solver=None, verbose=False):
    """Remez method with line search and adaptive active set expansion."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    no_improve = 0
    max_cons_start = 20
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive number of constraints
        b_cur = fast_conv(a)
        max_cur = b_cur.max()
        # Include more constraints as we converge
        tol = max(0.01, 0.5 * (1.0 - t / 5000.0))
        active_thresh = max_cur * (1.0 - tol)
        n_active = min(max_cons_start + t // 50, 80)
        
        a_cand, m_cand = remez_lp_full(a, max_cons=n_active, solver=solver)
        
        # Line search
        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                break
        
        if not accepted:
            no_improve += 1
            if no_improve > 15:
                # Perturb to escape
                rng = np.random.default_rng(t * 13 + 7)
                strategy = int(rng.integers(3))
                if strategy == 0:
                    idx = rng.choice(n, size=min(n//5, 300), replace=False)
                    a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                elif strategy == 1:
                    a = a * (1 + rng.normal(0, 0.01, n))
                else:
                    # Targeted: perturb around peak correlation location
                    ac = fast_conv(a)
                    peak = int(np.argmax(ac))
                    center = peak // 2
                    spread = max(5, n // 15)
                    pert_idx = np.arange(max(0, center - spread), min(n, center + spread))
                    a[pert_idx] *= rng.uniform(0.85, 1.15, len(pert_idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, no_improve={no_improve}, cons={n_active}")
    
    return best_a, best_m

def logsumexp_lbfgs(a0, time_limit, beta_start=10.0, beta_end=1000.0, n_continuation=5, verbose=False):
    """Optimize using log-sum-exp smoothing with L-BFGS-B and continuation."""
    from scipy.optimize import minimize as sp_minimize
    
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    betas = np.geomspace(beta_start, beta_end, n_continuation)
    
    for beta in betas:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        t_per_beta = remaining / (n_continuation + 1)
        
        def objective(x):
            a_curr = np.maximum(x, 0.0)
            s = a_curr.sum()
            if s < 1e-15:
                return 1e10, np.zeros_like(x)
            a_curr = a_curr / s
            b = fast_conv(a_curr)
            val = 2.0 * n * scipy_logsumexp(beta * b) / (beta * s * s)
            
            # Gradient via FFT-based cross-correlation
            w = np.exp(beta * b - scipy_logsumexp(beta * b))
            fft_size = 1 << (3 * n - 2).bit_length()
            rev_a = a_curr[::-1]
            A = np.fft.rfft(rev_a, fft_size)
            W = np.fft.rfft(np.zeros(fft_size), fft_size)
            # W should be w (length 2n-1)
            W_full = np.zeros(fft_size)
            W_full[:len(w)] = w
            W = np.fft.rfft(W_full)
            conv_result = np.fft.irfft(A * W, fft_size)
            c = conv_result[n-1 : 2*n-1]
            
            L = scipy_logsumexp(beta * b)
            grad = 4.0 * n * c / (s * s) - 4.0 * n * L / (beta * s**3)
            return val, grad
        
        def objective_only(x):
            a_curr = np.maximum(x, 0.0)
            s = a_curr.sum()
            if s < 1e-15:
                return 1e10
            a_curr = a_curr / s
            b = fast_conv(a_curr)
            return 2.0 * n * scipy_logsumexp(beta * b) / (beta * s * s)
        
        try:
            res = sp_minimize(objective_only, a.copy(), method='L-BFGS-B',
                           options={'maxiter': 200, 'maxfun': 500, 'gtol': 1e-8})
            a_new = np.maximum(res.x, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                m_new = fast_conv(a_new).max()
                if m_new < best_m - 1e-16:
                    best_m = m_new
                    best_a = a_new.copy()
                    a = a_new
        except Exception:
            pass
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with pair moves and bisection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            # Try multiple fractions
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'exponential':
        a = np.exp(-3.0 * x)
    elif kind == 'power':
        a = (1.0 - x) ** 2
    elif kind == 'bimodal':
        a = 0.5 * np.exp(-20*(x-0.3)**2) + 0.5 * np.exp(-20*(x-0.7)**2)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-method, multi-length strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            prev_score = best_score
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s, prev={prev_score:.8f})")
    
    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = make_init(2600, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Remez on parent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    print(f"\n=== Phase 1: Remez n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = remez_method(norm_sum1(best_seq), t1, solver=solver, verbose=True)
        try_update(a_out, f"Remez1_n{n0}")
    
    # === Phase 2: NMD mirror on best (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: NMD mirror n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t2, top_k=40, c0=3.5, verbose=True)
        try_update(a_out, f"NMD_n{len(best_seq)}")
    
    # === Phase 3: Remez again on best (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: Remez polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = remez_method(norm_sum1(best_seq), t3, solver=solver, verbose=True)
        try_update(a_out, f"Remez2_n{len(best_seq)}")
    
    # === Phase 4: NMD Euclidean (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: NMD Euclidean n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_euclidean(norm_sum1(best_seq), t4, c0=2.0, verbose=True)
        try_update(a_out, f"NE_n{len(best_seq)}")
    
    # === Phase 5: Try different lengths (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5_total = min(remaining * 0.15, 200.0)
    if t5_total > 60:
        print(f"\n=== Phase 5: Multi-length (budget={t5_total:.0f}s) ===")
        n_vals = [n0 - 200, n0 + 100, n0 + 300]
        n_vals = [nv for nv in n_vals if nv >= 1000]
        t_per_len = t5_total / max(1, len(n_vals))
        for nv in n_vals:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            a_alt = interpolate_sequence(best_seq, nv)
            t_r = min(t_per_len * 0.6, remaining - 15)
            t_n = min(t_per_len * 0.4, remaining - 15 - t_r)
            
            if t_r > 20:
                a_out, _ = remez_method(a_alt, t_r, solver=solver, verbose=False)
                try_update(a_out, f"Remez_n{nv}")
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), min(t_n, remaining - 5), top_k=30, c0=3.0, verbose=False)
                try_update(a_out, f"NMD_n{nv}")
    
    # === Phase 6: CD polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.05, 60.0)
    print(f"\n=== Phase 6: CD polish n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t6, verbose=False, rng_seed=42)
        try_update(a_out, f"CD_{len(best_seq)}")
    
    # === Phase 7: Final Remez + NMD on global best (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final push n={n_best} (budget={remaining:.0f}s) ===")
        t7a = min(remaining * 0.5, 150.0)
        if t7a > 20:
            a_out, _ = remez_method(norm_sum1(best_seq), t7a, solver=solver, verbose=True)
            try_update(a_out, f"Remez_final_{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7b = min(remaining * 0.5, 100.0)
            a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t7b, top_k=20, c0=2.0, verbose=False)
            try_update(a_out, f"NMD_final_{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = remez_method(norm_sum1(best_seq), remaining - 5, solver=solver, verbose=False)
            try_update(a_out, f"Remez_verylast_{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    # Ensure we return within time
    elapsed = time.monotonic() - t_global
    print(f"Total elapsed: {elapsed:.1f}s")
    
    return [float(x) for x in best_seq]
