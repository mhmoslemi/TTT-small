def soft_max_and_grad(a, t):
    """
    Compute soft max of autocorrelation and its gradient using FFT.
    f_t(a) = max_ac + t * log(sum exp((ac - max_ac)/t))
    grad[i] = 2 * sum_j w[i+j] * a[j] (cross-correlation of weights and a)
    """
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = ac.max()
    shifted = (ac - ac_max) / t
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    w = exp_shifted / w_sum
    f_val = ac_max + t * np.log(w_sum)

    # Gradient via FFT: grad[i] = 2 * (w cross-corr a)[i] = 2 * conv(a, w_rev)[2n-2-i]
    L = 1 << (3 * n - 1).bit_length()
    w_rev = w[::-1]
    a_pad = np.zeros(L)
    a_pad[:n] = a
    w_rev_pad = np.zeros(L)
    w_rev_pad[:2 * n - 1] = w_rev
    conv_full = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(w_rev_pad), L)
    grad = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
    return f_val, grad, ac


def nesterov_softmax(a0, time_limit, t_init=0.5, t_min=0.01, t_decay=0.9995,
                     verbose=False, seed=42):
    """
    Nesterov-accelerated gradient descent on the soft-max approximation.
    Uses adaptive step size and decreasing temperature.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    y = a.copy()
    x = a.copy()
    x_prev = a.copy()
    t = t_init
    step_size = 0.1
    lower_bound = 1.0 / (2 * n - 1) * 2 * n  # theoretical lower bound on score = 2n/(2n-1)
    f_lower = lower_bound / (2.0 * n)  # lower bound on max(ac) = 1/(2n-1)

    stall = 0
    restart_count = 0

    for it in range(50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if it > 1:
            beta = (it - 1.0) / (it + 1.0)
            y_trial = x + beta * (x - x_prev)
        else:
            y_trial = x.copy()
        y = project_simplex(y_trial)

        # Compute soft max and gradient
        f_val, grad, ac = soft_max_and_grad(y, t)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive step size: Polyak-like
        gap = f_val / (2.0 * n) - 1.0 / (2 * n - 1)
        if gap > 1e-10:
            alpha = min(gap / (grad_norm ** 2 + 1e-20), 2.0 / (it + 2.0))
        else:
            alpha = 0.01
        alpha = max(alpha, 1e-6)
        alpha = min(alpha, step_size * 1.1)

        # Backtracking line search
        x_new = y
        for ls in range(20):
            x_trial = y - alpha * grad
            x_trial = project_simplex(x_trial)
            f_trial, _, _ = soft_max_and_grad(x_trial, t)
            if f_trial < f_val - 1e-10 * alpha * grad_norm:
                x_new = x_trial
                break
            alpha *= 0.5
        else:
            x_new = y - alpha * grad
            x_new = project_simplex(x_new)

        step_size = max(alpha * 1.3, 0.01)
        step_size = min(step_size, 0.5)

        x_prev = x
        x = x_new

        # Track best using EXACT max
        J_new = compute_J(x)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart if stalled
        if stall > 3000:
            stall = 0
            restart_count += 1
            sigma = 0.02 * min(restart_count, 5)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = project_simplex(x_r)
            x = x_r
            x_prev = x_r.copy()
            y = x_r.copy()
            t = max(t * 2.0, t_init * 0.5)
            step_size = 0.05

        # Decrease temperature
        t = max(t * t_decay, t_min)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftNest it={it} ({elapsed:.0f}s): J={best_J:.6f}, t={t:.4f}, ss={step_size:.4f}")

    return best_a, best_J


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=500):
    """LP Frank-Wolfe with Nesterov acceleration for exact max."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()

    y = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix for top-K most binding lags
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()

        margins = M_curr - ac_y
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        # Build T_sel @ b <= M for selected lags
        T_sel_rows = []
        for k in binding_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            row = np.zeros(n)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            row[idx] = y[j_idx]
            T_sel_rows.append(row)
        T_sel = np.array(T_sel_rows)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        y = y_new
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, 2.0 * n * best_m


def coordinate_polish(a0, time_limit, verbose=False, seed=123):
    """Local coordinate descent / pairwise swap polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    iter_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:10]

        # Compute sensitivity at top peaks
        sensitivity = np.zeros(n)
        for k in top_lags[:5]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            sens_contrib = a[idx] * a[j_idx]
            sensitivity[idx] += sens_contrib
            sensitivity[j_idx] += sens_contrib

        improved = False
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            # Pick high-sensitivity source, low-sensitivity destination
            if np.any(sensitivity > 0):
                high = np.where(sensitivity > sensitivity.mean())[0]
                low = np.where(sensitivity < sensitivity.mean())[0]
            else:
                high = np.arange(n)
                low = np.arange(n)

            if len(high) == 0 or len(low) == 0:
                break

            src = high[rng.integers(len(high))]
            dst = low[rng.integers(len(low))]
            if src == dst or a[src] < 1e-10:
                continue

            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                # Random perturbation to escape
                n_pert = max(3, n // 200)
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * rng.uniform(0.01, 0.1)
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()

        iter_count += 1
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordPolish iter={iter_count} ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def make_cosh_init(n, width=0.3):
    """Initialize with cosh-like shape (high at center, low at edges)."""
    x = np.linspace(-1, 1, n)
    a = 1.0 / np.cosh(x / width)
    return norm_sum1(a)


def make_bimodal_init(n, sigma=0.15, offset=0.3):
    """Two Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - (0.5 - offset)) / sigma) ** 2) + np.exp(-((x - (0.5 + offset)) / sigma) ** 2)
    return norm_sum1(a)


def make_uniform_with_noise(n, seed=0):
    """Uniform with small random perturbation."""
    rng = np.random.default_rng(seed)
    a = 1.0 + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator: soft-max Nesterov + LP-FW + coordinate polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_cosh_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Soft-max Nesterov from parent ===
    print(f"\n=== Phase 1: Soft-max Nesterov (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_softmax(a_in, t1, t_init=0.3, t_min=0.005,
                                    t_decay=0.9998, verbose=True, seed=42)
        try_update(a_out, f"softmax_n{n0}")

    # === Phase 2: Soft-max Nesterov from diverse inits ===
    print(f"\n=== Phase 2: Soft-max Nesterov diverse inits (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2_total = min(remaining * 0.25, 400.0)
    if t2_total > 60:
        inits = [
            (make_cosh_init(n0, 0.2), "cosh0.2"),
            (make_bimodal_init(n0, 0.12, 0.25), "bimodal"),
            (make_uniform_with_noise(n0, seed=7), "uniform_noise"),
            (make_cosh_init(n0, 0.5), "cosh0.5"),
        ]
        t_per = max(15.0, t2_total / len(inits))
        for init, label in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 30)
            if t_avail < 10:
                break
            print(f"  SoftNest init '{label}' (t={t_avail:.0f}s)...")
            a_out, _ = nesterov_softmax(init, t_avail, t_init=0.5, t_min=0.01,
                                        t_decay=0.9997, verbose=False, seed=100)
            try_update(a_out, f"softmax_{label}")

    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True, top_K=400)
        try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 4: Soft-max Nesterov polish (lower temp) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Soft-max Nesterow polish (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_softmax(a_in, t4, t_init=0.1, t_min=0.003,
                                    t_decay=0.9999, verbose=True, seed=777)
        try_update(a_out, f"softmax_polish_n{n_best}")

    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate polish (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, seed=456)
        try_update(a_out, f"coordpolish_n{n_best}")

    # === Phase 6: LP-FW again ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP-FW round 2 (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 15 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True, top_K=500)
        try_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")
    for new_n in [n_best + 400, n_best - 400]:
        if new_n < 500 or new_n > 5000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t7a = min(remaining * 0.2, 120.0)
        if t7a > 15:
            a_out, _ = nesterov_softmax(a_long, t7a, t_init=0.3, t_min=0.01,
                                        t_decay=0.9998, verbose=False, seed=555)
            try_update(a_out, f"softmax_n{new_n}")

    # === Phase 8: Final LP if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 30 and n_best <= 3500:
        t8 = min(remaining * 0.3, 120.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=600)
            try_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final coord polish (n={n_best}) ===")
    if remaining > 15:
        t9 = min(remaining * 0.4, 100.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=False, seed=999)
            try_update(a_out, f"coordfinal_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_cosh_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
