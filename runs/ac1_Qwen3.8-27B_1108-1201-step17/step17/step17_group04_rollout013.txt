    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def subgradient_at_lag(a, k):
    """Compute subgradient of b[k] = sum_i a[i]*a[k-i] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k - idx
    grad[idx] = 2.0 * a[j_idx]
    return grad


def smooth_subgradient(a, ac, lam):
    """Compute smoothed subgradient using log-sum-exp."""
    n = len(a)
    # Compute softmax weights
    ac_clipped = ac - ac.max()  # numerical stability
    weights = np.exp(lam * ac_clipped)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        # Fallback to exact
        k = int(np.argmax(ac))
        return subgradient_at_lag(a, k)
    weights = weights / w_sum
    
    # Weighted sum of subgradients
    grad = np.zeros(n)
    # For efficiency, only include lags with significant weight
    threshold = 1e-6
    active = weights > threshold
    for k in np.where(active)[0]:
        grad += weights[k] * subgradient_at_lag(a, int(k))
    return grad


def mirror_descent_step(y, g, eta):
    """Exponential (mirror) descent step on the simplex."""
    log_a = np.log(np.maximum(y, 1e-30)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def lp_subgradient(a, M, top_K=200, solver=None):
    """Solve LP to get dual weights, return composite subgradient."""
    n = len(a)
    if solver is None:
        solver = get_solver()
    
    # Build constraint matrix for top-K lags
    ac = fast_autocorr(a)
    M_curr = ac.max()
    if M_curr < 1e-15:
        return np.zeros(n), None
    
    margins = M_curr - ac
    K_sel = min(top_K, 2 * n - 1)
    binding_idx = np.argsort(margins)[:K_sel]
    
    # Build T matrix (sparse would be better, but for n=3200 and K=200, dense is OK)
    T = np.zeros((K_sel, n))
    for col, k in enumerate(binding_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        T[col, idx] = a[j_idx]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M_curr]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None, None
    
    g_dir = b_val / b_sum
    return g_dir, b_val

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def subgradient_at_lag(a, k):
    """Compute subgradient of b[k] = (a*a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k - idx
    grad[idx] = 2.0 * a[j_idx]
    return grad


def smooth_subgradient(a, ac, lam, top_k=30):
    """Compute smoothed subgradient using log-sum-exp with top-k truncation."""
    n = len(a)
    ac_top = np.zeros(len(ac))
    top_idx = np.argsort(ac)[::-1][:top_k]
    ac_vals = ac[top_idx]
    ac_max = ac_vals.max()
    weights = np.exp(lam * (ac_vals - ac_max))
    w_sum = weights.sum()
    if w_sum < 1e-30:
        return subgradient_at_lag(a, int(np.argmax(ac)))
    weights = weights / w_sum
    grad = np.zeros(n)
    for wi, k in zip(weights, top_idx):
        if wi > 1e-8:
            grad += wi * subgradient_at_lag(a, int(k))
    return grad


def mirror_descent_step(y, g, eta):
    """Exponential (mirror) descent step on the simplex."""
    log_a = np.log(np.maximum(y, 1e-30)) - eta * g
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def lp_frank_wolfe(a0, time_limit, solver=None, verbose=False, top_K=300, seed=42):
    """
    LP Frank-Wolfe with Nesterov acceleration.
    Solves: max sum(b) s.t. conv(a, b)[k] <= M for top-K constraints, b >= 0
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()
    lp_fail_count = 0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Build constraint matrix for top-K lags
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T = np.zeros((K_sel, n))
        for col, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            T[col, idx] = y[j_idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            lp_fail_count += 1
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            lp_fail_count += 1
            if lp_fail_count > 20:
                # Fallback to subgradient for a while
                break
            continue

        lp_fail_count = 0
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def optimize_smoothed(a0, time_limit, solver=None, verbose=False, seed=42,
                       lam_start=1.0, lam_max=80.0, lam_growth=1.002,
                       lp_every=150, top_k_smooth=20, c_polyak=4.0):
    """
    Log-sum-exp smoothed mirror descent with adaptive lambda and LP subgradients.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    if solver is None:
        solver = get_solver()

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_every = 5000
    iter_count = 0
    lam = lam_start

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0

        # Nesterov extrapolation
        if iter_count > 1:
            beta_t = (iter_count - 1.0) / (iter_count + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()
        if M < 1e-15:
            break

        # Every lp_every iterations, try LP Frank-Wolfe step
        if iter_count % lp_every == 0:
            try:
                margins = M - ac
                K_sel = min(150, 2 * n - 1)
                binding_idx = np.argsort(margins)[:K_sel]
                T = np.zeros((K_sel, n))
                for col, k in enumerate(binding_idx):
                    k_int = int(k)
                    i_start = max(0, k_int - n + 1)
                    i_end = min(n, k_int + 1)
                    idx = np.arange(i_start, i_end)
                    j_idx = k_int - idx
                    T[col, idx] = y[j_idx]
                b_var = cp.Variable(n, nonneg=True)
                prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
                prob.solve(solver=solver, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                    b_sum = b_val.sum()
                    if b_sum > 1e-15:
                        g_dir = b_val / b_sum
                        alpha_fw = 2.0 / (iter_count / lp_every + 2.0)
                        x_new = (1.0 - alpha_fw) * y + alpha_fw * g_dir
                        x_new = np.maximum(x_new, 0.0)
                        sx = x_new.sum()
                        if sx > 1e-15:
                            x_new = x_new / sx
                            x_prev = x
                            x = x_new
                            J_new = compute_J(x)
                            if J_new < best_J - 1e-14:
                                best_J = J_new
                                best_a = x.copy()
                                stall = 0
                            else:
                                stall += 1
                            continue
            except Exception:
                pass

        # Smoothed subgradient step
        lam = min(lam * lam_growth, lam_max)
        g = smooth_subgradient(y, ac, lam, top_k=top_k_smooth)

        # Polyak step size
        lower_bound = 1.0 / (2 * n - 1)
        gap = max(M - lower_bound, 1e-8)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            eta = 1.0 / n
        else:
            eta = c_polyak * gap / g_norm_sq
            eta = min(eta, 3.0 / (iter_count + 1.0))
            eta = max(eta, 1e-10)

        x_new = mirror_descent_step(y, g, eta)
        x_prev = x
        x = x_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with adaptation
        if stall > restart_every:
            stall = 0
            sigma = 0.003 + 0.003 * min(iter_count / 50000.0, 1.0)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            lam = max(1.0, lam * 0.3)
            if verbose:
                print(f"    Restart ({elapsed:.0f}s): J={best_J:.6f}, lam={lam:.1f}")

        if verbose and iter_count % 50000 == 0:
            print(f"    Smooth t={iter_count} ({elapsed:.0f}s): J={best_J:.6f}, lam={lam:.1f}")

    return best_a, best_J


def peak_reduction(a0, time_limit, verbose=False, seed=77):
    """Coordinate moves: relocate mass from peak contributors to safe zones."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 2000:
        ac = fast_autocorr(a)
        top_lags = np.argsort(ac)[::-1][:10]

        # Compute sensitivity
        sensitivity = np.zeros(n)
        for k in top_lags[:6]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            sensitivity[idx] += contrib
            sensitivity[j_idx] += contrib

        high_sens = np.where((sensitivity > sensitivity.mean() + 0.5 * sensitivity.std()) & (a > 0.0005))[0]
        low_sens = np.where((sensitivity < sensitivity.mean() - 0.5 * sensitivity.std()) | (sensitivity == 0))[0]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            continue

        improved = False
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst or a[src] < 1e-8:
                continue
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.35]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                n_pert = max(10, n // 50)
                positions = rng.choice(n, size=n_pert, replace=False)
                for p in positions:
                    if a[p] > 1e-8:
                        amt = a[p] * rng.uniform(0.01, 0.08)
                        dst = int(rng.integers(n))
                        a[p] -= amt
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                if compute_J(a) < best_J:
                    best_J = compute_J(a)
                    best_a = a.copy()
                    stall = 0

    return best_a, best_J


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_symmetric_init(n):
    """Symmetric init: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.05
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_power_law_init(n, power=0.5):
    """Power law from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.01
    return norm_sum1(a)


def make_fourier_init(n, K=15, seed=0):
    """Fourier parametric: |sum c_k exp(2πik i/n)|^2 with random coefficients."""
    rng = np.random.default_rng(seed)
    coeffs = rng.normal(0, 1, (K + 1, 2))  # real and imaginary parts
    # Ensure DC component is large enough for non-negativity
    coeffs[0, 0] = 3.0 + rng.uniform(0, 2)
    coeffs[0, 1] = 0.0
    i = np.arange(n)
    signal = np.zeros(n, dtype=np.complex128)
    for k in range(K + 1):
        if k == 0:
            signal += coeffs[0, 0]
        else:
            signal += coeffs[k, 0] * np.exp(2j * np.pi * k * i / n)
            signal += coeffs[k, 1] * np.exp(-2j * np.pi * k * i / n)
    a = np.abs(signal) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_gaussian_center(n, sigma=0.3):
    """Gaussian centered in middle."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) / sigma) ** 2) + 0.01
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Multi-length exploration with smoothed optimization ===
    print(f"\n=== Phase 1: Multi-length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.20, 300.0)

    if t_phase1 > 60:
        candidate_lengths = [n0]
        # Add neighbors
        for delta in [200, 400, 600]:
            candidate_lengths.append(n0 + delta)
            candidate_lengths.append(n0 - delta)
        candidate_lengths = [l for l in candidate_lengths if 1500 <= l <= 5000]
        # Remove duplicates
        seen = set()
        unique_lengths = []
        for l in candidate_lengths:
            if l not in seen:
                seen.add(l)
                unique_lengths.append(l)
        candidate_lengths = unique_lengths[:5]

        n_trials = len(candidate_lengths)
        t_per = max(30.0, t_phase1 / n_trials)

        for new_n in candidate_lengths:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 30)
            if t_avail < 20:
                break
            print(f"  Trying n={new_n} (t={t_avail:.0f}s)")
            a_long = interpolate_sequence(best_seq, new_n)
            a_out, _ = optimize_smoothed(a_long, t_avail, solver=solver, verbose=False,
                                         seed=42 + new_n % 100)
            try_update(a_out, f"smooth_n{new_n}")

    # === Phase 2: LP Frank-Wolfe on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP Frank-Wolfe (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver=solver, verbose=True, top_K=400, seed=42)
        try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 3: Smoothed optimization on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smoothed optimization (n={n_best}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = optimize_smoothed(a_in, t3, solver=solver, verbose=True,
                                     seed=99, lam_start=5.0, lam_max=100.0,
                                     lam_growth=1.003, lp_every=100)
        try_update(a_out, f"smooth2_n{n_best}")

    # === Phase 4: Peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak reduction (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_reduction(a_in, t4, verbose=False, seed=77)
        try_update(a_out, f"peakred_n{n_best}")

    # === Phase 5: LP Frank-Wolfe after peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW round 2 (n={n_best}) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver=solver, verbose=True, top_K=500, seed=88)
        try_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 6: Alternative initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Alternative inits (n={n_best}) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 40:
        inits = [
            (make_fourier_init(n_best, K=20, seed=1), "fourier1"),
            (make_fourier_init(n_best, K=30, seed=42), "fourier2"),
            (make_power_law_init(n_best, power=0.8), "pow0.8"),
            (make_gaussian_center(n_best, sigma=0.25), "gauss_center"),
        ]
        t_per = t6 / len(inits)
        for init, lbl in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t_avail = min(t_per, remaining - 20)
            if t_avail < 10:
                break
            # Blend with current best for warm start
            blend = 0.5 * norm_sum1(best_seq) + 0.5 * init
            a_out, _ = optimize_smoothed(blend, t_avail, solver=solver, verbose=False,
                                         seed=200 + hash(lbl) % 100)
            try_update(a_out, f"alt_{lbl}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    t7 = min(remaining * 0.30, 300.0)
    if t7 > 30:
        # LP-FW for half the time
        t7a = t7 * 0.5
        if t7a > 20 and n_best <= 5000:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7a, solver=solver, verbose=True, top_K=500, seed=555)
            try_update(a_out, f"lpfw_final_n{n_best}")

        # Smoothed for the other half
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.4, 150.0)
        if t7b > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = optimize_smoothed(a_in, t7b, solver=solver, verbose=True,
                                         seed=777, lam_start=20.0, lam_max=150.0,
                                         lam_growth=1.005, lp_every=50)
            try_update(a_out, f"smooth_final_n{n_best}")

    # === Phase 8: Final peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: Final peak reduction (n={n_best}) ===")
        t8 = min(remaining * 0.6, 150.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_reduction(a_in, t8, verbose=False, seed=1234)
            try_update(a_out, f"peakred_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
