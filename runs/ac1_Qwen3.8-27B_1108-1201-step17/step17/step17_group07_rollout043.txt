def fast_conv(a):

    s = a.sum()
    if s < 1e-15:

    Using multiplicative weights for p and mirror descent for a.

    c = fast_conv(a)

    else:

    best_m = c.max()

    x_prev = a.copy()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute exact gradient using Hankel structure
        g = compute_hankel_gradient(y, p, grad_fft_size)
        
        # Mirror descent update for a
        eta_t = eta_a / (np.sqrt(t) + 1.0)

        a_new = np.maximum(a_new, 1e-30)

        M = c_new.max()

        c_scale = c_new.max()
        if c_scale > 1e-15:
            p_weights = np.exp(eta_c * c_new / c_scale)
            p = p * p_weights
            p /= p.sum()
        
        # Update Nesterov variables
        x_prev = x
        x = a_new
        
        # Track best
        if M < best_m - 1e-18:
            best_m = M
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Decay step sizes
        eta_a *= a_decay
        eta_c *= c_decay
        
        # Restart if stuck
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(1000, int(max_stall * 0.5))
            strategy = int(rng.integers(3))
            if strategy == 0:

            elif strategy == 1:

                x_r = best_a.copy()
                x_r[idx] *= rng.uniform(0.5, 1.5, len(idx))
            else:

            x_r = x_r / x_r.sum()

def lp_remez_single(a, max_cons=40, r=0.1, solver=None):
    """Single LP Remez step."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]

        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    constraints.append(a_var <= a * (1.0 + r))
    constraints.append(a_var >= a * max(0.0, 1.0 - r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_lp_method(a0, time_limit, solver=None, r_init=0.08, verbose=False):
    """LP-based Remez with adaptive trust region."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(456)
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand = lp_remez_single(a, max_cons=40, r=r, solver=solver)
        
        accepted = False
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.3, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.005, r * 0.8)
            if no_improve > 30:
                idx = rng.choice(n, size=min(n//5, 300), replace=False)
                a[idx] *= rng.uniform(0.8, 1.2, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with random pair moves."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(20, n // 100)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 5)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.1)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Minimax mirror descent with EXACT gradient via Hankel matrix structure (O(n log n))
2. LP-based Remez method for local polishing
3. Coordinate descent for final refinement
4. Multi-length and multi-init exploration

Key insight: The gradient of sum_k p[k]*c[k](a) w.r.t. a[i] is sum_j p[i+j]*a[j],
which exploits the Hankel structure Q[i,j]=p[i+j] and can be computed via a single FFT,
giving the EXACT subgradient of the smoothed objective at O(n log n) cost.
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_hankel_gradient(a, p, fft_size):
    """Compute g[i] = sum_j p[i+j] * a[j] for i=0..n-1 using FFT.
    Exploits Hankel structure Q[i,j] = p[i+j]."""
    n = len(a)
    ar = a[::-1].copy()
    P = np.zeros(fft_size)
    P[:2 * n - 1] = p
    A = np.zeros(fft_size)
    A[:n] = ar
    C = np.fft.irfft(np.fft.rfft(P) * np.fft.rfft(A), fft_size)
    return C[n - 1:2 * n - 1].copy()

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def minimax_mirror_descent(a0, time_limit, verbose=False):
    """Minimax mirror descent with exact Hankel gradient.
    Solves min_a max_k c[k] via alternating:
    - Mirror descent on a (exact gradient from current p)
    - Multiplicative weights update on p (concentrate on peak lags)
    """
    n = len(a0)
    a = norm_sum1(a0)

    grad_fft_size = 1 << (3 * n - 1).bit_length()
    conv_fft_size = 1 << (2 * n - 1).bit_length()

    # Initialize p from current autocorrelation with moderate temperature
    c = fast_conv(a)
    c_max = c.max()
    if c_max < 1e-15:
        p = np.ones(2 * n - 1) / (2 * n - 1)
        log_p = np.full(2 * n - 1, -np.log(2 * n - 1))
    else:
        log_p = 2.0 * c / c_max - np.log(2 * n - 1)
        log_p -= log_p.max()
        p = np.exp(log_p)
        p /= p.sum()

    best_a = a.copy()
    best_m = c.max()
    t0 = time.monotonic()

    eta_a = 1.5
    eta_c = 0.3
    a_decay = 0.9997
    c_decay = 0.99995

    x = a.copy()
    x_prev = a.copy()

    stall_count = 0
    max_stall = 8000
    rng = np.random.default_rng(42 + n)

    # Pre-allocate
    log_a = np.zeros(n)
    a_new = np.zeros(n)
    y = np.zeros(n)

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta = (t - 1.0) / (t + 1.5)
        np.add(x, beta * (x - x_prev), out=y)
        np.maximum(y, 1e-15, out=y)
        sy = y.sum()
        y /= sy

        # Compute exact gradient using Hankel structure
        g = compute_hankel_gradient(y, p, grad_fft_size)

        # Mirror descent update
        eta_t = eta_a / (np.sqrt(t) + 1.0)
        np.log(y, out=log_a)
        log_a -= eta_t * 2.0 * g
        max_log = log_a.max()
        np.exp(log_a - max_log, out=a_new)
        np.maximum(a_new, 1e-30, out=a_new)
        a_new /= a_new.sum()

        # Compute new autocorrelation
        c_new = fast_conv(a_new)
        M = c_new.max()

        # Update p using log-space multiplicative weights
        c_scale = c_new.max()
        if c_scale > 1e-15:
            log_p += eta_c * c_new / c_scale
            # Refold to prevent overflow
            log_p -= log_p.max()
            p = np.exp(log_p)
            p /= p.sum()

        # Update Nesterov variables
        x_prev = x
        x = a_new

        # Track best
        if M < best_m - 1e-18:
            best_m = M
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        eta_a *= a_decay
        eta_c *= c_decay

        # Restart if stuck
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(1000, int(max_stall * 0.5))
            strategy = int(rng.integers(4))
            if strategy == 0:
                x_r = best_a * (1 + rng.normal(0, 0.01, n))
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=max(n // 10, 50), replace=False)
                x_r[idx] *= rng.uniform(0.5, 1.5, len(idx))
            elif strategy == 2:
                # Reset p and small perturbation
                log_p = np.zeros(2 * n - 1) - np.log(2 * n - 1)
                p = np.ones(2 * n - 1) / (2 * n - 1)
                x_r = best_a * (1 + rng.normal(0, 0.005, n))
            else:
                # Warm restart: blend with a new random-ish init
                x_r = 0.7 * best_a + 0.3 * make_init_global(n, int(rng.integers(5)))
            x_r = np.maximum(x_r, 1e-10)
            x_r /= x_r.sum()
            x = x_r
            x_prev = x_r
            eta_a = max(0.2, eta_a * 3.0)
            eta_c = max(0.05, eta_c * 2.0)

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MMD t={t} ({elapsed:.1f}s): J={J:.8f}, eta_a={eta_a:.4f}, eta_c={eta_c:.4f}, stall={stall_count}")

    return best_a, best_m

def make_init_global(n, kind_idx):
    """Generate initial sequence of given length and type."""
    x = np.linspace(0.0, 1.0, n)
    if kind_idx == 0:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind_idx == 1:
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind_idx == 2:
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.1)
    elif kind_idx == 3:
        a = np.ones(n)
    elif kind_idx == 4:
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def lp_remez_single(a, max_cons=50, r=0.1, solver=None):
    """Single LP Remez step with linearized constraints."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()

    # Select active constraints: lags where b is close to max
    threshold = max_b * 0.998
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1]

    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])

    # Trust region
    constraints.append(a_var <= a * (1.0 + r))
    min_bound = np.maximum(a * (1.0 - r), 0.0)
    constraints.append(a_var >= min_bound)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_lp_method(a0, time_limit, solver=None, r_init=0.08, max_cons=50, verbose=False):
    """LP-based Remez with adaptive trust region and line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(789)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        a_cand = lp_remez_single(a, max_cons=max_cons, r=r, solver=solver)

        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.4, r * 1.08)
                break

        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.75)
            if no_improve > 40:
                idx = rng.choice(n, size=min(n // 5, 400), replace=False)
                a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                a = np.maximum(a, 0.0)
                a /= a.sum()
                no_improve = 0
                r = max(r, 0.05)

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")

    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with random pair moves for fine polish."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 80)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(300, n // 4)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue

            best_frac = None
            best_m_trial = best_m
            for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                if a[i] < -1e-15:
                    a[i] += amt
                    a[j] -= amt
                    continue
                a_safe = np.maximum(a, 0.0)
                a_safe /= a_safe.sum()
                m_trial = fast_conv(a_safe).max()
                if m_trial < best_m_trial - 1e-16:
                    best_m_trial = m_trial
                    best_frac = frac
                a[i] += amt
                a[j] -= amt

            if best_frac is not None:
                amt = a[i] * best_frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a /= a.sum()
                best_m = best_m_trial
                best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), remaining - 5, verbose=False, rng_seed=42)
            try_update(a_out, f"CD_final_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
