Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel combination of:

2. Softmax-weighted multi-peak subgradient with adaptive step size
3. Nesterov momentum acceleration
4. Multi-strategy initialization including "soft-binary" patterns
5. Multi-length search with shape transfer
6. LP-based Frank-Wolfe polish on best candidate

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except:
        return float('inf')

def spectral_flatten(a, alpha, fft_size):
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag[1:] = alpha * mag[1:].mean() + (1-alpha) * mag[1:]
    # Keep DC
    A_new = mag * np.exp(1j * phase)
    return np.fft.irfft(A_new, fft_size)[:len(a)]

def subgrad(a, ac, top_k, beta, fft_size):
    n = len(a)
    k_max = min(top_k, len(ac))
    # Get top peaks (use argpartition for efficiency)
    if k_max < len(ac):
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
    else:
        top_idx = np.arange(len(ac))
    top_vals = ac[top_idx]
    # Softmax weights
    log_w = beta * (top_vals - top_vals.max())
    w = np.exp(log_w)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # Subgradient of conv(a,a)[k] w.r.t. a:
        # For position j: a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        # Simplified: use convolution with delta
        lo1 = max(0, k_int - n + 1)
        hi1 = min(n, k_int + 1)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += wi * a[k_int - idx]
        
        lo2 = max(0, k_int)
        hi2 = min(n, k_int + n)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] += wi * a[idx - k_int]
    
    return grad

    """Main optimization loop."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    x_prev = a.copy()

    alpha = 0.1
    top_k = 50
    beta = 4.0
    stall = 0

        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        mu = (t - 1.0) / (t + 1.5)
        y = x + mu * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy > 1e-15:

            a_flat = spectral_flatten(y, alpha_curr, fft_size)

            m_flat = fast_autocorr(a_flat).max()
            if m_flat < fast_autocorr(y).max():

        g = subgrad(y, ac, top_k, beta, fft_size)

        eta_t = eta / (1.0 + 0.1 * eta * t / n)

        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sn
        
        x_prev = x
        x = a_new
        
        # Track best

        if m_new < best_m * (1 - 1e-10):
            best_m = m_new
            best_a = x.copy()
            stall = 0
        else:

            a_pert = best_a * (1.0 + strength * rng.standard_normal(n))

            stall = 0

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral magnitude flattening (Gerchberg-Saxton variant) as primary driver
2. Softmax-weighted multi-peak subgradient with Nesterov momentum
3. Adaptive step-size with periodic spectral re-equalization
4. Multi-strategy initialization: soft-binary patterns, ramps, parent transfer
5. Multi-length search with shape interpolation
6. LP-based Frank-Wolfe polish on best candidate

import numpy as np
import time

_lp_solver = None

def _get_lp_solver():

        return _lp_solver
    try:

            except Exception:

    except Exception:

def fast_autocorr(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def spectral_flatten_step(a, alpha, fft_size):
    """
    Flatten the magnitude spectrum while preserving phase.
    alpha in [0,1]: 0=no change, 1=perfectly flat AC spectrum.
    """
    n = len(a)
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    if len(mag) > 1:
        ac_mean = mag[1:].mean()
        mag[1:] = alpha * ac_mean + (1.0 - alpha) * mag[1:]
    A_new = mag * np.exp(1j * phase)
    return np.fft.irfft(A_new, fft_size)[:n]


def compute_subgrad(a, ac, top_k, beta):
    """
    Softmax-weighted subgradient from top autocorrelation peaks.
    """
    n = len(a)
    k_max = min(top_k, len(ac))
    if k_max < len(ac):
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
    else:
        top_idx = np.arange(len(ac))
    top_vals = np.maximum(ac[top_idx], 1e-30)
    log_w = beta * (top_vals - top_vals.max())
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # Forward: grad[j] += wi * a[k-j] for valid j
        lo = max(0, k_int - n + 1)
        hi = min(n, k_int + 1)
        if lo < hi:
            idx = np.arange(lo, hi)
            grad[lo:hi] += wi * a[k_int - idx]
        # Backward: grad[j] += wi * a[j-k] for valid j
        lo2 = max(0, k_int)
        hi2 = min(n, k_int + n)
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[lo2:hi2] += wi * a[idx2 - k_int]
    return grad


def optimize_core(a0, time_limit, fft_size, verbose=False):
    """
    Main optimization: alternating spectral flattening + subgradient with Nesterov.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    eta = 0.3 / n
    top_k = 50
    beta_w = 4.0
    stall = 0
    restart_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        mu = (t - 1.0) / (t + 1.5)
        y = x + mu * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Spectral flattening every 3 steps
        if t % 3 == 0:
            progress = min(1.0, t / 8000.0)
            alpha = 0.05 + 0.55 * progress
            a_flat = spectral_flatten_step(y, alpha, fft_size)
            a_flat = norm_sum1(a_flat)
            m_flat = fast_autocorr(a_flat).max()
            m_y = fast_autocorr(y).max()
            if m_flat < m_y:
                y = a_flat

        # Subgradient step
        ac = fast_autocorr(y)
        g = compute_subgrad(y, ac, top_k, beta_w)

        # Adaptive step size
        eta_t = eta / (1.0 + 0.05 * t / n)
        a_new = y - eta_t * g
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sn

        x_prev = x
        x = a_new

        # Track best
        m_new = fast_autocorr(x).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Periodic restart with increasing perturbation
        if stall > 4000:
            restart_count += 1
            strength = min(0.15, 0.03 + 0.02 * restart_count)
            a_pert = best_a + strength * rng.standard_normal(n) * np.maximum(best_a, 1e-10)
            x = norm_sum1(a_pert)
            x_prev = x.copy()
            stall = 0
            top_k = min(top_k + 15, 200)
            beta_w = max(beta_w - 0.4, 1.5)
            eta = eta * 1.1

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Core t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}, restarts={restart_count}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, top_K=200, verbose=False):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    solver = _get_lp_solver()
    if solver is None:
        return a0, fast_autocorr(a0).max()

    import cvxpy as cp

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x_mom = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select most binding constraints
        K_sel = min(top_K, 2 * n - 1)
        margins = M_curr - ac
        binding_idx = np.argsort(margins)[:K_sel]

        # Build constraint matrix (sparse) and solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        # Add constraints: T_sel @ b_var <= M_sel
        # T_sel[binding_idx, j] = y[j] if (binding_idx - j) is in [0, n)
        # For efficiency, build as dense (n up to ~5000, K up to 200)
        T_sel = np.zeros((K_sel, n))
        for row, k_val in enumerate(binding_idx):
            k_int = int(k_val)
            lo = max(0, k_int - n + 1)
            hi = min(n, k_int + 1)
            if lo < hi:
                T_sel[row, lo:hi] = y[k_int - np.arange(lo, hi)]
            lo2 = max(0, k_int)
            hi2 = min(n, k_int + n)
            if lo2 < hi2:
                T_sel[row, lo2:hi2] += y[np.arange(lo2, hi2) - k_int]

        M_sel = M_curr * np.ones(K_sel)
        constraints.append(T_sel @ b_var <= M_sel)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov update
        alpha_fw = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_fw) * y + alpha_fw * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Momentum
        beta_t = (t - 1.0) / (t + 2.0)
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx

        prev_y = y
        y = y_new
        x_mom = x_new

        # Track best
        for candidate in [y, x_mom]:
            mc = fast_autocorr(candidate).max()
            if mc < best_m:
                best_m = mc
                best_a = candidate.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def make_soft_binary(n, seed=0, density=0.5):
    """Random soft-binary pattern: ~density fraction at 2/n, rest near 0, then smoothed."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    n_ones = int(n * density)
    positions = rng.choice(n, size=n_ones, replace=False)
    a[positions] = 1.0
    # Smooth with a small kernel
    kernel_size = max(3, n // 100)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    return norm_sum1(a)


def make_ramp(n, direction='up'):

def make_sinusoidal(n, freq=1.0, phase=0.0):

def make_cosine(n, mode=1):

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
