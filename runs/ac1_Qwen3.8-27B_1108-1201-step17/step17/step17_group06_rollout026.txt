"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Continuation method on smooth max (log-sum-exp with increasing β)
2. L-BFGS-B optimization with FFT-based gradients
3. Multi-strategy initialization
4. Length sweeping
5. Final polish with coordinate descent
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_obj(a, beta):
    """f(a) = 2n * logsumexp(beta * conv(a,a)) / (sum a)^2"""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return 1e10
    conv = fast_conv(a)
    bc = beta * conv
    m = bc.max()
    LSE = m + np.log(np.exp(bc - m).sum())
    return 2.0 * n * LSE / (S * S)


def smooth_grad(a, beta):

    w = exp_bc / exp_bc_sum  # softmax

def obj_and_grad(a, beta):
    """Object function returning (value, gradient) for scipy."""
    val = smooth_obj(a, beta)
    grad = smooth_grad(a, beta)
    # Ensure gradient is finite
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(a)
    return val, grad
