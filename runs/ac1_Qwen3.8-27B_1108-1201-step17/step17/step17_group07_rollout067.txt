def fast_conv(a):
    """Compute full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_max_subgrad(a, beta, weight_threshold=1e-10):
    """
    Compute the gradient of logsumexp(beta * conv(a,a)) / beta.
    This smoothly approximates the subgradient of max(conv(a,a)).
    Returns (gradient, max_value, num_active_lags).

        weights = np.zeros_like(b)
        weights[int(np.argmax(b))] = 1.0
    else:
        weights = exp_shifted / total
    
    active = np.where(weights > weight_threshold)[0]
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)

            idx = np.arange(j_start, j_end + 1)
            g[idx] += weights[k] * a[k_int - idx]
    
    return g, M, len(active)

def downsample(a, factor):
    """Downsample by averaging blocks."""
    n = len(a)
    new_n = n // factor
    if new_n < 10:
        return a.copy()
    a_padded = np.zeros(new_n * factor)
    a_padded[:n] = a
    blocks = a_padded.reshape(new_n, factor)
    result = blocks.sum(axis=1)
    s = result.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return result / s

def upsample(a, new_n):
    """Upsample using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    ss = result.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return result / ss

def make_init(n, kind='triangle'):

        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0

        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'bump':
        a = np.exp(-20.0 * (x - 0.5) ** 2)
    elif kind == 'two_bump':
        a = np.exp(-40.0 * (x - 0.3) ** 2) + np.exp(-40.0 * (x - 0.7) ** 2)

        a = x ** 2 + (1.0 - x) ** 2 + 0.1
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

                                                 beta_start=20.0, beta_end=200.0,
                                                 eta_start=0.8, eta_end=0.03, verbose=False, seed=20+i)
            try_update(a_out, f"SMD_{kind}_n{n0}")
    
    # === Phase 6: CD polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 6: CD polish (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = coordinate_descent(norm_sum1(best_seq), t6, verbose=False, rng_seed=555)
        try_update(a_out, f"CD_n{len(best_seq)}")
    
    # === Phase 7: Final LP + NMD polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 7: Final LP+MD (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = lp_remez_polish(norm_sum1(best_seq), t7 * 0.5, solver=solver,
                                    max_cons=60, verbose=False, seed=7)
        try_update(a_out, f"LP_final_n{len(best_seq)}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = smooth_max_mirror_descent(norm_sum1(best_seq), remaining - 3,
                                                 beta_start=100.0, beta_end=500.0,
                                                 eta_start=0.2, eta_end=0.01, verbose=False, seed=8)
            try_update(a_out, f"SMD_final_n{len(best_seq)}")
    
    # === Phase 8: Alternate length exploration (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 8: Alt length (budget={remaining:.0f}s) ===")
        n_alt = n0 + 200
        a_alt = upsample(best_seq, n_alt)
        
        t8a = min(remaining * 0.5, 200.0)
        if t8a > 30:
            a_out, _ = smooth_max_mirror_descent(a_alt, t8a,
                                                 beta_start=30.0, beta_end=300.0,
                                                 eta_start=0.5, eta_end=0.02, verbose=True, seed=30)
            try_update(a_out, f"SMD_n{n_alt}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out, _ = lp_remez_polish(norm_sum1(best_seq), remaining - 5, solver=solver,
                                        max_cons=70, verbose=False, seed=9)
            try_update(a_out, f"LP_last_n{len(best_seq)}")
    
    # === Phase 9: Final SMD + LP on global best with remaining time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final push n={n_best} (budget={remaining:.0f}s) ===")
        t9a = min(remaining * 0.5, 100.0)
        if t9a > 10:
            a_out, _ = smooth_max_mirror_descent(norm_sum1(best_seq), t9a,
                                                 beta_start=80.0, beta_end=500.0,
                                                 eta_start=0.15, eta_end=0.005, verbose=False, seed=100)
            try_update(a_out, f"SMD_push_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = lp_remez_polish(norm_sum1(best_seq), remaining - 3, solver=solver,
                                        max_cons=50, verbose=False, seed=101)
            try_update(a_out, f"LP_push_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
