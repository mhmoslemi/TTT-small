def fast_autocorr(a):
    """Compute np.convolve(a, a) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_t_matrix(y, top_idx, n):
    """Build constraint matrix T where T[row, j] = y[m-j] if valid."""
    K = len(top_idx)
    T = np.zeros((K, n), dtype=np.float64)

        m = int(top_idx[row])
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            T[row, js] = y[m - js]
    return T


def solve_fw_lp(T, M, n):
    """Solve LP: maximize sum(b) s.t. T@b <= M, b >= 0. Returns normalized direction."""
    K = T.shape[0]
    if M < 1e-30:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    try:
        res = so.linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            bs = b.sum()
            if bs > 1e-15:
                return b / bs
    except Exception:
        pass
    return np.ones(n) / n

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss
