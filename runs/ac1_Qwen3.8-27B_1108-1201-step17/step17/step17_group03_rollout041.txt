def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   margin_frac=1e-3, use_acceleration=True):
    """
    LP Frank-Wolfe optimization for minimizing max(conv(a,a)) with sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        threshold = margin_frac * M_curr
        n_active = np.sum(margins <= threshold)

        if n_active > top_K * 4:
            active_idx = np.argsort(margins)[:top_K * 4]
        elif n_active < 5:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(margins <= threshold)[0]

        K = len(active_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1 or not use_acceleration:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            margin_frac = max(margin_frac * 0.85, 1e-5)

        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")

    return best_a, best_m


def peak_crush(a0, time_limit, verbose=True, rng_seed=42):

    """

        if total_contrib < 1e-20:
            stall += 1
            continue

                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt

                a = a / a.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def gradient_smooth_conv(a, T):
    """
    Compute gradient of 2n * T * log(sum_k exp(c_k/T)) where c = conv(a,a).
    Returns gradient with respect to a (unconstrained).
    Uses FFT for the cross-correlation computation.
    """
    n = len(a)
    c = fast_conv(a)
    max_c = np.max(c)
    log_w = (c - max_c) / T
    log_w -= np.max(log_w)
    p = np.exp(log_w)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(c)) / len(c)
    else:
        p = p / p_sum

    # g[i] = sum_{j=0}^{n-1} p[i+j] * a[j] for i = 0,...,n-1
    # Using: g[i] = conv(p_rev, a)[2n-2-i] where p_rev = p[::-1]
    p_rev = p[::-1]
    fft_size = 1 << (len(p_rev) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_result = np.fft.irfft(P * A, fft_size)
    g = conv_result[2 * n - 2 : n - 1 : -1]

    return 4.0 * n * g

            best_a = a.copy()
            steps_without_improve = 0
        else:
            steps_without_improve += 1

        if steps_without_improve > 3000 and steps_without_improve % 1000 == 0:
            pert = rng.normal(0, 0.01, n)
            x = x + pert
            steps_without_improve = 0

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Smooth t={step} ({elapsed:.0f}s): obj={obj:.6f}, T={T:.4f}, lr={lr:.6f}")

    return best_a, best_obj


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   margin_frac=1e-3, use_acceleration=True):
    """
    LP Frank-Wolfe for minimizing max(conv(a,a)) with sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        threshold = margin_frac * M_curr
        n_active = np.sum(margins <= threshold)

        if n_active > top_K * 4:
            active_idx = np.argsort(margins)[:top_K * 4]
        elif n_active < 5:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(margins <= threshold)[0]

        K = len(active_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1 or not use_acceleration:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            margin_frac = max(margin_frac * 0.85, 1e-5)

        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")

    return best_a, best_m


def peak_crush(a0, time_limit, verbose=True, rng_seed=42):
    """
    Coordinate descent: move mass from peak-contributing positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 1
            continue

        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                n_trials = 40
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update(a, best_seq, best_score, t_global, label=""):
    """Try to update best sequence if new candidate is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def make_parametric(n, family='sine'):
    """Create a parametric initial sequence."""
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return (np.sin(np.pi * x) ** 2).astype(np.float64)
    elif family == 'beta':
        x = np.linspace(0.01, 0.99, n)
        return (x ** 2 * (1 - x) ** 2).astype(np.float64)
    elif family == 'uniform':
        return np.ones(n, dtype=np.float64)
    else:
        x = np.linspace(0, 1, n)
        return (np.sin(np.pi * x) ** 2).astype(np.float64)
