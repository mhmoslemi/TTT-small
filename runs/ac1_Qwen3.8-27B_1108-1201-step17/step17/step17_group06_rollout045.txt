    n = len(a)

    conv_shifted = conv - conv.max()
    lse = np.log(np.sum(np.exp(mu * conv_shifted))) / mu + conv.max()

    n = len(a)

    # But let's use all weighted lags

            j_idx = k - idx

def fast_conv(a):
