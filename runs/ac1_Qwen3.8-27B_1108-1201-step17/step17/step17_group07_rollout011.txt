```
def project_to_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

This is O(n log n) per projection.

    """FFT-based full convolution of a with itself."""

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, b, top_k=30, soft_tau=None):
    """Compute subgradient of max(b) using top-k lags with optional soft weighting."""
    n = len(a)
    b = np.asarray(b, dtype=np.float64)
    M = b.max()
    
    if top_k >= len(b):
        top_idx = np.arange(len(b))
    else:
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    
    # Soft weighting
    if soft_tau is not None and soft_tau > 0:
        weights = np.exp((b[top_idx] - M) / soft_tau)
        weights = weights / weights.sum()
    else:
        weights = np.ones(len(top_idx)) / len(top_idx)
    
    g = np.zeros(n)
    for i, k in enumerate(top_idx):
        k = int(k)
        w = weights[i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += w * 2.0 * a[k - idx]
    
    return g

```python
def lp_remez_single(a, max_cons=100, r=0.1, solver=None):
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    scaled = b / max_b
    top_k = min(max_cons * 3, len(b))
    top_idx = np.argpartition(scaled, -top_k)[-top_k:]
    weights = np.exp(scaled[top_idx] - 1.0)
    weights = weights / weights.sum()
    thresholds = scaled[top_idx][::-1][:max_cons]
    active = top_idx[np.argsort(thresholds)[::-1]][:max_cons]
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    constraints.append(a_var >= a * 0.01)
    constraints.append(a_var <= a * (1.0 + r))
    constraints.append(M_var >= 0)
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
```
