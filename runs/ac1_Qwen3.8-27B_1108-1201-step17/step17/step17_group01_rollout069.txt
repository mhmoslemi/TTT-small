def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgrad(a, conv, top_k=30, temperature=0.003):
    n = len(a)
    conv_max = conv.max()
    threshold = conv_max - temperature * 8.0
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > top_k:
        order = np.argsort(conv[active])[::-1][:top_k]
        active = active[order]
    w = np.exp((conv[active] - conv_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

def compute_subgrad(a, conv, top_k=30, temperature=0.003):
    """Subgradient of max_k conv(k) via softmax-weighted active peaks."""
    n = len(a)
    conv_max = conv.max()
    threshold = conv_max - temperature * 8.0
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    if len(active) > top_k:
        order = np.argsort(conv[active])[::-1][:top_k]
        active = active[order]
    w = np.exp((conv[active] - conv_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad


def amd_optimize(a0, time_limit, top_k_init=25, temp_init=0.003, c0_init=3.0,
                 restart_every=5000, seed=42, verbose=False):
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_m = fast_conv(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    top_k = top_k_init
    temperature = temp_init
    c0 = c0_init
    stall_count = 0
    restart_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv_y = fast_conv(y)
        g = compute_subgrad(y, conv_y, top_k=top_k, temperature=temperature)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M = fast_conv(x).max()
        if M < best_m - 1e-15:
            best_m = M
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.06, 0.015 + 0.005 * min(restart_count, 8))
            pert = rng.normal(0, strength, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            top_k = min(top_k + 5, 80)
            temperature = max(temperature - 0.0004, 0.0008)
            c0 = 3.0 * (1.0 + 0.04 * min(restart_count, 10))
            restart_every = max(3000, restart_every + 500)
            if verbose:
                J = 2.0 * n * best_m
                print(f"    AMD restart {restart_count} t={t}: J={J:.6f}, tk={top_k}, temp={temperature:.4f}")

    return best_x, best_m


def coord_descent_polish(a0, time_limit, seed=42, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 250:
        conv = fast_conv(a)
        peak_k = int(np.argmax(conv))
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[peak_k - idx]

        n_top_src = max(5, min(20, n // 200))
        top_src = idx[np.argsort(contrib)[::-1][:n_top_src]]

        n_top_peaks = min(15, 2 * n - 1)
        top_peaks = np.argsort(conv)[::-1][:n_top_peaks]
        costs = np.zeros(n)
        for k2 in top_peaks:
            k2i = int(k2)
            lo2 = max(0, k2i - n + 1)
            hi2 = min(n - 1, k2i)
            idx2 = np.arange(lo2, hi2 + 1)
            costs[idx2] += a[idx2] * a[k2i - idx2]

        improved = False
        for src in top_src:
            if time.monotonic() - t0 > time_limit:
                break
            src_i = int(src)
            if a[src_i] < 1e-12:
                continue
            candidates = np.argsort(costs + 1e-10 * a)[:max(10, n // 30)]
            candidates = [int(c) for c in candidates if int(c) != src_i]
            for dst_i in candidates[:8]:
                found = False
                for frac in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7]:
                    amt = a[src_i] * frac
                    a_trial = a.copy()
                    a_trial[src_i] -= amt
                    a_trial[dst_i] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            J = 2.0 * n * best_m
            print(f"    CD ({time.monotonic()-t0:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def make_decreasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((n - x + 1) ** power)


def make_increasing(n, power=0.5):
    x = np.arange(n, dtype=np.float64)
    return norm_sum1((x + 1) ** power)


def make_mixed(n, pos1=0.2, pos2=0.65, width=0.1, ratio=1.0):
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - pos1) / width) ** 2) + ratio * np.exp(-((x - pos2) / (width * 1.5)) ** 2)
    return norm_sum1(a)


def make_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    a = np.sinc((x - 0.5) * bw) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_centered_power(n, power=0.5):
    x_c = np.abs(np.arange(n) - n / 2.0) + 1.0
    return norm_sum1(x_c ** (-power))


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_exponential_decay(n, rate=1.0):
    x = np.arange(n, dtype=np.float64) / n
    return norm_sum1(np.exp(-rate * x))


def make_two_block(n, frac=0.5):
    a = np.zeros(n)
    a[:int(frac * n)] = 1.0
    return norm_sum1(a)


def try_update(a, label):
    """Top-level function to track best sequence."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a_list = [float(v) for v in a]
    try:
        sc = evaluate_sequence(a_list)
    except Exception:
        return
    if np.isfinite(sc) and sc < _best_tracker['score']:
        _best_tracker['score'] = sc
        _best_tracker['seq'] = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)})")


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    _best_tracker['seq'] = None
    _best_tracker['score'] = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(v) for v in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _best_tracker['seq'] = parent.copy()
            _best_tracker['score'] = p_score
    except Exception:
        pass

    if _best_tracker['seq'] is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Default init: n={len(_best_tracker['seq'])}, score={_best_tracker['score']:.6f}")

    n0 = len(_best_tracker['seq'])
    print(f"Starting: n={n0}, score={_best_tracker['score']:.6f}")

    # Phase 1: AMD on diverse inits
    print(f"\n=== Phase 1: AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.40, 650.0)

    if t_phase1 > 30:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(make_decreasing(n0, 0.3)); labels.append("dec03")
        inits.append(make_decreasing(n0, 0.7)); labels.append("dec07")
        inits.append(make_increasing(n0, 0.3)); labels.append("inc03")
        inits.append(make_mixed(n0, 0.15, 0.65, 0.08)); labels.append("mixA")
        inits.append(make_mixed(n0, 0.1, 0.75, 0.12, 0.8)); labels.append("mixB")
        inits.append(make_sinc(n0, 4.0)); labels.append("sinc4")
        inits.append(make_sinc(n0, 7.0)); labels.append("sinc7")
        inits.append(make_sinc(n0, 10.0)); labels.append("sinc10")
        inits.append(make_cosine(n0, 1)); labels.append("cos1")
        inits.append(make_cosine(n0, 2)); labels.append("cos2")
        inits.append(make_centered_power(n0, 0.5)); labels.append("cctr05")
        inits.append(make_centered_power(n0, 1.0)); labels.append("cctr10")
        inits.append(make_exponential_decay(n0, 1.5)); labels.append("exp15")
        inits.append(make_exponential_decay(n0, 0.5)); labels.append("exp05")
        inits.append(make_random_smooth(n0, seed=3, n_freq=6)); labels.append("rnd1")
        inits.append(make_random_smooth(n0, seed=42, n_freq=10)); labels.append("rnd2")
        inits.append(make_two_block(n0, 0.5)); labels.append("blk50")

        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(8.0, t_phase1 / n_inits)
        for idx in range(n_inits):
            if time.monotonic() - t_global > time_budget - 30:
                break
            t_avail = min(t_per, time_budget - 30 - (time.monotonic() - t_global))
            if t_avail < 5:
                break
            lbl = unique_labels[idx]
            print(f"  AMD '{lbl}' t={t_avail:.0f}s")
            a_out, _ = amd_optimize(unique_inits[idx], t_avail,
                                    top_k_init=25, temp_init=0.003, c0_init=3.0,
                                    restart_every=4000, seed=100 + idx, verbose=False)
            try_update(a_out, f"amd_{lbl}")

    # Phase 2: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coord Descent ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 15:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = coord_descent_polish(a_in, t2, seed=42, verbose=False)
        try_update(a_out, "coord1")

    # Phase 3: AMD refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: AMD Refine ===")
    t3 = min(remaining * 0.18, 270.0)
    if t3 > 20:
        a_in = norm_sum1(_best_tracker['seq'])
        a_out, _ = amd_optimize(a_in, t3, top_k_init=40, temp_init=0.002,
                                c0_init=2.5, restart_every=6000, seed=999, verbose=False)
        try_update(a_out, "amd_refine")

    # Phase 4: Multi-n search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 4: Multi-n ===")
    candidate_ns = []
    for delta in [-500, -1000, 500, 1000, 1500, 2000]:
        new_n = n_best + delta
        if 1500 <= new_n <= 8000:
            candidate_ns.append(new_n)

    for new_n in candidate_ns[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  n={new_n}")
        a_interp = interpolate_seq(_best_tracker['seq'], new_n)
        try_update(a_interp, f"intp{new_n}")
        t4a = min(remaining * 0.30, 150.0)
        if t4a > 15:
            a_out, _ = amd_optimize(a_interp, t4a, top_k_init=30,
                                    temp_init=0.003, c0_init=3.0,
                                    restart_every=5000, seed=777, verbose=False)
            try_update(a_out, f"amd{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t4b = min(remaining * 0.20, 100.0)
            if t4b > 10:
                a_out, _ = coord_descent_polish(a_interp, t4b, seed=555, verbose=False)
                try_update(a_out, f"cd{new_n}")

    # Phase 5: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_best_tracker['seq'])
    print(f"\n=== Phase 5: Final (n={n_best}) ===")
    if remaining > 30:
        t5a = min(remaining * 0.50, 200.0)
        if t5a > 10:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = amd_optimize(a_in, t5a, top_k_init=50,
                                    temp_init=0.0015, c0_init=2.0,
                                    restart_every=8000, seed=2024, verbose=False)
            try_update(a_out, "amd_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t5b = min(remaining * 0.55, 150.0)
        if t5b > 10:
            a_in = norm_sum1(_best_tracker['seq'])
            a_out, _ = coord_descent_polish(a_in, t5b, seed=31415, verbose=False)
            try_update(a_out, "cd_final")

    # Final
    if _best_tracker['seq'] is not None:
        fs = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        if np.isfinite(fs):
            _best_tracker['score'] = fs
        print(f"\nFINAL: score={_best_tracker['score']:.6f}, n={len(_best_tracker['seq'])}, t={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        _best_tracker['seq'] = norm_sum1(np.sin(np.pi * x) ** 2)
        _best_tracker['score'] = evaluate_sequence([float(v) for v in _best_tracker['seq']])
        print(f"Fallback: score={_best_tracker['score']:.6f}")

    return [float(v) for v in _best_tracker['seq']]
