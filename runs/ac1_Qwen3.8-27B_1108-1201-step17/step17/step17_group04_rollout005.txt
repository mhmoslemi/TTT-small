This is O(n) to compute for all k! Much better than O(n log n) for FFT.

So the coordinate descent becomes:
- For each coordinate i: O(n_grid * n) to evaluate all candidates
- For n coordinates: O(n * n_grid * n) = O(n^2 * n_grid)
- For n=3200, n_grid=50: 3200^2 * 50 = 5 * 10^8 per sweep

That's still a lot. But each "evaluation" is just O(n) (compute delta_conv and find max), so the total per sweep is n * n_grid * n = 3200 * 50 * 3200 = 5*10^8 simple operations. In numpy, this should take a few seconds per sweep.

Actually, we can vectorize the candidate evaluation. For a given i and a set of candidate deltas, the delta_conv for all k can be computed as:
delta_conv_matrix = 2 * deltas[:, None] * a_valid[None, :] + (deltas^2)[:, None] * I(2i)

where a_valid is a[k-i] for valid k. Then the max of (conv + delta_conv) for each candidate is just the max along axis=1.

This is O(n_grid * n) but vectorized in numpy, so it's fast.

Let me implement this more efficiently:

```python
def coordinate_descent_fast(a0, time_limit, verbose=False, n_sweeps=20, n_grid=50, seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv(a)
        current_peak = conv.max()
        improved_any = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            ai = a[i]
            
            # Generate candidate values
            # Spread around current value
            scale = max(ai, 1.0/n)
            candidates = np.concatenate([
                np.linspace(0, 3*scale, n_grid//2),
                ai * np.exp(rng.normal(0, 0.5, n_grid//2))
            ])
            candidates = np.clip(candidates, 0, 0.5)
            # Remove values too close to current
            candidates = candidates[np.abs(candidates - ai) > 1e-6]
            
            if len(candidates) == 0:
                continue
            
            deltas = candidates - ai
            
            # Compute delta_conv for each candidate
            # delta_conv[k] = 2*delta*a[k-i] + delta^2 * I(k=2i)
            # a[k-i] is valid for k = i, i+1, ..., i+n-1 (clamped to [0, 2n-2])
            k_start = i
            k_end = min(2*n-1, i+n)
            k_range = np.arange(k_start, k_end)
            a_vals = a[k_range - i]  # = a[0], a[1], ..., a[min(n-1, k_end-1-i)]
            
            # delta_conv at valid k positions
            delta_conv_valid = 2 * deltas[:, None] * a_vals[None, :]
            # Add delta^2 term if 2i is in the valid range
            if k_start <= 2*i < k_end:
                pos_2i = 2*i - k_start
                delta_conv_valid[:, pos_2i] += deltas**2
            
            # Full conv at valid k
            conv_valid = conv[k_start:k_end]
            
            # For invalid k (outside [k_start, k_end]), conv doesn't change
            # The peak could be outside this range, so we need to track it
            conv_unchanged_max = 0
            if k_start > 0:
                conv_unchanged_max = max(conv_unchanged_max, conv[:k_start].max())
            if k_end < 2*n-1:
                conv_unchanged_max = max(conv_unchanged_max, conv[k_end:].max())
            
            # New peak for each candidate = max(max(delta_conv_valid + conv_valid), conv_unchanged_max)
            new_conv_valid = conv_valid[None, :] + delta_conv_valid
            new_peaks = np.maximum(new_conv_valid.max(axis=1), conv_unchanged_max)
            
            # Find best candidate
            best_idx = np.argmin(new_peaks)
            best_new_peak = new_peaks[best_idx]
            
            if best_new_peak < current_peak - 1e-12:
                a[i] = candidates[best_idx]
                # Update conv incrementally
                delta = candidates[best_idx] - ai
                conv[k_start:k_end] += 2 * delta * a_vals + delta**2 * np.where(k_range == 2*i, 1, 0)
                # Actually, we need to be more careful with the delta^2 term
                conv[k_start:k_end] = new_conv_valid[best_idx]
                current_peak = max(conv.max())
                ai = a[i]
                improved_any = True
        
        if not improved_any:
            # Try random perturbation
            n_pert = max(5, n // 50)
            for _ in range(n_pert):
                idx = rng.integers(n)
                a[idx] = max(0, a[idx] + rng.normal(0, 0.01/a.max()))
            a = norm_sum1(a)
            conv = fast_conv(a)
            current_peak = conv.max()
        
        J_new = 2.0 * n * current_peak
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a.copy()
        
        if verbose and sweep % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc sweep {sweep} ({elapsed:.0f}s): J={best_J:.6f}")
    
    return best_a, best_J
```
