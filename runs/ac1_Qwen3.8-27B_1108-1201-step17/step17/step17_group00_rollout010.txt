def fast_autocorr(a):

def norm_sum1(a):

def project_simplex(v, target_sum=1.0):
    """Project v onto simplex {x: sum x = target_sum, x >= 0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_soft_subgrad(a, ac, top_k, beta):
    """Compute soft subgradient of max(a*c) using top-k peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = np.exp(beta * top_vals)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # d(a*a)[k]/da[i] = 2*a[k-i] if i <= k < i+n
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[k_int - idx]
    return grad

def lp_direction(y, constraint_list, solver):
    """
    Solve LP: min M s.t. (y*g)[k] <= M for k in constraint_list, sum g = 1, g >= 0.
    Returns the optimal g direction.
    """
    n = len(y)
    K = len(constraint_list)
    if K == 0:
        return None
    
    coeffs = np.zeros((K, n))
    for jj, k in enumerate(constraint_list):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            coeffs[jj, idx] = y[k_int - idx]
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    g_sum = g_val.sum()
    if g_sum < 1e-15:
        return None
    return g_val / g_sum

def backtracking_step(y, g_dir, best_m, max_iters=20):

    sizes = [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.02, 0.01]
    ac_y = fast_autocorr(y)
    M_y = ac_y.max()
    
    for alpha in sizes:
        y_new = (1.0 - alpha) * y + alpha * g_dir
        s = y_new.sum()
        if s < 1e-15:
            continue
        y_new = y_new / s
        M_new = fast_autocorr(y_new).max()
        if M_new < M_y - 1e-15:
            return y_new, alpha
    return None, 0.0

            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-EH iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-EH done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m

def subgradient_optimize(a0, time_limit, top_k=80, beta_init=5.0, beta_min=1.0,
                          c0=3.0, restart_every=5000, verbose=False):
    """Subgradient mirror descent on the simplex with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta_init
    iter_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = project_simplex(y, 1.0)
        
        ac = fast_autocorr(y)
        g = compute_soft_subgrad(y, ac, cur_top_k, cur_beta)
        
        eta = gamma / (np.sqrt(t) + 1.0)
        # Mirror step (exponential on log scale)
        log_a = np.log(np.maximum(y, 1e-15)) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = project_simplex(a_new, 1.0)
        
        # Also try Euclidean step (under relaxation)
        a_euclid = y - eta * g
        a_euclid = project_simplex(a_euclid, 1.0)
        
        # Choose the better one
        M_mirror = fast_autocorr(a_new).max()
        M_euclid = fast_autocorr(a_euclid).max()
        
        if M_euclid < M_mirror:
            x_new = a_euclid
        else:
            x_new = a_new
        
        x_prev = x
        x = x_new
        iter_count += 1
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive restart
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.015, n) * best_a
            x_r = best_a + pert
            x_r = project_simplex(x_r, 1.0)
            x = x_r
            x_prev = x_r.copy()
            # Adapt parameters
            cur_top_k = min(cur_top_k + 15, 150)
            cur_beta = max(cur_beta - 0.3, beta_min)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    SG restart t={t} ({elapsed:.0f}s): J={J:.6f}, tk={cur_top_k}, b={cur_beta:.1f}")
    
    return best_a, best_m
