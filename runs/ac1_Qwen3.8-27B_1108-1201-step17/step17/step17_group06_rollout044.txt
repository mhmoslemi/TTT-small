```python
"""
Hybrid LP + Mirror Descent optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding as the main workhorse, with mirror descent for polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()

def build_lp_matrix(f, n):
    """Build the sparse A_ub matrix for the LP."""
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # k ranges from 0 to 2n-2, j ranges from 0 to n-1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # b_j appears in (f*b)(k) for k = j, j+1, ..., j+n-1 (clipped to 2n-2)
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        for k in range(k_start, k_end + 1):
            i = k - j  # = f_i coefficient
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_direction(f, M, n):
    """Solve LP: max sum(b) s.t. (f*b)(k) <= M for all k, b >= 0"""
    c = -np.ones(n)
    A = build_lp_matrix(f, n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        b = result.x
        b = np.maximum(b, 0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return None

def lp_optimize(f0, time_limit, verbose=False):
    """Iterate LP-based direction finding."""
    n = len(f0)
    f = f0.copy() / f0.sum()
    best_f = f.copy()
    best_J = eval_J(f)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if stall > 50:
            break
        conv = fast_conv(f)
        M = conv.max()
        g = lp_direction(f, M, n)
        if g is None:
            break
        improved = False
        for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_new = (1-t)*f + t*g
            f_new = np.maximum(f_new, 0)
            f_new /= f_new.sum()
            J_new = eval_J(f_new)
            if J_new < best_J - 1e-12:
                f = f_new
                best_J = J_new
                best_f = f.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if verbose:
                print(f"    LP stall={stall}, J={best_J:.6f}")
    
    return best_f, best_J
```

"""
Hybrid LP + Mirror Descent optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Main workhorse: LP-based direction finding (solves max sum(b) s.t. f*b <= M, b>=0).
Secondary: Nesterov-accelerated mirror descent for fine-tuning.
Includes multi-start, length sweep, and adaptive restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def precompute_lp_structure(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows = []
    cols = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * length)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def build_lp_matrix(f, n, rows, cols):
    """Build sparse LP constraint matrix with current f values."""
    vals = f[rows - cols]
    return csr_matrix((vals.astype(np.float64), (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, n, rows, cols):
    """Solve LP: max sum(b) s.t. (f*b)(k) <= M for all k, b >= 0. Returns normalized b."""
    A = build_lp_matrix(f, n, rows, cols)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 30, 'presolve': True})
        if result.success and result.x is not None:
            b = np.maximum(result.x, 0.0)
            s = b.sum()
            if s > 1e-15:
                return b / s
    except Exception:
        pass
    return None


def lp_optimize(f0, time_limit, verbose=False, seed=42):
    """Iterative LP-based optimization with adaptive t and restarts."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_J = eval_J(f)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(seed)

    rows, cols = precompute_lp_structure(n)
    max_stall = 80

    iteration = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(f)
        M = conv.max()
        g = solve_lp_direction(f, M, n, rows, cols)
        if g is None:
            break

        improved = False
        # Adaptive t values: start larger when stalled
        if stall < 20:
            t_vals = [0.3, 0.5, 0.7, 0.1, 0.2, 0.4]
        elif stall < 40:
            t_vals = [0.5, 0.7, 0.9, 0.1, 0.3]
        else:
            t_vals = [0.7, 0.9, 0.95, 0.5, 0.2]

        for t in t_vals:
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            J_new = eval_J(f_new)
            if J_new < best_J - 1e-10:
                f = f_new
                best_J = J_new
                best_f = f.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 10 == 0 and verbose:

                    f = f_r
                    best_J = J_r
                    best_f = f.copy()
                    stall = 0

            print(f"      LP iter={iteration}, J={best_J:.6f}, stall={stall}, t={time.monotonic()-t0:.0f}s")

    return best_f, best_J


def compute_subgradient(a, conv, top_k=80, beta_w=10.0):
    """Compute soft subgradient of max(conv) at a using top-K lags."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta_w
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g


def mirror_descent(a0, time_limit, top_k=100, beta_w=8.0, c0=4.0,
                   restart_every=5000, verbose=False):
    """Nesterov-accelerated mirror descent."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta_w

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        conv = fast_conv(y)
        g = compute_subgradient(y, conv, top_k=cur_top_k, beta_w=cur_beta)

        eta = c0 / (np.sqrt(t) + 1.0)

        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        val = eval_J(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.005, 0.03)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert

            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 20, 200)

        if verbose and t % 20000 == 0:

    return best_a, best_val


def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=80.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.1) ** 2) + np.exp(-width * (x - 0.9) ** 2) + 0.005
    return a / a.sum()


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 12):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_half_hump(n, p=1.5):
    """Half-hump: peaked in the middle, zero at edges."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** p + 0.01
    return a / a.sum()


def make_sinusoidal(n, freq=3.0, phase=0.0, p=2.0):
    x = np.linspace(0, 1, n)
    a = (1.0 + np.cos(2 * np.pi * freq * x + phase)) ** p + 0.01
    return a / a.sum()


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s
