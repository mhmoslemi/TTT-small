def fast_autocorr(a):
    """Compute conv(a, a) using FFT. Returns array of length 2*n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    # Fix small negative values from numerical errors
    result[result < 0] = 0
    return result

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def symmetrize(a):
    """Make a sequence symmetric."""
    return 0.5 * (a + a[::-1])


def fw_lp_step(a, M, active_set, solver, warm_start=None):
    """
    Solve the LP for Frank-Wolfe direction with given active constraints.
    Returns the LP solution (a direction on the simplex).
    """
    n = len(a)
    # Build constraint matrix for active constraints only
    # conv(a, b)[k] = sum_i a[i] * b[k-i] (valid indices)
    # = sum_i T[k,i] * b[i] where T[k,i] = a[k-i] if 0<=k-i<n else 0
    K = len(active_set)
    T = np.zeros((K, n))
    for idx, k in enumerate(active_set):
        k_int = int(k)
        # b[i] contributes to conv[k] when a[k-i] is valid, i.e., 0 <= k-i < n
        # So i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            i_vals = np.arange(i_start, i_end + 1)
            T[idx, i_vals] = a[k_int - i_vals]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(K),
                   cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def fw_with_constraint_generation(a0, time_limit, solver, verbose=True, 
                                    max_active=500, top_k_init=100):
    """
    Frank-Wolfe with adaptive constraint generation.
    For symmetric sequences.
    """
    n = len(a0)
    a = symmetrize(norm_sum1(a0))
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    # Active constraint set
    # Only consider k in [0, n-1] due to symmetry
    all_lags = list(range(n))  # First half of lags
    top_idx = np.argsort(ac[:n])[::-1][:top_k_init]
    active_set = set(int(x) for x in top_idx)
    
    y = a.copy()
    x = a.copy()
    prev_y = a.copy()
    iteration = 0
    consecutive_no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        # Update active set: add most violated constraints
        current_max = np.max(ac[:n])
        # Find lags where ac[k] > M_curr * 0.9 (near-max)
        near_max_mask = ac[:n] > current_max * 0.85
        near_max_lags = np.where(near_max_mask)[0]
        
        # Also keep the top constraints
        top_lags = np.argsort(ac[:n])[::-1][:max_active]
        
        # Combine: keep current active + near-max + top
        new_active = set(active_set)
        for lag in near_max_lags:
            new_active.add(int(lag))
        for lag in top_lags:
            new_active.add(int(lag))
        
        # Cap the active set size
        if len(new_active) > max_active:
            # Keep the most binding ones
            lag_list = list(new_active)
            lag_vals = np.array([ac[int(lag)] for lag in lag_list])
            order = np.argsort(lag_vals)[::-1][:max_active]
            new_active = set(int(lag_list[i]) for i in order)
        
        active_set = new_active
        
        # Solve LP
        direction = fw_lp_step(y, M_curr, sorted(active_set), solver)
        if direction is None:
            # Fallback: use uniform direction
            direction = np.ones(n) / n
        
        # Nesterov step
        alpha = 2.0 / (iteration + 2.0)
        y_new = (1.0 - alpha) * y + alpha * direction
        y_new = np.maximum(y_new, 0.0)
        y_new = symmetrize(y_new)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if iteration == 1:
            x_new = y_new.copy()
        else:
            beta_t = (iteration - 1.0) / (iteration + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            x_new = symmetrize(x_new)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        prev_y = y
        y = y_new
        x = x_new
        
        # Check x as well
        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-15:
            best_m = mx
            best_a = x.copy()
            consecutive_no_improve = 0
        
        # Restart if stalled for too long
        if consecutive_no_improve > 50 and iteration > 100:
            # Perturb
            rng = np.random.default_rng(iteration)
            pert = rng.normal(0, 0.01, n) * best_a

def subgradient_optimize(a0, time_limit, verbose=True, beta_init=10.0, 
                          beta_decay=0.002, c0=3.0, seed=42):
    """
    Smoothed subgradient method with Nesterov acceleration.
    For symmetric sequences.
    """
    n = len(a0)
    a = symmetrize(norm_sum1(a0))
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    iteration = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        cur_beta = max(3.0, beta_init - beta_decay * iteration)
        
        # Nesterov extrapolation
        beta_t = (iteration - 1.0) / (iteration + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = symmetrize(y)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute smoothed subgradient
        ac = fast_autocorr(y)
        shifted = ac - ac.max()
        w = np.exp(cur_beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w = w / w_sum
        w[w < 1e-14] = 0
        
        # Gradient: grad[j] = 2 * sum_m w[m] * y[m-j] (valid indices)
        # = 2 * conv(y[::-1], w)[n-1 : 2n-1]
        fft_size = 1 << (2 * n - 1).bit_length()
        y_rev = np.zeros(fft_size)
        y_rev[:n] = y[::-1]
        w_pad = np.zeros(fft_size)
        w_pad[:len(w)] = w
        conv = np.fft.irfft(np.fft.rfft(y_rev) * np.fft.rfft(w_pad), fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        # Mirror descent step (on log scale)
        eta = c0 / (np.sqrt(iteration) + 1.0)
        log_y = np.log(np.maximum(y, 1e-30))
        log_new = log_y - eta * grad
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = symmetrize(a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x.copy()
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = symmetrize(x_r)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
        
        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"  SG iter={iteration} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.2f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length, preserving symmetry."""
    old_n = len(f)
    if old_n == new_n:
        return symmetrize(f.copy())
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    f_interp = symmetrize(f_interp)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def make_symmetric_init(n, seed=0, type='hann'):
    """Create a symmetric initialization."""
    x = np.linspace(0, 1, n)
    if type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif type == 'raised':
        a = np.where(x < 0.5, 2*x, 2*(1-x))
        a = 0.5 + 0.5 * a
    elif type == 'flat':
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    a = symmetrize(a)
    return norm_sum1(a)

        max(1000, n_best - 400),
        max(1000, n_best - 200),
        n_best,
        n_best + 200,
        n_best + 400,
        n_best + 800,
    ]), key=lambda x: abs(x - n_best))
    
    for new_n in candidate_lengths:
        if new_n == n_best:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        if new_n > 6000 or new_n < 1000:
            continue
        
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        
        # Quick subgradient
        t_sg = min(remaining * 0.3, 100.0)
        if t_sg > 15:
            a_out, _ = subgradient_optimize(a_interp, t_sg, verbose=False,
                                             beta_init=8.0, beta_decay=0.002,
                                             c0=3.0, seed=new_n % 1000)
            try_update(a_out, f"sg_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        # Quick FW if time permits
        if remaining > 60 and new_n <= 4000:
            t_fw = min(remaining * 0.4, 150.0)
            if t_fw > 20:
                a_in = symmetrize(norm_sum1(best_seq))
                a_interp = interpolate_sequence(a_in, new_n)
                a_out, _ = fw_with_constraint_generation(a_interp, t_fw, solver,
                                                          verbose=False,
                                                          max_active=200, top_k_init=50)
                try_update(a_out, f"fw_n{new_n}")
    
    # === Phase 6: Final polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Polish (n={n_best}) ===")
    if remaining > 30:
        t6 = min(remaining * 0.4, 200.0)
        if t6 > 15:
            a_in = symmetrize(norm_sum1(best_seq))
            a_out, _ = coord_descent_sym(a_in, t6, verbose=True, seed=777)
            try_update(a_out, "coord_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and n_best <= 4000:
            t6b = min(remaining * 0.5, 150.0)
            if t6b > 15:
                a_in = symmetrize(norm_sum1(best_seq))
                a_out, _ = fw_with_constraint_generation(a_in, t6b, solver,
                                                          verbose=True,
                                                          max_active=200, top_k_init=30)
                try_update(a_out, "fw_final")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
