            stall += 1
        
        # Adaptive beta
        cur_beta = max(3.0, beta_init - beta_decay * iters)
        
        # Restart on stall
        if stall > restart_every:
            stall = 0
            if verbose:
                print(f"    MD restart at iter={iters}, obj={best_obj:.8f}, t={elapsed:.0f}s")
            # Perturb around best
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            
            # Sometimes full random restart
            if rng.random() < 0.2:
                x = rng.dirichlet(np.ones(n) * 0.3)
                x_prev = x.copy()
        
        if verbose and iters % 500 == 0:
            print(f"    MD iter={iters} ({elapsed:.0f}s): obj={best_obj:.8f}, stall={stall}")
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, seed=123, verbose=False):
    """
    Coordinate descent: move mass from high-contributing positions to low-sensitivity positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = objective_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 300
    moves = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        elapsed = time.monotonic() - t0
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Find top contributing lags
        n_top = min(20, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_lags[:min(8, n_top)]:
            if time.monotonic() - t0 > time_limit or stall >= max_stall:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Compute sensitivity at each destination
            sens = np.zeros(n)
            for k2 in top_lags[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for d in range(s2, e2 + 1):
                    partner = k2i - d
                    sens[d] += a[partner]
            
            # Find good destination (low sensitivity, far from source)
            dist = np.abs(np.arange(n) - src).astype(float)
            scores = sens * (1.0 + 0.1 * dist / n)
            # Exclude source and nearby
            scores[src] = 1e10
            nearby = np.abs(np.arange(n) - src) < max(3, n // 50)
            scores[nearby] = np.maximum(scores[nearby], sens.mean())
            
            best_dst = np.argmin(scores)
            if best_dst == src:
                best_dst = (src + n // 2) % n
            
            # Try moving mass
            found = False
            for frac in [0.01, 0.005, 0.02, 0.001, 0.03, 0.002]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                obj_trial = objective_normalized(a_trial)
                if obj_trial < best_obj - 1e-15:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    moves += 1
                    break
            
            if found:
                stall = 0
                if verbose and moves % 100 == 0:
                    print(f"    Coord ({elapsed:.0f}s): obj={best_obj:.8f}, moves={moves}")
        
        if not improved:
            stall += 1
            # Random perturbation every 20 stalls
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    obj_new = objective_normalized(a)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                        stall = 0
    
    if verbose:
        print(f"    Coord final ({time.monotonic()-t0:.0f}s): obj={best_obj:.8f}, moves={moves}")
    return best_a, best_obj

def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, seed=0, mode="smooth"):
    """Create diverse initializations."""
    rng = np.random.default_rng(seed)
    if mode == "uniform":
        return np.ones(n) / n
    elif mode == "smooth":
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for f in range(1, 8):
            a += rng.normal(0, 0.05) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.0)
        return norm_sum1(a)
    elif mode == "symmetric":
        a = make_init(n, seed=seed + 100, mode="smooth")
        a_sym = 0.5 * (a + a[::-1])
        return norm_sum1(a_sym)
    elif mode == "lowpass":
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(np.pi * x) ** 2
        a = np.maximum(a, 0.0)
        return norm_sum1(a)
    else:
        return np.ones(n) / n

def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Phase 1: Evaluate parent and uniform baselines
    Phase 2: Mirror descent from multiple initializations
    Phase 3: Coordinate descent polish
    Phase 4: Length exploration
    Phase 5: Final refinement
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: score={sc:.6f}, n={len(a)}, t={elapsed:.0f}s")
    
    # === Phase 1: Baselines ===
    print("=== Phase 1: Baselines ===")
    
    # Try parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            try_update(parent, "parent")
            print(f"  Parent: n={len(parent)}, score={best_score:.6f}")
    except Exception:
        pass
    
    # Try uniform sequences at various lengths
    for n_try in [1000, 2000, 3200, 4000, 5000]:
        u = np.ones(n_try) / n_try
        try_update(u, f"uniform_n{n_try}")
    
    if best_seq is None:
        n0 = 3200
        best_seq = np.ones(n0) / n0
        best_score = evaluate_safe(best_seq)
        print(f"  Default: n={n0}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"  Starting best: n={n0}, score={best_score:.6f}")
    
    # === Phase 2: Mirror Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror Descent (budget={remaining:.0f}s) ===")
    t_phase2 = min(remaining * 0.35, 600.0)
    
    if t_phase2 > 30:
        # Try multiple initializations
        n_try = n0
        inits = [
            (make_init(n_try, seed=1, mode="smooth"), "smooth1"),
            (make_init(n_try, seed=2, mode="symmetric"), "sym2"),
            (make_init(n_try, seed=3, mode="lowpass"), "lowpass3"),
            (make_init(n_try, seed=7, mode="smooth"), "smooth7"),
        ]
        if parent is not None:
            inits.insert(0, (norm_sum1(parent), "parent"))
        
        t_per = max(15.0, t_phase2 / len(inits))
        for idx, (init, lbl) in enumerate(inits):
            elapsed = time.monotonic() - t_global
            if elapsed > time_budget - 30:
                break
            t_avail = min(t_per, time_budget - 30 - elapsed)
            if t_avail < 10:
                break
            print(f"  MD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = mirror_descent(init, t_avail, beta_init=6.0, beta_decay=0.001,
                                       restart_every=2000, c0=4.0, seed=42 + idx, verbose=False)
            try_update(a_out, f"md_{lbl}")
    
    # === Phase 3: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coordinate Descent (budget={remaining:.0f}s) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    
    if t3 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t3, seed=42, verbose=True)
        try_update(a_out, f"coord1_n{n_best}")
    
    # === Phase 4: MD Polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: MD Polish (budget={remaining:.0f}s) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 250.0)
    
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t4, beta_init=8.0, beta_decay=0.0005,
                                   restart_every=5000, c0=3.0, seed=999, verbose=False)
        try_update(a_out, f"md_polish1_n{n_best}")
        
        # Second polish with different params
        remaining = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining * 0.10, 150.0)
        if t4b > 10:
            a_out, _ = mirror_descent(a_in, t4b, beta_init=10.0, beta_decay=0.0003,
                                       restart_every=4000, c0=5.0, seed=777, verbose=False)
            try_update(a_out, f"md_polish2_n{n_best}")
    
    # === Phase 5: Length Exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Length Exploration (budget={remaining:.0f}s) ===")
    n_best = len(best_seq)
    
    # Try nearby lengths
    candidate_lengths = sorted(set([
        max(500, n_best - 500), max(500, n_best - 200),
        n_best + 200, n_best + 500, n_best + 1000
    ]))
    candidate_lengths = [l for l in candidate_lengths if 500 <= l <= 8000]
    
    for new_n in candidate_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if new_n == n_best:
            continue
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_seq(best_seq, new_n)
        
        # Quick MD
        t5a = min(remaining * 0.15, 100.0)
        if t5a > 10:
            a_out, _ = mirror_descent(a_interp, t5a, beta_init=6.0, beta_decay=0.002,
                                       restart_every=1500, c0=4.0, seed=555, verbose=False)
            try_update(a_out, f"md_n{new_n}")
        
        # Quick coord
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.10, 80.0)
            if t5b > 8:
                a_out, _ = coordinate_descent(a_interp, t5b, seed=333, verbose=False)
                try_update(a_out, f"coord_n{new_n}")
    
    # === Phase 6: Final Coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Coord (budget={remaining:.0f}s) ===")
    n_best = len(best_seq)
    t6 = min(remaining * 0.25, 200.0)
    
    if t6 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t6, seed=2024, verbose=True)
        try_update(a_out, f"coord_final_n{n_best}")
    
    # === Phase 7: Final MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final MD (budget={remaining:.0f}s) ===")
    n_best = len(best_seq)
    t7 = min(remaining * 0.4, 200.0)
    
    if t7 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t7, beta_init=12.0, beta_decay=0.0002,
                                   restart_every=8000, c0=3.0, seed=31415, verbose=False)
        try_update(a_out, f"md_final_n{n_best}")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
