def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_true_gradient(a, t):
    """Compute exact gradient of smoothed max loss L=(1/t)log(sum exp(t*b[k])).
    Returns (grad, loss_value)."""
    n = len(a)
    ac = fast_autocorr(a)
    max_ac = ac.max()
    if t < 1e-8:
        p = np.zeros_like(ac)
        p[ac.argmax()] = 1.0
        loss = max_ac
    else:
        shifted = t * (ac - max_ac)
        exp_shifted = np.exp(shifted)
        p = exp_shifted / exp_shifted.sum()
        loss = max_ac + np.log(exp_shifted.sum()) / t
    # grad[i] = 2 * sum_j a[j] * p[j+i] = 2*(a cross_padded)[i]
    # Use FFT cross-correlation: (a ⋆ p)[i] = sum_j a[j]*p[j+i]
    n_p = len(p)
    N = 1 << (n + n_p - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    C = np.fft.irfft(np.conj(P) * A, N)
    grad = 2.0 * C[:n]
    return grad, loss

            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    # Phase 1: Smoothed-max mirror descent (true gradient)
    print(f"\n=== Phase 1: Smoothed-max GD (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.28, 480.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_max_mirror_descent(a_in, t1, t_start=5.0, t_end=1000.0,
                                             c0=4.5, restart_every=5000,
                                             verbose=True, rng_seed=42)
        try_update(a_out, "smgd_parent")

    # Phase 2: LP Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW v2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 440.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_v2(a_in, t2, solver, verbose=True, top_K_init=250)
        try_update(a_out, "lp_v2")

    # Phase 3: Peak surgery
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak surgery (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_surgery(a_in, t3, verbose=True, rng_seed=77)
        try_update(a_out, "surgery")

    # Phase 4: Second smoothed-max GD round
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: SMGD round 2 (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.15, 260.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_max_mirror_descent(a_in, t4, t_start=20.0, t_end=2000.0,
                                             c0=3.0, restart_every=6000,
                                             verbose=True, rng_seed=100)
        try_update(a_out, "smgd_r2")

    # Phase 5: LP-FW round 2
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP-FW round 2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_v2(a_in, t5, solver, verbose=True, top_K_init=400)
        try_update(a_out, "lp_r2")

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration (base n={n_best}) ===")
    trial_lengths = [n_best + 200, n_best - 200, n_best + 500]
    trial_lengths = [l for l in trial_lengths if 2000 <= l <= 6000]
    trial_lengths = list(dict.fromkeys(trial_lengths))
    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.4, 200.0)
        if t6a > 30:
            a_out, _ = smooth_max_mirror_descent(a_interp, t6a, t_start=10.0, t_end=800.0,
                                                 c0=4.0, restart_every=4000,
                                                 verbose=True, rng_seed=new_n % 997)
            try_update(a_out, f"smgd_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 4500:
            t6b = min(remaining * 0.4, 120.0)
            if t6b > 20:
                a_out, _ = lp_fw_v2(a_interp, t6b, solver, verbose=False, top_K_init=200)
                try_update(a_out, f"lp_n{new_n}")

    # Phase 7: Coordinate polish + surgery
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    t7 = min(remaining * 0.3, 100.0)
    if t7 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_surgery(a_in, t7, verbose=False, rng_seed=555)
        try_update(a_out, "surgery_f")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7b = min(remaining * 0.4, 60.0)
        if t7b > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t7b, verbose=False, rng_seed=789)
            try_update(a_out, "coord_f")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        n_fb = 4200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
