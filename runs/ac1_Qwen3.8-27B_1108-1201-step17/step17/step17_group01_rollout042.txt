def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_stochastic_subgradient(a, ac, top_k=30, beta=4.0, rng=None):
    n = len(a)
    k_max = min(top_k, len(ac))
    if rng is not None and k_max > 8:
        # Random subset of top peaks for speed
        all_top = np.argsort(ac)[::-1][:min(3 * k_max, len(ac))]
        chosen = rng.choice(all_top, size=k_max, replace=False)
        top_idx = chosen
    else:
        top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=40, beta=5.0, c0=5.0, restart_every=3000,
                 verbose=False, rng_seed=42, use_stochastic=True, stoch_frac=0.5):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_count = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        if use_stochastic and t % 2 == 0:
            # Use fewer peaks every other step for speed
            tk = max(5, int(cur_top_k * stoch_frac))
        else:
            tk = cur_top_k
        g = compute_stochastic_subgradient(y, ac, top_k=tk, beta=cur_beta, rng=rng)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:

        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.05, 0.01 + 0.005 * restart_count)
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 8, 100)
            cur_beta = max(cur_beta - 0.2, 2.0)
            gamma = c0 * (1.0 + 0.05 * min(restart_count, 10))
            if verbose:
                elapsed = time.monotonic() - t0

            # More aggressive restart: blend with random direction
            if restart_count % 3 == 0:
                # Mix with a random smooth perturbation
                freq = rng.integers(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                xx = np.linspace(0, 1, n)
                pert = 0.05 * best_a * (1 + 0.3 * np.sin(2 * np.pi * freq * xx + phase))
                x_r = best_a + pert
            else:
                strength = min(0.04, 0.01 + 0.003 * restart_count)
                pert = rng.normal(0, strength, n) * best_a
                x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 5, 120)
            cur_beta = max(cur_beta - 0.15, 2.0)
            gamma = c0 * (1.0 + 0.03 * min(restart_count, 15))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD+LS restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD+LS t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def freq_domain_optimize(a0, time_limit, iters=5000, eta0=0.4, verbose=False):
    """Gerchberg-Saxton style spectrum flattening."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        eta = eta0 * (1.0 + 0.3 * it / max(1, iters))
        P_target = P_mean * (P / P_mean) ** (1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.6 * a + 0.4 * a_new
            a = norm_sum1(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FreqDom it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def hybrid_optimize(a0, time_limit, verbose=False, rng_seed=42):
    """Alternates between AMD and frequency-domain steps."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    fft_size = 1 << (2 * n - 1).bit_length()

    x = a.copy()
    x_prev = a.copy()
    gamma = 5.0
    cur_top_k = 50
    cur_beta = 5.0
    stall = 0
    mode = 0  # 0=AMD, 1=freq

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        if mode == 0:
            # AMD step
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

            ac = fast_autocorr(y)
            g = compute_stochastic_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta, rng=rng)
            eta = gamma / (np.sqrt(t) + 1.0)
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new
            mode = 1
        else:
            # Freq domain step
            A = np.fft.rfft(x, fft_size)
            P = np.abs(A) ** 2
            P_mean = P.mean()
            if P_mean < 1e-30:
                mode = 0
                continue
            eta_f = 0.3
            P_target = P_mean * (P / P_mean) ** (1.0 - eta_f)
            P_target = np.maximum(P_target, 1e-30)
            phase = np.angle(A)
            A_new = np.sqrt(P_target) * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                mode = 0
                continue
            a_new = a_new / s_new
            x = x + 0.3 * (a_new - x)
            x = np.maximum(x, 1e-15)
            x = x / x.sum()
            x_prev = x
            mode = 0

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            strength = 0.02 + 0.005 * (stall // 3000)
            pert = rng.normal(0, min(strength, 0.05), n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            cur_top_k = min(cur_top_k + 5, 100)
            cur_beta = max(cur_beta - 0.2, 2.5)

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=42):
    """Local coordinate descent on pairs of indices."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 50

    # Precompute which indices are important (top peaks)
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        # Try moving mass from high-value indices to low-cost positions
        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_i = np.argmax(contrib)
            src = idx[max_i]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find candidate destinations (low total contribution)
            costs = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                js2 = max(0, k2i - n + 1)
                je2 = min(n - 1, k2i)
                if js2 > je2:
                    continue
                idx2 = np.arange(js2, je2 + 1)
                costs[idx2] += a[k2i - idx2]

            # Sample a few candidates
            n_cand = min(20, n // 5)
            cand = rng.choice(n, size=n_cand, replace=False)
            best_dst = cand[np.argmin(costs[cand] + 0.01 * a[cand])]

            if best_dst == src:
                continue

            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordPolish ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_ramp(n, power=1.0, start=0.0):
    """Asymmetric power-law ramp: (x+start)^power normalized."""
    x = np.arange(n, dtype=np.float64)
    a = (x + 1.0 + start) ** power
    return norm_sum1(a)


def make_exp_decay(n, rate=0.5, rising=False):
    x = np.arange(n, dtype=np.float64) / n
    if rising:
        a = np.exp(rate * x)
    else:
        a = np.exp(-rate * (1 - x))
    return norm_sum1(a)


def make_quad_asym(n, center=0.6, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    return norm_sum1(a)


def make_random_smooth(n, seed=0, n_freq=5, asymmetry=1.0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    # Apply asymmetry
    a *= (1.0 + asymmetry * x)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_chirp(n, n_cycles=3.0):
    """Chirp-like asymmetric shape."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2 * (0.5 + 0.5 * x)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase optimization."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_ramp(n_base, power=1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Multi-start AMD at base length ===
    print(f"\n=== Phase 1: Multi-start AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 500.0)

    if t_phase1 > 40:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_ramp(n0, power=1.0))
        labels.append("linear_ramp")

        inits.append(make_ramp(n0, power=0.5))
        labels.append("sqrt_ramp")

        inits.append(make_ramp(n0, power=2.0))
        labels.append("quad_ramp")

        inits.append(make_exp_decay(n0, rate=1.0, rising=True))
        labels.append("exp_rise")

        inits.append(make_quad_asym(n0, center=0.6, width=0.25))
        labels.append("asym_gauss")

        inits.append(make_quad_asym(n0, center=0.7, width=0.2))
        labels.append("asym_gauss2")

        inits.append(make_chirp(n0, n_cycles=2.0))
        labels.append("chirp")

        inits.append(make_random_smooth(n0, seed=13, n_freq=4, asymmetry=1.5))
        labels.append("smooth_asym1")

        inits.append(make_random_smooth(n0, seed=77, n_freq=6, asymmetry=2.0))
        labels.append("smooth_asym2")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        n_starts = len(inits)
        t_per = max(15.0, t_phase1 / n_starts)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 30:
                break
            t_avail = min(t_per, time_budget - 30 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=50, beta=5.0, c0=5.0,
                                     restart_every=2500, verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: Hybrid optimization on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Hybrid (n={n_best}) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = hybrid_optimize(a_in, t2, verbose=False, rng_seed=99)
        try_update(a_out, f"hybrid_n{n_best}")

    # === Phase 3: AMD with line search on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: AMD+LS (n={n_best}) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_with_linesearch(a_in, t3, top_k=60, beta=4.5, c0=5.0,
                                       restart_every=4000, verbose=False, rng_seed=123)
        try_update(a_out, f"amdls_n{n_best}")

    # === Phase 4: Frequency domain polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: FreqDom (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = freq_domain_optimize(a_in, t4, iters=5000, eta0=0.4, verbose=False)
        try_update(a_out, f"freqdom_n{n_best}")

    # === Phase 5: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Longer sequences ===")
    longer_lengths = sorted(set([n_best + 500, n_best + 1000, n_best + 2000, n_best + 4000]))
    longer_lengths = [l for l in longer_lengths if l <= 10000]

    for new_n in longer_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Scaling to n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t5a = min(remaining * 0.35, 200.0)
        if t5a > 20:
            a_out, _ = amd_optimize(a_long, t5a, top_k=60, beta=4.5, c0=5.0,
                                     restart_every=3000, verbose=False, rng_seed=555)
            try_update(a_out, f"amd_n{new_n}")

    # === Phase 6: Try shorter sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 50:
        shorter_lengths = sorted(set([n_best - 500, n_best - 1500, n_best - 3000]))
        shorter_lengths = [l for l in shorter_lengths if l >= 1000]
        for new_n in shorter_lengths[:2]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            print(f"  Trying n={new_n}...")
            a_short = interpolate_sequence(best_seq, new_n)
            try_update(a_short, f"interp_short_n{new_n}")
            t6 = min(remaining * 0.4, 120.0)
            if t6 > 15:
                a_out, _ = amd_optimize(a_short, t6, top_k=40, beta=5.0, c0=5.0,
                                         restart_every=2000, verbose=False, rng_seed=888)
                try_update(a_out, f"amd_short_n{new_n}")

    # === Phase 7: Coordinate polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coord Polish (n={n_best}) ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t7, verbose=False, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 8: Final AMD burst ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final AMD (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.6, 200.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t8, top_k=80, beta=3.5, c0=4.0,
                                     restart_every=5000, verbose=False, rng_seed=777)
            try_update(a_out, f"amd_final_n{n_best}")

    # === Phase 9: Hybrid on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final Hybrid (n={n_best}) ===")
        t9 = min(remaining * 0.7, 150.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = hybrid_optimize(a_in, t9, verbose=False, rng_seed=314)
            try_update(a_out, f"hybrid_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_ramp(n_fb, power=1.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
