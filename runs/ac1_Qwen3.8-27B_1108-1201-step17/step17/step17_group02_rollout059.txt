def fast_autocorr(a):
    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_conv_matrix_row(y, k, n):
    """Compute the row of the conv matrix for lag k: T[k] @ b = (y*b)[k]."""
    # (y*b)[k] = sum_j y[j] * b[k-j] for valid j
    # = sum_j b[j] * y[k-j] for valid j
    # So T[k, j] = y[k-j] if 0 <= k-j < n, else 0
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    row = np.zeros(n)
    if j_start <= j_end:
        idx = np.arange(j_start, j_end + 1)
        row[idx] = y[k - idx]
    return row

    iter_count = 0

    # Constraint management: start with top constraints, add violated ones
    K_max = max_K if max_K is not None else min(2 * n - 1, 4000)
    active_constraints = set()

    def update_active_constraints(current_point, M_bound, n_active_max):
        ac_cur = fast_autocorr(current_point)
        # Find constraints with value close to M_bound (potential violations)
        margins = M_bound - ac_cur
        # Add constraints with small margins (close to being tight)
        close_idx = np.where(margins < 0.05 * M_bound)[0]
        for idx in close_idx[:50]:
            active_constraints.add(int(idx))
        # Also add the top-K by value
        top_idx = np.argsort(ac_cur)[::-1][:100]
        for idx in top_idx:
            active_constraints.add(int(idx))
        # Cap the set size
        if len(active_constraints) > n_active_max:
            # Keep only the most relevant (those with highest values)
            vals = {k: ac_cur[k] for k in active_constraints}
            sorted_keys = sorted(vals.keys(), key=lambda k: vals[k], reverse=True)
            active_constraints.clear()
            for k in sorted_keys[:n_active_max]:
                active_constraints.add(k)

    def solve_lp(y_curr, M_bound, constraint_set):
        """Solve: max sum(b) s.t. (y*b)[k] <= M_bound for k in constraint_set, b >= 0."""
        n_var = len(y_curr)
        b_var = cp.Variable(n_var, nonneg=True)
        constraints = []
        for k in constraint_set:
            row = compute_conv_matrix_row(y_curr, k, n_var)
            constraints.append(row @ b_var <= M_bound)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            return None
        return b_val / b_sum

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Progressive M-tightening
        if t > 10:
            tighten_factor = 1.0 - initial_M_tighten - tighten_rate * (t // tighten_interval)
            tighten_factor = max(tighten_factor, 0.85)
        else:
            tighten_factor = 1.0 - initial_M_tighten
        M_target = M_curr * tighten_factor

        # Update active constraints
        if t % 5 == 1 or len(active_constraints) == 0:
            update_active_constraints(y, M_curr, K_max)

        if len(active_constraints) == 0:
            # Fallback: use top 200 by value
            top_idx = np.argsort(ac)[::-1][:200]
            active_constraints = set(int(i) for i in top_idx)

        # Try to solve LP with tightened M
        b_dir = solve_lp(y, M_target, active_constraints)

        lp_solves = 0
        if b_dir is None:
            # LP infeasible with tightened M, try with current M
            b_dir = solve_lp(y, M_curr, active_constraints)
            if b_dir is None:
                # Try with slightly relaxed M
                b_dir = solve_lp(y, M_curr * 1.01, active_constraints)
                if b_dir is None:
                    continue
            lp_solves += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = b_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        # Check improvement
        my = fast_autocorr(y).max()
        if my < best_m - 1e-16:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-16:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-T iter={t} ({elapsed:.0f}s): J={J:.6f}, M_tight={tighten_factor:.4f}, "
                  f"|constraints|={len(active_constraints)}, lp_solves={lp_solves}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    LP-T final ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def peak_shaving(a0, time_limit, verbose=True, rng_seed=42):
    """Targeted peak shaving: for each top peak, find and reduce the most responsible pair."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100
    n_peaks = 8

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_peaks]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            # Find top contributing pairs
            top_pairs = np.argsort(contrib)[::-1][:10]

            for pi in top_pairs:
                src = int(idx[pi])
                src_partner = int(partner[pi])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Determine which to reduce: reduce the one with lower "importance"
                # Importance = sum of contributions to all top peaks
                imp_src = 0.0
                imp_partner = 0.0
                for kp in top_k[:5]:
                    kpi = int(kp)
                    # Contribution of src to peak at kpi
                    if 0 <= kpi - src < n:
                        imp_src += a[src] * a[kpi - src]
                    # Contribution of partner to peak at kpi
                    if 0 <= kpi - src_partner < n:
                        imp_partner += a[src_partner] * a[kpi - src_partner]

                if imp_src >= imp_partner:
                    reduce_idx = src
                else:
                    reduce_idx = src_partner

                if a[reduce_idx] < 1e-12:
                    continue

                # Find safe destination: a position that minimizes impact on peaks
                # Try several candidates
                candidates = []
                # Positions far from peak support
                for d in range(n):
                    if d == reduce_idx:
                        continue
                    # Check if d is in support of any top peak
                    in_support = False
                    for kp in top_k[:5]:
                        kpi = int(kp)
                        ds = max(0, kpi - n + 1)
                        de = min(n - 1, kpi)
                        if ds <= d <= de:
                            in_support = True
                            break
                    if not in_support:
                        candidates.append(d)

                if not candidates:
                    candidates = [d for d in range(n) if d != reduce_idx]

                if not candidates:
                    continue

                # Pick the candidate with lowest sensitivity
                cand_arr = np.array(candidates[:50])
                sens = np.zeros(len(cand_arr))
                for ci, d in enumerate(cand_arr):
                    for kp in top_k[:5]:
                        kpi = int(kp)
                        if 0 <= kpi - d < n:
                            sens[ci] += a[d] * a[kpi - d]
                dst = int(cand_arr[np.argmin(sens)])

                # Binary search for maximum safe fraction
                lo, hi = 0.0, 0.5
                best_frac = 0.0
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    amt = a[reduce_idx] * mid
                    if amt < 1e-15:
                        hi = mid
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        hi = mid
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        best_frac = mid
                        lo = mid
                    else:
                        hi = mid

                if best_frac > 0.001:
                    amt = a[reduce_idx] * best_frac
                    a_new = a.copy()
                    a_new[reduce_idx] -= amt
                    a_new[dst] += amt
                    a_new = np.maximum(a_new, 0.0)
                    a_new = a_new / a_new.sum()
                    M_new = fast_autocorr(a_new).max()
                    if M_new < best_m - 1e-16:
                        a = a_new
                        best_m = M_new
                        best_a = a.copy()
                        improved = True

            if improved:
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            if stall > 20:
                n_pert = min(3, n)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakShave final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=5000,
                 rng_seed=42):
    """Adaptive mirror descent with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        # Compute subgradient
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 1.0, 2.0)

    return best_a, best_m


def spectral_init(n, n_modes=20, seed=0):
    """Generate a smooth non-negative sequence via Fourier synthesis."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):
    """U-shaped distribution."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with LP-FW M-tightening as the core method."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP-FW with M-tightening on parent (main optimization) ===
    print(f"\n=== Phase 1: LP-FW M-Tightening (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 750.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_tight(a_in, t1, solver, verbose=True,
                                         initial_M_tighten=0.0,
                                         tighten_rate=0.001,
                                         tighten_interval=8,
                                         max_K=min(6000, 2 * n0 - 1))
        try_update(a_out, "lp_tight_main")

    # === Phase 2: Peak shaving ===
    print(f"\n=== Phase 2: Peak Shaving (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t2, verbose=True, rng_seed=42)
        try_update(a_out, "peakshave")

    # === Phase 3: LP-FW round 2 with higher tightening ===
    print(f"\n=== Phase 3: LP-FW Round 2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_tight(a_in, t3, solver, verbose=True,
                                         initial_M_tighten=0.002,
                                         tighten_rate=0.0015,
                                         tighten_interval=5,
                                         max_K=min(8000, 2 * n_best - 1))
        try_update(a_out, "lp_tight_r2")

    # === Phase 4: Peak shaving round 2 ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Peak Shaving R2 (n={n_best}) ===")
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t4, verbose=True, rng_seed=123)
        try_update(a_out, "peakshave_r2")

    # === Phase 5: Length exploration ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Length Exploration ===")

    trial_lengths = []
    for delta in [200, -200, 400, -400]:
        new_n = n_best + delta
        if 2000 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.4, 200.0)
        if t5a > 30:
            a_out, _ = lp_frank_wolfe_tight(a_long, t5a, solver, verbose=False,
                                             initial_M_tighten=0.0,
                                             tighten_rate=0.001,
                                             tighten_interval=8,
                                             max_K=min(5000, 2 * new_n - 1))
            try_update(a_out, f"lp_n{new_n}")

    # === Phase 6: AMD initialization exploration ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        print(f"\n=== Phase 6: AMD Exploration (n={n_best}) ===")
        t6 = min(remaining * 0.2, 200.0)

        inits = []
        labels = []
        inits.append(make_u_shape(n_best, sharpness=2.0))
        labels.append("u2")
        inits.append(make_bimodal(n_best, width=0.2))
        labels.append("bimodal")
        inits.append(spectral_init(n_best, n_modes=30, seed=99))
        labels.append("spectral")

        t_per = max(15.0, t6 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 20)
            if t_avail < 10:
                break
            print(f"  AMD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=80, beta=6.0, c0=5.0,
                                     restart_every=4000, rng_seed=100 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 7: Final LP polish ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final LP Polish (n={n_best}) ===")
    if remaining > 30 and n_best <= 5000:
        t7 = min(remaining * 0.6, 200.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_tight(a_in, t7, solver, verbose=False,
                                             initial_M_tighten=0.003,
                                             tighten_rate=0.002,
                                             tighten_interval=3,
                                             max_K=min(10000, 2 * n_best - 1))
            try_update(a_out, "lp_final")

    # === Phase 8: Final peak shaving ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 8: Final Peak Shaving ===")
        t8 = min(remaining * 0.7, 100.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_shaving(a_in, t8, verbose=False, rng_seed=777)
            try_update(a_out, "peakshave_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
