                idx = np.arange(i_start, i_end)
                j_idx = k_max - idx
                contrib = x[idx] * x[j_idx]
                # Reduce the largest contributor
                src = idx[np.argmax(contrib)]
                if x[src] > 1e-10:
                    # Move mass to a low-sensitivity position
                    sens = np.zeros(n)
                    for kk in np.argsort(ac)[::-1][:5]:
                        kki = int(kk)
                        ss = max(0, kki - n + 1)
                        ee = min(n, kki + 1)
                        ii = np.arange(ss, ee)
                        sens[ii] += x[kki - ii]
                    dst = np.argmin(sens)
                    amt = x[src] * 0.05
                    x[src] -= amt
                    x[dst] += amt
                    x = np.maximum(x, 1e-15)
                    x = x / x.sum()
                    M_check = fast_autocorr(x).max()
                    if M_check < best_m:
                        best_m = M_check
                        best_a = x.copy()
                        stall = 0
                if stall > restart_every:
                    # Full restart with perturbation
                    x = best_a + rng.normal(0, 0.02, n) * best_a
                    x = np.maximum(x, 1e-15)
                    x = x / x.sum()
                    momentum = np.zeros(n)
                    stall = 0
        
        if verbose and t % 5000 == 0:
            J = 2.0 * n * best_m
            print(f"    MD t={t}: J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def peak_redistribute(a0, time_limit, verbose=True, rng_seed=77):
    """Direct peak redistribution: identify top peaks and redistribute mass."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_peaks = np.argsort(ac)[::-1][:8]
        
        improved = False
        
        for k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            contrib = a[idx] * a[j_idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_partner = j_idx[np.argmax(contrib)]
            
            if a[src] < 1e-12:
                continue
            
            # Sensitivity: how much does each position contribute to top peaks?
            sens = np.zeros(n)
            for kk in top_peaks[:5]:
                kki = int(kk)
                ss = max(0, kki - n + 1)
                ee = min(n, kki + 1)
                ii = np.arange(ss, ee)
                sens[ii] += a[kki - ii]
            
            # Find best destination (low sensitivity, not the same pair)
            exclude = set()
            exclude.add(src)
            exclude.add(src_partner)
            candidates = np.array([d for d in range(n) if d not in exclude])
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            cand_sens = sens[candidates]
            # Pick from the lowest sensitivity positions
            n_candidates = min(10, len(candidates))
            best_dsts = candidates[np.argsort(cand_sens)[:n_candidates]]
            
            found = False
            for dst in best_dsts:
                for frac in [0.01, 0.02, 0.05, 0.03, 0.005, 0.08, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break
            
            if not found and stall > 50:
                # Random perturbation
                j = int(rng.integers(n))
                frac = rng.uniform(0.005, 0.05)
                amt = a[src] * frac
                a[src] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
        
        if not improved:
            stall += 1
            # Multi-point perturbation
            n_pert = min(10, n)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            for i in pert_idx:
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakRedist ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakRedist final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent trying all pairs of positions for mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:10]
        
        # Build sensitivity
        sens = np.zeros(n)
        for kk in top_k[:6]:
            kki = int(kk)
            ss = max(0, kki - n + 1)
            ee = min(n, kki + 1)
            ii = np.arange(ss, ee)
            sens[ii] += a[kki - ii]
        
        improved = False
        
        # For each top peak, try moving mass from high contributors to low sensitivity positions
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            ss = max(0, ki - n + 1)
            ee = min(n, ki + 1)
            ii = np.arange(ss, ee)
            contrib = a[ii] * a[ki - ii]
            
            # Get top contributors
            top_contrib = np.argsort(contrib)[::-1][:5]
            
            for tc in top_contrib:
                src = ii[tc]
                if a[src] < 1e-12:
                    continue
                
                # Try moving to several low-sensitivity positions
                low_sens = np.argsort(sens)[:10]
                for dst in low_sens:
                    if dst == src:
                        continue
                    for frac in [0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            break
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = a_trial / a_trial.sum()
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                
                if time.monotonic() - t0 > time_limit:
                    break
            
            if time.monotonic() - t0 > time_limit:
                break
        
        if not improved:
            stall += 1
            # Random moves
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.003, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stall = 0
        
        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)

def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)

def make_power_shape(n, p=1.0):
    """a_i proportional to (distance from center)^p"""
    x = np.linspace(-1, 1, n)
    a = np.abs(x) ** p + 0.01
    return norm_sum1(a)

def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)
