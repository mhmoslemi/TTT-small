    """Compute subgradient using top-k lags."""
    n = len(y)
    k_max = min(top_k, len(ac))
    if k_max == len(ac):
        top_idx = np.arange(len(ac))
    else:
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
    top_idx = np.sort(top_idx)
    
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= k_max
    return g

def nesterov_mirror_descent_v2(a0, time_limit, top_k=25, c0=4.0, verbose=False):
    """NMD with backtracking and adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 2000
    rng = np.random.default_rng(42 + n)
    c = c0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        beta_t = (iteration - 1.0) / (iteration + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        ac = fast_conv(y)
        M = ac.max()
        
        # Adaptive top_k
        near_max = np.sum(ac >= M * 0.999)
        k_use = min(max(top_k, int(near_max) + 10), min(150, len(ac)))
        
        g = compute_subgrad_full(y, ac, k_use)
        
        # Backtracking step size
        eta = c / (np.sqrt(iteration) + 1.0)
        a_new = None
        for _ in range(10):
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_candidate = np.exp(log_a - max_log)
            a_candidate = np.maximum(a_candidate, 1e-30)
            a_candidate = a_candidate / a_candidate.sum()
            M_candidate = fast_conv(a_candidate).max()
            if M_candidate < M + eta * np.sum(g * (a - y)) + eta**2 / 2:
                a_new = a_candidate
                break
            eta *= 0.5
        if a_new is None:
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
            c = min(10.0, c * 1.01)  # Slightly increase if making progress
        else:
            stall_count += 1
            c = max(0.3, c * 0.999)
        
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(300, int(max_stall * 0.7))
            strategy = int(rng.integers(5))
            if strategy == 0:
                pert = rng.normal(0, 0.02, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n//4, 800), replace=False)
                x_r[idx] *= rng.uniform(0.5, 1.5, len(idx))
            elif strategy == 2:
                # Sinusoidal perturbation
                freq = rng.uniform(1, 10)
                phase = rng.uniform(0, 2*np.pi)
                t = np.linspace(0, 1, n)
                x_r = best_a * (1 + 0.03 * np.sin(2*np.pi*freq*t + phase))
            elif strategy == 3:
                # Polynomial perturbation
                t = np.linspace(0, 1, n)
                p = rng.uniform(-0.05, 0.05, size=4)
                x_r = best_a * (1 + p[0] + p[1]*t + p[2]*t**2 + p[3]*t**3)
            else:
                x_r = best_a * (1 + rng.normal(0, 0.003, n))
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.5, c * 0.7)
        
        if verbose and iteration % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD v2 iter={iteration} ({elapsed:.1f}s): J={J:.8f}, c={c:.3f}")
    
    return best_a, best_m

def lp_remez_v2(a, max_cons=50, r=0.1, solver=None, verbose=False):
    """LP Remez with better active set selection using second-order prediction."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Select active constraints: top-k by value, plus predicted future actives
    active = np.argsort(b)[::-1]
    
    # Use top max_cons
    active = active[:max_cons]
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    # Trust region
    constraints.append(a_var >= a * (1.0 - r))
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_lp_v2(a0, time_limit, solver=None, r_init=0.1, verbose=False):
    """LP-based Remez with adaptive trust region and multiple line search fractions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        a_cand = lp_remez_v2(a, max_cons=min(60, max(20, n//50)), r=r, solver=solver)
        
        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.5, r * 1.15)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.6)
            if no_improve > 15:
                # Structured perturbation to escape
                rng = np.random.default_rng(iteration)
                strategy = int(rng.integers(4))
                if strategy == 0:
                    idx = rng.choice(n, size=min(n//5, 500), replace=False)
                    a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                elif strategy == 1:
                    freq = rng.uniform(2, 8)
                    t = np.linspace(0, 1, n)
                    a = a * (1 + 0.02 * np.cos(2*np.pi*freq*t))
                elif strategy == 2:
                    a = a * (1 + rng.normal(0, 0.01, n))
                else:
                    # Swap mass between high and low regions
                    high_idx = np.argsort(a)[-n//10:]
                    low_idx = np.argsort(a)[:n//10]
                    amt = a[high_idx].sum() * 0.05
                    a[high_idx] -= amt / len(high_idx)
                    a[low_idx] += amt / len(low_idx)
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                no_improve = 0
                r = r_init * 0.5
        
        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez v2 iter={iteration} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")
    
    return best_a, best_m

def peak_shaving(a0, time_limit, verbose=False):
    """Directly target the top peaks in the convolution."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        M = b.max()
        
        # Find top peaks
        n_peaks = min(5, n // 10)
        peak_indices = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            pk_int = int(pk)
            
            # The peak at position pk is b[pk] = sum_j a[j]*a[pk-j]
            # To reduce it, we should decrease the products a[j]*a[pk-j]
            # The "most responsible" indices are where a[j]*a[pk-j] is largest
            
            j_start = max(0, pk_int - n + 1)
            j_end = min(n - 1, pk_int)
            
            if j_start > j_end:
                continue
            
            products = a[j_start:j_end+1] * a[pk_int - np.arange(j_start, j_end+1)]
            
            # Find indices with largest contributions
            n_contrib = max(2, (j_end - j_start + 1) // 4)
            top_contrib = np.argsort(products)[::-1][:n_contrib]
            
            # Try to reduce by transferring mass from high-contributing to low-contributing indices
            for _ in range(10):
                if time.monotonic() - t0 > time_limit:
                    break
                    
                i_idx = j_start + int(top_contrib[int(rng.integers(n_contrib))])
                # Choose a random target that doesn't contribute to this peak
                all_indices = np.setdiff1d(np.arange(n), np.arange(j_start, j_end+1))
                if len(all_indices) == 0:
                    break
                j_idx = int(all_indices[int(rng.integers(len(all_indices)))])
                
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[i_idx] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i_idx] -= amt
                    a_trial[j_idx] += amt
                    if a_trial[i_idx] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    a_trial = a_trial / s
                    m_trial = fast_conv(a_trial).max()
                    if m_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = m_trial
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            # Perturb
            idx = rng.choice(n, size=min(n//8, 300), replace=False)
            a[idx] *= rng.uniform(0.85, 1.15, len(idx))
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            if fast_conv(a).max() < best_m:
                best_m = fast_conv(a).max()
                best_a = a.copy()
    
    return best_a, best_m

def coordinate_descent_v2(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with binary search for optimal transfer amount."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 80)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_moves = min(200, n // 2)
        
        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            
            # Binary search for the best fraction to transfer from i to j
            lo, hi = 0.0, min(1.0, a[i] / (a[i] + a[j] + 1e-15))
            
            # Try several fractions including the binary search result
            fractions = [0.001, 0.005, 0.01, 0.05, 0.1, 0.2, 0.5]
            best_frac = None
            best_trial_m = best_m
            
            for frac in fractions:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                a_trial = a_trial / s
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_frac = frac
            
            if best_frac is not None and best_trial_m < best_m - 1e-16:
                amt = a[i] * best_frac
                a[i] -= amt
                a[j] += amt
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
        
        if not improved:
            stall += 1
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.05)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'bell':
        a = np.exp(-10.0 * (x - 0.5) ** 2)
    elif kind == 'U_shape':
        a = (2.0 * x - 1.0) ** 2 + 0.1
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s
