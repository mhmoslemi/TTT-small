       it+=1
       b=fast_conv(a); cur_m=float(np.max(b)); M=cur_m
       if cur_m < best_m - 1e-15: best_m=cur_m; best=a.copy()
       t_solve=time.monotonic()

def smooth_gradient_from_weights(a, p):
    n = len(a)
    if n == 0:
       return np.zeros(0)
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    need = 3*n - 2
    L = 1 << (need - 1).bit_length()
    Ar = np.fft.rfft(a, L)
    Pr = np.fft.rfft(p[::-1], L)
    conv = np.fft.irfft(Ar * Pr, L)
    g = conv[n-1:2*n-1][::-1]
    return 2.0 * g

"""
Searches for a non-negative step function minimizing
    2*n*max_j conv(a,a)_j / (sum a)^2.

Main ingredients:
- FFT-linear-convolution evaluation and full convolution-gradient computation
  for the smoothed max objective logsumexp(beta*conv(a,a)).
- Mirror-descent / multiplicative-weight optimization in the simplex.
- Active-set linearized Remez steps as LPs (SciPy HiGHS, CVXPY fallback).
- Cheap global-ish transforms: interpolation to other lengths, trimming,
  smoothing, reversal symmetrization.
- Repeated exact validation of the best candidates, with fallback safety.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

try:
    import cvxpy as cp
except Exception:
    cp = None

MAX_LEN = 7000


def fast_conv(a):
    """Linear convolution a * a via FFT, returned as a real non-negative array."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    # Correct tiny negative FFT roundoff; exact convolution is nonnegative.
    np.maximum(out, 0.0, out=out)
    return out


def norm_sum1(a):
    """Clip to [0,1000], force finite, normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def interpolate_sequence(a_base, n_target):
    """Linearly resample a normalized sequence to a new length."""
    a_base = np.asarray(a_base, dtype=np.float64).reshape(-1)
    n_target = int(n_target)
    if n_target <= 0:
        return np.ones(1, dtype=np.float64)
    if a_base.size == 0:
        return np.ones(n_target, dtype=np.float64) / float(n_target)
    a_base = norm_sum1(a_base)
    if a_base.size == n_target:
        return a_base
    x_old = np.linspace(0.0, 1.0, a_base.size)
    x_new = np.linspace(0.0, 1.0, n_target)
    out = np.interp(x_new, x_old, a_base)
    out = np.maximum(out, 0.0)
    return norm_sum1(out)


def make_init(n, kind="quartic"):
    """Smooth nonnegative initialization."""
    n = int(n)
    x = np.linspace(0.0, 1.0, n)
    t = 2.0 * x - 1.0
    if kind == "triangle":
        a = np.maximum(1.0 - np.abs(t), 0.0)
    elif kind == "raised_cos":
        a = 0.5 * (1.0 + np.cos(np.pi * t))
    elif kind == "gaussian":
        a = np.exp(-8.0 * t * t)
    elif kind == "bump":
        a = np.exp(-25.0 * t * t)
    else:  # quartic bump, good smooth tapered edges
        a = 1.0 - np.abs(t) ** 4
    a = np.maximum(a, 1e-12)
    return norm_sum1(a)

        return float(evaluate_sequence(vals))
    except Exception:

    """Active-set size for LP Remez, scaled mildly with length."""
    return int(max(40, min(230, int(0.045 * float(n)))))


def smooth_gradient_from_weights(a, p):
    """
    Gradient of sum_k p_k conv(a,a)_k.

    If p_k >= 0 sum to 1, this is also the gradient of the smoothed max
    logsumexp(beta*conv(a,a))/beta, up to the constant offset.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    # Need full convolution of a (length n) and reversed p (length 2n-1).
    need = 3 * n - 2
    L = 1 << (need - 1).bit_length()
    Ar = np.fft.rfft(a, L)
    Pr = np.fft.rfft(p[::-1], L)
    conv = np.fft.irfft(Ar * Pr, L)
    # h_j = sum_k p_k a_{k-j}, j=0..n-1, is conv[n-1 : 2n-1] reversed.
    g = conv[n - 1:2 * n - 1][::-1]
    return 2.0 * g


def mirror_candidate(y, grad, alpha):
    """Multiplicative-weight candidate proportional to y * exp(-alpha grad)."""
    y = np.asarray(y, dtype=np.float64).reshape(-1)
    grad = np.asarray(grad, dtype=np.float64).reshape(-1)
    logy = np.log(np.maximum(y, 1e-300))
    z = logy - alpha * grad
    nm = float(np.max(z))
    if not np.isfinite(nm):
        z = -alpha * grad
        nm = float(np.max(z))
        if not np.isfinite(nm):
            return y.copy()
    z = z - nm
    z = np.clip(z, -700.0, 700.0)
    out = np.exp(z)
    s = float(out.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return y.copy()
    out /= s
    np.maximum(out, 0.0, out=out)
    return out


def smooth_mirror_descent(a0, time_limit, beta_start=None, beta_end=None,
                          target_init=0.7, verbose=False):
    """
    Smoothed-max mirror descent with Nesterov-style extrapolation.
    The objective is logsumexp(beta * conv(a,a)) / beta, so low beta flatters,
    and high beta tracks the maximum lag.
    """
    a0 = norm_sum1(a0)
    n = a0.size
    best = a0.copy()
    best_m = float(np.max(fast_conv(best)))
    if time_limit <= 0.0:
        return best, best_m

    scale = max(1, n) / 2600.0
    if beta_start is None:
        beta_start = 300.0 * scale
    if beta_end is None:
        beta_end = 12000.0 * scale
    beta_end = max(beta_end, 1.1 * beta_start)

    t0 = time.monotonic()
    rng = np.random.default_rng(12345 + int(n))
    x = a0.copy()
    xprev = a0.copy()
    x_m = float(np.max(fast_conv(x)))
    target = float(target_init)
    stall = 0
    max_stall = 800
    it = 0
    last_print = -1e9

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(float(time_limit), 1e-9))
        beta = beta_start + (beta_end - beta_start) * progress

        # Nesterov extrapolation in the simplex.
        y = x + ((float(it) - 1.0) / (float(it) + 1.5)) * (x - xprev)
        y = np.maximum(y, 1e-15)
        sy = float(y.sum())
        if sy <= 1e-15 or not np.isfinite(sy):
            y = a0.copy()
        else:
            y = y / sy

        b = fast_conv(y)
        M = float(np.max(b))
        old_best = best_m
        if M < best_m - 1e-15:
            best_m = M
            best = y.copy()

        # Softmax weights over convolution lags.
        z = beta * (b - M)
        p = np.exp(z)
        sp = float(p.sum())
        if sp <= 0.0 or not np.isfinite(sp):
            p = np.ones_like(b, dtype=np.float64) / float(b.size)
        else:
            p /= sp

        grad = smooth_gradient_from_weights(y, p)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        gmax = float(np.max(np.abs(grad)))
        if not np.isfinite(gmax) or gmax < 1e-18:
            break

        alpha = min(1000.0, max(1e-12, target / gmax))

        # Short line search in mirror step size: full, half, quarter if needed.
        x_new = mirror_candidate(y, grad, alpha)
        Mx = float(np.max(fast_conv(x_new)))
        base_m = M if M < x_m else x_m
        if Mx > base_m + 1e-12 and alpha > 1e-9:
            x_half = mirror_candidate(y, grad, 0.5 * alpha)
            Mh = float(np.max(fast_conv(x_half)))
            if Mh < Mx:
                x_new = x_half
                Mx = Mh
        if Mx > base_m + 1e-12 and alpha * 0.25 > 1e-9:
            x_quarter = mirror_candidate(y, grad, 0.25 * alpha)
            Mq = float(np.max(fast_conv(x_quarter)))
            if Mq < Mx:
                x_new = x_quarter
                Mx = Mq

        xprev = x
        x = x_new
        x_m = Mx

        if x_m < best_m - 1e-15:
            best_m = x_m
            best = x.copy()

        if best_m < old_best - 1e-15:
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            max_stall = max(200, int(max_stall * 0.7))
            u = float(rng.random())
            if u < 0.4:
                target *= 0.6
            elif u < 0.7:
                target *= 1.4
            else:
                target = 0.25
            target = float(np.clip(target, 0.02, 2.0))

            strategy = int(rng.integers(3))
            if strategy == 0:
                xr = best * (1.0 + rng.normal(0.0, 0.01, n))
            elif strategy == 1:
                xr = best.copy()
                cnt = min(max(1, n // 10), 400)
                idx = rng.choice(n, cnt, replace=False)
                xr[idx] = xr[idx] * rng.uniform(0.5, 1.6, cnt)
            else:
                xr = best * (1.0 + rng.normal(0.0, 0.005, n))

            xr = np.maximum(xr, 1e-12)
            sxr = float(xr.sum())
            if sxr > 1e-15:
                xr /= sxr
            else:
                xr = best.copy()

            x = xr
            xprev = xr
            x_m = float(np.max(fast_conv(x)))
            it = 0

        if verbose and elapsed > last_print + 20.0 and it > 0:
            print(f"        SMD it={it} elapsed={elapsed:.1f}s "
                  f"score~{2.0 * n * best_m:.8f} beta={beta:.1f} target={target:.3f}")
            last_print = elapsed

    return best, best_m


def select_active_lags(b, M, max_cons):
    """Active convolution lags: top lags plus lags very close to the max."""
    b = np.asarray(b, dtype=np.float64).reshape(-1)
    M = float(M)
    ncoef = b.size
    max_cons = int(max(1, max_cons))
    ktop = min(max_cons, ncoef)
    if ktop >= ncoef:
        top = np.arange(ncoef, dtype=int)
    else:
        top = np.argpartition(b, -ktop)[-ktop:].astype(int)

    cap = max(2 * max_cons, 50)
    threshold = M - max(1e-12, 0.003 * M)
    if M > 0.0 and threshold > 0.0:
        near = np.where(b >= threshold)[0]
        if near.size > cap:
            near = near[np.argsort(b[near])[::-1][:cap]]
    else:
        near = np.array([], dtype=int)

    active = np.unique(np.concatenate((top, near)))
    if active.size > cap:
        active = active[np.argsort(b[active])[::-1][:cap]]
    return np.sort(active.astype(int))


def lp_remez_single_cvxpy(a, max_cons, r, b=None, M=None):
    """CVXPY fallback for one linearized active-set LP Remez step."""
    if cp is None:
        return None
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return None
    if b is None:
        b = fast_conv(a)
        M = float(np.max(b))
    else:
        b = np.asarray(b, dtype=np.float64).reshape(-1)
        M = float(M)
    if not np.isfinite(M) or M <= 0.0:
        return a.copy()

    active = select_active_lags(b, M, max_cons)
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [
        cp.sum(a_var) == 1,
        M_var >= 0,
        M_var <= M * 1.001 + 1e-9,
    ]
    delta = float(r) * max(float(np.max(a)), 1.0 / float(n))
    constraints.append(a_var <= a + delta)
    constraints.append(a_var >= np.maximum(0.0, a - delta))

    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        coeff = np.zeros(n, dtype=np.float64)
        if js <= je:
            length = je - js + 1
            coeff[js:je + 1] = 2.0 * a[k - (js + np.arange(length))]
        constraints.append(cp.sum(cp.multiply(coeff, a_var)) - M_var <= b[k])

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve()
    except Exception:
        return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    val = a_var.value
    if val is None:
        return None
    val = np.asarray(val, dtype=np.float64).reshape(-1)
    val = np.where(np.isfinite(val), val, 0.0)
    val = np.maximum(val, 0.0)
    s = float(val.sum())
    if s <= 1e-15:
        return None
    return val / s


def lp_remez_single(a, max_cons, r, b=None, M=None):
    """
    One linearized Remez LP:
      minimize M
      s.t. sum(a)=1, a in local box,
           linearized conv_k(a') <= M + conv_k(a)
    for selected active lags k.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return None
    if b is None:
        b = fast_conv(a)
        M = float(np.max(b))
    else:
        b = np.asarray(b, dtype=np.float64).reshape(-1)
        M = float(M)
    if not np.isfinite(M) or M <= 0.0:
        return a.copy()

    if linprog is not None:
        active = select_active_lags(b, M, max_cons)
        C = active.size
        A_ub = np.zeros((C, n + 1), dtype=np.float64)
        b_ub = np.empty(C, dtype=np.float64)
        for idx, k in enumerate(active):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                length = je - js + 1
                A_ub[idx, js:je + 1] = 2.0 * a[k - (js + np.arange(length))]
            A_ub[idx, n] = -1.0
            b_ub[idx] = b[k]

        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)

        delta = float(r) * max(float(np.max(a)), 1.0 / float(n))
        lower = np.maximum(0.0, a - delta)
        upper = np.minimum(1.0, a + delta)
        lower = np.minimum(lower, a + 1e-12)
        upper = np.maximum(upper, a - 1e-12)
        lower[n] = 0.0
        upper[n] = M * 1.001 + 1e-10
        if upper[n] < lower[n]:
            upper[n] = lower[n] + 1e-10
        bounds = np.column_stack((lower, upper))
        c = np.zeros(n + 1, dtype=np.float64)
        c[n] = 1.0

        res = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method="highs", options={"disp": False})
        except Exception:
            try:
                res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds)
            except Exception:
                res = None

        if res is not None and getattr(res, "status", None) == 0 and res.x is not None:
            a_new = np.asarray(res.x[:n], dtype=np.float64)
            a_new = np.where(np.isfinite(a_new), a_new, 0.0)
            a_new = np.maximum(a_new, 0.0)
            s = float(a_new.sum())
            if s > 1e-15:
                return a_new / s

    return lp_remez_single_cvxpy(a, max_cons, r, b, M)


def lp_remez_loop(a0, time_limit, max_cons=None, verbose=False):
    """Iterative linearized Remez with adaptive trust region and line search."""
    a = norm_sum1(a0)
    n = a.size
    if max_cons is None:
        max_cons = default_max_cons(n)
    best = a.copy()
    best_m = float(np.max(fast_conv(best)))
    if time_limit <= 0.0:
        return best, best_m

    t0 = time.monotonic()
    r = 0.1
    no_imp = 0
    mc = int(max_cons)
    it = 0
    rng = np.random.default_rng(777 + int(n))

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_conv(a)
        cur_m = float(np.max(b))
        if cur_m < best_m - 1e-15:
            best_m = cur_m
            best = a.copy()

        t_solve = time.monotonic()
        a_lp = lp_remez_single(a, mc, r, b, cur_m)
        solve_time = time.monotonic() - t_solve
        if solve_time > 8.0:
            mc = max(20, int(mc * 0.75))

        if a_lp is None:
            no_imp += 1
            r = min(5.0, r * 1.5)
            if no_imp > 30:
                break
            continue

        # Line-search along a -> a_lp. Evaluate a few convex combinations.
        best_local_m = cur_m
        best_local_a = a
        fractions = [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]
        if r > 2.0:
            fractions.append(1.2)

        for frac in fractions:
            cand = (1.0 - frac) * a + frac * a_lp
            cand = np.maximum(cand, 0.0)
            s = float(cand.sum())
            if s <= 1e-15:
                continue
            cand /= s
            Mc = float(np.max(fast_conv(cand)))
            if Mc < best_local_m - 1e-15:
                best_local_m = Mc
                best_local_a = cand

        if best_local_m < cur_m - 1e-14:
            a = best_local_a
            if best_local_m < best_m - 1e-15:
                best_m = best_local_m
                best = a.copy()
            no_imp = 0
            r = min(1.0, max(0.05, r * 1.05))
            if it % 20 == 0:
                mc = min(250, mc + 1)
        else:
            no_imp += 1
            if no_imp % 2 == 0:
                r = min(5.0, r * 1.5)
            else:
                r = max(0.02, r * 0.7)

            if no_imp in (6, 12, 20, 30):
                cnt = min(max(1, n // 8), 500)
                idx = rng.choice(n, cnt, replace=False)
                a[idx] = a[idx] * rng.uniform(0.6, 1.4, cnt)
                a = np.maximum(a, 0.0)
                s = float(a.sum())
                if s > 1e-15:
                    a /= s
                no_imp = 0

        if verbose and it % 50 == 0:
            print(f"        LP it={it} elapsed={time.monotonic()-t0:.1f}s "
                  f"score~{2.0 * n * best_m:.8f} r={r:.3f} mc={mc}")

    return best, best_m


def consider_candidate_seq(cand, best_score, best_seq, label=""):
    """Evaluate a candidate by FFT and update best if better."""
    cand = np.asarray(cand, dtype=np.float64).reshape(-1)
    if cand.size == 0 or cand.size > MAX_LEN:
        return best_score, best_seq
    cand = np.where(np.isfinite(cand), cand, 0.0)
    cand = np.maximum(cand, 0.0)
    s = float(cand.sum())
    if s <= 1e-12:
        return best_score, best_seq
    cand = cand / s
    M = float(np.max(fast_conv(cand)))
    sc = 2.0 * float(cand.size) * M + 1e-10
    if np.isfinite(sc) and sc < best_score:
        print(f"  >>> {label}: n={cand.size} score~{sc:.8f}")
        return sc, cand
    return best_score, best_seq


def apply_simple_transforms(a, best_score, best_seq):
    """Cheap candidate generation: trims, reversal symmetrization, smoothing."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size

    trims = [1, 2, 3, 5, 8, 12, 20, 30, 50]
    for k in trims:
        if n > 2 * k + 10:
            best_score, best_seq = consider_candidate_seq(
                a[k:n - k], best_score, best_seq, f"trim_both_{k}")
        if n > k + 10:
            best_score, best_seq = consider_candidate_seq(
                a[k:], best_score, best_seq, f"trim_left_{k}")
            best_score, best_seq = consider_candidate_seq(
                a[:-k], best_score, best_seq, f"trim_right_{k}")

    if n > 5:
        best_score, best_seq = consider_candidate_seq(
            0.5 * (a + a[::-1]), best_score, best_seq, "sym_rev")

    for m in (3, 5, 7, 9):
        if n > m:
            kernel = np.ones(m, dtype=np.float64) / float(m)
            cand = np.convolve(a, kernel, mode="same")
            best_score, best_seq = consider_candidate_seq(
                cand, best_score, best_seq, f"smooth_{m}")

    return best_score, best_seq


def scan_interpolated_lengths(a_base, best_score, best_seq, max_len=MAX_LEN):
    """Test several lengths by interpolation from current shape."""
    a_base = np.asarray(a_base, dtype=np.float64).reshape(-1)
    base_n = a_base.size
    candidates = set()
    for f in (0.5, 0.75, 1.0, 1.1, 1.25, 1.5, 2.0):
        candidates.add(int(base_n * f))
    candidates.update([2000, 3000, 4000, 5000, 6500, 7000])

    for n in sorted(candidates):
        if n < 100 or n > max_len or n == base_n:
            continue
        cand = interpolate_sequence(a_base, n)
        best_score, best_seq = consider_candidate_seq(
            cand, best_score, best_seq, f"interp_{n}")

    return best_score, best_seq


def polish_sequence(a0, time_limit, max_lp_cons=None, verbose=False):
    """Balanced polish: SMD, LP Remez, finer SMD, final LP Remez."""
    a0 = norm_sum1(a0)
    n = a0.size
    if max_lp_cons is None:
        max_lp_cons = default_max_cons(n)
    best = a0.copy()
    best_m = float(np.max(fast_conv(best)))
    if time_limit <= 0.0:
        return best, best_m

    t0 = time.monotonic()

    t1 = max(0.0, time_limit * 0.32)
    t1 = min(t1, max(0.0, time_limit - 2.0))
    if t1 > 5.0:
        a, m = smooth_mirror_descent(a0, t1, verbose=verbose)
        if m < best_m:
            best_m = m
            best = a

    t_left = time_limit - (time.monotonic() - t0)
    t2 = max(0.0, t_left * 0.50)
    t2 = min(t2, max(0.0, t_left - 2.0))
    if t2 > 5.0:
        a, m = lp_remez_loop(best, t2, max_cons=max_lp_cons, verbose=verbose)
        if m < best_m:
            best_m = m
            best = a

    t_left = time_limit - (time.monotonic() - t0)
    t3 = max(0.0, t_left * 0.30)
    t3 = min(t3, max(0.0, t_left - 2.0))
    if t3 > 3.0:
        scale = max(1, n) / 2600.0
        a, m = smooth_mirror_descent(best, t3,
                                     beta_start=3500.0 * scale,
                                     beta_end=25000.0 * scale,
                                     target_init=0.25,
                                     verbose=verbose)
        if m < best_m:
            best_m = m
            best = a

    t_left = time_limit - (time.monotonic() - t0)
    if t_left > 3.0:
        a, m = lp_remez_loop(best, t_left,
                             max_cons=min(250, max(max_lp_cons + 40,
                                                   default_max_cons(n) + 40)),
                             verbose=verbose)
        if m < best_m:
            best_m = m
            best = a

    return best, best_m


def explore_length(a_base, n_target, time_limit, max_lp_cons=None, verbose=False):
    """Interpolate to a target length, then polish at that length."""
    n_target = int(n_target)
    if n_target < 100 or n_target > MAX_LEN or time_limit <= 0.0:
        return None

    a = interpolate_sequence(a_base, n_target)
    best = a.copy()
    best_m = float(np.max(fast_conv(best)))
    if max_lp_cons is None:
        max_lp_cons = default_max_cons(n_target)

    t0 = time.monotonic()
    scale = float(n_target) / 2600.0

    t1 = max(0.0, time_limit * 0.45)
    t1 = min(t1, max(0.0, time_limit - 2.0))
    if t1 > 5.0:
        a, m = smooth_mirror_descent(a, t1,
                                     beta_start=600.0 * scale,
                                     beta_end=15000.0 * scale,
                                     target_init=0.5,
                                     verbose=verbose)
        if m < best_m:
            best_m = m
            best = a

    t_left = time_limit - (time.monotonic() - t0)
    if t_left > 5.0:
        a, m = lp_remez_loop(best, t_left, max_cons=max_lp_cons, verbose=verbose)
        if m < best_m:
            best_m = m
            best = a

    return best, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    budget = min(float(budget_s) - 45.0, 1770.0)
    if budget <= 10.0:
        return [1.0]

    best_seq = None
    best_score = float("inf")
    exact_best_seq = None
    exact_best_score = float("inf")

    # Load parent if available.
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and p.sum() > 1e-10:
            if p.size > MAX_LEN:
                p = interpolate_sequence(p, 4200)
            best_seq = p / p.sum()
            b = fast_conv(best_seq)
            best_score = 2.0 * float(len(best_seq)) * float(np.max(b)) + 1e-10
            print(f"Parent loaded: n={len(best_seq)}, approx={best_score:.8f}")
            exact = evaluate_safe(best_seq)
            if np.isfinite(exact):
                exact_best_seq = best_seq.copy()
                exact_best_score = exact
                if exact < best_score:
                    best_score = exact
    except Exception:
        print("No usable parent; using default initialization.")

    if best_seq is None:
        best_seq = make_init(2600, "quartic")
        b = fast_conv(best_seq)
        best_score = 2.0 * float(len(best_seq)) * float(np.max(b)) + 1e-10
        print(f"Default init: n={len(best_seq)}, approx={best_score:.8f}")

    if exact_best_seq is None:
        exact = evaluate_safe(best_seq)
        if np.isfinite(exact):
            exact_best_seq = best_seq.copy()
            exact_best_score = exact

    # Low-cost transformations and length scan before heavy polishing.
    best_score, best_seq = apply_simple_transforms(best_seq, best_score, best_seq)
    best_score, best_seq = scan_interpolated_lengths(best_seq, best_score, best_seq)

    exact = evaluate_safe(best_seq)
    if np.isfinite(exact) and exact < exact_best_score:
        exact_best_seq = best_seq.copy()
        exact_best_score = exact
        if exact < best_score:
            best_score = exact

    # Phase 1: strong polish of current best shape.
    rem = budget - (time.monotonic() - t_global)
    t1 = min(rem * 0.45, 900.0)
    if t1 > 60.0:
        a, _ = polish_sequence(best_seq, t1,
                               max_lp_cons=default_max_cons(len(best_seq)),
                               verbose=True)
        best_score, best_seq = consider_candidate_seq(a, best_score, best_seq, "phase1")
        best_score, best_seq = apply_simple_transforms(best_seq, best_score, best_seq)

        exact = evaluate_safe(best_seq)
        if np.isfinite(exact) and exact < exact_best_score:
            exact_best_seq = best_seq.copy()
            exact_best_score = exact
            if exact < best_score:
                best_score = exact

    # Phase 2: explore nearby / larger lengths.
    rem = budget - (time.monotonic() - t_global)
    if rem > 160.0:
        base_n = int(len(best_seq))
        raw = []
        for f in (1.1, 1.25, 1.5, 1.8, 2.0):
            raw.append(int(base_n * f))
        raw.extend([base_n + 300, base_n + 800, base_n + 1500])
        raw.extend([4000, 5500, 7000])
        lengths = []
        for n in sorted(set(raw)):
            if 150 <= n <= MAX_LEN and n != base_n and n not in lengths:
                lengths.append(n)

        max_count = 4 if rem > 700.0 else 3
        lengths = lengths[:max_count]

        if lengths:
            print(f"Length exploration: {lengths}, remaining={rem:.0f}s")
            each = max(45.0, rem / float(len(lengths)))
            for n in lengths:
                rem = budget - (time.monotonic() - t_global)
                if rem < 80.0:
                    break
                t = min(each, rem - 25.0)
                if t < 35.0:
                    break
                res = explore_length(best_seq, n, t,
                                     max_lp_cons=default_max_cons(n),
                                     verbose=True)
                if res is not None:
                    a, _ = res
                    best_score, best_seq = consider_candidate_seq(
                        a, best_score, best_seq, f"explore_{n}")
                best_score, best_seq = apply_simple_transforms(
                    best_seq, best_score, best_seq)

            exact = evaluate_safe(best_seq)
            if np.isfinite(exact) and exact < exact_best_score:
                exact_best_seq = best_seq.copy()
                exact_best_score = exact
                if exact < best_score:
                    best_score = exact

    # Phase 3: final focused polish on validated/global best.
    rem = budget - (time.monotonic() - t_global)
    if rem > 30.0:
        t_smd = min(rem * 0.18, 120.0)
        if t_smd > 10.0:
            scale = max(1, len(best_seq)) / 2600.0
            a, _ = smooth_mirror_descent(best_seq, t_smd,
                                         beta_start=6000.0 * scale,
                                         beta_end=35000.0 * scale,
                                         target_init=0.15,
                                         verbose=True)
            best_score, best_seq = consider_candidate_seq(
                a, best_score, best_seq, "final_smd")
            best_score, best_seq = apply_simple_transforms(
                best_seq, best_score, best_seq)

        rem = budget - (time.monotonic() - t_global)
        if rem > 15.0:
            maxc = min(250, max(150, default_max_cons(len(best_seq)) + 80))
            a, _ = lp_remez_loop(best_seq, rem - 5.0,
                                 max_cons=maxc, verbose=True)
            best_score, best_seq = consider_candidate_seq(
                a, best_score, best_seq, "final_lp")
            best_score, best_seq = apply_simple_transforms(
                best_seq, best_score, best_seq)

    # Final strict evaluation and safe fallback.
    final_exact = evaluate_safe(best_seq)
    if np.isfinite(final_exact) and final_exact < exact_best_score:
        exact_best_seq = best_seq.copy()
        exact_best_score = final_exact

    if exact_best_seq is not None:
        final_seq = exact_best_seq
    else:
        final_seq = best_seq

    final_seq = np.asarray(final_seq, dtype=np.float64).reshape(-1)
    final_seq = np.where(np.isfinite(final_seq), final_seq, 0.0)
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = np.ones(2600, dtype=np.float64) / 2600.0
    else:
        final_seq = final_seq / s

    print(f"FINAL approx={best_score:.8f}, validated_exact={exact_best_score:.8f}, "
          f"n={len(final_seq)}, time={time.monotonic() - t_global:.1f}s")
    return [float(x) for x in final_seq]
