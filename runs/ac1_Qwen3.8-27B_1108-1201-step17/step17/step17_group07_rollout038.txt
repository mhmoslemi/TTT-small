"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed mirror descent with Nesterov acceleration and FFT-based gradient (O(n log n)/iter)
2. Spectral flattening in frequency domain (flatten power spectrum to reduce autocorrelation peaks)
3. LP Remez with trust region for final polishing
4. Multi-length exploration and diverse initializations
Key insight: The gradient of the LSE-smoothed max of b[k] can be computed in O(n log n) via FFT,

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):

def compute_lse_gradient_fft(a, b, tau):
    """Compute gradient of LSE-smoothed max(b) w.r.t. a using FFT. O(n log n).
    g[j] = 2 * sum_{l=0}^{n-1} w[j+l] * a[l] where w[k] = exp(tau*b[k])/sum(exp(tau*b))
    """
    n = len(a)
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)

    scaled = tau * b / M
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # g[j] = 2 * sum_{l=0}^{n-1} w[j+l] * a[l]
    # Using convolution: w_flipped = w[::-1], then (w_flipped * a)[2n-2-j] = inner product
    w_flipped = w[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()

    W = np.fft.rfft(w_flipped, fft_size)
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(W * A, fft_size)

    g = 2.0 * result[n - 1:2 * n - 1][::-1]
    return g

    x_prev = a.copy()
    stall_count = 0

    grad_norm_ema = 1.0
    eta_base = 0.5
    n_iter_max = 50000000

    for t in range(1, n_iter_max):

        progress = min(t / 50000.0, 1.0)
        tau = tau_start * (tau_end / tau_start) ** progress

        # Nesterov extrapolation
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = x.copy()
            y_sum = x.sum()

        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        # Evaluate
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

            stall_count = 0

            strategy = int(rng.integers(4))
            if strategy == 0:
                x_r = best_a * (1 + rng.normal(0, 0.02, n))

                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n // 4, 400), replace=False)

                phases = np.angle(fft_x)

                x_r = np.real(np.fft.irfft(magnitudes * np.exp(1j * phases), n * 4))[:n]
            else:

            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

def spectral_flattening(a0, time_limit, alpha_start=0.3, alpha_end=0.05, verbose=False):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 100

    fft_size = 1 << ((2 * n - 1) - 1).bit_length()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(t / 1000.0, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        # Compute DFT
        A = np.fft.rfft(a, fft_size)
        magnitudes = np.abs(A)
        phases = np.angle(A)

        # Flatten: new_mag = mag^(1-alpha), which reduces dynamic range
        max_mag = magnitudes.max()
        if max_mag < 1e-30:
            break
        normalized_mag = magnitudes / max_mag
        new_mag = normalized_mag ** (1.0 - alpha)
        new_mag = new_mag * max_mag

        # Reconstruct
        A_new = new_mag * np.exp(1j * phases)
        a_new = np.real(np.fft.irfft(A_new, fft_size))[:n]

        # Clip and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-18:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Small perturbation to escape
            if stall > 20:
                idx = rng.choice(n, size=min(n // 10, 200), replace=False)
                a[idx] *= rng.uniform(0.5, 2.0, len(idx))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                stall = 0

        if stall > max_stall:
            break

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Spectral t={t} ({elapsed:.1f}s): J={J:.8f}, alpha={alpha:.4f}")

    return best_a, best_m


def lp_remez_single(a, max_cons=50, r=0.1, solver=None):
    """Single LP Remez step."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()

    threshold = max_b * 0.998
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1]

    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])

    constraints.append(a_var <= a * (1.0 + r))
    constraints.append(a_var >= a * max(0.0, 1.0 - r))

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with random pair moves."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 50)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(50, n // 5)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue

            for frac in [0.002, 0.005, 0.01, 0.03, 0.08, 0.15]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'bump':
        a = np.exp(-20.0 * (x - 0.3) ** 2) + np.exp(-20.0 * (x - 0.7) ** 2)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining LSE-NMD, spectral flattening, Remez, and CD."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    n_base = 2600
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: LSE-NMD from parent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: LSE-NMD n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = lse_nesterov_mirror_descent(norm_sum1(best_seq), t1, tau_start=100.0, tau_end=20.0, verbose=True)
        try_update(a_out, f"LSE-NMD_n{n0}")

    # === Phase 2: Spectral flattening (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: Spectral flattening n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = spectral_flattening(norm_sum1(best_seq), t2, alpha_start=0.4, alpha_end=0.05, verbose=True)
        try_update(a_out, f"Spectral_n{len(best_seq)}")

    # === Phase 3: LSE-NMD from best (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: LSE-NMD polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = lse_nesterov_mirror_descent(norm_sum1(best_seq), t3, tau_start=50.0, tau_end=10.0, verbose=True)
        try_update(a_out, f"LSE-NMD_polish_n{len(best_seq)}")

    # === Phase 4: LP Remez (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: LP Remez n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = remez_lp_method(norm_sum1(best_seq), t4, solver=solver, r_init=0.1, verbose=True)
        try_update(a_out, f"Remez_n{len(best_seq)}")

    # === Phase 5: Multi-init LSE-NMD (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5_total = min(remaining * 0.10, 150.0)
    if t5_total > 50:
        print(f"\n=== Phase 5: Multi-init LSE-NMD (budget={t5_total:.0f}s) ===")
        inits = ['raised_cos', 'gaussian', 'bump']
        t_per = t5_total / len(inits)
        for kind in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            a_init = make_init(n0, kind)
            a_out, _ = lse_nesterov_mirror_descent(a_init, min(t_per, remaining - 5), tau_start=60.0, tau_end=15.0, verbose=False)
            try_update(a_out, f"LSE-{kind}_n{n0}")

    # === Phase 6: CD polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 6: CD polish n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t6, verbose=False, rng_seed=999)
        try_update(a_out, f"CD_{len(best_seq)}")

    # === Phase 7: Alternate length exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n0 + 100
        print(f"\n=== Phase 7: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_alt = interpolate_sequence(best_seq, n_alt)

        t7a = min(remaining * 0.6, 200.0)
        if t7a > 30:
            a_out, _ = lse_nesterov_mirror_descent(a_alt, t7a, tau_start=80.0, tau_end=15.0, verbose=True)
            try_update(a_out, f"LSE-NMD_n{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.4, 100.0)
            if t7b > 15:
                a_out, _ = remez_lp_method(norm_sum1(best_seq), t7b, solver=solver, r_init=0.05, verbose=False)
                try_update(a_out, f"Remez_alt_{len(best_seq)}")

    # === Phase 8: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final polish n={n_best} (budget={remaining:.0f}s) ===")
        t8a = min(remaining * 0.4, 80.0)
        if t8a > 10:
            a_out, _ = lse_nesterov_mirror_descent(norm_sum1(best_seq), t8a, tau_start=30.0, tau_end=8.0, verbose=False)
            try_update(a_out, f"LSE_final_{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.4, 50.0)
            if t8b > 10:
                a_out, _ = spectral_flattening(norm_sum1(best_seq), t8b, alpha_start=0.2, alpha_end=0.03, verbose=False)
                try_update(a_out, f"Spectral_final_{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = remez_lp_method(norm_sum1(best_seq), remaining - 5, solver=solver, r_init=0.02, verbose=False)
            try_update(a_out, f"Remez_last_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
