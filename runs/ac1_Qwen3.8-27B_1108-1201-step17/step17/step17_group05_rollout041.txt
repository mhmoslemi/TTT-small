def fast_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """conv(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def compute_subgrad_topk(a, top_k=20, use_exp_weights=True, beta=4.0):
    """
    Compute subgradient using top-K peak lags.
    g[m] = 2 * sum_k w_k * a[k-m] (valid range)
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]

    if use_exp_weights and k_max > 1:
        shifted = top_vals - top_vals.max()
        w_top = np.exp(beta * shifted)
        w_sum = w_top.sum()
        if w_sum < 1e-30:
            w_top = np.ones(k_max) / k_max
        else:
            w_top = w_top / w_sum
    else:
        w_top = np.ones(k_max) / k_max

    # Build sparse weight vector
    w = np.zeros(len(ac))
    for wi, idx in zip(w_top, top_idx):
        w[int(idx)] = wi

    # g[m] = 2 * sum_k w[k] * a[k-m] for valid k-m
    # = 2 * (a[::-1] * w)[m + n - 1] in the full conv
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv_result = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad, ac.max(), top_idx

def lp_frank_wolfe_optimize(a0, time_limit, solver, verbose=True,

    LP-based Frank-Wolfe with adaptive constraint selection and Nesterov acceleration.

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    prev_y = a.copy()

    for t in range(1, n_iters_max + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-18:
            break

        frac = min(1.0, t / 500.0)
        K = int(K_init + frac * (K_max - K_init))
        K = min(K, 2 * n - 1)

        # Select top-K most binding constraints
        top_idx = np.argsort(ac)[::-1][:K]

        # Build constraint matrix A where A[i, j] = y[top_idx[i] - j] if valid
        A_mat = np.zeros((K, n))
        for i, m in enumerate(top_idx):

                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(cp.ones(n) @ b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}")

    return best_a, best_m


def coordinate_descent_optimize(a0, time_limit, verbose=True, seed=123):
    """
    Coordinate descent: for each peak lag, find best single-coordinate move.
    Uses exact 1D optimization via golden section search on the affected lag value.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:10]
        improvement = False

        for k in top_lags[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js_max = max(0, ki - n + 1)
            js_min = min(n - 1, ki)
            if js_max > js_min:
                continue
            idx = np.arange(js_max, js_min + 1)
            partners = ki - idx
            contrib = a[idx] * a[partners]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try moving mass to various destinations
            # Score destinations by how much they would DECREASE the peak
            # For destination d, the change in C[k'] is approximately:
            # -src_val * a[k'-src] + amt * a[k'-d] for each peak k'
            sens = np.zeros(n)
            for k2 in top_lags[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 > e2:
                    continue
                for d in range(s2, e2 + 1):
                    sens[d] += a[k2i - d]

            # Best destinations: low sensitivity, far from source
            dist = np.abs(np.arange(n) - src).astype(float)
            score = sens / (1.0 + dist / (n * 0.1))
            candidates = np.argsort(score)[:20]
            candidates = [d for d in candidates if d != src]

            for dst in candidates[:8]:
                for frac in [0.01, 0.03, 0.05, 0.02, 0.08, 0.015]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improvement = True
                        break
                if improvement:
                    break
            if improvement:
                stall = 0
                break

        if not improvement:
            stall += 1
            if stall % 15 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.04)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def make_exponential_init(n, seed=0, asymmetry=0.5):
    """Create exponential (non-symmetric) initialization."""
    rng = np.random.default_rng(seed)
    m = np.arange(n, dtype=np.float64)
    # Exponential tilt: more mass on one side
    rate = rng.uniform(1.0, 3.0) * asymmetry
    a = np.exp(rate * m / n)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_ramp_init(n, seed=0, power=2.0):
    """Create power-law ramp initialization."""
    m = np.arange(n, dtype=np.float64) + 1.0
    a = m ** power
    return a / a.sum()


def make_tilted_uniform_init(n, seed=0, tilt=0.3):
    """Create tilted uniform initialization (nearly uniform with slight non-symmetry)."""
    rng = np.random.default_rng(seed)
    m = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 1.0 + tilt * (2.0 * m - 1.0)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_quadratic_init(n, seed=0):
    """Create quadratic bowl initialization (more mass at edges)."""
    m = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = (m - 0.5) ** 2 + 0.1
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_blend_init(n, seed=0):
    """Blend of different initializations."""
    rng = np.random.default_rng(seed)
    a1 = make_exponential_init(n, seed=seed)
    a2 = make_ramp_init(n, power=rng.uniform(1.0, 4.0))
    a3 = make_tilted_uniform_init(n, seed=seed + 1)
    w1 = rng.dirichlet(np.ones(3) * 2)
    a = w1[0] * a1 + w1[1] * a2 + w1[2] * a3
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with multi-phase optimization.
    Phases:
    1. Coarse mirror descent with diverse inits
    2. Fine LP-FW refinement
    3. Mirror descent polish
    4. Coordinate descent
    5. Length exploration
    6. Final LP polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_tilted_uniform_init(n_base, seed=1, tilt=0.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Mirror Descent with diverse inits ===
    print(f"\n=== Phase 1: Mirror Descent (diverse inits) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 550.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_tilted_uniform_init(n0, seed=5, tilt=0.6))
        labels.append("tilt06")

        inits.append(make_exponential_init(n0, seed=10, asymmetry=0.8))
        labels.append("exp")

        inits.append(make_ramp_init(n0, power=3.0))
        labels.append("ramp3")

        inits.append(make_blend_init(n0, seed=20))
        labels.append("blend")

        inits.append(make_quadratic_init(n0, seed=30))
        labels.append("quad")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  MD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = mirror_descent_optimize(init, t_avail, verbose=False,
                                                beta_init=6.0, beta_min=2.0,
                                                c0=8.0, top_k_init=3,
                                                top_k_max=30, restart_every=3000,
                                                seed=100 + idx)
