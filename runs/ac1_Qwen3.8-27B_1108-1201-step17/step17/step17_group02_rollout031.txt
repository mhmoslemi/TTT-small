1. Parametric seeding with power-law and exponential forms

3. Fine-grained coordinate descent for final polish

import numpy as np
import time

def fast_autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    max_b = ac.max()
    return 2.0 * n * max_b / (s * s)

def subgradient(a, ac, top_k=None):
    """Compute subgradient of max_k (a*a)[k] at point a."""
    n = len(a)
    if top_k is None:
        top_k = 1
    # Find top peaks
    peak_vals = ac.max()
    # Get indices where ac is close to max
    threshold = peak_vals * (1 - 1e-8)
    peak_idx = np.where(ac >= threshold)[0]
    if len(peak_idx) == 0:
        peak_idx = [np.argmax(ac)]
    
    grad = np.zeros(n)
    for k in peak_idx[:top_k]:
        ki = int(k)
        # (a*a)[ki] = sum_i a[i] * a[ki-i]
        # d/d a[j]: a[ki-j] if 0 <= ki-j < n
        s = max(0, ki - n + 1)
        e = min(n - 1, ki)
        if s > e:
            continue
        idx = np.arange(s, e + 1)
        partner = ki - idx
        grad[idx] += a[partner]
    
    # Normalize by number of active constraints
    if len(peak_idx) > 1:
        grad /= len(peak_idx[:top_k])
    
    return grad

def nesterov_amd(a0, time_limit, verbose=True, rng_seed=42, c0=3.0, top_k=5):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_score = compute_objective(a)

    x = a.copy()  # current iterate
    y = a.copy()  # Nesterov lookahead
    x_prev = a.copy()
    
    stall = 0
    best_t = 0
    max_stall = 5000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov step
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute gradient
        ac = fast_autocorr(y)
        g = subgradient(y, ac, top_k=top_k)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_update = np.log(y) - eta * g
        max_log = log_update.max()
        a_new = np.exp(log_update - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new

        if score < best_score - 1e-12:
            best_score = score
            best_a = x.copy()
            stall = 0
            best_t = t

            stall += 1

        if stall > max_stall:

            n_pert = max(3, n // 20)

            y = x_new.copy()

    return best_a, best_score
