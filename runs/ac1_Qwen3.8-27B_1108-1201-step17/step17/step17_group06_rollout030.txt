import numpy as np

from scipy.optimize import minimize

def fast_conv(a):
    """FFT-based full linear convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax_max(x, beta):
    """Smooth approximation of max(x) using softmax."""
    x_max = x.max()
    exp_x = np.exp(beta * (x - x_max))
    return x_max + np.log(exp_x.sum()) / beta

def softmax_weights(x, beta):
    """Compute softmax weights for the smooth max."""
    x_max = x.max()
    exp_x = np.exp(beta * (x - x_max))
    return exp_x / exp_x.sum()

def smooth_objective(x, beta, n):
    """Compute the smooth objective and its gradient in x-space."""
    # x is the log-parameter, a is the actual sequence
    x_shifted = x - x.max()  # for numerical stability
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    # Compute conv(a, a)
    conv_a = fast_conv(a)
    
    # Compute softmax weights
    p = softmax_weights(conv_a, beta)

    grad_x = grad_a * a - a * dot
    
    return obj, grad_x

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Beta-annealed smooth optimization with L-BFGS-B (FFT-based gradients)
2. LP-guided Frank-Wolfe direction finding
3. Nesterov mirror descent with adaptive subgradients
4. Coordinate descent with pressure-based mass transfer
5. Length sweeping and final polish

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(p, a):
    """Compute c[i] = sum_k p[k] * a[k-i] for i=0,...,len(a)-1 using FFT."""
    n_p = len(p)
    n_a = len(a)
    p_rev = p[::-1]
    size = n_p + n_a - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_full = np.fft.irfft(P * A, fft_size)[:size]
    c = conv_full[2 * n_a - 2 : n_a - 2 : -1]
    return c


def softmax_max(x, beta):
    """Smooth approximation of max(x)."""
    x_max = x.max()
    exp_x = np.exp(beta * (x - x_max))
    return x_max + np.log(exp_x.sum()) / beta


def softmax_weights(x, beta):
    """Compute softmax weights for smooth max."""
    x_max = x.max()
    exp_x = np.exp(beta * (x - x_max))
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def log_param_to_a(x):
    """Convert log-parameters to normalized non-negative sequence."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def a_to_log_param(a):
    """Convert normalized sequence to log-parameters."""
    return np.log(np.maximum(a, 1e-30))


def smooth_objective_and_grad(x, beta, n):
    """Compute smooth objective and gradient w.r.t. log-parameters x."""
    a = log_param_to_a(x)
    conv_a = fast_conv(a)
    p = softmax_weights(conv_a, beta)
    smooth_max_val = softmax_max(conv_a, beta)
    obj = 2.0 * n * smooth_max_val
    c = fast_cross_corr(p, a)
    grad_a = 4.0 * n * c
    dot = np.dot(grad_a, a)
    grad_x = grad_a * a - a * dot
    return obj, grad_x


def smooth_obj_only(x, beta, n):
    """Objective only for scipy optimizer."""
    a = log_param_to_a(x)
    conv_a = fast_conv(a)
    return 2.0 * n * softmax_max(conv_a, beta)


def smooth_grad_only(x, beta, n):
    """Gradient only for scipy optimizer."""
    a = log_param_to_a(x)
    conv_a = fast_conv(a)
    p = softmax_weights(conv_a, beta)
    c = fast_cross_corr(p, a)
    grad_a = 4.0 * n * c
    dot = np.dot(grad_a, a)
    return grad_a * a - a * dot


def smooth_optimize(a0, time_limit, beta_start=5.0, beta_end=300.0, n_stages=10, verbose=True):
    """Beta-annealed smooth optimization using L-BFGS-B."""
    n = len(a0)
    x = a_to_log_param(a0)
    x = x - x.mean()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_J = eval_J(a0)
    betas = np.geomspace(beta_start, beta_end, n_stages)
    for idx, beta in enumerate(betas):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 3.0:
            break
        stop_flag = [False]

        def callback(xk):
            if time.monotonic() - t0 > time_limit - 1.0:
                stop_flag[0] = True
            return stop_flag[0]

        def fun(xx):
            return smooth_obj_only(xx, beta, n)

        def gfun(xx):
            return smooth_grad_only(xx, beta, n)

        try:
            result = minimize(fun, x, jac=gfun, method='L-BFGS-B',
                              options={'maxiter': 200, 'ftol': 1e-14,
                                       'gtol': 1e-10, 'maxfun': 5000},
                              callback=callback)
            if result.x is not None:
                x_new = result.x - result.x.mean()
                a_new = log_param_to_a(x_new)
                J_new = eval_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                x = x_new
                if verbose:
                    print(f"    Smooth stage {idx+1}/{n_stages} beta={beta:.1f}: J={best_J:.6f} (stop={stop_flag[0]})")
        except Exception as e:
            if verbose:
                print(f"    Smooth stage {idx+1} error: {e}")
    return best_a, best_J


def lp_guided_step(a, M, top_k=80):
    """Solve LP to find descent direction. Returns (direction, lp_sum) or (None, None)."""
    n = len(a)
    conv = fast_conv(a)
    top_idx = np.argsort(conv)[::-1][:top_k]
    top_idx = np.array(sorted([int(k) for k in top_idx]))
    K = len(top_idx)
    A_ub = np.zeros((K, n))
    for row, k in enumerate(top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            cols = np.arange(i_start, i_end + 1)
            A_ub[row, cols] = a[k_int - cols]
    c = -np.ones(n)
    b_ub = M * np.ones(K)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            b = res.x
            s = b.sum()
            if s > 1e-15:
                return b / s, s
    except Exception:
        pass
    return None, None


def frank_wolfe_lp(a0, time_limit, top_k=80, n_max_iters=500, verbose=True):
    """Frank-Wolfe with LP directions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    stall = 0
    t_fw = 1.0
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit - 2.0:
            break
        conv = fast_conv(a)
        M = conv.max()
        direction, lp_sum = lp_guided_step(a, M, top_k=top_k)
        if direction is None:
            stall += 1
            if stall > 20:
                break
            continue
        t_step = 2.0 / (t_fw + 2.0)
        a_new = (1.0 - t_step) * a + t_step * direction
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        J_new = eval_J(a_new)
        if J_new < best_J - 1e-15:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
        t_fw += 1.0
        if verbose and it % 50 == 0:
            print(f"    FW-LP iter {it}: J={best_J:.6f}, lp_sum={lp_sum:.4f}, t={time.monotonic()-t0:.0f}s")
    return best_a, best_J


def compute_subgradient(a, conv, top_k=100, beta_w=10.0):
    """Compute soft subgradient of max(conv) at a."""
    n = len(a)
    k_max = min(top_k, 2 * n - 1)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta_w
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g


def mirror_descent(a0, time_limit, top_k=100, beta_w=10.0, c0=4.0,
                   restart_every=5000, verbose=False):
    """Nesterov-accelerated mirror descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta_w
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        conv = fast_conv(y)
        g = compute_subgradient(y, conv, top_k=cur_top_k, beta_w=cur_beta)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        val = eval_J(a_new)
        if val < best_val - 1e-16:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.005, 0.02)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 20, 250)
            cur_beta = max(cur_beta - 1.0, 3.0)
            c0 = max(c0 * 0.92, 1.0)
    return best_a, best_val


def coordinate_descent(a0, time_limit, verbose=False, seed=42):
    """Coordinate descent with pressure-based mass transfer."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    while time.monotonic() - t0 < time_limit - 2.0 and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()
        n_lags = min(30, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_lags]
        pressure = np.zeros(n)
        for k in top_lags[:12]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                pressure[idx] += a[idx] * a[k_i - idx]
        valid = pressure * a > 1e-20
        if not valid.any():
            stall += 1
            break
        src_scores = np.where(valid, pressure * a, 0.0)
        src = int(np.argmax(src_scores))
        if a[src] < 1e-12:
            stall += 1
            continue
        dest_scores = np.zeros(n)
        for k in top_lags[:6]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                dest_scores[idx] += a[k_i - idx]
        dest_scores[src] = np.inf
        dest_scores += 0.1 * pressure
        dst = int(np.argmin(dest_scores))
        improved = False
        for frac in [0.003, 0.008, 0.015, 0.03, 0.06, 0.1, 0.2, 0.35]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            if a_t[src] < -1e-15:
                continue
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-16:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall % 20 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.002, 0.03)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break
    return best_a, best_J


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=60.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.15) ** 2) + np.exp(-width * (x - 0.85) ** 2) + 0.003
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 12):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent loaded: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3000
        best_a = make_power_ramp(n0, 1.0)
        best_score = eval_J(best_a)
        print(f"Initial (no parent): n={len(best_a)}, J={best_score:.6f}")

    n0 = len(best_a)
    print(f"\nInitial best: J={best_score:.6f}, n={n0}")

    # === Phase 1: Smooth optimization (main workhorse) ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t1 = min(remaining * 0.35, 600.0)
    print(f"\n=== Phase 1: Smooth Optimize (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = smooth_optimize(best_a.copy(), t1, beta_start=5.0,
                                   beta_end=300.0, n_stages=12, verbose=True)
        try_update(a_out, "smooth1")

    # Also try smooth opt from a different init
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t1b = min(remaining * 0.10, 150.0)
    if t1b > 20:
        init2 = 0.5 * best_a / best_a.sum() + 0.5 * make_bimodal(n0)
        init2 = init2 / init2.sum()
        a_out, _ = smooth_optimize(init2, t1b, beta_start=3.0,
                                   beta_end=200.0, n_stages=6, verbose=True)
        try_update(a_out, "smooth_bimodal")

    # === Phase 2: LP-guided Frank-Wolfe ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: FW-LP (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = frank_wolfe_lp(best_a.copy(), t2, top_k=100,
                                  n_max_iters=300, verbose=True)
        try_update(a_out, "fw_lp")

    # === Phase 3: Smooth opt again (refinement) ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t3 = min(remaining * 0.18, 300.0)
    print(f"\n=== Phase 3: Smooth Refine (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = smooth_optimize(best_a.copy(), t3, beta_start=50.0,
                                   beta_end=500.0, n_stages=8, verbose=True)
        try_update(a_out, "smooth_refine")

    # === Phase 4: Mirror descent ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Mirror Descent (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = mirror_descent(best_a.copy(), t4, top_k=120, beta_w=8.0,
                                  c0=3.5, restart_every=6000, verbose=True)
        try_update(a_out, "md")

    # === Phase 5: Coordinate descent ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    t5 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 5: Coord Descent (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = coordinate_descent(best_a.copy(), t5, verbose=True, seed=99)
        try_update(a_out, "coord")

    # === Phase 6: Length sweep ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    print(f"\n=== Phase 6: Length Sweep (budget={remaining:.0f}s) ===")
    n_best = len(best_a)
    lengths_to_try = []
    for d in [-300, -200, -100, 100, 200, 300, 500]:
        nl = n_best + d
        if 1500 <= nl <= 5000:
            lengths_to_try.append(nl)
    for nl in sorted(set(lengths_to_try)):
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        if remaining < 40:
            break
        if abs(nl - n_best) <= 50:
            continue
        print(f"  Length sweep: n={nl}...")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.25, 80.0)
        if t_md > 15:
            a_out, _ = smooth_optimize(a_int, t_md, beta_start=10.0,
                                       beta_end=200.0, n_stages=5, verbose=False)
            try_update(a_out, f"len{nl}_smooth")
        elapsed = time.monotonic() - t_global
        remaining = time_budget - elapsed
        if remaining > 20:
            t_cd = min(remaining * 0.10, 40.0)
            if t_cd > 10:
                a_cur = interpolate_seq(best_a, nl)
                a_out, _ = coordinate_descent(a_cur, t_cd, verbose=False, seed=nl)
                try_update(a_out, f"len{nl}_cd")

    # === Phase 7: Final polish ===
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    n_cur = len(best_a)
    print(f"\n=== Phase 7: Final Polish (budget={remaining:.0f}s, n={n_cur}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.40, 200.0)
        a_out, _ = smooth_optimize(best_a.copy(), t7, beta_start=80.0,
                                   beta_end=600.0, n_stages=6, verbose=True)
        try_update(a_out, "final_smooth")
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining > 20:
        t7b = min(remaining * 0.30, 100.0)
        a_out, _ = mirror_descent(best_a.copy(), t7b, top_k=150, beta_w=5.0,
                                  c0=2.5, restart_every=8000, verbose=False)
        try_update(a_out, "final_md")
    elapsed = time.monotonic() - t_global
    remaining = time_budget - elapsed
    if remaining > 15:
        t7c = min(remaining * 0.40, 80.0)
        a_out, _ = coordinate_descent(best_a.copy(), t7c, verbose=False, seed=271)
        try_update(a_out, "final_coord")

    # Final output
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        # Validate using evaluate_sequence
        result = [float(max(0.0, min(1000.0, x))) for x in best_a]
        eval_result = evaluate_sequence(result)
        print(f"Validation: evaluate_sequence = {eval_result:.6f}")
        return result
    else:
        n_fb = 3000
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
