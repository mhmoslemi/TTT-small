def lp_fw_lazy(a, M, solver, max_rounds=50):
    """Solve the LP subproblem with lazy constraint generation."""
    n = len(a)
    # Start with a few constraints (most binding for a*a)
    ac = fast_autocorr(a)
    top_k = min(100, 2*n-1)
    top_indices = np.argsort(ac)[::-1][:top_k]
    
    active_constraints = list(top_indices)
    
    for round_i in range(max_rounds):
        # Build constraint matrix for active constraints
        # (a*b)[k] = sum_j a[j] * b[k-j] for valid j
        # = sum_j b[j] * a[k-j] (by symmetry of the relation)
        # Actually: (a*b)[k] = sum_j a[j] * b[k-j]
        # For fixed k: (a*b)[k] = sum_{j: 0<=k-j<n} a[j] * b[k-j]
        #            = sum_{j: max(0,k-n+1) <= j <= min(k, n-1)} a[j] * b[k-j]
        # Let i = k-j, so j = k-i:
        # = sum_{i: max(0, k-n+1) <= k-i <= min(k, n-1)} a[k-i] * b[i]
        # = sum_{i: k-min(k,n-1) <= i <= k-max(0,k-n+1)} a[k-i] * b[i]
        # = sum_{i: max(0, k-n+1) <= i <= min(k, n-1)} a[k-i] * b[i]
        
        # So the constraint matrix T[k, i] = a[k-i] if 0 <= k-i < n, else 0
        # which means for constraint k: T[k, i] = a[k-i] for i in [max(0,k-n+1), min(k,n-1)]
        
        K = len(active_constraints)
        T = np.zeros((K, n))
        for col_i, k in enumerate(active_constraints):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            idx = np.arange(i_start, i_end + 1)
            T[col_i, idx] = a[k_int - idx]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M * np.ones(K),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=solver, verbose=False)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if b_val.sum() < 1e-15:
            return None
        b_val = b_val / b_val.sum()
        
        # Check if any inactive constraint is violated
        ab = fast_conv(a, b_val)
        violations = np.where(ab > M * (1 + 1e-6))[0]
        if len(violations) == 0:
            return b_val
        # Add violated constraints
        active_constraints.extend([int(v) for v in violations[:50]])
    
    return b_val

def fast_autocorr(a):
    """Compute conv(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv(a, b):
    """Compute conv(a,b) via FFT."""
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

        K = len(active)

        for col_i, k in enumerate(active):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            idx = np.arange(i_start, i_end + 1)
            T[col_i, idx] = a[k_int - idx]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M * np.ones(K),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=solver, verbose=False)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if b_val.sum() < 1e-15:
            return None
        b_val = b_val / b_val.sum()
        
        # Check violated constraints
        ab = fast_conv(a, b_val)
        viol = np.where(ab > M * (1 + 1e-8) + 1e-12)[0]
        if len(viol) == 0:
            return b_val
        # Add violated constraints (up to 50 per round)
        new_constr = [int(v) for v in viol[:50] if int(v) not in set(active)]
        if not new_constr:
            return b_val
        active.extend(new_constr)
    
    return b_val


def nesterov_fw(a0, time_limit, solver, verbose=True, nesterov=True, top_k_init=200):
    """
    Nesterov-accelerated Frank-Wolfe with LP oracle and lazy constraints.
    """
    n = len(a0)
    x = norm_sum1(a0).copy()
    y = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    n_iter = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        n_iter = t
        
        # Compute current peak
        ac = fast_autocorr(y)
        M = ac.max()
        
        # LP solve to get direction
        s = lp_fw_solve(y, M, solver, max_rounds=30, initial_top_k=top_k_init)
        if s is None:
            # Fallback: use negative subgradient projected to simplex
            g = compute_subgrad(y, beta=5.0)
            s = norm_sum1(np.exp(-g))
        
        # Step size
        alpha = 2.0 / (t + 2.0)
        
        # Update
        x_new = (1.0 - alpha) * x + alpha * s
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new / sx
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
        y_new = x_new + beta_t * (x_new - x)
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = x_new.copy()
        else:
            y_new = y_new / sy
        
        prev_x = x
        x = x_new
        
        # Track best
        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-14:
            best_m = mx
            best_a = x.copy()
        
        my = fast_autocorr(y).max()
        if my < best_m - 1e-14:
            best_m = my
            best_a = y.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NFW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def compute_subgrad(a, beta=5.0):
    """Compute soft subgradient of max(conv(a,a))."""
    n = len(a)
    ac = fast_autocorr(a)
    shifted = ac - ac.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(ac) / len(ac)
    else:
        w = w / ws
    w[w < 1e-12] = 0.0
    if w.sum() < 1e-15:
        w = np.ones_like(ac) / len(ac)
        w = w / w.sum()
    
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def coord_descent(a0, time_limit, verbose=True, seed=42):
    """
    Coordinate descent with pairwise mass transfers targeting peak lags.
    """
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    n = len(a)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M = ac.max()
        top_lags = np.argsort(ac)[::-1][:min(20, len(ac))]
        
        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            # Find the pair (i, j) with i + j = k (or k - n + 1 + (n-1-i) = k, etc.)
            # For lag k: c[k] = sum_i a[i] * a[k-i] where 0 <= i < n and 0 <= k-i < n
            # The most contributing pair is argmax_i a[i]*a[k-i]
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            max_c = contrib.max()
            if max_c < 1e-20:
                continue
            best_i = idx[np.argmax(contrib)]
            
            # Try to redistribute mass from best_i to reduce this contribution
            # Move mass from i to some position i' that's far from the complementary position
            comp = k_int - best_i
            # We want to move mass from best_i to a position that doesn't create
            # new peaks. Try positions far from comp.
            dist_to_comp = np.abs(np.arange(n) - comp).astype(float)
            candidates = np.argsort(dist_to_comp)[::-1][:50]  # farthest from comp
            candidates = [c for c in candidates if c != best_i]
            
            for dst in candidates[:20]:
                for frac in [0.01, 0.03, 0.05]:
                    amt = a[best_i] * frac
                    if amt < 1e-15 or a[best_i] < 1e-10:
                        break
                    a_trial = a.copy()
                    a_trial[best_i] -= amt
                    a_trial[dst] += amt
                    a_trial = norm_sum1(a_trial)
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.005, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = norm_sum1(a)
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
            
            if verbose and stall % 100 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def spectral_smooth(a, sigma=0.5):
    """Apply spectral smoothing (low-pass filter in Fourier domain)."""
    n = len(a)
    f = np.fft.rfft(a)
    freqs = np.fft.rfftfreq(n)
    # Gaussian low-pass
    mask = np.exp(-2.0 * (np.pi * sigma * freqs * n / n) ** 2)
    # Actually, let's use a simpler cutoff
    f[mask < 0.1] *= mask[mask < 0.1]
    result = np.fft.irfft(f, n)
    return norm_sum1(result)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated Frank-Wolfe with LP oracle (subgradient objective) and lazy constraint generation
2. Strongly-spherical step variant with backtracking
3. Adaptive coordinate descent with targeted peak-lag reduction
4. Multi-length exploration via interpolation
Key: LP-FW with lazy constraints is much faster than full-constraint LP,
allowing far more iterations within the time budget.
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv(a, b):
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def eval_obj(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return float(2.0 * n * ac.max() / (s * s))


def compute_subgrad(a, beta=5.0):
    n = len(a)
    ac = fast_autocorr(a)
    shifted = ac - ac.max()
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones_like(ac) / len(ac)
    else:
        w = w / ws
    w[w < 1e-12] = 0.0
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.ones_like(ac) / len(ac)
        w = w / w.sum()
    else:
        w = w / w_sum
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def lp_direction(a, M, solver, g, max_rounds=40, top_k=200, epsilon=1e-12):
    n = len(a)
    ac = fast_autocorr(a)
    n_constr = 2 * n - 1
    top_idx = np.argsort(ac)[::-1][:min(top_k, n_constr)]
    active = [int(x) for x in top_idx]
    active_set = set(active)
    rhs = M + epsilon

    for _ in range(max_rounds):
        K = len(active)
        T = np.zeros((K, n))
        for col_i, k in enumerate(active):
            k_int = k
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            idx = np.arange(i_start, i_end + 1)
            T[col_i, idx] = a[k_int - idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= rhs * np.ones(K),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(g @ b_var), constraints)
        prob.solve(solver=solver, verbose=False)

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if b_val.sum() < 1e-15:
            return None
        b_val = b_val / b_val.sum()

        ab = fast_conv(a, b_val)
        viol_mask = ab > rhs + 1e-10
        if not viol_mask.any():
            return b_val
        viol_idx = np.where(viol_mask)[0]
        new_constr = [int(v) for v in viol_idx[:50] if int(v) not in active_set]
        if not new_constr:
            return b_val
        active.extend(new_constr)
        active_set.update(new_constr)

    return b_val


def nesterov_fw(a0, time_limit, solver, top_k_init=200, use_backtrack=False, verbose=True):
    n = len(a0)
    x = norm_sum1(a0).copy()
    y = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    n_iter = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        n_iter = t

        ac = fast_autocorr(y)
        M = ac.max()

        g = compute_subgrad(y, beta=5.0)
        s_dir = lp_direction(y, M, solver, g, max_rounds=30, top_k=top_k_init)

        if s_dir is None:
            s_dir = norm_sum1(np.exp(-g - g.max()))

        if use_backtrack:
            alpha = 1.0
            for _bt in range(10):
                x_try = (1.0 - alpha) * x + alpha * s_dir
                x_try = norm_sum1(x_try)
                m_try = fast_autocorr(x_try).max()
                if m_try <= M + 1e-14:
                    break
                alpha *= 0.5
        else:
            alpha = 2.0 / (t + 2.0)

        x_new = (1.0 - alpha) * x + alpha * s_dir
        x_new = norm_sum1(x_new)

        beta_t = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
        y_new = x_new + beta_t * (x_new - x)
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = x_new.copy()
        else:
            y_new = y_new / sy

        prev_x = x.copy()
        x = x_new

        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-14:
            best_m = mx
            best_a = x.copy()

        my = fast_autocorr(y).max()
        if my < best_m - 1e-14:
            best_m = my
            best_a = y.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NFW iter={t} ({elapsed:.0f}s): J={J:.8f}")

    return best_a, best_m


def coord_descent(a0, time_limit, seed=42, verbose=True):
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    n = len(a)
    stall = 0
    max_stall = 800

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(15, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            best_i = idx[np.argmax(contrib)]
            comp = k_int - best_i

            dist_to_comp = np.abs(np.arange(n) - comp).astype(float)
            far_idx = np.argsort(dist_to_comp)[::-1][:30]
            candidates = [int(c) for c in far_idx if c != best_i]

            found = False
            for dst in candidates[:15]:
                for frac in [0.02, 0.05, 0.01, 0.08, 0.03]:
                    amt = a[best_i] * frac
                    if amt < 1e-15 or a[best_i] < 1e-10:
                        break
                    a_trial = a.copy()
                    a_trial[best_i] -= amt
                    a_trial[dst] += amt
                    a_trial = norm_sum1(a_trial)
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not improved:
            stall += 1
            if stall % 25 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.005, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = norm_sum1(a)
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

            if verbose and stall % 200 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Coord ({elapsed:.0f}s): J={J:.8f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord done ({elapsed:.0f}s): J={J:.8f}, stall={stall}")
    return best_a, best_m


def mirror_descent_polish(a0, time_limit, seed=0, c0=4.0, beta_init=6.0):
    n = len(a0)
    x = norm_sum1(a0).copy()
    x_prev = x.copy()
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        eta = c0 / (np.sqrt(it) + 1.0)
        beta = max(3.0, beta_init - 0.0005 * it)
        g = compute_subgrad(x, beta=beta)
        log_x = np.log(x + 1e-30)
        log_new = log_x - eta * g
        log_new -= log_new.max()
        x_new = np.exp(log_new)
        x_new = np.maximum(x_new, 1e-30)
        x_new = x_new / x_new.sum()

        x_prev = x.copy()
        x = x_new

        mx = fast_autocorr(x).max()
        if mx < best_m - 1e-14:
            best_m = mx
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.1, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

    return best_a, best_m


def interpolate_to_n(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a)
    return norm_sum1(a_interp)
