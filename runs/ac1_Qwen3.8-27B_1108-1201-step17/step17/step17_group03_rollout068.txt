"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 via a coordinated multi-phase search:

  1. Smooth mirror-descent FISTA on log-sum-exp(beta * autocorr) as a global
     optimizer / starting-point generator (annealed beta).
  2. LP Frank-Wolfe with the CORRECT linearization of the peak constraints
     T_sel[k] @ b <= (M + ac[k]) / 2  (previously 2M was used, which was loose
     for non-active lags), plus adaptive constraint growth and FISTA extrapolation.
  3. Peak-crushing coordinate descent for fast local polish.
  4. Multi-length interpolation exploration to escape length-local minima.

The LP is the workhorse; the corrected right-hand side makes the predicted-

import numpy as np
import time
import cvxpy as cp

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])

    print("  LP solver (fallback): ECOS")

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_parametric(n, family='sine'):
    x = np.linspace(0.0, 1.0, n)
    if family == 'sine':
        return norm_sum1(np.sin(np.pi * x) ** 2)
    if family == 'cosine':
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.02)
    if family == 'beta22':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    return norm_sum1(np.sin(np.pi * x) ** 2)


def fista_mirror(a0, time_limit, beta_start=8.0, beta_end=400.0,
                 n_anneal=14, verbose=True):
    """Mirror-descent FISTA on log-sum-exp(beta * ac). Annealed beta."""
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_a = x.copy()
    best_val = fast_autocorr(x).max()
    t0 = time.monotonic()
    betas = np.geomspace(beta_start, beta_end, n_anneal)
    t_iter = 0
    beta = float(beta_start)
    beta_idx = 0
    anneal_every = max(50, int(time_limit / (0.0005 * n * n_anneal)))
    anneal_every = max(anneal_every, 30)

    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        if t_iter % anneal_every == 0 and beta_idx < n_anneal - 1:
            beta_idx += 1
            beta = float(betas[beta_idx])
            if verbose and beta_idx % 2 == 0:
                print(f"    FISTA anneal t={t_iter} beta={beta:.0f} "
                      f"J={2*n*best_val:.6f}")

        beta_t = (t_iter - 1.0) / (t_iter + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        ac_shift = ac - ac.max()
        p = np.exp(beta * ac_shift)
        p = p / p.sum()

        # grad_j = 2 * sum_m y[m] * p[m+j]  via reversed convolution
        y_rev = y[::-1]
        N2 = 1 << (3 * n - 1 - 1).bit_length()
        YR = np.fft.rfft(y_rev, N2)
        Pp = np.fft.rfft(p, N2)
        convres = np.fft.irfft(YR * Pp, N2)
        grad = 2.0 * convres[n - 1:n - 1 + n]
        grad = np.maximum(grad, 0.0)

        eta = 1.0 / (2.0 * beta + 1.0)
        base_m = ac.max()
        accepted = False
        for _ in range(5):
            log_a = np.log(y) - eta * grad
            log_a = log_a - log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            eta *= 0.5
        if not accepted:
            log_a = np.log(y) - eta * grad
            log_a = log_a - log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_val - 1e-15:
            best_val = M_new
            best_a = x.copy()

        if verbose and t_iter % 8000 == 0:
            print(f"    FISTA t={t_iter} ({time.monotonic()-t0:.0f}s) "
                  f"J={2*n*best_val:.6f}")

    if verbose:
        print(f"    FISTA done ({time.monotonic()-t0:.0f}s) J={2*n*best_val:.6f}")
    return best_a, best_val


def lp_fw(a0, time_limit, solver, verbose=True, init_K=40, max_K=400,
          margin_frac=1e-3):
    """LP Frank-Wolfe with corrected linearization rhs=(M+ac[k])/2 and
    adaptive constraint growth + FISTA extrapolation."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    x = y.copy()

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margin = M_curr - ac
        thr = margin_frac * M_curr
        active = np.where(margin <= thr)[0]
        # adaptive K
        cur_K = min(init_K + (t // 40) * 8, max_K)
        if len(active) > cur_K * 3:
            sel = np.concatenate([active, np.argsort(margin)[:cur_K]])
        elif len(active) < 20:
            sel = np.argsort(margin)[:cur_K]
        else:
            sel = active
        sel = np.unique(sel)
        K = int(min(len(sel), max_K))
        sel = sel[:K]

        T_sel = np.zeros((K, n))
        rhs = np.zeros(K)
        for row, k_int in enumerate(sel):
            ki = int(k_int)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[ki - i]
            rhs[row] = 0.5 * (M_curr + ac[ki])

        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_sel @ b_var <= rhs])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            bt = (t - 1.0) / (t + 2.0)
            x_new = y_new + bt * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            margin_frac = max(margin_frac * 0.85, 1e-5)
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 4 == 0:
            print(f"    LP t={t} ({time.monotonic()-t0:.0f}s) "
                  f"J={2*n*best_m:.6f} K={K} m={margin_frac:.1e}")

    if verbose:
        print(f"    LP done ({time.monotonic()-t0:.0f}s) J={2*n*best_m:.6f}")
    return best_a, best_m


def peak_crush(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        # target the top few peaks, not just the argmax
        peaks = np.argsort(ac)[::-1][:5]
        improved = False
        for peak_k in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(peak_k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[ki - idx]
            total = contrib.sum()
            if total < 1e-20:
                continue
            n_top = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]
            for li in top_local:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                best_new_m = M_curr
                best_move = None
                for frac in [0.004, 0.01, 0.02, 0.04, 0.07]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    for _ in range(24):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_new_m:
                            best_new_m = M_trial
                            best_move = (src, dst, amt)
                if best_move is not None:
                    si, di, am = best_move
                    a[si] -= am
                    a[di] += am
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.004, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        if verbose and stall % 60 == 0:
            print(f"    PC ({time.monotonic()-t0:.0f}s) J={2*n*best_m:.6f} "
                  f"stall={stall}")
    if verbose:
        print(f"    PC done ({time.monotonic()-t0:.0f}s) J={2*n*best_m:.6f}")
    return best_a, best_m


def freq_perturb(a, strength=0.015, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A) + rng.normal(0, strength, len(np.angle(A)))
    A_new = np.abs(A) * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, "
              f"t={time.monotonic()-t_global:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score

    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3400
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def remaining():
        return total - (time.monotonic() - t_global)

    # Phase 1: FISTA mirror descent (global / diverse start)
    rem = remaining()
    t1 = min(rem * 0.08, 130.0)
    print(f"\n=== Phase 1: FISTA mirror (n={n0}, {t1:.0f}s) ===")
    if t1 > 40:
        a_out, _ = fista_mirror(norm_sum1(best_seq), t1, verbose=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                          t_global, "fista")

    # Phase 2: LP Frank-Wolfe main (corrected linearization)
    rem = remaining()
    n_best = len(best_seq)
    t2 = min(rem * 0.34, 620.0)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}, {t2:.0f}s) ===")
    if t2 > 40 and n_best <= 4800:
        a_out, _ = lp_fw(norm_sum1(best_seq), t2, solver, verbose=True,
                         init_K=40, max_K=400, margin_frac=2e-3)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                          t_global, "lp_main")

    # Phase 3: peak-crush polish
    rem = remaining()
    t3 = min(rem * 0.08, 150.0)
    print(f"\n=== Phase 3: PeakCrush (n={len(best_seq)}, {t3:.0f}s) ===")
    if t3 > 25:
        a_out, _ = peak_crush(norm_sum1(best_seq), t3, verbose=True,
                              rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                          t_global, "pc1")

    # Phase 4: tighter LP-FW round
    rem = remaining()
    n_best = len(best_seq)
    t4 = min(rem * 0.30, 560.0)
    print(f"\n=== Phase 4: LP-FW tight (n={n_best}, {t4:.0f}s) ===")
    if t4 > 40 and n_best <= 4800:
        a_out, _ = lp_fw(norm_sum1(best_seq), t4, solver, verbose=True,
                         init_K=60, max_K=500, margin_frac=5e-4)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                          t_global, "lp_tight")

    # Phase 5: multi-length exploration
    rem = remaining()
    n_best = len(best_seq)
    cands = sorted(set([max(1200, n_best - 300), n_best + 400,
                        n_best + 900, n_best + 1500]))
    cands = [c for c in cands if c <= 5200]
    rng = np.random.default_rng(7)
    for new_n in cands[:3]:
        rem = remaining()
        if rem < 120:
            break
        print(f"\n=== Phase 5: extend n={new_n} (rem {rem:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, new_n)
        t5 = min(rem * 0.22, 180.0)
        if t5 > 40 and new_n <= 4800:
            a_out, _ = lp_fw(a_long, t5, solver, verbose=False,
                             init_K=40, max_K=300, margin_frac=2e-3)
            best_seq, best_score = try_update(a_out, best_seq, best_score,
                                              t_global, f"lp_n{new_n}")
        rem = remaining()
        if rem < 60 and new_n <= 4000:
            a_p = freq_perturb(norm_sum1(best_seq), 0.02, rng)
            t5b = min(rem * 0.3, 90.0)
            if t5b > 30 and new_n <= 4000:
                a_out, _ = fista_mirror(a_p, t5b, verbose=False)
                best_seq, best_score = try_update(a_out, best_seq,
                                                  best_score, t_global,
                                                  f"fista_p{new_n}")

    # Phase 6: final LP-FW on absolute best
    rem = remaining()
    n_best = len(best_seq)
    t6 = min(rem * 0.22, 300.0)
    print(f"\n=== Phase 6: Final LP (n={n_best}, {t6:.0f}s) ===")
    if t6 > 40 and n_best <= 4800:
        a_out, _ = lp_fw(norm_sum1(best_seq), t6, solver, verbose=True,
                         init_K=60, max_K=520, margin_frac=2e-4)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                          t_global, "lp_final")

    # Phase 7: final peak-crush
    rem = remaining()
    t7 = min(rem * 0.25, 200.0)
    print(f"\n=== Phase 7: Final PC (n={len(best_seq)}, {t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = peak_crush(norm_sum1(best_seq), t7, verbose=True,
                              rng_seed=999)
        best_seq, best_score = try_update(a_out, best_seq, best_score,
                                          t_global, "pc_final")

    # Final validation + safety fallback
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
        return [float(x) for x in best_seq]

    print("ERROR: no valid sequence; using fallback")
    fb = make_parametric(3200, 'sine')
    return [float(x) for x in fb]
