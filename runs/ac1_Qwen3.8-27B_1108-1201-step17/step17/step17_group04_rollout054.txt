import numpy as np

from scipy.optimize import minimize as scipy_minimize

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

    Returns (gradient, lse_value).
    """
    n = len(a)
    ac = fast_autocorr(a)
    
    if top_k is not None and top_k < len(ac):
        # Only consider top-k peaks
        top_idx = np.argsort(ac)[::-1][:top_k]
        ac_sel = ac[top_idx]
    else:
        ac_sel = ac
        top_idx = np.arange(len(ac))
    
    # Log-sum-exp
    max_ac = ac_sel.max()
    logs = beta * (ac_sel - max_ac)
    logs = np.clip(logs, -50, 50)  # numerical stability
    log_sum = np.log(np.exp(logs).sum())
    lse_val = (max_ac + log_sum) / beta

    p = np.exp(logs) / np.exp(logs).sum()

        j_idx = k_int - idx

    return grad, lse_val

def adam_smooth_optimize(a0, time_limit, beta_start=10.0, beta_end=500.0,
                          top_k=50, lr=0.01, beta1=0.9, beta2=0.999,
                          eps=1e-8, verbose=False):
    """
    Adam optimizer on log-sum-exp smoothed peak.
    Works in log-space to maintain non-negativity.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    
    # Work in log-space
    log_a = np.log(np.maximum(a, 1e-15))
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        frac = min(1.0, t / 5000)
        beta = beta_start + (beta_end - beta_start) * frac
        # Decay lr
        lr_t = lr / (1 + 0.001 * t)
        
        # Compute gradient in log-space
        # d/d(log_a[i]) of LSE = a[i] * grad_conv[i]
        ac = fast_autocorr(a)
        if top_k is not None and top_k < len(ac):
            top_idx = np.argsort(ac)[::-1][:top_k]
            ac_sel = ac[top_idx]
        else:
            ac_sel = ac
            top_idx = np.arange(len(ac))
        
        max_ac = ac_sel.max()
        logs = beta * (ac_sel - max_ac)
        logs = np.clip(logs, -50, 50)
        log_sum = np.log(np.exp(logs).sum() + 1e-30)
        
        p = np.exp(logs - log_sum)
        
        grad_conv = np.zeros(n)
        for w, k in zip(p, top_idx):
            if w < 1e-12:
                continue
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad_conv[idx] += 2.0 * w * a[j_idx]
        
        # Gradient in log-space: dLSE/d(log_a[i]) = a[i] * grad_conv[i]
        grad_log = a * grad_conv
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * grad_log**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        log_a = log_a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back and normalize
        a = np.exp(log_a - log_a.max())
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    return best_a, best_J

def peak_shave(a0, time_limit, verbose=False, seed=42):
    """
    Direct peak shaving: identify peak and analytically compute optimal redistribution.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()

    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        k_peak = int(np.argmax(ac))
        peak_val = ac[k_peak]
        
        # Get contributors to the peak
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_peak - idx
        products = a[idx] * a[j_idx]
        
        # Find the pair (i, j) with the largest product
        max_prod_idx = np.argmax(products)
        i_src = idx[max_prod_idx]
        j_src = j_idx[max_prod_idx]
        
        # We want to reduce a[i_src] * a[j_src]
        # Either reduce a[i_src] or a[j_src] (or both)
        # Moving mass from i_src to some other position i_dst
        # The change in peak ≈ -a[j_src] * delta (if we move from i_src)
        # But this might increase other peaks
        
        # Find destination: position that minimizes increase in other peaks
        # Simplified: move to a position with low contribution to all top peaks
        
        # Get top peaks
        n_peaks = min(20, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]
        
        # Compute "danger" for each position: how much it contributes to top peaks
        danger = np.zeros(n)
        for k in top_lags:
            k_int = int(k)
            s = max(0, k_int - n + 1)
            e = min(n, k_int + 1)
            ii = np.arange(s, e)
            jj = k_int - ii
            danger[ii] += a[jj]
            danger[jj] += a[ii]
        
        # Move mass from the peak contributor to a low-danger position
        src_candidates = [i_src, j_src]
        # Pick the one with more mass
        src = max(src_candidates, key=lambda x: a[x])
        
        # Find best destination
        dst_candidates = np.where(danger < danger.mean())[0]
        if len(dst_candidates) == 0:
            dst_candidates = np.argsort(danger)[:n//10]
        
        # Try multiple amounts
        best_improvement = 0
        best_amount = 0
        best_dst = -1
        
        for dst in dst_candidates[rng.permutation(len(dst_candidates))[:20]]:
            if dst == src:
                continue
            for frac in [0.01, 0.02, 0.05, 0.08, 0.1, 0.15, 0.2]:
                amt = a[src] * frac
                if amt < 1e-10:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                improvement = best_J - J_trial
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_amount = amt
                    best_dst = dst
        
        if best_improvement > 1e-14:
            a[src] -= best_amount
            a[best_dst] += best_amount
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_J = compute_J(a)
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Perturb to escape
            n_pert = max(5, n // 200)
            positions = rng.choice(n, size=n_pert, replace=False)
            for p in positions:
                if a[p] > 1e-8:
                    amt = a[p] * rng.uniform(0.01, 0.1)
                    a[p] -= amt
                    a[int(rng.integers(n))] += amt
            a = np.maximum(a, 0)
            a = a / a.sum()
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
                stall = 0
    
    return best_a, best_J

def spectral_balance(a0, time_limit, verbose=False, seed=123):
    """
    Try to flatten the spectrum |F(a)|^2 to reduce time-domain peaks.
    Uses iterative approach in frequency domain.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    fft_size = 1 << (2*n - 1).bit_length()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A)**2
        power_sum = power.sum()
        
        if power_sum < 1e-20:
            break
        
        # Target: make power spectrum flat
        # Current power vs flat target
        flat_target = np.ones_like(power) * (power_sum / len(power))
        
        # Adjust: where power is high, reduce; where low, increase
        # Use a gentle correction
        alpha = 0.01
        ratio = np.minimum(2.0, np.maximum(0.5, flat_target / (power + 1e-20)))
        A_new = A * np.sqrt(ratio)
        
        # Add small random perturbation in frequency
        A_new += rng.normal(0, 0.001, size=A_new.shape) * np.abs(A_new).mean()
        
        # Transform back
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        J = compute_J(a_new)
        if J < best_J:
            best_J = J
            best_a = a_new.copy()
        else:
            # Keep old a, add more perturbation
            a = a_new  # still update to explore
    
    return best_a, best_J

def get_lp_solver():
    try:
        import cvxpy as cp
        for s in [cp.CLARABEL, cp.ECOS, cp.SCIP, cp.SCS]:
            try:
                x = cp.Variable(5, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
                prob.solve(solver=s, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    return s
            except Exception:
                continue
        return cp.ECOS
    except Exception:
        return None

def lp_direction(a, solver, top_K=300):
    """
    Solve LP to find feedback direction (Frank-Wolfe style).
    """
    if solver is None:
        return None
    import cvxpy as cp
    
    n = len(a)
    T_full = np.zeros((2*n-1, n))
    for j in range(n):
        T_full[j:j+n, j] = a
    
    ac = T_full @ a
    M = ac.max()
    
    # Select top-K constraints
    margins = M - ac
    K = min(top_K, 2*n-1)
    binding = np.argsort(margins)[:K]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_full[binding, :] @ b_var <= M + 1e-10]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    
    return b_val / b_sum

def frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300):
    """Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    y = a.copy()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        direction = lp_direction(y, solver, top_K)
        if direction is None:
            break
        
        alpha = 2.0 / (t + 2.0)
        y = (1 - alpha) * y + alpha * direction
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        if t == 1:
            x = y.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x = y + beta_t * (y - a)
            x = np.maximum(x, 0)
            x = x / x.sum()
        
        a = y  # for next LP
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter={t} ({elapsed:.0f}s): best_M={best_m:.8f}, J={2*n*best_m:.6f}")
    
    return best_a, best_m

2. Vectorized peak shaving with analytical optimal transfer amounts
3. LP Frank-Wolfe with Nesterov acceleration for periodic directional updates

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_topk_weights(ac, top_k, beta):
    """Compute softmax weights over top-k peaks."""
    if top_k >= len(ac):
        top_idx = np.arange(len(ac))
    else:
        if top_k == 1:
            top_idx = np.array([np.argmax(ac)])
        else:
            top_idx = np.argpartition(ac, -top_k)[-top_k:]
            top_idx = top_idx[np.argsort(ac[top_idx])[::-1]]
    
    ac_sel = ac[top_idx]
    max_ac = ac_sel.max()
    exp_vals = np.exp(beta * (ac_sel - max_ac))
    weights = exp_vals / (exp_vals.sum() + 1e-30)
    return top_idx, weights


def compute_gradient(a, top_idx, weights):
    """Compute weighted subgradient of the convolution peak."""
    n = len(a)
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        if w < 1e-12:
            continue
        k_int = int(k)
        s = max(0, k_int - n + 1)
        e = min(n, k_int + 1)
        ii = np.arange(s, e)
        jj = k_int - ii
        grad[ii] += 2.0 * w * a[jj]
    return grad


def adam_logspace(a0, time_limit, beta_start=5.0, beta_end=500.0,
                  top_k_start=80, top_k_end=5, lr=0.005,
                  beta1=0.9, beta2=0.999, eps=1e-8,
                  ramp_steps=3000, verbose=False):
    """
    Adam optimizer in log-space with adaptive smoothing.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    
    m = np.zeros(n)
    v = np.zeros(n)
    log_a = np.log(np.maximum(a, 1e-15))
    
    stall = 0
    restart_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive parameters
        frac = min(1.0, t / float(ramp_steps))
        beta = beta_start + (beta_end - beta_start) * frac
        top_k = max(top_k_end, int(round(top_k_start + (top_k_end - top_k_start) * frac)))
        lr_t = lr / (1.0 + 0.0005 * t)
        
        # Compute autocorrelation
        ac = fast_autocorr(a)
        
        # Get top-k and weights
        top_idx, weights = get_topk_weights(ac, top_k, beta)
        
        # Compute gradient in log-space: dLSE/d(log_a[j]) = a[j] * grad_conv[j]
        grad_conv = compute_gradient(a, top_idx, weights)
        grad_log = a * grad_conv
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        log_a = log_a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back and normalize
        max_log = log_a.max()
        a = np.exp(log_a - max_log)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        J = compute_J(a)
        if J < best_J - 1e-14:
            best_J = J
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stuck
        if stall > 3000:
            stall = 0
            restart_count += 1
            sigma = 0.02 + 0.01 * restart_count
            rng = np.random.default_rng(42 + restart_count)
            pert = rng.normal(0, sigma, n) * best_a
            a_new = best_a + pert
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()
            a = a_new
            log_a = np.log(a)
            m = np.zeros(n)
            v = np.zeros(n)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Adam restart {restart_count} ({elapsed:.0f}s): J={best_J:.6f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): J={best_J:.6f}, beta={beta:.0f}, tk={top_k}")
    
    return best_a, best_J


def peak_shave_fast(a0, time_limit, verbose=False, seed=42):
    """
    Fast peak shaving: vectorized destination search with analytical amounts.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        k_peak = int(np.argmax(ac))
        
        # Get all top peaks for danger computation
        n_peaks = min(25, len(ac))
        top_lags = np.zeros(n_peaks, dtype=int)
        if n_peaks == 1:
            top_lags[0] = k_peak
        else:
            top_lags = np.argpartition(ac, -n_peaks)[-n_peaks:]
        
        # Compute danger: contribution of each position to top peaks
        danger_src = np.zeros(n)
        danger_dst = np.zeros(n)
        for k in top_lags:
            k_int = int(k)
            s = max(0, k_int - n + 1)
            e = min(n, k_int + 1)
            ii = np.arange(s, e)
            jj = k_int - ii
            # If we reduce a[ii], the peak decreases by a[jj]
            danger_src[ii] += a[jj]
            # If we increase a[ii], the peak increases by a[jj]
            danger_dst[ii] += a[jj]
            danger_src[jj] += a[ii]
            danger_dst[jj] += a[ii]
        
        # Source: position with highest danger_src and substantial mass
        src_mask = a > 0.001
        danger_src_masked = np.where(src_mask, danger_src, -1)
        src = int(np.argmax(danger_src_masked))
        if a[src] < 1e-8:
            stall += 1
            continue
        
        # Destination: positions with low danger_dst
        dst_threshold = np.percentile(danger_dst, 20)
        dst_candidates = np.where(danger_dst < dst_threshold)[0]
        if len(dst_candidates) == 0:
            dst_candidates = np.argsort(danger_dst)[:max(10, n // 50)]
        dst_candidates = dst_candidates[dst_candidates != src]
        
        if len(dst_candidates) == 0:
            stall += 1
            continue
        
        # Try batch of (src, dst, amount) combinations
        n_try = min(50, len(dst_candidates))
        dst_sample = dst_candidates[rng.permutation(len(dst_candidates))[:n_try]]
        
        fracs = np.array([0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2])
        
        best_improvement = 0
        best_dst = -1
        best_amt = 0
        
        for dst in dst_sample:
            for frac in fracs:
                amt = a[src] * frac
                if amt < 1e-10:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                J_trial = compute_J(a_trial)
                imp = best_J - J_trial
                if imp > best_improvement:
                    best_improvement = imp
                    best_dst = int(dst)
                    best_amt = amt
        
        if best_improvement > 1e-14:
            a[src] -= best_amt
            a[best_dst] += best_amt
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_J = compute_J(a)
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                # Aggressive perturbation
                n_pert = max(10, n // 100)
                positions = rng.choice(n, size=n_pert, replace=False)
                for p in positions:
                    if a[p] > 1e-6:
                        amt = a[p] * rng.uniform(0.02, 0.15)
                        a[p] -= amt
                        a[rng.integers(n)] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    PeakShave final ({elapsed:.0f}s): J={best_J:.6f}")
    
    return best_a, best_J

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=400):
    """LP Frank-Wolfe with Nesterov acceleration."""
    import cvxpy as cp
    
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    y = a.copy()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build constraint matrix (sparse construction)
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y
        
        ac_vals = T_full @ y
        M = ac_vals.max()
        if M < 1e-16:
            break
        
        # Select top-K binding constraints
        margins = M - ac_vals
        K = min(top_K, 2 * n - 1)
        if K == 1:
            binding = np.array([np.argmin(margins)])
        else:
            binding = np.argpartition(margins, K - 1)[:K]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_full[binding, :] @ b_var <= M + 1e-10]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            break
        direction = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y = (1.0 - alpha) * y + alpha * direction
        y = np.maximum(y, 0.0)
        y = y / y.sum()
        
        if t == 1:
            x = y.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x = y + beta_t * (y - a)
            x = np.maximum(x, 0.0)
            x = x / x.sum()
        
        a = y
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m


def spectral_refine(a0, time_limit, verbose=False, seed=77):
    """
    Spectral refinement: iteratively adjust amplitude spectrum to flatten autocorrelation.
    Uses a "peak-aware" spectral adjustment.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    phase = np.zeros(n)
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    power = np.abs(A)
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Current autocorrelation
        ac = fast_autocorr(a)
        peak_vals = np.sort(ac)[::-1][:10]
        mean_ac = ac.mean()
        
        # Target: reduce peak-to-mean ratio
        # In frequency domain, adjust power to reduce time-domain peaks
        # Simple approach: where power is high, reduce slightly; where low, increase
        target = np.full_like(power, power.mean())
        # Blend: move power toward flat
        alpha = 0.005 + 0.005 * (1.0 - min(1.0, t / 50000.0))
        new_power = power + alpha * (target - power)
        
        # Add small random perturbation to escape local minima
        if t % 1000 == 0:
            new_power += rng.normal(0, 0.01 * power.mean(), size=new_power.shape)
        
        new_power = np.maximum(new_power, 1e-10)
        
        # Reconstruct
        A_new = new_power * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        J = compute_J(a_new)
        if J < best_J:
            best_J = J
            best_a = a_new.copy()
            a = a_new
            power = new_power.copy()
        else:
            # Keep exploring but don't update best
            a = a_new
            power = new_power.copy()
            
            # If stuck for too long, revert
            if t % 5000 == 0:
                a = best_a.copy()
                A = np.fft.rfft(a, fft_size)
                power = np.abs(A)
                phase = np.angle(A)
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Spectral ({elapsed:.0f}s): J={best_J:.6f}")
    
    return best_a, best_J


def multi_start_adam(a_base, time_limit, verbose=False):
    """
    Run Adam from multiple perturbed starts.
    """
    n = len(a_base)
    a0 = norm_sum1(a_base)
    best_a = a0.copy()
    best_J = compute_J(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(2024)
    
    n_starts = 3
    t_per = time_limit / n_starts
    
    configs = [
        {'beta_start': 5, 'beta_end': 300, 'top_k_start': 60, 'top_k_end': 5, 'lr': 0.004},
        {'beta_start': 10, 'beta_end': 800, 'top_k_start': 40, 'top_k_end': 3, 'lr': 0.006},
        {'beta_start': 3, 'beta_end': 150, 'top_k_start': 100, 'top_k_end': 10, 'lr': 0.003},
    ]
    
    for i in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Create perturbed start
        if i == 0:
            start = a0.copy()
        else:
            sigma = 0.05 * i
            pert = rng.normal(0, sigma, n) * a0
            start = a0 + pert
            start = np.maximum(start, 1e-10)
            start = start / start.sum()
        
        cfg = configs[i]
        remaining = time_limit - (time.monotonic() - t0)
        t_avail = min(t_per, remaining - 5)
        if t_avail < 10:

        a_out, J_out = adam_logspace(start, t_avail,
                                       beta_start=cfg['beta_start'],

                                       lr=cfg['lr'])
        if J_out < best_J:
            best_J = J_out
            best_a = a_out.copy()
            if verbose:
                print(f"    MultiStart {i}: J={best_J:.6f}")
    
    return best_a, best_J


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator: Adam log-space optimization + peak shaving + LP FW + spectral.
    Uses adaptive time allocation across phases.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    solver = get_lp_solver()
    if solver:
        print(f"LP solver: {solver}")
    else:
        print("No LP solver available")
    
    # === Phase 1: Multi-start Adam (primary workhorse) ===
    print(f"\n=== Phase 1: Multi-start Adam (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = multi_start_adam(a_in, t1, verbose=True)
        try_update(a_out, "multistart_adam")
    
    # === Phase 2: Peak shaving ===
    print(f"\n=== Phase 2: Peak shaving (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shave_fast(a_in, t2, verbose=True, seed=42)
        try_update(a_out, "peak_shave")
    
    # === Phase 3: LP Frank-Wolfe ===
    print(f"\n=== Phase 3: LP Frank-Wolfe (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30 and solver is not None and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True, top_K=400)
        try_update(a_out, f"fw_n{n_best}")
    
    # === Phase 4: Spectral refinement ===
    print(f"\n=== Phase 4: Spectral refinement (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_refine(a_in, t4, verbose=True)
        try_update(a_out, "spectral")
    
    # === Phase 5: Second Adam round (finer tuning) ===
    print(f"\n=== Phase 5: Adam fine-tune (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = adam_logspace(a_in, t5,
                                  beta_start=50, beta_end=1000,
                                  top_k_start=20, top_k_end=3,
                                  lr=0.002, ramp_steps=5000,
                                  verbose=True)
        try_update(a_out, "adam_fine")
    
    # === Phase 6: Peak shaving round 2 ===
    print(f"\n=== Phase 6: Peak shave round 2 ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shave_fast(a_in, t6, verbose=True, seed=999)
        try_update(a_out, "peak_shave_r2")
    
    # === Phase 7: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Longer length exploration ===")
    longer = [n_best + 400, n_best + 800]
    longer = [l for l in longer if l <= 5000]
    
    for new_n in longer[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t7a = min(remaining * 0.25, 150.0)
        if t7a > 20:
            a_out, _ = adam_logspace(a_long, t7a,
                                      beta_start=10, beta_end=300,
                                      top_k_start=50, top_k_end=5,
                                      lr=0.005)
            try_update(a_out, f"adam_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out, _ = peak_shave_fast(a_long, min(remaining * 0.3, 80.0), seed=555)
            try_update(a_out, f"peak_n{new_n}")
    
    # === Phase 8: Final LP if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 40 and solver is not None and n_best <= 3500:
        t8 = min(remaining * 0.35, 250.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=500)
            try_update(a_out, f"fw_final_n{n_best}")
    
    # === Phase 9: Final Adam polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
        t9 = min(remaining * 0.5, 200.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = adam_logspace(a_in, t9,
                                      beta_start=200, beta_end=2000,
                                      top_k_start=10, top_k_end=2,
                                      lr=0.001, ramp_steps=2000)
            try_update(a_out, "final_polish")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_safe(best_seq)
        print(f"Fallback uniform: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
