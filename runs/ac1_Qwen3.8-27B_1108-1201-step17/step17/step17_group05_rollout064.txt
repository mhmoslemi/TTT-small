def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def exact_subgrad(a, top_k=20, beta=10.0):
    n = len(a)
    C = fast_autocorr(a)
    M = C.max()
    k = min(top_k, len(C))
    sorted_idx = np.argsort(C)[::-1][:k]
    sorted_vals = C[sorted_idx]
    weights = np.exp(beta * (sorted_vals - M))
    ws = weights.sum()
    if ws < 1e-30:
        weights = np.zeros(k)
        weights[0] = 1.0
    else:
        weights = weights / ws
    g = np.zeros(n)
    for i in range(k):
        wi = weights[i]
        if wi < 1e-12:
            break
        ki = int(sorted_idx[i])
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1)
            g[js] += wi * 2.0 * a[ki - js]
    return g, M

def coord_step(a, n, best_m, rng):
    C = fast_autocorr(a)
    M = C.max()
    peaks = np.argsort(C)[::-1][:min(3, 2*n-1)]
    for k in peaks:
        ki = int(k)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[ki - idx]
        if contrib.max() < 1e-20:
            continue
        src = idx[np.argmax(contrib)]
        sv = a[src]
        if sv < 1e-12:
            continue
        impact = np.zeros(n)
        for k2 in peaks:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            d = np.arange(s2, e2 + 1)
            impact[d] += a[k2i - d]
        dist = np.abs(np.arange(n, dtype=float) - src) / max(n, 1)
        scores = impact + 0.5 * (1.0 - dist)
        scores[src] = np.inf
        dst = int(np.argmin(scores))
        for frac in [0.003, 0.008, 0.001, 0.015, 0.005, 0.02]:
            amt = a[src] * frac
            if amt < 1e-15 or a[src] <= 0:
                break
            at = a.copy()
            at[src] -= amt
            at[dst] += amt
            at = np.maximum(at, 0.0)
            at = at / at.sum()
            Mt = fast_autocorr(at).max()
            if Mt < M - 1e-14:
                return at
        # Random perturbation
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j and a[i] > 1e-10:
            amt = a[i] * rng.uniform(0.002, 0.01)
            a[i] -= amt
            a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    return a

def make_smooth_init(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)

    a = np.maximum(a, 0.0)
