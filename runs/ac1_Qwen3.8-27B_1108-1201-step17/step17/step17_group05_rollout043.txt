```python
def prox_simplex(z, n):
    """Project z onto the simplex {x: sum(x)=1, x>=0}."""
    # Algorithm from Duchi et al. (2008)
    z = z.copy()
    rho = 0
    z_sorted = np.sort(z)[::-1]
    cssv = np.cumsum(z_sorted)
    ind = np.arange(1, n+1)
    cond = z_sorted - (cssv - 1) / ind > 0
    rho = ind[cond][-1]
    theta = (cssv[cond][rho-1] - 1) / rho
    return np.maximum(z - theta, 0)
```

def accelerated_prox_grad(a, n, max_iter, beta_init, beta_growth, eta, time_limit):
    a = a.astype(float).copy()
    x = a.copy()
    y = a.copy()
    best_a = a.copy()
    best_val = np.inf
    t0 = time.monotonic()
    
    for t in range(1, max_iter+1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        alpha = (1 + np.sqrt(1 + 4*(t-1)**2)) / 2
        beta_t = (t-2) / alpha  # Previous momentum coefficient
        y = x + beta_t * (x - y)
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s
        
        # Compute convolution
        C = fast_autocorr(y)
        M = C.max()
        k_star = C.argmax()
        
        # Soft-max weights
        beta_t_smooth = beta_init * beta_growth ** (t / 1000)
        shifted = C - M
        w = np.exp(beta_t_smooth * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(C))
            w[k_star] = 1.0
        else:
            w = w / w_sum
        
        # Compute subgradient: g_j = 2 * sum_k w[k] * a[k-j]
        # = 2 * conv(w, a_rev)[n-1 : 2*n-1]
        # Use FFT
        total_len = len(w) + n - 1  # = 2(n-1) + n - 1 = 3n-3
        fft_size = 1 << (total_len - 1).bit_length()
        a_rev = y[::-1]
        W = np.fft.rfft(w, fft_size)
        A_rev = np.fft.rfft(a_rev, fft_size)

        g = 2.0 * g_half

        M_new = C_new.max()
        if M_new < best_val:
            best_val = M_new
            best_a = x_new.copy()
        
        x = x_new
    
    return best_a, best_val

"""
Accelerated Proximal Gradient (FISTA) for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Exploits convexity of the problem and FFT-based computations for efficiency.
"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear convolution conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_soft_gradient(y, beta, fft_size=None):
    """
    Compute soft subgradient of max(conv(y,y)).
    Returns (gradient, max_value, weights).
    """
    n = len(y)
    C = fast_autocorr(y)
    M = C.max()
    shifted = C - M
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(C)
        w[int(np.argmax(C))] = 1.0
    else:
        w = w / w_sum
    
    # g_j = 2 * sum_k w[k] * y[k-j] for 0<=k-j<n
    # = 2 * cross_correlation(w, y)[j]
    # Using FFT: ifft(fft(w) * conj(fft(y)))
    if fft_size is None:
        total = len(w) + n - 1  # = 3n - 2
        fft_size = 1 << (total - 1).bit_length()
    
    W = np.fft.rfft(w, fft_size)
    Y = np.fft.rfft(y, fft_size)
    g_full = np.fft.irfft(W * np.conj(Y), fft_size)
    g = 2.0 * g_full[:n]
    
    return g, M, w

def project_simplex(z):
    """Project z onto the simplex {x: sum(x)=1, x>=0} in O(n log n)."""
    n = len(z)
    if n == 0:
        return z
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = rho[cond][-1]
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(z - theta, 0.0)

def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def fista_optimize(a0, n, time_limit, beta_init=5.0, beta_final=50.0,
                   eta=0.25, verbose=True, seed=42):
    """
    Accelerated proximal gradient (FISTA) for minimizing max(conv(a,a))
    subject to sum(a)=1, a>=0.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    x = a.copy()
    y = a.copy()
    x_prev = a.copy()
    t_momentum = 1.0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    iterations = 0
    
    # Precompute FFT size
    total = 3 * n - 2
    fft_size = 1 << (total - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_momentum ** 2)) / 2.0
        momentum = (t_momentum - 1.0) / t_new
        y = x + momentum * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Adaptive beta
        frac = min(1.0, iterations / (time_limit * 100))
        beta_t = beta_init + (beta_final - beta_init) * frac
        
        # Compute soft gradient
        g, M_curr, w = compute_soft_gradient(y, beta_t, fft_size)
        
        # Proximal step
        z = y - eta * g
        x_new = project_simplex(z)
        
        # Evaluate
        C_new = fast_autocorr(x_new)
        M_new = C_new.max()
        if M_new < best_M:
            best_M = M_new
            best_a = x_new.copy()
        
        x_prev = x
        x = x_new
        
        if verbose and iterations % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"      iter={iterations} ({elapsed:.0f}s): J={J:.6f}, M={best_M:.8f}, beta={beta_t:.1f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"      FISTA done ({elapsed:.0f}s): J={J:.6f}, iters={iterations}")
    
    return best_a, best_M

**Option 6: Use the special structure of Q_k.**

Q_k is a "slanted identity" (ones on the anti-diagonal i+j=k). This means Q_k is a permutation matrix (or a sub-matrix). Specifically:
- If k < n: Q_k has non-zero entries at positions (i, k-i) for i = max(0, k-n+1) to min(k, n-1).
