    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_gradient_fft(a, p):
    """
    Compute gradient of sum_k p[k] * (a*a)[k] w.r.t. a.
    grad[j] = 2 * sum_l p[j+l] * a[l] (for valid indices)
    Uses FFT cross-correlation: O(n log n).
    """
    n = len(a)
    m = len(p)  # should be 2n-1
    # We need R[j] = sum_l p[j+l] * a[l] for j=0,...,n-1
    # This is cross-correlation of p and a at specific lags
    # R[j] = sum_l a[l] * p[j+l]
    # Let's compute via: FFT approach
    # Cross-correlation: corr[j] = sum_i x[i] * y[i+j]
    # We want: sum_l a[l] * p[j+l] = cross_corr(a, p)[j]
    # cross_corr(x, y)[j] = sum_i x[i] * y[i+j]
    # = IFFT(FFT(x) * conj(FFT(y)))  -- but need to be careful with indexing
    
    # Use the property: cross_corr(x,y)[j] = (x * flip(y))[len(x)-1+j] for circular
    # For linear, we need zero-padding
    
    # Simpler: pad both to size N >= n + m - 1
    N = 1 << (n + m - 1 - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    # cross_corr(a, p)[j] = sum_i a[i] * p[i+j]
    # = IFFT(FFT(a) * conj(FFT(p)))  gives circular cross-correlation
    # For linear, with zero-padding it's correct
    corr = np.fft.irfft(A * np.conj(P), N)
    # corr[j] for j=0,...,n-1 gives what we want
    grad = 2.0 * corr[:n]
    return grad


def mw_optimize(a0, time_limit, c_p=1.5, c_a=2.0, alpha=0.5, top_k=50, beta=3.0,
                warm_p=None, verbose=True, verbose_interval=10000):
    """
    Fast multiplicative weights optimizer for min_a max_k (a*a)[k] s.t. sum(a)=1, a>=0.
    
    Uses:
    - Multiplicative weights update on dual variable p (distribution over lags)
    - Exponentiated gradient update on primal a
    - FFT-based gradient computation: O(n log n) per iteration
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac0 = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac0.max()
    t0 = time.monotonic()
    
    # Initialize p
    if warm_p is not None:
        p = np.asarray(warm_p, dtype=np.float64).copy()
        p = np.maximum(p, 1e-30)
        p = p / p.sum()
    else:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    
    iter_count = 0
    stall_count = 0
    stall_threshold = 50000
    
    # FFT size for gradient computation
    N_grad = 1 << (n + 2 * n - 1 - 1).bit_length()
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Update best
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # If stalled for too long, add perturbation
        if stall_count > stall_threshold:
            stall_count = 0
            rng = np.random.default_rng(iter_count)
            pert = rng.normal(0, 0.01, n) * best_a
            a_new = best_a + pert
            a_new = np.maximum(a_new, 1e-10)
            a = a_new / a_new.sum()
            # Reset p to be concentrated on current peaks
            ac = fast_autocorr(a)
            p = ac.copy()
            p = np.maximum(p, 1e-30)
            p = p / p.sum()
        
        # Step sizes (diminishing)
        eta_p = c_p / (np.sqrt(t + 1.0))
        eta_a = c_a / (np.sqrt(t + 1.0))
        
        # Update p: multiplicative weights
        # Concentrate p on lags with high autocorrelation
        # Use a soft version: weight by (ac / mean_ac)^beta for concentration
        ac_clipped = np.maximum(ac, 1e-30)
        weights = ac_clipped ** beta
        w_sum = weights.sum()
        if w_sum < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w = weights / w_sum
        # Blend: p = (1-gamma)*p + gamma*w
        gamma = min(1.0, 5.0 / (t + 10.0))
        p = (1.0 - gamma) * p + gamma * w
        p = np.maximum(p, 1e-30)
        p = p / p.sum()
        
        # Compute gradient via FFT
        grad = compute_gradient_fft(a, p)
        
        # Update a: exponentiated gradient (multiplicative)
        # Clip gradient to avoid overflow/underflow
        grad_max = np.abs(grad).max()
        if grad_max > 1e-15:
            grad = grad / grad_max  # normalize to [-1, 1] range
            a = a * np.exp(-eta_a * grad)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        iter_count += 1
        
        if verbose and iter_count % verbose_interval == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    MW iter={iter_count} ({elapsed:.0f}s): score={score:.6f}, "
                  f"max_ac={best_m:.8f}, eta_a={eta_a:.4f}")
    
    elapsed = time.monotonic() - t0
    score = 2.0 * n * best_m
    if verbose:
        print(f"    MW done ({elapsed:.0f}s): score={score:.6f}, iters={iter_count}")
    return best_a, best_m, p


def mw_optimize_v2(a0, time_limit, verbose=True):
    """
    Improved MW with adaptive beta and multiple phase strategy.
    Phase 1: Large beta (concentrate on peak), large steps
    Phase 2: Medium beta, moderate steps  
    Phase 3: Small beta (spread out), small steps for convergence
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac0 = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac0.max()
    t0 = time.monotonic()
    
    p = np.ones(2 * n - 1) / (2 * n - 1)
    iter_count = 0
    stall = 0
    
    # Phase configuration: (time_fraction, c_p, c_a, beta, label)
    phases = [
        (0.30, 2.0, 3.0, 5.0, "P1"),
        (0.40, 1.5, 2.0, 3.0, "P2"),
        (0.30, 1.0, 1.0, 1.5, "P3"),
    ]
    
    phase_time = time_limit
    for frac, c_p, c_a, beta, label in phases:
        if time.monotonic() - t0 > time_limit:
            break
        phase_budget = min(frac * time_limit, time_limit - (time.monotonic() - t0))
        if phase_budget < 5:
            break
        
        if verbose:
            print(f"    MW-v2 {label}: beta={beta}, c_p={c_p}, c_a={c_a}, budget={phase_budget:.0f}s")
        
        t_phase = 0
        phase_start = time.monotonic()
        
        while time.monotonic() - phase_start < phase_budget:
            if time.monotonic() - t0 > time_limit:
                break
            
            ac = fast_autocorr(a)
            M_curr = ac.max()
            
            if M_curr < best_m - 1e-12:
                best_m = M_curr
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            
            if stall > 30000:
                # Restart with perturbation
                rng = np.random.default_rng(iter_count * 7 + t_phase)
                pert = rng.normal(0, 0.02, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                ac = fast_autocorr(a)
                p = ac.copy()
                p = np.maximum(p, 1e-30)
                p = p / p.sum()
                stall = 0
            
            t_phase += 1
            eta_p = c_p / np.sqrt(t_phase + 1.0)
            eta_a = c_a / np.sqrt(t_phase + 1.0)
            
            # Update p
            ac_clipped = np.maximum(ac, 1e-30)
            w = ac_clipped ** beta
            w_sum = w.sum()
            if w_sum > 1e-30:
                w = w / w_sum
            else:
                w = np.ones_like(ac) / len(ac)
            gamma = min(1.0, 3.0 / (t_phase + 5.0))
            p = (1.0 - gamma) * p + gamma * w
            p = np.maximum(p, 1e-30)
            p = p / p.sum()
            
            # Gradient
            grad = compute_gradient_fft(a, p)
            gmax = np.abs(grad).max()
            if gmax > 1e-15:
                a = a * np.exp(-eta_a * grad / gmax)
            a = np.maximum(a, 1e-30)
            a = a / a.sum()
            
            iter_count += 1
            
            if verbose and iter_count % 20000 == 0:
                elapsed = time.monotonic() - t0
                score = 2.0 * n * best_m
                print(f"      MW-v2 {label} iter={iter_count} ({elapsed:.0f}s): "
                      f"score={score:.6f}")
    
    elapsed = time.monotonic() - t0
    score = 2.0 * n * best_m
    if verbose:
        print(f"    MW-v2 done ({elapsed:.0f}s): score={score:.6f}, total_iters={iter_count}")
    return best_a, best_m, p


def lp_frank_wolfe(a0, time_limit, solver, initial_K=200, max_K=2000, add_per_iter=20, verbose=True):
    """LP-based Frank-Wolfe with cutting plane constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    ac_y = fast_autocorr(y)
    init_idx = np.argsort(ac_y)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    
    iter_count = 0
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-14:
            best_m = M_x
            best_a = x.copy()
        
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix
        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]
        
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # FW update with Nesterov acceleration
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): score={J:.6f}, |S|={len(constraint_set)}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): score={J:.6f}, iters={iter_count}")
    return best_a, best_m


def coordinate_polish(a0, time_limit, rng_seed=42, verbose=True):
    """Coordinate descent: move mass from peak-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        improved = False
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Score candidates by their contribution to peak lags
            scores = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
            
            cand_scores = scores[candidates]
            dst = candidates[np.argmin(cand_scores)]
            
            if dst == src:
                continue
            
            found = False
            for frac in [0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break
            
            if not found and stall > 50:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): score={J:.6f}, stall={stall}")
    
    return best_a, best_m


def soft_subgradient_optimize(a0, time_limit, top_k=30, beta=3.0, c0=2.0, verbose=True):
    """
    Nesterov-accelerated mirror descent with soft subgradient.
    Uses FFT for gradient computation when top_k is large.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac0 = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac0.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov lookahead
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation
        ac = fast_autocorr(y)
        
        # Soft subgradient: weighted combination of top-k lag gradients
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]
        
        # Mirror descent step (exponentiated gradient)
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 5000:
            stall_count = 0
            pert = rng.normal(0, 0.015, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            top_k = min(top_k + 10, 80)
            beta = max(beta - 0.3, 2.0)
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):

        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 0: Quick parametric search for potentially better init ===
    remaining = time_budget - (time.monotonic() - t_global)
    t0 = min(remaining * 0.04, 50.0)
    if t0 > 10:
        print(f"\n=== Phase 0: Parametric search (n={n0}, budget={t0:.0f}s) ===")
        x = np.linspace(0.0, 1.0, n0)
        i_arr = np.arange(n0, dtype=np.float64)
        candidates = []
        
        # Geometric
        for pwr in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
            a = (i_arr + 1.0) ** pwr
            candidates.append((f"geo_{pwr}", a))
            a2 = (n0 - i_arr) ** pwr
            candidates.append((f"rgeo_{pwr}", a2))
        
        # Sine-based
        for freq in [1, 2, 3]:
            for ph in [0.0, 0.25, 0.5]:
                a = np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.01)
                candidates.append((f"sin{freq}_{ph}", a))
        
        # Beta-distribution-like
        for alpha_b in [0.5, 1.0, 2.0, 3.0]:
            for beta_b in [0.5, 1.0, 2.0, 3.0]:
                a = np.maximum(i_arr + 1.0, 0.1) ** alpha_b * np.maximum(n0 - i_arr, 0.1) ** beta_b
                candidates.append((f"beta{alpha_b}_{beta_b}", a))
        
        # Exponential
        for rate in [1, 2, 5, 10, 20]:
            a = np.exp(-rate * x)
            candidates.append((f"exp_{rate}", a))
        
        # Sine-modulated geometric
        for amp in [0.1, 0.3, 0.5]:
            a = (i_arr + 1.0) ** 0.5 * (1.0 + amp * np.sin(2 * np.pi * 3 * x))
            a = np.maximum(a, 0.01)
            candidates.append((f"sinmod_{amp}", a))
        
        for label, a in candidates:
            a_n = norm_sum1(a)
            ac = fast_autocorr(a_n)
            val = 2.0 * n0 * ac.max()
            if val < best_score - 0.01:
                print(f"    Parametric [{label}]: {val:.6f}")
                check_and_update(a_n, label)
        
        # Also try blending with parent
        if parent is not None:
            parent_n = interpolate_sequence(parent, n0)
            for w in [0.3, 0.5, 0.7]:
                a_blend = w * best_seq + (1.0 - w) * parent_n
                check_and_update(a_blend, f"blend_{w}")
    
    # === Phase 1: Fast MW optimization (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: Fast MW optimization (n={n_best}) ===")
    t1 = min(remaining * 0.40, 700.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, m_out, p_out = mw_optimize_v2(a_in, t1, verbose=True)
        check_and_update(a_out, f"mw1_n{n_best}")
        # Save p for potential reuse
        mw_p = p_out
    
    # === Phase 2: LP-based Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP Frank-Wolfe (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30 and n_best <= 5500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, initial_K=300, max_K=2500,
                                   add_per_iter=30, verbose=True)
        check_and_update(a_out, f"lpfw_n{n_best}")
    
    # === Phase 3: Soft subgradient (AMD) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Soft subgradient (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = soft_subgradient_optimize(a_in, t3, top_k=50, beta=4.0,
                                              c0=2.5, verbose=False)
        check_and_update(a_out, f"sparse_n{n_best}")
    
    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate descent (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, rng_seed=77, verbose=True)
        check_and_update(a_out, f"coord_n{n_best}")
    
    # === Phase 5: Try different n values ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-n exploration ===")
    
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    shorter_lengths = [n_best - 500, n_best - 1000]
    all_lengths = [l for l in longer_lengths + shorter_lengths if 2000 <= l <= 6000]
    # Remove duplicates and sort by distance from current best
    all_lengths = sorted(set(all_lengths), key=lambda l: abs(l - n_best))
    
    for new_n in all_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)
        
        t5a = min(remaining * 0.40, 200.0)
        if t5a > 30:
            a_out, m_out, _ = mw_optimize_v2(a_long, t5a, verbose=False)
            check_and_update(a_out, f"mw_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 5000:
            t5b = min(remaining * 0.35, 150.0)
            if t5b > 30:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t5b, solver, initial_K=200,
                                           max_K=1500, add_per_iter=20, verbose=False)
                check_and_update(a_out, f"lp_n{new_n}")
    
    # === Phase 6: Final polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}) ===")
    if remaining > 30:
        t6 = min(remaining * 0.5, 200.0)
        if t6 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t6, rng_seed=314, verbose=True)
            check_and_update(a_out, f"finalcoord_n{n_best}")
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7 = min(remaining * 0.6, 150.0)
        if t7 > 20 and n_best <= 5500:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=400,
                                       max_K=3000, add_per_iter=40, verbose=True)
            check_and_update(a_out, f"finallp_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
