The subgradient at point a is:

where K* = {k : a^T T_k a = f(a)} is the set of active (maximizing) indices.

For optimization, we can use the "maximum" subgradient (T_{k*} a for k* in K*) or an averaged subgradient.

The mirror descent (entropy) update is:

where g is the subgradient and Z is the normalizing constant.

With Nesterov acceleration, we use a "lookahead" point y = a + beta*(a - a_prev) and evaluate the subgradient at y.

import numpy as np
import time

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, non-negative."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def soft_subgradient(a, ac, top_k=30, beta=10.0):
    """Compute soft subgradient using softmax weighting over top peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    
    # Softmax weights (numerically stable)
    shifted = top_vals - top_vals[0]
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    # Compute gradient
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        j1 = k - idx  # a[k-i]
        j2 = idx - k  # a[i-k]
        # Valid j1: already ensured by i_start, i_end
        # Valid j2: 0 <= j2 < n
        valid_j2 = (j2 >= 0) & (j2 < n)
        contrib = a[j1].copy()
        # Add a[j2] where valid and j2 != j1
        mask = valid_j2 & (j2 != j1)
        contrib[mask] += a[j2[mask]]
        # Where j2 == j1 (i.e., i = k/2), add a[i] once more
        same = (j2 == j1) & valid_j2
        contrib[same] += a[j1[same]]
        grad += wi * contrib
    return grad

def mirrored_descent_step(a, g, eta):
    """Mirror descent update in the probability simplex."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * g
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def optimize_amd(a0, time_limit, top_k=50, beta=8.0, c0=5.0, 
                 restart_every=3000, verbose=False, rng_seed=42):
    """Accelerated mirror descent with soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    eta = c0
    stall = 0
    best_stall_action = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute objective and subgradient at y
        ac = fast_autocorr(y)
        M_y = ac.max()
        g = soft_subgradient(y, ac, top_k=top_k, beta=beta)
        
        # Step size: eta_t = c0 / sqrt(t)
        eta_t = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent update
        x_new = mirrored_descent_step(y, g, eta_t)
        
        # Track best
        M_x = fast_autocorr(x_new).max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        x_prev = x
        x = x_new
        
        # Restart logic
        if stall > restart_every:
            stall = 0
            # Perturb around best
            strength = min(0.15, 0.03 + 0.02 * (best_stall_action))
            best_stall_action += 1
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart ({elapsed:.0f}s): J={J:.6f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse structured initializations (tilt-bowl, power-law, random smooth, parent)
2. Nesterov-accelerated mirror descent with adaptive soft subgradient (softmax-weighted peaks)

4. Multi-scale coarse-to-fine via interpolation
5. Peak shaving via targeted mass redistribution
6. Coordinate descent polish
7. Multi-length exploration

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def soft_subgradient_vec(a, ac, top_k=40, beta=10.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals[0]
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        j1 = k - idx
        j2 = idx - k
        contrib = a[j1].copy()
        valid_j2 = (j2 >= 0) & (j2 < n)
        diff_mask = valid_j2 & (j2 != j1)
        contrib[diff_mask] += a[j2[diff_mask]]
        same_mask = valid_j2 & (j2 == j1)
        contrib[same_mask] += a[j1[same_mask]]
        grad += wi * contrib
    return grad


def mirror_step(a, g, eta):
    log_a = np.log(np.maximum(a, 1e-30)) - eta * g
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def optimize_amd(a0, time_limit, top_k=50, beta=10.0, c0=5.0,
                 restart_every=4000, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_c0 = c0
    n_restarts = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = soft_subgradient_vec(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta_t = cur_c0 / (np.sqrt(t) + 1.0)
        x_new = mirror_step(y, g, eta_t)

        M_new = fast_autocorr(x_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = x
        x = x_new

        if stall > restart_every:
            stall = 0
            n_restarts += 1
            strength = min(0.2, 0.03 + 0.03 * n_restarts)
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 100)
            cur_beta = max(cur_beta - 0.5, 3.0)
            cur_c0 = c0 * (1.0 + 0.15 * min(n_restarts, 8))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD r={n_restarts} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def peak_shave(a0, time_limit, rng_seed=42, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:10]

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            costs = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                i2s = max(0, k2i - n + 1)
                i2e = min(n - 1, k2i)
                idx2 = np.arange(i2s, i2e + 1)
                costs[idx2] += a[k2i - idx2]
                j2 = idx2 - k2i
                vj = (j2 >= 0) & (j2 < n)
                costs[idx2[vj]] += a[j2[vj]]

            low_cost = np.argsort(costs)[:max(1, n // 5)]
            best_dst = low_cost[np.argmin(a[low_cost])]
            if best_dst == src:
                best_dst = int(rng.integers(n))
                if best_dst == src:
                    continue

            found = False
            for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[best_dst] += amt
                a_t = np.maximum(a_t, 0.0)
                s_t = a_t.sum()
                if s_t < 1e-15:
                    continue
                a_t = a_t / s_t
                M_t = fast_autocorr(a_t).max()
                if M_t < best_m - 1e-16:
                    a = a_t
                    best_m = M_t
                    best_a = a.copy()
                    found = True
                    break

            if not found:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakShave it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, rng_seed=42, verbose=False):

def make_tilt_bowl(n, tilt=0.3, sharp=2.0):
    x = np.linspace(0, 1, n)
    base = ((x - 0.5) ** 2) ** sharp
    tilt_factor = 1.0 + tilt * (2 * x - 1)
    a = base * tilt_factor
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_power_ramp(n, power=1.5, direction=1):
    x = np.arange(1, n + 1, dtype=np.float64)
    if direction == 1:
        a = x ** power
    else:
        a = (n + 1 - x) ** power
    return norm_sum1(a)


def make_double_exp(n, center=0.3, width=0.2):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    a += 0.01 * np.exp(-((x - (1 - center)) / (width * 0.7)) ** 2)
    return norm_sum1(a)


def make_cosine_tilt(n, mode=1, tilt=0.5):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a *= (1.0 + tilt * (2 * x - 1))
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_random_smooth(n, seed=0, n_freq=6):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.2 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_sinc_like(n, bandwidth=5.0, center=0.5):
    x = np.linspace(0, 1, n)
    t = (x - center) * bandwidth
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coord descent (n={n_best}) ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t7, rng_seed=777, verbose=False)
        try_update(a_out, f"coordescent_n{n_best}")

    # === Phase 8: Final AMD with aggressive params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final AMD (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.5, 200.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = optimize_amd(a_in, t8, top_k=80, beta=6.0, c0=3.5,
                                     restart_every=8000, verbose=False, rng_seed=500)
            try_update(a_out, f"amd_final_n{n_best}")

    # === Phase 9: One more peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 9: Final peak shave ===")
        t9 = min(remaining * 0.6, 100.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_shave(a_in, t9, rng_seed=1234, verbose=False)
            try_update(a_out, f"peakshave_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_tilt_bowl(n_fb, tilt=0.4, sharp=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
