def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_matrix(a, n):
    """Build the matrix T such that (T @ b)[k] = sum_i b_i * a[k-i] for valid k."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def alternating_lp_optimize(a0, time_limit, solver, max_iters=200, verbose=True):
    """
    Alternating LP block coordinate descent.
    Given b, solve: min_t s.t. (a*b)[k] <= t, sum a = 1, a >= 0
    Given a, solve: min_t s.t. (a*b)[k] <= t, sum b = 1, b >= 0
    """
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    b = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute structure
    # For the constraint (a * b)[k] = sum_i a_i * b[k-i], we need the matrix
    # that maps b to the convolution with a (or vice versa)

    def solve_lp(fixed, verbose_flag=False):
        """Given fixed vector, find the other vector minimizing max convolution."""
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = fixed

        var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [T @ var <= t_var, cp.sum(var) == 1]
        prob = cp.Problem(cp.Minimize(t_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return fixed.copy()

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return fixed.copy()

        val = np.maximum(np.asarray(var.value, dtype=np.float64), 0.0)
        s = val.sum()
        if s < 1e-15:
            return fixed.copy()
        return val / s

    stagnation = 0
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve for b given a
        b_new = solve_lp(a)
        # Solve for a given b_new
        a_new = solve_lp(b_new)

        # Check improvement
        m_a = fast_autocorr(a_new).max()
        if m_a < best_m - 1e-12:
            best_m = m_a
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        a = a_new
        b = b_new

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AltLP iter={it} ({elapsed:.0f}s): J={J:.6f}, stagnation={stagnation}")

        if stagnation > 20:
            break

    return best_a, best_m


def alternating_lp_fast(a0, time_limit, solver, max_iters=300, verbose=True):
    """
    Faster alternating LP using constraint generation.
    Only enforce a subset of constraints (binding + near-binding).
    """
    import cvxpy as cp
    n = len(a0)
    a = norm_sum1(a0)
    b = a.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    K_sel = min(200, 2 * n - 1)

    def solve_lp_constrained(fixed, active_idx, verbose_flag=False):
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = fixed

        conv_vals = T @ fixed
        # Select constraints: those closest to being binding (smallest margin)
        if active_idx is None or len(active_idx) == 0:
            T_sel = T
            var = cp.Variable(n, nonneg=True)
            t_var = cp.Variable()
            constraints = [T_sel @ var <= t_var, cp.sum(var) == 1]
        else:
            T_sel = T[active_idx]
            var = cp.Variable(n, nonneg=True)
            t_var = cp.Variable()
            constraints = [T_sel @ var <= t_var, cp.sum(var) == 1]

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return fixed.copy()

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return fixed.copy()

        val = np.maximum(np.asarray(var.value, dtype=np.float64), 0.0)
        s = val.sum()
        if s < 1e-15:
            return fixed.copy()
        return val / s

    stagnation = 0
    active_idx = None
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Determine active constraints for b given a
        T_a = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_a[j:j + n, j] = a
        conv_a = T_a @ b
        margins_a = conv_a.max() - conv_a
        active_a = np.argsort(margins_a)[:K_sel]

        # Solve for b
        b_new = solve_lp_constrained(a, active_a)

        # Determine active constraints for a given b_new
        T_b = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_b[j:j + n, j] = b_new
        conv_b = T_b @ a
        margins_b = conv_b.max() - conv_b
        active_b = np.argsort(margins_b)[:K_sel]

        # Solve for a
        a_new = solve_lp_constrained(b_new, active_b)

        m_a = fast_autocorr(a_new).max()
        if m_a < best_m - 1e-12:
            best_m = m_a
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        a = a_new
        b = b_new

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AltLP-fast iter={it} ({elapsed:.0f}s): J={J:.6f}, stg={stagnation}")

        if stagnation > 30:
            break

    return best_a, best_m


def freq_domain_optimize_v2(a0, time_limit, iters=5000, verbose=False):
    """
    Improved frequency-domain optimization.
    Uses spectrum concentration: push energy to low frequencies to flatten autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    # Target: concentrate |A|^2 at low frequencies
    # Use a smooth target spectrum
    freqs = np.fft.rfftfreq(fft_size) * 2 * np.pi  # angular frequencies
    # Target: |A(omega)|^2 proportional to 1/(1 + (omega/omega0)^2)^alpha
    # This gives a Lorentzian-type concentration at omega=0

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        r = fast_autocorr(a)
        m_curr = r.max()

        # Compute ratio of current max to mean autocorrelation
        mean_r = r.sum() / len(r)
        ratio = m_curr / max(mean_r, 1e-30)

        # Adaptive concentration parameter
        # If ratio is high (peaked autocorrelation), concentrate more
        alpha = np.clip(0.5 * (ratio - 1.0) + 1.0, 1.0, 8.0)

        # Target spectrum: concentrate at low freq
        # Use sqrt of exponential decay
        omega_scale = 2.0 * np.pi / (0.3 * n)
        target_P = np.exp(-np.abs(freqs) / omega_scale * alpha)
        # Normalize to match total energy
        target_P *= P.sum() / max(target_P.sum(), 1e-30)

        # Blend with current spectrum (damped update)
        damp = 0.3 + 0.4 * (1.0 - min(it / float(iters), 1.0))
        P_new = (1.0 - damp) * P + damp * target_P
        P_new = np.maximum(P_new, 1e-30)

        # Update magnitude, keep phase
        phase = np.angle(A)
        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        elif M_new < m_curr:
            # Slight improvement, accept with damping
            a = 0.7 * a + 0.3 * a_new
            a = norm_sum1(a)
        else:
            # No improvement, try smaller step
            a = 0.95 * a + 0.05 * a_new
            a = norm_sum1(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FreqDom2 it={it} ({elapsed:.0f}s): J={J:.6f}, alpha={alpha:.2f}")

    return best_a, best_m


def subgradient_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0,
                         restart_every=5000, verbose=False, rng_seed=42):
    """
    Multi-peak subgradient mirror descent with Nesterov acceleration.
    Uses multiplicative interior-point style updates.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    kickoff_iter = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        # Build subgradient from top peaks
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = gamma / (np.sqrt(t - kickoff_iter) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            kickoff_iter = t
            strength = min(0.15, 0.03 + 0.02 * min(stall_count // restart_every, 3))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 150)
            cur_beta = max(cur_beta - 0.5, 2.0)
            gamma = c0 * (1.0 + 0.15 * min((t - kickoff_iter) // restart_every, 5))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    SubGrad restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SubGrad t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    """
    Fine-grained coordinate descent: try small perturbations at positions
    contributing most to peak autocorrelation values.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-25:
                continue

            # Find the pair with maximum contribution
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try moving mass to positions with low autocorrelation cost
            # Compute "cost" of each position: sum of autocorrelation contributions
            costs = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                i_start = max(0, k2i - n + 1)
                i_end = min(n, k2i + 1)
                if i_start < i_end:
                    idx2 = np.arange(i_start, i_end)
                    costs[idx2] += a[k2i - idx2]

            # Candidate destinations: low cost, low current value
            candidates = np.argsort(costs * (1.0 + a))[:max(5, n // 10)]
            for best_dst in candidates[:5]:
                if best_dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 0.01:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

            if not improved and stall > 50:
                # Random perturbation to escape
                j2 = int(rng.integers(n))
                if j2 != src and src_val > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = src_val * frac
                    a[src] -= amt
                    a[j2] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Random mass transport
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_raised_cosine(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    return norm_sum1(a)


def make_hann_window(n):
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def make_super_gaussian(n, p=4.0):
    x = np.linspace(-1, 1, n)
    a = np.exp(-np.abs(x) ** p)
    return norm_sum1(a)


def make_flat_top(n):
    """Flat-top window: designed for minimal sidelobes."""
    x = np.arange(n) / (n - 1)
    a = (1 - 1.93 * np.cos(2 * np.pi * x) + 1.29 * np.cos(4 * np.pi * x)
         - 0.39 * np.cos(6 * np.pi * x) + 0.09 * np.cos(8 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def make_bell(n, width=0.4):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) ** 2) / (2 * width ** 2))
    return norm_sum1(a)


def make_dual_bell(n, separation=0.3):
    x = np.linspace(0, 1, n)
    a = (np.exp(-((x - (0.5 - separation)) ** 2) / (2 * 0.08 ** 2)) +
         np.exp(-((x - (0.5 + separation)) ** 2) / (2 * 0.08 ** 2)))
    a += 0.01 * np.ones(n)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining alternating LP, frequency-domain,
    subgradient, and coordinate descent methods.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_raised_cosine(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Alternating LP (main innovation) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Alternating LP (n={n0}) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60 and n0 <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = alternating_lp_fast(a_in, t1, solver, max_iters=300, verbose=True)
        try_update(a_out, f"altlp_n{n0}")

    # === Phase 2: Subgradient from LP result ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Subgradient polish (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_optimize(a_in, t2, top_k=60, beta=4.0, c0=4.0,
                                         restart_every=5000, verbose=True, rng_seed=42)
        try_update(a_out, f"subgrad_n{len(best_seq)}")

    # === Phase 3: More Alternating LP with constraint generation ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: AltLP round 2 (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40 and len(best_seq) <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = alternating_lp_fast(a_in, t3, solver, max_iters=200, verbose=True)
        try_update(a_out, f"altlp2_n{len(best_seq)}")

    # === Phase 4: Frequency domain ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Freq-domain v2 (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = freq_domain_optimize_v2(a_in, t4, iters=5000, verbose=True)
        try_update(a_out, f"freqdom_n{len(best_seq)}")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coord descent (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_polish(a_in, t5, verbose=True, rng_seed=123)
        try_update(a_out, f"coorddesc_n{len(best_seq)}")

    # === Phase 6: Try different initializations with AltLP ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Multi-init AltLP ===")
    n_trial = len(best_seq)
    if remaining > 120 and n_trial <= 5000:
        inits = [
            (make_hann_window(n_trial), "hann"),
            (make_super_gaussian(n_trial, p=4.0), "superG4"),
            (make_flat_top(n_trial), "flattop"),
            (make_bell(n_trial, width=0.35), "bell35"),
            (make_dual_bell(n_trial, separation=0.25), "dualbell"),
        ]

        for init, lbl in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_per = min(remaining * 0.3, 120.0)
            if t_per < 20:
                break
            print(f"  AltLP from '{lbl}' (t={t_per:.0f}s)...")
            a_out, _ = alternating_lp_fast(init, t_per, solver, max_iters=100, verbose=False)
            try_update(a_out, f"altlp_{lbl}")

    # === Phase 7: Multi-length search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Multi-length (base n={n_best}) ===")
    for new_n in [n_best + 400, n_best - 500]:
        if new_n < 1000 or new_n > 6000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t7a = min(remaining * 0.35, 150.0)
        if t7a > 30:
            a_out, _ = subgradient_optimize(a_long, t7a, top_k=50, beta=4.0, c0=4.0,
                                             restart_every=4000, verbose=False, rng_seed=77)
            try_update(a_out, f"subgrad_n{new_n}")

    # === Phase 8: Final AltLP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final AltLP (n={n_best}) ===")
    if remaining > 60 and n_best <= 5000:
        t8 = min(remaining * 0.4, 200.0)
        if t8 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = alternating_lp_fast(a_in, t8, solver, max_iters=200, verbose=True)
            try_update(a_out, f"altlp_final_n{n_best}")

    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final coord polish (n={n_best}) ===")
    if remaining > 30:
        t9 = min(remaining * 0.6, 120.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_polish(a_in, t9, verbose=True, rng_seed=999)
            try_update(a_out, f"coorddesc_final_n{n_best}")

    # === Phase 10: One more subgradient if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 10: Final subgradient ===")
        t10 = min(remaining * 0.7, 100.0)
        if t10 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = subgradient_optimize(a_in, t10, top_k=80, beta=3.0, c0=3.5,
                                             restart_every=8000, verbose=False, rng_seed=555)
            try_update(a_out, f"subgrad_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_raised_cosine(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
