import numpy as np
import time
import cvxpy as cp

_global_solver = None

def _fast_autocorr(a):
    """Compute full linear autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def _norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _compute_soft_subgrad(a, beta=10.0, top_k=None):
    """
    Compute soft subgradient of max(conv(a,a)).
    If top_k is None, use all lags.
    """
    n = len(a)
    ac = _fast_autocorr(a)
    if top_k is not None and top_k < len(ac):
        top_idx = np.argsort(ac)[::-1][:top_k]
        w = np.zeros(len(ac))
        vals = ac[top_idx]
        shifted = vals - vals.max()
        w_top = np.exp(beta * shifted)
        w_top = w_top / w_top.sum()
        for wi, idx in zip(w_top, top_idx):
            w[int(idx)] = wi
    else:
        shifted = ac - ac.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w = w / ws
        w[w < 1e-14] = 0.0
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones_like(ac) / len(ac)
        else:
            w = w / ws
    
    # grad[j] = 2 * sum_m w[m] * a[m-j] for valid m-j
    # = 2 * (a_rev * w)[n-1 + j] where a_rev = a[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = a[::-1]
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a_rev
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_padded) * np.fft.rfft(w_padded), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, ac.max(), ac


def _mirror_descent(a0, time_limit, beta_init=10.0, beta_min=4.0,
                    c0=5.0, restart_every=5000, top_k_switch=200,
                    top_k_val=100, seed=42):
    """
    Nesterov-accelerated mirror descent on the simplex.
    """
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    ac = _fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_beta = beta_init
    iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        beta_t = (iterations - 1.0) / (iterations + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        if iterations < top_k_switch:
            g, M_curr, _ = _compute_soft_subgrad(y, beta=cur_beta, top_k=None)
        else:
            g, M_curr, _ = _compute_soft_subgrad(y, beta=cur_beta, top_k=top_k_val)
        
        eta = gamma / (np.sqrt(iterations) + 1.0)
        log_y = np.log(y)
        log_a_new = log_y - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x.copy()
        x = a_new
        
        M_new = _fast_autocorr(x).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        cur_beta = max(beta_min, beta_init - 0.002 * iterations)
        
        if stall_count > restart_every:
            stall_count = 0
            pert_scale = max(0.01, 0.05 * (1.0 - stall_count / (2.0 * restart_every)))
            pert = rng.normal(0, pert_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if rng.random() < 0.3:
                x_r = rng.dirichlet(np.ones(n) * 0.3)
                x = x_r
                x_prev = x_r.copy()
    
    return best_a, best_m


def _lp_fw_optimize(a0, time_limit, solver, relaxation=1.05, top_K=500,
                    use_full=True):
    """

    """
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = _fast_autocorr(y)
        M_curr = ac.max()
        
        M_relax = M_curr * relaxation
        
        if use_full and n <= 5000:

            T = np.zeros((K_sel, n))
            for j in range(n):
                T[j:j + n, j] = y
        else:

            top_idx = np.argsort(ac)[::-1][:K_sel]

def _fast_autocorr(a):
    """Compute full linear autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _evaluate_safe(a):

def _norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _compute_soft_subgrad(a, beta=10.0, top_k=None):
    """Compute soft subgradient of max(conv(a,a)) using FFT."""
    n = len(a)
    ac = _fast_autocorr(a)
    if top_k is not None and top_k < len(ac):
        top_idx = np.argsort(ac)[::-1][:top_k]
        w = np.zeros(len(ac))
        vals = ac[top_idx]
        shifted = vals - vals.max()
        w_top = np.exp(beta * shifted)
        w_top = w_top / w_top.sum()
        for wi, idx in zip(w_top, top_idx):
            w[int(idx)] = wi
    else:
        shifted = ac - ac.max()
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w = w / ws
        w[w < 1e-14] = 0.0
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones_like(ac) / len(ac)
        else:
            w = w / ws
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = a[::-1]
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a_rev
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_padded) * np.fft.rfft(w_padded), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, ac.max()

def _interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

        pass

    if best_seq is None:
        n_base = 3200
        best_seq = _make_antisym_init(n_base, seed=1)
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # Phase 1: Mirror descent with diverse inits including antisymmetric
    print("\n=== Phase 1: Mirror Descent (diverse inits) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.25, 400.0)

    if t_phase1 > 60:
        inits = []
        labels = []
        if parent is not None:
            inits.append(_norm_sum1(parent))
            labels.append("parent")
            # Antisymmetric version of parent
            a_anti = _make_antisym_init(n0, seed=42)
            blend = 0.8 * _norm_sum1(parent) + 0.2 * a_anti
            inits.append(_norm_sum1(blend))
            labels.append("parent_anti")
        inits.append(_make_antisym_init(n0, seed=7))
        labels.append("antisym")
        inits.append(_make_smooth_init(n0, seed=13, freqs=[1, 2, 3]))
        labels.append("smooth_low")

        # Remove duplicates
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(20.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  MD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = _mirror_descent(init, t_avail, beta_init=10.0, beta_min=4.0,
                                       c0=5.0, restart_every=3000, seed=42 + idx)
            _try_update(a_out, f"md_{lbl}")

    # Phase 2: LP-FW with relaxation on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 40 and n_best <= 4000:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _lp_fw_optimize(a_in, t2, solver, relaxation=1.03)
        _try_update(a_out, f"lpfw_n{n_best}")

    # Phase 3: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord Descent (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _coord_descent(a_in, t3, seed=42)
        _try_update(a_out, f"coord_n{n_best}")

    # Phase 4: Try different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = [n_best - 300, n_best + 300, n_best + 600, n_best - 600]
    candidate_lengths = sorted(set(l for l in candidate_lengths if 1500 <= l <= 5000))
    print(f"\n=== Phase 4: Multi-length (candidates: {candidate_lengths}) ===")

    for new_n in candidate_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Length n={new_n}...")
        a_interp = _interpolate_sequence(best_seq, new_n)
        t_sub = min(remaining * 0.25, 120.0)
        if t_sub < 20:
            break
        a_out, _ = _mirror_descent(a_interp, t_sub * 0.6, seed=777 + new_n)
        _try_update(a_out, f"md_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 3500:
            t_lp = min(remaining * 0.3, 80.0)
            if t_lp > 20:
                a_out2, _ = _lp_fw_optimize(_norm_sum1(a_out), t_lp, solver, relaxation=1.02)
                _try_update(a_out2, f"lp_n{new_n}")

    # Phase 5: Second MD round on best with fresh params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: MD Round 2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 20:
        a_in = _norm_sum1(best_seq)
        # Add small antisymmetric perturbation before
        rng = np.random.default_rng(999)
        s = np.zeros(n_best)
        half = n_best // 2
        s[:half] = rng.standard_normal(half)
        s[n - half:] = -s[:half][::-1]
        sn = np.sqrt(np.sum(s ** 2))
        if sn > 1e-15:
            a_in = _norm_sum1(a_in + 0.01 * s / sn)
        a_out, _ = _mirror_descent(a_in, t5, beta_init=12.0, beta_min=5.0,
                                   c0=3.0, restart_every=6000, seed=555)
        _try_update(a_out, f"md2_n{n_best}")

    # Phase 6: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Coord (n={n_best}) ===")
    t6 = min(remaining * 0.25, 150.0)
    if t6 > 15:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _coord_descent(a_in, t6, seed=777)
        _try_update(a_out, f"coord_final_n{n_best}")

    # Phase 7: If time remains, another LP-FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 40 and n_best <= 3500:
        print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
        t7 = min(remaining * 0.6, 120.0)
        if t7 > 20:
            a_in = _norm_sum1(best_seq)
            a_out, _ = _lp_fw_optimize(a_in, t7, solver, relaxation=1.01)
            _try_update(a_out, f"lpfw_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = _make_antisym_init(n_fb, seed=1)
        best_score = _evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
