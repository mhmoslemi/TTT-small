where (f*f) is the continuous convolution of f with itself, and (g*g)_p is the periodic convolution of g with itself.

The peak of this product is maximized when BOTH factors are large simultaneously. To minimize the peak, we want:
- When (f*f)(x) is large, (g*g)_p should be small
- When (g*g)_p is large, (f*f)(x) should be small

This is a "dual optimization" problem!

For the continuous part, (f*f)(x) is largest at x=1 (the center), where it equals (integral f)^2 / (support length) for a uniform f.

For the periodic part, (g*g)_p(k) is largest at certain lags.

If we design f to be small at the center (x=1) and large at the edges, and g to have peaks at lags that correspond to the "edges" in the convolution, we can achieve a lower product peak.

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:

def fast_conv(a, b=None):
    if b is None:
        b = a
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def compute_objective(a):
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max() / (s * s)


def make_envelope_periodic_init(n, period, envelope_type='u', seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if envelope_type == 'u':
        f = 0.5 + 0.5 * (2 * x - 1) ** 2
    elif envelope_type == 'w':
        f = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(2 * np.pi * x)
        f = np.maximum(f, 0.05)
    elif envelope_type == 'v':
        f = np.abs(2 * x - 1) + 0.1
    else:
        f = np.ones(n) * 0.5
    g = np.ones(n)
    if period == 2:
        g[::2] = 1.3
        g[1::2] = 0.7
    elif period == 3:
        pattern = np.array([1.4, 0.6, 0.9])
        for i in range(n):
            g[i] = pattern[i % 3]
    elif period == 4:
        pattern = np.array([1.5, 0.5, 1.2, 0.7])
        for i in range(n):
            g[i] = pattern[i % 4]
    elif period == 5:
        pattern = np.array([1.5, 0.5, 1.3, 0.6, 0.9])
        for i in range(n):
            g[i] = pattern[i % 5]
    else:
        pattern = np.ones(period)
        for i in range(period):
            pattern[i] = 1.0 + 0.5 * rng.normal(0, 0.3)
        pattern = np.maximum(pattern, 0.1)
        for i in range(n):
            g[i] = pattern[i % period]
    a = f * g
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_ring_init(n, num_rings, seed=0):
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    for _ in range(num_rings):
        pos = int(rng.integers(0, n))
        width = int(rng.integers(1, max(2, n // (2 * num_rings))))
        for d in range(-width, width + 1):
            idx = (pos + d) % n
            a[idx] += np.exp(-0.5 * (d / max(1, width // 2)) ** 2)
    return norm_sum1(a)


def make_smooth_init(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in [1, 2, 3, 4, 5]:
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def peak_deflation_optimize(a0, time_limit, verbose=True, seed=42, top_k_peaks=5, max_iters=50000):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    it = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall and it < max_iters:
        it += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 0 and stall % 50 == 0:
            # Random perturbation to escape local optima
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.04)
                amt = a[i] * frac
                a[i] = max(0, a[i] - amt)
                a[j] += amt
                a = a / a.sum()

        # Find top peaks
        peak_indices = np.argsort(ac)[::-1][:top_k_peaks]
        peak_vals = ac[peak_indices]

        # For the top peak, find main contributors and move mass
        for pk in range(min(3, len(peak_indices))):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(peak_indices[pk])
            k_val = ac[k]

            # Find valid source indices for this lag
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            partners = k - js
            contrib = a[js] * a[partners]

            # Find top contributors
            n_top_contrib = min(5, len(js))
            top_c_idx = np.argsort(contrib)[::-1][:n_top_contrib]

            for ci in top_c_idx:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(js[ci])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Compute sensitivity for each destination
                # sens[d] = sum of a[d'] where d' pairs with d at any top peak lag
                sens = np.zeros(n)
                for pk2 in range(min(5, len(peak_indices))):
                    k2 = int(peak_indices[pk2])
                    s2_start = max(0, k2 - n + 1)
                    s2_end = min(n - 1, k2)
                    for d in range(s2_start, s2_end + 1):
                        partner = k2 - d
                        if 0 <= partner < n:
                            sens[d] += a[partner]

                # Best destination: low sensitivity, far from source
                abs_dist = np.abs(np.arange(n) - src).astype(float)
                score = sens + 0.01 * (1.0 - abs_dist / n)
                # Exclude source and its partner at this lag
                partner_src = k - src
                if 0 <= partner_src < n:
                    score[partner_src] = np.inf
                score[src] = np.inf

                candidates = np.argsort(score)[:20]
                best_move = False
                for dst in candidates:
                    dst = int(dst)
                    if dst == src or score[dst] >= np.inf:
                        continue
                    # Try multiple fractions
                    for frac in [0.05, 0.03, 0.02, 0.08, 0.01, 0.1]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-14:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            stall = 0
                            best_move = True
                            break
                    if best_move:
                        break

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Deflation iter={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Deflation final ({elapsed:.0f}s): J={J:.6f}, iters={it}")
    return best_a, best_m


def lp_frank_wolfe_peak(a0, time_limit, solver, verbose=True, top_K=80):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    prev_y = a.copy()

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select top-K most binding constraints
        K_sel = min(top_K, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix T: T[m,j] = y[m-j] if valid
        T = np.zeros((K_sel, n))
        for col_idx, m in enumerate(top_idx):
            m_int = int(m)
            j_start = max(0, m_int - n + 1)
            j_end = min(n - 1, m_int)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            T[col_idx, js] = y[m_int - js]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K_sel), cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        my = fast_autocorr(y_new).max()
        if my < best_m:
            best_m = my
            best_a = y_new.copy()

        y = y_new
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def multi_peak_balance(a0, time_limit, verbose=True, seed=77):
    """Optimize by balancing the top peaks rather than just minimizing the max."""
    n = len(a0)

            best_a = a.copy()

        # Focus on the single worst peak
        k_peak = int(np.argmax(ac))
        j_start = max(0, k_peak - n + 1)
        j_end = min(n - 1, k_peak)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        partners = k_peak - js
        contrib = a[js] * a[partners]

        # Find the biggest contributor and try to reduce it
        max_c_pos = int(np.argmax(contrib))
        src = int(js[max_c_pos])
        src_val = a[src]
        if src_val < 1e-12:
            continue

        # Try moving to all positions, pick best
        n_trials = min(50, n)
        trial_positions = rng.choice(n, size=n_trials, replace=False)

        best_trial_m = M_curr

        for dst in trial_positions:
            dst = int(dst)

            for frac in [0.03, 0.02, 0.05]:

                a_t = a.copy()

                m_t = fast_autocorr(a_t).max()
                if m_t < best_trial_m:

            i = int(rng.integers(n))

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Deflation Round 2 ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_deflation_optimize(a_in, t5, verbose=True, seed=999,
                                            top_k_peaks=3, max_iters=100000)
        try_update(a_out, "deflation_r2")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = [n_best + 200, n_best - 200, n_best + 400]
    candidate_lengths = [l for l in candidate_lengths if 1000 <= l <= 6000]
    print(f"\n=== Phase 6: Length Exploration ===")
    for new_n in candidate_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.30, 100.0)
        if t6a > 15:
            a_out, _ = peak_deflation_optimize(a_long, t6a, verbose=False, seed=555,
                                                top_k_peaks=5, max_iters=30000)
            try_update(a_out, f"defl_n{new_n}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Polish (n={n_best}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.7, 200.0)
        a_in = norm_sum1(best_seq)
        # Final deflation with very small steps
        a_out, _ = peak_deflation_optimize(a_in, t7, verbose=True, seed=1234,
                                            top_k_peaks=3, max_iters=200000)
        try_update(a_out, "final_polish")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        best_seq = make_smooth_init(3200, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
