def exact_subgradient(a, ac, active_thresh=1e-6):
    """Compute exact subgradient of f(a) = max_k (a*a)_k at point a.
    Returns (subgradient, active_set)."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n), np.array([0])
    
    # Active set: all k where ac[k] >= M - active_thresh * M
    active = np.where(ac >= M - active_thresh * M)[0]
    
    # For fair weighting, use uniform weights over active set
    # Or weight by proximity to the maximum
    margins = M - ac[active]
    # Use exponential weights: w_k = exp(-margins_k / (0.01*M))
    temp = 0.01 * M
    w = np.exp(-margins / temp)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        ki = int(k)
        # (a*a)_k = sum_i a_i * a_{i+k}
        # d/d a_j of (a*a)_k = a_{j+k} + a_{j-k} (with boundary = 0)
        # For j in [0, n-1]:
        #   contribution from i=j (a_j * a_{j+k}): a_{j+k} if j+k < n
        #   contribution from i=j-k (a_{j-k} * a_j): a_{j-k} if j-k >= 0
        s1 = max(0, ki - n + 1)
        e1 = min(n, ki + 1)
        idx = np.arange(s1, e1)
        part1 = ki - idx  # a_{j-k} part: partner index
        grad[idx] += wi * a[part1]
        
        s2 = max(0, ki - n + 1)
        e2 = min(n, ki + 1)
        idx2 = np.arange(s2, e2)
        part2 = idx2 + ki  # a_{j+k} part: partner index
        grad[idx2] += wi * a[part2]
    
    return grad, active

def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=5000,

    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # Compute soft subgradient

        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            s1 = max(0, ki - n + 1)
            e1 = min(n, ki + 1)
            idx = np.arange(s1, e1)
            grad[idx] += 2.0 * wi * a[ki - idx]

        # Mirror descent step
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 1.0, 2.0)

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                candidates.append(s)
        except Exception:
            continue
    if candidates:
        get_solver._result = candidates[0]
    else:
        get_solver._result = cp.ECOS
    print(f"  Using LP solver: {get_solver._result}")
    return get_solver._result


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_u_shape(n, sharpness=2.0):
    """U-shaped (high at edges) distribution."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_inv_u(n, sharpness=2.0):
    """Inverted-U (high in middle) distribution."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * (1 - x ** 2)
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.01
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    """Fourier-synthesis initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def amd_optimize(a0, time_limit, top_k=60, beta=6.0, c0=4.0,
                 restart_every=5000, verbose=False, rng_seed=42):
    """Nesterov-accelerated mirror descent with adaptive peak weighting."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        # Compute weighted subgradient from top peaks
        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            ki = int(k)
            s1 = max(0, ki - n + 1)
            e1 = min(n, ki + 1)
            idx = np.arange(s1, e1)
            grad[idx] += 2.0 * wi * a[ki - idx]

        # Mirror descent (exponentiated gradient)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive restart
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 1.0, 2.0)

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, 
                    initial_K=None, max_K=None, K_growth=50, K_interval=15):
    """
    LP-based Frank-Wolfe with Nesterov acceleration.
    
    At each iteration:
    1. Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in selected set, b >= 0
    2. Update y = (1-alpha)*y + alpha*(b/sum(b))
    3. Nesterov extrapolate x
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    total_lags = 2 * n - 1
    if initial_K is None:
        initial_K = min(300, total_lags)
    if max_K is None:
        max_K = total_lags
    
    K_sel = initial_K
    y = a.copy()
    x = a.copy()
    iter_count = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select constraints: top-K most binding
        if K_sel >= total_lags:
            binding_idx = np.arange(total_lags)
        else:
            margins = M_curr - ac
            binding_idx = np.argsort(margins)[:K_sel]

        # Build constraint matrix: T[k, i] = y[k-i] if 0 <= k-i < n
        K = len(binding_idx)
        # Use sparse construction
        T_sel = np.zeros((K, n))
        for row in range(K):
            ki = int(binding_idx[row])
            j_start = max(0, ki - n + 1)
            j_end = min(n, ki + 1)
            if j_start < j_end:
                T_sel[row, j_start:j_end] = y[ki - j_start:ki + 1][::-1]

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # FW update with Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        # Check improvement
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

        # Grow constraint set
        if K_sel < max_K:
            if t % K_interval == 0:
                K_sel = min(K_sel + K_growth, max_K)

    return best_a, best_m
