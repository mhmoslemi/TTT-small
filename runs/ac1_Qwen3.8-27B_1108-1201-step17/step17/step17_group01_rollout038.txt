def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_max_grad(c, beta):
    """Compute the gradient of the smooth max approximation.
    
    smooth_max(c) = (1/beta) * log(sum(exp(beta * c)))
    grad = exp(beta * c) / sum(exp(beta * c))  (softmax weights)
    """
    c_shifted = beta * c - beta * c.max()
    weights = np.exp(c_shifted)
    weights = weights / weights.sum()
    return weights  # These are the softmax weights for each lag

    f(a) = (1/beta) * log(sum_k exp(beta * c[k]))
    df/da[i] = 2 * sum_k w[k] * a[k-i] where w = softmax(beta*c)
    
    Uses cross-correlation via FFT: (w * a)[i] = sum_k w[k]*a[k-i]
    """
    n = len(a)
    bc = beta * c
    bc -= bc.max()
    w = np.exp(bc)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w = w / w_sum

    # Cross-correlation: (w ⋆ a)[i] = sum_k w[k] * a[k-i]
    # FFT size needs to be >= (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    grad = 2.0 * corr[:n]
    return grad


def compute_hard_subgrad(a, c, top_k=50, beta_w=5.0):
    """Compute subgradient using weighted top-k peaks (fallback)."""
    n = len(a)
    k_max = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = np.maximum(c[top_idx], 1e-30)
    w = top_vals ** beta_w
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    # Build sparse weight vector
    w_full = np.zeros(len(c))
    w_full[top_idx] = w

    # Use FFT cross-correlation
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    grad = 2.0 * corr[:n]
    return grad


def mirror_descent_optimize(a0, time_limit, beta_start=None, beta_end=None,
                            c0=3.0, restart_every=3000, verbose=False, seed=42):
    """Nesterov-accelerated mirror descent with smooth-max objective."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    if beta_start is None:
        beta_start = n / 100.0
    if beta_end is None:
        beta_end = n / 5.0

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    total_iters = 500000
    restart_count = 0

    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive beta: interpolate from smooth to sharp
        progress = min(t / (total_iters * 0.7), 1.0)
        beta_t = beta_start + (beta_end - beta_start) * progress

        # Nesterov momentum
        if t > 1:
            beta_t_mom = (t - 1.0) / (t + 2.0)
            y = x + beta_t_mom * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute gradient at y
        c_y = fast_conv(y)
        g = compute_smooth_grad(y, c_y, beta_t)

        # Mirror descent step (exponential weights)
        eta = gamma / (np.sqrt(t) + 1.0)
        # Gradient clipping for stability
        g_norm = np.abs(g).max()
        if g_norm > 0:
            g_clipped = g / (g_norm + 1e-30) * min(g_norm, 10.0)
        else:
            g_clipped = g

        log_update = np.log(y) - eta * g_clipped
        max_log = log_update.max()
        a_new = np.exp(log_update - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        # Track best
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive restart
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.15, 0.03 + 0.02 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Increase beta to be sharper after restart
            gamma = c0 * (1.0 + 0.15 * min(restart_count, 5))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      MD restart #{restart_count} t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta_t:.1f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta_t:.1f}, stall={stall_count}")

    return best_a, best_m


def freq_sculpt(a0, time_limit, eta=0.4, verbose=False):
    """Frequency domain amplitude sculpting to flatten the convolution."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        C = A * A
        C_mag = np.abs(C)
        C_phase = np.angle(C)

        C_mag_mean = C_mag.mean()
        if C_mag_mean < 1e-30:
            break

        # Adaptive eta: start aggressive, become gentle
        eta_t = eta * max(0.3, 1.0 - 0.01 * it)
        C_mag_new = C_mag_mean * (C_mag / C_mag_mean) ** (1.0 - eta_t)
        C_mag_new = np.maximum(C_mag_new, 1e-30)

        # Reconstruct A_new with modified magnitude
        A_new = np.sqrt(C_mag_new) * np.exp(1j * C_phase / 2.0)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.6 * a + 0.4 * a_new
            a = norm_sum1(a)

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      FreqSculpt it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def local_descent(a0, time_limit, verbose=False, seed=42):
    """Local coordinate descent: move mass from high-contributing positions to low ones."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M_curr = c.max()
        n_top = min(20, len(c))
        top_lags = np.argsort(c)[::-1][:n_top]

        improved = False

        # For each top lag, find the position contributing most and move mass
        for k in top_lags[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            # Positions contributing to c[k_int]: pairs (i, k_int-i)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue

            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]

            # Find best destination: position where moving mass reduces max conv
            # Try a few random destinations
            for _ in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                src_val = a[src]
                if src_val < 1e-12:
                    break

                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] = max(0, a_trial[src] - amt)
                    a_trial[dst] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      LocalDesc ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_u_shape(n, depth=1.0):
    """U-shaped sequence: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = 1.0 + depth * np.cos(np.pi * (2 * x - 1))
    a = np.maximum(a, 0.01)
    return norm_sum1(a)


def make_multi_dip(n, n_dips=3, depth=1.0):
    """Multi-dip sequence: several dips at regular intervals."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    for i in range(n_dips):
        center = (i + 1) / (n_dips + 1)
        width = 0.1 / n_dips
        a -= depth * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 0.01)
    return norm_sum1(a)


def make_log_u(n, power=1.5):
    """Logarithmic U-shape."""
    x = np.linspace(0.01, 0.99, n)
    a = x ** power + (1 - x) ** power
    a = np.maximum(a, 0.01)
    return norm_sum1(a)


def make_cosine_window(n, k=1):
    """Raised cosine window (low at edges)."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 - np.cos(2 * np.pi * k * x))
    a = np.maximum(a, 0.01)
    return norm_sum1(a)


def make_geom_u(n, ratio=0.7):
    """Geometric U-shape: a[i] = r^min(i, n-1-i)."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = ratio ** d
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coarse_to_fine(a_fine, coarse_n, time_limit, verbose=False):
    """Optimize at coarse resolution, then upsample and refine."""
    n = len(a_fine)
    # Downsample
    a_coarse = interpolate_sequence(a_fine, coarse_n)
    # Optimize at coarse level
    a_opt_coarse, _ = mirror_descent_optimize(
        a_coarse, time_limit * 0.4,
        beta_start=coarse_n / 50.0, beta_end=coarse_n / 3.0,
        c0=3.5, restart_every=1000, verbose=False
    )
    # Upsample back
    a_upsampled = interpolate_sequence(a_opt_coarse, n)
    # Refine at full resolution
    a_refined, _ = mirror_descent_optimize(
        a_upsampled, time_limit * 0.5,
        beta_start=n / 50.0, beta_end=n / 3.0,
        c0=2.5, restart_every=2000, verbose=False
    )
    return a_refined


def propose_candidate(budget_s=1800.0):

    best_score = float('inf')

        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

def compute_smooth_grad(a, c, beta):
    """Compute gradient of smooth max of convolution using FFT.
    
    f(a) = (1/beta) * log(sum_k exp(beta * c[k]))
    df/da[i] = 2 * sum_k w[k] * a[k-i] where w = softmax(beta*c)
    """
    n = len(a)
    bc = beta * c
    bc -= bc.max()
    w = np.exp(bc)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w = w / w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    grad = 2.0 * corr[:n]
    return grad

def make_multi_dip(n, n_dips=3, depth=1.0):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
