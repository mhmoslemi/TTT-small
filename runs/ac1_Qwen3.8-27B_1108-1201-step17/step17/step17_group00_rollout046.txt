        return None
    
    g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    gs = g_val.sum()
    if gs < 1e-15:
        return None
    return g_val / gs

def line_search(y, g, best_m, n):
    # Minimize f((1-t)*y + t*g) for t in [0,1]
    # Use golden section
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    
    def obj(t):
        a = (1-t)*y + t*g
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return best_m  # don't improve
        a = a / s
        return fast_autocorr(a).max()
    
    a, b = 0.0, 1.0
    c = b - gr*(b-a)
    d = a + gr*(b-a)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(12):
        if fc < fd:
            b = d; d = c; fd = fc; c = b - gr*(b-a); fc = obj(c)
        else:
            a = c; c = d; fc = fd; d = a + gr*(b-a); fd = obj(d)
    
    t_best = (a + b) / 2
    return t_best

Let me also think about the **parametric initialization** more carefully. What shapes are good for low autocorrelation?

For a sequence on [0,1], the goal is to make the autocorrelation as flat as possible (low peaks). This is related to the "Sidelobe" problem in radar/communications.

Good shapes:
- **Wigner distribution**: a(x) ≈ sqrt(x(1-x)) (triangular spectrum)
- **Raised cosine**: a(x) = 0.5*(1 + cos(pi*x)) on [0,1]
- **Gaussian**: a(x) = exp(-x^2/(2*sigma^2))
- **Optimal**: The sequence that equalizes all autocorrelation peaks

The theoretical minimum of J for large n is related to the "flat spectrum" problem. For a probability density on [0,1], the autocorrelation at lag 0 is integral a(x)^2 dx, and at other lags it's smaller. The ratio max_lag / integral^2 gives J/(2n).

For a uniform distribution a(x) = 1, the autocorrelation is triangular: (1-|k|/n), so max = 1 (at k=0). J = 2n*1/n^2 = 2/n... wait that doesn't seem right.

Let me recalculate. For a = (1/n) * ones(n):
- sum(a) = 1
- (a*a)_k = sum_i (1/n)*(1/n) = (n-|k|)/n^2 for |k| < n, and similar for |k| >= n
- Actually for linear convolution: (a*a)_k = sum_{i=max(0,k-n+1)}^{min(n-1,k)} (1/n)^2
  = (min(n-1,k) - max(0,k-n+1) + 1) / n^2
  At k=0: n/n^2 = 1/n
  At k=n-1: 1/n^2
- max = 1/n (at k=0)
- J = 2n * (1/n) / 1^2 = 2

So uniform gives J=2. We need to do better.

For a(x) = sin(pi*x) on [0,1] (approximately):
- This has more mass in the middle
- The autocorrelation should be more spread out

The previous best is 1.504, which is significantly better than 2.

For the "optimal" sequence, we want to equalize all the autocorrelation peaks. In the limit of large n, this is related to finding a function on [0,1] whose autocorrelation is as flat as possible.

I recall that for the problem of minimizing max_k (a*a)_k subject to sum(a)=1, a>=0, the optimal solution for large n approaches a specific profile. I believe it's related to the "optimal window" in signal processing.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-constraint LP-based Frank-Wolfe with Nesterov acceleration
2. Diverse parametric initializations
3. Multi-restart strategy
4. Coordinate descent polish
5. Multi-length exploration
"""

import numpy as np
import time
import cvxpy as cp
from scipy import sparse

# Global cache for convolution structure
_conv_cache = {}

def get_conv_indices(n):
    """Precompute row/col indices for the convolution constraint matrix."""
    if n in _conv_cache:
        return _conv_cache[n]
    size = 2 * n - 1
    rows_parts = []
    cols_parts = []
    for k in range(size):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        count = i_end - i_start
        rows_parts.append(np.full(count, k, dtype=np.int32))
        cols_parts.append(np.arange(i_start, i_end, dtype=np.int32))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    _conv_cache[n] = (rows, cols, size)
    print(f"  Conv structure for n={n}: {len(rows)} nonzeros")
    return _conv_cache[n]

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def get_solver():
    for s in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS

def solve_lp_oracle(y, solver, max_K=None):
    """Solve LP: min M s.t. sum(g)=1, g>=0, (y*g)_k <= M."""
    n = len(y)
    rows, cols, size = get_conv_indices(n)
    vals = y[rows - cols]
    
    if max_K is not None and size > max_K:
        ac = fast_autocorr(y)
        top_k = np.argsort(ac)[::-1][:max_K]
        C = sparse.csr_matrix((vals[rows == ...], ...))  # Need subsetting
        # Actually, let me subset differently
        # Build full then subset
        C_full = sparse.csr_matrix((vals, (rows, cols)), shape=(size, n))
        C = C_full[top_k, :].tocsr()
    else:
        C = sparse.csr_matrix((vals, (rows, cols)), shape=(size, n))
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(g) == 1, C @ g <= M]
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver, verbose=False, max_time=30)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    g_val = np.asarray(g.value, dtype=np.float64)
    g_val = np.maximum(g_val, 0.0)
    gs = g_val.sum()
    if gs < 1e-15:
        return None
    return g_val / gs

def fw_optimize(a0, time_limit, solver, max_K=None, verbose=True):
    """Frank-Wolfe with LP oracle and Nesterov acceleration."""
    n = len(a0)
    y = norm_sum1(a0.copy())
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    x_acc = y.copy()
    
    iter_count = 0
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov lookahead
        if t > 1:
            beta_t = 0.5 * (1.0 - np.sqrt((1.0 + (t-1.0)**2) / (1.0 + t**2)))
            # Actually use the standard beta_t = (t-1)/(t+2)
            beta_t = (t - 1.0) / (t + 2.0)
            y_look = x_acc + beta_t * (x_acc - y)
            y_look = np.maximum(y_look, 0.0)
            sl = y_look.sum()
            if sl > 1e-15:
                y_look = y_look / sl
            else:
                y_look = y.copy()
        else:
            y_look = y.copy()
        
        # Solve LP oracle at y_look (or at y for simpler version)
        g = solve_lp_oracle(y_look, solver, max_K)
        if g is None:
            # Fallback: use negative gradient direction
            ac = fast_autocorr(y)
            k_star = int(np.argmax(ac))
            grad = np.zeros(n)
            i_start = max(0, k_star - n + 1)
            i_end = min(n, k_star + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] = 2.0 * y[k_star - idx]
            # Simple gradient step
            eta = 1.0 / np.sqrt(t + 1)
            y_new = y - eta * grad
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy > 1e-15:
                y_new = y_new / sy
            else:
                y_new = y.copy()
            x_acc = y_new
            y = y_new
            iter_count += 1
            continue
        
        # Update y
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        # Update Nesterov point
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx > 1e-15:
                x_new = x_new / sx
            else:
                x_new = y_new.copy()
        
        # Check improvements
        M_new = fast_autocorr(y_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = y_new.copy()
        M_x = fast_autocorr(x_new).max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x_new.copy()
        
        x_acc = x_new
        y = y_new
        iter_count += 1
        
        if verbose and t % 25 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m

def coordinate_descent(a0, time_limit, rng_seed=42, verbose=False):
    """Coordinate descent: move mass from peak-contributing positions."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:15]
        improved = False
        
        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination
            scores = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                idx2 = np.arange(s2, e2 + 1)
                scores[idx2] += a[k2i - idx2]
            
            candidates = np.where(scores < np.median(scores))[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            dst = candidates[np.argmin(a[candidates])]
            if dst == src:
                continue
            
            found = False
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if not found:
                pass
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        else:
            stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def parametric_init(n, time_limit, best_a, best_J):
    """Try various parametric families."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    def try_update(a, label):
        nonlocal best_a, best_J
        a_n = norm_sum1(a)
        ac = fast_autocorr(a_n)
        val = 2.0 * n * ac.max()
        if val < best_J:
            best_J = val
            best_a = a_n.copy()
    
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
        try_update((i_arr + 1.0) ** p, f"p{p}")
        try_update((n - i_arr) ** p, f"ip{p}")
    
    # Beta-like
    for a_p in [0.5, 1.0, 1.5, 2.0]:
        for b_p in [0.5, 1.0, 1.5, 2.0]:
            try_update(np.maximum(i_arr + 1, 0.5)**a_p * np.maximum(n - i_arr, 0.5)**b_p, f"b{a_p}_{b_p}")
    
    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for s in [0.1, 0.2, 0.3, 0.4]:
            try_update(np.exp(-((x-c)**2)/(2*s*s)), f"g{c}_{s}")
    
    # Sine
    for f in [1, 2, 3]:
        try_update(np.maximum(np.sin(np.pi*f*x)**2, 0.01), f"s{f}")
    
    # Exp
    for r in [1, 2, 5, 10]:
        try_update(np.exp(-r*x), f"e{r}")
        try_update(np.exp(r*(x-1)), f"er{r}")
    
    # Sinc-like
    for f in [1, 2]:
        try_update(np.maximum(np.sinc(f*x-0.5)**2, 0.001), f"sinc{f}")
    
    # Blend with existing best
    w_best = best_a / best_a.sum()
    for w in [0.3, 0.5, 0.7]:
        try_update(w*w_best + (1-w)*w_best, f"bw{w}")  # This is just w_best, skip
        a_uniform = np.ones(n) / n
        try_update(w*w_best + (1-w)*a_uniform, f"bwu{w}")
    
    return best_a, best_J
