        M_new = b_new.max()
        
        # Check non-active constraints
        non_active = np.setdiff1d(np.arange(len(b_new)), active_set)
        if len(non_active) > 0:
            # Find if any non-active constraint is violated (b_new[k] > M_lp)
            # Actually, we should check if b_new[k] >= M_new (i.e., is at the max level)
            # Better: add the most "promising" non-active constraint
            max_non_active = b_new[non_active].max() if len(non_active) > 0 else 0
            if max_non_active > 0 and max_non_active >= M_new * 0.99:
                # Add the most violated one
                worst_non_active = non_active[np.argmax(b_new[non_active])]
                if len(active_set) < max_active * 2:
                    active_set = np.append(active_set, worst_non_active)
                else:
                    # Remove the least "binding" active constraint
                    # The least binding one is the one with the smallest b value
                    active_b_vals = b_new[active_set]
                    min_idx = np.argmin(active_b_vals)
                    active_set = np.delete(active_set, min_idx)
                    active_set = np.append(active_set, worst_non_active)
        
        iter_count += 1
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez iter={iter_count} ({elapsed:.1f}s): J={J:.8f}, |active|={len(active_set)}")
    
    return best_a, best_m

def get_solver():

    if _global_solver is not None:

def fast_conv(a):

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def solve_remez_lp(a, active_set, solver):
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    for k in active_set:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start > j_end:
            continue
        idx = np.arange(j_start, j_end + 1)
        coeffs = 2.0 * a[k_int - idx]
        if coeffs.max() < 1e-15:
            continue
        constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) <= M_var + b[k_int])
    if len(constraints) <= 1:
        return a.copy(), M
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return a.copy(), M
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), M
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), M
    a_new = a_new / s
    return a_new, M

def remez_method(a0, time_limit, solver, max_active=200, verbose=False, refresh_interval=10):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    b = fast_conv(a)
    active_set = set(np.argsort(b)[::-1][:min(max_active, len(b))].tolist())
    iter_count = 0
    fail_count = 0

    while time.monotonic() - t0 < time_limit:
        a_lp, M_lp = solve_remez_lp(a, sorted(active_set), solver)
        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            a_trial = (1 - t_frac) * a + t_frac * a_lp
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                fail_count = 0
                break
        if not accepted:
            fail_count += 1
            if fail_count > 20:
                break
        if accepted or iter_count % refresh_interval == 0:
            b_new = fast_conv(a)
            M_new = b_new.max()
            threshold = M_new * 0.99
            near_max = np.where(b_new >= threshold)[0]
            if len(near_max) > 0:
                if len(near_max) <= max_active:
                    active_set = set(near_max.tolist())
                else:
                    top_vals = b_new[near_max]
                    top_order = np.argsort(top_vals)[::-1][:max_active]
                    active_set = set(near_max[top_order].tolist())
        iter_count += 1
        if verbose and iter_count % 10 == 0:

            J = 2.0 * n * best_m
            print(f"    Remez iter={iter_count} ({elapsed:.1f}s): J={J:.8f}, |active|={len(active_set)}")
    return best_a, best_m

def make_init(n, kind='triangle'):

    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'parabolic':
        a = np.maximum(1.0 - 4.0 * (x - 0.5) ** 2, 0.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization: NMD (global) -> Remez (local) -> NMD (polish)
    -> Remez (polish) -> length exploration -> final burst.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1750.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 2600
        best_seq = make_init(n_init, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: NMD global optimization (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 1: NMD n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_mirror_descent(normalize(best_seq), t1, top_k=35, c0=5.0, verbose=True)
        try_update(a_out, f"NMD1_n{n0}")

    # === Phase 2: Remez main (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 2: Remez n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = remez_method(normalize(best_seq), t2, solver, max_active=180, verbose=True, refresh_interval=8)
        try_update(a_out, f"Remez1_n{len(best_seq)}")

    # === Phase 3: NMD polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: NMD polish n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = nesterov_mirror_descent(normalize(best_seq), t3, top_k=20, c0=2.0, verbose=True)
        try_update(a_out, f"NMD2_n{len(best_seq)}")

    # === Phase 4: Remez polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: Remez polish n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = remez_method(normalize(best_seq), t4, solver, max_active=250, verbose=True, refresh_interval=5)
        try_update(a_out, f"Remez2_n{len(best_seq)}")

    # === Phase 5: Multi-init exploration (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t5 = min(remaining * 0.10, 100.0)
        if t5 > 20:
            print(f"\n=== Phase 5: Multi-init NMD (budget={t5:.0f}s) ===")
            inits = ['raised_cos', 'gaussian', 'parabolic']
            t_per = t5 / len(inits)
            for kind in inits:
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining < 10:
                    break
                a_init = make_init(n0, kind)
                a_out, _ = nesterov_mirror_descent(a_init, min(t_per, remaining - 5),
                                                   top_k=30, c0=4.0, verbose=False)
                try_update(a_out, f"NMD_{kind}_n{n0}")

    # === Phase 6: Alternative length (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        for delta in [100, -100]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            n_alt = len(best_seq) + delta
            if n_alt < 500:
                continue
            a_alt = interpolate_sequence(best_seq, n_alt)
            t_alt = min(remaining * 0.4, 100.0)
            print(f"\n=== Phase 6: Alt length n={n_alt} (budget={t_alt:.0f}s) ===")
            if t_alt > 20:
                a_out, _ = nesterov_mirror_descent(a_alt, t_alt, top_k=25, c0=3.0, verbose=False)
                try_update(a_out, f"NMD_n{n_alt}")
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 30:
                    t_rem = min(remaining * 0.5, 80.0)
                    a_out2, _ = remez_method(normalize(a_out), t_rem, solver, max_active=150, verbose=False)
                    try_update(a_out2, f"Remez_n{n_alt}")

    # === Phase 7: Final Remez burst on global best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 7: Final Remez burst (budget={remaining:.0f}s) ===")
        a_out, _ = remez_method(normalize(best_seq), remaining - 5, solver,
                                 max_active=300, refresh_interval=5, verbose=False)
        try_update(a_out, f"Remez_final_{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
