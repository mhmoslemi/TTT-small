Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral annealing in the frequency domain (Metropolis-Hastings on power spectrum)
2. Iterative power-spectrum flattening with adaptive non-negativity projection
3. Fine-grained pair-exchange local search targeting top autocorrelation peaks
4. Multi-length interpolation search
5. Adaptive mirror descent with full-peak subgradient weighting
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interp_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / ss

def spectral_flatten_iter(a, n_iters, alpha):
    """Single pass of spectral flattening."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = (A * np.conj(A)).real
    P_mean = P.mean()
    if P_mean < 1e-30:
        return a.copy()
    P_ratio = P / P_mean
    P_target = P_mean * np.power(np.maximum(P_ratio, 1e-30), 1.0 - alpha)
    phase = np.angle(A)
    mag = np.sqrt(np.maximum(P_target, 1e-30))
    A_new = mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:

    return a_new / s_new

def spectral_annealing(a0, time_limit, T0=0.001, T_end=1e-5, decay=0.9995, verbose=False):
    """
    Simulated annealing in the power-spectrum domain.
    Perturb |DFT(a)|, project back, accept/reject.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = eval_score(a)
    cur_J = best_J
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    T = T0
    n_accepted = 0
    n_rejected = 0
    n_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        T = max(T_end, T * decay)
        
        # Compute current spectrum
        A = np.fft.rfft(a, fft_size)
        P = (A * np.conj(A)).real
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Generate spectral perturbation
        # Mix of: (1) multiplicative noise, (2) flattening bias
        noise_scale = 0.02 + 0.05 * (T / T0)
        noise = rng.normal(0, noise_scale, P.shape)
        P_pert = P * (1.0 + noise)
        
        # Also add a slight flattening bias
        flatten = 0.01 * (T / T0)
        P_pert = P_pert * (1.0 - flatten) + P_mean * flatten
        
        P_pert = np.maximum(P_pert, 1e-30)
        
        # Reconstruct
        phase = np.angle(A)
        mag = np.sqrt(P_pert)
        A_new = mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            T *= 0.9
            continue
        a_new = a_new / s_new
        
        # Evaluate
        J_new = eval_score(a_new)
        n_iters += 1
        
        if J_new < cur_J:
            a = a_new
            cur_J = J_new
            n_accepted += 1
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
        else:
            # Metropolis acceptance
            delta = J_new - cur_J
            if delta < 0:
                a = a_new
                cur_J = J_new
                n_accepted += 1
            else:
                accept_prob = np.exp(-delta / max(T, 1e-10))
                if rng.random() < accept_prob:
                    a = a_new
                    cur_J = J_new
                    n_accepted += 1
                else:
                    n_rejected += 1
        
        if verbose and n_iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpectralAnneal it={n_iters} ({elapsed:.1f}s): best_J={best_J:.6f}, cur_J={cur_J:.6f}, T={T:.6f}, acc={n_accepted}, rej={n_rejected}")
    
    return best_a, best_J

def pair_exchange_search(a0, time_limit, verbose=False):
    """
    Fine-grained local search: transfer small amounts between pairs of positions
    to reduce the maximum autocorrelation peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    best_J = 2.0 * n * best_M
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peaks
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            
            # Find the pairs (i, k-i) that contribute to this peak
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            # Contribution from each pair: a[i] * a[k-i]
            # (for i != k-i, the pair contributes twice to the convolution)
            pairs = {}
            for i in idx:
                j = k_int - i
                if j != i:
                    key = (min(i, j), max(i, j))
                    pairs[key] = 2.0 * a[i] * a[j]
                else:
                    key = (i, i)
                    pairs[key] = a[i] ** 2
            
            # Find the largest contributing pair
            if not pairs:
                continue
            max_pair = max(pairs, key=pairs.get)
            
            # Try to reduce this pair's contribution by moving mass
            src = max_pair[0]
            # Move mass from src to a random position that doesn't increase other peaks
            # Candidate destinations: positions where a is small
            small_positions = np.argsort(a)[:max(1, n // 10)]
            
            for dst in small_positions[:5]:
                if dst == src:
                    continue
                
                # Try several transfer amounts
                for frac in [0.001, 0.002, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M - 1e-16:
                        a = a_trial
                        best_M = M_trial
                        best_J = 2.0 * n * best_M
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_J = 2.0 * n * best_M
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairExchange ({elapsed:.1f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J
