where C = {b : T b <= M*1, b >= 0} and grad = -1.

But projecting onto C is hard (it's a polytope defined by 2n-1 + n constraints).

import numpy as np
import time
import cvxpy as cp

_solver_cache = None

def fast_autocorr(a):
    """Compute linear autocorrelation conv(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence using the global evaluator."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_solver():
    """Find the best available LP solver."""
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:

def build_T(y, K_sel):
    """Build constraint matrix rows for selected lags. T[idx, i] = y[k-i]."""
    n = len(y)
    T = np.zeros((len(K_sel), n))
    for idx in range(len(K_sel)):
        k_int = int(K_sel[idx])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_end > i_start:
            T[idx, i_start:i_end] = y[k_int - i_end + 1 : k_int - i_start + 1][::-1]
    return T

def lp_direction(y, M, T, solver):
    """Solve the LP to get the FW direction. Returns normalized direction or None."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(T.shape[0])]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception as e:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

def lp_fw_optimize(a0, time_limit, solver, verbose=True, use_all_constraints=True,
                   top_K=500, nesterov=True, restart_every=500, verbose_every=20):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    all_k = np.arange(2 * n - 1)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select constraints
        if use_all_constraints:
            K_sel = all_k
        else:
            K_sel = np.argsort(ac)[::-1][:min(top_K, 2*n-1)]

        T = build_T(y, K_sel)
        M_vec = np.full(len(K_sel), M_curr)

        # Solve LP
        b_dir = lp_direction(y, M_curr, T, solver)
        if b_dir is None:
            # Fallback: simple subgradient step
            k_star = int(np.argmax(ac))
            i_start = max(0, k_star - n + 1)
            i_end = min(n, k_star + 1)
            g = np.zeros(n)
            g[i_start:i_end] = y[k_star - i_end + 1 : k_star - i_start + 1][::-1]
            g_sum = g.sum()
            if g_sum > 0:
                b_dir = g / g_sum
            else:
                b_dir = np.ones(n) / n

        # Nesterov update
        if nesterov:
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * b_dir
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sy

            if t == 1:
                x_new = b_dir.copy()
            else:
                beta_t = (t - 1.0) / (t + 2.0)
                x_new = y_new + beta_t * (y_new - y)
                x_new = np.maximum(x_new, 0.0)
                sx = x_new.sum()
                if sx < 1e-15:
                    x_new = y_new.copy()
                else:
                    x_new = x_new / sx
        else:
            y_new = b_dir.copy()
            x_new = b_dir.copy()

        y = y_new
        x = x_new

        # Track best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        # Restart if stalled
        if t > restart_every:
            y = best_a.copy()
            x = best_a.copy()
            # Slight perturbation
            pert = np.random.default_rng(t).normal(0, 0.01, n) * best_a
            y = np.maximum(best_a + pert, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            x = y.copy()

        if verbose and t % verbose_every == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Aggressive coordinate descent: move mass from peak-contributing positions to low-sensitivity positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300
    n_top = 20

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity map
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            in_peak[idx] = True
            partner = ki - idx
            sens[idx] += a[partner]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            dst = candidates[np.argmin(cand_sens)]

            if dst == src:
                continue

            found = False
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if not found and stall > 50:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def make_u_shape(n, sharpness=2.0):
    """U-shaped distribution (high at edges, low in middle)."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)

def make_bimodal(n, width=0.15):
    """Bimodal distribution with peaks near edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.08) / width) ** 2) + np.exp(-((x - 0.92) / width) ** 2) + 0.003
    return norm_sum1(a)

def spectral_init(n, n_modes=20, seed=0):
    """Generate a smooth non-negative sequence via Fourier synthesis."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    n0 = len(best_seq)
    print(f"Starting n={n0}, score={best_score:.6f}")

    # === Phase 1: Quick multi-init exploration ===
    print(f"\n=== Phase 1: Multi-init LP-FW (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.15, 250.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, sharpness=1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, sharpness=3.0))
        labels.append("u3.0")
        inits.append(make_bimodal(n0, width=0.12))
        labels.append("bimodal")
        inits.append(spectral_init(n0, n_modes=25, seed=42))
        labels.append("spectral")

        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_u_shape(n0, 2.5)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(25.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 20:
                break
            print(f"  LP-FW init '{lbl}' (t_avail={t_avail:.0f}s)...")
            use_all = (n0 <= 3000)
            a_out, _ = lp_fw_optimize(init, t_avail, solver, verbose=False,
                                       use_all_constraints=use_all,
                                       top_K=min(1000, 2*n0-1),
                                       nesterov=True, restart_every=400)
            try_update(a_out, f"lp_{lbl}")

    # === Phase 2: Main LP-FW on best (long run with all constraints) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Main LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.45, 800.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 5000)
        a_out, _ = lp_fw_optimize(a_in, t2, solver, verbose=True,
                                   use_all_constraints=use_all,
                                   top_K=min(2000, 2*n_best-1),
                                   nesterov=True, restart_every=600,
                                   verbose_every=30)
        try_update(a_out, "lp_main")

    # === Phase 3: Coordinate + Peak-pair polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Polishing (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        t3a = t3 * 0.6
        a_out, _ = peak_pair_reduction(a_in, t3a, verbose=True, rng_seed=77)
        try_update(a_out, "peakpair")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t3b = min(remaining * 0.3, 150.0)
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t3b, verbose=True, rng_seed=456)
            try_update(a_out, "coord")

    # === Phase 4: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP-FW round 2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 5000)
        a_out, _ = lp_fw_optimize(a_in, t4, solver, verbose=True,
                                   use_all_constraints=use_all,
                                   top_K=min(3000, 2*n_best-1),
                                   nesterov=True, restart_every=800,
                                   verbose_every=50)
        try_update(a_out, "lp_r2")

    # === Phase 5: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final polish (n={n_best}) ===")
    if remaining > 20:
        t5 = min(remaining * 0.7, 100.0)
        if t5 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t5, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10 and n_best <= 4000:
            t5b = min(remaining * 0.6, 60.0)
            if t5b > 10:
                a_in = norm_sum1(best_seq)
                use_all = (n_best <= 3000)
                a_out, _ = lp_fw_optimize(a_in, t5b, solver, verbose=False,
                                           use_all_constraints=use_all,
                                           top_K=min(2000, 2*n_best-1),
                                           nesterov=True, restart_every=1000)
                try_update(a_out, "lp_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n_fb = 4200
        best_seq = make_u_shape(n_fb, sharpness=2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
