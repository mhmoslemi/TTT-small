def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_gradient(a, ac, tau):
    """Compute gradient of tau*log(sum exp(ac_k/tau)) w.r.t. a using FFT."""
    n = len(a)
    ac_scaled = ac / tau
    ac_scaled -= ac_scaled.max()
    exp_vals = np.exp(ac_scaled)
    p = exp_vals / exp_vals.sum()
    
    # grad[i] = 2 * sum_j p[i+j] * a[j] (for valid indices)
    # This equals 2 * (flip(a) * p)[i+n-1]
    a_flip = a[::-1].astype(np.float64)
    size = len(ac)
    fft_size = 1 << (size - 1).bit_length()
    Af = np.fft.rfft(a_flip, fft_size)
    Pf = np.fft.rfft(p.astype(np.float64), fft_size)
    conv_full = np.fft.irfft(Af * Pf, fft_size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return np.maximum(grad, 0.0)


def smooth_adam(a0, time_limit, tau_decay=0.9997, lr=0.8, beta1=0.9, beta2=0.999,
                restart_every=3000, verbose=True):
    """Multiplicative Adam with log-sum-exp smoothing and adaptive tau."""
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    tau = max(ac.max() * 0.5, 1e-6)
    tau_min = max(ac.max() * 0.005, 1e-10)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    stall = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        tau = max(tau * tau_decay, tau_min)
        
        g = lse_gradient(a, ac, tau)
        
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * (g * g)
        
        bc1 = 1.0 - beta1 ** t
        bc2 = 1.0 - beta2 ** t
        m_c = m_hat / bc1
        v_c = v_hat / bc2
        
        step = lr / (np.sqrt(v_c) + 1e-8)
        
        log_a = np.log(np.maximum(a, 1e-30)) - step * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        a_new = a_new / s
        a = a_new
        
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x = best_a + pert
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            a = x.copy()
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def parametric_init(n):
    """Generate diverse parametric initial sequences."""
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    candidates = []
    
    # Sin^2 family
    for freq in [1, 2, 3]:
        for phase in [0.0, 0.25, 0.5, 0.75]:
            a = np.maximum(np.sin(np.pi * freq * x + phase) ** 2, 0.01)
            candidates.append(a / a.sum())
    
    # Power law
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (i_arr + 1.0) ** p
        candidates.append(a / a.sum())
        a = (n - i_arr) ** p
        candidates.append(a / a.sum())
    
    # Beta-like
    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.5, 1.0, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            candidates.append(a / a.sum())
    
    # Gaussian bumps
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            candidates.append(a / a.sum())
    
    # Cosine taper
    a = np.cos(np.pi * (x - 0.5)) ** 2
    candidates.append(a / a.sum())
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Chirp-like
    for rate in [1, 3, 5]:
        a = 1.0 / (i_arr + 1.0) ** (1.0 / rate)
        candidates.append(a / a.sum())
    
    return candidates

            check_and_update(cand_blend, f"blend_{w}")
    
    n_best = len(best_seq)
    
    # === Phase 2: Smoothed Adam ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Smoothed Adam (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_adam(a_in, t2, tau_decay=0.9998, lr=0.6,
                               beta1=0.9, beta2=0.999, restart_every=2000, verbose=True)
        check_and_update(a_out, f"smooth_n{n_best}")
    
    # === Phase 3: LP-FW v2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW v2 (n={n_best}) ===")
    t3 = min(remaining * 0.30, 500.0)
    if t3 > 30 and n_best <= 5500:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_v2(a_in, t3, solver, initial_K=500,
                                      max_K=4000, add_per_iter=50, verbose=True)
        check_and_update(a_out, f"lp2_n{n_best}")
    
    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent_v2(a_in, t4, rng_seed=111, verbose=True)
        check_and_update(a_out, f"coord_n{n_best}")
    
    # === Phase 5: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    nearby_lengths = []
    for delta in [200, 400, 500, -200, -400]:
        new_n = n_best + delta
        if 2000 <= new_n <= 5500:
            nearby_lengths.append(new_n)
    
    seen = set()
    unique_lengths = []
    for ln in nearby_lengths:
        if ln not in seen:
            seen.add(ln)
            unique_lengths.append(ln)
    
    for new_n in unique_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print(f"\n=== Phase 5: n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)
        
        # Quick smoothed run
        t5a = min(remaining * 0.25, 150.0)
        if t5a > 20:
            a_out, _ = smooth_adam(a_long, t5a, tau_decay=0.9998, lr=0.5,
                                   beta1=0.9, beta2=0.999, restart_every=1500, verbose=False)
            check_and_update(a_out, f"smooth_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100 and new_n <= 5000:
            t5b = min(remaining * 0.35, 200.0)
            if t5b > 20:
                a_out, _ = lp_frank_wolfe_v2(norm_sum1(best_seq), t5b, solver,
                                              initial_K=400, max_K=3000,
                                              add_per_iter=40, verbose=False)
                check_and_update(a_out, f"lp2_n{new_n}")
    
    # === Phase 6: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP-FW (n={n_best}) ===")
    if remaining > 60 and n_best <= 5000:
        t6 = min(remaining * 0.40, 250.0)
        if t6 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_v2(a_in, t6, solver, initial_K=600,
                                          max_K=5000, add_per_iter=60, verbose=True)
            check_and_update(a_out, f"lp_final_n{n_best}")
    
    # === Phase 7: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.50, 100.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_v2(a_in, t7, rng_seed=271, verbose=False)
            check_and_update(a_out, f"coord_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
