import numpy as np

def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A.conj(), fft_size)[:size]

def evaluate_safe(a):

        return float(evaluate_sequence(a_list))

def norm_sum1(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_gradient_softmax(a, ac, beta):
    """
    Compute the subgradient of the log-sum-exp approximation to max(ac).

    # Compute softmax weights
    ac_clipped = np.minimum(ac, ac.max())  # numerical stability
    exp_vals = np.exp(beta * (ac_clipped - ac.max()))
    p = exp_vals / exp_vals.sum()
    
    # For efficiency, only use lags with significant weight
    threshold = p.max() * 1e-6
    active = p > threshold
    if active.sum() == 0:
        active = np.zeros(len(ac), bool)
        active[np.argmax(ac)] = True
    
    grad = np.zeros(n)
    active_idx = np.where(active)[0]

        j_idx = k_int - idx
        grad[idx] += 2.0 * wk * a[j_idx]
    
    return grad

def spectral_equalize(a, alpha=0.5, n_iter=10):
    """
    Push the spectrum toward flatness via iterative projection.
    alpha: blending factor (0 = no change, 1 = fully flat)
    """
    n = len(a)
    a = a.copy()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        
        # Blend toward flat
        P_new = (1 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 0)
        
        # Reconstruct with new amplitude, same phase
        phase = np.angle(A)
        mag = np.sqrt(P_new)
        A_new = mag * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:

        a = a_new / s
    
    return a

        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Use top-K lags for efficiency (those with highest softmax weight)
        # But also compute full softmax for the adaptive part
        g = compute_gradient_softmax(y, ac, beta=beta, )
        
        # Mirror descent step (exponential family / entropic mirror)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y + 1e-30) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart on stall
        if stall > restart_every:
            stall = 0
            # Perturb and restart
            strength = min(0.15, 0.03 + 0.02 * (total_iters // (restart_every * 10)))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            
            # Spectral equalization nudge
            x = spectral_equalize(x, alpha=0.3, n_iter=3)
            
            if verbose:
                J = 2.0 * n * best_M
                print(f"    AMD restart ({time.monotonic()-t0:.0f}s): J={J:.6f}, beta={beta:.1f}")
        
        if verbose and total_iters % 10000 == 0:
            J = 2.0 * n * best_M
            print(f"    AMD it={total_iters} ({time.monotonic()-t0:.0f}s): J={J:.6f}")
    
    return best_a, best_M

def local_perturbation_search(a0, time_limit, n_trials=500, verbose=False):
    """
    Local search via random perturbations: coordinate swaps, Gaussian kicks.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    trial = 0
    while time.monotonic() - t0 < time_limit and trial < n_trials:
        trial += 1
        a_try = a.copy()
        
        mode = rng.integers(0, 4)
        if mode == 0:
            # Coordinate swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-10:
                continue
            frac = rng.uniform(0.01, 0.15)
            amt = a[i] * frac
            a_try[i] -= amt
            a_try[j] += amt
        elif mode == 1:
            # Gaussian kick
            a_try += rng.normal(0, 0.005, n) * a
        elif mode == 2:
            # Block shift
            block_size = int(rng.integers(10, max(11, n // 10)))
            start = int(rng.integers(0, max(1, n - block_size)))
            shift = int(rng.integers(-5, 6))
            if shift != 0:
                a_try = np.roll(a_try[start:start+block_size], shift)
                a_try[start:start+block_size] = np.clip(
                    a_try[start:start+block_size], 0, None)
            # Actually this gets complicated, let's just do a simple version
            a_try = a.copy()
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.1)
                amt = a[i] * frac
                a_try[i] -= amt
                a_try[j] += amt
        else:
            # Mirror perturbation: flip a random segment
            i = int(rng.integers(n))
            length = int(rng.integers(5, max(6, n // 5)))
            j = min(n - 1, i + length)
            a_try[i:j+1] = np.flip(a[i:j+1])
        
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            continue
        a_try = a_try / s
        
        M_try = fast_autocorr(a_try).max()
        if M_try < best_M - 1e-14:
            best_M = M_try
            best_a = a_try.copy()
            a = a_try.copy()
    
    return best_a, best_M

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_structured_init(n, type_idx=0):
    """Create various structured initializations."""
    x = np.linspace(0, 1, n)
    if type_idx == 0:  # half-sine
        a = np.sin(np.pi * x) ** 2
    elif type_idx == 1:  # cosine window
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif type_idx == 2:  # Gaussian center
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif type_idx == 3:  # U-shape
        a = (x - 0.5) ** 2 + 0.1
    elif type_idx == 4:  # power law
        a = x ** 0.5
    elif type_idx == 5:  # multi-hump
        a = np.zeros(n)
        for c in [0.25, 0.5, 0.75]:
            a += np.exp(-20 * (x - c) ** 2)
    elif type_idx == 6:  # random smooth
        rng = np.random.default_rng(type_idx * 1000)
        a = np.ones(n) * 0.5
        for f in range(1, 8):
            a += rng.normal(0, 0.15/np.sqrt(f)) * np.sin(2*np.pi*f*x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) / n
    return norm_sum1(a)

def propose_candidate(budget_s=1800.0):
    """
    Main search function. Combines:
    1. Multi-init softmax-AMD at various n
    2. Spectral equalization
    3. Local perturbation search
    4. Final polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")
    
    if best_seq is None:
        n_base = 4000
        best_seq = make_structured_init(n_base, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # Phase 1: AMD on parent with different parameters
    print(f"\n=== Phase 1: Softmax-AMD on parent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 60:
        # Run AMD with different beta schedules
        for beta_s, beta_e, c0_val, label in [
            (100.0, 5.0, 4.0, "high_beta"),
            (50.0, 3.0, 5.0, "med_beta"),
            (200.0, 10.0, 3.5, "very_high_beta"),

                break
            t_avail = min(t1 / 3.0, remaining - 30)
            if t_avail < 20:
                break
            print(f"  AMD {label} (t={t_avail:.0f}s)...")
            a_out, _ = softmax_amd(norm_sum1(best_seq), t_avail,
                                    beta_start=beta_s, beta_end=beta_e, c0=c0_val)

        # Spectral equalization on best

            try_update(a_spec, f"spec_eq_n{n0}")
