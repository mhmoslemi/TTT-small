        # = [max(-k+1, -n+1), min(n-k, 1)]

        # I think the cleanest way is:
        # For each valid i (where both a[i] and a[k-i] exist):
        #   d(conv[k])/d(a[i]) += a[k-i]
        #   d(conv[k])/d(a[k-i]) += a[i]

                grad[partner] += wi * a[i_val]

Hmm, this subgradient computation is O(n * top_k) which could be slow for large n and top_k. Let me optimize it.

Actually, for a fixed k, the valid i range is [max(0, k-n+1), min(n-1, k)]. The derivative contribution to grad[j] from this k is:
- If j is in [max(0, k-n+1), min(n-1, k)]: grad[j] += a[k-j]
- If (k-j) is in [max(0, k-n+1), min(n-1, k)], i.e., j is in [k-min(n-1,k), k-max(0,k-n+1)]: grad[j] += a[j-k]

For the first condition: j in [max(0,k-n+1), min(n-1,k)], the contribution is a[k-j].
For the second condition: j-k in [max(0,k-n+1), min(n-1,k)], i.e., j in [k+max(0,k-n+1), k+min(n-1,k)].

If k < n: 
- First: j in [0, k], contribution a[k-j] (j goes 0 to k, k-j goes k to 0)
- Second: j in [k, k+n-1] intersected with [0, n-1] = [k, n-1], contribution a[j-k] (j-k goes 0 to n-1-k)

So for k < n:
- grad[0..k] gets contributions from the "first" term: a[k], a[k-1], ..., a[0]
- grad[k..n-1] gets contributions from the "second" term: a[0], a[1], ..., a[n-1-k]
- At j=k: both terms contribute a[0] + a[0] = 2*a[0] (if k < n/2... no, just 2*a[0])

Wait, at j=k: first term gives a[k-k] = a[0], second term gives a[k-k] = a[0]. Total: 2*a[0].

For k >= n:
- First: j in [k-n+1, n-1], contribution a[k-j] (k-j goes 1 to k-n+1... wait)
  j from k-n+1 to n-1: k-j from k-(k-n+1) to k-(n-1) = n-1 to k-n+1
  So contribution to grad[j] is a[k-j] where k-j ranges from n-1 down to k-n+1
- Second: j in [k, k+n-1] intersected with [0, n-1]. Since k >= n, k > n-1, so this is empty.

So for k >= n: only the first term applies, and it's just grad[j] += a[k-j] for j in [k-n+1, n-1].

This is getting complex. Let me just use a vectorized approach:

For a given k, the convolution value is:
conv[k] = sum_{i} a[i] * a[k-i]

The subgradient w.r.t. a is simply the "cross-correlation" of the indicator of valid indices with a shifted version of a.

Actually, the simplest vectorized approach: for a given k, the valid pairs are (i, k-i) where 0 <= i < n and 0 <= k-i < n. This means i in [max(0, k-n+1), min(n-1, k)].

The gradient contribution is: for each valid i, grad[i] += a[k-i] and grad[k-i] += a[i].

But we can vectorize: 
- indices1 = arange(max(0,k-n+1), min(n, k+1))
- indices2 = k - indices1
- grad[indices1] += a[indices2]  (this adds the contribution to each i from its partner k-i)
- grad[indices2] += a[indices1]  (this adds the contribution to each partner from i)

But this double-counts when i == k-i (i.e., 2i = k). In that case, the derivative is 2*a[i], which is what we'd get: grad[i] += a[i] from both lines.

Wait no, if i == k-i, then indices1 contains i and indices2 also contains i. So grad[i] gets a[i] from the first line and a[i] from the second line, total 2*a[i]. That's correct!

So the vectorized subgradient for a single k is:
```python
i_start = max(0, k - n + 1)
i_end = min(n, k + 1)
idx1 = np.arange(i_start, i_end)
idx2 = k - idx1
grad[idx1] += a[idx2]
grad[idx2] += a[idx1]
```

And for multiple k values with weights:

for w_k, k in zip(weights, top_indices):

This is O(n * top_k) in the worst case but typically much less since i_end - i_start is at most n.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
1. Nesterov-accelerated mirror descent with adaptive multi-peak subgradient
2. Frequency-domain spectral equalization (Gerchberg-Saxton variant)
3. Pairwise mass transport coordinate descent
4. Simulated annealing with structured perturbations
5. Multi-length exploration with interpolation

import numpy as np
import time

def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_nonneg(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def multi_peak_subgrad(a, conv, top_k=50, beta=5.0):
    """Compute weighted subgradient focusing on top peaks of convolution."""
    n = len(a)
    k_max = min(top_k, len(conv))
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.ones(n) / n
    w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx1 = np.arange(i_start, i_end)
        idx2 = k_int - idx1
        grad[idx1] += wi * a[idx2]
        grad[idx2] += wi * a[idx1]
    return grad

def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0,
                 restart_every=5000, verbose=False, rng_seed=42):
    """Nesterov-accelerated mirror descent for the convolution max minimization."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_conv = fast_conv(a)
    best_m = best_conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    n_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        beta_t = (n_iters - 1.0) / (n_iters + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        conv_y = fast_conv(y)
        g = multi_peak_subgrad(y, conv_y, top_k=cur_top_k, beta=cur_beta)
        
        eta = gamma / (np.sqrt(n_iters) + 1.0)
        log_a = np.log(y, where=y > 1e-30) - eta * g
        log_a = np.where(y > 1e-30, log_a, -30.0)
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        conv_x = fast_conv(x)
        M_new = conv_x.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            strength = min(0.08, 0.01 + 0.005 * (n_iters // restart_every))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 5, 100)
            cur_beta = max(cur_beta - 0.2, 2.0)
            gamma = c0 * (1.0 + 0.05 * min(n_iters // restart_every, 10))
    
    return best_a, best_m

def spectral_equalize(a0, time_limit, iters=5000, eta0=0.5, verbose=False):
    """Iterative spectral equalization to flatten the power spectrum."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_conv = fast_conv(a)
    best_m = best_conv.max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        eta = eta0 * (1.0 + 0.3 * it / max(1, iters))
        P_target = P_mean * (P / P_mean) ** (1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)
        
        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        conv_new = fast_conv(a_new)
        M_new = conv_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.7 * a + 0.3 * a_new
            a = normalize_nonneg(a)
    
    return best_a, best_m

def pairwise_swap_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent via pairwise mass transfers."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_conv = fast_conv(a)
    best_m = best_conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M_curr = conv.max()
        # Focus on indices where conv is large
        n_top = min(20, len(conv))
        top_k = np.argsort(conv)[::-1][:n_top]
        
        improved = False
        
        # Try moving mass from high-contribution positions to low-contribution positions
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            # Contribution of a[j] to conv[k] is a[k-j] (partner)
            partners = k_int - idx
            contrib = a[idx] * a[partners]
            
            if contrib.max() < 1e-20:
                continue
            
            # Source: position with highest contribution
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Try moving to various destinations
            for _ in range(20):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src] = src_val - amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        src_val = a[src]  # update for next fraction
                        break
                    elif frac > 0.01:
                        break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        else:
            stall = 0
    
    return best_a, best_m

def simulated_annealing(a0, time_limit, verbose=False, rng_seed=42):
    """Simulated annealing with structured perturbations."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_conv = fast_conv(a)
    best_m = best_conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    T = 0.1
    cool_rate = 0.9995
    min_T = 1e-6
    
    while time.monotonic() - t0 < time_limit and T > min_T:
        # Structured perturbation
        perturb_type = rng.integers(3)
        a_trial = a.copy()
        
        if perturb_type == 0:
            # Gaussian perturbation
            a_trial += rng.normal(0, T * 0.1, n)
        elif perturb_type == 1:
            # Pair swap
            i, j = rng.integers(0, n, 2)
            if i != j:
                amt = rng.uniform(0.01, 0.1) * min(a[i], a[j], a.sum() * 0.05)
                a_trial[i] -= amt
                a_trial[j] += amt
        else:
            # Block perturbation
            block_len = rng.integers(5, max(6, n // 10))
            start = rng.integers(0, max(1, n - block_len))
            a_trial[start:start+block_len] *= np.exp(rng.normal(0, T * 0.2, block_len))
        
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            T *= cool_rate
            continue
        a_trial = a_trial / s_t
        
        M_trial = fast_conv(a_trial).max()
        M_curr = fast_conv(a).max()
        
        if M_trial < M_curr:
            a = a_trial
            if M_trial < best_m:
                best_m = M_trial
                best_a = a_trial.copy()
        elif rng.uniform() < np.exp(-(M_trial - M_curr) / max(T, 1e-10)):
            a = a_trial
        
        T *= cool_rate
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_initial(n, kind='cosine'):
    x = np.linspace(0, 1, n)
    if kind == 'cosine':
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif kind == 'sinc':
        a = np.sinc((x - 0.5) * 8.0) ** 2
    elif kind == 'uniform':
        a = np.ones(n)
    elif kind == 'gaussian':
        a = np.exp(-10 * (x - 0.5) ** 2)
    elif kind == 'bimodal':
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.001)
    return normalize_nonneg(a)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
1. Nesterov-accelerated mirror descent with adaptive multi-peak subgradient
2. Frequency-domain spectral equalization (Gerchberg-Saxton variant)
3. Pairwise mass transport coordinate descent
4. Simulated annealing with structured perturbations
5. Multi-length exploration with interpolation

import numpy as np
import time

_results = {'best_seq': None, 'best_score': float('inf')}

def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize_nonneg(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def multi_peak_subgrad(a, conv, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(conv))
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.ones(n) / n
    w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx1 = np.arange(i_start, i_end)
        idx2 = k_int - idx1
        grad[idx1] += wi * a[idx2]
        grad[idx2] += wi * a[idx1]
    return grad

def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0,

    best_a = a.copy()

        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        conv_y = fast_conv(y)
        g = multi_peak_subgrad(y, conv_y, top_k=top_k, beta=beta)
        
        eta = gamma / (np.sqrt(n_iters) + 1.0)
        log_y = np.where(y > 1e-30, np.log(np.maximum(y, 1e-30)), -30.0)
