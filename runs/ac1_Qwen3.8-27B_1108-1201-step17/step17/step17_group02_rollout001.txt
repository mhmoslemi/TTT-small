def fast_autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:

def norm_sum1(a):
    """Normalize to sum = 1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_objective(a):
    """Compute J(a) for normalized sequence (sum=1)."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

# Parametric families
def param_sin_power(n, p):
    """a[i] = sin(pi*(i+0.5)/n)^p"""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** p
    return norm_sum1(a)

def param_cos_mod(n, alpha, beta=2.0):
    """a[i] = (1 + alpha*cos(2*pi*i/n)) * sin(pi*(i+0.5)/n)^beta"""
    x = np.linspace(0, 1, n)
    a = (1 + alpha * np.cos(2 * np.pi * x)) * np.sin(np.pi * x) ** beta
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return param_sin_power(n, beta)
    return a / s

def param_fourier(n, coeffs, base_power=2.0):
    """a[i] = sin(pi*(i+0.5)/n)^base_power * (1 + sum c_k*cos(2*pi*k*i/n))"""
    x = np.linspace(0, 1, n)
    base = np.sin(np.pi * x) ** base_power
    mod = np.ones(n)
    for k, c in enumerate(coeffs, 1):
        mod += c * np.cos(2 * np.pi * k * x)
    a = base * mod
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return base / base.sum()
    return a / s

def mirror_descent(a0, time_limit, verbose=True, seed=42):
    """Mirror descent with adaptive temperature and Nesterov momentum."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    temp = 0.5  # Temperature for soft subgradient
    min_temp = 0.05
    temp_decay = 0.9995
    stall = 0
    best_stall = 0
    c0 = 4.0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov step
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute autocorrelation and soft subgradient
        ac = fast_autocorr(y)
        # Soft subgradient: weight peaks by exp(ac/temp)
        ac_clipped = np.maximum(ac, 1e-30)
        weights = np.exp((ac_clipped - ac.max()) / temp)
        # Focus on top peaks
        weights[ac < ac.max() * 0.5] = 0  # Only consider peaks above 50% of max
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.zeros(len(ac))
            weights[np.argmax(ac)] = 1.0
        else:
            weights = weights / w_sum
        
        # Compute subgradient
        g = np.zeros(n)
        for k, w in zip(np.nonzero(weights)[0], weights[np.nonzero(weights)[0]]):
            if w < 1e-10:
                continue
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            g[idx] += 2.0 * w * y[j_idx]
        
        # Mirror descent update
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive temperature
        temp = max(min_temp, temp * temp_decay)
        
        # Restart
        if stall > 5000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            temp = 0.3  # Reset temperature
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}, temp={temp:.4f}, stall={stall}")
    
    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    # Start with moderate K, increase over time
    K_cur = min(200, 2 * n - 1)
    K_max = 2 * n - 1
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Select most binding constraints
        if K_cur >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            # Use top K peaks as binding constraints
            binding_idx = np.argsort(ac)[::-1][:K_cur]
        
        # Build constraint matrix (sparse)
        # T[k, j] = y[k-j] if 0 <= k-j < n
        # We'll build it as a dense matrix for small K
        rows = len(binding_idx)
        T = np.zeros((rows, n))
        for row_i, k in enumerate(binding_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            j_idx = ki - idx
            T[row_i, idx] = y[j_idx]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(rows)]
        # Add a normalizing constraint to prevent b from being too small
        constraints.append(cp.sum(b_var) >= 0.1)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov FW step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        # Track best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        # Adapt K
        if t % 10 == 0:
            K_cur = min(K_cur + 20, K_max)
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-W t={t} ({elapsed:.0f}s): J={J:.6f}, K={K_cur}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:10]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination (low contribution to peaks)
            dst = int(rng.integers(n))
            
            found = False
            for frac in [0.01, 0.005, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if not found:
                # Try random source
                for _ in range(3):
                    src2 = int(rng.integers(n))
                    if a[src2] < 1e-10 or src2 == dst:
                        continue
                    for frac in [0.005, 0.01, 0.02]:
                        amt = a[src2] * frac
                        a_trial = a.copy()
                        a_trial[src2] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = a_trial / a_trial.sum()
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found = True
                            break
                    if found:
                        break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.003, 0.02)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m
