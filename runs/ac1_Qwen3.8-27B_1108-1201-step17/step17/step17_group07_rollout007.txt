def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:n]

def fast_conv(a):
    """Full autocorrelation (a*a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Compute H @ v where H_ij = w_{i+j}. w has length 2n-1, v has length n. O(n log n)."""
    n = len(v)
    u_rev = w[::-1]
    conv = fft_convolve(u_rev, v)

    return conv[n-1:2*n-1][::-1]

def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1) / rho > 0
    rho_star = np.max(rho[cond])
    theta = (cssv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - theta, 0)

        # Line search: blend between a and a_cand
        accepted = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - alpha) * a + alpha * a_cand
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m * (1 - 1e-12):
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                break
        
        if not accepted:
            # Try small perturbation
            a_cand2 = a + 0.01 * (a_cand - a)
            a_cand2 = np.maximum(a_cand2, 0)
            a_cand2 = a_cand2 / a_cand2.sum()
            m2 = fast_conv(a_cand2).max()
            if m2 < best_m:
                a = a_cand2
                best_m = m2
                best_a = a.copy()
        
        if verbose and outer % 10 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    SoftMax QP outer={outer} ({elapsed:.1f}s): score={score:.8f}, sigma={sigma:.4f}")
    
    return best_a, best_m

def lp_remez_polish(a0, time_limit, solver=None, max_cons=60, r_init=0.1, verbose=False):
    """LP Remez polishing."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv(a).max()
    a = a / s
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Active set: lags near the max
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) > max_cons:
            # Use the top max_cons
            active = np.argsort(b)[::-1][:max_cons]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # LP: minimize M_var
        # s.t. sum(a) = 1
        #      a >= 0
        #      2 * sum_j a'_j * a[k-j] - M_var <= b[k] for k in active (linearization)
        #      a' within trust region of a
        
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1]
        
        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                coeffs = 2.0 * a[k_int - idx]
                constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
        
        # Trust region
        constraints.append(a_var >= a * (1.0 - r))
        constraints.append(a_var <= a * (1.0 + r))
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
        except Exception:
            no_improve += 1
            r = max(0.005, r * 0.7)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 1
            r = max(0.005, r * 0.7)
            continue
        
        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            no_improve += 1
            r = max(0.005, r * 0.7)
            continue
        a_new = a_new / s_new
        
        # Check improvement
        m_new = fast_conv(a_new).max()
        if m_new < best_m * (1 - 1e-12):
            a = a_new
            best_m = m_new
            best_a = a.copy()
            no_improve = 0
            r = min(0.5, r * 1.05)
        else:
            no_improve += 1
            r = max(0.003, r * 0.8)
            if no_improve > 30:
                # Perturb
                rng = np.random.default_rng(t)
                idx = rng.choice(n, size=min(n//5, 500), replace=False)
                a[idx] *= rng.uniform(0.8, 1.2, len(idx))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                r = 0.05
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m
            print(f"    LP Remez t={t} ({elapsed:.1f}s): score={score:.8f}, r={r:.4f}")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, rng_seed=42, verbose=False):
    """Coordinate descent with random moves."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, fast_conv(a).max()
    a = a / s
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 100)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_moves = min(200, n // 5)
        
        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.SCS

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.1)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(budget_s=1800.0):
    """
    Main search: Soft-max QP with FISTA (Hankel Hessian via FFT) + LP Remez + CD.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = make_init(2600, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_parent = len(best_seq)
    
    # === Phase 1: Soft-max QP on parent length (50% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.50, 800.0)
    print(f"\n=== Phase 1: SoftMax QP n={n_parent} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_out, _ = soft_max_qp_method(best_seq, n_parent, t1, 
                                       sigma_start=1.0, sigma_end=0.02,
                                       n_outer=100, fista_iters=100, verbose=True)
        try_update(a_out, f"SoftMax_n{n_parent}")
    
    # === Phase 2: LP Remez polish (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: LP Remez n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_remez_polish(best_seq, t2, solver=solver, max_cons=80, r_init=0.08, verbose=True)
        try_update(a_out, f"Remez_n{len(best_seq)}")
