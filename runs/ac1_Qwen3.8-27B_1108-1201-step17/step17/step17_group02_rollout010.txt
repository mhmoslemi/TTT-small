def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_max_approx(v, tau=1.0):
    """Smooth approximation of max using log-sum-exp."""
    v = np.asarray(v, dtype=np.float64)
    v_max = v.max()
    if v_max <= 0:
        return 0.0, v / max(len(v), 1)
    shifted = v / tau - v_max / tau
    exp_shifted = np.exp(shifted)
    M = v_max + tau * np.log(exp_shifted.sum())
    w = exp_shifted / exp_shifted.sum()
    return M, w


def eval_score(a):
    """Compute the score for a sequence (normalized to sum=1)."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    ac = fast_autocorr(a_norm)
    return float(2.0 * n * ac.max())

def smooth_max_optimize(a0, time_limit, tau_initial=0.01, tau_final=0.001, 
                        lr_initial=0.1, momentum=0.9, verbose=True, seed=42):
    """Optimize using smooth max of autocorrelation with SGD + momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    v = np.zeros(n)  # momentum
    lr = lr_initial
    tau = tau_initial
    stall = 0
    max_stall = 500
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal tau
        tau = tau_initial * (tau_final / tau_initial) ** (min(t, 10000) / 10000.0)
        lr = lr_initial / (1.0 + 0.01 * t)
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Compute subgradient
        top_k = min(100, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        w_top = ac[top_idx] - ac[top_idx].min() + 1e-10
        w_top = w_top / w_top.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for ki, wki in zip(top_idx, w_top):
            ki = int(ki)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            partner = ki - idx
            valid = (partner >= 0) & (partner < n)
            if np.any(valid):
                grad[idx[valid]] += 2.0 * wki * a[partner[valid]]
        
        # SGD step with momentum
        v = momentum * v - lr * grad
        a_new = a + v
        
        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        M_new = fast_autocorr(a).max()
        
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothMax t={t} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.5f}, stall={stall}")
        
        if stall > max_stall:
            # Perturb and restart
            pert = rng.normal(0, 0.02, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            v = np.zeros(n)
            stall = 0
            lr = lr_initial * 0.5
    
    return best_a, best_m

                T_sel[p, i_start:i_end+1] = y[i_start:i_end+1][::-1]
        
        # LP: maximize sum(b) subject to T_sel @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(len(binding_idx))]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        t = iter_count + 1
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        beta_t = (t - 1.0) / (t + 2.0)
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and iter_count % 2 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")
        
        # Grow constraint set
        K_sel = min(K_sel + grow_rate, 2 * n - 1)
        
        # Check time per iteration
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_m


def peak_reduction_v2(a0, time_limit, verbose=True, seed=77):
    """Aggressive peak reduction via targeted coordinate moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 300
    n_top = 20
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the pair with maximum contribution
            best_local = np.argmax(contrib)
            src = idx[best_local]
            src_partner = partner[best_local]
            
            if a[src] < 1e-12:
                continue
            
            # Compute sensitivity: how much does each position contribute to peaks?
            # For each position d, sum over top peaks of a[partner] where d is involved
            sens = np.zeros(n)
            for kp in top_k[:10]:
                kp = int(kp)
                s2 = max(0, kp - n + 1)
                e2 = min(n - 1, kp)
                idx2 = np.arange(s2, e2 + 1)
                partner2 = kp - idx2
                sens[idx2] += a[partner2]
            
            # Find minimum sensitivity destination (excluding src and partner)
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            mask[src_partner] = False
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            cand_sens = sens[candidates]
            dst = candidates[np.argmin(cand_sens)]
            
            if dst == src:
                continue
            
            # Try to move mass from src to dst
            found = False
            for frac in [0.01, 0.005, 0.02, 0.03, 0.05, 0.08, 0.15, 0.25]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if not found and stall > 50:
                # Random kick
                j = int(rng.integers(n))
                if j != src and j != src_partner:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if not improved:
            stall += 1
            # Multi-point perturbation
            if stall % 10 == 0:
                n_pert = rng.integers(3, min(20, n))
                perturb_idx = rng.choice(n, size=n_pert, replace=False)
                for pidx in perturb_idx:
                    j = int(rng.integers(n))
                    if j != pidx:
                        amt = a[pidx] * rng.uniform(0.01, 0.1)
                        a[pidx] -= amt
                        a[j] += amt
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakRed ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakRed final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def adam_optimize(a0, time_limit, verbose=True, seed=42):
    """Adam optimizer on smooth max of autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Compute subgradient (top-k weighted)
        top_k = min(80, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        # Softmax weights over top peaks
        vals = ac[top_idx]
        val_min = vals.min()
        w = np.exp((vals - val_min) / 0.01)
        w = w / w.sum()
        
        grad = np.zeros(n)
        for ki, wki in zip(top_idx, w):
            ki = int(ki)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            partner = ki - idx
            valid = (partner >= 0) & (partner < n)
            if np.any(valid):
                grad[idx[valid]] += 2.0 * wki * a[partner[valid]]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to simplex
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
        
        # Decay learning rate
        if t % 500 == 0:
            lr *= 0.95
        
        # Periodic perturbations
        if t % 2000 == 0:
            pert = rng.normal(0, 0.01, n) * best_a
            a_cand = best_a + pert
            a_cand = np.maximum(a_cand, 1e-15)
            a_cand = a_cand / a_cand.sum()
            M_cand = fast_autocorr(a_cand).max()
            if M_cand < best_m:
                a = a_cand
                best_m = M_cand
                best_a = a_cand.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Adam t={t} ({elapsed:.0f}s): J={J:.6f}, lr={lr:.4f}")
    
    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return a / a.sum()


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return a / a.sum()


def make_sine(n, power=2):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** power
    return a / a.sum()


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def pair_swap_optimize(a0, time_limit, verbose=True, seed=321):
    """Direct pair-swap optimization: move mass between pairs of positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 500
    n_top = 30
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        # For each top peak, try to reduce it by moving mass
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            
            # Sort by contribution (most important first)
            order = np.argsort(contrib)[::-1]
            
            for oi in order[:5]:  # Try top 5 contributors
                src = idx[oi]
                src_p = partner[oi]
                if a[src] < 1e-14:
                    continue
                
                # Try moving to various destinations
                # Prefer positions that don't contribute to any top peak
                best_reduction = 0
                best_dst = -1
                
                # Sample candidate destinations
                n_cand = min(50, n)
                cand_idx = rng.choice(n, size=n_cand, replace=False)
                
                for dst in cand_idx:
                    if dst == src or dst == src_p:
                        continue
                    for frac in [0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < M_curr - 1e-16:
                            reduction = M_curr - M_trial
                            if reduction > best_reduction:
                                best_reduction = reduction
                                best_dst = dst
                
                if best_dst >= 0:
                    amt = a[src] * 0.1
                    a[src] -= amt
                    a[best_dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Larger perturbation
                n_pert = rng.integers(5, min(50, n))
                i_idx = rng.choice(n, size=n_pert, replace=False)
                j_idx = rng.choice(n, size=n_pert, replace=False)
                for ii, jj in zip(i_idx, j_idx):
                    if ii != jj and a[ii] > 1e-10:
                        amt = a[ii] * rng.uniform(0.05, 0.3)
                        a[ii] -= amt
                        a[jj] += amt
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairSwap ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Strategy:
    1. Load parent or initialize
    2. Phase 1: Adam optimization (fast gradient-based)
    3. Phase 2: LP-Frank-Wolfe (constraint-based)
    4. Phase 3: Peak reduction (coordinate moves)
    5. Phase 4: LP-FW with more constraints
    6. Phase 5: Final polish
    7. If time remains, try different lengths
    """
    global _best_seq, _best_score
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)
    
    _best_seq = None
    _best_score = float('inf')
    
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    def try_update(a, label):
        nonlocal _best_seq, _best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        ac = fast_autocorr(a_norm)
        J = 2.0 * len(a_norm) * ac.max()
        if np.isfinite(J) and J < _best_score:
            _best_score = J
            _best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_norm = parent / parent.sum()
            p_ac = fast_autocorr(p_norm)
            p_J = 2.0 * len(p_norm) * p_ac.max()
            print(f"Parent: n={len(parent)}, J={p_J:.6f}")
            _best_seq = p_norm.copy()
            _best_score = p_J
    except Exception:
        pass
    
    if _best_seq is None:
        n_base = 4200
        _best_seq = make_sine(n_base, power=2)
        _best_score = eval_score(_best_seq)
        print(f"Default init: n={len(_best_seq)}, J={_best_score:.6f}")
    
    n0 = len(_best_seq)
    a_work = _best_seq.copy()
    
    # === Phase 1: Adam (fast, 30% of time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: Adam optimization (n={n0}, t={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = adam_optimize(a_work, t1, verbose=True, seed=42)
        try_update(a_out, "adam1")
        a_work = _best_seq.copy()
    
    # === Phase 2: LP-FW (25% of time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t2 = min(remaining * 0.25, 400.0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_cur}, t={t2:.0f}s) ===")
    if t2 > 30 and n_cur <= 5000:
        # Use top-K constraints, growing
        K0 = min(400, 2 * n_cur - 1)
        a_out, _ = lp_frank_wolfe_v2(a_work, t2, solver, verbose=True, top_K=K0, grow_rate=30)
        try_update(a_out, "lp1")
        a_work = _best_seq.copy()
    
    # === Phase 3: Peak reduction (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t3 = min(remaining * 0.15, 250.0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 3: Peak reduction (n={n_cur}, t={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = peak_reduction_v2(a_work, t3, verbose=True, seed=77)
        try_update(a_out, "peak1")
        a_work = _best_seq.copy()
    
    # === Phase 4: LP-FW with more constraints (15% of time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t4 = min(remaining * 0.15, 250.0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 4: LP-FW expanded (n={n_cur}, t={t4:.0f}s) ===")
    if t4 > 30 and n_cur <= 5000:
        K0 = min(800, 2 * n_cur - 1)
        a_out, _ = lp_frank_wolfe_v2(a_work, t4, solver, verbose=True, top_K=K0, grow_rate=80)
        try_update(a_out, "lp2")
        a_work = _best_seq.copy()
    
    # === Phase 5: Adam with different params (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t5 = min(remaining * 0.10, 150.0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 5: Adam round 2 (n={n_cur}, t={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = adam_optimize(a_work, t5, verbose=True, seed=999)
        try_update(a_out, "adam2")
        a_work = _best_seq.copy()
    
    # === Phase 6: Peak reduction round 2 (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t6 = min(remaining * 0.10, 150.0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 6: Peak reduction round 2 (n={n_cur}, t={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = peak_reduction_v2(a_work, t6, verbose=True, seed=456)
        try_update(a_out, "peak2")
    
    # === Phase 7: Length exploration (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 7: Length exploration (remaining={remaining:.0f}s) ===")
    
    trial_lengths = []
    for delta in [500, 1000, -500, 1500, -1000]:
        new_n = n_cur + delta
        if 2000 <= new_n <= 5000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    
    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        
        # Interpolate current best to new length
        x_old = np.linspace(0, 1, len(_best_seq))
        x_new = np.linspace(0, 1, new_n)
        a_interp = np.interp(x_new, x_old, _best_seq)
        a_interp = np.maximum(a_interp, 0.0)
        a_interp = a_interp / a_interp.sum()
        
        t7 = min(remaining * 0.4, 200.0)
        if t7 > 20:
            a_out, _ = adam_optimize(a_interp, t7, verbose=False, seed=new_n % 1000)
            try_update(a_out, f"adam_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 40 and new_n <= 4000:
            t7b = min(remaining * 0.3, 100.0)
            if t7b > 15:
                a_out, _ = lp_frank_wolfe_v2(a_interp, t7b, solver, verbose=False, 
                                              top_K=300, grow_rate=20)
                try_update(a_out, f"lp_n{new_n}")
    
    # === Final: one more LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_cur = len(_best_seq)
    if remaining > 30 and n_cur <= 5000:
        print(f"\n=== Final LP-FW (n={n_cur}, t={remaining:.0f}s) ===")
        a_out, _ = lp_frank_wolfe_v2(_best_seq, remaining * 0.8, solver, verbose=True,
                                      top_K=600, grow_rate=100)
        try_update(a_out, "lp_final")
    
    # Final validation
    if _best_seq is not None:
        final_score = eval_score(_best_seq)
        print(f"\nFINAL: J={final_score:.6f}, n={len(_best_seq)}, time={time.monotonic()-t_start:.0f}s")
        if final_score < _best_score:

def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute J = 2*n*max(ac)/(sum(a))^2 for a non-negative sequence."""

def compute_gradient_topk(a, top_k=80):
    """Compute subgradient of max autocorrelation using top-k weighted peaks."""
    n = len(a)
    ac = fast_autocorr(a)
    k_sel = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_sel]
    vals = ac[top_idx]
    val_min = vals.min()
    # Softmax-like weights
    w = np.exp((vals - val_min) * 20.0)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_sel) / k_sel
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for ki, wki in zip(top_idx, w):
        ki = int(ki)
        i_start = max(0, ki - n + 1)
        i_end = min(n - 1, ki)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        partner = ki - idx
        valid = (partner >= 0) & (partner < n)
        if np.any(valid):
            grad[idx[valid]] += 2.0 * wki * a[partner[valid]]
    return grad


def adam_optimize(a0, time_limit, top_k=80, lr_init=0.05, verbose=True, seed=42):
    """Adam optimizer on smooth max of autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:

    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    cur_top_k = top_k
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        grad = compute_gradient_topk(a, top_k=cur_top_k)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to simplex
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new

                stall = 0

def lp_frank_wolfe(a0, time_limit, solver, top_K=400, grow_rate=40, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration and growing constraint set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    iter_count = 0
    K_sel = min(top_K, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-18:
            break

        # Select binding constraints (top-K peaks)
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix
        T_sel = np.zeros((len(binding_idx), n))
        for p, k in enumerate(binding_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n - 1, ki)
            if i_start <= i_end:
                # T_sel[p, i] = y[ki - i] for i in [i_start, i_end]
                T_sel[p, i_start:i_end + 1] = y[i_start:i_end + 1][::-1]

        # LP: maximize sum(b) subject to T_sel @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(len(binding_idx))]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov accelerated step
        t = iter_count + 1
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        beta_t = (t - 1.0) / (t + 2.0)
        x_new = y_new + beta_t * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and iter_count % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

        # Grow constraint set
        K_sel = min(K_sel + grow_rate, 2 * n - 1)

        if time.monotonic() - t0 > time_limit:
            break

    return best_a, best_m

def spectral_init(n, n_modes=20, seed=0):

def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, a_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_norm = parent / parent.sum()
            p_ac = fast_autocorr(p_norm)
            p_J = float(2.0 * len(p_norm) * p_ac.max())
            print(f"Parent: n={len(parent)}, J={p_J:.6f}")
            _best_seq = p_norm.copy()
            _best_score = p_J
    except Exception:
        pass

    if _best_seq is None:
        n_base = 4200
        _best_seq = make_sine(n_base, power=2)
        _best_score = eval_score(_best_seq)
        print(f"Default init: n={len(_best_seq)}, J={_best_score:.6f}")

    n0 = len(_best_seq)
    a_work = _best_seq.copy()

    # === Phase 1: Adam optimization (25% of budget) ===
    remaining = time_budget - (time.monotonic() - _t_start)
    t1 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 1: Adam (n={n0}, t={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = adam_optimize(a_work, t1, top_k=80, lr_init=0.05, verbose=True, seed=42)
        try_update_global(a_out, "adam1")
        a_work = _best_seq.copy()

    # === Phase 2: LP-FW (20% of budget) ===
    remaining = time_budget - (time.monotonic() - _t_start)
    t2 = min(remaining * 0.20, 350.0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_cur}, t={t2:.0f}s) ===")
    if t2 > 30 and n_cur <= 5000:
        K0 = min(350, 2 * n_cur - 1)
        a_out, _ = lp_frank_wolfe(a_work, t2, solver, top_K=K0, grow_rate=30, verbose=True)
        try_update_global(a_out, "lp1")
        a_work = _best_seq.copy()

    # === Phase 3: Peak reduction (15% of budget) ===
    remaining = time_budget - (time.monotonic() - _t_start)
    t3 = min(remaining * 0.15, 250.0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 3: Peak reduction (n={n_cur}, t={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = peak_reduction(a_work, t3, n_top=25, verbose=True, seed=77)
        try_update_global(a_out, "peak1")
        a_work = _best_seq.copy()

    # === Phase 4: LP-FW expanded (15% of budget) ===
    remaining = time_budget - (time.monotonic() - _t_start)
    t4 = min(remaining * 0.15, 250.0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 4: LP-FW expanded (n={n_cur}, t={t4:.0f}s) ===")
    if t4 > 30 and n_cur <= 5000:
        K0 = min(700, 2 * n_cur - 1)
        a_out, _ = lp_frank_wolfe(a_work, t4, solver, top_K=K0, grow_rate=60, verbose=True)
        try_update_global(a_out, "lp2")
        a_work = _best_seq.copy()

    # === Phase 5: Pair-swap random search (10% of budget) ===
    remaining = time_budget - (time.monotonic() - _t_start)
    t5 = min(remaining * 0.10, 150.0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 5: Pair-swap (n={n_cur}, t={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = pair_swap_optimize(a_work, t5, verbose=True, seed=321)
        try_update_global(a_out, "pairswap1")
        a_work = _best_seq.copy()

    # === Phase 6: Adam round 2 (10% of budget) ===
    remaining = time_budget - (time.monotonic() - _t_start)
    t6 = min(remaining * 0.10, 150.0)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 6: Adam round 2 (n={n_cur}, t={t6:.0f}s) ===")
    if t6 > 20:
        a_out, _ = adam_optimize(a_work, t6, top_k=120, lr_init=0.03, verbose=True, seed=999)
        try_update_global(a_out, "adam2")
        a_work = _best_seq.copy()

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - _t_start)
    n_cur = len(_best_seq)
    print(f"\n=== Phase 7: Length exploration (remaining={remaining:.0f}s) ===")

    trial_lengths = []
    for delta in [500, 1000, -500, 1500, -1000]:
        new_n = n_cur + delta
        if 2000 <= new_n <= 5000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))

    for new_n in trial_lengths:
        remaining = time_budget - (time.monotonic() - _t_start)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")

        a_interp = interpolate_to_length(_best_seq, new_n)

        t7a = min(remaining * 0.35, 180.0)
        if t7a > 20:
            a_out, _ = adam_optimize(a_interp, t7a, top_k=80, lr_init=0.04, verbose=False, seed=new_n % 1000)
            try_update_global(a_out, f"adam_n{new_n}")

        remaining = time_budget - (time.monotonic() - _t_start)
        if remaining > 50 and new_n <= 4000:
            t7b = min(remaining * 0.30, 120.0)
            if t7b > 15:
                a_out, _ = lp_frank_wolfe(a_interp, t7b, solver, top_K=300, grow_rate=30, verbose=False)
                try_update_global(a_out, f"lp_n{new_n}")

    # === Phase 8: Final polish ===
    remaining = time_budget - (time.monotonic() - _t_start)
    n_cur = len(_best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final polish (n={n_cur}, t={remaining:.0f}s) ===")
        # Peak reduction
        t8a = min(remaining * 0.4, 120.0)
        if t8a > 15:
            a_out, _ = peak_reduction(_best_seq, t8a, n_top=30, verbose=False, seed=456)
            try_update_global(a_out, "peak_final")

        remaining = time_budget - (time.monotonic() - _t_start)
        if remaining > 20 and n_cur <= 5000:
            t8b = min(remaining * 0.5, 100.0)
            if t8b > 10:
                a_out, _ = lp_frank_wolfe(_best_seq, t8b, solver, top_K=500, grow_rate=80, verbose=False)
                try_update_global(a_out, "lp_final")

    # Final validation
    if _best_seq is not None:
        final_score = eval_score(_best_seq)
        if final_score < _best_score:
            _best_score = final_score
        print(f"\nFINAL: J={_best_score:.6f}, n={len(_best_seq)}, time={time.monotonic()-_t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        _best_seq = make_sine(n_fb, power=2)
        _best_score = eval_score(_best_seq)
        print(f"Fallback: J={_best_score:.6f}")

    result = [float(x) for x in _best_seq]
    print(f"Returning sequence of length {len(result)}")
    return result
