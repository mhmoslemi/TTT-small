```python
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def solve_slp_lp(a, active_lags, b_vals):
    n = len(a)
    K = len(active_lags)
    
    # Variables: d_0, ..., d_{n-1}, t (total n+1)
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: 2 * g_k^T d - t <= -b_k(a)
    rows = []
    cols = []
    vals = []
    rhs = []
    
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(len(rhs))
            cols.append(j)
            vals.append(2.0 * a[k - j])
        # -t term
        rows.append(len(rhs))
        cols.append(n)  # index of t
        vals.append(-1.0)
        rhs.append(-b_vals[k])
    
    # Lower bounds: d_i >= -a_i
    # This is handled by the bounds parameter
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K, n + 1))
    b_ub = np.array(rhs)
    
    # Equality: sum(d) = 0
    A_eq = csr_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = [(-a[i], None) for i in range(n)] + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        d = result.x[:n]
        t_val = result.x[n]
        return d, t_val
    else:
        return None, None
```

```python
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (s * s)

def build_slp_lp(a, active_lags, b_vals):
    """Build the SLP LP problem. Returns (c, A_ub, b_ub, A_eq, b_eq, bounds)."""
    n = len(a)
    K = len(active_lags)
    
    # Variables: [d_0, ..., d_{n-1}, t] (n+1 total)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: 2 * sum_j a_{k-j} * d_j - t <= -b_k(a)
    row_idx = []
    col_idx = []
    vals = []
    rhs = np.zeros(K)
    
    for idx, k in enumerate(active_lags):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        js = np.arange(j_start, j_end + 1)
        coeffs = 2.0 * a[k_int - js]
        
        row_idx.extend([idx] * len(js))
        col_idx.extend(js.tolist())
        vals.extend(coeffs.tolist())
        
        # -t term
        row_idx.append(idx)
        col_idx.append(n)
        vals.append(-1.0)
        
        rhs[idx] = -b_vals[k_int]
    
    A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(K, n + 1))
    b_ub = rhs
    
    # Equality: sum(d) = 0
    A_eq = csr_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([0.0])
    
    # Bounds: d_i >= -a_i, t >= 0
    lower = -a.copy()
    lower = np.append(lower, 0.0)
    upper = np.full(n + 1, np.inf)
    bounds = list(zip(lower, upper))
    
    return c, A_ub, b_ub, A_eq, b_eq, bounds

def slp_step(a, b, top_k=100, eps=0.05):
    """One SLP iteration. Returns direction d or None."""
    M = b.max()
    
    # Active set: top lags by value, or lags above (1-eps)*M
    threshold = M * (1 - eps)
    active_mask = b > threshold
    active_lags = np.where(active_mask)[0]
    
    # Cap the active set size
    if len(active_lags) > top_k:
        # Take the top_k largest
        top_indices = np.argsort(b)[::-1][:top_k]
        active_lags = top_indices
    
    if len(active_lags) == 0:
        active_lags = np.array([int(np.argmax(b))])
    
    c, A_ub, b_ub, A_eq, b_eq, bounds = build_slp_lp(a, active_lags, b)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        return result.x[:len(a)]
    else:
        return None

def slp_optimize(a0, time_limit, top_k=100, eps=0.05, max_stall=30, seed=42):
    """SLP optimizer."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(seed)
    
    alpha = 1.0  # step damping
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_conv(a)
        M = b.max()
        
        d = slp_step(a, b, top_k=top_k, eps=eps)
        if d is None:
            stall += 3
            continue
        
        # Linesearch
        improved = False
        for t_factor in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05, 0.01]:
            if t_factor < 1e-6:
                continue
            a_new = a + t_factor * d
            if a_new.min() < -1e-10:
                continue
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()
            M_new = fast_conv(a_new).max()
            if M_new < M * (1 - 1e-9):
                a = a_new
                alpha = t_factor
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                improved = True
                break
        
        if improved:
            stall = 0
            # Gradually increase alpha back
            alpha = min(1.0, alpha * 1.05)
        else:
            stall += 1
            alpha = max(0.01, alpha * 0.5)
            if stall % 5 == 0:
                # Perturb
                scale = 0.002 * (stall // 5)
                pert = rng.normal(0, scale, n) * a
                a_new = a + pert
                a_new = np.maximum(a_new, 0.0)
                a_new /= a_new.sum()
                M_new = fast_conv(a_new).max()
                if M_new < best_M:
                    a = a_new
                # Don't update best here, just perturb current
    
    return best_a, best_M
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Successive Linear Programming (SLP)
with adaptive active sets, backtracking linesearch, and coordinate descent polish.
The key innovation over mirror descent is using exact linearization of the quadratic
convolution terms in LP subproblems, yielding much faster convergence near the optimum.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (s * s)


def build_and_solve_lp(a, active_lags, b_vals):
    """
    Build and solve the SLP LP:
    Minimize t
    Subject to: 2*sum_j a_{k-j}*d_j - t <= -b_k(a) for k in active_lags
                sum(d) = 0
                d_i >= -a_i, t >= 0
    Returns direction d or None.
    """
    n = len(a)
    K = len(active_lags)

    c = np.zeros(n + 1)
    c[n] = 1.0

    row_idx = []
    col_idx = []
    vals = []
    rhs = np.empty(K)

    for idx, k in enumerate(active_lags):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        js = np.arange(j_start, j_end + 1)
        coeffs = 2.0 * a[k_int - js]

        row_idx.extend([idx] * len(js))
        col_idx.extend(js.tolist())
        vals.extend(coeffs.tolist())

        row_idx.append(idx)
        col_idx.append(n)
        vals.append(-1.0)

        rhs[idx] = -b_vals[k_int]

    A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(K, n + 1))
    b_ub = rhs

    row_eq = np.arange(n)
    col_eq = np.arange(n)
    vals_eq = np.ones(n)
    A_eq = csr_matrix((vals_eq, (row_eq, col_eq)), shape=(1, n + 1))
    b_eq = np.array([0.0])

    lower = -a.copy()
    lower = np.append(lower, 0.0)
    upper = np.full(n + 1, np.inf)
    bounds = list(zip(lower.tolist(), upper.tolist()))

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.status == 0:
            return result.x[:n]
    except Exception:
        pass
    return None


def slp_optimize(a0, time_limit, top_k=100, eps=0.05, max_stall=40, seed=42, verbose=False):
    """SLP optimizer with adaptive active set and backtracking linesearch."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(seed)
    alpha = 1.0
    cur_top_k = top_k
    cur_eps = eps
    iters = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_conv(a)
        M = b.max()
        iters += 1

        # Build active set
        threshold = M * (1.0 - cur_eps)
        active_mask = b > threshold
        active_lags = np.where(active_mask)[0]

        if len(active_lags) > cur_top_k:
            top_indices = np.argsort(b)[::-1][:cur_top_k]
            active_lags = top_indices
        elif len(active_lags) < 5:
            # Always include top 5 at minimum
            active_lags = np.argsort(b)[::-1][:min(5, 2 * n - 1)]

        # Solve LP
        d = build_and_solve_lp(a, active_lags, b)
        if d is None:
            stall += 3
            cur_top_k = max(20, cur_top_k // 2)
            continue

        # Check if direction is meaningful
        d_norm = np.max(np.abs(d))
        if d_norm < 1e-14:
            stall += 1
            continue

        # Backtracking linesearch
        improved = False
        for t_factor in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.05, 0.01, 0.005]:
            if t_factor < 1e-4:
                continue
            a_new = a + t_factor * d
            if a_new.min() < -1e-10:
                continue
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            M_new = fast_conv(a_new).max()
            if M_new < M * (1.0 - 1e-9):
                a = a_new
                alpha = min(1.0, t_factor * 1.1)
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                improved = True
                break

        if improved:
            stall = 0
            # Gradually expand active set
            if iters % 20 == 0:
                cur_eps = max(0.01, cur_eps * 0.95)
                cur_top_k = min(300, cur_top_k + 5)
        else:
            stall += 1
            alpha = max(0.005, alpha * 0.5)
            if stall % 8 == 0:
                # Perturb to escape
                scale = 0.001 * min(stall // 8, 5.0)
                pert = rng.normal(0, scale, n) * a
                a_new = a + pert
                a_new = np.maximum(a_new, 0.0)
                a_new /= a_new.sum()
                M_new = fast_conv(a_new).max()
                if M_new < best_M * 1.01:
                    a = a_new
            # Adapt parameters
            if stall % 20 == 0:
                cur_eps = min(0.15, cur_eps * 1.3)
                cur_top_k = min(400, cur_top_k + 20)

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SLP iter={iters} ({elapsed:.0f}s): M={best_M:.8f}, stall={stall}, k={cur_top_k}, eps={cur_eps:.3f}")

    return best_a, best_M

                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.002, 0.02)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        b_new = fast_conv(a)
                        if b_new.max() < best_M:
                            best_M = b_new.max()
                            best_a = a.copy()
                            b = b_new
                            stall = 0
                            break

        if verbose and stall % 30 == 0:
            print(f"    CD ({time.monotonic()-t0:.0f}s): M={best_M:.8f}, stall={stall}")

    return best_a, best_M


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=80.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.15) ** 2) + np.exp(-width * (x - 0.85) ** 2) + 0.005
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 8):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s
```
