import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient(a, ac, top_k, beta):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=5000,
                 rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 1.0, 2.0)

    return best_a, best_m


def lp_peak_minimize(a0, time_limit, solver, verbose=True, max_constraints=5000):
    """Direct LP: minimize M s.t. (a0*b)_k <= M, sum(b)=1, b>=0.
    Uses cutting plane to add constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Initial constraint set: top constraints by (a*a) values
    ac = fast_autocorr(a)
    K_init = min(200, 2 * n - 1)
    constraint_set = set(np.argsort(ac)[::-1][:K_init])

    b = a.copy()  # current iterate
    b_sum = b.sum()
    b = b / b_sum

    for outer in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP: minimize M s.t. (a*b)_k <= M for k in constraint_set, sum(b)=1, b>=0
        M_var = cp.Variable()
        b_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(b_var) == 1]

        # Add selected constraints
        for k in constraint_set:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            # (a * b)_k = sum_j a[j] * b[k-j]
            # For fixed a, this is sum_j a[k-j] * b[j] for valid j
            coeff = a[partner]
            constraints.append(coeff @ b_var <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_new_sum = b_new.sum()
        if b_new_sum < 1e-15:
            continue
        b_new = b_new / b_new_sum

        # Evaluate
        ac_new = fast_autocorr(b_new)
        M_new = ac_new.max()

        if M_new < best_m:
            best_m = M_new
            best_a = b_new.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    LP-CP iter={outer} ({elapsed:.0f}s): J={J:.6f}")

        b = b_new

        # Check for violated constraints (cutting plane)
        if len(constraint_set) < max_constraints:
            violated = [k for k in range(2 * n - 1) if ac_new[k] > M_var.value + 1e-8 and k not in constraint_set]
            if violated:
                # Add the most violated ones
                violated_sorted = sorted(violated, key=lambda k: ac_new[k], reverse=True)
                n_add = min(50, len(violated_sorted), max_constraints - len(constraint_set))
                for k in violated_sorted[:n_add]:
                    constraint_set.add(k)

        # Also periodically add new top constraints
        if outer % 10 == 0 and len(constraint_set) < max_constraints:
            top_now = np.argsort(ac_new)[::-1][:50]
            for k in top_now:
                if k not in constraint_set:
                    constraint_set.add(k)
                    if len(constraint_set) >= max_constraints:
                        break

        if verbose and outer % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-CP iter={outer} ({elapsed:.0f}s): M={M_new:.6f}, |C|={len(constraint_set)}")

    return best_a, best_m


def evolutionary_search(a0, time_limit, pop_size=30, verbose=True, rng_seed=100):
    """Evolutionary strategy with tournament selection, arithmetic crossover, and Gaussian mutation."""
    n = len(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Initialize population
    a0_norm = norm_sum1(a0)
    population = [a0_norm.copy()]

    # Add diverse initial members
    inits = [
        make_u_shape(n, 1.5),
        make_u_shape(n, 3.0),
        make_bimodal(n, 0.2),
        spectral_init(n, 20, 42),
        spectral_init(n, 15, 99),
    ]
    for init in inits:
        if len(population) >= pop_size:
            break
        population.append(init)

    # Fill with mutations of best
    while len(population) < pop_size:
        parent = population[0]
        perm = rng.permutation(n)
        child = parent[perm].copy()
        child = np.maximum(child, 0)
        child = child / child.sum()
        population.append(child)

    # Evaluate all
    fitness = []
    for ind in population:
        m = fast_autocorr(ind).max()
        fitness.append(m)

    # Sort by fitness
    order = np.argsort(fitness)
    population = [population[i] for i in order]
    fitness = [fitness[i] for i in order]

    best_a = population[0].copy()
    best_m = fitness[0]
    if verbose:
        J = 2.0 * n * best_m
        print(f"    ES init: J={J:.6f}")

    gen = 0
    while time.monotonic() - t0 < time_limit:
        gen += 1
        new_pop = []

        # Elitism: keep top 2
        for i in range(min(2, len(population))):
            new_pop.append(population[i].copy())

        # Generate offspring
        while len(new_pop) < pop_size:
            if time.monotonic() - t0 > time_limit - 5:
                break

            # Tournament selection
            candidates = rng.integers(0, len(population), size=3)
            parent_idx = candidates[np.argmin([fitness[c] for c in candidates])]
            parent = population[parent_idx]

            # Crossover (arithmetic) with random partner
            if rng.random() < 0.5 and len(population) > 1:
                partner_idx = rng.integers(0, len(population))
                partner = population[partner_idx]
                alpha = rng.uniform(0.3, 0.7)
                child = alpha * parent + (1 - alpha) * partner
            else:
                child = parent.copy()

            # Mutation
            mutation_rate = 0.1 / np.sqrt(n)
            mask = rng.random(n) < mutation_rate
            sigma = 0.05 + 0.05 * np.sqrt(fitness[parent_idx] / max(best_m, 1e-15))
            child[mask] += rng.normal(0, sigma, mask.sum())
            child = np.maximum(child, 0)
            cs = child.sum()
            if cs < 1e-15:
                child = parent.copy()
            else:
                child = child / cs

            # Also try permutation mutation occasionally
            if rng.random() < 0.1:
                perm = rng.permutation(n)
                child = child[perm].copy()
                child = np.maximum(child, 0)
                cs = child.sum()
                if cs > 1e-15:
                    child = child / cs

            new_pop.append(child)

        # Evaluate new population
        new_fitness = []
        for ind in new_pop:
            m = fast_autocorr(ind).max()
            new_fitness.append(m)

        # Combine and select
        all_pop = new_pop[:pop_size]
        all_fit = new_fitness[:pop_size]

        # Update best
        best_in_gen = np.argmin(all_fit)
        if all_fit[best_in_gen] < best_m:
            best_m = all_fit[best_in_gen]
            best_a = all_pop[best_in_gen].copy()

        # Sort and keep best pop_size
        order = np.argsort(all_fit)
        population = [all_pop[i] for i in order[:pop_size]]
        fitness = [all_fit[i] for i in order[:pop_size]]

        if verbose and gen % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ES gen={gen} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def nesterov_subgrad(a0, time_limit, top_k=80, beta=6.0, c0=5.0,
                      restart_every=6000, rng_seed=555):
    """Nesterov-accelerated subgradient with line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=top_k, beta=beta)

        # Subgradient step in log space with backtracking
        eta = gamma / (np.sqrt(t) + 1.0)

        # Try a few step sizes
        best_y = y.copy()
        best_y_m = fast_autocorr(y).max()

        for scale in [1.0, 0.5, 0.2, 2.0]:
            log_a = np.log(y) - eta * scale * g
            max_log = log_a.max()
            a_cand = np.exp(log_a - max_log)
            a_cand = np.maximum(a_cand, 1e-30)
            s_cand = a_cand.sum()
            if s_cand < 1e-15:
                continue
            a_cand = a_cand / s_cand
            m_cand = fast_autocorr(a_cand).max()
            if m_cand < best_y_m:
                best_y_m = m_cand
                best_y = a_cand

        x_prev = x
        x = best_y

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

    return best_a, best_m


def coordinate_refine(a0, time_limit, rng_seed=123):
    """Targeted coordinate descent focusing on peak-contributing pairs."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:20]

        improved = False
        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_partner = partner[np.argmax(contrib)]

            # Try moving mass away from the contributing pair
            for reduce_idx in [src, src_partner]:
                if a[reduce_idx] < 1e-12:
                    continue
                # Find best destination (low autocorrelation contribution)
                sens = np.zeros(n)
                for kk in top_k[:8]:
                    kki = int(kk)
                    ss = max(0, kki - n + 1)
                    ee = min(n - 1, kki)
                    ii = np.arange(ss, ee + 1)
                    sens[ii] += a[kki - ii]

                exclude = {src, src_partner}
                candidates = np.array([d for d in range(n) if d not in exclude])
                if len(candidates) == 0:
                    continue
                dst = candidates[np.argmin(sens[candidates])]

                for frac in [0.01, 0.02, 0.05, 0.005]:
                    amt = a[reduce_idx] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m

def fast_autocorr(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient(a, ac, top_k, beta):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    LP-CP iter={outer} ({elapsed:.0f}s): J={J:.6f}, |C|={len(constraint_list)}")

        b = b_new

        # Check for violated constraints (cutting plane)
        # Find lags where (a*b)_k > M_lp + tolerance
        violation = ac_new - M_lp
        violated_mask = violation > max(1e-6, 0.01 * abs(M_lp))
        violated_indices = np.where(violated_mask)[0]

        if len(violated_indices) > 0 and len(constraint_list) < max_constraints:
            # Add most violated constraints
            violated_sorted = violated_indices[np.argsort(ac_new[violated_indices])[::-1]]
            n_add = min(100, len(violated_sorted), max_constraints - len(constraint_list))
            for k in violated_sorted[:n_add]:
                if k not in constraint_set:
                    constraint_set.add(k)
                    constraint_list.append(int(k))

        # Periodically refresh with current top constraints
        if outer % 15 == 0 and len(constraint_list) < max_constraints:
            top_now = np.argsort(ac_new)[::-1][:50]
            added = 0
            for k in top_now:
                if k not in constraint_set:
                    constraint_set.add(k)
                    constraint_list.append(int(k))
                    added += 1
                    if added >= 30:
                        break

        # Convergence check
        if outer > 50 and abs(M_new - best_m) < 1e-10 and M_lp >= M_new - 1e-8:
            # LP bound matches actual peak, converged
            if verbose:
                print(f"    LP-CP converged at iter={outer}")
            break

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=500):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    K_sel = min(top_K, 2 * n - 1)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select binding constraints
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build LP: max sum(b) s.t. (y*b)_k <= M for k in binding, b>=0
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        for k in binding_idx:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            coeff = y[partner]
            constraints.append(coeff @ b_var <= M_curr)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

        # Adaptively increase K
        if not (K_sel >= 2 * n - 1) and t % 30 == 0:
            K_sel = min(K_sel + 50, 2 * n - 1)

    return best_a, best_m


def evolutionary_search(a0, time_limit, pop_size=30, verbose=True, rng_seed=100):
    """Evolutionary strategy with tournament selection, arithmetic crossover, and Gaussian mutation."""
    n = len(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    # Initialize population with diverse starting points
    a0_norm = norm_sum1(a0)
    init_list = [
        a0_norm.copy(),
        make_u_shape(n, 1.5),
        make_u_shape(n, 3.0),
        make_u_shape(n, 5.0),
        make_bimodal(n, 0.2),
        make_bimodal(n, 0.1),
        spectral_init(n, 20, 42),
        spectral_init(n, 15, 99),
        make_power_law(n, 1.5),
        make_cosine_bump(n, 0.5, 0.3),
    ]

    population = []
    for init in init_list:
        if len(population) < pop_size:
            population.append(init)

    # Fill remaining with random mutations
    while len(population) < pop_size:
        parent = population[rng.integers(len(population))]
        mask = rng.random(n) < 0.3
        child = parent.copy()
        child[mask] += rng.normal(0, 0.05, mask.sum())
        child = np.maximum(child, 0)
        cs = child.sum()
        if cs > 1e-15:
            child = child / cs
        else:
            child = parent.copy()
        population.append(child)

    # Evaluate
    fitness = []
    for ind in population:
        m = fast_autocorr(ind).max()
        fitness.append(m)

    # Sort
    order = np.argsort(fitness)
    population = [population[i].copy() for i in order]
    fitness = [fitness[i] for i in order]

    best_a = population[0].copy()
    best_m = fitness[0]
    if verbose:
        J = 2.0 * n * best_m
        print(f"    ES init: J={J:.6f}, pop={len(population)}")

    gen = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        gen += 1
        new_pop = []

        # Elitism
        for i in range(min(3, len(population))):
            new_pop.append(population[i].copy())

        # Generate offspring
        while len(new_pop) < pop_size:
            if time.monotonic() - t0 > time_limit - 3:
                break

            # Tournament selection (size 3)
            cands = rng.integers(0, len(population), size=3)
            parent_idx = cands[np.argmin([fitness[c] for c in cands])]
            parent = population[parent_idx]

            # Crossover
            if rng.random() < 0.6 and len(population) > 1:
                partner_idx = rng.integers(0, min(10, len(population)))
                partner = population[partner_idx]
                alpha = rng.uniform(0.2, 0.8)
                child = alpha * parent + (1.0 - alpha) * partner
            else:
                child = parent.copy()

            # Gaussian mutation
            # Adaptive mutation rate based on fitness
            rel_fitness = fitness[parent_idx] / max(best_m, 1e-15)
            sigma = 0.03 + 0.04 * rel_fitness
            mask = rng.random(n) < 0.15
            child[mask] += rng.normal(0, sigma, mask.sum())
            child = np.maximum(child, 0)
            cs = child.sum()
            if cs < 1e-15:
                child = parent.copy()
            else:
                child = child / cs

            # Occasional permutation (to explore non-smooth structures)
            if rng.random() < 0.05:
                # Block swap
                block_size = max(1, n // 20)
                i = rng.integers(0, n - block_size)
                j = rng.integers(0, n - block_size)
                if abs(i - j) > block_size:
                    child[i:i+block_size], child[j:j+block_size] = child[j:j+block_size].copy(), child[i:i+block_size].copy()
                    child = np.maximum(child, 0)
                    cs = child.sum()
                    if cs > 1e-15:
                        child = child / cs

            new_pop.append(child)

        # Evaluate new population
        new_fitness = []
        for ind in new_pop:
            m = fast_autocorr(ind).max()
            new_fitness.append(m)

        # Update best
        best_in_gen = np.argmin(new_fitness)
        prev_best = best_m
        if new_fitness[best_in_gen] < best_m:
            best_m = new_fitness[best_in_gen]
            best_a = new_pop[best_in_gen].copy()
            no_improve = 0
        else:
            no_improve += 1

        # Combine: keep best pop_size from new population
        order = np.argsort(new_fitness)
        population = [new_pop[i].copy() for i in order[:pop_size]]
        fitness = [new_fitness[i] for i in order[:pop_size]]

        # If stagnating, inject more diversity
        if no_improve > 30 and gen % 10 == 0:
            # Replace worst half with fresh random initializations
            half = pop_size // 2
            for i in range(half):
                fresh = population[rng.integers(len(population))]
                perm = rng.permutation(n)
                child = fresh[perm].copy()
                mask = rng.random(n) < 0.2
                child[mask] += rng.normal(0, 0.08, mask.sum())
                child = np.maximum(child, 0)
                cs = child.sum()
                if cs > 1e-15:
                    child = child / cs
                population[-(i + 1)] = child
                fitness[-(i + 1)] = fast_autocorr(child).max()

        if verbose and gen % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ES gen={gen} ({elapsed:.0f}s): J={J:.6f}, no_improve={no_improve}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    ES final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def coordinate_refine(a0, time_limit, rng_seed=123):
    """Targeted coordinate descent on peak-contributing pairs."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 400:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:25]

        improved = False
        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            src_local = np.argmax(contrib)
            src = idx[src_local]
            src_partner = partner[src_local]

            # Build sensitivity
            sens = np.zeros(n)
            for kk in top_k[:10]:
                kki = int(kk)
                ss = max(0, kki - n + 1)
                ee = min(n - 1, kki)
                ii = np.arange(ss, ee + 1)
                sens[ii] += a[kki - ii]

            for reduce_idx in [src, src_partner]:
                if a[reduce_idx] < 1e-12:
                    continue
                exclude = {int(src), int(src_partner)}
                candidates = np.array([d for d in range(n) if d not in exclude])
                if len(candidates) == 0:
                    continue
                dst = candidates[np.argmin(sens[candidates])]

                for frac in [0.01, 0.02, 0.005, 0.03, 0.05]:
                    amt = a[reduce_idx] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        best_seq = make_u_shape(n_base, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP Peak Minimization (cutting plane) - most promising ===
    print(f"\n=== Phase 1: LP Peak Min (cutting plane) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 440.0)
    if t1 > 40 and n0 <= 6000:
        a_in = norm_sum1(best_seq)
        max_c = min(5000, 2 * n0 - 1)
        a_out, _ = lp_peak_minimize(a_in, t1, solver, verbose=True, max_constraints=max_c)
        try_update(a_out, "lp_cp")

    # === Phase 2: Multi-init AMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Multi-init AMD ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 60:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        inits.append(norm_sum1(best_seq))
        labels.append("best")
        inits.append(make_u_shape(n0, 2.5))
        labels.append("u2.5")
        inits.append(make_u_shape(n0, 4.0))
        labels.append("u4.0")
        inits.append(make_bimodal(n0, 0.15))
        labels.append("bimodal")
        inits.append(spectral_init(n0, 25, 42))
        labels.append("spectral")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 3))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(12.0, t2 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=80, beta=7.0, c0=5.0,
                                     restart_every=5000, rng_seed=42 + idx)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 3: Evolutionary Search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Evolutionary Search ===")
    t3 = min(remaining * 0.25, 450.0)
    if t3 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = evolutionary_search(a_in, t3, pop_size=30, verbose=True, rng_seed=200)
        try_update(a_out, "evolutionary")

    # === Phase 4: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 4: LP-FW (n={n_now}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30 and n_now <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K=min(2000, 2 * n_now - 1))
        try_update(a_out, "lp_fw")

    # === Phase 5: LP Peak Min again (round 2) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 5: LP CP Round 2 ===")
    t5 = min(remaining * 0.10, 180.0)
    if t5 > 30 and n_now <= 6000:
        a_in = norm_sum1(best_seq)
        max_c = min(6000, 2 * n_now - 1)
        a_out, _ = lp_peak_minimize(a_in, t5, solver, verbose=True, max_constraints=max_c)
        try_update(a_out, "lp_cp_r2")

    # === Phase 6: Coordinate Refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coordinate Refine ===")
    t6 = min(remaining * 0.08, 120.0)
    if t6 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_refine(a_in, t6, rng_seed=456)
        try_update(a_out, "coord")

    # === Phase 7: Length Exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 7: Length Exploration ===")

    trial_lengths = []
    for delta in [800, 1200, -800, 1600, -1200]:
        new_n = n_now + delta
        if 1500 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.35, 180.0)
        if t7a > 30:
            a_out, _ = amd_optimize(a_long, t7a, top_k=60, beta=6.0, c0=4.0,
                                     restart_every=5000, rng_seed=new_n % 1000)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50 and new_n <= 5000:
            t7b = min(remaining * 0.30, 120.0)
            if t7b > 25:
                a_in = norm_sum1(best_seq)
                max_c = min(3000, 2 * new_n - 1)
                a_out, _ = lp_peak_minimize(a_in, t7b, solver, verbose=False, max_constraints=max_c)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_now = len(best_seq)
    print(f"\n=== Phase 8: Final Polish (n={n_now}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t8, top_k=120, beta=9.0, c0=3.5,
                                     restart_every=10000, rng_seed=999)
            try_update(a_out, "amt_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.5, 60.0)
            if t8b > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_refine(a_in, t8b, rng_seed=789)
                try_update(a_out, "coord_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20 and n_now <= 5000:
            t8c = min(remaining * 0.4, 50.0)
            if t8c > 10:
                a_in = norm_sum1(best_seq)
                max_c = min(5000, 2 * n_now - 1)
                a_out, _ = lp_peak_minimize(a_in, t8c, solver, verbose=False, max_constraints=max_c)
                try_update(a_out, "lp_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
