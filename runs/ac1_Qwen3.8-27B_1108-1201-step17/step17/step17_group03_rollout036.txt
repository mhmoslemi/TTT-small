def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=300, margin_frac=1e-3, 
                   margin_schedule='geometric', margin_min=1e-5):
    """LP-based Frank-Wolfe with adaptive margin."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        # Adaptive margin
        if margin_schedule == 'geometric':
            margin = max(margin_frac * (0.95 ** (t // 10)), margin_min)
        else:
            margin = margin_frac
        
        # Select active constraints
        margins = M_curr - ac
        threshold = margin * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()
        
        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 10:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]
        
        K = len(active_idx)
        
        # Build constraint matrix
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, int(k_int) - n + 1)
            i_end = min(n, int(k_int) + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[int(k_int) - i]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_val if False else b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe update
        alpha = 2.0 / (t + 2.0)
        
        # Check if FW step improves
        y_trial = (1.0 - alpha) * y + alpha * g_dir
        y_trial = np.maximum(y_trial, 0.0)
        sy = y_trial.sum()
        if sy < 1e-15:
            y_trial = y.copy()
        else:
            y_trial = y_trial / sy
        
        # Also try a pure gradient-like step
        m_trial = fast_autocorr(y_trial).max()
        
        # Nesterov acceleration
        if t == 1:
            x_new = y_trial.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_trial + beta_t * (y_trial - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_trial.copy()
            else:
                x_new = x_new / sx
        
        y = y_trial
        x = x_new
        
        # Update best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, K={K}")
    
    return best_a, best_m

def peak_coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Targeted coordinate descent focusing on peak autocorrelation."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peaks
        n_peaks = min(5, len(ac))
        peak_indices = np.argsort(ac)[::-1][:n_peaks]
        
        improved = False
        
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            pk_int = int(pk)
            i_start = max(0, pk_int - n + 1)
            i_end = min(n, pk_int + 1)
            idx = np.arange(i_start, i_end)
            
            # Contribution of each pair to this peak
            contrib = a[idx] * a[pk_int - idx]
            
            # Find the largest contributors
            n_top = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]
            
            for li in top_local:
                if time.monotonic() - t0 > time_limit:
                    break
                
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                # Try moving mass from src to a random destination
                best_move = None
                best_new_m = M_curr
                
                n_trials = 40
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    
                    # Try multiple fractions
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_new_m - 1e-15:
                            best_new_m = M_trial
                            best_move = (src, dst, amt)
                
                if best_move is not None:
                    src_i, dst_i, amt = best_move
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 100 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PC ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def freq_perturb(a, strength=0.02, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x / 2) ** 2)
    elif family == 'beta':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x * (1 - x))
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)

def fast_autocorr(a):
    """Compute full autoconvolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver: ECOS (fallback)")
    return cp.ECOS


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=300,
                   margin_frac=2e-3, margin_min=5e-6, decay_rate=0.97):
    """LP-based Frank-Wolfe with geometrically shrinking margin."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Geometrically decaying margin
        margin = max(margin_frac * (decay_rate ** (t // 5)), margin_min)

        # Select active constraints
        margins = M_curr - ac
        threshold = margin * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 4:
            active_idx = np.argsort(margins)[:top_K * 4]
        elif n_active < 10:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)

        # Build constraint matrix (sparse-friendly)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            ki = int(k_int)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[ki - i]

        # Solve LP: max sum(b) s.t. T_sel @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Frank-Wolfe step
        alpha = 2.0 / (t + 2.0)
        y_trial = (1.0 - alpha) * y + alpha * g_dir
        y_trial = np.maximum(y_trial, 0.0)
        sy = y_trial.sum()
        if sy < 1e-15:
            y_trial = y.copy()
        else:
            y_trial = y_trial / sy

        # Nesterov acceleration
        if t == 1:
            x_new = y_trial.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_trial + beta_t * (y_trial - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_trial.copy()
            else:
                x_new = x_new / sx

        y = y_trial
        x = x_new

        # Update best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.7f}, K={K}, margin={margin:.2e}")

    return best_a, best_m


def lse_mirror_descent(a0, time_limit, verbose=True, c0=3.0, beta_start=60.0):
    """Mirror descent with log-sum-exp smoothing of max autoconvolution."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute log-sum-exp weights over autoconvolution
        ac = fast_autocorr(y)
        beta_lse = max(beta_start, 30.0 + 0.05 * t)
        ac_shifted = ac - ac.max()
        weights = np.exp(beta_lse * ac_shifted)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            w = np.ones_like(ac) / len(ac)
        else:
            w = weights / w_sum

        # Compute gradient using top peaks only (for efficiency)
        top_k = min(40, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]

        grad = np.zeros(n)
        for k_int in top_idx:
            wk = w[k_int]
            if wk < 1e-12:
                continue
            ki = int(k_int)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            for i in range(i_start, i_end):
                if ki == 2 * i:
                    grad[i] += wk * y[i]
                else:
                    grad[i] += 2.0 * wk * y[ki - i]

        # Mirror descent step with backtracking
        eta = c0 / (np.sqrt(t) + 1.0)
        accepted = False
        base_m = fast_autocorr(y).max()

        for _ in range(5):
            log_a = np.log(y) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            else:
                eta *= 0.5

        if not accepted:
            log_a = np.log(y) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2500:
            stall = 0
            pert = rng.normal(0, 0.012, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 8000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE-MD t={t} ({elapsed:.0f}s): J={J:.7f}, beta_lse={beta_lse:.1f}")

    return best_a, best_m


def peak_coordinate_descent(a0, time_limit, verbose=True, rng_seed=42):
    """Single-peak targeted coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total = contrib.sum()

        if total < 1e-20:
            stall += 1
            continue

        n_top = min(25, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break

            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

            for frac in [0.003, 0.008, 0.015, 0.03, 0.06, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                n_trials = 35
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m - 1e-16:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.04)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 80 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PC ({elapsed:.0f}s): J={J:.7f}, stall={stall}")

    return best_a, best_m


def freq_perturb(a, strength=0.02, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)

        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x * (1 - x))
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)

                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"pc_n{new_n}")

    # === Phase 5: Freq diversification + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Freq-diversify (n={n_best}) ===")
    rng = np.random.default_rng(2024)

    for trial in range(4):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        strength = 0.015 + 0.01 * trial
        a_pert = freq_perturb(best_seq, strength=strength, rng=rng)

        t5 = min(remaining * 0.09, 100.0)
        if t5 > 30:
            a_out, _ = lp_frank_wolfe(a_pert, t5, solver, verbose=False, top_K=250,
                                      margin_frac=2e-3, margin_min=1e-5, decay_rate=0.97)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 6: Second LP pass with very tight margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP tight-margin (n={n_best}) ===")
    t6 = min(remaining * 0.14, 180.0)
    if t6 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t6, solver, verbose=True, top_K=500,
                                  margin_frac=1e-4, margin_min=2e-6, decay_rate=0.98)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 7: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final PC (n={len(best_seq)}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.30, 180.0)
        if t7 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_coordinate_descent(a_in, t7, verbose=True, rng_seed=7777)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.7f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning now")

    return [float(x) for x in best_seq]
