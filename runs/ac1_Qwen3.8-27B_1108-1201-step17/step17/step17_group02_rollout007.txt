import numpy as np

_global_best_score = float('inf')


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def cross_corr_p_a(p, a):
    """Compute g[i] = sum_{m=0}^{n-1} p[i+m] * a[m] for i=0,...,n-1."""
    n = len(a)
    fft_size = 1 << (3 * n).bit_length()
    Pa = np.fft.rfft(p, fft_size)
    Aa = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(Pa * np.conj(Aa), fft_size)
    return cross[:n]


def soft_max_obj_grad(theta, T):
    """Compute soft-max objective and gradient w.r.t. theta where a = exp(theta)."""
    a = np.exp(theta)
    n = len(a)
    S = a.sum()

    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    sum_exp = exp_Tc.sum()
    p = exp_Tc / sum_exp
    
    m_soft = (Tc_max + np.log(sum_exp)) / T
    
    obj = 2.0 * n * m_soft / (S * S)
    
    g = cross_corr_p_a(p, a)
    
    grad_a = 4.0 * n * (g * S - m_soft) / (S ** 3)
    grad_theta = grad_a * a
    
    return obj, grad_theta

    # Final check with exact objective
    final_obj = 2.0 * n * fast_autocorr(best_a).max()
    if final_obj < best_obj:
        best_obj = final_obj
    
    sc = 2.0 * n * best_obj
    if sc < _global_best_score:
        _global_best_score = sc
        _global_best = best_a.copy()
    
    return best_a, best_obj


def lbfgs_exact(a0, time_limit, verbose=False, top_k=20):
    """L-BFGS with exact max subgradient (top-k averaging)."""
    global _global_best, _global_best_score
    n = len(a0)
    a = norm_sum1(a0)
    theta0 = np.log(np.maximum(a, 1e-15))
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    theta = theta0.copy()
    
    def func_grad(th):
        return exact_max_obj_grad(th, top_k=top_k)
    
    max_iters = 200
    while time.monotonic() - t0 < time_limit:
        try:
            res = minimize(func_grad, theta, jac=True, method='L-BFGS-B',
                          options={'maxiter': max_iters, 'maxfun': max_iters * 3,
                                   'ftol': 1e-14, 'gtol': 1e-10,
                                   'iprint': -1})
        except Exception:
            break
        
        if res is not None and res.x is not None:
            a_new = norm_sum1(np.exp(res.x))
            obj_new = 2.0 * n * fast_autocorr(a_new).max()
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                theta = res.x.copy()
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"      Exact L-BFGS ({elapsed:.0f}s): obj={obj_new:.8f}")
            else:
                break
        else:
            break
    
    sc = 2.0 * n * fast_autocorr(best_a).max()
    if sc < _global_best_score:
        _global_best_score = sc
        _global_best = best_a.copy()
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False, rng_seed=123):
    """Aggressive coordinate descent with autocorrelation-aware moves."""
    global _global_best, _global_best_score
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 300
    n_top = 10
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        # Build sensitivity map
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]
        
        # Try moves on top peaks
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            # Try several source positions (not just the worst)
            top_contrib = np.argsort(contrib)[::-1][:5]
            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                candidates = np.where(~in_peak)[0]
                if len(candidates) == 0:
                    candidates = np.arange(n)
                cand_sens = sens[candidates]
                
                # Try several destination candidates
                best_moves = np.argsort(cand_sens)[:10]
                for bm in best_moves:
                    dst = candidates[bm]
                    if dst == src:
                        continue
                    
                    for frac in [0.01, 0.005, 0.02, 0.03]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_obj - 1e-14:
                            a = a_trial
                            best_obj = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if time.monotonic() - t0 > time_limit:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            n_pert = rng.integers(2, min(8, n))
            perturbs = rng.integers(0, n, n_pert)
            dests = rng.integers(0, n, n_pert)
            for i, j in zip(perturbs, dests):
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_obj:
                best_obj = M_new
                best_a = a.copy()
                stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Coord ({elapsed:.0f}s): obj={best_obj:.8f}, stall={stall}")
    
    sc = 2.0 * n * best_obj
    if sc < _global_best_score:
        _global_best_score = sc
        _global_best = best_a.copy()
    
    return best_a, best_obj


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_w_shape(n, sharpness=1.5):
    x = np.linspace(0, 1, n)
    a = 1.0 + sharpness * (np.cos(4 * np.pi * x) + 1) ** 2
    return norm_sum1(a)


def make_cosine(n, k=1):
    x = np.linspace(0, 1, n)
    a = np.abs(np.cos(k * np.pi * x)) ** 2 + 0.01
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def gradient_flow(a0, time_limit, lr=0.01, decay=0.999, verbose=False, rng_seed=99):
    """Simple projected gradient descent with the exact subgradient."""
    global _global_best, _global_best_score
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    cur_lr = lr
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        if M_curr < best_obj - 1e-13:
            best_obj = M_curr
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute subgradient (averaged over top peaks)
        threshold = M_curr * 0.995
        active = np.where(ac >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(ac)])
        
        # Gradient of max(c) at a: for active lag k, grad = 2*a (shifted)
        grad = np.zeros(n)
        for k in active[:20]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            grad[idx] += 2.0 * a[ki - idx]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-30:
            break
        grad = grad / gnorm
        
        # Update
        a_new = a - cur_lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        # Check if we improved (gradient step might overshoot)
        M_new = fast_autocorr(a_new).max()
        if M_new < M_curr:
            a = a_new
            cur_lr = min(cur_lr * 1.1, 0.1)
        else:
            # Try smaller step
            for mult in [0.5, 0.25, 0.1]:
                a_try = a - cur_lr * mult * grad
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try / a_try.sum()
                M_try = fast_autocorr(a_try).max()
                if M_try < M_curr:
                    a = a_try
                    cur_lr = min(cur_lr * mult * 1.1, 0.1)
                    break
            else:
                cur_lr *= decay
        
        if stall > 500:
            # Perturb and restart
            pert = rng.normal(0, 0.01, n) * a
            a = a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0
            cur_lr = 0.01
    
    sc = 2.0 * n * best_obj
    if sc < _global_best_score:
        _global_best_score = sc
        _global_best = best_a.copy()
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    global _global_best, _global_best_score
    
    _global_best = None
    _global_best_score = float('inf')
    
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)
    
    def try_update(a, label):
        global _global_best, _global_best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = 2.0 * len(a_norm) * fast_autocorr(a_norm).max()
        if np.isfinite(sc) and sc < _global_best_score:
            _global_best_score = sc
            _global_best = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = 2.0 * len(parent) * fast_autocorr(parent).max()
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            _global_best = parent.copy()
            _global_best_score = p_score
    except Exception:
        pass
    
    if _global_best is None:
        n_base = 4000
        _global_best = make_u_shape(n_base, 2.0)
        _global_best_score = 2.0 * n_base * fast_autocorr(_global_best).max()
        print(f"Default init: n={n_base}, score={_global_best_score:.6f}")
    
    n0 = len(_global_best)
    
    # === Phase 1: Multi-init L-BFGS with soft-max ===
    print(f"\n=== Phase 1: L-BFGS soft-max (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 600.0)
    
    if t_phase1 > 60:
        inits = []
        labels = []
        
        if parent is not None:
            inits.append(parent.copy())
            labels.append("parent")
        
        inits.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, 2.5))
        labels.append("u2.5")
        inits.append(make_u_shape(n0, 4.0))
        labels.append("u4.0")
        inits.append(make_w_shape(n0, 1.0))
        labels.append("w1.0")
        inits.append(make_cosine(n0, 1))
        labels.append("cos1")
        inits.append(make_cosine(n0, 2))
        labels.append("cos2")
        inits.append(spectral_init(n0, 20, seed=42))
        labels.append("spectral1")
        inits.append(spectral_init(n0, 15, seed=7))
        labels.append("spectral2")
        
        if parent is not None:
            blend = 0.6 * parent + 0.4 * make_u_shape(n0, 2.0)
            inits.append(norm_sum1(blend))
            labels.append("blend")
        
        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed = time.monotonic() - t_global
            if elapsed > time_budget - 30:
                break
            t_avail = min(t_per, time_budget - 30 - elapsed)
            if t_avail < 10:
                break
            print(f"  L-BFGS '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = lbfgs_optimize(init, t_avail, verbose=True, rng_seed=42 + idx)
            try_update(a_out, f"lbfgs_{lbl}")
    
    # === Phase 2: Exact subgradient L-BFGS ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Exact subgradient L-BFGS (n={len(_global_best)}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_in = _global_best.copy()
        a_out, _ = lbfgs_exact(a_in, t2, verbose=True, top_k=30)
        try_update(a_out, "exact_lbfgs")
    
    # === Phase 3: Gradient flow ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Gradient flow (n={len(_global_best)}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_in = _global_best.copy()
        a_out, _ = gradient_flow(a_in, t3, lr=0.005, verbose=False)
        try_update(a_out, "gradflow")
    
    # === Phase 4: More L-BFGS rounds on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: L-BFGS round 2 (n={len(_global_best)}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = _global_best.copy()
        a_out, _ = lbfgs_optimize(a_in, t4, verbose=True, rng_seed=200)
        try_update(a_out, "lbfgs_r2")
    
    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coordinate descent (n={len(_global_best)}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 15:
        a_in = _global_best.copy()
        a_out, _ = coordinate_descent(a_in, t5, verbose=True, rng_seed=456)
        try_update(a_out, "coord")
    
    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_global_best)
    print(f"\n=== Phase 6: Length exploration ===")
    
    trial_lengths = []
    for delta in [-500, 500, -1000, 1000, 1500, -1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 7000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    
    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(_global_best, new_n)
        
        t6a = min(remaining * 0.3, 200.0)
        if t6a > 20:
            a_out, _ = lbfgs_optimize(a_long, t6a, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"lbfgs_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.2, 100.0)
            if t6b > 10:
                a_out, _ = lbfgs_exact(a_long, t6b, verbose=False, top_k=30)
                try_update(a_out, f"exact_n{new_n}")
    
    # === Phase 7: Final L-BFGS + exact on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_global_best)
    print(f"\n=== Phase 7: Final optimization (n={n_best}) ===")
    
    t7a = min(remaining * 0.3, 200.0)
    if t7a > 20:
        a_in = _global_best.copy()
        a_out, _ = lbfgs_optimize(a_in, t7a, verbose=True, rng_seed=999)
        try_update(a_out, "final_lbfgs")
    
    remaining = time_budget - (time.monotonic() - t_global)
    t7b = min(remaining * 0.25, 150.0)
    if t7b > 15:
        a_in = _global_best.copy()
        a_out, _ = lbfgs_exact(a_in, t7b, verbose=True, top_k=50)
        try_update(a_out, "final_exact")
    
    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final coordinate polish ===")
    t8 = min(remaining * 0.4, 200.0)
    if t8 > 15:
        a_in = _global_best.copy()
        a_out, _ = coordinate_descent(a_in, t8, verbose=True, rng_seed=789)
        try_update(a_out, "final_coord")
    
    # === Phase 9: One more gradient flow ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 9: Final gradient flow ===")
        t9 = min(remaining * 0.5, 150.0)
        if t9 > 15:
            a_in = _global_best.copy()
            a_out, _ = gradient_flow(a_in, t9, lr=0.003, verbose=False)
            try_update(a_out, "final_gradflow")
    
    # === Final validation ===
    if _global_best is not None:
        final_score = 2.0 * len(_global_best) * fast_autocorr(_global_best).max()
        print(f"\nFINAL: score={final_score:.6f}, n={len(_global_best)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        _global_best = make_u_shape(n_fb, 2.0)
        final_score = 2.0 * n_fb * fast_autocorr(_global_best).max()
        print(f"Fallback: score={final_score:.6f}")
    
    return [float(x) for x in _global_best]
