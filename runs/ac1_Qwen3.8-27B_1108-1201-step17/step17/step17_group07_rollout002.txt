def fast_conv(a):

    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

def compute_exact_subgrad(y, ac):
    """Compute exact subgradient using all lags at the maximum."""
    n = len(y)
    M = ac.max()
    # Use lags within 1e-12 of max (or top few if numeric issues)
    active = np.where(ac >= M - 1e-14 * max(1.0, M))[0]
    if len(active) == 0:
        active = np.array([np.argmax(ac)])
    if len(active) > 200:
        active = np.argsort(ac)[::-1][:200]
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= len(active)
    return g

def nesterov_mirror_descent(a0, time_limit, top_k=50, c0=3.0, verbose=False):
    """Nesterov mirror descent with adaptive subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 2000
    rng = np.random.default_rng(42 + n)
    c = c0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        ac = fast_conv(y)
        M = ac.max()
        
        # Adaptive: use exact subgrad if few active, top-k if many
        near_max = np.sum(ac >= M - 1e-14 * max(1.0, M))
        if near_max <= 200:
            g = compute_exact_subgrad(y, ac)
        else:
            k_use = min(max(top_k, int(near_max)), 100)
            top_idx = np.argpartition(ac, -k_use)[-k_use:]
            g = np.zeros(n)
            for k in top_idx:
                k_int = int(k)
                j_start = max(0, k_int - n + 1)
                j_end = min(n - 1, k_int)
                if j_start <= j_end:
                    idx = np.arange(j_start, j_end + 1)
                    g[idx] += 2.0 * y[k_int - idx]
            g /= k_use
        
        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(300, int(max_stall * 0.5))
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.015, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n//3, 800), replace=False)
                x_r[idx] *= rng.uniform(0.6, 1.4, len(idx))
            elif strategy == 2:
                x_r = best_a * (1 + rng.normal(0, 0.003, n))

                x_r = x_r * (1 + rng.normal(0, 0.005, n))
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            c = max(0.3, c * 0.5)
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.10f}, stall={stall_count}")
    
    return best_a, best_m

def remez_lp_improved(a, budget_s, solver=None, verbose=False):
    """Improved LP Remez with quadratic correction and adaptive active set."""
    n = len(a)
    a = norm_sum1(a).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = 0.05
    no_improve = 0
    quad_correction = np.zeros(2 * n - 1)  # Tracks (d*d)[k] from last step
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > budget_s:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Adaptive active set: all lags within adaptive threshold
        threshold = max_b - max(0.001 * max_b, 1e-10)
        active = np.where(b >= threshold)[0]
        max_cons = 60
        if len(active) > max_cons:
            active = np.argsort(b)[::-1][:max_cons]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1]
        
        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                coeffs = 2.0 * a[k_int - idx]
                # Quadratic correction: subtract known (d*d)[k] from previous step
                rhs = b[k_int] - quad_correction[k_int]
                constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= rhs)
        
        constraints.append(a_var <= a * (1.0 + r))
        constraints.append(a_var >= a * max(0.0, 1.0 - r))
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
        except Exception:
            no_improve += 1
            r = max(0.002, r * 0.5)
            if no_improve > 30:
                break
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 1
            r = max(0.002, r * 0.5)
            if no_improve > 30:
                break
            continue
        
        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s = a_new.sum()
        if s < 1e-15:
            no_improve += 1
            continue
        a_new = a_new / s
        
        # Compute quadratic correction for next iteration
        d = a_new - a
        dd = fast_conv(d)
        quad_correction = np.maximum(dd, 0) * 0.5  # Conservative: only positive part

        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_new
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                d = a_trial - a
                quad_correction = np.maximum(fast_conv(d), 0) * 0.5
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.15, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.001, r * 0.6)
            if no_improve > 25:
                # Perturb to escape
                rng = np.random.default_rng(12345 + t)
                idx = rng.choice(n, size=min(n//4, 500), replace=False)
                a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()

                r = 0.02
        
        if verbose and t % 20 == 0:

            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.10f}, r={r:.4f}, active={len(active)}")
    
    return best_a, best_m

def newton_equalize(a0, budget_s, verbose=False):
    """Newton-based step to equalize the top convolution lags."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > budget_s:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Select active lags: top m where m adapts
        m_active = min(20, n // 10)
        active = np.argsort(b)[::-1][:m_active]
        
        # Build Jacobian J[i][j] = 2*a[k_i - j] for valid j
        # Solve J @ da = r with constraint sum(da) = 0, a + da >= 0
        # Use least squares with the sum constraint
        
        J = np.zeros((m_active, n))
        r = np.zeros(m_active)
        for i, k in enumerate(active):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                J[i, idx] = 2.0 * a[k_int - idx]
            r[i] = M - b[k_int]  # We want to reduce b[k] to M (already at M for top)
            # Actually we want to equalize: target T = mean of top values
            r[i] = b[k_int] - b[active].mean()  # Push toward mean
        
        # Add sum constraint as a row
        J_aug = np.vstack([J, np.ones(n) * 0.01])
        r_aug = np.append(r, 0.0)
        
        try:
            da, residuals, rank, sv = np.linalg.lstsq(J_aug, r_aug, rcond=None)
        except Exception:
            da = np.zeros(n)
        
        # Clip to maintain non-negativity and limit step
        step = np.linalg.norm(da)
        if step > 1e-15:
            da = da / step
        max_step = 0.02
        da = np.clip(da, -a * 0.5, a * 0.5)
        da_sum = da.sum()
        da -= da_sum / n  # Re-center
        
        # Line search
        accepted = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_trial = a + alpha * da
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                break
        
        if not accepted:
            # Try random perturbation
            if t % 100 == 0:
                idx = rng.choice(n, size=min(n//5, 300), replace=False)
                a[idx] *= rng.uniform(0.85, 1.15, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m_check = fast_conv(a).max()
                if m_check < best_m:
                    best_m = m_check
                    best_a = a.copy()
                else:
                    a = best_a.copy()  # Revert
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Newton t={t} ({elapsed:.1f}s): J={J:.10f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, budget_s, rng_seed=42, verbose=False):
    """Coordinate descent with pair moves and bisection for best frac."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 80)
    
    while time.monotonic() - t0 < budget_s and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > budget_s:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14 or a[j] < 1e-14:
                continue
            
            # Bisection for optimal fraction
            lo, hi = 0.0, min(a[i] * 0.5, a[j] * 0.5)
            best_frac = 0.0
            best_m_pair = fast_conv(a).max()
            
            for _ in range(8):
                mid = (lo + hi) / 2
                a_trial = a.copy()
                a_trial[i] -= mid
                a_trial[j] += mid
                if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                    hi = mid
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m_pair:
                    best_m_pair = m_trial
                    best_frac = mid
                    lo = mid
                else:
                    hi = mid
            
            if best_frac > 1e-12 and best_m_pair < best_m - 1e-16:
                a[i] -= best_frac
                a[j] += best_frac
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_m_pair
                best_a = a.copy()
                improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'symmetric_bump':
        a = np.exp(-((x - 0.5) / 0.15) ** 2)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        a_out, _ = remez_lp_improved(norm_sum1(best_seq), t1, solver=solver, verbose=True)
        try_update(a_out, f"Remez1_n{n0}")
    
    # === Phase 2: NMD with exact subgradient (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: NMD n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t2, top_k=50, c0=3.0, verbose=True)
        try_update(a_out, f"NMD_n{len(best_seq)}")
    
    # === Phase 3: Newton equalization (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Newton equalize n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = newton_equalize(norm_sum1(best_seq), t3, verbose=True)
        try_update(a_out, f"Newton_n{len(best_seq)}")
    
    # === Phase 4: Remez again on best (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Remez2 n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = remez_lp_improved(norm_sum1(best_seq), t4, solver=solver, verbose=False)
        try_update(a_out, f"Remez2_n{len(best_seq)}")
    
    # === Phase 5: NMD polish with small c (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: NMD polish n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), t5, top_k=30, c0=1.0, verbose=False)
        try_update(a_out, f"NMDpolish_n{len(best_seq)}")
    
    # === Phase 6: Coordinate descent polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 6: CD polish n={len(best_seq)} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t6, rng_seed=999, verbose=False)
        try_update(a_out, f"CD_n{len(best_seq)}")
    
    # === Phase 7: Try slightly different length (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n0 + 100
        print(f"\n=== Phase 7: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_alt = interpolate_sequence(best_seq, n_alt)
        
        t7a = min(remaining * 0.4, 100.0)
        if t7a > 20:
            a_out, _ = remez_lp_improved(a_alt, t7a, solver=solver, verbose=False)
            try_update(a_out, f"Remez_n{n_alt}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.4, 80.0)
            if t7b > 15:
                a_out, _ = nesterov_mirror_descent(a_alt, t7b, top_k=30, c0=2.0, verbose=False)
                try_update(a_out, f"NMD_n{n_alt}")
    
    # === Phase 8: Final combined push on global best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final push n={n_best} (budget={remaining:.0f}s) ===")
        
        # Remez with very small trust region
        t8a = min(remaining * 0.35, 100.0)
        if t8a > 10:
            a_out, _ = remez_lp_improved(norm_sum1(best_seq), t8a, solver=solver, verbose=False)
            try_update(a_out, f"RemezFinal_{n_best}")
        
        # Newton
        remaining = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining * 0.25, 80.0)
        if t8b > 10:
            a_out, _ = newton_equalize(norm_sum1(best_seq), t8b, verbose=False)
            try_update(a_out, f"NewtonFinal_{n_best}")
        
        # CD
        remaining = time_budget - (time.monotonic() - t_global)
        t8c = min(remaining * 0.25, 80.0)
        if t8c > 10:
            a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t8c, rng_seed=555, verbose=False)
            try_update(a_out, f"CDFinal_{n_best}")
        
        # Final NMD
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = nesterov_mirror_descent(norm_sum1(best_seq), remaining - 5, top_k=20, c0=0.5, verbose=False)
            try_update(a_out, f"NMDLast_{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
