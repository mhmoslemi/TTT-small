grad[2]/2 = b[2]*a0 + b[3]*a1 + b[4]*a2 = p2*a0 + p3*a1 + p4*a2

Comparing with (p*c):

(p*c)[3] = p1*a0 + p2*a1 + p3*a2 = grad[1]/2 ✓

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based exact gradient of smooth max (log-sum-exp) approximation
2. Adam optimizer with simplex projection
3. Adaptive beta scheduling (smooth → sharp)
4. Multi-strategy initialization and periodic restarts
5. Coordinate descent polish targeting top peaks
"""

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def simplex_project(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    r = idx[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(a, conv, beta):
    """
    Compute gradient of J_beta(a) = 2n * (1/beta) * log(sum_k exp(beta*conv[k]))
    using FFT-based method. Returns gradient w.r.t. a (before scaling by 2n/beta).
    
    Actually, let's just compute the gradient of max_k conv[k] approximated by log-sum-exp.
    The gradient of (1/beta)*log(sum exp(beta*conv[k])) is sum_k p_k * grad(conv[k])
    where p_k = softmax(beta*conv).
    
    We compute: grad_a[j] = sum_k p[k] * d(conv[k])/d(a[j])
    d(conv[k])/d(a[j]) = a[k-j]*I(0<=k-j<n) + a[j-k]*I(0<=j-k<n)
    
    Using the forward-backward trick:
    grad_a = 2 * (p * reverse(a))[n-1 : 2n-1]
    where * is full linear convolution.
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Compute softmax weights
    log_conv = np.log(np.maximum(conv, 1e-30))
    shifted = beta * log_conv
    shifted -= shifted.max()
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(conv_len) / conv_len
    else:
        p = p / p_sum
    
    # Compute gradient via FFT
    # grad[j] = 2 * sum_k p[k] * a[k-j] (with valid index constraints)
    # = 2 * (p * reverse(a))[j+n-1]
    
    a_rev = a[::-1]
    fft_size = 1 << (conv_len + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    temp = np.fft.irfft(P * AR, fft_size)[:conv_len + n - 1]
    
    grad = 2.0 * temp[n - 1:2 * n - 1]
    
    return grad


def adam_simplex(a0, time_limit, beta_start=5.0, beta_end=50.0,
                 lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8,
                 restart_every=5000, verbose=False, seed=42):
    """Adam optimizer with simplex projection for the normalized problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Adaptive beta
        progress = min(t / (restart_every * 3), 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute gradient
        conv = fast_conv(a)
        grad = compute_gradient_fft(a, conv, beta)
        # Scale by 2n (the actual J gradient)
        grad *= 2.0 * n
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = simplex_project(a_new)
        
        # Check improvement
        J_new = eval_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Restart
        if stall > restart_every:
            stall = 0
            # Diverse perturbation
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n)
            a_r = best_a + pert * best_a
            a = simplex_project(a_r)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Adam restart ({elapsed:.0f}s): J={best_J:.6f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): J={best_J:.6f}, beta={beta:.1f}")
    
    return best_a, best_J


def coordinate_descent(a0, time_limit, verbose=False, seed=42):
    """Targeted coordinate descent focusing on top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M_idx = np.argsort(conv)[::-1][:20]
        M = conv.max()
        
        # For each top peak, identify contributing pairs and try to reduce
        improved = False
        
        for k_peak in M_idx[:8]:
            k_int = int(k_peak)
            # Pairs (i, k-i) that contribute to this peak
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            
            if i_start > i_end:
                continue
            
            # Find the pair with largest product
            idx = np.arange(i_start, i_end + 1)
            products = a[idx] * a[k_int - idx]
            
            if products.max() < 1e-15:
                continue
            
            # Source: position with max contribution
            src_local = int(np.argmax(products))
            src1 = i_start + src_local
            src2 = k_int - src1
            
            # Try to move mass from the larger of the two to reduce the product
            if a[src1] >= a[src2]:
                src = src1
            else:
                src = src2
            
            if a[src] < 1e-12:
                continue
            
            # Find best destination: try positions that don't contribute to top peaks
            # Score each position by how much it would increase other peaks
            hot_positions = set()
            for kp in M_idx[:10]:
                kp_i = int(kp)
                hs = max(0, kp_i - n + 1)
                he = min(n - 1, kp_i)
                for h in range(hs, he + 1):
                    hot_positions.add(h)
            
            # Try non-hot positions first
            candidates = [j for j in range(n) if j != src and j not in hot_positions and a[j] < 0.5 * a.max()]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Score by minimum contribution to peaks
            def score_pos(j):
                sc = 0.0
                for kp in M_idx[:5]:
                    kp_i = int(kp)
                    # If we add mass to position j, it contributes to peak kp via a[j]*a[kp-j]
                    if 0 <= kp_i - j < n:
                        sc += a[kp_i - j]
                    if 0 <= j - kp_i < n:
                        sc += a[j - kp_i]
                return sc
            
            # Only score a subset for speed
            if len(candidates) > 200:
                candidates = rng.choice(candidates, 200, replace=False).tolist()
            
            scores = np.array([score_pos(j) for j in candidates])
            dst = candidates[int(np.argmin(scores))]
            
            # Try multiple transfer fractions
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t /= st
                J_t = eval_J(a_t)
                if J_t < best_J - 1e-14:
                    a = a_t
                    best_J = J_t
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.005, 0.03)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J


def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, c1=0.2, c2=0.8, width=50.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - c1) ** 2) + np.exp(-width * (x - c2) ** 2) + 0.003
    return a / a.sum()


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact FFT-based gradient of log-sum-exp (smooth max) approximation
2. Adam optimizer with simplex projection and adaptive beta scheduling
3. Multi-strategy initialization with diverse restarts
4. Targeted coordinate descent for final polish
5. Length interpolation exploration
"""

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_ab(a, b):
    """Compute full linear convolution of a and b using FFT."""
    len_a = len(a)
    len_b = len(b)
    size = len_a + len_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_conv(a)
    return 2.0 * n * conv.max()


def simplex_project(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.arange(1, n + 1)
    cond = mu * rho > (cssv - 1.0)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    r = idx[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, conv, beta):
    """
    Compute gradient of 2n * (1/beta) * log(sum_k exp(beta*conv[k])).
    
    Uses the identity: d(conv[k])/d(a[j]) = 2*a[k-j]*I(0<=k-j<n)
    So grad[j] = 4n * sum_m p[m+j]*a[m] = 4n * (p * a_rev)[j+n-1]
    
    where p = softmax(beta*conv), a_rev = reversed(a).
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Compute softmax weights (using log-sum-exp for stability)
    shifted = beta * np.log(np.maximum(conv, 1e-30))
    shifted -= shifted.max()
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(conv_len) / conv_len
    else:
        p = p / p_sum
    
    # Compute convolution p * a_rev via FFT
    a_rev = a[::-1]
    temp = fast_conv_ab(p, a_rev)
    
    # Extract gradient: grad[j] = 4n * temp[j+n-1]
    grad = 4.0 * n * temp[n - 1:2 * n - 1]
    
    return grad


def adam_simplex(a0, time_limit, beta_start=5.0, beta_end=50.0,
                 lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8,
                 restart_every=4000, lr_decay=0.95, verbose=False, seed=42):
    """Adam optimizer with simplex projection for normalized optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    cur_lr = lr
    total_restarts = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Adaptive beta: start smooth, end sharp
        progress = min(float(t) / float(restart_every * 4), 1.0)
        beta_cur = beta_start + (beta_end - beta_start) * (progress ** 0.7)
        
        # Compute gradient
        conv = fast_conv(a)
        grad = compute_gradient(a, conv, beta_cur)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        a_new = a - cur_lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = simplex_project(a_new)
        
        # Evaluate
        J_new = eval_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Decay learning rate slowly
        if t % 2000 == 0:
            cur_lr *= lr_decay
        
        # Restart logic
        if stall > restart_every:
            stall = 0
            total_restarts += 1
            cur_lr = lr * (0.8 ** total_restarts)
            cur_lr = max(cur_lr, 0.0005)
            
            # Diverse perturbation strategies
            strategy = total_restarts % 4
            if strategy == 0:
                # Gaussian perturbation
                scale = rng.uniform(0.01, 0.04)
                a_r = best_a + rng.normal(0, scale, n) * best_a
            elif strategy == 1:
                # Block perturbation
                n_blocks = rng.integers(3, 10)
                block_size = n // n_blocks
                a_r = best_a.copy()
                for _ in range(rng.integers(1, 4)):
                    b = rng.integers(n_blocks)
                    a_r[b*block_size:(b+1)*block_size] *= rng.uniform(0.5, 1.5)
            elif strategy == 2:
                # Frequency perturbation (low-pass)
                freq = rng.integers(1, 5)
                x = np.linspace(0, 1, n)
                mod = 1.0 + 0.05 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 6.28))
                a_r = best_a * mod
            else:
                # Sparse large perturbation
                a_r = best_a.copy()
                n_spikes = rng.integers(3, 15)
                for _ in range(n_spikes):
                    pos = rng.integers(n)
                    a_r[pos] *= rng.uniform(0.3, 2.0)
            
            a = simplex_project(a_r)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Restart {total_restarts} ({elapsed:.0f}s): J={best_J:.6f}, lr={cur_lr:.5f}")
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): J={best_J:.6f}, beta={beta_cur:.1f}, lr={cur_lr:.5f}")
    
    return best_a, best_J


def coordinate_descent_v2(a0, time_limit, verbose=False, seed=42):
    """Targeted coordinate descent focusing on reducing top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 150
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        top_lags = np.argsort(conv)[::-1][:15]
        
        improved = False
        
        for k_peak in top_lags[:6]:
            k_int = int(k_peak)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            
            idx_arr = np.arange(i_start, i_end + 1)
            products = a[idx_arr] * a[k_int - idx_arr]
            
            if products.max() < 1e-15:
                continue
            
            # Identify the dominant contributing pair
            src_local = int(np.argmax(products))
            src1 = i_start + src_local
            src2 = k_int - src1
            
            # Move mass from the larger element
            if a[src1] >= a[src2]:
                src = src1
            else:
                src = src2
            
            if a[src] < 1e-12:
                continue
            
            # Find candidate destinations (positions that contribute least to peaks)
            # Precompute hot zone
            hot_mask = np.zeros(n, dtype=bool)
            for kp in top_lags[:10]:
                kp_i = int(kp)
                hs = max(0, kp_i - n + 1)
                he = min(n - 1, kp_i)
                hot_mask[hs:he+1] = True
            
            # Try a mix of cold and random positions
            cold_positions = np.where(~hot_mask)[0]
            if len(cold_positions) > 50:
                candidates = rng.choice(cold_positions, 50, replace=False)
            else:
                candidates = rng.choice(n, 50, replace=False)
            
            # Also include some random positions
            random_positions = rng.choice(n, 20, replace=False)
            candidates = np.concatenate([candidates, random_positions])
            candidates = candidates[candidates != src]
            
            if len(candidates) == 0:
                continue
            
            # Score: how much would adding mass at position j increase the max peak?
            scores = np.zeros(len(candidates))
            for ci, j in enumerate(candidates):
                sc = 0.0
                for kp in top_lags[:5]:
                    kp_i = int(kp)
                    if 0 <= kp_i - j and kp_i - j < n:
                        sc += a[kp_i - j] ** 2
                    if 0 <= j - kp_i and j - kp_i < n:
                        sc += a[j - kp_i] ** 2
                scores[ci] = sc
            
            # Try best few destinations with multiple fractions
            best_dst_idx = np.argsort(scores)[:5]
            found_improvement = False
            
            for dit in best_dst_idx:
                dst = int(candidates[dit])
                for frac in [0.003, 0.008, 0.015, 0.03, 0.06, 0.12]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_t = a.copy()
                    a_t[src] -= amt
                    a_t[dst] += amt
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t /= st
                    J_t = eval_J(a_t)
                    if J_t < best_J - 1e-13:
                        a = a_t
                        best_J = J_t
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found_improvement = True
                        break
                if found_improvement:
                    break
        
        if not improved:
            stall += 1
            if stall % 15 == 0:
                # Escalating random perturbation
                n_trials = min(stall // 10 + 1, 10)
                for _ in range(n_trials):
                    i1 = int(rng.integers(n))
                    i2 = int(rng.integers(n))
                    if i1 != i2 and a[i1] > 1e-10:
                        amt = a[i1] * rng.uniform(0.01, 0.05)
                        a[i1] -= amt
                        a[i2] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, c1=0.2, c2=0.8, width=40.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - c1) ** 2) + np.exp(-width * (x - c2) ** 2) + 0.003
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier components."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 12):
        coeff = rng.normal(0, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    print(f"Budget: {time_budget:.0f}s")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent loaded: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass
    
    if parent is None:
        n0 = 3200
        parent = make_power_ramp(n0, 1.0)
        print(f"No parent, using power ramp: n={len(parent)}, J={eval_J(parent):.6f}")
    
    best_a = parent.copy()
    best_J = eval_J(best_a)
    n = len(best_a)
    
    def try_update(a, label):
        nonlocal best_a, best_J
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Adam on parent (primary optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam primary (n={n}, {remaining*0.35:.0f}s) ===")
    t1 = remaining * 0.35
    a_out, _ = adam_simplex(best_a.copy(), t1, beta_start=5.0, beta_end=40.0,
                            lr=0.008, restart_every=3000, lr_decay=0.93,
                            verbose=True, seed=1)
    try_update(a_out, "adam_p1")
    
    # === Phase 2: Adam with different hyperparameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adam v2 ({remaining*0.18:.0f}s) ===")
    t2 = remaining * 0.18
    a_out, _ = adam_simplex(best_a.copy(), t2, beta_start=15.0, beta_end=80.0,
                            lr=0.004, restart_every=2500, lr_decay=0.9,
                            verbose=True, seed=7)
    try_update(a_out, "adam_p2")
    
    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coordinate Descent ({remaining*0.15:.0f}s) ===")
    t3 = remaining * 0.15
    a_out, _ = coordinate_descent_v2(best_a.copy(), t3, verbose=True, seed=42)
    try_update(a_out, "cd_p3")
    
    # === Phase 4: Adam after CD (fine-tuning) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Adam fine-tune ({remaining*0.12:.0f}s) ===")
    t4 = remaining * 0.12
    a_out, _ = adam_simplex(best_a.copy(), t4, beta_start=25.0, beta_end=100.0,
                            lr=0.002, restart_every=5000, lr_decay=0.95,
                            verbose=True, seed=13)
    try_update(a_out, "adam_p4")
    
    # === Phase 5: Multi-start exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-start ({remaining*0.25:.0f}s) ===")
    
    inits = []
    labels = []
    inits.append(make_power_ramp(n, 0.6))
    labels.append("ramp0.6")
    inits.append(make_power_ramp(n, 1.4))
    labels.append("ramp1.4")
    inits.append(make_bimodal(n, 0.15, 0.85, 30.0))
    labels.append("bimodal")
    inits.append(make_smooth_random(n, seed=7))
    labels.append("smooth7")
    
    # Blend with parent
    b1 = 0.5 * best_a + 0.5 * make_power_ramp(n, 1.5)
    inits.append(b1 / b1.sum())
    labels.append("blend_a")
    
    b2 = 0.6 * best_a + 0.4 * make_bimodal(n)
    inits.append(b2 / b2.sum())
    labels.append("blend_b")
    
    t_per = max(15.0, remaining * 0.25 / len(inits))
    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_avail = min(t_per, remaining - 15)
        if t_avail < 10:
            break
        print(f"  Multi-start [{lbl}] ({t_avail:.0f}s)...")
        a_out, _ = adam_simplex(init, t_avail, beta_start=5.0, beta_end=30.0,
                                lr=0.006, restart_every=2500, verbose=False,
                                seed=100 + idx * 17)
        try_update(a_out, f"ms_{lbl}")
    
    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length exploration ===")
    n_best = len(best_a)
    
    length_trials = []
    for delta in [-300, -150, 150, 300, 500]:
        nl = n_best + delta
        if 2000 <= nl <= 5000:
            length_trials.append(nl)
    
    for nl in sorted(length_trials):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if abs(nl - n_best) <= 50:
            continue
        print(f"  Trying n={nl}...")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.3, 80.0)
        if t_md > 15:
            a_out, _ = adam_simplex(a_int, t_md, beta_start=5.0, beta_end=30.0,
                                    lr=0.006, restart_every=2000, verbose=False,
                                    seed=nl % 1000)
            try_update(a_out, f"len_{nl}")
    
    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final Polish ===")
    n_fin = len(best_a)
    
    if remaining > 40:
        # CD polish
        t_cd = remaining * 0.35
        print(f"  CD polish ({t_cd:.0f}s)...")
        a_out, _ = coordinate_descent_v2(best_a.copy(), t_cd, verbose=True, seed=999)
        try_update(a_out, "final_cd")
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        # Final Adam with very high beta (essentially exact max)
        t_fa = remaining * 0.5
        print(f"  Final Adam ({t_fa:.0f}s)...")
        a_out, _ = adam_simplex(best_a.copy(), t_fa, beta_start=40.0, beta_end=200.0,
                                lr=0.001, restart_every=8000, lr_decay=0.97,
                                verbose=True, seed=7777)
        try_update(a_out, "final_adam")
    
    # === Final evaluation ===
    final_J = eval_J(best_a)
    best_J = min(best_J, final_J)
    elapsed = time.monotonic() - t_global
    print(f"\nFINAL RESULT: J={best_J:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
    
    result = [float(x) for x in best_a]
    return result
