def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def get_subgradient_adaptive(a, ac, top_k=None, beta=3.0):
    """
    Compute adaptive subgradient: includes all lags where conv > threshold * max.
    Falls back to top_k if too many peaks.
    """
    n = len(a)
    grad = np.zeros(n)
    M = ac.max()
    if M < 1e-30:
        return grad

    # Adaptive: include all prominent peaks (above 80% of max)
    if top_k is None:
        threshold = 0.8 * M
        peak_idx = np.where(ac >= threshold)[0]
    else:
        peak_idx = np.argsort(ac)[::-1][:top_k]

    if len(peak_idx) == 0:
        peak_idx = np.array([int(np.argmax(ac))])

    # Weight by peak value raised to beta
    peak_vals = np.maximum(ac[peak_idx], 1e-30)
    w = peak_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(peak_idx)) / len(peak_idx)
    else:
        w = w / w_sum

    for wi, k in zip(w, peak_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]

    return grad


def spectral_shape_init(n, method='highpass', beta_boost=0.3, seed=42):
    """
    Generate initialization using spectral shaping.
    Methods: 'highpass' (boost high freq), 'random_phase' (flat mag + random phase).
    """
    rng = np.random.default_rng(seed)
    L = 1 << (2 * n - 1 - 1).bit_length()

    if method == 'highpass':
        # Start with uniform, apply high-pass boost to DFT magnitude
        a = np.ones(n) / n
        A = np.fft.rfft(a, L)
        freqs = np.arange(len(A))
        # Boost high frequencies
        max_f = len(A) - 1
        boost = (freqs / max_f) ** beta_boost + 0.1
        A_mag_new = np.abs(A) * boost
        A_new = A_mag_new * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, L)[:n]
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s

    elif method == 'random_phase':
        # Flat magnitude with random phases
        best_a = np.ones(n) / n
        best_J = compute_J(best_a)
        for trial in range(50):
            phases = rng.uniform(0, 2 * np.pi, len(np.arange(0, L)))
            phases = np.concatenate([phases[0:1], phases, phases[1:-1][::-1]])
            # For real signal: use rfft Convention
            half_len = L // 2 + 1
            h_phases = rng.uniform(0, 2 * np.pi, half_len)
            mag = np.ones(half_len)
            A = mag * np.exp(1j * h_phases)
            a_new = np.fft.irfft(A, L)[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            J = compute_J(a_new)
            if J < best_J:
                best_J = J
                best_a = a_new.copy()
        return best_a

    elif method == 'tukey_window':
        # Tukey (raised cosine) window - good for reducing sidelobes
        r = 0.3  # Tukey parameter
        x = np.arange(n)
        w = np.ones(n)
        edge = int(r * n / 2)
        if edge > 0:
            w[:edge] = 0.5 * (1 + np.cos(np.pi * (np.arange(edge) / edge - 1)))
            w[-edge:] = 0.5 * (1 + np.cos(np.pi * (np.arange(edge) / edge - 1)))
        w = np.maximum(w, 0)
        s = w.sum()
        return w / s

    else:
        return np.ones(n) / n


def iht_flatten(a0, time_limit=30.0, n_iters=500, seed=42):
    """
    IHT-style flattening: iteratively reduce the peak of conv(a,a)
    by adjusting in the frequency domain.
    """
    n = len(a0)
    L = 1 << (2 * n - 1 - 1).bit_length()
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current conv
        A = np.fft.rfft(a, L)
        conv = np.fft.irfft(A * A, L)[:2 * n - 1]

        # Compute target: reduce peaks, boost valleys
        M = conv.max()
        m = conv.mean()
        if M < 1e-15:
            break

        # Frequency domain adjustment:
        # The "error" is conv - m (positive at peaks, negative at valleys)
        # DFT of error = |A|^2 - DFT(constant sequence of length 2n-1 with value m)
        # We want to reduce |A|^2 where it contributes to peaks
        # Simple approach: scale |A| by a factor that reduces overall peak

        # Compute the "peakiness" ratio
        peak_ratio = M / (m + 1e-30)

        # Apply a gentle spectral sharpening: boost frequencies that
        # correspond to anti-correlation (reduce autocorrelation peaks)
        half_len = L // 2 + 1
        freqs = np.arange(half_len)
        # High-pass boost (reduces low-freq components that cause broad peaks)
        boost = 1.0 + 0.05 * (freqs / (half_len - 1)) ** 2
        A_mag = np.abs(A) * boost
        A_new = A_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()

        # Also try a small time-domain perturbation every 50 iters
        if it % 50 == 49:
            a_pert = a + rng.normal(0, 0.01, n) * a
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert / s_p
                J_p = compute_J(a_pert)
                if J_p < best_J:
                    best_J = J_p
                    best_a = a_pert.copy()
                    a = a_pert.copy()

        a = a_new

    return best_a, best_J

        y = np.maximum(y, 1e-15)

            stall += 1

                x = x / x.sum()
                x_prev = x.copy()
                stall = 0
            continue

        # Polyak step with line search
        gap = M - lower_bound
        eta_base = cur_c * gap / (g_norm_sq + 1e-20)

        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new

        else:

                A_mag = np.abs(A) * (1 + rng.normal(0, 0.05, len(A)))
                A_new = np.maximum(A_mag, 0) * np.exp(1j * np.angle(A))

        if verbose and t % 50000 == 0:

def peak_reduction_v2(a0, time_limit, verbose=False, seed=77):
    """
    Improved peak reduction: uses gradient-based sensitivity to find
    optimal source-destination pairs for mass transfer.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find the dominant peak
        k_peak = int(np.argmax(ac))

        # Compute sensitivity: for each position i, how much does reducing a[i]
        # reduce the peak?
        # d(conv[k_peak])/d(a[i]) = 2 * a[k_peak - i] (for valid i)
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_peak - idx

        # Sensitivity of the peak w.r.t. a[i]
        sensitivity = np.zeros(n)
        sensitivity[idx] = 2.0 * a[j_idx]

        # Also consider second-largest peak
        ac_copy = ac.copy()
        ac_copy[k_peak] = 0
        k_peak2 = int(np.argmax(ac_copy))
        i_start2 = max(0, k_peak2 - n + 1)
        i_end2 = min(n, k_peak2 + 1)
        idx2 = np.arange(i_start2, i_end2)
        j_idx2 = k_peak2 - idx2
        sensitivity[idx2] += 2.0 * a[j_idx2]

        # Find positions to move mass FROM (high sensitivity + high value)
        # and positions to move mass TO (low sensitivity)
        high_score = sensitivity * a  # contribution to peak
        low_sens = np.where(sensitivity < np.percentile(sensitivity, 20))[0]
        high_sens = np.where((sensitivity > np.percentile(sensitivity, 70)) & (a > 1e-6))[0]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1

            if time.monotonic() - t0 > time_limit:
                break

            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            # Try multiple transfer fractions

                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if found:
                stall = 0

        if not improved:
            stall += 1
            if stall % 30 == 0:
                # Larger perturbation

                perturb_positions = rng.choice(n, size=n_pert, replace=False)
                for p in perturb_positions:
                    if a[p] > 1e-8:
                        frac = rng.uniform(0.01, 0.05)
                        amt = a[p] * frac
                        a[p] -= amt
                        dst = int(rng.integers(n))
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:

        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix for top-K binding constraints
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # CW step size
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        y = y_new
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_symmetric_init(n):
    """Create a symmetric initialization (high at edges, low in middle)."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_power_law_init(n, power=0.5):
    """Power law from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.001
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.15):
    """Gaussian bumps at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)

    Main search orchestrator combining spectral initialization,
    improved subgradient, peak reduction, and LP refinement.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Spectral initialization + IHT (250s) ===
    print(f"\n=== Phase 1: Spectral init + IHT (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.14, 250.0)

    if t_phase1 > 30:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_symmetric_init(n0))
        labels.append("symmetric")
        inits.append(make_power_law_init(n0, power=0.7))
        labels.append("power0.7")
        inits.append(make_gaussian_edges(n0, sigma=0.12))
        labels.append("gauss_edges")
        inits.append(spectral_shape_init(n0, 'highpass', 0.3, seed=1))
        labels.append("highpass")
        inits.append(spectral_shape_init(n0, 'highpass', 0.5, seed=2))
        labels.append("highpass_strong")
        inits.append(spectral_shape_init(n0, 'tukey_window', seed=3))
        labels.append("tukey")
        inits.append(spectral_shape_init(n0, 'random_phase', seed=4))
        labels.append("rand_phase")

        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_symmetric_init(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend")
            blend2 = 0.6 * norm_sum1(parent) + 0.4 * spectral_shape_init(n0, 'highpass', 0.4, seed=5)
            inits.append(norm_sum1(blend2))
            labels.append("blend_hp")

        # Evaluate all inits
        for init, lbl in zip(inits, labels):
            sc = evaluate_safe(init)
            print(f"    Init '{lbl}': score={sc:.6f}")

        # Run IHT on the top 3 inits
        init_scores = [(evaluate_safe(init), init, lbl) for init, lbl in zip(inits, labels)]
        init_scores.sort(key=lambda x: x[0])
        t_per_iht = max(10.0, t_phase1 / 8)
