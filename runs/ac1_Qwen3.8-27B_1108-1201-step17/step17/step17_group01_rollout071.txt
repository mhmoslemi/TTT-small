  n=len(a)

"""
Search for low max-convolution sequences by combining:
- fast FFT-based evaluation of P(z)^2 coefficients,
- annealed exponentiated-gradient / mirror descent in the simplex,
- peak-shaving spectral square-root projection,
- low-cost one-pair and mean-direction coordinate moves,
- occasional smooth restarts and length interpolation.

The returned sequence is nonnegative and scale-invariant with respect to the
score 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time


def next_power_of_two(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_nonneg(a):
    x = np.asarray(a, dtype=np.float64).ravel()
    if x.size == 0:
        return x
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    x = x / s
    s = float(x.sum())
    if s <= 1e-300:
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    return x / s


def convolution_and_objective(a, size, n):
    arr = np.asarray(a, dtype=np.float64).ravel()
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf"), np.zeros(2 * n - 1, dtype=np.float64)
    A = np.fft.rfft(arr, size)
    c = np.fft.irfft(A * A, size)[: (2 * n - 1)]
    c = np.maximum(c, 0.0)
    m = float(c.max())
    if not np.isfinite(m):
        return float("inf"), c
    return float(2.0 * float(n) * m / (s * s)), c


def make_positive_weights(c, sigma):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = c.size
    w = np.zeros(L, dtype=np.float64)
    if L == 0:
        return w
    m = float(c.max())
    if not np.isfinite(m) or m <= 1e-300:
        w[int(np.argmax(c))] = 1.0
        return w
    pos = c > 0.0
    if not np.any(pos):
        w[0] = 1.0
        return w
    rel = (m - c[pos]) / m
    vals = np.exp(-np.maximum(rel, 0.0) / max(float(sigma), 1e-12))
    w[pos] = vals
    s = float(w.sum())
    if not np.isfinite(s) or s <= 1e-300:
        w[:] = 0.0
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return w


def gradient_from_weights(a, w, size):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    K = 2 * n - 2
    wr = np.ascontiguousarray(w[::-1])
    Aw = np.fft.rfft(wr, size)
    Aa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Aw * Aa, size)
    start = K - n + 1
    stop = K + 1
    g = 2.0 * conv[start:stop][::-1]
    g = np.where(g > 0.0, g, 0.0)
    return g


def exp_update_gmax(a, g, gm, step):
    a = np.asarray(a, dtype=np.float64).ravel()
    g = np.asarray(g, dtype=np.float64).ravel()
    if gm <= 1e-300 or not np.isfinite(gm):
        return None
    u = a * np.exp(-float(step) * (g / gm))
    u = np.maximum(u, 0.0)
    s = float(u.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return u / s


def smooth_perturbation(a, rng, amp, exp_scale):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return a
    x = np.arange(n, dtype=np.float64)
    z = np.zeros(n, dtype=np.float64)
    freqs = min(40, max(1, n // 2))
    two_pi = 2.0 * np.pi
    for f in range(1, freqs + 1):
        coefficient = float(amp) / np.sqrt(float(f))
        ph1 = float(rng.uniform(0.0, two_pi))
        ph2 = float(rng.uniform(0.0, two_pi))
        z += coefficient * (np.sin(two_pi * f * x / float(n) + ph1) +
                            np.cos(two_pi * f * x / float(n) + ph2))
    mean = float(z.mean())
    std = float(z.std())
    if std > 1e-15:
        z = (z - mean) / std
    else:
        z = np.zeros_like(z)
    mult = np.exp(float(exp_scale) * z)
    return normalize_nonneg(a * mult)


def reheat_mixture(a, best, rng, mix, amp, exp_scale):
    a = np.asarray(a, dtype=np.float64).ravel()
    if best is None or len(np.asarray(best).ravel()) == 0:
        out = a.copy()
    else:
        best = np.asarray(best, dtype=np.float64).ravel()
        out = (1.0 - float(mix)) * a + float(mix) * best
    return smooth_perturbation(out, rng, amp, exp_scale)


def interpolate_normalized(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    old_n = a.size
    if old_n == 0:
        return np.asarray([], dtype=np.float64)
    if int(new_n) == old_n:
        return normalize_nonneg(a)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, int(new_n))
    b = np.interp(x_new, x_old, a)
    return normalize_nonneg(b)

def clip_threshold_for_max(c, M):
    c = np.asarray(c, dtype=np.float64).ravel()
    S = float(c.sum())
    if S <= 1e-300 or not np.isfinite(S):
        return 0.0
    m = float(c.max())
    if not np.isfinite(m) or m <= 1e-300:
        return 0.0
    M = float(M)
    if M <= 1e-300:
        return 0.0
    if M >= m:
        return m
    T = max(1e-300, M * S)
    for _ in range(60):
        Scl = float(np.minimum(c, T).sum())
        if Scl <= 1e-300 or not np.isfinite(Scl):
            break
        T_new = M * Scl
        if abs(T_new - T) <= 1e-18 * max(1.0, T):
            T = T_new
            break
        T = 0.5 * T + 0.5 * T_new
    return max(1e-300, T)

        a_new = np.maximum(a_new, 0.0)

        obj_new, c_new = convolution_and_objective(a_new, size, n)

            stall += 1

                if current_obj > best_obj * 1.10:
                    a = best.copy()
                    current_obj, c = convolution_and_objective(a, size, n)
                reduce = 0.025
                stall = 0

    return best, best_obj


def run_local_pair_search(a_init, sec, seed, s_count=70, d_count=70, max_evals=80):
    a = normalize_nonneg(a_init)
    n = int(a.size)
    if n < 4:
        return a, float("inf")
    size = next_power_of_two(3 * n + 1)
    obj, c = convolution_and_objective(a, size, n)
    current_obj = obj
    best = a.copy()
    best_obj = current_obj
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    sigma = 0.05
    stall = 0
    tol = 1e-14
    s_count = max(8, min(int(s_count), n))
    d_count = max(8, min(int(d_count), n))

    while time.monotonic() - t0 < sec:
        if current_obj > best_obj * 1.05:
            a = best.copy()
            current_obj, c = convolution_and_objective(a, size, n)

        w = make_positive_weights(c, sigma)
        g = gradient_from_weights(a, w, size)
        gm = float(g.max())
        if gm <= 1e-300 or not np.isfinite(gm):
            break
        gn = g / gm
        src_score = a * gn
        src_idx = np.argsort(src_score, kind="stable")[::-1][:s_count]
        dst_idx = np.argsort(gn, kind="stable")[:d_count]
        gs = gn[src_idx].reshape(-1, 1)
        gd = gn[dst_idx].reshape(1, -1)
        gains = gd - gs
        gains_flat = gains.ravel()
        if gains_flat.size == 0:
            break
        order = np.argsort(gains_flat, kind="stable")

        if gains_flat[order[0]] > -0.02 and stall > 5:
            stall += 4
        else:
            improved = False
            evaluated = 0
            for pos in order:
                if evaluated >= max_evals:
                    break
                i = int(pos) // d_count
                j = int(pos) % d_count
                si = int(src_idx[i])
                dj = int(dst_idx[j])
                if si == dj:
                    continue
                v = float(a[si])
                if v <= 1e-300:
                    continue
                for frac in (0.05, 0.10, 0.20, 0.08):
                    amt = v * float(frac)
                    if amt <= 1e-300 or amt > v:
                        continue
                    trial = a.copy()
                    trial[si] -= amt
                    trial[dj] += amt
                    obj_try, c_try = convolution_and_objective(trial, size, n)
                    evaluated += 1
                    if obj_try < current_obj - tol:
                        a = trial
                        c = c_try
                        current_obj = obj_try
                        if current_obj < best_obj - tol:
                            best_obj = current_obj
                            best = a.copy()
                        improved = True
                        break
                if improved:
                    break

            if improved:
                stall = 0
            else:
                stall += 1
                if stall % 8 == 0:
                    sigma *= 0.72
                    if sigma < 0.002:
                        sigma = 0.16
                if stall > 70:
                    a = reheat_mixture(a, best, rng, 0.30, 0.22, 0.28)
                    current_obj, c = convolution_and_objective(a, size, n)
                    if current_obj > best_obj * 1.10:
                        a = best.copy()
                        current_obj, c = convolution_and_objective(a, size, n)
                    stall = 0
                    sigma = 0.05
        if stall > 70:
            a = reheat_mixture(a, best, rng, 0.30, 0.22, 0.28)
            current_obj, c = convolution_and_objective(a, size, n)
            if current_obj > best_obj * 1.10:
                a = best.copy()
                current_obj, c = convolution_and_objective(a, size, n)
            stall = 0
            sigma = 0.05

    return best, best_obj


def run_mean_direction_search(a_init, sec, seed, low_count=80):
    a = normalize_nonneg(a_init)
    n = int(a.size)
    if n < 4:
        return a, float("inf")
    size = next_power_of_two(3 * n + 1)
    obj, c = convolution_and_objective(a, size, n)
    current_obj = obj
    best = a.copy()
    best_obj = current_obj
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    stale = 0
    tol = 1e-14
    low_count = max(8, min(int(low_count), n))

    while time.monotonic() - t0 < sec:
        if current_obj > best_obj * 1.05:
            a = best.copy()
            current_obj, c = convolution_and_objective(a, size, n)

        w = make_positive_weights(c, 0.055)
        g = gradient_from_weights(a, w, size)
        gm = float(g.max())
        if gm <= 1e-300 or not np.isfinite(gm):
            break
        gn = g / gm
        low = np.argsort(gn, kind="stable")[:low_count]
        x = np.zeros(n, dtype=np.float64)
        x[low] = 1.0 / float(low.size if low.size else n)

        best_trial_obj = current_obj
        best_trial_a = None
        best_trial_c = None
        for rho in (0.16, 0.10, 0.06, 0.24, 0.32):
            trial = (1.0 - rho) * a + rho * x
            trial = normalize_nonneg(trial)
            obj_try, c_try = convolution_and_objective(trial, size, n)
            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_trial_a = trial
                best_trial_c = c_try

        if best_trial_a is not None and best_trial_obj < current_obj - tol:
            a = best_trial_a
            c = best_trial_c
            current_obj = best_trial_obj
            if current_obj < best_obj - tol:
                best_obj = current_obj
                best = a.copy()
            stale = 0
        else:
            stale += 1
            if stale > 45:
                a = reheat_mixture(a, best, rng, 0.30, 0.22, 0.28)
                current_obj, c = convolution_and_objective(a, size, n)
                if current_obj > best_obj * 1.10:
                    a = best.copy()
                    current_obj, c = convolution_and_objective(a, size, n)
                stale = 0

    return best, best_obj


def propose_candidate():
    start_time = time.monotonic()
    budget = 1750.0

    parent = None
    try:
        p = height_sequence_1
        if p is not None:
            parent = p
    except Exception:
        parent = None

    best_seq = None
    init = None

    if parent is not None:
        try:
            arr = np.asarray(parent, dtype=np.float64).ravel()
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.maximum(arr, 0.0)
            if arr.size > 0 and float(arr.sum()) > 1e-8:
                best_seq = normalize_nonneg(arr)
                init = best_seq
        except Exception:
            best_seq = None
            init = None

    if best_seq is None:
        best_seq = default_initial(4200, 7)
        init = best_seq

    best_val = actual_eval(best_seq)
    if not np.isfinite(best_val):
        n = int(best_seq.size)
        size = next_power_of_two(3 * n + 1)
        best_val, _ = convolution_and_objective(best_seq, size, n)

    print(f"Start: n={len(best_seq)}, actual={best_val:.6f}", flush=True)

    # Phase 1: strong mirror polish from parent / current best.
    remain = budget - (time.monotonic() - start_time)
    if remain > 180.0:
        t = min(650.0, remain - 130.0)
        cand, _ = run_mirror_polish(init, t, 101)
        best_seq, best_val, upd = better_candidate(best_seq, best_val, cand)
        if upd:
            init = normalize_nonneg(best_seq)
            print(f"After mirror-1: n={len(best_seq)}, actual={best_val:.6f}", flush=True)

    # Phase 2: spectral square-root projection to flatten the peak profile.
    remain = budget - (time.monotonic() - start_time)
    if remain > 220.0:
        t = min(260.0, remain - 230.0)
        cand, _ = run_square_projection(init, t, 202, floor_obj=1.50)
        best_seq, best_val, upd = better_candidate(best_seq, best_val, cand)
        if upd:
            init = normalize_nonneg(best_seq)
            print(f"After spectral: n={len(best_seq)}, actual={best_val:.6f}", flush=True)

    # Phase 3: local pair moves.
    remain = budget - (time.monotonic() - start_time)
    if remain > 180.0:
        t = min(320.0, remain - 210.0)
        cand, _ = run_local_pair_search(init, t, 303, 70, 70, 80)
        best_seq, best_val, upd = better_candidate(best_seq, best_val, cand)
        if upd:
            init = normalize_nonneg(best_seq)
            print(f"After local-1: n={len(best_seq)}, actual={best_val:.6f}", flush=True)

    # Phase 4: mirror polish from a smooth perturbation of the best state.
    remain = budget - (time.monotonic() - start_time)
    if remain > 260.0:
        rng = np.random.default_rng(999)
        base = normalize_nonneg(best_seq)
        pert = reheat_mixture(base, base, rng, 0.50, 0.34, 0.42)
        t = min(360.0, remain - 260.0)
        cand, _ = run_mirror_polish(pert, t, 404)
        best_seq, best_val, upd = better_candidate(best_seq, best_val, cand)
        if upd:
            init = normalize_nonneg(best_seq)
            print(f"After mirror-2: n={len(best_seq)}, actual={best_val:.6f}", flush=True)

    # Phase 5: another local round.
    remain = budget - (time.monotonic() - start_time)
    if remain > 160.0:
        t = min(230.0, remain - 190.0)
        cand, _ = run_local_pair_search(normalize_nonneg(best_seq), t, 505, 85, 85, 80)
        best_seq, best_val, upd = better_candidate(best_seq, best_val, cand)
        if upd:
            init = normalize_nonneg(best_seq)
            print(f"After local-2: n={len(best_seq)}, actual={best_val:.6f}", flush=True)

    # Phase 6: mean-direction search.
    remain = budget - (time.monotonic() - start_time)
    if remain > 140.0:
        t = min(200.0, remain - 160.0)
        cand, _ = run_mean_direction_search(normalize_nonneg(best_seq), t, 606, 90)
        best_seq, best_val, upd = better_candidate(best_seq, best_val, cand)
        if upd:
            init = normalize_nonneg(best_seq)
            print(f"After mean: n={len(best_seq)}, actual={best_val:.6f}", flush=True)

    # Phase 7: try a slightly longer sequence if enough time remains.
    remain = budget - (time.monotonic() - start_time)
    if remain > 320.0:
        n0 = int(len(best_seq))
        new_n = min(n0 + 1000, 7000)
        if new_n != n0 and new_n >= 1200:
            cand0 = interpolate_normalized(best_seq, new_n)
            best_seq, best_val, upd = better_candidate(best_seq, best_val, cand0)
            if upd:
                init = normalize_nonneg(best_seq)
                print(f"Interpolated longer: n={len(best_seq)}, actual={best_val:.6f}", flush=True)

            remain = budget - (time.monotonic() - start_time)
            if remain > 340.0:
                t = min(270.0, remain - 300.0)
                cand, _ = run_mirror_polish(init, t, 707)
                best_seq, best_val, upd = better_candidate(best_seq, best_val, cand)
                if upd:
                    init = normalize_nonneg(best_seq)

            remain = budget - (time.monotonic() - start_time)

                t = min(150.0, remain - 170.0)
                cand, _ = run_local_pair_search(normalize_nonneg(best_seq), t, 708, 60, 60, 70)
                best_seq, best_val, upd = better_candidate(best_seq, best_val, cand)
                if upd:
                    init = normalize_nonneg(best_seq)
                    print(f"After longer local: n={len(best_seq)}, actual={best_val:.6f}", flush=True)

    # Final validation and cleanup.
    final_val = actual_eval(best_seq)
    if not np.isfinite(final_val):
        arr = np.asarray(best_seq, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)

            arr = arr / s
        else:
            arr = default_initial(4200, 99)
        best_seq = arr
        final_val = actual_eval(best_seq)

    if not np.isfinite(final_val):
        best_seq = default_initial(4200, 99)
        final_val = actual_eval(best_seq)

    out = np.asarray(best_seq, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-12:
        out = default_initial(4200, 101)
    else:
        out = out / s

    print(f"Final: n={out.size}, actual={final_val:.6f}, elapsed={time.monotonic()-start_time:.1f}s", flush=True)
    return [float(x) for x in out.tolist()]
