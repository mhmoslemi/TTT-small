"""
 Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
 1. Multi-family parametric initialization
 2. Aggressive LP-based Frank-Wolfe with admissible step size and cutting planes
 3. Nesterov mirror descent with adaptive soft-subgradient
 4. Exact-line-search coordinate descent polish

import numpy as np
import time
import cvxpy as cp

_MODULE_SOLVER = None


def get_solver():
    global _MODULE_SOLVER
    if _MODULE_SOLVER is not None:
        return _MODULE_SOLVER
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _MODULE_SOLVER = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _MODULE_SOLVER = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _MODULE_SOLVER


def fast_autocorr(a):
    """Compute full convolution (autocorrelation) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    ac = fast_autocorr(a)
    return float(2.0 * len(a) * ac.max() / (s * s))


def norm_sum1(a):
    """Normalize so sum = 1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def interp_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def parametric_init(n, time_limit):
    """Search over parametric families."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = np.ones(n, dtype=np.float64) / n
    best_val = 2.0 * n * fast_autocorr(best_a).max()

    def try_local(a, label):
        nonlocal best_a, best_val
        a_n = norm_sum1(a)
        ac = fast_autocorr(a_n)
        val = 2.0 * n * ac.max()
        if val < best_val:
            best_val = val
            best_a = a_n.copy()
            if int(best_val * 1e6) % 100 == 0:
                print(f"    Param [{label}]: J={val:.6f}")

    # Raised cosine family
    for p in [0.5, 0.7, 1.0, 1.2, 1.5, 2.0, 3.0]:
        a = (0.5 * (1.0 + np.cos(2.0 * np.pi * x))) ** p
        a = np.maximum(a, 1e-10)
        try_local(a, f"cos{p}")

    # Polynomial taper
    for p in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
        a = (1.0 - (2.0 * x - 1.0) ** 2) ** p
        a = np.maximum(a, 1e-10)
        try_local(a, f"poly{p}")

    # Power law
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = (i_arr + 1.0) ** p
        try_local(a, f"pow{p}")
        a = (n - i_arr) ** p
        try_local(a, f"ipow{p}")

    # Beta distribution
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for beta in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a = np.maximum(i_arr + 1.0, 0.1) ** (alpha - 1) * np.maximum(n - i_arr, 0.1) ** (beta - 1)
            try_local(a, f"b{alpha}_{beta}")

    # Gaussian
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.15, 0.25, 0.35, 0.5]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            try_local(a, f"gc{c}_{sigma}")

    # Bimodal
    for c1 in [0.25, 0.35]:
        for c2 in [0.65, 0.75]:
            for sigma in [0.08, 0.12, 0.18]:
                a = np.exp(-((x - c1) ** 2) / (2 * sigma * sigma)) + np.exp(-((x - c2) ** 2) / (2 * sigma * sigma))
                try_local(a, f"bi{c1}_{c2}_{sigma}")

    # Sine squared
    for freq in [1, 2, 3]:
        for ph in [0.0, 0.25, 0.5]:
            a = np.maximum(np.sin(np.pi * freq * x + ph * np.pi) ** 2, 0.001)
            try_local(a, f"sn{freq}_{ph}")

    # Exponential
    for rate in [0.5, 1, 2, 5, 10]:
        try_local(np.exp(-rate * x), f"er{rate}")
        try_local(np.exp(rate * x), f"enr{rate}")

    elapsed = time.monotonic() - t0
    print(f"  Parametric done ({elapsed:.0f}s): best J={best_val:.6f}")
    return best_a, best_val


def lp_fw_optimize(a0, time_limit, solver, initial_K=500, max_K=5000, add_per_iter=50, verbose=True):
    """LP-based Frank-Wolfe with cutting planes and admissible step size."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_autocorr(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    # Initialize constraint set with top-K convolution positions
    init_idx = np.argsort(ac_y)[::-1][:min(initial_K, len(ac_y))]
    constraint_set = set(int(k) for k in init_idx)

    x = y.copy()  # Nesterov iterate
    iter_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-14:
            best_m = ac_x.max()
            best_a = x.copy()

        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix (sparse-friendly)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
                constraint_list = sorted(constraint_set)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints and add them
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Admissible step size: try several alphas
        alpha_base = 2.0 / (t + 2.0)
        best_alpha = alpha_base
        best_alpha_val = 2.0 * n * fast_autocorr((1 - alpha_base) * y + alpha_base * g_dir).max()

        # Try a few alternatives
        for trial_alpha in [alpha_base * 2.0, alpha_base * 0.5, alpha_base * 1.5]:
            trial_alpha = min(max(trial_alpha, 0.01), 1.0)
            trial = (1 - trial_alpha) * y + trial_alpha * g_dir
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts > 1e-15:
                trial = trial / ts
                tv = 2.0 * n * fast_autocorr(trial).max()
                if tv < best_alpha_val:
                    best_alpha_val = tv
                    best_alpha = trial_alpha

        # Update y with best alpha
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        # Nesterov update for x
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - x)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        x = x_new
        iter_count += 1

        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): bestJ={J:.6f}, |S|={len(constraint_set)}, M_cur={M_curr:.6f}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def mirror_descend_optimize(a0, time_limit, top_k=80, beta=4.0, c0=4.0, restart_every=5000, seed=42, verbose=False):
    """Nesterov-accelerated mirror descent with soft subgradient and adaptive restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        # Compute soft subgradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            # Restart with perturbation
            pert_scale = 0.03 * (1.0 + 0.5 * np.sin(np.pi * np.arange(n) / n))
            pert = rng.normal(0, 1.0, n) * pert_scale * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 30, 200)
            cur_beta = max(cur_beta - 0.8, 2.0)
            c0 = min(c0 * 1.1, 8.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({elapsed:.0f}s): J={J:.6f}, tk={cur_top_k}, b={cur_beta:.1f}")

    return best_a, best_m


def coord_polish(a0, time_limit, seed=77, verbose=True):
    """Coordinate descent with multi-scale mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac0 = fast_autocorr(a)
    best_m = ac0.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 600:
        ac = fast_autocorr(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score all positions by their contribution to top peaks
            scores = np.zeros(n)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[idx2] += a[idx2]
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            dst = candidates[np.argmin(cand_scores)]

            if dst == src:
                continue

            # Try multiple transfer fractions
            found = False
            fracs = [0.05, 0.10, 0.15, 0.25, 0.40, 0.60]
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 100:
                # Random exploration
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def local_subgradient_step(a, ac, k_peak, n_top=5):
    """Compute exact subgradient at top peak positions."""
    n = len(a)
    top_idx = np.argsort(ac)[::-1][:min(n_top, len(ac))]
    grad = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        for i_val, j_val in zip(idx, j_idx):
            if i_val != j_val:
                grad[i_val] += a[j_val]
            else:
                grad[i_val] += a[i_val]
    return grad


def fw_exact_subgrad(a0, time_limit, verbose=True):
    """Frank-Wolfe with exact subgradient (no LP, very fast iterations)."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    x = y.copy()
    iter_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        # Compute subgradient using top 3 peaks
        top_idx = np.argsort(ac)[::-1][:3]
        grad = np.zeros(n)
        for k in top_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            for i_val, j_val in zip(idx, j_idx):
                if i_val != j_val:
                    grad[i_val] += a[j_val] if False else y[j_val]  # BUG fix below
                else:
                    grad[i_val] += y[i_val]

        # Proper subgradient computation
        grad = np.zeros(n)
        for k in top_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            for i_val in range(i_start, i_end):
                j_val = k_int - i_val
                if 0 <= j_val < n:
                    if i_val != j_val:
                        grad[i_val] += 2.0 * y[j_val]
                    else:
                        grad[i_val] += y[i_val]

        # FW linear minimization: min grad^T z s.t. z in simplex
        # Solution: z = e_{argmin grad}
        min_idx = np.argmin(grad)
        e_min = np.zeros(n)
        e_min[min_idx] = 1.0

        # Also try away-step: move away from argmax grad coordinate
        max_idx = np.argmax(grad)
        if y[max_idx] > 0.01:
            # Away step
            alpha_away = y[max_idx] * 0.3
            y_away = y - alpha_away * (np.zeros(n))
            y_away[max_idx] -= alpha_away
            y_away[min_idx] += alpha_away
            y_away = np.maximum(y_away, 0.0)
            s_a = y_away.sum()
            if s_a > 1e-15:
                y_away = y_away / s_a
                M_a = fast_autocorr(y_away).max()
                if M_a < fast_autocorr((1 - 2.0/(t+2.0)) * y + (2.0/(t+2.0)) * e_min).max():
                    g_dir = y_away
                else:
                    g_dir = e_min
            else:
                g_dir = e_min
        else:
            g_dir = e_min

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        # Nesterov
        if t == 1:
            x = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - x)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx > 1e-15:
                x = x_new / sx
            else:
                x = y_new.copy()

        if fast_autocorr(x).max() < best_m - 1e-14:
            best_m = fast_autocorr(x).max()
            best_a = x.copy()

        iter_count += 1
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW-exact iter={t} ({elapsed:.0f}s): J={J:.6f}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    FW-exact done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_score(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = eval_score(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 3000
        x = np.linspace(0, 1, n_base)
        best_seq = (0.5 * (1 + np.cos(2 * np.pi * x))) ** 1.5
        best_seq = norm_sum1(best_seq)
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Parametric search ===
    print(f"\n=== Phase 1: Parametric (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 60.0)
    if t1 > 10:
        a_param, _ = parametric_init(n0, t1)
        check_update(a_param, "parametric")

    # === Phase 2: Main LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 60 and n_best <= 8000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_fw_optimize(a_in, t2, solver, initial_K=500, max_K=5000,
                                  add_per_iter=50, verbose=True)
        check_update(a_out, f"lp_main_n{n_best}")

    # === Phase 3: Mirror descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Mirror descent (n={n_best}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descend_optimize(a_in, t3, top_k=80, beta=4.0,
                                           c0=4.0, restart_every=4000, seed=42, verbose=False)
        check_update(a_out, f"md_n{n_best}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord descent (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_polish(a_in, t4, seed=77, verbose=True)
        check_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Scale to longer lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 2000]
    longer_lengths = [l for l in longer_lengths if l <= 8000]

    for new_n in longer_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n=== Phase 5: Scale to n={new_n} ===")
        a_long = interp_to_length(best_seq, new_n)

        # Quick LP-FW on longer sequence
        t5a = min(remaining * 0.30, 200.0)
        if t5a > 30 and new_n <= 7000:
            a_out, _ = lp_fw_optimize(a_long, t5a, solver, initial_K=400, max_K=3000,
                                      add_per_iter=40, verbose=False)
            check_update(a_out, f"lp_scale_n{new_n}")

        # Mirror descent on longer sequence
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            a_cur = norm_sum1(best_seq) if len(best_seq) == n_best else norm_sum1(a_long)
            a_long2 = interp_to_length(a_cur, new_n) if len(a_cur) != new_n else a_cur
            t5b = min(remaining * 0.40, 150.0)
            if t5b > 20:
                a_out, _ = mirror_descend_optimize(a_long2, t5b, top_k=60, beta=4.5,
                                                   c0=3.5, restart_every=3000, seed=123, verbose=False)
                check_update(a_out, f"md_scale_n{new_n}")

    # === Phase 6: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP-FW (n={n_best}) ===")
    if remaining > 60 and n_best <= 7000:
        t6 = min(remaining * 0.50, 300.0)
        if t6 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_fw_optimize(a_in, t6, solver, initial_K=800, max_K=6000,
                                      add_per_iter=60, verbose=True)
            check_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final coordination polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord (n={n_best}) ===")
    if remaining > 20:
        t7 = min(remaining * 0.60, 120.0)
        if t7 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_polish(a_in, t7, seed=314, verbose=True)
            check_update(a_out, f"coord_final_n{n_best}")

    # === Phase 8: Quick mirror descent final ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.70, 80.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descend_optimize(a_in, t8, top_k=100, beta=3.0,
                                               c0=5.0, restart_every=3000, seed=999, verbose=False)
            check_update(a_out, f"md_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 3000
        x = np.linspace(0, 1, n_fb)
        best_seq = (0.5 * (1 + np.cos(2 * np.pi * x))) ** 1.5
        best_seq = norm_sum1(best_seq)
        best_score = eval_score(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
