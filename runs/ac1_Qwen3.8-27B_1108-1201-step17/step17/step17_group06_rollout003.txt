def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def compute_h(q, a):
    """Compute h[i] = sum_{j=0}^{n-1} q[i+j] * a[j] for i=0,...,n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    size = 1 << (3 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a_rev, size)
    Q = np.fft.rfft(q, size)
    conv = np.fft.irfft(A * Q, size)
    return conv[n-1:n-1+n]

def softmax_and_obj(b, T):
    """Compute objective and gradient with respect to b (log-parametrization)."""
    # b -> a = softmax(b)
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    a = exp_b / exp_b.sum()
    n = len(a)
    
    # Compute convolution
    C = fast_conv(a)
    
    # Compute softmax of C / T
    C_scaled = C / T
    C_scaled -= C_scaled.max()
    exp_C = np.exp(C_scaled)
    sum_exp_C = exp_C.sum()
    q = exp_C / sum_exp_C
    M = T * (np.log(sum_exp_C) + C_scaled.max() - C_scaled.max())  # = T * log(sum(exp(C/T)))
    # Actually: M = T * log(sum(exp(C/T))) = T * (log(sum(exp(C_scaled))) + C_scaled.max())
    # Wait: exp(C/T) = exp(C_scaled + C_scaled_max/T * T/T) ... let me redo
    # C_scaled = C/T - max(C/T)
    # sum(exp(C/T)) = exp(max(C/T)) * sum(exp(C/T - max(C/T))) = exp(max(C/T)) * sum(exp(C_scaled))
    # M = T * log(sum(exp(C/T))) = T * (max(C/T) + log(sum(exp(C_scaled))))
    #   = T * (C.max()/T + log(sum(exp(C_scaled))))
    #   = C.max() + T * log(sum(exp(C_scaled)))
    
    M = C.max() + T * np.log(sum_exp_C)
    
    # Compute h[i] = sum_j q[i+j] * a[j]
    h = compute_h(q, a)
    
    # Compute S_q[i] = sum_{j=0}^{n-1} q[i+j]

    q_cumsum = np.concatenate([[0], np.cumsum(q_padded)])
    S_q = q_cumsum[n:n+n] - q_cumsum[:n]  # S_q[i] = sum_{j=0}^{n-1} q[i+j]
    
    # dM/da[i] = h[i] + a[i] * S_q[i]
    dM_da = h + a * S_q
    
    # f = 2*n*M (since sum(a) = 1)
    obj = 2.0 * n * M
    
    # df/db[i] = a[i] * (dM_da[i] * 2*n - sum_j a[j] * dM_da[j] * 2*n)
    #         = 2*n * a[i] * (dM_da[i] - dot(a, dM_da))
    dot_adM = np.dot(a, dM_da)
    grad_db = 2.0 * n * a * (dM_da - dot_adM)
    
    return obj, grad_db

def lfgs_optimize(a0, T_schedule, max_iter=200):
    """Run L-BFGS-B with temperature schedule."""
    n = len(a0)
    b = np.log(a0 + 1e-30)  # initial log parameters
    # Normalize b so that softmax(b) = a0
    # Actually, softmax is shift-invariant, so this is fine as long as a = softmax(b) ≈ a0
    # Let's verify:
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    a_check = exp_b / exp_b.sum()
    # a_check should be approximately a0 (up to normalization)
    # Since a0 is already normalized, and b = log(a0), softmax(log(a0)) = a0 / sum(a0) = a0
    # So this is exact!
    
    best_b = b.copy()
    best_obj = np.inf
    
    for T in T_schedule:
        result = minimize(
            fun=lambda x: softmax_and_obj(x, T)[0],
            x0=b,
            jac=lambda x: softmax_and_obj(x, T)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'gtol': 1e-8, 'ftol': 1e-12, 'iprint': 0}
        )
        b = result.x
        obj = softmax_and_obj(b, T)[0]
        if obj < best_obj:
            best_obj = obj
            best_b = b.copy()
    
    # Get final a
    b_shifted = best_b - best_b.max()
    exp_b = np.exp(b_shifted)
    a = exp_b / exp_b.sum()
    return a, best_obj

def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    C = fast_conv(a_norm)
    return 2.0 * n * C.max()
