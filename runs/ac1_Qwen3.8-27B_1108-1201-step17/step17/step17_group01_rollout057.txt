```python
def fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient(a, ac, n_active=None, tau=None, top_k=100):
    """Compute weighted subgradient of max autocorrelation.
    
    Uses soft weighting based on exp((c[k]-M)/tau) for peaks near the max.
    """
    n = len(a)
    M = ac.max()
    
    if tau is not None and tau > 0:
        # Soft weighting: exp((c-M)/tau)
        # Only consider peaks above M - 5*tau to avoid underflow
        threshold = M - 5 * tau
        active = np.where(ac > threshold)[0]
        w = np.exp((ac[active] - M) / tau)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(active)) / len(active)
        else:
            w = w / w_sum
    else:
        # Hard top-k
        k = min(top_k, 2*n-1)
        active = np.argsort(ac)[::-1][:k]
        top_vals = ac[active]
        w = top_vals ** 5  # sharp weighting
        w_sum = w.sum()
        w = w / w_sum
    
    # Compute subgradient
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        if wi < 1e-15:
            continue
        k_int = int(k)
        # Part 1: i such that k-i is in [0, n-1] → i in [k-n+1, k]
        i1s = max(0, k_int - n + 1)
        i1e = min(n - 1, k_int)

        # Part 2: i such that i-k is in [0, n-1] → i in [k, k+n-1]

    return grad
```

= exp(-2πi(N-1)k/N) * conj(A[k])

Hmm, this is getting complicated. Let me just use the direct formula:

g[i] = 2 * sum_j w[i+j] * a[j] (where i+j is in [0, 2n-2] and j is in [0, n-1])

This is the cross-correlation (w ⊛ a)[i] evaluated at i.

I can compute this using:
- Zero-pad w to length fft_size (>= (2n-1) + n - 1 = 3n-2)
- Zero-pad a to length fft_size
- Compute: ifft(FFT(w_padded) * conj(FFT(a_padded)))

Wait, let me just verify with the small example.
a = [x, y, z], w = [w0, w1, w2, w3, w4]
g[0] = 2 * (w[0]*a[0] + w[1]*a[1] + w[2]*a[2]) = 2*(w0*x + w1*y + w2*z)
g[1] = 2 * (w[1]*a[0] + w[2]*a[1] + w[3]*a[2]) = 2*(w1*x + w2*y + w3*z)
g[2] = 2 * (w[2]*a[0] + w[3]*a[1] + w[4]*a[2]) = 2*(w2*x + w3*y + w4*z)

This is indeed the cross-correlation: (w ⊛ a)[i] = sum_j w[i+j] * a[j].

Using FFT: (w ⊛ a) = ifft(FFT(w) * conj(FFT(a)))
Let's verify: FFT(w)[k] * conj(FFT(a))[k] = W[k] * conj(A[k])
ifft(W[k] * conj(A[k]))[i] = (1/N) sum_k W[k] * conj(A[k]) * exp(2πiki/N)

This is the cross-correlation. Yes, this is correct.

So the efficient computation is:
```python
def compute_subgradient_fft(a, w):
    n = len(a)
    w_len = len(w)  # 2n-1
    fft_size = 1 << (w_len + n - 1 - 1).bit_length()  # >= 3n-2
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    correl = np.fft.ifft(W * np.conj(A), fft_size).real[:n]
    return 2.0 * correl
```

This is O(n log n) instead of O(n * |active_set|)! Much faster.

But wait, I need to construct w (the weight vector of length 2n-1). If I'm using soft weighting:
w[k] = exp((c[k] - M) / tau) / Z for all k where c[k] > M - 5*tau

This w is dense (many non-zero entries), so the FFT approach is definitely better.

For the hard top-k approach, w is sparse (only k non-zero entries), so the direct O(n*k) approach might be faster for small k. But for k=100 and n=4200, FFT with size 16384 should still be faster than 420,000 multiply-adds.

Let me benchmark: FFT of size 16384 takes maybe 0.1ms. The direct approach with k=100 peaks and n=4200 positions: 100 * 4200 = 420,000 operations, which in numpy is maybe 0.5ms. So FFT wins by a factor of 5.

```python
def md_optimize(a0, time_limit, verbose=False):
    """Mirror descent with Nesterov acceleration and FFT-based subgradient."""
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    
    # Parameters
    c0 = 3.0  # step size constant
    tau0 = 0.01  # initial temperature for soft active set
    tau_decay = 0.999  # decay rate for tau
    restart_every = 5000
    stall_count = 0
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Soft active set
        tau = tau0 * (tau_decay ** (t // 1000))
        threshold = M - 5 * tau
        active_mask = ac > threshold
        w_full = np.zeros(len(ac))
        w_full[active_mask] = np.exp((ac[active_mask] - M) / tau)
        w_sum = w_full.sum()
        if w_sum < 1e-30:
            w_full = np.ones_like(ac) / len(ac)
        else:
            w_full = w_full / w_sum
        
        # Subgradient via FFT
        g = compute_subgradient_fft(y, w_full)
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart if stalled
        if stall_count > restart_every:
            stall_count = 0
            rng = np.random.default_rng(t)
            pert = rng.normal(0, 0.05, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
    
    return best_a, best_m
```

```python
def spectral_equalize(a0, time_limit, verbose=False):
    """Iterative spectral flattening with non-negativity projection."""
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Exclude DC (index 0) from flattening
        P_ac = P[1:]
        P_ac_mean = P_ac.mean()
        
        # Compress dynamic range
        # P_new = P_mean * (P/P_mean)^alpha for alpha < 1
        alpha = 0.5  # moderate compression
        P_new = P.copy()
        P_new[1:] = P_ac_mean * (P_ac / P_ac_mean) ** alpha
        P_new = np.maximum(P_new, 1e-30)
        
        # Reconstruct
        mag_new = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try a damped update
            a = 0.7 * a + 0.3 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_m
```

```python
def local_search(a0, time_limit, verbose=False):
    """Local coordinate descent: move mass between pairs of positions."""
    n = len(a0)
    a = a0 / a0.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        # Find positions that contribute most to the max autocorrelation
        ac = fast_autocorr(a)
        M = ac.max()
        max_k = np.argmax(ac)
        
        # The positions that contribute to ac[max_k] are pairs (i, max_k-i)
        i_start = max(0, int(max_k) - n + 1)
        i_end = min(n, int(max_k) + 1)
        
        # Find the pair with the largest product
        i_range = np.arange(i_start, i_end)
        products = a[i_range] * a[max_k - i_range]
        src = i_range[np.argmax(products)]
        
        # Try moving mass from src to a random other position
        dst = int(rng.integers(n))
        if dst == src:
            continue
        
        # Try different fractions
        found = False
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_try = a.copy()
            a_try[src] -= amt
            a_try[dst] += amt
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            
            M_try = fast_autocorr(a_try).max()
            if M_try < best_m:
                a = a_try
                best_m = M_try
                best_a = a.copy()
                found = True
                stall = 0
                break
        
        if not found:
            stall += 1
    
    return best_a, best_m
```
