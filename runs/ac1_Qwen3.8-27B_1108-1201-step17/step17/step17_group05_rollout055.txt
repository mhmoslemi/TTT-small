def fast_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """conv(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def compute_subgrad_focused(a, top_k=20, beta=15.0):
    """
    Compute subgradient focusing on top-k peaking lags with soft-max weighting.
    Uses FFT for efficiency.
    """
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals.max()
    w_top = np.exp(beta * shifted)
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        w_top = np.ones(k_max) / k_max
    else:
        w_top = w_top / w_sum
    w = np.zeros(len(ac))
    for wi, idx in zip(w_top, top_idx):
        w[int(idx)] = wi
    # grad[j] = 2 * sum_m w[m] * a[m-j] for valid indices
    # = 2 * convolve(a[::-1], w, mode='full')[n-1 : 2*n-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev_padded = np.zeros(fft_size)
    a_rev_padded[:n] = a[::-1]
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev_padded) * np.fft.rfft(w_padded), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, ac.max(), top_idx


def projected_gradient_simplex(d, a):
    """
    Project (a - d) onto the simplex {x >= 0, sum(x) = 1}.
    Uses bisection on the threshold.
    """
    v = a - d
    lo = v.min()
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        t = np.maximum(v - mid, 0.0).sum()
        if t > 1.0:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    result = np.maximum(v - mid, 0.0)
    s = result.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return result / s


def accelerated_subgrad_optimize(a0, time_limit, verbose=True, top_k=15,
                                 beta_init=20.0, beta_min=5.0,
                                 lr=0.02, decay=0.9995,
                                 restart_every=4000, seed=42):
    """
    Nesterov-accelerated projected subgradient with adaptive soft-max.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    lr_curr = lr
    stall = 0
    cur_beta = beta_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov extrapolation
        beta_n = (it - 1.0) / (it + 2.0)
        y = x + beta_n * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Subgradient
        g, M_curr, top_idx = compute_subgrad_focused(y, top_k=top_k, beta=cur_beta)

        # Normalized step
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm

        x_prev = x.copy()
        x = projected_gradient_simplex(lr_curr * g, x)

        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive beta: increase focus over time
        cur_beta = max(beta_min, beta_init - 0.005 * it)

        # Decay learning rate
        lr_curr *= decay

        # Restart if stalled
        if stall > restart_every:
            stall = 0
            # Perturb around best
            scale = max(0.02, 0.08 * (stall / (2 * restart_every) if stall > 0 else 0.5))
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            lr_curr = max(lr_curr, 0.01)
            # Random restart with some probability
            if rng.random() < 0.2:
                # Try a "bump" initialization
                xx = np.linspace(0, 1, n)
                x_r = np.exp(-50 * (xx - 0.3) ** 2) + 0.1 * np.ones(n)
                x_r = norm_sum1(x_r)
                x = x_r
                x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    ASG restart iter={it} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    ASG iter={it} ({elapsed:.0f}s): J={J:.6f}, stall={stall}, beta={cur_beta:.1f}")

    return best_a, best_m


def lp_frank_wolfe_optimize(a0, time_limit, solver, verbose=True,
                            n_constraints=None, max_iters=10000):
    """
    LP-based Frank-Wolfe with Nesterov acceleration.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select constraints
        if n_constraints is None:
            n_constraints = min(2 * n - 1, max(n, 500))
        K_sel = min(n_constraints, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K_sel]

        T = np.zeros((K_sel, n))
        for col_idx, m in enumerate(top_idx):
            m_int = int(m)
            j_start = max(0, m_int - n + 1)
            j_end = min(n - 1, m_int)
            js = np.arange(j_start, j_end + 1)
            T[col_idx, js] = y[m_int - js]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K_sel),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coord_descent_2d(a0, time_limit, verbose=True, seed=42):
    """
    2D coordinate descent: move mass between pairs of coordinates.
    Focuses on reducing the peak convolution coefficients.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500
    iterations = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        iterations += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Find top peak lags
        n_top = min(10, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]

        improvement_found = False

        # For each peak lag, identify contributing positions
        for lag_idx in range(min(5, n_top)):
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(top_lags[lag_idx])
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]

            if contrib.max() < 1e-20:
                continue

            # Source: position with largest contribution to this peak
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # For each potential destination, compute impact on all top lags
            # Score: how much would moving mass from src to dst reduce the max?
            best_dst = -1
            best_score = float('inf')
            best_frac = 0.0

            # Try a subset of destinations (for speed)
            n_try = min(50, n)
            if n <= 100:
                dst_candidates = list(range(n))
            else:
                # Focus on positions far from src and in "quiet" regions
                all_pos = np.arange(n)
                dist = np.abs(all_pos - src).astype(float)
                # Score by distance (prefer far) and low local autocorrelation
                peak_sens = np.zeros(n)
                for kl in top_lags[:5]:
                    kli = int(kl)
                    s2 = max(0, kli - n + 1)
                    e2 = min(n - 1, kli)
                    for d2 in range(s2, e2 + 1):
                        partner = kli - d2
                        if partner == src:
                            peak_sens[d2] += a[d2]
                        else:
                            peak_sens[d2] += a[partner] * 0.5
                scores = peak_sens / (1.0 + dist / n)
                dst_candidates = np.argsort(scores)[:n_try]

            for dst in dst_candidates:
                dst_int = int(dst) if hasattr(dst, '__int__') else dst
                if dst_int == src:
                    continue
                # Try a few fractions
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15 or amt >= src_val:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_score:
                        best_score = M_trial
                        best_dst = dst_int
                        best_frac = frac

            if best_dst >= 0 and best_score < best_m - 1e-14:
                amt = a[src] * best_frac
                a[src] -= amt
                a[best_dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = best_score
                best_a = a.copy()
                improvement_found = True
                stall = 0
                if verbose and iterations % 100 == 0:
                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print(f"    Coord2D iter={iterations} ({elapsed:.0f}s): J={J:.6f}")

        if not improvement_found:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

            if verbose and stall % 100 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Coord2D iter={iterations} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord2D final ({elapsed:.0f}s): J={J:.6f}, iters={iterations}")

    return best_a, best_m


def fine_tune_pair(a0, time_limit, verbose=True):
    """
    Finer-tuned pair optimization: for the top contributing pairs to peak lags,
    do a local 1D optimization of the transfer fraction.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Top few lags
        top_lags = np.argsort(ac)[::-1][:5]
        improved = False

        for k_int in [int(x) for x in top_lags[:3]]:
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            order = np.argsort(contrib)[::-1]

            for oi in order[:3]:
                src = idx[oi]
                src_val = a[src]
                if src_val < 1e-10:
                    continue

                # Try moving to several nearby and far positions
                for dst_offset in [-n//3, -n//4, -n//6, n//6, n//4, n//3, n//2, -n//2]:
                    dst = src + dst_offset
                    if dst < 0 or dst >= n or dst == src:
                        continue
                    # Binary search for optimal fraction
                    lo, hi = 0.0, min(0.5, a[src] * 0.9)
                    best_frac_local = 0.0
                    best_m_local = M_curr
                    for _ in range(15):
                        mid = (lo + hi) / 2.0
                        amt = a[src] * mid
                        a_t = a.copy()
                        a_t[src] -= amt
                        a_t[dst] += amt
                        a_t = np.maximum(a_t, 0.0)
                        a_t = a_t / a_t.sum()
                        m_t = fast_autocorr(a_t).max()
                        if m_t < best_m_local:
                            best_m_local = m_t
                            best_frac_local = mid
                            lo = mid
                        else:
                            hi = mid

                    if best_frac_local > 0 and best_m_local < best_m - 1e-14:
                        amt = a[src] * best_frac_local
                        a[src] -= amt
                        a[dst] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        best_m = fast_autocorr(a).max()
                        best_a = a.copy()
                        improved = True

        if not improved:
            break

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    FineTune ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def make_bump_init(n, center=0.3, width=0.15, seed=0):
    """Create a bump-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2) + 0.05 * np.ones(n)
    return norm_sum1(a)


def make_tapered_init(n, seed=0):
    """Create a tapered (triangular-ish) initialization."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 0.5 + 0.1 * np.ones(n)
    rng = np.random.default_rng(seed)
    a += rng.normal(0, 0.02, n)
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_two_bump_init(n, seed=0):
    """Two-bump initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.25) / 0.12) ** 2) + np.exp(-((x - 0.75) / 0.12) ** 2)
    a += 0.05 * np.ones(n)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase: subgradient, LP-FW, coordinate descent, fine-tune, multi-length.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_tapered_init(n_base, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Accelerated subgradient on parent ===
    print(f"\n=== Phase 1: Accelerated Subgradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = accelerated_subgrad_optimize(a_in, t1, verbose=False,
                                                 top_k=12, beta_init=25.0,
                                                 beta_min=8.0, lr=0.015,
                                                 decay=0.9997, restart_every=3000,
                                                 seed=42)
        try_update(a_out, f"asg_n{n0}")

    # === Phase 2: LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        n_cons = min(2 * n_best - 1, max(n_best, 800))
        a_out, _ = lp_frank_wolfe_optimize(a_in, t2, solver, verbose=True,
                                            n_constraints=n_cons)
        try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 3: Subgradient from different init ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Alt Subgradient (n={n0}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 30:
        a_init = make_bump_init(n0, center=0.35, width=0.12)
        a_out, _ = accelerated_subgrad_optimize(a_init, t3, verbose=False,
                                                 top_k=10, beta_init=15.0,
                                                 beta_min=6.0, lr=0.02,
                                                 decay=0.9995, restart_every=4000,
                                                 seed=123)
        try_update(a_out, f"asg_bump_n{n0}")

    # === Phase 4: 2D Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: 2D Coord (n={n_best}) ===")
    t4 = min(remaining * 0.10, 180.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_descent_2d(a_in, t4, verbose=True, seed=77)
        try_update(a_out, f"coord2d_n{n_best}")

    # === Phase 5: Fine-tune pairs ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: FineTune (n={n_best}) ===")
    t5 = min(remaining * 0.08, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = fine_tune_pair(a_in, t5, verbose=True)
        try_update(a_out, f"fintune_n{n_best}")

    # === Phase 6: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP-FW Round 2 (n={n_best}) ===")
    t6 = min(remaining * 0.15, 250.0)
    if t6 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        n_cons = min(2 * n_best - 1, max(n_best, 1000))
        a_out, _ = lp_frank_wolfe_optimize(a_in, t6, solver, verbose=True,
                                            n_constraints=n_cons)
        try_update(a_out, f"lpfw2_n{n_best}")

    # === Phase 7: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    # Try a few nearby lengths
    lengths_to_try = []
    for delta in [-200, 200, -400, 400]:
        nn = n_best + delta
        if 1500 <= nn <= 5000 and nn not in lengths_to_try:
            lengths_to_try.append(nn)

    for new_n in lengths_to_try[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"\n=== Phase 7: Length n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.20, 120.0)
        if t7a > 20:
            a_out, _ = accelerated_subgrad_optimize(a_long, t7a, verbose=False,
                                                     top_k=12, beta_init=20.0,
                                                     beta_min=7.0, lr=0.015,
                                                     decay=0.9997, restart_every=3000,
                                                     seed=555)
            try_update(a_out, f"asg_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and new_n <= 4000:
            t7b = min(remaining * 0.25, 100.0)
            if t7b > 20:
                n_cons = min(2 * new_n - 1, max(new_n, 600))
                a_out, _ = lp_frank_wolfe_optimize(a_long, t7b, solver, verbose=True,
                                                    n_constraints=n_cons)
                try_update(a_out, f"lpfw_n{new_n}")

    # === Phase 8: Final polish LP-FW ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP-FW (n={n_best}) ===")
    if remaining > 30 and n_best <= 4500:
        t8 = min(remaining * 0.30, 150.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            n_cons = min(2 * n_best - 1, max(n_best, 800))
            a_out, _ = lp_frank_wolfe_optimize(a_in, t8, solver, verbose=True,
                                                n_constraints=n_cons)
            try_update(a_out, f"lpfw_final_n{n_best}")

    # === Phase 9: Final coordinate + fine-tune ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final Coord+FineTune (n={n_best}) ===")
    if remaining > 20:
        t9a = min(remaining * 0.4, 100.0)
        if t9a > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_descent_2d(a_in, t9a, verbose=False, seed=999)
            try_update(a_out, f"coord_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t9b = min(remaining * 0.4, 60.0)
            if t9b > 5:
                a_in = norm_sum1(best_seq)
                a_out, _ = fine_tune_pair(a_in, t9b, verbose=False)
                try_update(a_out, f"fintune_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_tapered_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
