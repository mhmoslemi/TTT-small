    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = symmetrize(y)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = symmetrize(a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.3, 0.03 * (1.0 + 0.5 * restart_count))
            pert = rng.normal(0, strength, n)
            pert = symmetrize(pert)
            x_r = best_a + pert * best_a
            x_r = np.maximum(x_r, 1e-10)
            x_r = symmetrize(x_r)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 150)
            cur_beta = max(cur_beta - 0.4, 1.5)
            gamma = c0 * (1.0 + 0.15 * min(restart_count, 8))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart #{restart_count} t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 8000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def freq_domain_optimize(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(symmetrize(a0))
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    it = 0
    max_iters = 50000

    while time.monotonic() - t0 < time_limit and it < max_iters:
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        progress = min(1.0, it / max_iters)
        eta = 0.3 + 0.5 * (1.0 - progress)

        P_target = P_mean * (P / P_mean) ** (1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = symmetrize(a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            mix = 0.3 + 0.4 * progress
            a = (1.0 - mix) * a + mix * a_new
            a = symmetrize(a)
            a = np.maximum(a, 0.0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FreqDom it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def freq_capping_optimize(a0, time_limit, verbose=False):
    n = len(a0)
    a = norm_sum1(symmetrize(a0))
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    it = 0
    max_iters = 30000
    cap_factor = 1.5

    while time.monotonic() - t0 < time_limit and it < max_iters:
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        progress = min(1.0, it / max_iters)
        cap_level = P_mean * cap_factor * (1.0 + 0.5 * progress)

        P_new = np.minimum(P, cap_level)
        P_new = np.maximum(P_new, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = symmetrize(a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            it += 1
            continue
        a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.7 * a + 0.3 * a_new
            a = symmetrize(a)
            a = np.maximum(a, 0.0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FreqCap it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300):
    n = len(a0)
    a = norm_sum1(symmetrize(a0))
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    import cvxpy as cp

    y = a.copy()
    prev_y = a.copy()
    it = 0
    max_iters = 5000

    while time.monotonic() - t0 < time_limit and it < max_iters:
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = np.zeros((len(binding_idx), n))
        for row, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            T_sel[row, idx] = y[j_idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            it += 1
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            it += 1
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_val = symmetrize(b_val)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            it += 1
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = symmetrize(y_new)
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        prev_y = y
        y = y_new
        it += 1

        yy = symmetrize(y)
        my = fast_autocorr(yy).max()
        if my < best_m:
            best_m = my
            best_a = yy.copy()

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(symmetrize(a0))
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination using cost from multiple peaks
            costs = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                i_start = max(0, k2i - n + 1)
                i_end = min(n, k2i + 1)
                idx2 = np.arange(i_start, i_end)
                j2 = k2i - idx2
                costs[idx2] += a[j2]
                costs[j2] += a[idx2]

            n_candidates = max(10, n // 5)
            candidates = np.argsort(costs)[:n_candidates]
            # Prefer positions with low value and low cost
            scores = costs[candidates] + 10.0 * a[candidates]
            best_dst = candidates[np.argmin(scores)]

            if best_dst == src:
                best_dst = int(rng.integers(n))
                if best_dst == src:
                    continue

            found = False
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[n - 1 - src] -= amt
                a_trial[best_dst] += amt
                a_trial[n - 1 - best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = symmetrize(a_trial)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n // 2))
            j = int(rng.integers(n // 2))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[n - 1 - i] -= amt
                a_trial[j] += amt
                a_trial[n - 1 - j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = symmetrize(a_trial)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    M_new = fast_autocorr(a_trial).max()
                    if M_new < best_m:
                        a = a_trial
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordDescent ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_power_law_sym(n, power=1.0):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    d = np.abs(x - center) + 1.0
    a = d ** (-power)
    return norm_sum1(symmetrize(a))


def make_cosine_sym(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(symmetrize(a))


def make_sinc_like_sym(n, bandwidth=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bandwidth
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(symmetrize(a))


def make_multihump_sym(n, n_humps=3, spread=0.15):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_humps):
        center = (i + 1) / (n_humps + 1)
        a += np.exp(-((x - center) ** 2) / (2 * spread ** 2))
        a += np.exp(-((x - (1 - center)) ** 2) / (2 * spread ** 2))
    a += 0.01
    return norm_sum1(symmetrize(a))


def make_random_smooth_sym(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        if f % 2 == 0:
            coeff = rng.normal(0, 0.15 / np.sqrt(f))
            a += coeff * np.cos(np.pi * f * x)
        else:
            coeff = rng.normal(0, 0.15 / np.sqrt(f))
            phase = 0.0
            a += coeff * np.cos(np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(symmetrize(a))


def make_bell_sym(n, width=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) ** 2) / (2 * width ** 2))
    a += 0.01
    return norm_sum1(symmetrize(a))

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent (n={n_best}) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_in = norm_sum1(symmetrize(best_seq))
        a_out, _ = coordinate_descent(a_in, t5, verbose=True, rng_seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 6: Multi-length search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-length search ===")

    candidate_lengths = []
    for delta in [500, 1000, 2000, -500, -1000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 7000:
            candidate_lengths.append(new_n)
    candidate_lengths = list(dict.fromkeys(candidate_lengths))[:3]

    for new_n in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t6a = min(remaining * 0.35, 120.0)
        if t6a > 15:
            a_out, _ = amd_optimize(a_long, t6a, top_k=60, beta=4.0, c0=3.5,
                                     restart_every=5000, verbose=False)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t6b = min(remaining * 0.25, 60.0)
            if t6b > 10:
                a_out, _ = freq_domain_optimize(a_long, t6b, verbose=False)
                try_update(a_out, f"freqdom_n{new_n}")

    # === Phase 7: Final LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 5000:
        t7 = min(remaining * 0.25, 120.0)
        if t7 > 20:
            a_in = norm_sum1(symmetrize(best_seq))
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_K=400)
            try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final AMD + coordinate ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 30:
        t8a = min(remaining * 0.4, 80.0)
        if t8a > 10:
            a_in = norm_sum1(symmetrize(best_seq))
            a_out, _ = amd_optimize(a_in, t8a, top_k=120, beta=2.5, c0=2.5,
                                     restart_every=10000, verbose=False)
            try_update(a_out, f"amd_final_n{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t8b = min(remaining * 0.5, 60.0)
            if t8b > 10:
                a_in = norm_sum1(symmetrize(best_seq))
                a_out, _ = coordinate_descent(a_in, t8b, verbose=False, rng_seed=777)
                try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = make_cosine_sym(n_fb, mode=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
