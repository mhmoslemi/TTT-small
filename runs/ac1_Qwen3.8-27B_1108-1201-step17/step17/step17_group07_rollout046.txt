            g[idx] += 2.0 * y[k_int - idx]

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Choose pairs: one from "active" region, one from elsewhere
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            # Try several transfer amounts
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15]:
                amt = a[i] * frac
                if amt >= a[i]:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = (1.0 - (2.0 * x - 1.0) ** 4)
    elif kind == 'sin_sq':
        a = np.sin(np.pi * x) ** 2
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
