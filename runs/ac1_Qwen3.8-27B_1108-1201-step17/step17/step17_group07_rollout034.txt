            conv = np.fft.irfft(A * A, L)[:2*n-1]
            
            # Softmax of conv/mu
            cv = conv / mu
            cv -= cv.max()
            exp_cv = np.exp(cv)
            sw = exp_cv.sum()
            softmax = exp_cv / sw
            
            # Gradient via cross-correlation
            S = np.fft.rfft(softmax, L)
            cross = np.fft.irfft(S * np.conj(A), L)[:n]
            grad_a = 2.0 * cross
            
            # Natural gradient on simplex
            dot_ga_a = np.dot(grad_a, a)
            grad_x = a * (grad_a - dot_ga_a)

            gmax = np.abs(grad_x).max()
            if gmax > 1.0:
                grad_x = grad_x / gmax
            
            # Adam update
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_x
            v = beta2 * v + (1 - beta2) * grad_x * grad_x
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            x = x + lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Update a = softmax(x)
            x_shifted = x - x.max()
            a = np.exp(x_shifted)
            a_sum = a.sum()
            if a_sum < 1e-15:
                a = np.ones(n) / n
            else:
                a /= a_sum
            
            # Check objective periodically
            if step % check_interval == 0:
                actual_m = conv.max()
                if actual_m < best_m - 1e-15:
                    best_m = actual_m
                    best_a = a.copy()
                    if verbose and actual_m < best_m * 1.001:
                        J = 2.0 * n * best_m
                        print(f"      SA phase={phase} mu={mu:.6f} J={J:.10f}")
    
    # Final check
    final_conv = fast_conv(a)
    final_m = final_conv.max()
    if final_m < best_m:
        best_m = final_m
        best_a = a.copy()
    
    return best_a, best_m

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth Adam optimization with log-sum-exp (softmax) approximation and FFT-accelerated gradients
2. LP-based Remez with progressive active set and line search
3. Multi-length exploration
4. Coordinate descent polish
Key innovation: O(n log n) gradient computation via FFT cross-correlation,
enabling thousands of smooth gradient steps per second.
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_adam(a0, time_limit, verbose=False):
    """Smooth Adam optimization with log-sum-exp approximation."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    conv0 = fast_conv(a)
    M0 = conv0.max()
    
    # FFT size for cross-correlation
    L = 1 << (3 * n - 1).bit_length()
    
    # Parameterization: a = softmax(x)
    x = np.log(a + 1e-30).copy()
    
    # Adam
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_m = M0
    t0 = time.monotonic()
    
    # Schedule: 30 phases of decreasing mu
    n_phases = 30
    mu_start = M0 * 3.0
    mu_end = M0 / 300.0
    log_mus = np.linspace(np.log(mu_start), np.log(mu_end), n_phases)
    
    steps_per_phase = 20000
    check_every = 200
    
    for phase in range(n_phases):
        mu = np.exp(log_mus[phase])
        lr = 0.01 * (0.5 + 0.5 * mu / M0)
        
        for step in range(steps_per_phase):
            if time.monotonic() - t0 > time_limit:
                break
            
            # FFT of a
            A = np.fft.rfft(a, L)
            
            # Convolution
            conv = np.fft.irfft(A * A, L)[:2*n-1]
            
            # Softmax
            cv = conv / mu
            cv -= cv.max()
            exp_cv = np.exp(cv)
            sw = exp_cv.sum()
            if sw < 1e-30:
                break
            softmax_cv = exp_cv / sw
            
            # Gradient via cross-correlation: grad_a[i] = 2 * sum_m softmax[i+m] * a[m]
            S = np.fft.rfft(softmax_cv, L)
            cross = np.fft.irfft(S * np.conj(A), L)[:n]
            grad_a = 2.0 * cross
            
            # Natural gradient on simplex
            dot_ga_a = np.dot(grad_a, a)
            grad_x = a * (grad_a - dot_ga_a)
            
            # Gradient clipping
            gnorm = np.sqrt(np.sum(grad_x * grad_x))
            if gnorm > 1.0:
                grad_x = grad_x / gnorm
            
            # Adam
            t_adam += 1
            m_adam = beta1 * m_adam + (1 - beta1) * grad_x
            v_adam = beta2 * v_adam + (1 - beta2) * grad_x * grad_x
            m_hat = m_adam / (1 - beta1 ** t_adam)
            v_hat = v_adam / (1 - beta2 ** t_adam)
            x = x + lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Update a
            x_shifted = x - x.max()
            a_new = np.exp(x_shifted)
            a_sum = a_new.sum()
            if a_sum < 1e-15:
                a_new = a  # keep old
            else:
                a_new /= a_sum
                a = a_new
            
            # Periodic check
            if step % check_every == 0:
                cm = conv.max()
                if cm < best_m - 1e-14:
                    best_m = cm
                    best_a = a.copy()
                    if verbose and phase % 5 == 0:
                        J = 2.0 * n * best_m
                        elapsed = time.monotonic() - t0
                        print(f"      SA p={phase:2d} mu={mu:.2e} J={J:.10f} ({elapsed:.0f}s)")
    
    # Final evaluation
    final_conv = fast_conv(a)
    if final_conv.max() < best_m:
        best_m = final_conv.max()
        best_a = a.copy()
    
    return best_a, best_m

def remez_step(a, active_lags, solver=None):
    """Single Remez LP step."""
    n = len(a)
    b = fast_conv(a)
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) <= M_var + b[k_int])
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    return a_new / s

def remez_method(a0, time_limit, solver=None, verbose=False):
    """LP Remez with adaptive active set."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Select active lags: top 60 by value
        k_max = min(60, 2*n-1)
        active = np.argsort(b)[::-1][:k_max]
        
        # Solve LP
        a_cand = remez_step(a, active, solver=solver)
        if a_cand is None:
            continue
        
        # Line search
        accepted = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1 - alpha) * a + alpha * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial /= s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-14:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                break
        
        if not accepted:
            # Try smaller step or perturb
            rng = np.random.default_rng(it)
            a_pert = a * (1 + 0.01 * rng.normal(0, 1, n))
            a_pert = np.maximum(a_pert, 0)
            a_pert /= a_pert.sum()
            m_pert = fast_conv(a_pert).max()
            if m_pert < best_m:
                a = a_pert
                best_m = m_pert
                best_a = a.copy()
            else:
                # Stall: break
                if it > 20:
                    break
        
        if verbose and it % 10 == 0:
            J = 2.0 * n * best_m
            print(f"      RM it={it:4d} J={J:.10f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_m

def cd_polish(a0, time_limit, verbose=False):
    """Coordinate descent with random pair moves."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if stall > 100:
            break
        
        improved = False
        for _ in range(min(200, n)):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            
            for frac in [0.001, 0.005, 0.01, 0.05, 0.1]:
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial /= a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0
