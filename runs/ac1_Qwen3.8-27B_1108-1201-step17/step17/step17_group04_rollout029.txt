"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import cvxpy as cp

# --- Global solver setup ---
_global_solver = None

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:

        except Exception:

def fast_autocorr(a):
    """Compute full linear self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_max_ac(a):
    """Compute max of self-convolution."""
    ac = fast_autocorr(a)
    return ac.max(), ac

def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def smoothed_gradient(a, ac, T, top_K=None):
    """
    Compute smoothed gradient using log-sum-exp over top-K lags.
    Returns (gradient, M) where M = max(ac).
    """
    n = len(a)
    size = len(ac)
    if top_K is not None and top_K < size:
        top_idx = np.argpartition(ac, -top_K)[-top_K:]
        vals = ac[top_idx]
    else:
        top_idx = np.arange(size)
        vals = ac
    
    M = vals.max()
    # Log-sum-exp weights
    shifted = T * (vals - M)
    # Numerical stability
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-10:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        # Handle the case where i == k-i (diagonal term)
        contrib = 2.0 * a[j_idx]
        # For diagonal terms (i == k-i), the factor should be 1 not 2
        diag_mask = (idx == j_idx)
        contrib[diag_mask] = a[j_idx[diag_mask]]
        grad[idx] += wi * contrib
    
    return grad, M

def spectral_smooth(a, strength=0.3):
    """
    Apply spectral smoothing: reduce peaks in the self-convolution domain.
    Works by modifying the FFT to reduce concentration.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    A = np.fft.rfft(a, fft_size)
    # A*A gives the spectrum of the self-convolution
    B = A * A  # This is the DTFT of the linear convolution (zero-padded)
    
    # b = ifft(B) should give the self-convolution
    # We want to reduce the peak of b. In the frequency domain,
    # we can apply a soft "equalization" to B.
    
    b = np.fft.irfft(B, fft_size)
    M = b.max()
    
    # Create a target: reduce the peak while preserving total energy
    # Simple approach: clip b at (1-strength)*M and redistribute
    # But this is in the time domain of the convolution...
    
    # Alternative: modify |A| to make |A|^2 more uniform
    # |A|^2 proportional to B (up to real/imag issues)
    # Actually B = A * A (complex), not |A|^2
    
    # Let's try a different approach: 
    # Modify a directly by reducing its "peakedness" in a spectral sense
    # a_new = a - strength * (a - mean(a)) * (some spectral filter)
    
    # Simpler: apply a high-pass filter to a to reduce low-frequency concentration
    A_abs = np.abs(A)
    # Gentle equalization
    A_eq = A * (1.0 - strength * A_abs / (A_abs.max() + 1e-10))
    a_new = np.fft.irfft(A_eq, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def nesterov_smoothed(a0, time_limit, verbose=False, T_init=5.0, T_max=100.0,
                       top_K=30, c=3.0, restart_every=5000, nesterov=True,
                       seed=42, spectral_every=10000):
    """
    Nesterov mirror descent with log-sum-exp smoothed gradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_ac_max = fast_autocorr(a).max()
    best_J = 2.0 * n * best_ac_max
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_T = T_init
    it = 0
    
    # Lower bound for Polyak step (optimistic)
    J_lower = 1.50  # We know we can get below 1.51
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov extrapolation
        if nesterov and it > 1:
            beta_t = (it - 1.0) / (it + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s
        
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Adaptive temperature: increase over time
        cur_T = min(T_init + it / (5000.0), T_max)
        
        # Compute smoothed gradient
        g, M = smoothed_gradient(y, ac, cur_T, top_K=top_K)
        
        # Step size: Polyak-like
        gap = 2.0 * n * M - J_lower
        if gap < 0.01:
            gap = 0.01
        eta = c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 3.0 / (it + 1.0))
        eta = max(eta, 1e-10)
        
        # Mirror descent step (exponential)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Spectral smoothing periodically
        if it % spectral_every == 0:
            x_spectral = spectral_smooth(x, strength=0.1)
            J_spec = compute_J(x_spectral)
            if J_spec < best_J:
                x = x_spectral
                x_prev = x.copy()
                best_J = J_spec
                best_a = x.copy()
        
        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart with adaptation
        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.005 * min(it // restart_every, 5)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_T = T_init  # Reset temperature on restart
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Nesterov restart ({elapsed:.0f}s): J={best_J:.6f}, T={cur_T:.1f}")
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov t={it} ({elapsed:.0f}s): J={best_J:.6f}, T={cur_T:.1f}")
    
    return best_a, best_J

def peak_reduction_moves(a0, time_limit, verbose=False, seed=77):
    """

    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Get top peaks
        n_peaks = min(10, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]
        
        # Compute sensitivity
        sensitivity = np.zeros(n)
        for k in top_lags[:6]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contribution = a[idx] * a[j_idx]
            sensitivity[idx] += contribution
            sensitivity[j_idx] += contribution
        
        high_sens = np.where((sensitivity > sensitivity.mean() + 0.5 * sensitivity.std()) & (a > 0.001))[0]
        low_sens = np.where(sensitivity < sensitivity.mean())[0]
        
        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            continue
        
        improved = False
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            
            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst:
                continue
            
            src_val = a[src]
            if src_val < 1e-8:
                continue
            
            fracs = [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
            found = False
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if found:
                stall = 0
            else:
                if stall > 50:
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        J_new = compute_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
        
        if not improved:
            stall += 1
            if stall % 30 == 0:
                n_pert = max(3, n // 200)
                perturb_positions = rng.choice(n, size=n_pert, replace=False)
                for p in perturb_positions:
                    if a[p] > 1e-8:
                        frac = rng.uniform(0.01, 0.05)
                        amt = a[p] * frac
                        a[p] -= frac * a[p]
                        dst = int(rng.integers(n))
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=400):
    """LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build constraint matrix for top-K
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y
        
        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        
        T_sel = T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_symmetric_init(n):
    """Symmetric initialization (high at edges)."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)

def make_power_law_init(n, power=0.5):
    """Power law from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.001
    return norm_sum1(a)

def make_gaussian_edges(n, sigma=0.15):
    """Gaussian bumps at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)

def make_asymmetric_init(n, frac=0.6):
    """Asymmetric: more mass on one side."""
    x = np.linspace(0, 1, n)
    center = frac
    a = np.exp(-((x - center) / 0.3) ** 2) + 0.01
    return norm_sum1(a)

def fast_autocorr(a):
    """Compute full linear self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()

    """Compute objective and return (J, ac)."""
    n = len(a)

def smoothed_gradient(a, ac, T, top_K=None):
    """
    Compute smoothed gradient using softmax weights over top-K lags.
    Returns (gradient, M) where M = max(ac).
    """
    n = len(a)
    size = len(ac)
    if top_K is not None and top_K < size:
        top_idx = np.argpartition(ac, -top_K)[-top_K:]
        vals = ac[top_idx]
    else:
        top_idx = np.arange(size)
        vals = ac

    M = vals.max()
    shifted = T * (vals - M)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    w = exp_shifted / exp_shifted.sum()

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        contrib = 2.0 * a[j_idx]
        diag_mask = (idx == j_idx)
        if np.any(diag_mask):
            contrib[diag_mask] = a[j_idx[diag_mask]]
        grad[idx] += wi * contrib

    return grad, M

        # x^12 + x^6 + x^4 + x + 1 primitive polynomial
        bit = ((reg >> 11) ^ (reg >> 5) ^ (reg >> 3) ^ (reg >> 0)) & 1
        reg = ((reg >> 1) | (bit << (m - 1))) & ((1 << m) - 1)
    # Wrap/interpolate to desired length
    mseq_len = len(seq)
    if mseq_len >= n:
        # Take first n or interpolate
        x_old = np.linspace(0, 1, mseq_len)
        x_new = np.linspace(0, 1, n)
        seq_mapped = np.interp(x_new, x_old, seq)
    else:
        # Tile and truncate
        repeats = n // mseq_len + 1
        seq_tiled = np.tile(seq, repeats)[:n]
        seq_mapped = seq_tiled
    # Map to non-negative and smooth
    a = (seq_mapped + 1.0) / 2.0  # Map to [0, 1]
    # Apply smoothing to avoid discontinuities
    kernel_size = max(3, n // 500)
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.ones(kernel_size) / kernel_size
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0.01)
    return norm_sum1(a_smooth)

def nesterov_smoothed(a0, time_limit, verbose=False, T_init=5.0, T_max=100.0,
                       top_K=25, c=3.0, restart_every=4000, seed=42):
    """
    Nesterov mirror descent with log-sum-exp smoothed gradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    it = 0

    J_lower = 1.50

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov extrapolation
        if it > 1:
            beta_t = (it - 1.0) / (it + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()

        # Adaptive temperature
        cur_T = min(T_init + 0.01 * it, T_max)

        # Compute smoothed gradient
        g, M = smoothed_gradient(y, ac, cur_T, top_K=top_K)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break

        # Step size: Polyak-like with cap
        J_curr = 2.0 * n * M
        gap = J_curr - J_lower
        if gap < 0.01:
            gap = 0.01
        eta = c * gap / g_norm_sq
        eta = min(eta, 4.0 / (it + 1.0))
        eta = max(eta, 1e-10)

        # Mirror descent step (exponential)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with adaptation
        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.003 * min(it // restart_every, 10)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Nesterov restart ({elapsed:.0f}s): J={best_J:.6f}")

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov t={it} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J

def peak_reduction_moves(a0, time_limit, verbose=False, seed=77):
    """
    Targeted coordinate moves for peak reduction using sensitivity analysis.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        n_peaks = min(12, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # Compute sensitivity
        sensitivity = np.zeros(n)
        for k in top_lags[:8]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contribution = a[idx] * a[j_idx]
            sensitivity[idx] += contribution
            sensitivity[j_idx] += contribution

        sens_mean = sensitivity.mean()
        sens_std = sensitivity.std() + 1e-20

        high_sens = np.where((sensitivity > sens_mean + 0.5 * sens_std) & (a > 0.002))[0]
        low_sens = np.where(sensitivity < sens_mean - 0.3 * sens_std)[0]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            if stall > 50:
                # Random perturbation to escape
                n_pert = max(5, n // 150)
                for _ in range(n_pert):
                    p = int(rng.integers(n))
                    if a[p] > 1e-8:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[p] * frac
                        a[p] -= amt
                        dst = int(rng.integers(n))
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0
            continue

        improved = False
        for _ in range(40):
            if time.monotonic() - t0 > time_limit:
                break

            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            fracs = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.15, 0.25]
            found = False
            best_trial_J = best_J
            best_trial_a = None
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_trial_a = a_trial
            if best_trial_a is not None:
                a = best_trial_a
                best_J = best_trial_J
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=400):
    """LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 30000):
        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix for top-K
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_symmetric_init(n):
    """Symmetric initialization (high at edges)."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)

def make_power_law_init(n, power=0.5):
    """Power law from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.001
    return norm_sum1(a)

def make_gaussian_edges(n, sigma=0.15):
    """Gaussian bumps at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)

def make_asymmetric_init(n, frac=0.65):
    """Asymmetric: mass concentrated at one edge."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - frac) / 0.25) ** 2) + 0.5 * np.exp(-((x - (1 - frac)) / 0.4) ** 2) + 0.01
    return norm_sum1(a)
