"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth-max (log-sum-exp) mirror descent with adaptive temperature annealing
2. Top-K subgradient mirror descent with Nesterov acceleration  
3. Directed restarts based on convolution pressure analysis
4. Multi-length optimization with interpolation
5. Coordinate descent polish
"""

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution a*a using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_conv(a)
    return float(2.0 * n * conv.max() / (s * s))


def smooth_max_weights(c, T):
    """Compute softmax weights for smooth max approximation.
    w_k = exp(c_k / T) / sum(exp(c_j / T))
    """
    c_scaled = c / T
    c_scaled -= c_scaled.max()
    w = np.exp(c_scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform
        return np.ones_like(c) / len(c)
    return w / w_sum


def grad_smooth_max(a, T):
    """Compute gradient of T*log(sum(exp(c_k/T))) w.r.t. a.
    Uses FFT-based cross-correlation. O(n log n).
    """
    n = len(a)
    c = fast_conv(a)  # length 2n-1
    w = smooth_max_weights(c, T)  # length 2n-1
    
    # grad[i] = 2 * sum_j a[j] * w[j+i] for j=0..n-1, j+i < 2n-1
    # = 2 * (w * flip(a))[i + n - 1] (full convolution, element i+n-1)
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    A_flip = np.fft.rfft(np.flip(a), fft_size)
    conv_w = np.fft.irfft(W * A_flip, fft_size)[:3 * n - 2]
    grad = 2.0 * conv_w[n - 1:2 * n - 1]
    
    return grad


def grad_topk(a, top_k=80, beta_w=10.0):
    """Compute top-K subgradient of max(conv(a,a)) w.r.t. a.
    O(K * n) with vectorized numpy.
    """
    n = len(a)
    c = fast_conv(a)  # length 2n-1
    k_max = min(top_k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k_max]
    top_vals = np.maximum(c[top_idx], 1e-30)
    w = top_vals ** beta_w
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        # grad[i] += 2 * wi * a[k-i] for valid i
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            # j = k - i, so as i goes from i_start to i_end, j goes from k-i_start to k-i_end
            j_start = k_int - i_end
            j_end = k_int - i_start
            grad[i_start:i_end + 1] += 2.0 * wi * a[j_start:j_end + 1][::-1]
    
    return grad, c

    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()
        peak_lags = np.argsort(conv)[::-1][:20]
        
        # Compute pressure
        pressure = np.zeros(n)
        for k in peak_lags[:10]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                pressure[idx] += a[idx] * a[k_i - idx]
        
        # Find source
        valid = (pressure > 1e-20) & (a > 1e-15)
        if not valid.any():
            stall += 1
            break
        src_scores = np.where(valid, pressure * a, 0.0)
        src = int(np.argmax(src_scores))
        
        # Find destination
        dest_scores = np.zeros(n)
        for k in peak_lags[:5]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                dest_scores[idx] += a[k_i - idx]
        dest_scores[src] = np.inf
        dst = int(np.argmin(dest_scores))
        
        # Try transfers
        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-12:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random escape
            if stall % 20 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.01, 0.1)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break
    
    return best_a, best_J


def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=50.0, center1=0.05, center2=0.95):
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - center1) ** 2) + np.exp(-width * (x - center2) ** 2) + 0.01
    return a / a.sum()


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s
