**Even better idea: Use the KKT conditions**

At the optimum, for the problem $\min_a \max_k r_k(a)$ s.t. $a \geq 0, \sum a_i = 1$:
- There exists $M^* = \max_k r_k(a^*)$
- The set of "active" lags is $K^* = \{k : r_k(a^*) = M^*\}$
- There exist dual variables $\lambda_k \geq 0$ for $k \in K^*$, $\sum_{k \in K^*} \lambda_k = 1$
- KKT: $\sum_{k \in K^*} \lambda_k \nabla r_k(a^*)_i = \mu$ for all $i$ with $a^*_i > 0$
  where $\mu$ is the dual variable for the sum constraint.

This means that at the optimum, the "blended subgradient" (weighted average of subgradients at active lags) is constant across all positive coordinates.

1. Start with a feasible $a$.
2. Compute $r = a * a$, find active set $K^*$.
3. Solve for $\lambda$ (dual weights) such that $\sum_{k \in K^*} \lambda_k \nabla r_k$ is as "flat" as possible across positive coordinates.
4. Update $a$ to move toward the KKT solution.

Step 3 is interesting: we want $\sum_k \lambda_k \frac{\partial r_k}{\partial a_i} = \mu$ for all active $i$.

$\frac{\partial r_k}{\partial a_i} = 2 a_{k-i}$ (if $0 \leq k-i < n$ and $k \neq 2i$), or $2a_i$ (if $k = 2i$).

This is a system of linear equations in $\lambda$ (and $\mu$). If there are $m$ active lags and $p$ positive coordinates, we have $p$ equations and $m+1$ unknowns ($m$ $\lambda$'s and $\mu$). If $p > m+1$, the system is overdetermined, and we solve in least-squares sense.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS optimization with log-sum-exp smoothing in softmax parameterization (novel core)
2. KKT-informed Frank-Wolfe with dynamic active set
3. Exact coordinate descent with 1D line search
4. Multi-scale length exploration
Key innovation: smooth reparameterization enables use of quasi-Newton methods,
achieving much faster convergence than subgradient methods for this problem.

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize
from scipy.signal import fftconvolve

# --- Global state ---
_best_seq_global = None
_best_score_global = float('inf')


def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def logsumexp_objective_and_grad(x, beta):
    """
    Compute smoothed objective and gradient in softmax parameterization.
    L(x) = (1/beta) * log(sum_k exp(beta * r_k)) where r = softmax(x) * softmax(x)
    """
    a = softmax(x)
    n = len(a)
    r = fast_autocorr(a)
    r_max = np.max(r)
    shifted = beta * (r - r_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    value = r_max + log_sum / beta
    # Weights
    w = exp_shifted / exp_shifted.sum()
    # c_j = sum_k w_k * a_{k-j} (valid cross-correlation)
    c = np.correlate(w, a, mode='valid')
    r_bar = np.dot(w, r)
    grad = 2.0 * a * (c - r_bar)
    return value, grad


def lbfgs_smooth(a_init, time_limit, beta_start=5.0, beta_end=200.0,
                 n_restarts=3, verbose=True, seed=42):
    """
    L-BFGS optimization with log-sum-exp smoothing and annealing.
    Returns best a found.
    """
    n = len(a_init)
    best_a = norm_sum1(a_init)
    best_J = compute_J(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Initialize x from log of a_init
    a_safe = np.maximum(a_init, 1e-15)
    x0 = np.log(a_safe / a_safe.sum())

    # Anneal beta
    n_phases = max(n_restarts, 3)
    time_per_phase = time_limit / n_phases

    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit - 10:
            break

        # Linearly interpolate beta
        frac = phase / max(n_phases - 1, 1)
        beta = beta_start + (beta_end - beta_start) * frac

        t_phase_start = time.monotonic()
        t_available = min(time_per_phase * (1.0 + 0.5 * phase),
                          time_limit - (time.monotonic() - t0) - 10)
        if t_available < 5:
            break

        def obj_func(x):
            v, _ = logsumexp_objective_and_grad(x, beta)
            return v

        def grad_func(x):
            _, g = logsumexp_objective_and_grad(x, beta)
            return g

        # Initial x: for phase 0, use log(a_init); for later, use previous result
        if phase == 0:
            x_start = x0.copy()
        else:
            # Perturb slightly for diversity
            x_start = x0 + rng.normal(0, 0.1 * (1 - frac), n)

        maxiter = max(50, int(t_available * 50))  # rough estimate

        try:
            res = scipy_minimize(obj_func, x_start, jac=grad_func,
                                 method='L-BFGS-B',
                                 options={'maxiter': maxiter, 'gtol': 1e-8,
                                          'maxls': 50, 'iprint': 0})
            x_opt = res.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS phase {phase} error: {e}")
            continue

        a_opt = softmax(x_opt)
        a_opt = norm_sum1(a_opt)
        J_opt = compute_J(a_opt)

        if J_opt < best_J - 1e-10:
            best_J = J_opt
            best_a = a_opt.copy()
            x0 = x_opt.copy()

        # Also try the unsmoothed objective from this point
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS phase {phase} (beta={beta:.1f}, {elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J

    a_safe = np.maximum(a_init, 1e-15)
    x0 = np.log(a_safe / a_safe.sum())

    # Use very high beta for near-exact optimization
    beta = 500.0

    def obj_func(x):
        v, _ = logsumexp_objective_and_grad(x, beta)
        return v

    def grad_func(x):
        _, g = logsumexp_objective_and_grad(x, beta)
        return g

    try:
        res = scipy_minimize(obj_func, x0, jac=grad_func,
                             method='L-BFGS-B',
                             options={'maxiter': 200, 'gtol': 1e-10,
                                      'maxls': 50, 'iprint': 0})
        a_opt = softmax(res.x)
        a_opt = norm_sum1(a_opt)
        J_opt = compute_J(a_opt)
        if J_opt < best_J:
            best_a = a_opt
            best_J = J_opt
    except Exception:
        pass

    if verbose:
        print(f"    L-BFGS exact ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")
    return best_a, best_J


def kkt_frank_wolfe(a_init, time_limit, verbose=True, top_K=300, seed=99):
    """
    Frank-Wolfe with KKT-informed active set and adaptive step size.
    Uses the structure: at optimum, blended subgradient is constant on support.
    """
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    y = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        # Dynamic active set: lags within adaptive threshold of max
        # Use KKT-inspired: include lags that would be "binding"
        n_active = min(top_K, len(ac))
        sorted_idx = np.argsort(ac)[::-1]
        # Include all lags within 5% of max, up to top_K
        threshold = M_curr * 0.95
        active_idx = sorted_idx[ac[sorted_idx] >= threshold][:n_active]
        if len(active_idx) < 5:
            active_idx = sorted_idx[:5]

        # Build constraint matrix for active lags
        # (y * b)_k = sum_i y_i * b_{k-i} <= M_curr
        # This is linear in b when y is fixed
        # T[k, j] = y_{k-j} if 0 <= k-j < n, else 0
        K_act = len(active_idx)
        T = np.zeros((K_act, n))
        for ii, k in enumerate(active_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            for j in range(j_start, j_end):
                T[ii, j] = y[k_int - j]

        # Solve LP: max sum(b) s.t. T @ b <= M, b >= 0
        # Use scipy linprog
        from scipy.optimize import linprog
        c = -np.ones(n)  # minimize -sum(b)
        A_ub = T
        b_ub = M_curr * np.ones(K_act)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            continue

        if res.status != 0:
            continue

        b_val = np.maximum(res.x, 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Adaptive step size (using FW step or backtracking)
        alpha = 2.0 / (t + 2.0)

        # Try multiple step sizes and pick best
        best_alpha = alpha
        best_trial_M = M_curr
        for trial_alpha in [alpha, alpha * 0.5, alpha * 1.5, alpha * 2, 0.1, 0.05]:
            y_trial = (1 - trial_alpha) * y + trial_alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            sy = y_trial.sum()
            if sy < 1e-15:
                continue
            y_trial = y_trial / sy
            M_trial = fast_autocorr(y_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_alpha = trial_alpha

        y_new = (1 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        My = fast_autocorr(y_new).max()
        if My < best_M:
            best_M = My
            best_a = y_new.copy()

        y = y_new

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    KKT-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_M


def coordinate_descent(a_init, time_limit, verbose=True, seed=123):
    """
    Coordinate descent with exact 1D line search for mass transfers.
    Focuses on reducing the peak autocorrelation.
    """
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lag = int(np.argmax(ac))

        # Find top contributors to the peak
        i_start = max(0, peak_lag - n + 1)
        i_end = min(n, peak_lag + 1)
        idx = np.arange(i_start, i_end)
        j_idx = peak_lag - idx
        contributions = a[idx] * a[j_idx]

        # Sort by contribution (descending)
        order = np.argsort(contributions)[::-1]
        top_contributors = idx[order[:10]]

        # Find low-contribution positions (safe to add mass)
        total_contrib = np.zeros(n)
        for k_offset in range(-3, 4):
            k = peak_lag + k_offset
            if 0 <= k < len(ac):
                s = max(0, k - n + 1)
                e = min(n, k + 1)
                ii = np.arange(s, e)
                jj = k - ii
                total_contrib[ii] += a[ii] * a[jj]

        low_positions = np.where(total_contrib < np.percentile(total_contrib, 25))[0]

        improved = False
        n_attempts = 30

        for _ in range(n_attempts):
            if time.monotonic() - t0 > time_limit:
                break

            # Source: high contributor to peak
            if len(top_contributors) > 0 and rng.random() < 0.7:
                src = top_contributors[rng.integers(len(top_contributors))]
            else:
                src = int(rng.integers(n))

            # Destination: low contribution position
            if len(low_positions) > 0 and rng.random() < 0.8:
                dst = low_positions[rng.integers(len(low_positions))]
            else:
                dst = int(rng.integers(n))

            if src == dst or a[src] < 1e-10:
                continue

            # Binary search for optimal transfer amount
            lo, hi = 1e-8, a[src] * 0.9
            best_frac = None
            best_trial_J = best_J

            for _ in range(12):
                mid = (lo + hi) / 2
                amt = a[src] * mid
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_frac = mid
                    lo = mid
                else:
                    hi = mid

            if best_frac is not None and best_trial_J < best_J - 1e-12:
                amt = a[src] * best_frac
                a[src] -= amt
                a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_J = best_trial_J
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 30 == 0:
                n_pert = max(3, n // 200)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    if verbose:
        print(f"    CoordDescent final ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")
    return best_a, best_J


def make_u_shape_init(n, sharpness=2.0):
    """Create U-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = (2 * x - 1) ** 2 + 1.0 / sharpness
    return norm_sum1(a)


def make_bimodal_init(n, width=0.15):
    """Bimodal: peaks at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.02) / width) ** 2) + np.exp(-((x - 0.98) / width) ** 2) + 0.01
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def subgradient_polish(a_init, time_limit, verbose=False, seed=777):
    """
    Fast subgradient method with Nesterov acceleration for final polish.
    Uses exact peak targeting with adaptive step size.
    """
    n = len(a_init)
    a = norm_sum1(a_init)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        y = y / s

        ac = fast_autocorr(y)
        M = ac.max()
        k_peak = int(np.argmax(ac))

        # Exact subgradient at peak
        g = np.zeros(n)
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_peak - idx
        g[idx] = 2.0 * y[j_idx]

        # Polyak step size
        lower_bound = 1.0 / (2 * n)  # rough lower bound
        gap = M - lower_bound
        eta = 3.0 * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)

        # Mirror descent step
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = x.copy()

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining L-BFGS, Frank-Wolfe, and coordinate descent.
    """
    global _best_seq_global, _best_score_global

    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_u_shape_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: L-BFGS with log-sum-exp (primary method) ===
    print(f"\n=== Phase 1: L-BFGS Smooth (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 700.0)

    if t1 > 30:
        # Multiple initializations
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape_init(n0, sharpness=2.0))
        labels.append("u_shape")

        inits.append(make_bimodal_init(n0, width=0.12))
        labels.append("bimodal")

        # Blend
        if parent is not None:
            blend = 0.5 * norm_sum1(parent) + 0.5 * make_u_shape_init(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Deduplicate
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for u in unique_inits:
                if np.sum(np.abs(init - u)) < 0.03:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = t1 / n_inits

        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 30)
            if t_avail < 10:
                break
            print(f"  L-BFGS init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = lbfgs_smooth(init, t_avail,
                                     beta_start=5.0, beta_end=300.0,
                                     n_restarts=4, verbose=True,
                                     seed=42 + idx)
            try_update(a_out, f"lbfgs_{lbl}")

    # === Phase 2: L-BFGS exact finish ===
    print(f"\n=== Phase 2: L-BFGS Exact Finish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = lbfgs_exact_finish(a_in, t2, verbose=True)
        try_update(a_out, "lbfgs_exact")

    # === Phase 3: KKT Frank-Wolfe ===
    print(f"\n=== Phase 3: KKT Frank-Wolfe (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = kkt_frank_wolfe(a_in, t3, verbose=True, top_K=200)
        try_update(a_out, f"kkt_fw_n{n_best}")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t4, verbose=True, seed=456)
        try_update(a_out, f"coord_n{len(a_in)}")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: Subgradient Polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_polish(a_in, t5, verbose=True, seed=888)
        try_update(a_out, f"subgrad_n{len(a_in)}")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length Exploration (current n={n_best}) ===")

    # Try slightly longer
    longer_options = [n_best + 400, n_best + 800]
    longer_options = [l for l in longer_options if l <= 5000]

    for new_n in longer_options[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.3, 150.0)
        if t6a > 15:
            a_out, _ = lbfgs_smooth(a_long, t6a,
                                     beta_start=5.0, beta_end=200.0,
                                     n_restarts=3, verbose=False,
                                     seed=555 + new_n)
            try_update(a_out, f"lbfgs_n{new_n}")

    # Try slightly shorter
    shorter = n_best - 400
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 50 and shorter >= 1000:
        print(f"  Trying shorter n={shorter}")
        a_short = interpolate_sequence(best_seq, shorter)
        t6b = min(remaining * 0.3, 150.0)
        if t6b > 15:
            a_out, _ = lbfgs_smooth(a_short, t6b,
                                     beta_start=5.0, beta_end=200.0,
                                     n_restarts=3, verbose=False,
                                     seed=777)
            try_update(a_out, f"lbfgs_n{shorter}")

    # === Phase 7: Final L-BFGS on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final L-BFGS (n={len(best_seq)}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.4, 200.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = lbfgs_smooth(a_in, t7,
                                 beta_start=50.0, beta_end=500.0,
                                 n_restarts=3, verbose=True,
                                 seed=9999)
        try_update(a_out, "lbfgs_final")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.4, 100.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t8, verbose=False, seed=2024)
        try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_u_shape_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    _best_seq_global = best_seq
    _best_score_global = best_score
    return [float(x) for x in best_seq]
