"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_initialization(n, seed=0, method='sinc'):

    x = np.linspace(0, 1, n)
    if method == 'sinc':
        bw = 5.0 + (seed % 5) * 2.0
        t = (x - 0.5) * bw
        a = np.sinc(t) ** 2
    elif method == 'cosine':
        mode = 1 + seed % 3
        a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    elif method == 'gaussian':
        center = 0.3 + 0.4 * (seed % 3) / 2.0
        width = 0.1 + 0.1 * seed
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif method == 'bimodal':
        a = np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02)
    elif method == 'plateau':
        a = (x > 0.2) & (x < 0.8)
        a = a.astype(float) + 0.01
    else:
        rng = np.random.default_rng(seed)
        a = np.ones(n) * 0.5
        for f in range(1, 8):
            a += rng.normal(0, 0.1/np.sqrt(f)) * np.sin(2*np.pi*f*x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def spectral_precondition(a, n_iters=200, alpha=0.7, verbose=False):
    """Flatten the spectrum via iterative projection (soft Gerchberg-Saxton)."""
    n = len(a)
    a = norm_sum1(a.copy())
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    fft_size = 1 << (2*n - 1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        # Soft flattening
        P_flat = P_mean * (P / P_mean) ** (1.0 - alpha)
        P_flat = np.maximum(P_flat, 1e-30)
        phase = np.angle(A)
        A_new = np.sqrt(P_flat) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        M = fast_autocorr(a_new).max()
        if M < best_m:
            best_m = M
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def compute_subgradient(a, ac, top_k=50, temperature=1.0):
    """Compute smoothed subgradient of max(ac) with respect to a."""
    n = len(a)
    k_max = min(top_k, len(ac))
    
    # Softmax weights
    ac_top = ac[:k_max]  # or all ac
    # Actually use all ac but weight by softmax
    scaled = ac / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback to top-k
        top_idx = np.argsort(ac)[::-1][:k_max]
        w = np.zeros(len(ac))
        w[top_idx] = 1.0 / k_max
    else:
        w = w / w_sum
    
    # Compute gradient: g[i] = sum_k w[k] * d(ac[k])/da[i]
    # d(ac[k])/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n, k!=2i) + a[i] (if k=2i)
    # Using FFT for efficiency:
    # g[i] = sum_k w[k] * a[k-i] + sum_k w[k] * a[i-k] (with corrections)
    
    # First term: sum_k w[k] * a[k-i] for valid k-i
    # This is a cross-correlation of w and a
    # w has length 2n-1, a has length n
    # For a given i: sum_{k: 0<=k-i<n} w[k] * a[k-i] = sum_{j: 0<=j<n} w[i+j] * a[j]
    # = (w * a)[i] where * is cross-correlation
    
    # Let's just do it directly for the top contributing lags
    # For efficiency, only use lags with significant weight
    
    threshold = w.max() * 1e-4
    active = np.where(w > threshold)[0]
    
    g = np.zeros(n)
    for k in active:
        wk = w[k]
        if wk < 1e-12:
            continue
        # d(ac[k])/da[i] for i = 0..n-1
        # = a[k-i] if 0 <= k-i < n, plus a[i-k] if 0 <= i-k < n (and i != k/2)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j = k - idx  # = k - i
        g[idx] += wk * a[j]
        # For the "other direction": a[i-k] where 0 <= i-k < n
        # i-k >= 0 → i >= k, i-k < n → i < k+n
        i2_start = max(k, 0)
        i2_end = min(n, k + n)
        if i2_start < i2_end:
            idx2 = np.arange(i2_start, i2_end)
            j2 = idx2 - k  # = i - k
            # Avoid double-counting at i = k/2
            mask = idx2 != (k - idx2)  # i != k-i, i.e., 2i != k
            g[idx2[mask]] += wk * a[j2[mask]]
    
    return g

def compute_subgradient_fast(a, ac, top_k=80, beta=3.0):
    """Faster subgradient using only top-k peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        g[idx] += 2.0 * wi * a[j]  # Factor of 2 accounts for both directions
    
    return g

        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart strategy
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            strength = min(0.15, 0.02 + 0.02 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            # Adapt parameters
            cur_top_k = min(cur_top_k + 15, 150)
            cur_beta = max(cur_beta - 0.5, 2.0)
            gamma = c0 * (1.0 + 0.15 * min(restart_count, 5))
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart #{restart_count} t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}, beta={cur_beta:.1f}")
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, max_passes=50):
    """Coordinate descent: for each coordinate, find optimal value via line search."""
    n = len(a0)
    a = norm_sum1(a0.copy())
    best_a = a.copy()
    best_J = 2.0 * n * fast_autocorr(a).max() / (a.sum()**2)
    t0 = time.monotonic()
    
    # Precompute: for coordinate i, the range of k values that are affected
    # conv[k] involves a[i] when 0 <= k-i < n, i.e., k in [i, i+n-1]
    
    for p in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        S = a.sum()
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            ai = a[i]
            if ai < 1e-15:
                continue
            
            k_lo = i
            k_hi = min(i + n - 1, 2*n - 2)
            
            # Background max (k values not affected by a[i])
            if k_lo > 0:
                bg_lo = ac[:k_lo].max()
            else:
                bg_lo = 0.0
            if k_hi < 2*n - 2:
                bg_hi = ac[k_hi+1:].max()
            else:
                bg_hi = 0.0
            B = max(bg_lo, bg_hi)
            
            # Coefficients D[k] for k in [k_lo, k_hi]
            # conv[k] = sum_j a[j]*a[k-j], a[i] contributes when j=i or k-j=i
            # j=i: need 0<=k-i<n → k in [i, i+n-1] ✓ (that's our range)
            # k-j=i: j=k-i, need 0<=k-i<n → same range
            # D[k] = a[k-i] (from j=i term) + a[k-i] (from k-j=i term, if k-i≠i)
            # If k-i = i (k=2i): D[k] = 2*a[i]
            # Otherwise: D[k] = 2*a[k-i]
            
            ks = np.arange(k_lo, k_hi + 1)
            js = ks - i  # = k - i, range [0, n-1]
            D = 2.0 * a[js]
            # Correction for k = 2i: D should be 2*a[i], not 2*a[k-i]
            # But a[k-i] = a[2i-i] = a[i], so D = 2*a[i] is already correct!
            
            # C[k] = ac[k] - ai * D[k] for k in range
            C = ac[k_lo:k_hi+1] - ai * D
            
            # Try candidate values for a[i]: 0, ai/2, ai, 2*ai
            S0 = S - ai
            candidates = [0.0, ai*0.25, ai*0.5, ai, ai*1.5, ai*2.0]
            
            best_x = ai
            best_f = float('inf')
            
            for x in candidates:
                if x < 0:
                    continue
                new_max = max(B, (C + x * D).max())
                new_S = S0 + x
                if new_S < 1e-15:
                    continue
                new_J = 2.0 * n * new_max / (new_S ** 2)
                if new_J < best_f:
                    best_f = new_J
                    best_x = x
            
            if best_x != ai:
                a[i] = best_x
                improved = True
        
        # Check overall improvement
        ac_new = fast_autocorr(a)
        S_new = a.sum()
        if S_new < 1e-15:
            a = best_a.copy()
            break
        J_new = 2.0 * n * ac_new.max() / (S_new ** 2)
        if J_new < best_J:
            best_J = J_new
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
