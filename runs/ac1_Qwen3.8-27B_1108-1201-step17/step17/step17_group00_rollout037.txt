def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    return np.fft.irfft(A * A, n=fft_size)[:size]


def _norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _softmax_weights(ac, beta):
    ac = np.asarray(ac, dtype=np.float64)
    shifted = ac - ac.max()
    exp_vals = np.exp(beta * shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        return np.ones_like(ac) / len(ac)
    return exp_vals / total


def _compute_gradient_soft(a, ac, beta, top_k=200):

    else:

    rng = np.random.default_rng(99)

    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(a)
        M = ac.max()

        active_thresh = M - max(1e-4 * M, 1e-8)
        active = np.where(ac >= active_thresh)[0]
        if len(active) < 2:
            active = np.argsort(ac)[::-1][:max(2, len(ac) // 4)]

        if len(active) == 0:
            break

        k_max = int(active[0])
        k_min = int(active[-1])

        j_start = max(0, k_max - n + 1)
        j_end = min(n - 1, k_max)
        if j_start > j_end:
            j_start, j_end = 0, n - 1

        idx = np.arange(j_start, j_end + 1)
        contrib = a[idx] * a[k_max - idx]
        if contrib.max() < 1e-20:
            continue
        src = int(idx[np.argmax(contrib)])

        impact = np.zeros(n)
        for k2 in active[:30]:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            if s2 <= e2:
                idx2 = np.arange(s2, e2 + 1)
                impact[idx2] += np.abs(a[k2i - idx2])

        impact[src] = np.inf
        in_peak_zone = np.zeros(n, dtype=bool)
        for k2 in active[:10]:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n - 1, k2i)
            in_peak_zone[s2:e2 + 1] = True

        candidates = np.where(~in_peak_zone)[0]
        if len(candidates) == 0:
            candidates = np.arange(n)
        cand_impact = impact[candidates]
        dst = int(candidates[np.argmin(cand_impact)])

        if dst == src or a[src] < 1e-12:
            if it % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.005)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    val_new = 2.0 * n * _fast_autocorr(a).max()
                    if val_new < best_val:
                        best_val = val_new
                        best_a = a.copy()
            continue

        improved = False
        for frac in [0.001, 0.002, 0.004, 0.008, 0.015, 0.03]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            val_trial = 2.0 * n * _fast_autocorr(a_trial).max()
            if val_trial < best_val - 1e-12:
                a = a_trial
                best_val = val_trial
                best_a = a.copy()
                improved = True
                break

        if not improved and it % 200 == 0:
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.0005, 0.003)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    val_new = 2.0 * n * _fast_autocorr(a).max()
                    if val_new < best_val:
                        best_val = val_new
                        best_a = a.copy()

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Balance iter={it} ({elapsed:.0f}s): J={best_val:.6f}, |active|={len(active)}")

    return best_a, best_val


def _lp_solve_direction(coeffs, a_curr, solver):
    n = coeffs.shape[1]
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    s = g_val.sum()
    if s < 1e-15:
        return None
    return g_val / s
