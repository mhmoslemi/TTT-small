"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Peak-focused subgradient with Nesterov acceleration
2. Coordinate descent with golden-section line search
3. Block-level optimization
4. Multiple restarts with diverse initializations
5. Length adaptation
Key differences from previous: peak-focused subgradient (not soft-max over all),
golden-section line search, block moves, lopsided Gaussian inits.

import numpy as np
import time

def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def peak_subgradient(a, top_k=5, alpha=50.0):
    """
    Compute subgradient focusing on top_k peaking lags.
    Returns gradient vector and the max value.
    """
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    
    # Get top k lags by value
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    
    # Soft weights over top k
    shifted = top_vals - top_vals.max()
    w = np.exp(alpha * shifted)
    w = w / w.sum()
    
    # Gradient: 2 * sum_m w[m] * a[m-j] for valid m-j
    # For each lag m, the gradient contribution at position j is 2*w[m]*a[m-j]
    # We can compute this via convolution
    total_len = n + k_max  # approximate
    # Actually, let's just do it directly for the top lags
    grad = np.zeros(n)
    for wi, m in zip(w, top_idx):
        m_i = int(m)
        # grad[j] += 2 * wi * a[m_i - j] for valid j (0 <= m_i - j < n)
        j_start = max(0, m_i - n + 1)
        j_end = min(n - 1, m_i)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            grad[js] += 2.0 * wi * a[m_i - js]
    
    return grad, M

def nesterov_subgradient(a0, time_limit, verbose=True, seed=42,
                          top_k=5, alpha=50.0, c0=3.0):
    """
    Nesterov-accelerated subgradient method on the simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    y = a.copy()
    x = a.copy()
    stall = 0
    max_stall = 3000
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        if stall > max_stall:
            # Restart with perturbation
            pert = rng.normal(0, 0.02, n) * best_a
            y = best_a + pert
            y = np.maximum(y, 1e-10)
            y = y / y.sum()
            x = y.copy()
            stall = 0
            if verbose:
                print(f"    NS restart at t={t}, J={2*n*best_m:.6f}")
        
        # Compute subgradient at y
        g, M_curr = peak_subgradient(y, top_k=top_k, alpha=alpha)
        
        # Step size
        eta = c0 / np.sqrt(t)
        
        # Mirror descent step (exponential)
        log_y = np.log(np.maximum(y, 1e-30))
        log_new = log_y - eta * g / M_curr  # normalize gradient by M
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s
        
        # Nesterov update
        beta_t = (t - 1.0) / (t + 1.0)
        x_new = a_new + beta_t * (a_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = a_new.copy()
        else:
            x_new = x_new / sx
        
        y = a_new
        x = x_new
        
        # Track best
        M_y = fast_autocorr(y).max()
        if M_y < best_m:
            best_m = M_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        M_x = fast_autocorr(x).max()
        if M_x < best_m:
            best_m = M_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and t % 500 == 0:
            print(f"    NS t={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, stall={stall}")
    
    return best_a, best_m

def coord_desc_line_search(a0, time_limit, verbose=True, seed=42,
                           max_epochs=50, pairs_per_epoch=200):
    """
    Coordinate descent with golden-section line search.
    For each promising (i,j) pair, find optimal transfer amount.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for epoch in range(max_epochs):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find peak lags
        top_lags = np.argsort(ac)[::-1][:10]
        
        # Generate candidate (i, j) pairs
        pairs = []
        for k in top_lags[:5]:
            k_i = int(k)
            # Positions contributing to this lag
            js = max(0, k_i - n + 1)
            je = min(n - 1, k_i)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_i - idx]
            # Most contributing positions are candidate sources
            top_sources = idx[np.argsort(contrib)[::-1][:5]]
            
            # Candidate destinations: positions with low "sensitivity"
            # (positions that don't contribute much to peak lags)
            sens = np.zeros(n)
            for k2 in top_lags:
                k2_i = int(k2)
                s2 = max(0, k2_i - n + 1)
                e2 = min(n - 1, k2_i)
                for d in range(s2, e2 + 1):
                    partner = k2_i - d
                    sens[d] += a[partner]
            
            low_sens = np.where(sens < np.percentile(sens, 25))[0]
            
            for si in top_sources:
                # Pick a few random destinations from low sensitivity
                if len(low_sens) > 0:
                    dests = rng.choice(low_sens, size=min(3, len(low_sens)), replace=False)
                    for dj in dests:
                        if dj != si:
                            pairs.append((si, dj))
        
        # Also add some random pairs
        for _ in range(20):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                pairs.append((i, j))
        
        # Deduplicate
        pairs = list(set(pairs))
        rng.shuffle(pairs)
        
        improvements = 0
        for i, j in pairs[:pairs_per_epoch]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            
            # Golden section search for optimal t (transfer from i to j)
            # t in [0, a[i]]
            t_lo = 0.0
            t_hi = a[i] * 0.95  # don't transfer too much
            
            def obj_at_t(t):
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    return 1e10
                a_trial = a_trial / s
                return fast_autocorr(a_trial).max()
            
            # Golden section
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            c = t_hi - gr * (t_hi - t_lo)
            d = t_lo + gr * (t_hi - t_lo)
            fc = obj_at_t(c)
            fd = obj_at_t(d)
            
            for _ in range(15):
                if fc < fd:
                    t_hi = d
                    d = c
                    fd = fc
                    c = t_hi - gr * (t_hi - t_lo)
                    fc = obj_at_t(c)
                else:
                    t_lo = c
                    c = d
                    fc = fd
                    d = t_lo + gr * (t_hi - t_lo)
                    fd = obj_at_t(d)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = obj_at_t(t_opt)
            
            if f_opt < best_m - 1e-14:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_m = f_opt
                best_a = a.copy()
                improvements += 1
        
        if verbose and epoch % 5 == 0:
            print(f"    CD epoch={epoch} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}, imp={improvements}")
        
        if improvements == 0:
            # Stuck - try random perturbation
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    t = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= t
                    a[j] += t
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        break
    
    return best_a, best_m

def block_optimize(a0, time_limit, verbose=True, seed=42, block_size=16):

    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    n_blocks = (n + block_size - 1) // block_size
    block_sums = np.array([a[i*block_size:(i+1)*block_size].sum() for i in range(n_blocks)])
    
    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find blocks with highest contribution to peak
        peak_lags = np.argsort(ac)[::-1][:5]

                # Lag k involves a[s:e] with a[k-s:k-e]

                    q = k_i - p
                    if 0 <= q < n:
                        block_contrib[bi] += a[p] * a[q]

        dst_block = int(np.argmin(block_contrib))
        
        if src_block == dst_block:
            continue
        
        # Within source block, find the position with highest value

        s_end = min(n, s_start + block_size)
        d_start = dst_block * block_size
        d_end = min(n, d_start + block_size)
        
        src_pos = s_start + int(np.argmax(a[s_start:s_end]))
        # Choose destination position: one with low local sensitivity
        dst_pos = d_start + int(np.argmin(a[d_start:d_end]))
        
        if a[src_pos] < 1e-12:
            continue
        
        # Try a few transfer amounts
        for frac in [0.02, 0.01, 0.005, 0.05, 0.03]:
            t = a[src_pos] * frac
            a_trial = a.copy()
            a_trial[src_pos] -= t
            a_trial[dst_pos] += t

def make_lopsided_gaussian(n, center_frac=0.3, width_frac=0.15, seed=0):
    """Create a lopsided Gaussian initialization."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    center = center_frac * n
    width = width_frac * n
    a = np.exp(-(x - center)**2 / (2 * width**2))
    # Add small noise
    a += rng.normal(0, 0.01, n) * a
    a = np.maximum(a, 0)
    return norm_sum1(a)

def make_smooth_asymmetric(n, seed=0):
    """Create a smooth asymmetric initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.3 * np.sin(np.pi * x) + 0.2 * np.cos(2 * np.pi * x)
    for f in [1, 2, 3, 4]:
        a += rng.normal(0, 0.05) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2*np.pi))
    a = np.maximum(a, 0)
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss
