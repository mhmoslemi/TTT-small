                                            'factr': 1e10})
            theta_new = result.x
            a_new = softmax_from_logits(theta_new)
            m_new = fast_conv(a_new).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a_new.copy()
            theta = theta_new
        except Exception as e:
            if verbose:
                print(f"    L-BFGS alpha={alpha:.0f} failed: {e}")
            continue
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    L-BFGS alpha={alpha:.0f} ({elapsed:.1f}s): J={J:.8f}")
    
    return best_a, best_m

def smooth_mirror_descent(a0, time_limit, alpha_init=30.0, alpha_max=300.0,
                          c0=3.0, verbose=False):
    """Nesterov mirror descent with smooth maximum subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 2000
    rng = np.random.default_rng(42 + n)
    c = c0
    alpha = alpha_init
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        ac = fast_conv(y)
        
        f_val, weights = smooth_max_value_and_weights(ac, alpha)
        g = compute_smoothed_subgradient(y, weights)
        
        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        alpha = min(alpha_max, alpha_init * (1.0 + 0.0005 * t))
        
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(300, int(max_stall * 0.7))
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.008, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n//4, 400), replace=False)
                x_r[idx] *= rng.uniform(0.6, 1.4, len(idx))
            elif strategy == 2:
                x_r = best_a * (1 + rng.normal(0, 0.003, n))
            else:
                # Block perturbation
                block_size = max(1, n // 20)
                b_start = int(rng.integers(0, n - block_size))
                block = best_a[b_start:b_start+block_size].copy()
                block *= rng.uniform(0.5, 1.5, block_size)
                x_r = best_a.copy()
                x_r[b_start:b_start+block_size] = block
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            alpha = alpha_init
            c = max(0.3, c * 0.75)
        
        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothMD t={t} ({elapsed:.1f}s): J={J:.8f}, alpha={alpha:.1f}")
    
    return best_a, best_m

def lp_remez_step(a, trust_r=0.05, max_cons=50, solver=None):
    """Single Remez LP step."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    threshold = max_b * 0.998
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1, a_var <= a * (1.0 + trust_r)]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy(), max_b
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), max_b
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), max_b
    return a_new / s, float(M_var.value) if M_var.value is not None else max_b

def remez_loop(a0, time_limit, solver=None, r_init=0.06, verbose=False):
    """Remez loop with adaptive trust region and line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(123 + n)
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand, _ = lp_remez_step(a, trust_r=r, max_cons=60, solver=solver)
        
        accepted = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_trial = (1 - frac) * a + frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.2, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.8)
            if no_improve > 15:
                idx = rng.choice(n, size=min(n//5, 500), replace=False)
                a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = max(0.02, r)
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=777):
    """Coordinate descent with adaptive step sizes focusing on critical coordinates."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 100)
    
    # Identify critical coordinates (those with high values)
    critical_threshold = np.sort(a)[::-1][min(n//4, 500)]
    critical_mask = a >= critical_threshold
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_trials = min(200, n // 2)
        
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                    break
            # Choose source: biased toward critical coordinates
            if rng.random() < 0.7:
                idx_crit = np.where(critical_mask)[0]
                if len(idx_crit) > 0:
                    i = int(rng.choice(idx_crit))
                else:
                    i = int(rng.integers(n))
            else:
                i = int(rng.integers(n))
            # Choose target: uniform
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15]:
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    # Update critical mask occasionally
                    if rng.random() < 0.1:
                        critical_threshold = np.sort(a)[::-1][min(n//4, 500)]
                        critical_mask = a >= critical_threshold
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > max_stall // 2:
                # Perturb to escape
                idx = rng.choice(n, size=min(n//10, 200), replace=False)
                a[idx] *= rng.uniform(0.8, 1.2, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                stall = 0
    
    return best_a, best_m

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'bimodal':
        a = np.exp(-20*(x-0.3)**2) + np.exp(-20*(x-0.7)**2)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    n_base = 2600
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            parent_loaded = True
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Smooth Mirror Descent from parent (25% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 1: Smooth Mirror Descent n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = smooth_mirror_descent(norm_sum1(best_seq), t1, 
                                         alpha_init=30.0, alpha_max=200.0,
                                         c0=3.0, verbose=True)
        try_update(a_out, f"SmoothMD_parent_n{n0}")
    
    # === Phase 2: L-BFGS with homotopy (30% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 2: L-BFGS Homotopy n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 60:
        a_out, _ = l_bfgs_smooth(norm_sum1(best_seq), t2,
                                  alpha_start=30.0, alpha_end=400.0,
                                  n_alphas=5, verbose=True)
        try_update(a_out, f"LBFGS_n{len(best_seq)}")
    
    # === Phase 3: Remez LP (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: Remez LP n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = remez_loop(norm_sum1(best_seq), t3, solver=solver,
                              r_init=0.05, verbose=True)
        try_update(a_out, f"Remez_n{len(best_seq)}")
    
    # === Phase 4: Coordinate Descent Polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: CD Polish n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t4,
                                             verbose=False, rng_seed=555)
        try_update(a_out, f"CD_n{len(best_seq)}")
    
    # === Phase 5: L-BFGS with higher alpha (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: L-BFGS High Alpha n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = l_bfgs_smooth(norm_sum1(best_seq), t5,
                                  alpha_start=100.0, alpha_end=800.0,
                                  n_alphas=3, verbose=True)
        try_update(a_out, f"LBFGS_high_n{len(best_seq)}")
    
    # === Phase 6: Multi-init exploration (if time remains) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        print(f"\n=== Phase 6: Multi-init exploration (budget={remaining:.0f}s) ===")
        inits = ['raised_cos', 'bimodal', 'gaussian']
        t_per = remaining / (len(inits) + 1)
        for kind in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            a_init = make_init(n0, kind)
            # Run smooth MD then L-BFGS
            t_md = t_per * 0.4
            t_lbfgs = t_per * 0.6
            if t_md > 15:
                a_mid, _ = smooth_mirror_descent(a_init, t_md, alpha_init=20.0,
                                                 alpha_max=100.0, c0=2.5, verbose=False)
            else:
                a_mid = a_init
            if t_lbfgs > 15 and time.monotonic() - t_global < time_budget - 10:
                a_out, _ = l_bfgs_smooth(a_mid, t_lbfgs, alpha_start=50.0,
                                         alpha_end=300.0, n_alphas=3, verbose=False)
                try_update(a_out, f"MultiInit_{kind}_n{n0}")
    
    # === Phase 7: Final L-BFGS + Remez + CD cascade ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final Cascade n={n_best} (budget={remaining:.0f}s) ===")
        
        # L-BFGS with very high alpha
        t7a = min(remaining * 0.35, 100.0)
        if t7a > 15:
            a_out, _ = l_bfgs_smooth(norm_sum1(best_seq), t7a,
                                      alpha_start=200.0, alpha_end=1000.0,
                                      n_alphas=2, verbose=True)
            try_update(a_out, f"FinalLBFGS_{n_best}")
        
        # Remez
        remaining = time_budget - (time.monotonic() - t_global)
        t7b = min(remaining * 0.35, 80.0)
        if t7b > 15:
            a_out, _ = remez_loop(norm_sum1(best_seq), t7b, solver=solver,
                                  r_init=0.02, verbose=False)
            try_update(a_out, f"FinalRemez_{n_best}")
        
        # CD
        remaining = time_budget - (time.monotonic() - t_global)
        t7c = min(remaining * 0.30, 60.0)
        if t7c > 10:
            a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t7c,
                                                 verbose=False, rng_seed=31337)
            try_update(a_out, f"FinalCD_{n_best}")
    
    # === Phase 8: Try slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        for n_alt in [n_best + 100, n_best - 100]:
            if n_alt < 100:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 40:
                break
            print(f"\n=== Alt length n={n_alt} (budget={remaining:.0f}s) ===")
            a_alt = interpolate_sequence(best_seq, n_alt)
            t_alt = remaining * 0.7
            if t_alt > 20:
                a_out, _ = l_bfgs_smooth(a_alt, t_alt, alpha_start=50.0,
                                         alpha_end=300.0, n_alphas=3, verbose=False)
                try_update(a_out, f"AltLen_{n_alt}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
