def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

    best_m = fast_autocorr(a).max()

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if adaptive_beta and stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.9, 2.0)
            cur_top_k = min(cur_top_k + 5, len(ac))

            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 1.0, 2.0)

    return best_a, best_m


def solve_lp_direction(y, M_curr, binding_idx, solver):
    n = len(y)
    T_rows = []
    for k in binding_idx:
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        idx = np.arange(i_start, i_end)
        coeffs = y[ki - idx]
        T_rows.append((idx, coeffs))

    b_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(b_var) >= 0.01]
    for idx, coeffs in T_rows:
        constraints.append(cp.sum(b_var[idx] * coeffs) <= M_curr)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, use_all_constraints=False,
                   top_K=300, line_search=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    alpha_decay = 0.0

    K_sel = 2 * n - 1 if use_all_constraints else min(top_K, 2 * n - 1)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        if use_all_constraints or K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            conv_vals = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                ki = int(k)
                i_start = max(0, ki - n + 1)
                i_end = min(n, ki + 1)
                idx = np.arange(i_start, i_end)
                conv_vals[k] = np.dot(y[idx], y[ki - idx])
            margins = M_curr - conv_vals
            binding_idx = np.argsort(margins)[:K_sel]

        g_dir = solve_lp_direction(y, M_curr, binding_idx, solver)
        if g_dir is None:
            g_dir = np.ones(n) / n

        if line_search:
            trial_alphas = [2.0 / (t + 2.0), 1.0 / (t + 1.0), 0.1, 0.05, 0.02]
            best_alpha = 2.0 / (t + 2.0)
            best_trial_m = fast_autocorr(y).max()
            for alpha in trial_alphas:
                y_trial = (1.0 - alpha) * y + alpha * g_dir
                y_trial = np.maximum(y_trial, 0.0)
                sy = y_trial.sum()
                if sy < 1e-15:
                    continue
                y_trial = y_trial / sy
                m_trial = fast_autocorr(y_trial).max()
                if m_trial < best_trial_m:
                    best_trial_m = m_trial
                    best_alpha = alpha
            alpha = best_alpha
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if not use_all_constraints and t % 10 == 0:
            K_sel = min(K_sel + 30, 2 * n - 1)

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def peak_surgery(a0, time_limit, verbose=True, rng_seed=77, n_top_peaks=20):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top_peaks]

        sens = np.zeros(n)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            sens[idx] += a[partner]

        improved_this_round = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            low_sens = np.argsort(sens)[:n // 4]
            dst = low_sens[np.argmin([0 if d == src else 1 for d in low_sens])]
            if dst == src:
                dst = low_sens[np.argmin([0 if d == src else 1 for d in low_sens[1:]])] + 1 if len(low_sens) > 1 else (src + 1) % n

            found = False
            for frac in [0.01, 0.005, 0.02, 0.03, 0.05, 0.002, 0.08]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_round = True
                    found = True
                    break

            if not found and stall > 50:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True

        if not improved_this_round:
            stall += 1
            n_pert = min(10, n)
            pert = rng.normal(0, 0.01, n) * a
            a_new = a + pert
            a_new = np.maximum(a_new, 0.0)
            if a_new.sum() > 1e-15:
                a_new = a_new / a_new.sum()
                M_new = fast_autocorr(a_new).max()
                if M_new < best_m:
                    a = a_new
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakSurgery ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakSurgery final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_ramp(n, power=2.0):
    x = np.linspace(0, 1, n)
    a = x ** power + 0.01
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_asymmetric_u(n, asym=0.3):
    x = np.linspace(0, 1, n)
    center = 0.5 + asym * 0.1
    a = 1.0 + 2.0 * ((x - center) / 0.5) ** 2
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_u_shape(n_base, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-init AMD (short) ===
    print(f"\n=== Phase 1: Multi-init AMD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.10, 150.0)

    if t_phase1 > 40:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, 3.0))
        labels.append("u3.0")
        inits.append(make_u_shape(n0, 5.0))
        labels.append("u5.0")
        inits.append(make_asymmetric_u(n0, 0.2))
        labels.append("asym_u")

        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  AMD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, top_k=80, beta=7.0, c0=5.0,
                                     restart_every=3000, adaptive_beta=True,
                                     verbose=False, rng_seed=42 + idx * 7)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: LP-FW (main optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2500)
        top_K = min(5000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                   use_all_constraints=use_all, top_K=top_K,
                                   line_search=True)
        try_update(a_out, "lp_main")

    # === Phase 3: Peak surgery ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak surgery (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_surgery(a_in, t3, verbose=True, rng_seed=77, n_top_peaks=25)
        try_update(a_out, "peak_surgery")

    # === Phase 4: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW round 2 (n={n_best}) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        use_all = (n_best <= 2500)
        top_K = min(6000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True,
                                   use_all_constraints=use_all, top_K=top_K,
                                   line_search=True)
        try_update(a_out, "lp_r2")

    # === Phase 5: AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: AMD polish (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t5, top_k=100, beta=8.0, c0=4.0,
                                 restart_every=6000, adaptive_beta=True, verbose=False, rng_seed=500)
        try_update(a_out, "amd_polish")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")

    trial_lengths = []
    for delta in [500, -500, 1000, -1000, 1500]:
        new_n = n_best + delta
        if 2000 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 60]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.30, 150.0)
        if t6a > 20:
            a_out, _ = amd_optimize(a_long, t6a, top_k=60, beta=6.0, c0=4.0,
                                     restart_every=4000, adaptive_beta=True, verbose=False, rng_seed=new_n % 999)
            try_update(a_out, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 3000:
            t6b = min(remaining * 0.40, 120.0)
            if t6b > 15:
                a_in = norm_sum1(best_seq)
                use_all = (new_n <= 2000)
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, verbose=False,
                                           use_all_constraints=use_all, top_K=min(3000, 2 * new_n - 1),
                                           line_search=True)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final LP-FW + Peak surgery ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final optimization (n={n_best}) ===")

    if remaining > 30:
        t7a = min(remaining * 0.5, 200.0)
        if t7a > 20:
            a_in = norm_sum1(best_seq)
            use_all = (n_best <= 2500)
            top_K = min(8000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t7a, solver, verbose=True,
                                       use_all_constraints=use_all, top_K=top_K,
                                       line_search=True)
            try_update(a_out, "lp_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7b = min(remaining * 0.6, 100.0)
        if t7b > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_surgery(a_in, t7b, verbose=False, rng_seed=999, n_top_peaks=30)
            try_update(a_out, "peak_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
