def fast_autocorr(a):
    """Compute full convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Use the standard simplex projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_subgradient(a, k):
    """Compute 2*T_k * a, the subgradient of a^T T_k a at point a.
    (T_k * a)[i] = a[k-i] if valid, else 0."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    idx = np.arange(i_start, i_end)
    partner = k - idx
    grad[idx] = 2.0 * a[partner]
    return grad


def soft_subgrad_multi(a, ac, top_k=30, beta=4.0):
    """Soft subgradient: weighted sum of subgradients at top peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        partner = k_int - idx
        grad[idx] += 2.0 * wi * a[partner]
    return grad


def nesterov_subgrad(a0, time_limit, top_k=40, beta=5.0, c0=4.0,
                     proj_type='simplex', verbose=False, rng_seed=42):
    """Nesterov-accelerated subgradient method with simplex projection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    gamma = c0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = soft_subgrad_multi(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)

        if proj_type == 'simplex':
            a_new = proj_simplex(y - eta * g, 1.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
        else:
            # Mirror descent (exponential)
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 5000:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 100)
            cur_beta = max(cur_beta - 1.0, 2.0)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Subgrad t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def qp_frank_wolfe(a0, time_limit, solver, top_K=10, verbose=True, rng_seed=42):
    """QP-based Frank-Wolfe: minimize b^T T_k b + (1/eta)||b-a||^2 at active lags."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 50

    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select active lags
        k_max = min(top_K, 2 * n - 1)
        active_idx = np.argsort(ac)[::-1][:k_max]
        active_vals = np.maximum(ac[active_idx], 1e-30)
        # Weights: emphasize the most active lags
        w = active_vals ** 3.0
        w = w / w.sum()

        # Build combined Hessian: sum_j w_j * T_{k_j} + (1/eta) * I
        eta = max(0.1, 0.5 / (1 + 0.1 * t))
        inv_eta = 1.0 / eta

        # Build the combined quadratic coefficient vector
        # The objective is: sum_j w_j * (b^T T_{k_j} b) + inv_eta * ||b - a||^2
        # = b^T (sum_j w_j T_{k_j} + inv_eta * I) b - 2*inv_eta * a^T b + const
        #
        # For cvxpy, we need to express this as a QP.
        # Since n can be large, let's build the QP efficiently.

        # The cross-term: sum_j w_j * sum_i b[i]*b[k_j - i]
        # We can build this as a quadratic form with a sparse matrix.
        # For efficiency, let's use the quadratic_over_lin trick or build the matrix.

        if n <= 2000:
            # Build the full matrix for small n
            Q = inv_eta * np.eye(n)
            for wi, k in zip(w, active_idx):
                ki = int(k)
                i_start = max(0, ki - n + 1)
                i_end = min(n, ki + 1)
                idx = np.arange(i_start, i_end)
                partner = ki - idx
                Q[idx, partner] += wi

            b_var = cp.Variable(n, nonneg=True)
            constraints = [cp.sum(b_var) == 1]
            objective = cp.quad_form(b_var, Q, assume_PSD=True) - 2.0 * inv_eta * a @ b_var
            prob = cp.Problem(cp.Minimize(objective), constraints)

            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue

            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        else:
            # For large n, use a simplified approach:
            # Only use the single most active lag
            k_star = int(active_idx[0])
            b_var = cp.Variable(n, nonneg=True)
            constraints = [cp.sum(b_var) == 1]

            # b^T T_k b = sum_i b[i] * b[k-i]
            # Build as sum of products
            i_start = max(0, k_star - n + 1)
            i_end = min(n, k_star + 1)
            idx = np.arange(i_start, i_end)
            partner = k_star - idx

            # This is a quadratic form: b[idx]^T * I * b[partner]
            # In cvxpy: cp.quad_over_lin doesn't help here
            # Use: sum_i b[i]*b[partner_i] = b[idx] @ b[partner]
            cross_term = b_var[idx] @ b_var[partner]

            objective = cross_term + inv_eta * cp.sum_squares(b_var - a)
            prob = cp.Problem(cp.Minimize(objective), constraints)

            try:
                prob.solve(solver=solver, verbose=False)
            except Exception:
                continue

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue

            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        b_val = b_val / b_sum

        # Frank-Wolfe update with Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        if sa < 1e-15:
            continue
        a_new = a_new / sa

        # Nesterov: also try the accelerated point
        x = a_new.copy()
        if t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            a_accel = x + beta_t * (x - a)
            a_accel = np.maximum(a_accel, 0.0)
            sa2 = a_accel.sum()
            if sa2 > 1e-15:
                a_accel = a_accel / sa2
                M_acc = fast_autocorr(a_accel).max()
                if M_acc < best_m:
                    best_m = M_acc
                    best_a = a_accel.copy()
                    a = a_accel
                else:
                    a = a_new
            else:
                a = a_new
        else:
            a = a_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            # Perturb and reset
            pert = rng.normal(0, 0.01, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            stall = 0
            top_K = min(top_K + 5, 30)

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    QP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=None, use_all=False):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    if use_all:
        K_sel = 2 * n - 1
    elif top_K is not None:
        K_sel = min(top_K, 2 * n - 1)
    else:
        K_sel = min(300, 2 * n - 1)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix: T_sel[j, i] = y[k_j - i] if valid
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        if not use_all and K_sel < 2 * n - 1:
            # Select most-violated constraints (those closest to M)
            constraint_vals = ac.copy()
            margins = M_curr - constraint_vals
            binding_idx = np.argsort(margins)[:K_sel]
        else:
            binding_idx = np.arange(2 * n - 1)

        # Build T matrix for selected constraints
        T_sel = np.zeros((len(binding_idx), n))
        for row, k in enumerate(binding_idx):
            ki = int(k)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            idx = np.arange(i_start, i_end)
            T_sel[row, idx] = y[ki - idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        for candidate in [y, x]:
            M_cand = fast_autocorr(candidate).max()
            if M_cand < best_m:
                best_m = M_cand
                best_a = candidate.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

        if not use_all and t % 15 == 0:
            K_sel = min(K_sel + 30, 2 * n - 1)

    return best_a, best_m


def coordinate_descent(a0, time_limit, verbose=True, rng_seed=123):
    """Pair-move coordinate descent with sensitivity-based targeting."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300
    n_top = 20

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity map
        sens = np.zeros(n)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        improved_this_round = False

        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination (low sensitivity)
            candidates = np.where(sens < sens.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            candidates = np.setdiff1d(candidates, [src])
            if len(candidates) == 0:
                continue
            dst = candidates[np.argmin(sens[candidates])]

            found = False
            for frac in [0.01, 0.005, 0.02, 0.03, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved_this_round = True
                    found = True
                    break

            if not found and stall > 50:
                j = int(rng.integers(n))
                if j != src:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True

        if not improved_this_round:
            stall += 1
            # Random perturbation
            n_pert = min(3, n)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved_this_round = True
                        break

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    """U-shaped distribution."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    """Fourier-synthesized smooth sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining multiple optimization strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = make_u_shape(n_base, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Nesterov subgradient (fast, many iterations) ===
    print(f"\n=== Phase 1: Nesterov subgrad (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)

                break

    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 2: QP-FW (n={n_best}) ===")
