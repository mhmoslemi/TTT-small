    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2. Returns (J, n, max_conv)."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf, n, np.inf
    conv = fast_conv(a)
    M = conv.max()
    return 2.0 * n * M / (S * S), n, M


def subgrad_max_conv(a, conv, top_k=100, beta=10.0):
    """Compute subgradient of max(conv(a,a)) with respect to a using top-K lags."""
    n = len(a)
    k_max = min(top_k, 2 * n - 1)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_i = int(k)
        i_s = max(0, k_i - n + 1)
        i_e = min(n - 1, k_i)
        if i_s <= i_e:
            idx = np.arange(i_s, i_e + 1)
            j_idx = k_i - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g


def mirror_descent(a0, time_limit, top_k=100, beta_w=10.0, c0=5.0,
                   restart_every=5000, use_polyak=True, verbose=False):
    """Nesterov-accelerated mirror descent with FFT subgradients and Polyak averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = obj_value(a)[0]
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)

    x = a.copy()
    x_prev = a.copy()
    polyak_sum = a.copy()
    polyak_count = 0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta_w
    cur_c0 = c0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # Compute subgradient
        conv = fast_conv(y)
        g = subgrad_max_conv(y, conv, top_k=cur_top_k, beta_w=cur_beta)

        # Step size (decreasing)
        eta = cur_c0 / (np.sqrt(t) + 1.0)

        # Mirror step (exponential family on simplex)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        # Polyak averaging
        if use_polyak:
            polyak_sum += x
            polyak_count += 1

        # Evaluate and track best
        val = obj_value(a_new)[0]
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Also check Polyak average periodically
        if use_polyak and t % 500 == 0 and polyak_count > 1000:
            polyak_avg = polyak_sum / polyak_count
            polyak_avg = np.maximum(polyak_avg, 1e-15)
            polyak_avg /= polyak_avg.sum()
            pval = obj_value(polyak_avg)[0]
            if pval < best_val - 1e-15:
                best_val = pval
                best_a = polyak_avg.copy()
                stall = 0

        # Restart logic
        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.003, 0.02)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            polyak_sum = x_r.copy()
            polyak_count = 1
            cur_top_k = min(cur_top_k + 20, 250)
            cur_beta = max(cur_beta - 0.5, 2.0)
            cur_c0 = max(cur_c0 * 0.85, 1.5)
            if verbose:
                print(f"    MD restart t={t}: J={best_val:.6f}, k={cur_top_k}, b={cur_beta:.1f}, c0={cur_c0:.2f}")

        if verbose and t % 5000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}")

    return best_a, best_val


def block_coord_descent(a0, time_limit, verbose=False, seed=42):
    """Block coordinate descent: transfer mass between pairs of positions based on pressure."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = obj_value(a)[0]
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        conv = fast_conv(a)
        M = conv.max()

        # Get top lag positions
        n_lags = min(50, 2 * n - 1)
        top_lags = np.argsort(conv)[::-1][:n_lags]

        # Compute per-position contribution to peaks
        peak_contrib = np.zeros(n)
        for k in top_lags[:20]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                j_idx = k_i - idx
                peak_contrib[idx] += a[idx] * a[j_idx]

        # Find best transfer: source = high peak_contrib * mass, dest = low peak impact
        src_scores = peak_contrib * a
        src_scores[src_scores < 1e-20] = 0.0
        src = int(np.argmax(src_scores))
        if a[src] < 1e-12:
            stall += 1
            continue

        # Destination: minimize increase in peak regions
        dest_impact = np.zeros(n)
        for k in top_lags[:10]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                j_idx = k_i - idx
                dest_impact[idx] += a[j_idx]
        dest_impact[src] = np.inf
        dest_impact += 0.05 * peak_contrib
        dst = int(np.argmin(dest_impact))

        # Try multiple transfer fractions
        improved = False
        for frac in [0.003, 0.007, 0.015, 0.03, 0.05, 0.08, 0.15, 0.25]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0.0)
            st = a_t.sum()
            if st < 1e-15:
                continue
            a_t /= st
            J_t = obj_value(a_t)[0]
            if J_t < best_J - 1e-15:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            # Random kicks to escape
            if stall % 50 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.002, 0.03)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = obj_value(a)[0]
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and stall % 200 == 0:
            print(f"    BCD ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, w1=0.2, w2=0.8, width=60.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - w1) ** 2) + np.exp(-width * (x - w2) ** 2) + 0.002
    return a / a.sum()


def make_smooth_random(n, seed=0, n_freq=15, amp=0.08):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.8
    for freq in range(1, n_freq + 1):
        coeff = rng.normal(0, amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_triangular(n, center=0.5, width=0.3):
    """Triangular distribution."""
    x = np.linspace(0, 1, n)
    a = np.maximum(0, 1 - np.abs(x - center) / width)
    a += 0.001
    return a / a.sum()


def make_fourier_optimal(n, seed=0):
    """Construct a sequence whose |FFT|^2 is close to flat (for flat autocorrelation)."""
    rng = np.random.default_rng(seed)
    # In Fourier domain, want |A(k)|^2 to produce flat autocorrelation
    # This means A(k) should have roughly constant magnitude
    # Try: random phases with constant magnitude
    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
    A[0] = np.sqrt(n)  # DC component
    for k in range(1, fft_size // 2 + 1):
        phase = rng.uniform(0, 2 * np.pi)
        mag = 1.0
        A[k] = mag * np.exp(1j * phase)
    a = np.fft.irfft(A, fft_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length, normalized to sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = obj_value(parent)[0]
            print(f"Parent: n={len(parent)}, J_internal={pJ:.6f}")
            # Verify with actual evaluator
            pJ_real = evaluate_sequence([float(x) for x in parent])
            print(f"Parent: J_real={pJ_real:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = obj_value(a)[0]
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize with multiple candidates
    n0 = len(parent) if parent is not None else 3200

    print(f"\n=== Initial Candidates (n={n0}) ===")
    candidates = {}

    # Uniform
    uniform = np.ones(n0) / n0
    candidates['uniform'] = uniform

    # Parent
    if parent is not None:
        candidates['parent'] = parent.copy()

    # Power ramps
    for p_exp in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates[f'ramp_{p_exp}'] = make_power_ramp(n0, p_exp)

    # Bimodal
    candidates['bimodal_20_80'] = make_bimodal(n0, 0.2, 0.8, 50.0)
    candidates['bimodal_30_70'] = make_bimodal(n0, 0.3, 0.7, 40.0)
    candidates['bimodal_15_85'] = make_bimodal(n0, 0.15, 0.85, 70.0)

    # Smooth random
    for seed in [1, 7, 42, 123, 999]:
        candidates[f'smooth_{seed}'] = make_smooth_random(n0, seed=seed)

    # Triangular
    candidates['tri_05'] = make_triangular(n0, 0.5, 0.4)
    candidates['tri_03'] = make_triangular(n0, 0.3, 0.5)

    # Fourier-based
    for seed in [5, 11, 23]:
        candidates[f'fourier_{seed}'] = make_fourier_optimal(n0, seed=seed)

    # Evaluate all
    for label, cand in candidates.items():
        val = obj_value(cand)[0]
        print(f"  {label}: J={val:.6f}")
        try_update(cand, label)

    # === Phase 1: MD from top candidates ===
    print(f"\n=== Phase 1: MD from top candidates ===")
    remaining = time_budget - (time.monotonic() - t_global)

    # Sort candidates by value, take top 5
    sorted_cands = sorted(candidates.items(), key=lambda x: obj_value(x[1])[0])
    top_cands = sorted_cands[:5]

    t_per = max(20.0, remaining * 0.25 / len(top_cands))
    for label, cand in top_cands:
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 60:
            break
        t_avail = min(t_per, time_budget - 60 - elapsed)
        if t_avail < 15:
            break
        print(f"  MD [{label}] ({t_avail:.0f}s)...")
        a_out, _ = mirror_descent(cand, t_avail, top_k=100, beta_w=10.0, c0=5.0,
                                  restart_every=5000, use_polyak=True, verbose=False)
        try_update(a_out, f"md_{label}")

    # === Phase 2: MD from parent (longer run) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Extended MD ===")
    if parent is not None:
        t2 = min(remaining * 0.18, 300.0)
        if t2 > 30:
            print(f"  MD [parent] ({t2:.0f}s)...")
            a_out, _ = mirror_descent(parent, t2, top_k=120, beta_w=8.0, c0=4.0,
                                      restart_every=8000, use_polyak=True, verbose=True)
            try_update(a_out, "md_parent_ext")

    # Also MD from current best
    remaining = time_budget - (time.monotonic() - t_global)
    t2b = min(remaining * 0.15, 250.0)
    if t2b > 30:
        print(f"  MD [best] ({t2b:.0f}s)...")
        a_out, _ = mirror_descent(best_a.copy(), t2b, top_k=150, beta_w=7.0, c0=3.5,
                                  restart_every=10000, use_polyak=True, verbose=True)
        try_update(a_out, "md_best_ext")

    # === Phase 3: Block Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Block Coord Descent ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        print(f"  BCD ({t3:.0f}s)...")
        a_out, _ = block_coord_descent(best_a.copy(), t3, verbose=True, seed=77)
        try_update(a_out, "bcd1")

    # === Phase 4: MD after BCD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: MD after BCD ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = mirror_descent(best_a.copy(), t4, top_k=150, beta_w=6.0, c0=3.0,
                                  restart_every=12000, use_polyak=True, verbose=False)
        try_update(a_out, "md_after_bcd")

    # === Phase 5: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Length Sweep ===")
    n_best = len(best_a)
    lengths_to_try = []
    for d in [-500, -300, -100, 100, 300, 500, 800, 1200]:
        nl = n_best + d
        if 1000 <= nl <= 8000:
            lengths_to_try.append(nl)
    for nl in [1500, 2000, 2500, 3000, 4000, 5000, 6000]:
        if nl not in lengths_to_try and abs(nl - n_best) > 100:
            lengths_to_try.append(nl)

    for nl in sorted(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if abs(nl - n_best) <= 50:
            continue
        print(f"  Length n={nl}...")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.25, 80.0)
        if t_md > 15:
            a_out, _ = mirror_descent(a_int, t_md, top_k=80, beta_w=10.0, c0=4.0,
                                      use_polyak=True, verbose=False)
            try_update(a_out, f"len{nl}_md")

    # === Phase 6: Final refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Refinement ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 20:
        a_out, _ = mirror_descent(best_a.copy(), t6, top_k=200, beta_w=5.0,
                                  c0=2.5, restart_every=15000, use_polyak=True, verbose=True)
        try_update(a_out, "final_md")

    # === Phase 7: Final BCD ===
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.20, 180.0)
    if t7 > 15:
        a_out, _ = block_coord_descent(best_a.copy(), t7, verbose=True, seed=314)
        try_update(a_out, "final_bcd")

    # === Phase 8: One more MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 8: Extra MD ===")
        a_out, _ = mirror_descent(best_a.copy(), remaining - 20,
                                  top_k=200, beta_w=4.0, c0=2.0,
                                  restart_every=20000, use_polyak=True, verbose=False)
        try_update(a_out, "extra_md")

    # Final validation with actual evaluator
    if best_a is not None:
        final_J_internal = obj_value(best_a)[0]
        final_J_real = evaluate_sequence([float(x) for x in best_a])
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: J_internal={final_J_internal:.6f}, J_real={final_J_real:.6f}, n={len(best_a)}, time={elapsed:.0f}s")

        # If the actual evaluator gives a different (better) result from internal,
        # try to optimize for the actual evaluator
        # Build a list of candidates and pick the best by actual evaluation
        final_candidates = [best_a]
        if parent is not None:
            final_candidates.append(parent)
        final_candidates.append(np.ones(n0) / n0)
        for label, cand in sorted_cands[:3]:
            final_candidates.append(cand)

        best_real = np.inf
        best_final_a = best_a
        for fc in final_candidates:
            fc = np.maximum(fc, 0.0)
            fc = fc / fc.sum()
            rJ = evaluate_sequence([float(x) for x in fc])
            if rJ < best_real:
                best_real = rJ
                best_final_a = fc.copy()
            print(f"  Candidate n={len(fc)}: J_real={rJ:.6f}")

        print(f"\nBEST by real eval: J={best_real:.6f}, n={len(best_final_a)}")
        result = [float(x) for x in best_final_a]
        return result
    else:
        n_fb = 3200
        fb = np.ones(n_fb) / n_fb
        print(f"\nFALLBACK: uniform n={n_fb}")
        return [float(x) for x in fb]
