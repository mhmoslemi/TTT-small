def fast_conv(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))

def normalize(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def subgrad_max_conv(a, b, threshold_frac=0.99):
    """Compute subgradient of max conv at point a.
    Returns weighted average of Q_k @ a for active k.
    Q_k @ a is a reversed/shifted copy of a."""
    n = len(a)
    M = b.max()
    active = np.where(b >= M * threshold_frac)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        # Q_k @ a: (Q_k a)[j] = a[k-j] if 0<=k-j<n else 0
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            g[j_start:j_end+1] += a[k_int - np.arange(j_start, j_end + 1)]
    
    g /= len(active)
    return g

def polyak_subgradient(a0, time_limit, verbose=False, seed=42):
    """Polyak-averaged subgradient method for min max_k conv(a,a)_k s.t. sum(a)=1, a>=0.
    
    Uses the fact that subgradient at active lag k is Q_k @ a (shifted-reversed a).
    Polyak averaging: x_avg = (1/t) * sum_{s=1}^{t} x_s often gives better
    results than the last iterate for non-smooth problems.
    """
    n = len(a0)
    a = normalize(a0)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    
    # Step size parameters
    # For min f(x) s.t. x in S, polyak step: eta_t = R / (L * sqrt(t))
    # where R is distance from initial to optimum, L is Lipschitz constant
    # L for max of convex quadratics on simplex is bounded by 2*max|Q_k| = 2
    R_est = 1.0  # initial estimate of distance to optimum
    c_step = 2.0  # multiplier
    
    a_avg = a.copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    
    x = a.copy()
    stall = 0
    max_stall = 5000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(x)
        M = b.max()
        
        # Compute subgradient
        g = subgrad_max_conv(x, b, threshold_frac=0.99)
        
        # Step size: eta = c * R_est / (sqrt(t) * ||g||)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        eta = c_step * R_est / (np.sqrt(t) * g_norm)
        
        # Gradient step in Euclidean space
        a_new = x - eta * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = x.copy()
        else:
            a_new = a_new / s
        
        # Polyak average
        a_avg = (1 - 1.0/t) * a_avg + (1.0/t) * a_new
        
        # Check both current and average
        m_new = fast_conv(a_new).max()
        m_avg = fast_conv(a_avg).max()
        
        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        elif m_avg < best_m - 1e-16:
            best_m = m_avg
            best_a = a_avg.copy()
            stall = 0
        else:
            stall += 1
        
        x = a_new
        
        if stall > max_stall:
            # Restart with perturbation
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.01, n)
                x_new = best_a * (1 + pert)
            elif strategy == 1:
                # Fourier mode perturbation
                mode = int(rng.integers(1, max(2, n//10)))
                freq = 2.0 * np.pi * mode / n
                pert_amplitude = 0.05
                pert = pert_amplitude * np.cos(freq * np.arange(n) + rng.uniform(0, 2*np.pi))
                x_new = best_a * (1 + pert)
            elif strategy == 2:
                # Random coordinate perturbation
                idx = rng.choice(n, size=min(n//5, 500), replace=False)
                x_new = best_a.copy()
                x_new[idx] *= rng.uniform(0.5, 1.5, len(idx))
            else:
                x_new = best_a * (1 + rng.normal(0, 0.005, n))
            
            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new
            a_avg = x_new.copy()
            stall = 0
            max_stall = max(2000, int(max_stall * 0.7))
            c_step = max(0.5, c_step * 0.7)
    
    return best_a

        M = b.max()
        
        # Select active set: all lags above 99.5% of max, capped at max_active
        threshold = M * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) > max_active:
            # Use the top max_active
            active = active[np.argsort(b[active])[::-1]][:max_active]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Build LP with second-order correction
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1]
        
        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                # Linearization: b[k](a') ≈ b[k](a) + grad_bk(a) @ (a' - a)
                # grad_bk(a)[j] = 2*a[k-j]
                # So: b[k](a') ≈ b[k](a) + 2 * sum_j a[k-j]*(a'[j] - a[j])
                #     = b[k](a) + 2 * sum_j a[k-j]*a'[j] - 2 * sum_j a[k-j]*a[j]
                #     = b[k](a) + 2 * sum_j a[k-j]*a'[j] - b[k](a)  [since sum a[k-j]^2 = b[k](a) when j and k-j are both valid]
                # Wait, that's only true if the linearization is exact at a, which it is:
                # b[k](a) = sum_j a[j]*a[k-j], so grad = 2*a[k-j] and f(a) + grad@(a'-a) = b[k](a) + 2*sum a[k-j]*(a'[j]-a[j])
                # = b[k](a) + 2*sum a[k-j]*a'[j] - 2*b[k](a)
                # = 2*sum_j a[k-j]*a'[j] - b[k](a)
                # So the linearized constraint is: 2*sum_j a[k-j]*a'[j] - b[k](a) <= M
                coeffs = 2.0 * a[k_int - idx]
                rhs = M  # We want b[k](a') <= M
                constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - b[k_int] <= rhs - b[k_int])
                # Simplifies to: sum(cp.multiply(coeffs, a_var[idx])) <= M + b[k_int] - b[k_int] + ... 
                # Let me redo: linearized b[k](a') ≈ b[k](a) + grad @ (a' - a)
                # = b[k] + 2*sum a[k-j]*(a'[j]-a[j])
                # We want this <= M
                # => 2*sum a[k-j]*a'[j] - 2*sum a[k-j]*a[j] <= M - b[k]
                # => 2*sum a[k-j]*a'[j] - b[k] <= M - b[k]  [since 2*sum a[k-j]*a[j] = 2*b[k]... wait no]
                # Actually 2*sum_j a[k-j]*a[j] is NOT b[k] in general.
                # b[k] = sum_j a[j]*a[k-j]
                # 2*sum_j a[k-j]*a[j] = 2*b[k]
                # So: b[k] + 2*b[k] - 2*sum a[k-j]*a'[j] ... no wait
                # Let me be more careful.
                # f(a') = (a')^T Q_k (a') where (Q_k a')[j] = a'[k-j]
                # grad f at a = Q_k a + Q_k^T a = 2 Q_k a (since Q_k is symmetric for the relevant sub-matrix)
                # Actually Q_k is symmetric: (Q_k)_{ij} = 1 if i+j=k, so Q_k = Q_k^T
                # So grad f(a) = 2 Q_k a
                # Linearization: f(a') ≈ f(a) + (2 Q_k a)^T (a' - a) = b[k] + 2*(Q_k a)^T a' - 2*(Q_k a)^T a
                # Note (Q_k a)^T a = a^T Q_k a = b[k]
                # So: f(a') ≈ b[k] + 2*(Q_k a)^T a' - 2*b[k] = 2*(Q_k a)^T a' - b[k]
                # Constraint: 2*sum_j (Q_k a)[j] * a'[j] - b[k] <= M
                # i.e., 2*sum_j a[k-j] * a'[j] <= M + b[k]
                
        # Trust region
        constraints.append(a_var <= a * (1.0 + r))
        # Also add lower bound to prevent too much change
        constraints.append(a_var >= a * max(0.0, 1.0 - r))
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
        except Exception:
            no_improve += 1
            r = max(0.01, r * 0.5)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 1
            r = max(0.01, r * 0.5)
            continue
        
        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s = a_new.sum()
        if s < 1e-15:
            no_improve += 1
            continue
        a_new = a_new / s
        
        # Line search
        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_new
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.4, r * 1.15)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.005, r * 0.6)
            if no_improve > 15:
                # Spectral escape
                mode = int(rng.integers(1, max(2, n//10)))
                freq = 2.0 * np.pi * mode / n
                amp = rng.uniform(0.03, 0.1)
                phase = rng.uniform(0, 2*np.pi)
                a = best_a * (1 + amp * np.cos(freq * np.arange(n) + phase))
                a = normalize(a)
                no_improve = 0
                r = r_init

            print(f"    Remez2 t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, active={len(active)}")
    
    return best_a

def spectral_equalize(a, time_limit, verbose=False, seed=42):
    """Try to flatten the power spectrum to reduce peak autocorrelation.
    
    The idea: compute the DFT of a, then adjust the magnitudes to make
    |A(f)|^2 more uniform. Keep phases. Then inverse DFT and project to simplex.
    """
    n = len(a)
    a = normalize(a)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    iters = 0
    for _ in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        iters += 1
        
        A = np.fft.rfft(a.astype(np.float64))
        P = np.abs(A)**2
        P_mean = P.mean()
        P_max = P.max()
        
        # Flatten: reduce peaks, boost valleys
        # Target: make P close to P_mean
        # Use a soft clipping
        alpha = 0.3  # flattening strength
        
        P_new = P.copy()
        P_new[P > P_mean] = P_mean + (P[P > P_mean] - P_mean) * (1 - alpha)
        P_new[P < P_mean] = P_mean - (P_mean - P[P < P_mean]) * (1 - alpha)
        
        # Scale to preserve total energy
        P_new *= P.sum() / P_new.sum()
        
        # New magnitudes, keep phases
        A_new_magnitude = np.sqrt(P_new)
        A_new_phase = np.angle(A)
        A_new = A_new_magnitude * np.exp(1j * A_new_phase)
        
        a_new = np.fft.irfft(A_new, 2*n-1)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller alpha
            alpha *= 0.9
            if alpha < 0.01:
                break
    
    return best_a

def fourier_optimize(a0, time_limit, verbose=False, seed=42):
    """Optimize the sequence parameterized in Fourier space.
    
    Write a[j] = base[j] + sum_{k=1}^{K} c_k * cos(2*pi*k*j/n + phi_k)
    where base is the current best. Optimize c_k and phi_k to reduce max autocorrelation.
    """
    n = len(a0)
    a = normalize(a0)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Try adding/removing Fourier modes
    modes_to_try = list(range(1, min(20, n//10 + 1)))
    
    for mode in modes_to_try:
        if time.monotonic() - t0 > time_limit:
            break
        
        freq = 2.0 * np.pi * mode / n
        
        # Try different amplitudes and phases
        best_local = best_m
        for amp in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1]:
            for _ in range(5):  # random phases
                phase = rng.uniform(0, 2*np.pi)
                a_trial = a * (1 + amp * np.cos(freq * np.arange(n) + phase))
                a_trial = normalize(a_trial)
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_local - 1e-14:
                    best_local = m_trial
                    best_a = a_trial.copy()
                    a = a_trial
                    best_m = best_m if best_m < m_trial else m_trial
    
    return best_a

def make_init(n, kind='raised_cos'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * (x - 0.5))) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * (x - 0.5)) ** 4, 0.0)
    elif kind == 'sech':
        a = 1.0 / np.cosh(3.0 * (x - 0.5))
    else:
        a = (1.0 + np.cos(2.0 * np.pi * (x - 0.5))) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Polyak-averaged subgradient method (different from Nesterov - often better for non-smooth)
2. Spectral equalization via amplitude flattening in Fourier space
3. Nesterov mirror descent with spectral-mode restarts
4. LP Remez with second-order-corrected linearization and larger active sets
5. Fourier-mode structured perturbations for local minimum escape
6. Multi-length exploration with interpolation
Key insight: the subgradient at active lag k is Q_k@a (a shifted-reversed copy of a),
and flattening the power spectrum reduces peak autocorrelation.

import numpy as np

import time


def fast_conv(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_subgradient(a, b, threshold_frac=0.99, max_active=50):
    """Compute subgradient of max_k conv(a,a)[k] at point a.
    For active lag k, the subgradient contribution is Q_k @ a = shifted reversed a."""
    n = len(a)
    M = b.max()
    active = np.where(b >= M * threshold_frac)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    if len(active) > max_active:
        active = active[np.argsort(b[active])[::-1]][:max_active]

    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += a[k_int - idx]
    g /= len(active)
    return g


def polyak_subgradient(a0, time_limit, verbose=False, seed=42):
    """Polyak-averaged subgradient method for min max_k conv(a,a)[k] s.t. sum(a)=1, a>=0.
    Uses Euclidean gradient steps with Polyak averaging of iterates."""
    n = len(a0)
    a = normalize(a0)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    a_avg = a.copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()

    x = a.copy()
    stall = 0
    max_stall = 5000
    c_step = 2.5

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(x)
        M = b.max()

        g = compute_subgradient(x, b, threshold_frac=0.99, max_active=40)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        eta = c_step / (np.sqrt(t) * g_norm)

        a_new = x - eta * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = x.copy()
        else:
            a_new = a_new / s

        a_avg = (1 - 1.0 / t) * a_avg + (1.0 / t) * a_new

        m_new = fast_conv(a_new).max()
        m_avg = fast_conv(a_avg).max()

        if m_new < best_m - 1e-16:
            best_m = m_new
            best_a = a_new.copy()
            stall = 0
        elif m_avg < best_m - 1e-16:
            best_m = m_avg
            best_a = a_avg.copy()
            stall = 0
        else:
            stall += 1

            strategy = int(rng.integers(5))
            if strategy == 0:

                freq = 2.0 * np.pi * mode / n
                amp = rng.uniform(0.03, 0.08)

            elif strategy == 2:

                x_new = best_a.copy()
                x_new[sorted_idx[:n // 10]] *= 1.1
                x_new[sorted_idx[-n // 10:]] *= 0.9

            x_new = np.maximum(x_new, 1e-10)
            x_new = x_new / x_new.sum()
            x = x_new

    return best_a


def nesterov_mirror_v2(a0, time_limit, top_k=30, c0=3.5, verbose=False, seed=42):

        sy = y.sum()

                g[idx] += 2.0 * y[k_int - idx]

def lp_remez_v2(a, time_limit, solver=None, max_active=50, r_init=0.12, verbose=False, seed=42):
    """LP Remez with larger active set, second-order correction, and spectral escape."""
    n = len(a)
    a = normalize(a)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)
        M = b.max()

        threshold = M * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) > max_active:
            active = active[np.argsort(b[active])[::-1]][:max_active]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        constraints = [cp.sum(a_var) == 1]

        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                coeffs = 2.0 * a[k_int - idx]
                # Linearized constraint: 2*(Q_k a)^T a' <= M + b[k]
                # which is equivalent to: sum(coeffs * a_var[idx]) <= M + b[k]
                constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) <= M + b[k_int])

        constraints.append(a_var <= a * (1.0 + r))
        lower = a * max(0.0, 1.0 - r)
        constraints.append(a_var >= lower)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
        except Exception:
            no_improve += 1
            r = max(0.01, r * 0.5)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 1
            r = max(0.01, r * 0.5)
            continue

        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s = a_new.sum()
        if s < 1e-15:
            no_improve += 1
            continue
        a_new = a_new / s

        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_new
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.35, r * 1.1)
                break

        if not accepted:
            no_improve += 1
            r = max(0.005, r * 0.6)
            if no_improve > 12:
                mode = int(rng.integers(1, max(2, n // 10)))
                freq = 2.0 * np.pi * mode / n
                amp = rng.uniform(0.03, 0.08)
                phase = rng.uniform(0, 2 * np.pi)
                a = best_a * (1 + amp * np.cos(freq * np.arange(n) + phase))
                a = normalize(a)
                no_improve = 0
                r = r_init

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez2 t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, act={len(active)}")

    return best_a


def spectral_equalize(a, time_limit, verbose=False, seed=42):
    """Flatten the power spectrum to reduce peak autocorrelation."""
    n = len(a)
    a = normalize(a)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    alpha = 0.3
    no_improve = 0
    max_no_improve = 30

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a.astype(np.float64))
        P = np.abs(A) ** 2
        P_mean = P.mean()

        P_new = P.copy()
        mask_high = P > P_mean
        mask_low = P < P_mean
        P_new[mask_high] = P_mean + (P[mask_high] - P_mean) * (1 - alpha)
        P_new[mask_low] = P_mean - (P_mean - P[mask_low]) * (1 - alpha)
        P_new = np.maximum(P_new, 1e-20)
        P_new *= P.sum() / P_new.sum()

        A_new_magnitude = np.sqrt(P_new)
        A_new_phase = np.angle(A)
        A_new = A_new_magnitude * np.exp(1j * A_new_phase)

        a_new = np.fft.irfft(A_new, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-14:
            best_m = m_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            alpha = min(0.5, alpha * 1.05)
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
            alpha = max(0.05, alpha * 0.85)

    return best_a


def fourier_optimize(a0, time_limit, verbose=False, seed=42):
    """Try adding/removing Fourier modes to reduce max autocorrelation."""
    n = len(a0)
    a = normalize(a0)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    modes_to_try = list(range(2, min(25, n // 8 + 1)))

    for mode in modes_to_try:
        if time.monotonic() - t0 > time_limit:
            break

        freq = 2.0 * np.pi * mode / n

        for amp in [0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1]:
            for _ in range(4):
                if time.monotonic() - t0 > time_limit:
                    break
                phase = rng.uniform(0, 2 * np.pi)
                a_trial = a * (1 + amp * np.cos(freq * np.arange(n) + phase))
                a_trial = normalize(a_trial)
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-14:
                    best_m = m_trial
                    best_a = a_trial.copy()
                    a = a_trial

    return best_a


def make_init(n, kind='raised_cos'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * (x - 0.5))) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * (x - 0.5)) ** 4, 0.0)
    elif kind == 'sech':
        a = 1.0 / np.cosh(4.0 * (x - 0.5))
    else:
        a = (1.0 + np.cos(2.0 * np.pi * (x - 0.5))) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def get_solver():
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.SCS


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator. Phases:
    1. Polyak subgradient on parent (25%)
    2. Spectral equalization (10%)
    3. NMD with spectral restarts (20%)
    4. LP Remez (20%)
    5. Fourier mode optimization (5%)
    6. Multi-length exploration (10%)
    7. Final combined polish (10%)
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: score={sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    n_base = 2600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_init(n_base, 'raised_cos')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    solver = get_solver()
    print(f"Solver: {solver}, budget: {time_budget:.0f}s")

    # === Phase 1: Polyak subgradient (25%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 1: Polyak subgradient n={len(best_seq)} ({t1:.0f}s) ===")
    if t1 > 30:
        a_out = polyak_subgradient(normalize(best_seq), t1, verbose=True, seed=42)
        try_update(a_out, f"Polyak_n{len(best_seq)}")

    # === Phase 2: Spectral equalization (10%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 2: Spectral equalization ({t2:.0f}s) ===")
    if t2 > 15:
        a_out = spectral_equalize(normalize(best_seq), t2, verbose=False)
        try_update(a_out, f"Spectral_n{len(best_seq)}")

    # === Phase 3: NMD with spectral restarts (20%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 3: NMD v2 n={len(best_seq)} ({t3:.0f}s) ===")
    if t3 > 30:
        a_out = nesterov_mirror_v2(normalize(best_seq), t3, top_k=30, c0=3.5, verbose=True, seed=7)
        try_update(a_out, f"NMDv2_n{len(best_seq)}")

    # === Phase 4: LP Remez (20%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: LP Remez n={len(best_seq)} ({t4:.0f}s) ===")
    if t4 > 30:
        a_out = lp_remez_v2(normalize(best_seq), t4, solver=solver, max_active=50,
                            r_init=0.12, verbose=True, seed=77)
        try_update(a_out, f"Remez2_n{len(best_seq)}")

    # === Phase 5: Fourier mode optimization (5%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 5: Fourier opt n={len(best_seq)} ({t5:.0f}s) ===")
    if t5 > 10:
        a_out = fourier_optimize(normalize(best_seq), t5, verbose=False, seed=123)
        try_update(a_out, f"Fourier_n{len(best_seq)}")

    # === Phase 6: Multi-length exploration (10%) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 180.0)
    if t6 > 40:
        n_alt = len(best_seq) + 200
        print(f"\n=== Phase 6: Alt length n={n_alt} ({t6:.0f}s) ===")
        a_alt = interpolate_sequence(best_seq, n_alt)

        t6a = t6 * 0.6
        a_out = nesterov_mirror_v2(a_alt, t6a, top_k=25, c0=3.0, verbose=False, seed=99)
        try_update(a_out, f"NMDv2_n{n_alt}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = min(remaining * 0.4, 60.0)
            a_out = lp_remez_v2(normalize(best_seq), t6b, solver=solver,
                                max_active=40, r_init=0.08, verbose=False)
            try_update(a_out, f"Remez2_final_n{len(best_seq)}")

    # === Phase 7: Final polish (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 7: Final polish n={n_best} ({remaining:.0f}s) ===")

        t7a = remaining * 0.4
        if t7a > 10:
            a_out = nesterov_mirror_v2(normalize(best_seq), t7a, top_k=15, c0=2.0,
                                       verbose=False, seed=200)
            try_update(a_out, f"NMD_final_{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out = lp_remez_v2(normalize(best_seq), remaining * 0.4, solver=solver,
                                max_active=60, r_init=0.05, verbose=False)
            try_update(a_out, f"Remez_last_{n_best}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 5:
            a_out = spectral_equalize(normalize(best_seq), remaining * 0.3, verbose=False)
            try_update(a_out, f"Spectral_last_{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
