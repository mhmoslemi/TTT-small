"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-temperature subgradient mirror descent (annealed)
2. Interleaved spectral flattening (Gerchberg-Saxton inspired)
3. Structured restarts with diverse perturbations
4. Multi-length exploration

import numpy as np
import time

def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    """Safely evaluate."""

def soft_subgradient(a, c, tau, top_k=None):
    """
    Compute soft subgradient of max(c) where c = conv(a,a).
    w[k] ∝ exp(c[k]/tau)
    grad[i] = sum_k w[k] * dc[k]/da[i]
    """
    n = len(a)
    # Compute soft weights
    c_max = c.max()
    exponents = (c - c_max) / tau
    # Clip for numerical stability
    exponents = np.clip(exponents, -50, 0)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = w / w_sum
    
    # Truncate to top_k if specified
    if top_k is not None and top_k < len(c):
        top_idx = np.argsort(w)[::-1][:top_k]
        w_new = np.zeros_like(w)
        w_new[top_idx] = w[top_idx]
        s = w_new.sum()
        if s > 1e-30:
            w = w_new / s
    
    # Compute gradient
    grad = np.zeros(n)
    active_k = np.where(w > 1e-12)[0]
    
    for k in active_k:
        wk = w[k]
        k_int = int(k)
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        # First term: a[k-i]
        j1 = k_int - idx  # should be in [0, n-1]
        contrib = a[j1].copy()
        # Second term: a[i-k] (if valid and not double-counting)
        j2 = idx - k_int
        valid2 = (j2 >= 0) & (j2 < n)
        if np.any(valid2):
            # Check for double counting (i = k-i, i.e., 2i = k)
            double_mask = (2 * idx == k_int)
            contrib[valid2 & ~double_mask] += a[j2[valid2 & ~double_mask]]
            contrib[double_mask] += a[j1[double_mask]]  # add a second a[k-i] for the case 2i=k
        grad[idx] += wk * contrib
    
    return grad

def spectral_flatten_step(a, strength=0.1):
    """
    One step of spectral flattening: modify the spectrum to be more uniform,
    while maintaining non-negativity.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    mag_mean = mag.mean()
    if mag_mean < 1e-30:
        return a.copy()
    
    # Compress the magnitude
    mag_new = mag ** (1 - strength) * (mag_mean ** strength)
    # Actually: we want new_mag = mag_mean * (mag/mag_mean)^(1-strength)
    # = mag^(1-strength) * mag_mean^strength
    phase = np.angle(A)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def amd_soft(a0, time_limit, tau_init=0.02, tau_end=0.001, top_k=80,
             c0=3.0, restart_every=5000, rng_seed=42, 
             spectral_every=100, spectral_strength=0.05, verbose=False):
    """
    Mirror descent with soft subgradient and temperature annealing.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    c_best = fast_conv(a)
    best_m = c_best.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        # Temperature annealing
        progress = min(1.0, iter_count / (restart_every * 3))
        tau = tau_init * (tau_end / tau_init) ** progress
        tau = max(tau, tau_end)
        
        # Nesterov momentum
        beta_t = (iter_count - 1) / (iter_count + 1.5)
        if iter_count == 1:
            y = x.copy()
        else:
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
        
        c = fast_conv(y)
        g = soft_subgradient(y, c, tau=tau, top_k=top_k)
        
        # Mirror descent step (exponential family)
        eta = gamma / (np.sqrt(iter_count) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Periodic spectral step
        if spectral_every > 0 and iter_count % spectral_every == 0:
            x = spectral_flatten_step(x, strength=spectral_strength)
        
        # Track best
        c_new = fast_conv(x)
        M_new = c_new.max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart
        if stall_count > restart_every:
            stall_count = 0
            # Structured perturbation
            pert_type = rng.integers(0, 4)
            if pert_type == 0:
                # Gaussian noise
                strength = 0.05 + 0.03 * rng.random()
                pert = rng.normal(0, strength, n) * best_a
                x_r = best_a + pert
            elif pert_type == 1:
                # Sinusoidal perturbation
                freq = rng.integers(2, 20)
                amp = 0.05 + 0.1 * rng.random()
                idx = np.arange(n)
                pert = amp * best_a * np.sin(2 * np.pi * freq * idx / n)
                x_r = best_a + pert
            elif pert_type == 2:
                # Edge boost
                edge_width = max(1, n // 20)
                x_r = best_a.copy()
                x_r[:edge_width] *= 1.1 + 0.2 * rng.random()
                x_r[-edge_width:] *= 1.1 + 0.2 * rng.random()
            else:
                # Random position swap
                x_r = best_a.copy()
                for _ in range(3):
                    i = rng.integers(0, n)
                    j = rng.integers(0, n)
                    if i != j:
                        swap_val = 0.01 * best_a.max()
                        x_r[i] += swap_val
                        x_r[j] -= swap_val
            
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            
            # Adam
            top_k = min(top_k + 5, 150)
            gamma = c0 * (1.0 + 0.05 * min(iter_count // restart_every, 10))
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart (t={iter_count}, {elapsed:.0f}s): J={J:.6f}, tau={tau:.5f}, top_k={top_k}")
        
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD it={iter_count} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.5f}")
    
    return best_a, best_m

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coordinate_polish(a0, time_limit, rng_seed=42, verbose=False):
    """
    Coordinate descent: try moving mass from high-contribution positions
    to low-contribution positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        c = fast_conv(a)
        M_curr = c.max()
        
        # Find the peak lag
        k_peak = int(np.argmax(c))
        
        # Find positions contributing most to the peak
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        idx = np.arange(i_start, i_end)
        j1 = k_peak - idx
        contrib = a[idx] * a[j1]
        
        if contrib.max() < 1e-20:
            stall += 1
            continue
        
        # Source: position with highest contribution
        src_local = idx[np.argmax(contrib)]
        src_val = a[src_local]
        if src_val < 1e-12:
            stall += 1
            continue
        
        # Destination: random position far from the peak
        for attempt in range(20):
            dst = int(rng.integers(n))
            if dst == src_local:
                continue
            
            # Try different fractions
            found = False
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src_local] -= amt
                a_trial[dst] += amt
                if a_trial[src_local] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    stall = 0
                    found = True
                    break
            
            if found:
                break
        
        if not found:
            stall += 1
    
    return best_a, best_m
