1. Smooth-max (logsumexp) mirror descent with adaptive beta
2. LP-based Remez for local polishing
3. Multi-scale optimization (downsample -> optimize -> upsample)
4. Diverse restarts and coordinate descent
Key insight: gradient of conv(a,a)_k w.r.t. a_j is a_{k-j},

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_max_subgrad(a, beta, weight_threshold=1e-8):
    """
    Compute subgradient of logsumexp(beta * conv(a,a)) / beta.
    This approximates the subgradient of max(conv(a,a)).
    Returns (gradient, max_value, num_active).
    """
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    
    # Compute softmax weights
    shifted = beta * (b - M)
    # Clip to avoid underflow
    shifted = np.maximum(shifted, -50)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        # Fallback to hard max
        weights = np.zeros_like(b)
        weights[int(np.argmax(b))] = 1.0
    else:
        weights = exp_shifted / total
    
    # Only use lags with sufficient weight
    active = np.where(weights > weight_threshold)[0]
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += weights[k] * a[k_int - idx]
    
    return g, M, len(active)

def smooth_max_mirror_descent(a0, time_limit, beta_start=20.0, beta_end=200.0, 
                                eta_start=0.5, eta_end=0.01, verbose=False):
    """Mirror descent with smooth-max gradient and adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    rng = np.random.default_rng(12345 + n)
    stall_count = 0
    max_stall = 2000
    total_iters = 1000000
    
    for t in range(1, total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive parameters
        progress = min(t / 5000.0, 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        eta = eta_start + (eta_end - eta_start) * progress
        
        # Compute gradient
        g, M, n_active = smooth_max_subgrad(a, beta)
        
        # Mirror descent update (exponential family)
        log_a = np.log(a + 1e-30) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        # Check improvement
        M_new = fast_conv(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart if stalled
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, int(max_stall * 0.7))
            strategy = int(rng.integers(5))
            if strategy == 0:
                # Small perturbation
                x_r = best_a * (1 + rng.normal(0, 0.01, n))
            elif strategy == 1:
                # Random subset scaling
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n//5, 400), replace=False)
                x_r[idx] *= rng.uniform(0.5, 2.0, len(idx))
            elif strategy == 2:
                # Add a bump
                x_r = best_a.copy()
                center = rng.integers(0, n)
                width = max(5, n // 20)
                bump = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2) * 0.1
                x_r += bump
            elif strategy == 3:
                # Mirror + interpolate
                x_r = 0.5 * best_a + 0.5 * best_a[::-1]
            else:
                # Stronger perturbation
                x_r = best_a * (1 + rng.normal(0, 0.05, n))
            
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            a = x_r
            eta = eta_start * 0.5  # Reset step size
            beta = beta_start * 0.5  # Reset beta
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SMD t={t} ({elapsed:.1f}s): J={J:.8f}, beta={beta:.1f}, active={n_active}, stall={stall_count}")
    
    return best_a, best_m

def lp_remez_polish(a0, time_limit, solver=None, max_cons=50, verbose=False):
    """LP-based Remez method for polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = 0.1
    no_improve = 0
    rng = np.random.default_rng(42)
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Select active constraints (top lags)
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) > max_cons:
            active = np.argsort(b)[::-1][:max_cons]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Set up LP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1]
        
        for k in active:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                coeffs = 2.0 * a[k_int - idx]
                constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int] + 1e-10)
        
        # Trust region
        constraints.append(a_var <= a * (1.0 + r))
        constraints.append(a_var >= a * max(0.0, 1.0 - r))
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver:
                prob.solve(solver=solver, verbose=False)
            else:
                prob.solve(verbose=False)
        except Exception:
            r *= 0.5
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            r *= 0.5
            no_improve += 1
            continue
        
        a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
        s = a_new.sum()
        if s < 1e-15:
            no_improve += 1
            continue
        a_new = a_new / s
        
        # Line search
        accepted = False
        for frac in [1.0, 0.8, 0.5, 0.3, 0.1]:
            a_trial = (1 - frac) * a + frac * a_new
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-15:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.3, r * 1.15)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.6)
            if no_improve > 15:
                idx = rng.choice(n, size=min(n//5, 300), replace=False)
                a[idx] *= rng.uniform(0.7, 1.4, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, active={len(active)}")
    
    return best_a, best_m

def coordinate_descent(a0, time_limit, verbose=False, rng_seed=77):
    """Coordinate descent with pair moves and bisection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_moves = min(200, n // 2)
        
        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-15:
                continue
            
            # Try multiple fractions
            best_frac = None
            best_m_trial = best_m
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15, 0.3]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m_trial:
                    best_m_trial = m_trial
                    best_frac = (i, j, amt, a_trial)
            
            if best_frac is not None:
                ii, jj, amt, a_trial = best_frac
                a = a_trial
                best_m = best_m_trial
                best_a = a.copy()
                improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def downsample(a, factor):
    """Downsample by averaging blocks."""
    n = len(a)
    new_n = n // factor
    if new_n < 10:
        return a.copy()

def upsample(a, new_n):
    """Upsample using linear interpolation."""
    old_n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    ss = result.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return result / ss
