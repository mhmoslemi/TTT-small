                    stall = 0
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    CD final ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a

def aggressive_coord_descent(a0, time_limit, verbose=True, rng_seed=777):
    """
    More aggressive coordinate descent with multi-coordinate moves.
    Tries to simultaneously adjust multiple coordinates.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top peaks
        n_peaks = min(5, len(ac))
        peak_indices = np.argsort(ac)[::-1][:n_peaks]
        
        improved = False
        for peak_k in peak_indices:
            if improved:
                break
            peak_k = int(peak_k)
            
            i_start = max(0, peak_k - n + 1)
            i_end = min(n, peak_k + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[peak_k - idx]
            
            # Try reducing the top contributor by transferring to a "quiet" location
            n_top = min(10, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]
            
            for li in top_local:
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                # Find a "quiet" destination (where the local convolution is low)
                # Try to move to a location that doesn't contribute to any peak
                candidates = []
                for _ in range(50):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    # Check if dst is in a "quiet" region
                    candidates.append(dst)
                
                best_new_M = M_curr
                best_move = None
                
                for dst in candidates[:20]:
                    for frac in [0.003, 0.007, 0.015, 0.03]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_new_M - 1e-18:
                            best_new_M = M_trial
                            best_move = (src, dst, amt)
                
                if best_move is not None:
                    src_i, dst_i, amt = best_move
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M - 1e-18:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
        
        if not improved:
            stall += 1
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    AggCD final ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a

def freq_perturb(a, strength=0.02, rng=None):
    """Perturb the phase of the FFT of a."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def nesterov_mirror_descent(a0, time_limit, verbose=True):
    """
    Nesterov mirror descent with soft subgradient.
    """
    n = len(a0)
    x = norm_sum1(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    gamma = 4.0
    top_k = 50
    beta_sub = 5.0
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute soft subgradient
        ac = fast_autocorr(y)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta_sub
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        # Gradient: weighted sum of 2*C_k*y for active k
        grad = np.zeros(n)
        for wi, k_val in zip(w, top_idx):
            k_int = int(k_val)
            # grad[i] += 2 * wi * y[k_int - i] (if valid)
            i_min = max(0, k_int - n + 1)
            i_max = min(n, k_int + 1)
            for i in range(i_min, i_max):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * wi * y[j]
        
        # Mirror descent step (exponential family)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_x = np.log(y) - eta * grad
        max_log = log_x.max()
        x_new = np.exp(log_x - max_log)
        x_new = np.maximum(x_new, 1e-30)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / s_new
        
        x_prev = x
        x = x_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-16:
            best_M = M_new
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.01, n) * best_x
            x_r = best_x + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    NMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_x

def propose_candidate(budget_s=1800.0):
    """
    Main search function. Tries multiple strategies and lengths to find
    the best sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    print(f"Total time budget: {time_budget:.0f}s")
    
    # Load parent
    parent = get_parent()
    
    best_seq = None
    best_score = float('inf')
    
    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = eval_score(best_seq)
        print(f"Parent: n={len(parent)}, score={best_score:.6f}")
    else:
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = eval_score(a_norm)
        if np.isfinite(sc) and sc < best_score:
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            best_seq = a_norm.copy()
            best_score = sc
    
    # === Phase 1: Try different initializations at the current length ===
    print(f"\n=== Phase 1: Multi-init at n={n0} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    
    # Quick FW from different parametric inits
    families = ['sine', 'triangle', 'quadratic', 'bump']
    t_per_init = min(remaining * 0.04, 30.0)
    
    for fam in families:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < t_per_init + 5:
            break
        a_init = make_parametric(n0, fam)
        a_out = frank_wolfe_optimize(a_init, t_per_init, verbose=False)
        try_update(a_out, f"fw_{fam}")
    
    # === Phase 2: Main FW optimization from best ===
    print(f"\n=== Phase 2: Main FW (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out = frank_wolfe_optimize(a_in, t2, verbose=True)
        try_update(a_out, "fw_main")
    
    # === Phase 3: Nesterov mirror descent ===
    print(f"\n=== Phase 3: NMD (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out = nesterov_mirror_descent(a_in, t3, verbose=True)
        try_update(a_out, "nmd")
    
    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: CD polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out = coord_descent_polish(a_in, t4, verbose=True, rng_seed=42)
        try_update(a_out, "cd1")
    
    # === Phase 5: Second FW with different params ===
    print(f"\n=== Phase 5: FW round 2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out = frank_wolfe_optimize(a_in, t5, verbose=True)
        try_update(a_out, "fw_r2")
    
    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = sorted(set([n_best + 300, n_best + 600, n_best + 1000]))
    longer_lengths = [l for l in longer_lengths if 2000 <= l <= 6000]
    
    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print(f"\n  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        
        t6a = min(remaining * 0.15, 150.0)
        if t6a > 30:
            a_out = frank_wolfe_optimize(a_long, t6a, verbose=False)
            try_update(a_out, f"fw_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.1, 100.0)
            if t6b > 20:
                a_in = norm_sum1(best_seq)
                a_out = coord_descent_polish(a_in, t6b, verbose=False, rng_seed=99)
                try_update(a_out, f"cd_n{new_n}")
    
    # === Phase 7: Frequency diversification ===
    print(f"\n=== Phase 7: Freq-diversify (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    rng = np.random.default_rng(123)
    
    for trial in range(3):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        a_pert = freq_perturb(best_seq, strength=0.015, rng=rng)
        t7 = min(remaining * 0.08, 80.0)
        if t7 > 20:
            a_out = frank_wolfe_optimize(a_pert, t7, verbose=False)
            try_update(a_out, f"freq{trial}")
            # Quick CD after
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                a_out2 = coord_descent_polish(a_out, min(remaining * 0.04, 40.0), verbose=False, rng_seed=trial + 200)
                try_update(a_out2, f"freq_cd{trial}")
    
    # === Phase 8: Aggressive coordinate descent ===
    print(f"\n=== Phase 8: Aggressive CD (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 20:
        a_in = norm_sum1(best_seq)
        a_out = aggressive_coord_descent(a_in, t8, verbose=True, rng_seed=555)
        try_update(a_out, "aggcd")
    
    # === Phase 9: Final NW + CD cycle ===
    print(f"\n=== Phase 9: Final cycle ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t9a = min(remaining * 0.4, 100.0)
        if t9a > 20:
            a_in = norm_sum1(best_seq)
            a_out = frank_wolfe_optimize(a_in, t9a, verbose=False)
            try_update(a_out, "final_fw")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t9b = min(remaining * 0.5, 80.0)
            if t9b > 15:
                a_in = norm_sum1(best_seq)
                a_out = coord_descent_polish(a_in, t9b, verbose=False, rng_seed=999)
                try_update(a_out, "final_cd")
    
    # === Final validation ===
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')
        best_score = eval_score(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")
    
    return [float(x) for x in best_seq]
