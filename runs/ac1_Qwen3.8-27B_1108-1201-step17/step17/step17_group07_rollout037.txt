def solve_remez_lp(a, active_lags, r, solver, lower_bound_frac=0.0):

        prob.solve(solver=solver, verbose=False)

        return a.copy(), b.max(), False

    s = a_new.sum()
    if s < 1e-15:

    if ss < 1e-15:

    nmd_restart_count = 0
    iteration = 0

    print(f"\n=== Phase 1: LP Remez main loop (n={n}, until t={time_budget*0.55:.0f}s) ===")

    while time.monotonic() < t_phase1_end:
        iteration += 1
        b = fast_conv(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b

        threshold = max_b * (1.0 - epsilon)
        active = np.where(b >= threshold)[0]

        max_active = min(45, len(active))
        if len(active) > max_active:
            active = active[np.argsort(b[active])[::-1][:max_active]]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Two-phase LP for better progress
        r_big = r
        r_small = r * 0.4
        a_cand, ok = solve_remez_lp_two_phase(a, active, r_big, r_small, solver)

        if ok:
            accepted = False
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
                a_trial = (1.0 - alpha) * a + alpha * a_cand
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_conv(a_trial).max()
                if m_trial < max_b - 1e-16:
                    a = a_trial
                    max_b = m_trial
                    lp_stall = 0
                    accepted = True
                    break

            if accepted:
                if iteration % 5 == 1 or 2.0 * n * max_b < best_score - 1e-6:
                    try_update(a, f"LP_it{iteration}")
                r = min(0.12, r * 1.03)
                epsilon = max(0.0008, epsilon * 0.997)
            else:
                lp_stall += 1
                r = max(0.008, r * 0.85)
        else:
            lp_stall += 1
            r = max(0.008, r * 0.85)

        # NMD restart when LP stalls
        if lp_stall >= max_lp_stall:
            nmd_restart_count += 1
            elapsed = time.monotonic() - t_global
            print(f"  LP stall #{lp_stall} -> NMD restart #{nmd_restart_count} "
                  f"(t={elapsed:.0f}s, score={2*n*max_b:.8f})")

            batch_size = 3000 if nmd_restart_count < 5 else 2000
            a_nmd, m_nmd = nmd_batch(a, num_iters=batch_size, top_k=25,
                                      c0=3.0, rng_seed=100 + nmd_restart_count * 7)
            if m_nmd < max_b:
                a = a_nmd
                max_b = m_nmd
                try_update(a, f"NMD_r{nmd_restart_count}")

            lp_stall = 0
            r = 0.04
            epsilon = min(0.008, epsilon * 1.3)
            max_lp_stall = max(8, int(max_lp_stall * 0.85))

        # PerIODIC LOG
        if iteration % 20 == 0:
            elapsed = time.monotonic() - t_global
            print(f"  it={iteration} t={elapsed:.0f}s: score={2*n*max_b:.8f} "
                  f"r={r:.4f} eps={epsilon:.5f} |A|={len(active)}")

    # Phase 2: NMD exploration from different starting points (15% time)
    t_phase2_end = t_global + time_budget * 0.70
    print(f"\n=== Phase 2: Multi-start NMD (until t={time_budget*0.70:.0f}s) ===")

    nmd_starts = [
        ("perturb_small", a * (1 + np.random.default_rng(77).normal(0, 0.01, n))),
        ("raised_cos", None),
        ("gaussian_env", None),
    ]

    for label, init in nmd_starts:
        if time.monotonic() >= t_phase2_end:
            break
        if init is None:
            x = np.linspace(0.0, 1.0, n)
            if label == "raised_cos":
                init = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
            else:
                init = np.exp(-8.0 * (x - 0.5) ** 2 / 0.1)
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 1e-15:
            init = init / s

        t_per = min(120.0, t_phase2_end - time.monotonic() - 5)
        if t_per < 15:
            break
        a_out, m_out = nmd_with_perturb_restarts(init, t_per, top_k=20, c0=3.0,
                                                   rng_seed=200 + hash(label) % 1000)
        try_update(a_out, f"NMD_{label}")

    # Phase 3: Tight LP polish on global best (20% time)
    t_phase3_end = t_global + time_budget * 0.90
    print(f"\n=== Phase 3: Tight LP polish (until t={time_budget*0.90:.0f}s) ===")

    a = norm_sum1(best_seq)
    b = fast_conv(a)
    max_b = b.max()
    r = 0.015
    epsilon = 0.002
    lp_stall = 0
    polish_iter = 0

    while time.monotonic() < t_phase3_end:
        polish_iter += 1
        b = fast_conv(a)
        max_b = b.max()

        threshold = max_b * (1.0 - epsilon)
        active = np.where(b >= threshold)[0]
        max_active = min(30, len(active))
        if len(active) > max_active:
            active = active[np.argsort(b[active])[::-1][:max_active]]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        a_cand, M_val, ok = solve_remez_lp(a, active, r, solver, lower_bound_frac=0.05)

        if ok:
            accepted = False
            for alpha in [1.0, 0.6, 0.4, 0.2, 0.1]:
                a_trial = (1.0 - alpha) * a + alpha * a_cand
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_conv(a_trial).max()
                if m_trial < max_b - 1e-16:
                    a = a_trial
                    max_b = m_trial
                    lp_stall = 0
                    accepted = True
                    break

            if accepted:
                if polish_iter % 3 == 1 or 2.0 * n * max_b < best_score - 1e-7:
                    try_update(a, f"Polish_{polish_iter}")
                r = min(0.02, r * 1.02)
                epsilon = max(0.0005, epsilon * 0.99)
            else:
                lp_stall += 1
                r = max(0.003, r * 0.8)
                if lp_stall > 20:
                    # Micro NMD escape
                    a_esc, m_esc = nmd_batch(a, num_iters=1000, top_k=15, c0=2.0,
                                              rng_seed=500 + polish_iter)
                    if m_esc < max_b:
                        a = a_esc
                        max_b = m_esc
                    lp_stall = 0
                    r = 0.01
        else:
            lp_stall += 1
            r = max(0.003, r * 0.8)

        if polish_iter % 15 == 0:
            elapsed = time.monotonic() - t_global
            print(f"  Polish it={polish_iter} t={elapsed:.0f}s: "
                  f"score={2*n*max_b:.8f} r={r:.5f} eps={epsilon:.6f}")

    # Phase 4: Final burnout - smallest trust region (10% time)
    t_phase4_end = t_global + time_budget - 5
    print(f"\n=== Phase 4: Final burnout (until t={time_budget:.0f}s) ===")

    a = norm_sum1(best_seq)
    b = fast_conv(a)
    max_b = b.max()
    r = 0.005
    epsilon = 0.001
    burnout_iter = 0

    while time.monotonic() < t_phase4_end:
        burnout_iter += 1
        b = fast_conv(a)
        max_b = b.max()

        threshold = max_b * (1.0 - epsilon)
        active = np.where(b >= threshold)[0]
        max_active = min(20, len(active))
        if len(active) > max_active:
            active = active[np.argsort(b[active])[::-1][:max_active]]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        a_cand, M_val, ok = solve_remez_lp(a, active, r, solver, lower_bound_frac=0.1)

        if ok:
            for alpha in [1.0, 0.5, 0.3, 0.1]:
                a_trial = (1.0 - alpha) * a + alpha * a_cand
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                m_trial = fast_conv(a_trial).max()
                if m_trial < max_b - 1e-16:
                    a = a_trial
                    max_b = m_trial
                    try_update(a, f"Burnout_{burnout_iter}")
                    break
            r = max(0.001, r * 0.95)
        else:
            # Tiny NMD nudge
            if burnout_iter % 5 == 0:
                a_esc, m_esc = nmd_batch(a, num_iters=500, top_k=10, c0=1.0,
                                          rng_seed=900 + burnout_iter)
                if m_esc < max_b:
                    a = a_esc
                    max_b = m_esc
            r = max(0.001, r * 0.9)

        if burnout_iter % 10 == 0:
            elapsed = time.monotonic() - t_global
            print(f"  Burnout it={burnout_iter} t={elapsed:.0f}s: "
                  f"score={2*n*max_b:.8f} r={r:.5f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
