    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)

def compute_entropic_grad(a, T):
    """Compute entropic gradient via FFT. Returns (grad, b) where b = conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    a_rev = a[::-1].copy()
    conv_size = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    w_conv_arev = np.fft.irfft(W * AR, fft_size)[:conv_size]
    grad = 2.0 * w_conv_arev[n - 1: n - 1 + n]
    return grad, b

def topk_subgrad(y, b, top_k):
    """Compute subgradient using top-k active lags. O(n*k)."""

        top_idx = np.argpartition(b, -k_max)[-k_max:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    g = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[ki - idx]
    g /= k_max
    return g

def lp_remez_step(a, max_cons=50, r=0.08, solver=None):
    """Single LP Remez step with linearized constraints."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1, a_var <= a * (1.0 + r)]

    for k in active:
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[ki - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[ki])

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver is not None:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def coordinate_descent_polish(a0, time_limit, rng_seed=123, verbose=False):
    """Coordinate descent with random pair moves and bisection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 80)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_a, best_m

        a = 1.0 / np.cosh(4.0 * (x - 0.5))
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator using entropic gradient mirror descent + Remez."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    n_base = 2600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = make_init(n_base, 'raised_cos')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n0 = len(best_seq)

    # === Phase 1: Entropic gradient MD from parent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: Entropic MD n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = mirror_descent_entropic(norm_sum1(best_seq), t1, T0=20.0, T_growth=1.0005,
                                           c0=4.0, verbose=True, use_symmetry=True)
        try_update(a_out, f"EntMD_sym_n{n0}")

    # === Phase 2: Top-k NMD from parent (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: Top-k NMD n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_topk(norm_sum1(best_seq), t2, top_k=40, c0=4.0,
                                 verbose=False, use_symmetry=True)
        try_update(a_out, f"NMDtk_sym_n{len(best_seq)}")
        # Also try without symmetry
        a_out, _ = nesterov_topk(norm_sum1(best_seq), min(t2 * 0.5, remaining - 30), top_k=30,
                                 c0=3.0, verbose=False, use_symmetry=False)
        try_update(a_out, f"NMDtk_asym_n{len(best_seq)}")

    # === Phase 3: LP Remez on best (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 3: LP Remez n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_out, _ = remez_lp_method(norm_sum1(best_seq), t3, solver=solver, r_init=0.1, verbose=True)
        try_update(a_out, f"Remez_n{len(best_seq)}")

    # === Phase 4: Multi-init exploration (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4_total = min(remaining * 0.15, 200.0)
    if t4_total > 60:
        print(f"\n=== Phase 4: Multi-init (budget={t4_total:.0f}s) ===")
        inits = ['gaussian', 'sech', 'quartic']
        t_per = t4_total / len(inits)
        for kind in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            a_init = make_init(n0, kind)
            a_out, _ = mirror_descent_entropic(a_init, min(t_per, remaining - 5),
                                               T0=10.0, c0=3.0, verbose=False, use_symmetry=True)
            try_update(a_out, f"EntMD_{kind}_n{n0}")

    # === Phase 5: Alt length exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 60:
        n_alt = n0 + 400
        print(f"\n=== Phase 5: Alt length n={n_alt} (budget={t5:.0f}s) ===")
        a_alt = interpolate_sequence(best_seq, n_alt)
        a_out, _ = mirror_descent_entropic(a_alt, t5 * 0.6, T0=15.0, c0=3.0,
                                           verbose=False, use_symmetry=True)
        try_update(a_out, f"EntMD_n{n_alt}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out, _ = remez_lp_method(norm_sum1(best_seq), min(remaining * 0.5, 60),
                                       solver=solver, r_init=0.05, verbose=False)
            try_update(a_out, f"Remez_final_{len(best_seq)}")

    # === Phase 6: Final Remez polish (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.05, 80.0)
    print(f"\n=== Phase 6: Remez polish (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_out, _ = remez_lp_method(norm_sum1(best_seq), t6, solver=solver, r_init=0.03, verbose=False)
        try_update(a_out, f"Remez_polish_{len(best_seq)}")

    # === Phase 7: Entropic MD polish + CD (remaining) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 7: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
        t7a = min(remaining * 0.5, 100.0)
        if t7a > 10:
            a_out, _ = mirror_descent_entropic(norm_sum1(best_seq), t7a, T0=50.0,
                                               T_growth=1.001, c0=1.5, verbose=False, use_symmetry=True)
            try_update(a_out, f"EntMD_polish_{len(best_seq)}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), remaining - 5, rng_seed=777)
            try_update(a_out, f"CD_{len(best_seq)}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
