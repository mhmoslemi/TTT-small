7. Normalize: a_new = a_new / sum(a_new)

def spectral_flatten(a, n_iters, alpha=0.1, L=None):
    """Flatten the self-convolution using frequency domain operations."""
    n = len(a)
    if L is None:
        L = 1 << (2*n - 1).bit_length()
    
    a_work = a.copy().astype(np.float64)
    best_a = a_work.copy()
    best_max = fast_autocorr(a_work).max()
    
    for it in range(n_iters):
        # Zero-pad and FFT
        a_pad = np.zeros(L)
        a_pad[:n] = a_work
        A = np.fft.rfft(a_pad)
        
        # Compute convolution
        b = np.fft.irfft(np.abs(A)**2, L)[:2*n-1]
        b_max = b.max()
        
        if b_max < best_max:
            best_max = b_max
            best_a = a_work.copy()
        
        # Compute target
        target = b.sum() / len(b)  # average value
        
        # Find offending lags
        offending = b > target * 1.5  # lags significantly above average
        if not np.any(offending):
            break
        
        # For each offending lag, compute the "penalty" in frequency domain
        # Penalty[m] = sum over offending k of cos(2*pi*m*k/L) * (b[k] - target)
        # Use FFT for efficiency
        penalty_signal = np.zeros(2*n-1)
        penalty_signal[:len(b)] = np.maximum(b - target, 0)
        # The penalty in freq domain is the FFT of penalty_signal
        penalty_freq = np.fft.rfft(np.concatenate([penalty_signal, np.zeros(L - len(penalty_signal))]))
        
        # Update amplitudes
        A_amp = np.abs(A)
        A_phase = np.angle(A)
        
        # Reduce amplitudes where penalty is high
        # penalty_freq.real gives the "real" part of the penalty
        A_amp -= alpha * penalty_freq.real / A_amp.max()
        A_amp = np.maximum(A_amp, 0)

        A_new = A_amp * np.exp(1j * A_phase)
        a_new = np.fft.irfft(A_new, L)[:n]
        
        # Project: clip and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_work = a_new / s
    
    return best_a, best_max

def get_solver():

        return _solver_name
    for s in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]:

            prob.solve(solver=s, verbose=False)

def fast_conv(a, b=None):
    """Compute full linear convolution using FFT."""
    if b is None:
        b = a
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build the convolution constraint matrix T where T[m,j] = a[m-j] if valid."""
    n = len(a)
    K = 2 * n - 1
    T = np.zeros((K, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_direction(a, M, solver, max_constraints=None):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(a,b)[m] <= M for all m.
    Returns optimal b (unnormalized) or None.
    """
    n = len(a)
    K = 2 * n - 1

    if max_constraints is not None and max_constraints < K:
        # Use only the most binding constraints
        ab = fast_conv(a, a)
        top_idx = np.argsort(ab)[::-1][:max_constraints]
        T_full = build_conv_matrix(a)
        T = T_full[top_idx, :]
        rhs = M * np.ones(len(top_idx))
    else:
        T = build_conv_matrix(a)
        rhs = M * np.ones(K)

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= rhs]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_fw(a0, time_limit, solver, verbose=True,
                   max_constraints=None, max_iters=2000):
    """
    LP Frank-Wolfe with Nesterov acceleration.
    The LP oracle: max sum(b) s.t. b>=0, conv(a_current, b) <= M_current.
    """
    n = len(a0)
    x = normalize_nonneg(a0)
    x_prev = x.copy()
    best_x = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()

    # Adaptive constraint count
    if max_constraints is None:
        max_constraints = 2 * n - 1 if n <= 2500 else 3000

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current M
        ac = fast_autocorr(x)
        M_curr = ac.max()

        # Solve LP for direction
        b_dir = solve_lp_direction(x, M_curr, solver, max_constraints)
        if b_dir is None:
            continue

        g = b_dir / b_dir.sum()

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        if t == 1:
            y = x
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

        x_new = (1.0 - alpha) * y + alpha * g
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y.copy()
        else:
            x_new = x_new / sx

        x_prev = x.copy()
        x = x_new

        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_x = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    NFW iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_x, best_M


def spectral_flatten_init(a0, n_iters=50, alpha=0.08, seed=0):
    """
    Improve a sequence by flattening its convolution spectrum.
    Uses frequency-domain amplitude adjustment.
    """
    n = len(a0)
    L = 1 << (2 * n - 1).bit_length()
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    rng = np.random.default_rng(seed)

    for it in range(n_iters):
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        A_amp = np.abs(A)
        A_phase = np.angle(A)

        b = np.fft.irfft(A_amp ** 2, L)[:2 * n - 1]
        b_max = b.max()
        if b_max < best_M - 1e-14:
            best_M = b_max
            best_a = a.copy()

        b_mean = b.sum() / len(b)
        k_peak = int(np.argmax(b))

        # Reduce amplitudes of bins contributing to the peak
        m_vals = np.arange(A_amp.shape[0])
        cos_peak = np.cos(2 * np.pi * m_vals * k_peak / L)
        scale = 1.0 - alpha * np.maximum(cos_peak, 0)
        scale = np.maximum(scale, 0.5)
        A_amp = A_amp * scale

        # Also try boosting bins that are in "valleys"
        k_val = int(np.argmin(b))
        cos_val = np.cos(2 * np.pi * m_vals * k_val / L)
        scale_up = 1.0 + 0.3 * alpha * np.maximum(cos_val, 0)
        A_amp = A_amp * scale_up

        A_new = A_amp * np.exp(1j * A_phase)
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        # Occasionally add random perturbation to escape local structure
        if it % 20 == 0 and it > 0:
            pert = rng.normal(0, 0.02, n)
            a_new = a_new + pert * a_new
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()

        a = a_new

    return best_a, best_M


def coordinate_polish(a0, time_limit, verbose=True, seed=42):
    """
    Coordinate descent: try moving mass from high-impact positions to low-impact ones.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    iterations = 0
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < best_M:
            best_M = M_curr
            best_a = a.copy()

        # Find the peak lag
        k_peak = int(np.argmax(ac))

        # Find positions with high contribution to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        valid_js = np.arange(js, je + 1)
        if len(valid_js) == 0:
            continue
        contrib = a[valid_js] * a[k_peak - valid_js]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            continue

        # Source: position with highest contribution
        src_local = np.argmax(contrib)
        src = valid_js[src_local]

        # Destination: position with lowest "sensitivity" (least impact on peak lags)
        # Sensitivity = how much moving mass there affects the top lags
        top_lags = np.argsort(ac)[::-1][:15]
        sens = np.zeros(n)
        for kl in top_lags:
            kli = int(kl)
            s2 = max(0, kli - n + 1)
            e2 = min(n - 1, kli)
            for d in range(s2, e2 + 1):
                partner = kli - d
                if 0 <= partner < n:
                    sens[d] += a[partner]

        # Best destination: low sensitivity, not too close to source
        dist = np.abs(np.arange(n) - src).astype(float)
        mask = (sens < np.mean(sens)) & (dist > n * 0.05)
        candidates = np.where(mask)[0]
        if len(candidates) == 0:
            candidates = np.where(sens < np.mean(sens))[0]
        if len(candidates) == 0:
            candidates = np.arange(n)

        # Pick destination with lowest sensitivity / best distance
        scores = sens[candidates] / (1.0 + dist[candidates] / n)
        dst = candidates[np.argmin(scores)]

        if dst == src:
            # Try random pair
            src = int(rng.integers(n))
            dst = int(rng.integers(n))
            if src == dst:
                continue

        # Try transfer
        src_val = a[src]
        if src_val < 1e-12:
            continue

        improved = False
        for frac in [0.01, 0.005, 0.02, 0.002, 0.03, 0.008, 0.015, 0.05]:
            amt = src_val * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved and iterations % 50 == 0:
            # Random perturbation to escape
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 != i2 and a[i1] > 1e-10:
                amt = a[i1] * rng.uniform(0.005, 0.02)
                a[i1] -= amt
                a[i2] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()

        if verbose and iterations % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    CD iter={iterations} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_M


def make_smooth_init(n, seed=0):
    """Create a smooth non-negative initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)

def interpolate_sequence(f, new_n):

    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Multi-phase: spectral init -> LP-FW (Nesterov) -> coordinate polish -> length adapt.
    """
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize_nonneg(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_smooth_init(n_base, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Spectral flattening (quick, for better start) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: Spectral Flattening ===")
    t_p1 = min(remaining * 0.05, 60.0)
    if t_p1 > 10:
        a_spec, _ = spectral_flatten_init(best_seq, n_iters=80, alpha=0.06, seed=7)
        try_update(a_spec, "spectral")

        # Also try from different inits
        a_smooth = make_smooth_init(n0, seed=42)
        a_spec2, _ = spectral_flatten_init(a_smooth, n_iters=60, alpha=0.1, seed=13)
        try_update(a_spec2, "spectral_smooth")

    # === Phase 2: LP Frank-Wolfe with Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 2: LP-FW Nesterov (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t_p2 = min(remaining * 0.45, 700.0)
    if t_p2 > 30:
        a_in = normalize_nonneg(best_seq)
        mc = 2 * n_best - 1 if n_best <= 2000 else 4000
        a_out, _ = nesterov_lp_fw(a_in, t_p2, solver, verbose=True,
                                   max_constraints=mc, max_iters=5000)
        try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 3: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 3: Coordinate Descent ===")
    n_best = len(best_seq)
    t_p3 = min(remaining * 0.15, 200.0)
    if t_p3 > 15:
        a_in = normalize_nonneg(best_seq)
        a_out, _ = coordinate_polish(a_in, t_p3, verbose=True, seed=99)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 4: Second LP-FW round (with fresh start from best) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 4: LP-FW Round 2 ===")
    n_best = len(best_seq)
    t_p4 = min(remaining * 0.20, 300.0)
    if t_p4 > 20:
        a_in = normalize_nonneg(best_seq)
        mc = 2 * n_best - 1 if n_best <= 2000 else 4000
        a_out, _ = nesterov_lp_fw(a_in, t_p4, solver, verbose=True,
                                   max_constraints=mc, max_iters=3000)
        try_update(a_out, f"lpfw_r2_n{n_best}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    candidate_lengths = [n_best - 400, n_best + 400, n_best - 800, n_best + 800]
    candidate_lengths = [l for l in candidate_lengths if 1500 <= l <= 5000]

    for new_n in candidate_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 60:
            break
        print(f"\n=== Phase 5: Length n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        # Quick LP-FW
        t_trial = min(remaining * 0.2, 150.0)
        if t_trial > 15:
            mc = 2 * new_n - 1 if new_n <= 2000 else 3500
            a_out, _ = nesterov_lp_fw(a_long, t_trial, solver, verbose=False,
                                       max_constraints=mc, max_iters=2000)
            try_update(a_out, f"lpfw_n{new_n}")

        # Quick coordinate
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 20:
            t_cd = min(remaining * 0.1, 60.0)
            if t_cd > 10:
                a_out, _ = coordinate_polish(a_long, t_cd, verbose=False, seed=77)
                try_update(a_out, f"coord_n{new_n}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Polish (n={n_best}) ===")
    if remaining > 20:
        # Coordinate descent
        t_final_cd = min(remaining * 0.4, 120.0)
        if t_final_cd > 10:
            a_in = normalize_nonneg(best_seq)
            a_out, _ = coordinate_polish(a_in, t_final_cd, verbose=True, seed=123)
            try_update(a_out, f"final_coord_n{n_best}")

        # Final LP
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 15:
            t_final_lp = min(remaining * 0.5, 120.0)
            if t_final_lp > 10 and n_best <= 4000:
                a_in = normalize_nonneg(best_seq)
                mc = 2 * n_best - 1 if n_best <= 2000 else 3500
                a_out, _ = nesterov_lp_fw(a_in, t_final_lp, solver, verbose=False,
                                           max_constraints=mc, max_iters=2000)
                try_update(a_out, f"final_lp_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_smooth_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
