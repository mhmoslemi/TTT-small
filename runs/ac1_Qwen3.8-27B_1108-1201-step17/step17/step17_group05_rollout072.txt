where cross_corr(f, w)[j] = Σ_m f[m] * w[m+j]

And we showed that cross_corr(f, w)[j] = conv(f_rev, w)[j + n - 1] where f_rev = f[::-1].

No factor of 1/β! The 1/β cancels with the β in the derivative of the log.

Let me verify: h = (1/β) * log(Σ exp(β*c_k))

def compute_grad(f, beta):
    n = len(f)
    cv = fast_conv(f)
    M = cv.max()
    shifted = cv - M
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w = w / ws
    h_val = M + np.log(ws) / beta
    
    f_rev = f[::-1]
    conv_len = n + (2*n-1) - 1

def mirror_descent(f0, time_limit, seed=42, verbose=False):
    n = len(f0)

    t = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Adaptive beta
        progress = min(1.0, t / 50000.0)
        beta = 30.0 - 25.0 * progress  # 30 to 5
        
        grad, h_val, M_curr = compute_grad(f, beta)
        
        # Step size
        eta = 4.0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent update
        log_f = np.log(np.maximum(f, 1e-300))
        log_f_new = log_f - eta * grad
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        f_new = np.maximum(f_new, 1e-300)
        f_new = f_new / f_new.sum()
        
        # Evaluate
        M_new = fast_conv(f_new).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
        
        f = f_new
        
        # Restart

            stall = 0

def coord_descent(f0, time_limit, seed=42, verbose=False):

    iters = 0
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        iters += 1
        cv = fast_conv(f)
        M_curr = cv.max()
        
        # Find top peak lags
        n_top = min(10, 2*n-1)
        top_lags = np.argsort(cv)[::-1][:n_top]
        
        improved = False
        
        for k_star in top_lags[:5]:
            k_star = int(k_star)
            if improved:
                break
            
            # Find contributing positions
            js = max(0, k_star - n + 1)
            je = min(n-1, k_star)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = f[idx] * f[k_star - idx]
            
            # Try moving mass from high-contribution positions to low-sensitivity positions
            for ci in np.argsort(contrib)[::-1][:5]:
                src = int(idx[ci])
                if f[src] < 1e-12:
                    continue
                
                # Find best destination
                # We want f[k_star - p] to be small
                # Compute sensitivity for top lags
                sens = np.zeros(n)
                for k2 in top_lags[:n_top]:
                    k2 = int(k2)
                    s2 = max(0, k2 - n + 1)
                    e2 = min(n-1, k2)
                    for p in range(s2, e2+1):
                        sens[p] += f[k2 - p]
                
                # Find positions with low sensitivity and some distance from src
                candidates = np.where((sens < np.percentile(sens, 30)))[0]
                if len(candidates) == 0:
                    candidates = np.arange(n)
                
                # Score by sensitivity
                best_p = candidates[np.argmin(sens[candidates])]
                if best_p == src:
                    continue
                
                # Try several fractions
                for frac in [0.01, 0.005, 0.02, 0.002, 0.05]:
                    amt = f[src] * frac
                    if amt < 1e-15:
                        continue
                    f_trial = f.copy()
                    f_trial[src] -= amt
                    f_trial[int(best_p)] += amt
                    f_trial = np.maximum(f_trial, 0)
                    st = f_trial.sum()
                    if st < 1e-15:
                        continue
                    f_trial = f_trial / st
                    M_trial = fast_conv(f_trial).max()
                    if M_trial < M_curr - 1e-14:
                        f = f_trial
                        M_curr = M_trial
                        if M_curr < best_M:
                            best_M = M_curr
                            best_f = f.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and f[i] > 1e-10:
                    amt = f[i] * rng.uniform(0.01, 0.05)
                    f[i] -= amt
                    f[j] += amt
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                    M_new = fast_conv(f).max()
                    if M_new < best_M:
                        best_M = M_new

            print(f"    CD it={iters} ({elapsed:.0f}s): J={J:.6f} stall={stall}")

def lp_fw_kick(f, M, max_time=5.0):
    """Solve LP to find a direction that reduces the convolution peak."""
    from scipy.optimize import linprog
    n = len(f)
    # Build constraint matrix T: T[k, j] = f[k-j] if 0 <= k-j < n else 0
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = f
    
    # LP: max sum(b) s.t. T @ b <= M, b >= 0
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=T, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if res.status == 0 and res.fun is not None:
            b = np.maximum(res.x, 0)
            b_sum = b.sum()
            if b_sum > 1e-10:
                return b / b_sum
    except:
        pass
    return None

def make_smooth_init(n, seed=0, symmetric=True):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    freqs = [1, 2, 3, 4, 5]
    for freq in freqs:
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2*np.pi)
        a += coeff * np.sin(2*np.pi*freq*x + phase)
    a = np.maximum(a, 0)
    if symmetric:
        a = 0.5*(a + a[::-1])
    return normalize(a)

def make_bell_init(n, seed=0, center=0.5, width=0.2):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5*((x-center)/width)**2)
    return normalize(a)

def make_random_init(n, seed=0, concentration=1.0):
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n) * concentration)
    return a

def make_block_init(n, n_blocks=10, seed=0):
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    block_size = n // n_blocks
    for i in range(n_blocks):
        start = i * block_size
        end = start + block_size if i < n_blocks - 1 else n
        a[start:end] = rng.uniform(0.1, 1.0)
    return normalize(a)
