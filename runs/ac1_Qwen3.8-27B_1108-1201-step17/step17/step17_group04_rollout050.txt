def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:

def compute_softmax_weights(ac, tau):
    """Compute softmax weights for the soft-max approximation."""
    scaled = ac / tau
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-30:
        return np.ones(len(ac)) / len(ac)
    return w / s


def compute_gradient_fft(a, w):
    """
    Compute gradient of soft-max(conv(a,a)) wrt a using FFT.
    grad[j] = 2 * sum_m w[j+m] * a[m] (cross-correlation)
    """
    n = len(a)
    size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(A) * W, size)
    return 2.0 * corr[:n]

        M = ac.max()

        grad = compute_gradient_fft(y, w)

        J_current = 2.0 * n * M
        J_lower = 1.0 + 1.0 / (2.0 * n - 1)  # theoretical lower bound ~1
        gap = J_current - J_lower
