def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        # Anneal temperature (sqrt schedule for slower ramp)
        elapsed_frac = min((t) / (time_limit * 4.0), 1.0)
        T = T0 * (T_end / T0) ** (elapsed_frac ** 0.7)

        p, ac = compute_P(a, T)
        grad = smooth_gradient_fft(a, p)

        # Learning rate schedule
        lr = lr0 / (1.0 + 0.005 * t)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        a = a_new
        J = compute_J(a)
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Periodic restart with perturbation
        if stall > restart_every:
            stall = 0
            sigma = 0.02 + 0.005 * min(stall // restart_every, 5)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-8)
            x_r = x_r / x_r.sum()
            a = x_r
            m = np.zeros(n)
            v = np.zeros(n)
            if verbose:
                print(f"    Adam restart: J={best_J:.6f}, T={T:.0f}")

        if verbose and t % 5000 == 0:
            print(f"    Adam t={t}: J={best_J:.6f}, T={T:.0f}")

    return best_a, best_J


def lbfgs_polish(a0, time_limit, T=500.0, max_iter=200, seed=0, verbose=False):
    """L-BFGS-B polish on the scale-invariant smoothed objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()

    def objective(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x_norm = x / s
        ac = fast_autocorr(x_norm)
        max_ac = ac.max()
        shifted = T * (ac - max_ac)
        shifted = np.clip(shifted, -700, 700)
        val = max_ac + np.log(np.exp(shifted).sum()) / T
        return 2.0 * n * val

    def gradient(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return np.ones(n)
        x_norm = x / s
        ac = fast_autocorr(x_norm)
        max_ac = ac.max()
        shifted = T * (ac - max_ac)
        shifted = np.clip(shifted, -700, 700)
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(ac))
            p[int(np.argmax(ac))] = 1.0
        else:
            p = p / p_sum

        # Gradient of the scale-invariant smoothed objective
        # f_T/da[j] = (4n/S^3) * (S * (a*p_cross)[j] - ac.p)
        # Simplified (removing scale factor since we normalize):
        S = 1.0  # since x_norm has sum 1
        acp = np.dot(ac, p)
        cross = smooth_gradient_fft(x_norm, p) / 2.0  # (a*p_cross)[j]
        # Effective gradient on the simplex
        grad = 2.0 * n * (S * cross - acp)
        # Project to tangent space of simplex (remove component along 1)
        grad = grad - np.mean(grad)
        return grad

    best_val = objective(a)

    for attempt in range(3):
        if time.monotonic() - t0 > time_limit:
            break
        x0 = best_a.copy() * (1.0 + 0.01 * attempt)
        x0 = np.maximum(x0, 1e-10)
        x0 = x0 / x0.sum()

        try:
            res = minimize(objective, x0, jac=gradient,
                          method='L-BFGS-B',
                          bounds=[(0, 1) for _ in range(n)],
                          options={'maxiter': max_iter, 'maxfun': max_iter * 5,
                                   'ftol': 1e-10, 'gtol': 1e-6,
                                   'iprint': -1})
        except Exception:
            continue

        if res.success or res.fun < best_val:
            x_opt = np.maximum(res.x, 0.0)
            s = x_opt.sum()
            if s > 1e-15:
                x_opt = x_opt / s
                J_opt = compute_J(x_opt)
                if J_opt < best_J:
                    best_J = J_opt
                    best_a = x_opt.copy()
                best_val = objective(x_opt)

        if verbose:
            print(f"    L-BFGS attempt {attempt}: J={best_J:.6f}")

    return best_a, best_J
