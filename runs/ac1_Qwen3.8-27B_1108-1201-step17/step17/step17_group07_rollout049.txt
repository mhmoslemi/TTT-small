```python
def proj_simplex(v, target_sum=1.0):
    """Project v onto the probability simplex {a >= 0, sum(a) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Scale to target sum
    v = v / target_sum * 1.0  # normalize to sum 1 first
    # Then project onto standard simplex (sum = 1, a >= 0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Rescale to target_sum
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n * target_sum
    return w / s * target_sum
```

        # Gradient at momentum point
        z_clip = np.maximum(z, 1e-30)
        z_clip = z_clip / z_clip.sum()
        grad_z = softmax_conv_grad(z_clip, T)
        
        # Step size
        L = 4.0 * T + 1e-10
        step = base_step / L
        
        # FISTA update
        alpha = (k - 1.0) / (k + 2.0)
        y = x + alpha * (x - z)
        y = np.maximum(y, 0.0)
        
        x_new = proj_simplex(y - step * grad_z)
        
        # Evaluate
        M_new = fast_conv(x_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            max_stall = max(500, int(max_stall * 0.6))
            base_step = max(0.1, base_step * 0.5)
            # Perturb
            pert = rng.normal(0, 0.01, n)
            x_r = best_a * (1 + pert)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            z = x_r
        
        # Update momentum
        k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k * k))
        
        z = x
        x = x_new
        k = min(k_new, 100.0)  # cap momentum to avoid instability
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it} T={T:.2f} step={step:.4f} ({elapsed:.1f}s): "
                  f"max_b={best_m:.10f}, obj={2*n*best_m:.8f}, stall={stall}")
    
    return best_a, best_m

def lp_remez_step(a, max_cons=50, r=0.05, threshold_frac=0.999, solver=None):
    """Single LP Remez step with wider active set."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Active set: all lags within threshold of max
    threshold = max_b * threshold_frac
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = active[np.argsort(b[active])[::-1][:max_cons]]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k_idx in active:
        k_int = int(k_idx)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            # Linearization: b_k(a') ≈ b_k(a) + grad_k(a)^T (a' - a)
            # = b_k(a) + sum_i grads_k[i] * (a'[i] - a[i])
            # We want: b_k(a') <= M_var
            # Linearized: b_k(a) + sum_i coeffs[i] * a'[i] - sum_i coeffs[i]*a[i] <= M_var
            # sum_i coeffs[i] * a[i] = b_k(a) (by construction)
            # So: b_k(a) + sum_i coeffs[i] * a'[i] - b_k(a) <= M_var
            # sum_i coeffs[i] * a'[i] <= M_var
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) <= M_var)
    
    # Trust region: a'[j] in [a[j]*(1-r), a[j]*(1+r)]
    constraints.append(a_var >= a * (1.0 - r))
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy(), False
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), False
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), False
    return a_new / s, True

def remez_method(a0, time_limit, solver=None, r_init=0.05, verbose=False, seed=123):
    """LP-based Remez with adaptive trust region."""
    n = len(a0)
    a = proj_simplex(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(seed)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand, success = lp_remez_step(a, max_cons=50, r=r, 
                                         threshold_frac=0.998, solver=solver)
        
        if not success:
            no_improve += 1
            r = max(0.005, r * 0.7)
            continue
        
        # Quadratic line search
        accepted = False
        b_vals = []
        trial_fracs = [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]
        
        for t_frac in trial_fracs:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            b_vals.append((t_frac, m_trial))
            if m_trial < best_m - 1e-14:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.15, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.75)
            if no_improve > 30:
                # Perturb to escape
                idx = rng.choice(n, size=min(n // 5, 500), replace=False)
                a[idx] *= rng.uniform(0.8, 1.2, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = min(0.1, r * 2.0)
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Remez t={t} ({elapsed:.1f}s): obj={2*n*best_m:.8f}, r={r:.5f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=999):
    """Coordinate descent with random pair moves and bisection."""
    n = len(a0)
    a = proj_simplex(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 80)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            # Try multiple transfer fractions
            for frac in [0.001, 0.005, 0.01, 0.03, 0.08, 0.15, 0.3]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sech':
        a = 1.0 / np.cosh(4.0 * (x - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-10)
    ss = f_interp.sum()
    return f_interp / ss
