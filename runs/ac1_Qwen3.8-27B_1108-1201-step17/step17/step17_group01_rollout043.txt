def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_soft_subgrad(a, ac, beta=100.0):
    """Compute soft-max exponential subgradient using all lags."""
    n = len(a)
    ac_shifted = ac - ac.max()
    p = np.exp(beta * ac_shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(ac)) / len(ac)
    else:
        p = p / p_sum
    # h[i] = sum_j a[j] * p[i+j] for valid i, j
    # This is cross-correlation of p and a
    h = np.correlate(p, a, mode='valid')[:n]
    return 2.0 * h


def compute_topk_subgrad(a, ac, top_k=50, beta=5.0):
    """Compute weighted subgradient using top-k peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx] - ac.min(), 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, use_soft=True, soft_beta=200.0, top_k=50,
                 top_beta=5.0, c0=5.0, restart_every=5000, rng_seed=42):
    """Nesterov-accelerated mirror descent with multiplicative updates."""
    global _global_best, _global_best_score, _global_t0
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = top_beta
    cur_soft_beta = soft_beta

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        if use_soft and t % 3 == 0:
            g = compute_soft_subgrad(y, ac, beta=cur_soft_beta)
        else:
            g = compute_topk_subgrad(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every and stall_count > 0:
            stall_count = 0
            pert_strength = min(0.15, 0.03 + 0.01 * min(stall_count // 200, 10))
            pert = rng.normal(0, pert_strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 15, 150)
            cur_beta = max(cur_beta - 0.2, 2.0)
            cur_soft_beta = max(cur_soft_beta * 0.9, 50.0)
            gamma = c0 * (1.0 + 0.05 * min((t // restart_every), 10))

    # Check global
    J = 2.0 * n * best_m
    if J < _global_best_score:
        _global_best_score = J
        _global_best = best_a.copy()
    return best_a, best_m


def additive_subgrad_descend(a0, time_limit, lr0=0.01, decay=0.9999, rng_seed=42):
    """Additive projected subgradient descent."""
    global _global_best, _global_best_score
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    lr = lr0
    stall = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        k_star = int(np.argmax(ac))
        M = ac[k_star]

        # Subgradient at k_star
        g = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        g[idx] = 2.0 * a[k_star - idx]

        # Also add contribution from a few other peaks
        top_idx = np.argsort(ac)[::-1][:20]
        for k2 in top_idx[1:5]:
            k2i = int(k2)
            s2 = max(0, k2i - n + 1)
            e2 = min(n, k2i + 1)
            idx2 = np.arange(s2, e2)
            g[idx2] += 0.5 * a[k2i - idx2]

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        lr *= decay

        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            lr = lr0 * 0.5

    J = 2.0 * n * best_m
    if J < _global_best_score:
        _global_best_score = J
        _global_best = best_a.copy()
    return best_a, best_m


def coordinate_descent(a0, time_limit, n_peaks=10, rng_seed=42):
    """Peak-targeted coordinate descent: move mass from peak contributors to low-cost positions."""
    global _global_best, _global_best_score
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_peaks]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            max_contrib_idx = np.argmax(contrib)
            src = idx[max_contrib_idx]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute cost: if we move mass from src to dst, how much does max peak change?
            # Approximate: cost at position d = sum over top peaks of contribution if we had mass at d
            cost = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n, k2i + 1)
                idx2 = np.arange(s2, e2)
                cost[idx2] += a[k2i - idx2]

            # Find low-cost destination with low current value
            sorted_cost = np.argsort(cost)
            candidates = sorted_cost[:max(10, n // 10)]
            # Prefer positions with low a value and low cost
            scores = cost[candidates] + a[candidates] * M_curr * n
            best_dst = candidates[np.argmin(scores)]

            if best_dst == src:
                best_dst = candidates[(np.argmin(scores) + 1) % len(candidates)]
                if best_dst == src:
                    continue

            # Try moving a fraction of mass
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = min(src_val * frac, a[src] * 0.5)
                if amt < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break

    J = 2.0 * n * best_m
    if J < _global_best_score:
        _global_best_score = J
        _global_best = best_a.copy()
    return best_a, best_m


def freq_domain_flatten(a0, time_limit, iters=5000, eta0=0.7):
    """Spectrum flattening via iterative projection."""
    global _global_best, _global_best_score
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        eta = eta0 * (1.0 + 0.3 * it / max(1, iters))
        P_target = P_mean * (P / P_mean) ** (1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)

        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.7 * a + 0.3 * a_new
            a = norm_sum1(a)

    J = 2.0 * n * best_m
    if J < _global_best_score:
        _global_best_score = J
        _global_best = best_a.copy()
    return best_a, best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_tilted_gaussian(n, mu=0.3, sigma=0.25, beta=2.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - mu) ** 2) / (2 * sigma ** 2)) * np.exp(beta * x)
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def make_tilted_cosine(n, alpha=1.0, mu=0.4):
    x = np.linspace(0, 1, n)
    a = (1.0 + np.cos(np.pi * (x - mu) * 2.0 / (1.0 - mu))) ** 0.5
    a = a * (1.0 + alpha * x)
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def make_exp_ramp(n, rate=1.0, offset=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x) + offset
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def make_quadratic_tilt(n, alpha=1.0):
    x = np.linspace(0, 1, n)
    a = (1.0 + alpha * x + 0.5 * x ** 2)
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def make_random_asym(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    n_freq = rng.integers(3, 12)
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.2 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    # Add asymmetry
    a *= (1.0 + rng.uniform(0.5, 2.0) * x)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_power_asym(n, power=0.5, direction=1):
    x = np.linspace(0, 1, n)
    if direction > 0:
        a = (x + 0.1) ** power
    else:
        a = (1.0 - x + 0.1) ** power
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def solution_interpolation(a1, a2, alpha=0.5):
    """Interpolate between two solutions of possibly different lengths."""
    if len(a1) == len(a2):
        a = alpha * a1 + (1 - alpha) * a2
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return a1.copy()
        return a / s
    else:
        # Interpolate to same length
        n = max(len(a1), len(a2))
        a1_i = interpolate_seq(a1, n)
        a2_i = interpolate_seq(a2, n)
        a = alpha * a1_i + (1 - alpha) * a2_i
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return a1_i
        return a / s

        inits.append(make_tilted_cosine(best_n, alpha=0.8, mu=0.5))
        labels.append("tcoss_b")

        # Exponential ramps
        inits.append(make_exp_ramp(best_n, rate=2.0, offset=0.2))
        labels.append("exp_ramp_a")
        inits.append(make_exp_ramp(best_n, rate=3.0, offset=0.5))
        labels.append("exp_ramp_b")

        # Quadratic tilt
        inits.append(make_quadratic_tilt(best_n, alpha=2.0))
        labels.append("quad_tilt")

        # Power asymmetric
        inits.append(make_power_asym(best_n, power=0.3, direction=1))
        labels.append("power_asym_a")
        inits.append(make_power_asym(best_n, power=1.5, direction=-1))
        labels.append("power_asym_b")

        # Random asymmetric
        inits.append(make_random_asym(best_n, seed=7))
        labels.append("rand_asym_a")
        inits.append(make_random_asym(best_n, seed=123))
        labels.append("rand_asym_b")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, len(init)//30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(8.0, t2_total / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed = time.monotonic() - _global_t0
            if elapsed > time_limit - 60:
                break
            t_avail = min(t_per, time_limit - 60 - elapsed)
            if t_avail < 6:
                break

            # Quick AMD
            a_out, _ = amd_optimize(init, t_avail, use_soft=True,
                                    soft_beta=200.0, top_k=50, top_beta=5.0,
                                    c0=4.0, restart_every=3000, rng_seed=100+idx)
            J = 2.0 * len(a_out) * fast_autocorr(a_out).max()
            if J < _global_best_score:
                _global_best_score = J
                _global_best = a_out.copy()
                print(f"  [{lbl}] J={J:.6f} (n={len(a_out)}, t={elapsed:.0f}s)")

    # Phase 3: Frequency domain on best
    remaining = time_limit - (time.monotonic() - _global_t0)
    print(f"\n=== Phase 3: Freq domain (n={len(_global_best)}) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        a_in = norm_sum1(_global_best)
        a_out, _ = freq_domain_flatten(a_in, t3, iters=5000, eta0=0.7)
        J = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if J < _global_best_score:
            _global_best_score = J
            _global_best = a_out.copy()
            print(f"  Freq domain: J={J:.6f}")

    # Phase 4: AMD polish on best with different hyperparams
    remaining = time_limit - (time.monotonic() - _global_t0)
    print(f"\n=== Phase 4: AMD polish ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_in = norm_sum1(_global_best)
        # Run 2 variants
        t4a = t4 * 0.6
        t4b = t4 * 0.4
        a_out, _ = amd_optimize(a_in, t4a, use_soft=True, soft_beta=500.0,
                                top_k=100, top_beta=3.0, c0=3.0,
                                restart_every=6000, rng_seed=999)
        J = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if J < _global_best_score:
            _global_best_score = J
            _global_best = a_out.copy()
            print(f"  AMD polish A: J={J:.6f}")

        remaining = time_limit - (time.monotonic() - _global_t0)
        if remaining > 20:
            a_out, _ = amd_optimize(a_in, min(t4b, remaining - 60),
                                    use_soft=False, top_k=150, top_beta=3.0,
                                    c0=6.0, restart_every=5000, rng_seed=777)
            J = 2.0 * len(a_out) * fast_autocorr(a_out).max()
            if J < _global_best_score:
                _global_best_score = J
                _global_best = a_out.copy()
                print(f"  AMD polish B: J={J:.6f}")

    # Phase 5: Coordinate descent
    remaining = time_limit - (time.monotonic() - _global_t0)
    print(f"\n=== Phase 5: Coord descent ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 10:
        a_in = norm_sum1(_global_best)
        a_out, _ = coordinate_descent(a_in, t5, n_peaks=15, rng_seed=42)
        J = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if J < _global_best_score:
            _global_best_score = J
            _global_best = a_out.copy()
            print(f"  Coord descent: J={J:.6f}")

        # Second round with different seed
        remaining = time_limit - (time.monotonic() - _global_t0)
        if remaining > 10:
            t5b = min(remaining * 0.1, 60.0)
            if t5b > 5:
                a_out, _ = coordinate_descent(a_in, t5b, n_peaks=20, rng_seed=314)
                J = 2.0 * len(a_out) * fast_autocorr(a_out).max()
                if J < _global_best_score:
                    _global_best_score = J
                    _global_best = a_out.copy()
                    print(f"  Coord descent 2: J={J:.6f}")

    # Phase 6: Try different lengths
    remaining = time_limit - (time.monotonic() - _global_t0)
    print(f"\n=== Phase 6: Length search ===")
    best_n = len(_global_best)
    lengths_to_try = [best_n + 500, best_n + 1000, best_n - 500, best_n - 1000]
    lengths_to_try = [l for l in lengths_to_try if 2000 <= l <= 8000 and l != best_n]

    for new_n in lengths_to_try[:3]:
        remaining = time_limit - (time.monotonic() - _global_t0)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_scaled = interpolate_seq(_global_best, new_n)
        sc = evaluate_safe(a_scaled)
        if np.isfinite(sc) and sc < _global_best_score:
            _global_best_score = sc
            _global_best = a_scaled.copy()
            print(f"    Interpolated: J={sc:.6f}")

        t6 = min(remaining * 0.25, 150.0)
        if t6 > 15:
            a_out, _ = amd_optimize(a_scaled, t6, use_soft=True, soft_beta=250.0,
                                    top_k=60, top_beta=4.0, c0=4.0,
                                    restart_every=4000, rng_seed=555)
            J = 2.0 * len(a_out) * fast_autocorr(a_out).max()
            if J < _global_best_score:
                _global_best_score = J
                _global_best = a_out.copy()
                print(f"    AMD at n={new_n}: J={J:.6f}")

    # Phase 7: Additive subgradient on best
    remaining = time_limit - (time.monotonic() - _global_t0)
    print(f"\n=== Phase 7: Additive subgrad ===")
    t7 = min(remaining * 0.10, 80.0)
    if t7 > 10:
        a_in = norm_sum1(_global_best)
        a_out, _ = additive_subgrad_descend(a_in, t7, lr0=0.005, decay=0.99995, rng_seed=42)
        J = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if J < _global_best_score:
            _global_best_score = J
            _global_best = a_out.copy()
            print(f"  Additive subgrad: J={J:.6f}")

    # Phase 8: Final AMD with aggressive settings
    remaining = time_limit - (time.monotonic() - _global_t0)
    print(f"\n=== Phase 8: Final AMD ===")
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 15:
        a_in = norm_sum1(_global_best)
        a_out, _ = amd_optimize(a_in, t8 * 0.5, use_soft=True, soft_beta=400.0,
                                top_k=120, top_beta=3.0, c0=3.5,
                                restart_every=8000, rng_seed=2024)
        J = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if J < _global_best_score:
            _global_best_score = J
            _global_best = a_out.copy()
            print(f"  Final AMD A: J={J:.6f}")

        remaining = time_limit - (time.monotonic() - _global_t0)
        if remaining > 10:
            t8b = min(remaining * 0.3, 100.0)
            if t8b > 10:
                a_out, _ = amd_optimize(a_in, t8b, use_soft=False,
                                        top_k=200, top_beta=2.5, c0=7.0,
                                        restart_every=10000, rng_seed=3141)
                J = 2.0 * len(a_out) * fast_autocorr(a_out).max()
                if J < _global_best_score:
                    _global_best_score = J
                    _global_best = a_out.copy()
                    print(f"  Final AMD B: J={J:.6f}")

    # Phase 9: Final coordinate descent
    remaining = time_limit - (time.monotonic() - _global_t0)
    print(f"\n=== Phase 9: Final coord descent ===")
    t9 = min(remaining * 0.5, 100.0)
    if t9 > 10:
        a_in = norm_sum1(_global_best)
        a_out, _ = coordinate_descent(a_in, t9, n_peaks=20, rng_seed=4242)
        J = 2.0 * len(a_out) * fast_autocorr(a_out).max()
        if J < _global_best_score:
            _global_best_score = J
            _global_best = a_out.copy()
            print(f"  Final coord: J={J:.6f}")

    # Final validation
    if _global_best is not None:
        final_score = evaluate_safe(_global_best)
        if np.isfinite(final_score):
            _global_best_score = final_score
        print(f"\nFINAL: score={_global_best_score:.6f}, n={len(_global_best)}, "
              f"time={time.monotonic()-_global_t0:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        _global_best = make_tilted_gaussian(n_fb, mu=0.3, sigma=0.25, beta=1.5)
        _global_best_score = evaluate_safe(_global_best)
        print(f"Fallback: score={_global_best_score:.6f}")

    return [float(x) for x in _global_best]
