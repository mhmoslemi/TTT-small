import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A*A, fft_size)[:size]

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0: return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgrad_at_max(a):
    """Compute exact subgradient of max(autocorr(a)) at the argmax lag."""
    n = len(a)
    ac = fast_autocorr(a)
    k = np.argmax(ac)
    # g[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    g = np.zeros(n)
    # For i in [0, n-1], k-i is in [k-n+1, k]
    # Valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
    i_start = max(0, k - n + 1)
    i_end = min(n-1, k)
    g[i_start:i_end+1] = 2 * a[k - i_start:k - i_end + 1][::-1]
    return g, ac.max()

import numpy as np
import time


def fast_autocorr(a):
    """Compute linear autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def proj_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_objective(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max() / (s * s)


def subgrad_exact(a, k):
    """Exact subgradient of conv(a,a)[k] w.r.t. a: g[i] = 2*a[k-i] if valid."""
    n = len(a)
    g = np.zeros(n)
    i_range = np.arange(n)
    j_range = k - i_range
    valid = (j_range >= 0) & (j_range < n)
    g[valid] = 2.0 * a[j_range[valid]]
    return g


def subgrad_topk(a, top_k=10, beta=80.0):
    """Weighted subgradient from top-k peaking lags with sharp softmax."""
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    vals = ac[top_idx]
    w = np.exp(beta * (vals - vals[0]))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    i_range = np.arange(n)
    for wi, ki in zip(w, top_idx):
        ki = int(ki)
        j_range = ki - i_range
        valid = (j_range >= 0) & (j_range < n)
        g[valid] += 2.0 * wi * a[j_range[valid]]
    return g, ac.max()


def subgrad_argmax(a):
    """Subgradient at the single argmax lag."""
    n = len(a)
    ac = fast_autocorr(a)
    k = int(np.argmax(ac))
    g = subgrad_exact(a, k)
    return g, ac.max(), k


def accelerated_subgrad(a0, time_limit, verbose=True, seed=42, use_symmetric=False,

    """
    Nesterov-accelerated subgradient method on the simplex.

    else:
        a = a / s

    x_prev = a.copy()

        if sharp:

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g_normed = g / g_norm

        c_armijo = 0.01
        eta_trial = eta
        moved = False
        for _ in range(15):
            a_new = x - eta_trial * g_normed
            a_new = proj_simplex(a_new)

                a_new = proj_simplex(a_new)
            M_new = fast_autocorr(a_new).max()
            if M_new <= M_y - c_armijo * eta_trial * g_normed[0]:  # simplified Armijo
                moved = True
                break
            # Simpler: just check if it improved

                moved = True
                break
            eta_trial *= 0.5

        if not moved:
            # Use a very small step
            eta_trial = max(eta * 0.01, 1e-6)
            a_new = proj_simplex(x - eta_trial * g_normed)
            if use_symmetric:
                a_new = 0.5 * (a_new + a_new[::-1])
                a_new = proj_simplex(a_new)

        x_prev = x.copy()
        x = a_new

        M_x = fast_autocorr(x).max()
        if M_x < best_m - 1e-14:

            r = rng.random()
            if r < 0.4:
                # Perturb around best
                scale = rng.uniform(0.05, 0.2)
                pert = rng.normal(0, 1, n) * best_a * scale
                x = proj_simplex(best_a + pert)
            elif r < 0.7:

                x_rand = rng.dirichlet(np.ones(n) * 0.5)

            else:

                half = n // 2
                pert = rng.normal(0, 0.1, half)
                pert_full = np.zeros(n)
                pert_full[:half] = pert
                pert_full[half:] = pert[::-1]
                x = proj_simplex(best_a + pert_full * best_a)
            if use_symmetric:
                x = 0.5 * (x + x[::-1])
                x = proj_simplex(x)
            x_prev = x.copy()
            eta = 2.0 / np.sqrt(n)

            if verbose:
                elapsed = time.monotonic() - t0

def coordinate_descent(a0, time_limit, verbose=True, seed=123, use_symmetric=False):
    """
    Coordinate descent: identify peak-contributing pairs and move mass to
    low-sensitivity positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Get top contributing lags
        n_top = min(15, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_top]

        improvement = False

        for lag in top_lags[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(lag)
            # Valid j range for this lag
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue

            # Find the pair with maximum product
            js = np.arange(j_start, j_end + 1)
            products = a[js] * a[k - js]
            if products.max() < 1e-15:
                continue
            src_idx = js[np.argmax(products)]

            # Find low-sensitivity destination
            # Sensitivity of position d: sum over top lags of a[lag - d] (if valid)
            sens = np.zeros(n)
            for k2 in top_lags[:10]:
                k2 = int(k2)
                d_start = max(0, k2 - n + 1)
                d_end = min(n - 1, k2)
                if d_start <= d_end:
                    ds = np.arange(d_start, d_end + 1)
                    sens[ds] += a[k2 - ds]

            # Candidate destinations: low sensitivity, far from source
            dist = np.abs(np.arange(n) - src_idx).astype(float)
            scores = sens + 0.1 * (1.0 - dist / n)  # prefer low sens + far
            candidates = np.argsort(scores)[:n // 4]

            best_improve = None
            best_trial_m = best_m
            for dst in candidates:
                if dst == src_idx:
                    continue
                for frac in [0.03, 0.01, 0.05, 0.005, 0.02, 0.08]:
                    amt = a[src_idx] * frac
                    if amt < 1e-12:
                        break
                    a_trial = a.copy()
                    a_trial[src_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    if use_symmetric:
                        a_trial = 0.5 * (a_trial + a_trial[::-1])
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_trial_m - 1e-13:
                        best_trial_m = M_trial
                        best_improve = (src_idx, dst, amt, a_trial.copy())

            if best_improve is not None:
                src_i, dst_i, amt_i, a_new = best_improve
                a = a_new
                if best_trial_m < best_m:
                    best_m = best_trial_m
                    best_a = a.copy()
                improvement = True
                break

        if not improvement:
            stall += 1
            # Random kick every 20 stalls
            if stall % 20 == 0:
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 != i2 and a[i1] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    a[i1] -= a[i1] * frac
                    a[i2] += a[i1] * frac / max(a[i1], 1e-15) * a[i1] * frac
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    if use_symmetric:
                        a = 0.5 * (a + a[::-1])
                        a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        else:
            stall = 0

        if verbose and stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_flat_spectrum_init(n, seed=0, offset=0.0):
    """Initialize from flat-spectrum random phase sequence."""
    rng = np.random.default_rng(seed)
    # Random phases with conjugate symmetry
    phases = rng.uniform(0, 2 * np.pi, n)
    # Enforce conjugate symmetry: A[k] = conj(A[n-k])
    A = np.exp(1j * phases)
    A[0] = 1.0  # DC
    if n > 1:
        for k in range(1, (n // 2) + 1):
            if n - k < n:
                A[n - k] = np.conj(A[k])
    # IFFT to get real sequence
    a = np.real(np.fft.ifft(A)) * n
    # Add offset to ensure positivity
    a = a - a.min() + max(offset, 0.1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_structured_init(n, seed=0, n_freqs=5):
    """Structured initialization with multiple sinusoidal components."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n) * 0.5
    for f in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, 0.15) / np.sqrt(f)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_cosine_init(n, k=1):
    """Raised cosine-like initialization."""
    x = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + np.cos(2 * np.pi * k * x)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-0.5 * ((x - center) / sigma_frac) ** 2)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def symmetrize(a):
    """Make symmetric: a[i] = a[n-1-i]."""
    return 0.5 * (a + a[::-1])


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-phase optimization with diverse initializations,
    accelerated subgradient, and coordinate descent polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        # Evaluate using the actual evaluate_sequence
        a_list = [float(x) for x in a_norm]
        try:
            sc = float(evaluate_sequence(a_list))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_norm = parent / parent.sum()
            p_list = [float(x) for x in p_norm]
            p_score = float(evaluate_sequence(p_list))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = p_norm.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_structured_init(n_base, seed=1)
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Build diverse initializations ===
    inits = []
    labels = []

    if parent is not None:
        inits.append(parent / parent.sum())
        labels.append("parent")

    inits.append(make_flat_spectrum_init(n0, seed=7))
    labels.append("flat_spec")

    inits.append(make_flat_spectrum_init(n0, seed=42))
    labels.append("flat_spec2")

    inits.append(make_structured_init(n0, seed=13, n_freqs=3))
    labels.append("low_freq")

    inits.append(make_structured_init(n0, seed=99, n_freqs=8))
    labels.append("high_freq")

    inits.append(make_cosine_init(n0, k=1))
    labels.append("cos1")

    inits.append(make_cosine_init(n0, k=2))
    labels.append("cos2")

    inits.append(make_gaussian_init(n0, sigma_frac=0.25))
    labels.append("gauss")

    # Symmetric versions
    for i, (init, lbl) in enumerate(list(zip(inits[:3], labels[:3]))):
        inits.append(symmetrize(init))
        labels.append(f"sym_{lbl}")

    # De-duplicate
    seen_sigs = set()
    unique_inits = []
    unique_labels = []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n0 // 40)], 3))
        if sig not in seen_sigs:
            seen_sigs.add(sig)
            unique_inits.append(init)
            unique_labels.append(lbl)
    inits = unique_inits
    labels = unique_labels

    # === Phase 1: Accelerated subgradient on diverse inits ===
    print(f"\n=== Phase 1: Accelerated Subgradient ({len(inits)} inits) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = remaining * 0.45
    n_inits = len(inits)
    t_per_init = max(15.0, t_phase1 / n_inits)

    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed_so_far = time.monotonic() - t_global
        t_avail = min(t_per_init, time_budget - 60 - elapsed_so_far)
        if t_avail < 10:
            break
        sym = lbl.startswith("sym_")
        print(f"  AS init '{lbl}' (t={t_avail:.0f}s)...")
        a_out, _ = accelerated_subgrad(init, t_avail, verbose=False,
                                        seed=100 + idx, use_symmetric=sym,
                                        sharp=True, max_stall=150)
        try_update(a_out, f"as_{lbl}")

    # === Phase 2: Second round on best with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: AS Round 2 (t_avail={remaining:.0f}s) ===")
    if remaining > 60:
        t2 = min(remaining * 0.2, 300.0)
        a_in = best_seq.copy()
        # Try both sharp and top-k
        a_out1, _ = accelerated_subgrad(a_in, t2 * 0.5, verbose=False,
                                         seed=200, sharp=True, max_stall=200)
        try_update(a_out1, "as_r2_sharp")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_out2, _ = accelerated_subgrad(a_in, min(remaining * 0.3, 150.0),
                                             verbose=False, seed=300, sharp=False,
                                             max_stall=200)
            try_update(a_out2, "as_r2_topk")

    # === Phase 3: Symmetric AS on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Symmetric AS ===")
    if remaining > 40:
        t3 = min(remaining * 0.15, 200.0)
        a_sym = symmetrize(best_seq)
        a_out, _ = accelerated_subgrad(a_sym, t3, verbose=False,
                                        seed=400, use_symmetric=True,
                                        sharp=True, max_stall=150)
        try_update(a_out, "as_sym")

    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coordinate Descent ===")
    if remaining > 30:
        t4 = min(remaining * 0.2, 250.0)
        a_out, _ = coordinate_descent(best_seq, t4, verbose=False, seed=500)
        try_update(a_out, "cd")

    # === Phase 5: AS on coordinate descent result ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: AS after CD ===")
    if remaining > 30:
        t5 = min(remaining * 0.2, 200.0)
        a_out, _ = accelerated_subgrad(best_seq, t5, verbose=False,
                                        seed=600, sharp=True, max_stall=200)
        try_update(a_out, "as_after_cd")

    # === Phase 6: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    new_n_try = n_best + 400
    if remaining > 80 and new_n_try <= 5000:
        print(f"\n=== Phase 6: Length {new_n_try} ===")
        a_interp = interpolate_to_length(best_seq, new_n_try)
        t6a = min(remaining * 0.3, 180.0)
        a_out, _ = accelerated_subgrad(a_interp, t6a, verbose=False,
                                        seed=700, sharp=True, max_stall=150)
        try_update(a_out, f"as_n{new_n_try}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.3, 120.0)
            a_out, _ = coordinate_descent(best_seq if len(best_seq) == n_best else a_out,
                                          t6b, verbose=False, seed=800)
            try_update(a_out, f"cd_n{new_n_try}")

    # === Phase 7: Final intensive AS with many restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final Intensive AS ===")
    if remaining > 20:
        t7 = remaining * 0.6
        a_out, _ = accelerated_subgrad(best_seq, t7, verbose=False,
                                        seed=900, sharp=True, max_stall=100)
        try_update(a_out, "as_final")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 8: Final CD ===")
        t8 = remaining * 0.7
        a_out, _ = coordinate_descent(best_seq, t8, verbose=True, seed=1000)
        try_update(a_out, "cd_final")

    # === Final validation ===
    if best_seq is not None:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_structured_init(n_fb, seed=1)
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
