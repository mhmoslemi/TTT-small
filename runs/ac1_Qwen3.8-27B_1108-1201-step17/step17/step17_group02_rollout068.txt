def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_lp_gradient(a, p):
    n = len(a)
    b = fast_conv(a)
    if p <= 1.0:
        w = np.ones_like(b)
    else:
        w = np.power(np.maximum(b, 0.0), p - 1)
    fft_size = 1 << ((len(w) + n - 1) - 1).bit_length()
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    W = np.fft.fft(w_pad)
    A2 = np.fft.fft(a_pad)
    R = np.fft.ifft(W * np.conj(A2)).real[:n]
    return 2.0 * p * R

def fft_projected_grad(a0, time_limit, p_schedule=None, verbose=True, rng_seed=0):

def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def amd_optimize(a0, time_limit, top_k=50, beta=5.0, c0=4.0, restart_every=5000,
                 adaptive_beta=True, verbose=False, rng_seed=42):

    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        ac = fast_conv(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        x_prev = x
        x = a_new
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        if adaptive_beta and stall_count > restart_every // 2:
            cur_beta = max(cur_beta * 0.92, 2.0)
            cur_top_k = min(cur_top_k + 3, len(ac))
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.025, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 8, 150)
            cur_beta = max(cur_beta - 1.0, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")

def isotonic_decrease(a):
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    weights = np.ones(n)
    block_start = 0
    for i in range(1, n):
        if a[i] > a[block_start] + 1e-15:
            block_end = i
            avg = np.sum(a[block_start:block_end+1]) / (block_end - block_start + 1)
            a[block_start:block_end+1] = avg
            j = block_start - 1
            while j >= 0 and a[j] > avg + 1e-15:
                block_start = j
                j -= 1
                block_start = j + 1
                block_end = i
                avg = np.sum(a[block_start:block_end+1]) / (block_end - block_start + 1)
                a[block_start:block_end+1] = avg
            block_start = i
        else:
            block_start = i
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_monotone_decreasing(n, sharpness=1.0):

def make_u_shape(n, sharpness=2.0):

def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels
        n_inits = len(inits)
        t_per = max(15.0, t_phase1 / n_inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  PGD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = fft_projected_grad(init, t_avail, verbose=False, rng_seed=42 + idx)
            try_update(a_out, f"pgd_{lbl}")
    # Phase 2: LP-FW on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW (n={n_best}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 40:
        a_in = norm_sum1(best_seq)
        top_K_init = min(200, 2 * n_best - 1)
        top_K_max = min(4000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_phase2")
    # Phase 3: AMD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: AMD polish (n={n_best}) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t3, top_k=80, beta=7.0, c0=4.0,
                                 restart_every=6000, adaptive_beta=True, verbose=False, rng_seed=200)
        try_update(a_out, "amd_p3")
    # Phase 4: LP-FW round 2 with more constraints
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW round 2 (n={n_best}) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        top_K_init = min(500, 2 * n_best - 1)
        top_K_max = min(6000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K_init=top_K_init, top_K_max=top_K_max)
        try_update(a_out, "lp_phase4")
    # Phase 5: Peak pair reduction
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Peak-pair (n={n_best}) ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_pair_reduction(a_in, t5, verbose=False, rng_seed=77)
        try_update(a_out, "peakpair")
    # Phase 6: Isotonic projection + re-optimize
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Isotonic + re-optimize ===")
    if remaining > 60:
        a_iso = isotonic_decrease(best_seq)
        sc_iso = evaluate_safe(a_iso)
        if np.isfinite(sc_iso) and sc_iso < best_score:
            best_score = sc_iso
            best_seq = a_iso.copy()
            print(f"  Isotonic improved: {sc_iso:.6f}")
        t6a = min(remaining * 0.30, 200.0)
        if t6a > 30:
            a_out, _ = lp_frank_wolfe(a_iso, t6a, solver, verbose=True,
                                       top_K_init=300, top_K_max=min(5000, 2*n_best-1))
            try_update(a_out, "lp_iso")
    # Phase 7: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")
    trial_lengths = []
    for delta in [500, -500, 1000, -1000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 100]
    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t7a = min(remaining * 0.35, 250.0)
        if t7a > 30:
            a_out, _ = fft_projected_grad(a_long, t7a * 0.4, verbose=False, rng_seed=new_n % 1000)
            try_update(a_out, f"pgd_n{new_n}")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 40:
                t7b = min(remaining * 0.5, 200.0)
                if t7b > 30:
                    a_in = norm_sum1(best_seq) if len(best_seq) == new_n else a_long
                    a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False,
                                               top_K_init=300, top_K_max=min(4000, 2*new_n-1))
                    try_update(a_out, f"lp_n{new_n}")
    # Phase 8: Final LP-FW polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP polish (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.7, 200.0)
        if t8 > 20:
            a_in = norm_sum1(best_seq)
            top_K_init = min(1000, 2 * n_best - 1)
            top_K_max = min(8000, 2 * n_best - 1)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True,
                                       top_K_init=top_K_init, top_K_max=top_K_max)
            try_update(a_out, "lp_final")
    # Phase 9: Final AMD + peak pair
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final polish ===")
        t9 = min(remaining * 0.5, 60.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = amd_optimize(a_in, t9, top_k=100, beta=8.0, c0=3.5,
                                     restart_every=5000, verbose=False, rng_seed=555)
            try_update(a_out, "amd_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_pair_reduction(a_in, min(remaining * 0.6, 30.0), verbose=False, rng_seed=999)
            try_update(a_out, "pp_final")
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_monotone_decreasing(n_fb, 1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    return [float(x) for x in best_seq]
