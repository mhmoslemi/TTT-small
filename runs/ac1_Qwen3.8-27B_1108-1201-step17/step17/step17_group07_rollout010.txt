"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient with mirror descent (uses all lags, not just top-k)
2. Adaptive temperature annealing for smooth-max approximation
3. LP Remez with many active constraints for polishing
4. Nesterov acceleration in the mirror map
5. Multi-temperature restarts to escape local minima
6. Multi-length exploration
Key insight: The log-sum-exp gradient naturally weights each lag by its
contribution to the smoothed max, giving a proper (non-sub)gradient that
captures more structural information than top-k subgradient methods.
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_lse_grad(a, mu):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Returns (grad, smoothed_value)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Softmax weights over lags
    log_w = (b - b_max) / mu
    log_w -= log_w.max()  # numerical stability
    exp_w = np.exp(log_w)
    w = exp_w / exp_w.sum()
    
    # Smoothed max value
    smooth_val = b_max + mu * np.log(exp_w.sum())
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (for valid indices)
    # This is 2 * cross_correlation(w, a) computed via FFT
    a_rev = a[::-1]
    conv_size = len(w) + n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Ar = np.fft.rfft(np.asarray(a_rev, dtype=np.float64), fft_size)
    conv_result = np.fft.irfft(W * Ar, fft_size)[:conv_size]
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    
    return grad, smooth_val, b

def lse_mirror_descent(a0, time_limit, mu_start=1.0, mu_end=0.01, verbose=False):
    """Log-sum-exp smoothed mirror descent with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    mu = mu_start
    eta = 0.5
    stall = 0
    max_stall = 2000
    total_iters = 0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta = (t - 1.0) / (t + 1.5)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 1e-30)
        y = y / y.sum()
        
        # Compute gradient at y
        grad, smooth_val, b_y = compute_lse_grad(y, mu)
        
        # Project gradient: remove mean component (tangent to simplex)
        grad_mean = (grad * y).sum()
        grad_tangent = grad - grad_mean
        
        # Mirror descent step: a_new[j] = y[j] * exp(-eta * grad_tangent[j]) / Z
        log_shift = -eta * grad_tangent
        log_shift -= log_shift.max()  # stability
        weights = np.exp(log_shift)
        a_new = y * weights
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        total_iters += 1
        
        # Evaluate
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
            # Slightly increase eta on improvement
            eta = min(2.0, eta * 1.002)
        else:
            stall += 1
            # Backtrack: decrease eta
            eta = max(0.01, eta * 0.95)
        
        # Anneal temperature
        progress = min(1.0, t / max(1, int(time_limit * 50)))
        mu = mu_start * (mu_end / mu_start) ** progress
        
        # Stall detection and restart
        if stall > max_stall:
            stall = 0
            max_stall = max(300, int(max_stall * 0.7))
            rng = np.random.default_rng(t)
            strategy = int(rng.integers(5))
            if strategy == 0:
                # Small perturbation
                x = best_a * (1 + rng.normal(0, 0.01, n))
            elif strategy == 1:
                # Random subset perturbation
                x = best_a.copy()
                idx = rng.choice(n, size=min(n // 4, 500), replace=False)
                x[idx] *= rng.uniform(0.5, 1.5, len(idx))
            elif strategy == 2:
                # Temperature bump
                mu = mu * 3.0
                x = best_a * (1 + rng.normal(0, 0.005, n))
            elif strategy == 3:
                # Shift perturbation
                shift = int(rng.integers(max(2, n // 50)))
                x = np.roll(best_a, shift)
            else:
                # Reflection-like
                x = best_a.copy()
                x[:n//2] = best_a[:n//2] * rng.uniform(0.7, 1.3, n//2)
            
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            eta = 0.3
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE-MD t={t} ({elapsed:.1f}s): J={J:.8f}, mu={mu:.4f}, eta={eta:.4f}, stall={stall}")
    
    return best_a, best_m

def lp_remez_step(a, max_cons=80, r=0.05, solver=None):
    """Single LP Remez step with linearized constraints."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Select active constraints: top lags near the maximum
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            # Linearization: b[k](a') ≈ 2*(a[k-idx] @ a'[idx]) - b[k](a)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    # Trust region
    constraints.append(a_var <= a * (1.0 + r))
    constraints.append(a_var >= a * (1.0 - r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_polish(a0, time_limit, solver=None, r_init=0.05, verbose=False):
    """LP Remez polishing with adaptive trust region."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    
    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand = lp_remez_step(a, max_cons=60, r=r, solver=solver)
        
        accepted = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_trial = (1 - frac) * a + frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.15, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.005, r * 0.8)
            if no_improve > 15:
                rng = np.random.default_rng(t)
                idx = rng.choice(n, size=min(n // 5, 300), replace=False)
                a[idx] *= rng.uniform(0.8, 1.2, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = r_init * 0.5
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")
    
    return best_a, best_m

def high_precision_grad_step(a, mu, n_steps=5, verbose=False):
    """Multiple small gradient steps at a given temperature for fine-tuning."""
    n = len(a)
    a_curr = a.copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    eta = 0.1
    
    for step in range(n_steps):
        grad, smooth_val, b = compute_lse_grad(a_curr, mu)
        grad_mean = (grad * a_curr).sum()
        grad_t = grad - grad_mean
        
        # Try multiple step sizes
        best_local_m = best_m
        best_local_a = best_a.copy()
        
        for trial_eta in [eta * 0.5, eta, eta * 2.0]:
            log_shift = -trial_eta * grad_t
            log_shift -= log_shift.max()
            weights = np.exp(log_shift)
            a_trial = a_curr * weights
            a_trial = np.maximum(a_trial, 1e-30)
            a_trial = a_trial / a_trial.sum()
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_local_m:
                best_local_m = m_trial
                best_local_a = a_trial.copy()
                eta = trial_eta
        
        a_curr = best_local_a
        if best_local_m < best_m:
            best_m = best_local_m
            best_a = a_curr.copy()
    
    return best_a, best_m

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        sigma = 0.25
        a = np.exp(-(x - 0.5) ** 2 / (2 * sigma ** 2))
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sinc_like':
        a = np.sinc(2.0 * (x - 0.5)) ** 2
        a = np.maximum(a, 0.0)
    elif kind == 'battlette':
        # Bartlett-Hanning (raised cosine with narrower main lobe)
        a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.0)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)
    new_x = np.linspace(0.0, 1.0, new_n + 1)
    f_interp = np.interp(new_x, old_x, np.concatenate([f_norm, [0.0]]))
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
