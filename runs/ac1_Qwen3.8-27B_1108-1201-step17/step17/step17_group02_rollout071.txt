"""
Search for a nonnegative sequence minimizing
    2*n*max(conv(a,a)) / (sum(a))^2
by combining:
  - smooth low-dimensional Fourier/softmax initialization,
  - full-coordinate log-softmax Adam with log-sum-exp smoothing,
  - fast coordinate/local mass-transfer polishing using exact convolution updates,
  - LP-based Frank-Wolfe style directions when a solver is available.
"""

_LP_SOLVER = None


def _elapsed():
    if _RUN_START <= 0:
        return 0.0
    return time.monotonic() - _RUN_START


def _remaining():
    return max(0.0, _RUN_BUDGET - _elapsed())


def _norm_sum1(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.array([1.0])
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _fast_C(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    C = np.fft.irfft(A * A, size)[: 2 * n - 1]
    C = np.where(np.isfinite(C), C, 0.0)
    if C.size and float(C.min()) < -1e-12:
        C = np.maximum(C, 0.0)
    return C


def _cross_valid(r, a):
    """
    Returns c[j] = sum_p r[j+p] * a[p], j=0..n-1,
    where r has length 2n-1 and a has length n.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    r = np.asarray(r, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if r.size != 2 * n - 1:
        r = r[: 2 * n - 1]
    need = 3 * n - 2
    size = _next_pow2(need)
    R = np.zeros(size, dtype=np.float64)
    R[: 2 * n - 1] = r[::-1]
    A = np.zeros(size, dtype=np.float64)
    A[:n] = a
    conv = np.fft.irfft(np.fft.rfft(R) * np.fft.rfft(A), size)[:need]
    idx = np.arange(n)
    return conv[(2 * n - 2) - idx]


def _lse(C, tau):
    C = np.asarray(C, dtype=np.float64).ravel()
    if C.size == 0:
        return 0.0, np.array([], dtype=np.float64), 0.0
    M = float(C.max())
    if not np.isfinite(M):
        M = 0.0
        C = np.zeros_like(C)
    diff = tau * (C - M)
    w = np.exp(np.maximum(diff, -745.0))
    Z = float(w.sum())
    if Z <= 0.0 or not np.isfinite(Z):
        w = np.zeros_like(C)
        w[int(np.argmax(C))] = 1.0
        Z = 1.0
    r = w / Z
    F = M + float(np.log(Z)) / tau
    return M, r, F


def _softmax(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.ones(n, dtype=np.float64) / float(n)
    v = v - float(v.max())
    e = np.exp(v)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _project_simplex(z):

def _interpolate_sequence(a, new_n):

def _get_solver():
    global _LP_SOLVER
    if cp is None:
        return None
    if _LP_SOLVER is not None:
        return _LP_SOLVER

    candidates = []
    for name in ["CLARABEL", "PDLP", "ECOS", "SCS", "MOSEK"]:
        if hasattr(cp, name):
            candidates.append(getattr(cp, name))

    for solver in candidates:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=solver, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):

def _fourier_basis(n, m):
    x = np.linspace(0.0, 1.0, n)
    cols = []
    fmax = max(0, m // 2)
    for f in range(0, fmax + 1):
        if f == 0:
            cols.append(np.ones(n, dtype=np.float64) / np.sqrt(float(n)))
        else:
            c = np.cos(2.0 * np.pi * f * x) / np.sqrt(float(n) / 2.0)
            s = np.sin(2.0 * np.pi * f * x) / np.sqrt(float(n) / 2.0)
            cols.append(c)
            cols.append(s)
    if len(cols) == 0:
        B = np.ones((n, 1), dtype=np.float64)
    else:
        B = np.column_stack(cols)
    if B.shape[1] < m:
        pad = np.zeros((n, m - B.shape[1]), dtype=np.float64)
        B = np.column_stack((B, pad))
    return B[:, :m]


def _move_apply(C, a, i, j, delta, sign):
    if i == j or delta == 0.0:
        return
    n = a.size
    twd = 2.0 * sign * delta
    C[j : j + n] += twd * a
    C[i : i + n] -= twd * a
    q = sign * delta * delta
    C[2 * i] += q
    C[2 * j] += q
    C[i + j] -= 2.0 * q


def _local_search(a0, time_limit, verbose=False):
    t0 = time.monotonic()
    a = _norm_sum1(a0)
    n = a.size
    if n <= 1:
        return a, 0.0

    C = _fast_C(a)
    best_a = a.copy()
    best_m = float(C.max())

    rng = np.random.default_rng(987654321)
    stall = 0
    moves = 0

    top_src = np.argsort(a * a)[::-1][:14]
    top_dest = np.argsort(a)[:14]
    sens = np.ones(n, dtype=np.float64)

    while _remaining() > 20 and (time.monotonic() - t0) < time_limit:
        if moves % 30 == 0:
            Macc = float(C.max())
            if Macc < best_m:
                best_m = Macc
                best_a = a.copy()
                stall = 0

            tau = 30000.0
            Macc, r, _ = _lse(C, tau)
            sens = 2.0 * _cross_valid(r, a)
            score = a * np.maximum(sens, 0.0)
            top_src = np.argsort(score)[::-1][:12]
            top_dest = np.argsort(sens)[:12]
            if n > 24:
                rs = rng.integers(0, n, 8)
                rd = rng.integers(0, n, 8)
                top_src = np.unique(np.concatenate((top_src, rs)))[:14]
                top_dest = np.unique(np.concatenate((top_dest, rd)))[:14]

        improved = False
        for ii in top_src[:10]:
            i = int(ii)
            ai = float(a[i])
            if ai < 1e-8:
                continue
            for jj in top_dest[:10]:
                j = int(jj)
                if i == j:
                    continue
                if sens[i] <= sens[j] + 1e-15:
                    continue
                for frac in (0.01, 0.005, 0.02):
                    delta = frac * ai
                    if delta <= 0.0:
                        continue
                    _move_apply(C, a, i, j, delta, 1.0)
                    m = float(C.max())
                    if m < best_m - 1e-16:
                        a[i] -= delta
                        a[j] += delta
                        best_m = m
                        best_a = a.copy()
                        stall = 0
                        moves += 1
                        improved = True
                        break
                    else:
                        _move_apply(C, a, i, j, delta, -1.0)
                if improved:
                    break
            if improved:
                break

                if i == j or float(a[i]) < 1e-8:
                    continue
                frac_opts = (0.002, 0.005, 0.01, 0.02)
                delta = float(frac_opts[int(rng.integers(len(frac_opts)))]) * float(a[i])
                _move_apply(C, a, i, j, delta, 1.0)
                m = float(C.max())
                if m < best_m - 1e-16:
                    a[i] -= delta
                    a[j] += delta
                    best_m = m
                    best_a = a.copy()
                    stall = 0
                    moves += 100
                    improved = True
                    break

            if not improved:

                    pert = rng.normal(0.0, sigma, n) * a
                    a2 = np.maximum(a + pert, 0.0)
                    s2 = float(a2.sum())
                    if s2 > 1e-15:

            el = time.monotonic() - t0
            print(f"    local ({el:.0f}s): M={best_m:.10f} J={2.0*n*best_m:.6f} stall={stall}")

def _fourier_adam(n, m, time_limit, init_a=None, rand_sigma=0.8, tau_start=1000.0, lr=0.55, verbose=False):
    t0 = time.monotonic()
    B = _fourier_basis(n, m)
    mB = B.shape[1]
    rng = np.random.default_rng(1729 + 101 * n + 7 * m)

    if init_a is not None:
        a0 = _norm_sum1(init_a)
        local_best_a = a0.copy()
        local_best_m = float(_fast_C(a0).max())
        loga = np.log(np.maximum(a0, 1e-12))
        v = B.T @ loga
        v = np.where(np.isfinite(v), v, 0.0)
        v -= float(np.mean(v))
    else:
        local_best_a = _norm_sum1(np.ones(n, dtype=np.float64))
        local_best_m = float(_fast_C(local_best_a).max())
        v = rng.normal(0.0, rand_sigma, mB)

    best_v = v.copy()
    vhat = np.zeros(mB, dtype=np.float64)
    mhat = np.zeros(mB, dtype=np.float64)
    b1 = 0.9
    b2 = 0.999
    t = 0
    lr_cur = lr
    low_count = 0

    while _remaining() > 30 and (time.monotonic() - t0) < time_limit:
        t += 1
        v = v - float(v.max())
        a = _softmax(B @ v)
        C = _fast_C(a)
        tau = tau_start * min(50.0, 1.0 + 0.01 * t)
        M, r, Fsm = _lse(C, tau)

        if M < local_best_m:
            local_best_m = M
            local_best_a = a.copy()
            best_v = v.copy()
            low_count = 0

        grad_a = 2.0 * _cross_valid(r, a)
        ga_mean = float(np.dot(grad_a, a))
        gs = grad_a - a * ga_mean
        grad = B.T @ gs

        if not np.all(np.isfinite(grad)):
            v = best_v.copy()
            vhat.fill(0.0)
            mhat.fill(0.0)
            t = 1
            lr_cur = lr
            continue

        mhat = b1 * mhat + (1.0 - b1) * grad * grad
        vhat = b2 * vhat + (1.0 - b2) * grad
        denom = np.sqrt(mhat) / (1.0 - b1 ** t) + 1e-8
        dv = lr_cur * (vhat / (1.0 - b2 ** t)) / denom

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.125):
            v_trial = v - mult * dv
            a_trial = _softmax(B @ v_trial)
            C_trial = _fast_C(a_trial)
            M_tr, r_tr, F_tr = _lse(C_trial, tau)
            if F_tr <= Fsm + 1e-16:
                v = v_trial
                accepted = True
                break

        if not accepted:
            lr_cur = max(1e-3, lr_cur * 0.5)
            low_count += 1
            if low_count > 12:
                v = best_v + rng.normal(0.0, 0.35, mB)
                vhat.fill(0.0)
                mhat.fill(0.0)
                t = 1
                lr_cur = lr
                low_count = 0
        else:
            low_count = max(0, low_count - 1)

        if verbose and (t % 200 == 0):
            el = time.monotonic() - t0
            print(f"    fourier m={m} ({el:.0f}s): M={local_best_m:.10f}")

    return local_best_a, local_best_m


def _fourier_progressive(a0, time_limit, verbose=False):

def _select_lags(C, K, n):
    m = C.size
    K = max(1, int(K))
    if K >= m:
        return np.arange(m, dtype=int)

    base = [
        n - 1,
        0,
        1,
        2,
        3,
        n - 2,
        n - 3,
        n,
        n + 1,
        2 * n - 4,
        2 * n - 3,
        2 * n - 2,
    ]
    extras = []
    seen = set()
    for k in base:
        if 0 <= k < m and k not in seen:
            seen.add(k)
            extras.append(k)

    order = np.argsort(C)[::-1]
    selected = []
    for k in extras:
        if len(selected) >= K:
            break
        selected.append(int(k))

    for k in order:
        if len(selected) >= K:
            break
        ki = int(k)
        if ki not in seen:
            selected.append(ki)
            seen.add(ki)

    for k in order:
        if len(selected) >= K:
            break
        ki = int(k)
        if ki not in seen:
            selected.append(ki)
            seen.add(ki)

    return np.array(selected[:K], dtype=int)

        if n <= 1200:
            Kmax = min(2 * n - 1, 2200)
        elif n <= 2500:
            Kmax = 1300
        elif n <= 5000:
            Kmax = 950
        else:
            Kmax = 700

    if "ECOS" in solver_s or "SCS" in solver_s:
        Kmax = min(Kmax, 650)

    Kmax = min(Kmax, 2 * n - 1)
    K = max(80, min(280, Kmax, 2 * n - 1))

    y = np.maximum(_norm_sum1(a0), 1e-12)
    y = y / float(y.sum())
    best_a = y.copy()
    best_m = float(_fast_C(y).max())

    y_old = y.copy()
    x = None
    t = 0
    fail_count = 0

    while _remaining() > 30 and (time.monotonic() - t0) < time_limit:
        C = _fast_C(y)
        M = float(C.max())
        if M < best_m:
            best_m = M
            best_a = y.copy()

        idx = _select_lags(C, K, n)
        Ksel = idx.size
        if Ksel == 0:
            break

        A = np.zeros((Ksel, n), dtype=np.float64)
        for row, k in enumerate(idx):
            kk = int(k)
            lo = max(0, kk - n + 1)
            hi = min(kk, n - 1)
            if hi >= lo:
                A[row, lo : hi + 1] = y[lo : hi + 1]

        b = cp.Variable(n, nonneg=True)
        try:
            if variant == "maxsum":
                constraints = [
                    A @ b <= M + 1e-15,
                    cp.sum(b) <= 100.0,
                ]
                prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
            else:
                mu = cp.Variable()
                constraints = [
                    A @ b <= mu,
                    cp.sum(b) == 1.0,
                    mu >= 0.0,
                ]
                prob = cp.Problem(cp.Minimize(mu), constraints)

            ts = time.monotonic()
            prob.solve(solver=solver, verbose=False)
            dt = time.monotonic() - ts
        except Exception:
            dt = 10.0
            fail_count += 1
            if fail_count > 4:
                break
            continue

        if prob.status not in ("optimal", "optimal_inaccurate"):
            fail_count += 1
            if fail_count > 5:
                break
            if dt > 8.0:
                K = max(60, K - 80)
            continue

        fail_count = 0
        b_val = b.value
        if b_val is None:
            continue
        b_val = np.asarray(b_val, dtype=np.float64).ravel()
        b_val = np.where(np.isfinite(b_val), b_val, 0.0)
        b_val = np.maximum(b_val, 0.0)
        b_sum = float(b_val.sum())

        if b_sum < 1e-9:
            direction = y.copy()
        else:
            direction = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * direction
        y_new = _project_simplex_floor(y_new, 1e-12)

        if x is None:
            x_new = y_new.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = _project_simplex_floor(x_new, 1e-12)

        y_old = y
        y = y_new
        x = x_new

        My = float(_fast_C(y).max())
        if My < best_m:
            best_m = My
            best_a = y.copy()

        Mx = float(_fast_C(x).max())
        if Mx < best_m:
            best_m = Mx
            best_a = x.copy()

        if verbose and (t % 5 == 0):
            el = time.monotonic() - t0
            print(f"    LP {variant} iter={t} ({el:.0f}s): J={2.0*n*best_m:.6f} K={K} dt={dt:.2f}")

        if t % 4 == 0:
            if dt < 2.5:
                K = min(Kmax, K + 80)
            elif dt > 5.0:
                K = max(60, K - 80)
            if t > 8 and dt > 25.0:
                break

        t += 1

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    global _RUN_START, _RUN_BUDGET, _BEST_SEQ, _BEST_SCORE, _LP_SOLVER

    _RUN_START = time.monotonic()
    _RUN_BUDGET = min(1795.0, max(80.0, float(budget_s) - 10.0))
    _BEST_SEQ = None
    _BEST_SCORE = float("inf")
    _LP_SOLVER = None

    solver = _get_solver()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.size > 0 and float(p.sum()) > 1e-10:
            parent = p / float(p.sum())
            print(f"Loaded parent length={p.size}")
    except Exception:
        parent = None

    if parent is None:
        n0 = 4200
        x = np.linspace(0.0, 1.0, n0)
        parent = 0.25 + np.cos(np.pi * x) ** 2
        parent = _norm_sum1(parent)
        print(f"Using generated fallback length={n0}")

    _update_best(parent, "parent")

    # Phase 0: quick local polish of the starting point.
    if _remaining() > 220:
        tl = min(70.0, _remaining() * 0.05)
        if tl > 25:
            a_out, _ = _local_search(_BEST_SEQ, tl, verbose=False)
            _update_best(a_out, "local0")

    # Phase 1: low-dimensional smooth Fourier search.
    if _remaining() > 300:
        tl = min(300.0, _remaining() * 0.20)
        a_out, _ = _fourier_progressive(_BEST_SEQ, tl, verbose=False)
        _update_best(a_out, "fourier_prog")

    # Phase 2: full-coordinate smoothed Adam.
    if _remaining() > 260:
        tl = min(220.0, _remaining() * 0.14)
        a_out, _ = _full_adam(_BEST_SEQ, tl, tau_start=1800.0, lr=0.15, verbose=False)
        _update_best(a_out, "adam")

    # Phase 3: LP/Frank-Wolfe directions.
    if solver is not None and _remaining() > 300:
        n_best = len(_BEST_SEQ)
        if n_best <= 5500:
            tl = min(470.0, _remaining() * 0.35)
            if tl > 60:
                a_out, _ = _lp_frank_wolfe(_BEST_SEQ, tl * 0.55, "maxsum", Kmax=0, verbose=True)
                _update_best(a_out, "lp_maxsum")
                if _remaining() > 60:
                    a_out, _ = _lp_frank_wolfe(_BEST_SEQ, min(tl * 0.45, _remaining()), "mincross", Kmax=0, verbose=True)
                    _update_best(a_out, "lp_mincross")

    # Phase 4: local coordinate polishing.
    if _remaining() > 130:
        tl = min(220.0, max(35.0, _remaining() - 120.0))
        if tl > 20:
            a_out, _ = _local_search(_BEST_SEQ, tl, verbose=True)
            _update_best(a_out, "local")

    # Phase 5: modest length exploration.
    if _remaining() > 260:
        n_best = len(_BEST_SEQ)
        new_n = int(min(5800, max(n_best + 250, n_best * 1.06)))
        if new_n != n_best:
            a_long = _interpolate_sequence(_BEST_SEQ, new_n)
            tl = min(150.0, _remaining() * 0.35)
            if tl > 40:
                a_out, _ = _local_search(a_long, tl, verbose=False)
                _update_best(a_out, f"long_{new_n}")
                if _remaining() > 70:
                    a_out, _ = _full_adam(a_long, min(60.0, _remaining() - 30), tau_start=2000.0, lr=0.12, verbose=False)
                    _update_best(a_out, f"long_adam_{new_n}")

    # Phase 6: final fast polish.
    if _remaining() > 35:
        tl = max(10.0, min(70.0, _remaining() - 12.0))
        a_out, _ = _local_search(_BEST_SEQ, tl, verbose=False)
        _update_best(a_out, "final_local")

    # Final fallback / validation.
    if _BEST_SEQ is None:
        n0 = 4200
        x = np.linspace(0.0, 1.0, n0)
        _BEST_SEQ = _norm_sum1(0.25 + np.cos(np.pi * x) ** 2)
        _update_best(_BEST_SEQ, "fallback")

    final = np.asarray(_BEST_SEQ, dtype=np.float64).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.maximum(final, 0.0)
    s = float(final.sum())
    if s < 0.01:
        x = np.linspace(0.0, 1.0, 4200)
        final = _norm_sum1(0.25 + np.cos(np.pi * x) ** 2)
    else:
        final = final / s

    try:
        sc = evaluate_sequence([float(x) for x in final])
        print(f"FINAL: score={sc:.6f} n={len(final)} time={_elapsed():.0f}s")
    except Exception:
        pass

    return [float(x) for x in final]
