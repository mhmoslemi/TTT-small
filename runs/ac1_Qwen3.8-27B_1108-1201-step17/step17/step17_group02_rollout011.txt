    e = np.exp(theta - theta_max)
    return e / e.sum()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS optimization with log-sum-exp smoothed objective in log-parameterization
2. Multi-phase approach with increasing smoothing (beta) and active set size

import numpy as np

from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def softmax(theta):
    theta = np.asarray(theta, dtype=np.float64)
    theta_max = theta.max()
    e = np.exp(theta - theta_max)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(theta) / len(theta)
    return e / s

def lbfgs_obj_grad(theta, n, beta, verbose=False):
    """Compute objective and gradient for L-BFGS."""
    a = softmax(theta)
    ac = fast_autocorr(a)
    
    # Active lags: those within 3/beta of the max
    max_val = ac.max()
    if max_val < 1e-30:
        return 0.0, np.zeros(n)
    cutoff = max_val - 5.0 / beta
    
    active = np.where(ac >= cutoff)[0]
    vals = ac[active]
    
    # Log-sum-exp
    v_max = vals.max()
    log_sum = v_max + np.log(np.sum(np.exp(vals - v_max)))
    obj = (2.0 * n / beta) * log_sum
    
    # Weights
    w = np.exp(vals - v_max)
    w /= w.sum()
    
    # Gradient of sum_k w_k * C_k(a) w.r.t. a
    grad_a = np.zeros(n)
    for wi, k in zip(w, active):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        j = np.arange(j_start, j_end)
        grad_a[j] += 2.0 * wi * a[ki - j]
    
    grad_a *= (2.0 * n / beta)
    
    # Chain rule through softmax: dL/dtheta_j = a_j * (g_j - sum(a*g))
    grad_theta = a * (grad_a - np.dot(a, grad_a))
    
    return obj, grad_theta

def lbfgs_optimize(a_init, n, beta, max_iter=500, time_limit=60.0, ftol=1e-12):
    """Run L-BFGS-B on the smoothed objective."""
    t0 = time.monotonic()
    a = norm_sum1(a_init)
    theta = np.log(np.maximum(a, 1e-30))
    
    def fun(x):
        if time.monotonic() - t0 > time_limit:
            return (np.inf, np.zeros_like(x))
        return lbfgs_obj_grad(x, n, beta)
    
    result = minimize(fun, theta, method='L-BFGS-B', 
                      options={'maxiter': max_iter, 'ftol': ftol, 
                               'gtol': 1e-10, 'maxfun': max_iter * 3})
    
    theta_opt = result.x
    a_opt = softmax(theta_opt)
    a_opt = norm_sum1(a_opt)
    
    return a_opt, result

def make_sine(n, power=2.0):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** power
    a = np.maximum(a, 1e-15)

def make_power_law(n, p=1.0):
    x = np.linspace(0, 1, n)
    a = x * (1 - x)
    a = a ** p + 1e-10
    return norm_sum1(a)

def make_raised_cosine(n):

def make_kaiser(n, alpha=3.0):

    x = np.linspace(-1, 1, n)
    a = i0(alpha * np.sqrt(np.maximum(1 - x**2, 0)))
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent polish with top-peak focus."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:20]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue
            
            # Find best destination
            best_m_trial = best_m
            best_move = None
            for _ in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m_trial:
                        best_m_trial = M_trial
                        best_move = (a_trial.copy(), M_trial)
            
            if best_move is not None and best_move[1] < best_m - 1e-16:
                a = best_move[0]
                best_m = best_move[1]
                best_a = a.copy()
                improved = True
        
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_m

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS optimization with log-sum-exp smoothed objective in softmax parameterization
   (enforces a>=0, sum(a)=1 automatically; quasi-second-order convergence)
2. Multi-phase strategy: moderate beta -> high beta -> very high beta
3. Multiple diverse initializations (parent, sine, power-law, Kaiser, raised cosine)
4. Multi-length exploration via smooth interpolation
5. Coordinate descent for final micro-optimization
Key insight: the problem is non-convex (max of indefinite quadratic forms), but the
log-sum-exp smoothing makes it amenable to quasi-Newton methods. Using increasingly
high beta gradually focuses on the true maximum lag.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import i0


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def softmax(theta):
    """Numerically stable softmax."""
    theta = np.asarray(theta, dtype=np.float64)
    theta_max = theta.max()
    e = np.exp(theta - theta_max)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(theta) / len(theta)
    return e / s


def lbfgs_obj_grad(theta, n, beta):
    """Compute smoothed objective and gradient for L-BFGS.
    
    Objective: (2n/beta) * log(sum_k exp(beta * C_k(a)))
    where a = softmax(theta), C_k are autocorrelation values.
    Uses cutoff to include only lags within 5/beta of the max.
    """
    a = softmax(theta)
    ac = fast_autocorr(a)
    
    max_val = ac.max()
    if max_val < 1e-30:
        return 0.0, np.zeros(n)
    
    # Active lags: within 5/beta of max (captures >99% of log-sum-exp mass)
    cutoff = max_val - 5.0 / beta
    active = np.where(ac >= cutoff)[0]
    vals = ac[active]
    
    # Log-sum-exp (numerically stable)
    v_max = vals.max()
    shifted = vals - v_max
    log_sum = v_max + np.log(np.sum(np.exp(shifted)))
    obj = (2.0 * n / beta) * log_sum
    
    # Weights for each active lag
    w = np.exp(shifted)
    w /= w.sum()
    
    # Gradient: sum_k w_k * 2 * n / beta * d C_k / d a
    # d C_k / d a_i = 2 * a_{k-i} for valid i
    grad_a = np.zeros(n)
    for wi, k in zip(w, active):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n, ki + 1)
        if j_start >= j_end:
            continue
        j = np.arange(j_start, j_end)
        grad_a[j] += 2.0 * wi * a[ki - j]
    
    grad_a *= (2.0 * n / beta)
    
    # Chain rule through softmax: dL/dtheta_j = a_j * (g_j - sum(a*g))
    grad_theta = a * (grad_a - np.dot(a, grad_a))
    
    return obj, grad_theta


def lbfgs_optimize(a_init, n, beta, max_iter=500, time_limit=120.0, ftol=1e-12, gtol=1e-
10):
    """Run L-BFGS-B on the smoothed objective. Returns optimized a and result."""
    t0 = time.monotonic()
    a = norm_sum1(a_init)
    theta = np.log(np.maximum(a, 1e-30))
    
    calls = [0]
    
    def fun(x):
        calls[0] += 1
        if time.monotonic() - t0 > time_limit:
            raise RuntimeError("time limit")
        return lbfgs_obj_grad(x, n, beta)
    
    try:
        result = minimize(fun, theta, method='L-BFGS-B', jac=True,
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxfun': max_iter * 3})
    except RuntimeError:
        result = None
    except Exception:
        result = None
    
    if result is not None and result.x is not None:
        theta_opt = result.x
    else:
        theta_opt = theta
    
    a_opt = softmax(theta_opt)
    a_opt = norm_sum1(a_opt)
    
    return a_opt, result


def make_sine(n, power=2.0):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Aggressive coordinate descent focusing on top autocorrelation peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:15]
        
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue
            
            # Try moving mass from src to low-contribution positions
            best_m_trial = best_m
            best_move = None
            n_tries = 8
            for _ in range(n_tries):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.003, 0.005, 0.01, 0.02, 0.03]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m_trial:
                        best_m_trial = M_trial
                        best_move = a_trial.copy()
            
            if best_move is not None and best_m_trial < best_m - 1e-16:
                a = best_move
                best_m = best_m_trial
                best_a = a.copy()
                improved = True
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    CoordDescent final: J={J:.6f}, time={elapsed:.0f}s")
    
    return best_a, best_m
