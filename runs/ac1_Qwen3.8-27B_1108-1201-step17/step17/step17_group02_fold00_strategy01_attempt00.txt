        # Ensure sequence is properly formatted and nonnegative
        a = norm_sum1(sequence)
        if np.any(a < 0):
            raise ValueError("Negative values detected.")
    except Exception:

        a = sequence

    # Initialize best sequence and score
    best_seq = a.copy()
    best_score = evaluate_safe(best_seq)

    sum_factor_search_max = 1.5

    sum_grid = np.arange(sum_factor_search_min, sum_factor_search_max + 0.01, 0.01)

    # Main optimization loop
    start_time = time.monotonic()
    for d in d_grid:

def make_two_peak_sequence(n, d_start, beta, gamma):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    try:
        center_dist = d_start / np.exp(beta)
        exponents = np.exp(-((x - 0.5)**2) / (2 * gamma**2))
        a += center_dist * exponents[1:]  # Properly distribute initial energy concentration
        a += center_dist * exponents[:-1] # Ensure symmetric structure maintenance
    except Exception:
        pass
    return a
