"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient with mirror descent (progressively increasing sharpness)
2. Exact peak-targeting coordinate descent for final polish
3. Multi-length exploration
"""

import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation (linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Wrapper around evaluate_sequence with safety checks."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def logsumexp_gradient(a, beta):
    """
    Compute the gradient of the log-sum-exp of the autocorrelation.
    Returns (gradient, lse_value)
    """
    n = len(a)
    c = fast_autocorr(a)  # length 2n-1
    # Numerically stable log-sum-exp
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    log_sum = c_max + np.log(exp_c.sum()) / beta
    p = exp_c / exp_c.sum()  # softmax weights
    
    # Gradient: g_i = 2 * sum_k p[k] * a[k-i] for valid k-i
    # = 2 * (p * a_reversed)[i + n - 1]
    a_rev = a[::-1]
    # Convolve p (length 2n-1) with a_rev (length n)
    # Result has length 3n-2
    # We want indices n-1 to 2n-2 (inclusive), which is n values
    fft_size = 1 << (len(p) + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    Ar = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(P * Ar, fft_size)[:len(p) + n - 1]
    g = 2.0 * conv[n-1 : n-1+n]
    
    return g, log_sum

def lse_mirror_descent(a0, time_limit, betas=None, steps_per_beta=2000, verbose=True):
    """
    Mirror descent with log-sum-exp smoothed objective.
    Progressively increases beta for sharper approximation to max.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if betas is None:
        betas = [5, 10, 20, 50, 100, 200, 500, 1000]
    
    for beta_idx, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        
        if verbose:
            print(f"  LSE-MD beta={beta} ({time.monotonic()-t0:.1f}s elapsed)")
        
        # Adaptive step size
        base_eta = 1.0 / (np.sqrt(beta) * 4.0)
        
        for step in range(steps_per_beta):
            if time.monotonic() - t0 > time_limit:
                break
            
            g, lse = lse_gradient(a, beta)
            
            # Step size with decay
            eta = base_eta / (1.0 + 0.001 * step)
            
            # Multiplicative update (mirror descent on simplex)
            log_a = np.log(np.maximum(a, 1e-300)) - eta * g
            log_a -= log_a.max()  # numerical stability
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            # Check improvement
            M_new = fast_autocorr(a_new).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a_new.copy()
            
            a = a_new
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True, rng_seed=42):

    Coordinate descent: try moving mass from peak-contributing positions
    to low-sensitivity positions.
    """

    best_m = fast_autocorr(a).max()

        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        sens = np.zeros(n)
        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]
        
        improved = False
        
        # Try to reduce the top peaks
        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:

import numpy as np

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_gradient(a, beta):
    """
    Compute gradient of L_beta(a) = (1/beta) * log(sum_k exp(beta * c[k]))
    where c = a * a (linear convolution).
    Returns (gradient_vector, lse_value).
    """
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.minimum(scaled, 700.0)
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = exp_scaled / total
    lse_val = c_max + np.log(total) / beta

    # Equivalent to: g = 2 * conv(p, a_reversed)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_len = len(p) + n - 1  # 3n-2

    conv_full = np.fft.irfft(P * Ar, fft_size)[:conv_len]
    g = 2.0 * conv_full[n - 1:2 * n - 1]

    return g, lse_val
