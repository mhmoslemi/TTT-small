def fast_autocorr(a):
    """Compute full autocorrelation conv(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_conv(a, b):
    """Compute conv(a,b) via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    B = np.fft.rfft(b.astype(np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def smoothed_max_grad(a, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*ac))) w.r.t. a."""
    n = len(a)
    ac = fast_autocorr(a)
    w = np.exp(beta * (ac - ac.max()))
    w /= w.sum()
    # grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    return 2.0 * conv[n-1:2*n-1]

def softmax_obj_grad(z, beta):
    """Objective and gradient in softmax coordinates."""
    za = z - z.max()
    a = np.exp(za)
    a /= a.sum()
    n = len(a)
    ac = fast_autocorr(a)
    m = ac.max()
    exp_b = np.exp(beta * (ac - m))
    obj = (m + np.log(exp_b.sum()) / beta)
    
    w = exp_b / exp_b.sum()
    # grad_a
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w)] = w
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    grad_a = 2.0 * conv[n-1:2*n-1]
    
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return obj, grad_z

The LP-FW iteration:

2. Compute `ac = conv(a,a)`, `M = max(ac)`

6. Update: `a_new = (1-t)a + t*b*`

def peak_shave(a0, time_limit, verbose=True, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top peak lags
        n_peaks = min(20, len(ac))
        peak_idx = np.argsort(ac)[::-1][:n_peaks]
        
        improvement = False
        
        for k in peak_idx[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            # Valid j range for this lag
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            js = np.arange(j_start, j_end + 1)
            partners = k_int - js
            contrib = a[js] * a[partners]
            
            if len(js) == 0:
                continue
            
            # Find the source with highest contribution
            src_idx = js[np.argmax(contrib)]
            src_val = a[src_idx]
            if src_val < 1e-12:
                continue
            
            # Compute sensitivity for all positions
            # sens[j] = sum over peak lags of a[k-j] (contribution coefficient)
            sens = np.zeros(n)
            for k2 in peak_idx[:15]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                for j in range(s2, e2 + 1):
                    sens[j] += a[k2i - j]
            
            # Find best destination: low sensitivity, far from source
            dist = np.abs(np.arange(n) - src_idx).astype(float)
            score = sens / (1 + dist / n)
            # Exclude source itself
            score[src_idx] = np.inf
            dst_idx = np.argmin(score)
            
            if dst_idx == src_idx:
                continue
            
            # Try several transfer fractions
            best_trial = None
            best_trial_m = M_curr
            for frac in [0.05, 0.02, 0.01, 0.005, 0.1, 0.03, 0.015, 0.08]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src_idx] -= amt
                a_trial[dst_idx] += amt
                if a_trial[src_idx] < 0:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_trial_m - 1e-14:
                    best_trial_m = M_trial
                    best_trial = a_trial
            
            if best_trial is not None:
                a = best_trial
                if best_trial_m < best_m:
                    best_m = best_trial_m
                    best_a = a.copy()
                improvement = True
                stall = 0
                break  # Restart the peak loop
        
        if not improvement:
            stall += 1
            # Random perturbation
            if stall % 20 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt

                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 50 == 0:

            J = 2.0 * n * best_m
            print(f"    PeakShave ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m
