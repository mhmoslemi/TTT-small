def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:

def build_constraint_matrix_lags(lags, n, a):
    """Build constraint matrix rows for given lags.
    For lag k, the constraint is: sum_j b_j * a[k-j] <= M
    where a[k-j] is 0 if k-j is out of [0, n-1].
    So row k of the matrix has entry (k, j) = a[k-j] if k-j in [0,n-1], else 0.
    """
    K = len(lags)
    T = np.zeros((K, n))
    for ii, k in enumerate(lags):
        k_int = int(k)
        # T[ii, j] = a[k_int - j] if 0 <= k_int - j < n, else 0
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            T[ii, j_start:j_end] = a[k_int - np.arange(j_start, j_end)]
    return T


def a_fw(a0, time_budget, solver, verbose=False):
    """Away-step Frank-Wolfe with backtracking line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adaptive K
    K_base = min(300, 2*n-1)
    K_max = min(1000, 2*n-1)
    
    iter_count = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_budget:
        iter_count += 1
        elapsed = time.monotonic() - t0

        M_curr = ac.max()

        K = K_base + min(K_max - K_base, iter_count // 50)

        b_var = cp.Variable(n, nonneg=True)

            T_sel @ b_var <= M_vals,
            cp.sum(b_var) <= 1.0

        except Exception:
            continue

            continue

        if b_sum < 1e-15:
            continue
        v = b_val / b_sum  # FW vertex (normalized to sum 1)

        i_end = min(n, k_star + 1)

        grad[idx] = 2.0 * a[j_idx]

        d_fw = v - a
        
        # Away step: find coordinate to move away from
        # The coordinate that most increases the objective
        away_candidates = np.where(a > 1e-8)[0]

            # Move away from coordinate with highest grad value
            j_away = away_candidates[np.argmax(grad[away_candidates])]
            a_j_away = a[j_away]
        else:
            j_away = -1
            a_j_away = 0.0
        
        # Backtracking line search
        best_F = M_curr
        best_update = None

        for gamma in [gamma0, 2*gamma0, 0.5*gamma0, 0.25*gamma0, 1.5*gamma0]:
            a_trial = a + gamma * d_fw
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            F_trial = fast_autocorr(a_trial).max()
            if F_trial < best_F - 1e-12:
                best_F = F_trial
                best_update = ('fw', gamma, a_trial)

                if s_t < 1e-15:
                    continue

            a = best_update[2]

        else:

            a = a + gamma0 * d_fw
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stall_count += 1
        
        # Track best
        cur_M = fast_autocorr(a).max()
        if cur_M < best_M:

    return best_a, best_M


def logsumexp_adam(a0, time_budget, verbose=False, seed=123):
    """
    Log-sum-exp smoothed optimization with Adam.
    Minimizes (1/tau)*log(sum exp(tau * ac_k)) which approximates max(ac_k).
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_adam = 0
    
    # Annealing schedule for tau
    tau_start = 1.0
    tau_end = 50.0
    
    while time.monotonic() - t0 < time_budget:
        t_adam += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / time_budget
        
        # Anneal tau
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.5)
        
        # Compute autocorrelation
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Compute log-sum-exp gradient
        # soft_max weights: w_k = exp(tau * ac_k) / sum_j exp(tau * ac_j)
        # To avoid overflow, subtract max
        ac_shifted = ac - ac.max()
        exp_vals = np.exp(tau * ac_shifted)
        w = exp_vals / exp_vals.sum()
        
        # Gradient = sum_k w_k * (2 * shift(a, k))
        # where (shift(a, k))[i] = a[k-i] (with bounds)
        grad = np.zeros(n)
        # Only use top contributing lags for efficiency
        n_top = min(20, len(ac))
        top_w_idx = np.argsort(w)[::-1][:n_top]
        for k in top_w_idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * w[k_int] * a[j_idx]
        
        # Adam update
        lr = 0.1 / (1.0 + 0.1 * t_adam ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        
        # Mirror descent step (log-space)
        log_a = np.log(np.maximum(a, 1e-30)) - lr * m_hat / (np.sqrt(v_hat) + eps)
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Occasionally add noise to escape local minima
        if t_adam % 500 == 0:
            noise = rng.normal(0, 0.01, n) * a
            a_new = a_new + noise
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()
        
        a = a_new
        
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
        
        if verbose and t_adam % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam t={t_adam} ({elapsed:.0f}s): J={best_J:.6f}, tau={tau:.1f}")
    
    return best_a, best_J

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propertwo(a, time_budget, solver, verbose=False):
    """
    Alternative: property-two style iterative method.
    At each step, solve for the "best response" given the current peak constraint.
    """
    n = len(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Binary search on M
    # Lower bound: 1/(2n-1) (sum of all ac values = 1)
    # Upper bound: current M
    M_lo = 0.0
    M_hi = best_M
    
    K = min(200, 2*n-1)
    
    for outer in range(50):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Midpoint
        M_mid = (M_lo + M_hi) / 2
        
        # Try to find b with sum(b) = 1 and (b*a)[k] <= M_mid for all top-K k
        ac = fast_autocorr(a)
        top_idx = np.argsort(ac)[::-1][:K]
        T_sel = build_constraint_matrix_lags(top_idx, n, a)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [
            T_sel @ b_var <= M_mid,
            cp.sum(b_var) == 1.0
        ]
        prob = cp.Problem(cp.Minimize(cp.sum_squares(b_var - a)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            M_lo = M_mid
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            M_lo = M_mid
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            M_lo = M_mid
            continue
        b_val = b_val / s_b
        
        # Check if feasible: verify peak constraint
        ac_b = fast_autocorr(b_val)
        if ac_b.max() <= M_mid * 1.01:
            # Feasible, try smaller M
            M_hi = M_mid
            # Update a toward b
            a_new = 0.5 * a + 0.5 * b_val
            a_new = norm_sum1(a_new)
            M_new = fast_autocorr(a_new).max()
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
            a = a_new
        else:
            M_lo = M_mid
        
        if verbose and outer % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Prop2 outer={outer} ({elapsed:.0f}s): J={J:.6f}, M=[{M_lo:.8f},{M_hi:.8f}]")
    
    return best_a, best_M

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)

def build_constraint_rows(lags, n, a):
    """Build constraint matrix rows for given lags.
    Row for lag k: T[k, j] = a[k-j] if 0<=k-j<n else 0.
    """
    K = len(lags)
    T = np.zeros((K, n))
    for ii in range(K):
        k_int = int(lags[ii])
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start < j_end:
            T[ii, j_start:j_end] = a[k_int - np.arange(j_start, j_end)]
    return T

def logsumexp_adam(a0, time_budget, verbose=False, seed=123):
    """
    Log-sum-exp smoothed optimization with Adam in log-space.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_adam = 0

    tau_start = 2.0
    tau_end = 80.0

    while time.monotonic() - t0 < time_budget:
        t_adam += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_budget:
            break
        progress = min(1.0, elapsed / time_budget)

        # Anneal tau
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.4)

        # Compute autocorrelation
        ac = fast_autocorr(a)
        M = ac.max()

        # Softmax weights (top-30 for efficiency)
        ac_shifted = ac - ac.max()
        exp_vals = np.exp(tau * ac_shifted)
        w = exp_vals / exp_vals.sum()

        # Gradient using top contributing lags
        grad = np.zeros(n)
        n_top = min(30, len(ac))
        top_w_idx = np.argsort(w)[::-1][:n_top]
        for k in top_w_idx:
            k_int = int(k)
            if w[k_int] < 1e-8:
                continue
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * w[k_int] * a[j_idx]

        # Adam update
        lr = 0.05 / (1.0 + 0.05 * t_adam ** 0.5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        # Mirror descent step in log-space
        log_a = np.log(np.maximum(a, 1e-30)) - lr * m_hat / (np.sqrt(v_hat) + eps)
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        # Periodic noise
        if t_adam % 800 == 0:
            noise = rng.normal(0, 0.005, n) * a
            a_new = a_new + noise
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()

        a = a_new

        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()

        if verbose and t_adam % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam t={t_adam} ({elapsed:.0f}s): J={best_J:.6f}, tau={tau:.1f}")

    return best_a, best_J


def peak_reduction_refined(a0, time_budget, verbose=False, seed=77):
    """Refined peak reduction with binary search on transfer amount."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()

    t0 = time.monotonic()

        # Compute importance (contribution to peak)
        importance = np.zeros(n)
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_star - idx
        contrib = a[idx] * a[j_idx]
        importance[idx] += contrib
        importance[j_idx] += contrib

        # Also consider second peak
        ac2 = ac.copy()
        ac2[k_star] = -1
        k_star2 = int(np.argmax(ac2))
        i_start2 = max(0, k_star2 - n + 1)
        i_end2 = min(n, k_star2 + 1)
        idx2 = np.arange(i_start2, i_end2)
        j_idx2 = k_star2 - idx2
        contrib2 = a[idx2] * a[j_idx2]
        importance[idx2] += 0.5 * contrib2
        importance[j_idx2] += 0.5 * contrib2

        # Source and destination selection
        threshold_hi = np.percentile(importance[importance > 0], 80) if np.any(importance > 0) else 0
        threshold_lo = np.percentile(importance, 20)
        high_imp = np.where((importance > threshold_hi) & (a > 0.0001))[0]
        low_imp = np.where(importance <= threshold_lo)[0]

        if len(high_imp) == 0 or len(low_imp) == 0:
            stall += 1
            continue

        improved = False

        for _ in range(40):
            if time.monotonic() - t0 > time_budget:
                break

            src = high_imp[rng.integers(len(high_imp))]
            dst = low_imp[rng.integers(len(low_imp))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            # Binary search for optimal transfer
            lo, hi = 0.0, src_val * 0.4
            best_amt = 0.0
            best_trial_J = compute_J(a)

            for _ in range(6):
                mid = (lo + hi) / 2.0
                if mid < 1e-10:

                a_trial = a.copy()
                a_trial[src] -= mid
                a_trial[dst] += mid
                a_trial = np.maximum(a_trial, 0.0)

                J_trial = compute_J(a_trial)

                    best_amt = mid
                    lo = mid
                else:
                    hi = mid

                a[src] -= best_amt
                a[dst] += best_amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new

                    improved = True

        else:
            stall += 1

                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()

    if verbose:

    return best_a, best_J


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    t1 = min(remaining * 0.30, 550.0)

    print(f"\n=== Phase 2: LSE-Adam (n={len(best_seq)}) ===")

    # === Phase 3: AFW round 2 ===
    print(f"\n=== Phase 3: AFW round 2 (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)

    # === Phase 4: Peak reduction ===
    print(f"\n=== Phase 4: Peak reduction (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    # === Phase 5: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration ===")

    lengths_to_try = [n_best + 400, n_best + 800, n_best - 400]

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out, _ = peak_reduction_refined(norm_sum1(best_seq), min(remaining * 0.15, 60.0), verbose=False, seed=333)
            a_out = np.maximum(a_out, 0.0)
