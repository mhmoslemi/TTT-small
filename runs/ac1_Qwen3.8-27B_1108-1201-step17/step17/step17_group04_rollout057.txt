                        amt = a[p] * frac
                        dst = int(rng.integers(n))
                        a[p] -= amt
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                c = fast_conv(a)
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    stall = 0

        if verbose and iter_count % 50 == 0:
            print(f"    CPD iter={iter_count} ({elapsed:.0f}s): obj={best_obj:.7f}, stall={stall}")

    return best_a, best_obj

def smooth_optimize(a0, time_limit, verbose=False, seed=42):
    """Smooth optimization using log-sum-exp approximation with L-BFGS-B."""
    n = len(a0)
    a_init = norm_sum1(a0).copy()
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    t0 = time.monotonic()

    # Parameterize a = softmax(z) so a > 0 and sum(a) = 1 automatically
    z0 = np.log(a_init + 1e-30)

    def objective(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = fast_conv(a)
        # Log-sum-exp approximation of max
        beta = 50.0
        scaled = beta * c
        max_s = scaled.max()
        lse = (max_s + np.log(np.sum(np.exp(scaled - max_s)))) / beta
        return lse

    def gradient(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        c = fast_conv(a)
        beta = 50.0
        scaled = beta * c
        max_s = scaled.max()
        weights = np.exp(scaled - max_s)
        weights = weights / weights.sum()
        # Gradient of lse w.r.t. a
        grad_a = np.zeros(n)
        for k in range(len(c)):
            if weights[k] < 1e-10:
                continue
            wk = weights[k]
            k_int = int(k)
            lo_i = max(0, k_int - n + 1)
            hi_i = min(n, k_int + 1)
            idx = np.arange(lo_i, hi_i)
            jdx = k_int - idx
            valid = (jdx >= 0) & (jdx < n)
            grad_a[idx[valid]] += 2.0 * wk * a[jdx[valid]]
        # Chain rule: z -> a = softmax(z)
        # d(softmax(z))[i]/dz[j] = a[i]*(delta_ij - a[j])
        grad_z = a * (grad_a - np.dot(grad_a, a))
        return grad_z

    # Run L-BFGS-B with time limit
    best_z = z0.copy()
    best_val = objective(z0)

    def callback(xk):
        nonlocal best_z, best_val
        val = objective(xk)
        if val < best_val:
            best_z = xk.copy()
            best_val = val
        return time.monotonic() - t0 > time_limit - 10

    try:
        result = scipy_opt.minimize(objective, z0, jac=gradient,
                                     method='L-BFGS-B', callback=callback,
                                     options={'maxiter': 500, 'gtol': 1e-8})
    except Exception as e:
        if verbose:
            print(f"    Smooth opt error: {e}")

    if best_val < objective(z0):
        a_new = np.exp(best_z - best_z.max())
        a_new = a_new / a_new.sum()
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj

def nesterov_mirror(a0, time_limit, verbose=False, top_k=10, beta=2.0,
                    c_param=3.0, seed=42):
    """Nesterov mirror descent with adaptive subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_conv(y)
        M = ac.max()

        # Subgradient from top-k peaks
        g = np.zeros(n)
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        for wi, k in zip(w, top_idx):
            k_int = int(k)
            lo_i = max(0, k_int - n + 1)
            hi_i = min(n, k_int + 1)
            idx = np.arange(lo_i, hi_i)
            jdx = k_int - idx
            valid = (jdx >= 0) & (jdx < n)
            g[idx[valid]] += 2.0 * wi * a[jdx[valid]]

        # Step size
        lower_bound = 1.0 / n
        gap = M - lower_bound
        eta = c_param * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)

        # Mirror descent (exponential)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        J_new = compute_obj(x)
        if J_new < best_obj - 1e-14:
            best_obj = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            sigma = 0.01
            x = best_a + rng.normal(0, sigma, n) * best_a
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()

        if verbose and t % 50000 == 0:
            print(f"    Mirror t={t} ({time.monotonic()-t0:.0f}s): obj={best_obj:.7f}")

    return best_a, best_obj

def interp_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_out = np.interp(new_x, old_x, f_norm)
    f_out = np.maximum(f_out, 0.0)
    ss = f_out.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_out / ss

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator combining coordinate pair descent, smooth optimization,
    and Nesterov mirror descent for minimizing 2*n*max(conv(a,a))/(sum(a))^2."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = np.cos(np.pi * x) ** 2 + 0.01
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Coordinate pair descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 650.0)
    print(f"\n=== Phase 1: Coordinate pair descent ({t1:.0f}s) ===")
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_pair_descent(a_in, t1, verbose=True, seed=42,
                                           n_peaks_to_consider=15, batch_size=400)
        try_update(a_out, "cpd_main")

    # === Phase 2: Smooth L-BFGS optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: Smooth L-BFGS ({t2:.0f}s) ===")
    if t2 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_optimize(a_in, t2, verbose=True)
        try_update(a_out, "smooth_lbgfs")

    # === Phase 3: Nesterov mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 3: Nesterov mirror ({t3:.0f}s) ===")
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror(a_in, t3, verbose=True, top_k=12, beta=2.5,
                                   c_param=4.0, seed=99)
        try_update(a_out, "mirror_descent")

    # === Phase 4: Second CPD round (finer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 4: CPD round 2 ({t4:.0f}s) ===")
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_pair_descent(a_in, t4, verbose=True, seed=777,
                                           n_peaks_to_consider=20, batch_size=500)
        try_update(a_out, "cpd_round2")

    # === Phase 5: Try slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        for delta_n in [200, -200]:
            new_n = n0 + delta_n
            if new_n < 500:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t5 = min(remaining * 0.15, 120.0)
            if t5 < 20:
                break
            print(f"\n=== Phase 5: Try n={new_n} ({t5:.0f}s) ===")
            a_long = interp_seq(best_seq, new_n)
            a_out, _ = coordinate_pair_descent(a_long, t5, verbose=False, seed=555,
                                               n_peaks_to_consider=15, batch_size=200)
            try_update(a_out, f"cpd_n{new_n}")

    # === Phase 6: Final CPD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.30, 400.0)
    print(f"\n=== Phase 6: Final CPD polish ({t6:.0f}s) ===")
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_pair_descent(a_in, t6, verbose=True, seed=12345,
                                           n_peaks_to_consider=25, batch_size=600)
        try_update(a_out, "cpd_final_polish")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.cos(np.pi * x) ** 2 + 0.01
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
