    except Exception:
        return float('inf')

def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    return normalize(f_new)

def spectrum_flatten(a, eta=1.0, num_iters=500):
    """
    Aggressive spectrum flattening.
    Iteratively equalizes the magnitude of the DFT of the zero-padded sequence.
    """
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(num_iters):
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Flattening: push P toward uniform
        P_target = P_mean * (P / P_mean) ** (1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)
        
        # Preserve phase, adjust magnitude
        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        M_new = fast_conv(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped update
            a = 0.7 * a + 0.3 * a_new
            a = normalize(a)
    
    return best_a

def subgradient_step(a, conv_a, top_k=30, beta=3.0):
    """Compute the subgradient using top_k active lags."""
    n = len(a)
    k_sel = min(top_k, len(conv_a))
    top_idx = np.argsort(conv_a)[::-1][:k_sel]
    top_vals = np.maximum(conv_a[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_sel) / k_sel
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        for j in range(n):
            kj = k_int - j
            if 0 <= kj < n:
                grad[j] += wi * 2.0 * a[kj]
    return grad

def nesterov_descent(a0, time_limit, top_k=40, beta=4.0, c0=3.0, verbose=False):
    """Nesterov-accelerated mirror descent with subgradient."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute subgradient
        ac = fast_conv(y)
        g = subgradient_step(y, ac, top_k=top_k, beta=beta)
        
        # Multiplicative update (mirror descent on simplex)
        eta_t = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta_t * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Re-randomization on stall
        if stall > 5000:
            stall = 0
            pert = rng.normal(0, 0.05, n) * best_a
            x = np.maximum(best_a + pert, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            top_k = min(top_k + 10, 100)
            beta = max(beta - 0.2, 2.0)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NSD t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def lp_direction(a, top_K=100, solver=None):

    binding = np.argsort(margins)[:k_sel]

    # (using the "other" sequence a fixed, b is the variable)
