import numpy as np

def fast_autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

import time
import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

            x = cp.Variable(10, nonneg=True)

            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except:
            continue
    return cp.ECOS

    """Generate diverse parametric initializations."""
    t0 = time.monotonic()

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:

        try_candidate((n - i_arr) ** p, f"q{p}")

    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for beta in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a = np.maximum(i_arr + 1, 0.5) ** alpha * np.maximum(n - i_arr + 0.5, 0.5) ** (beta - 1)

            a = np.exp(-((x - c) ** 2) / (2 * sigma ** 2))
            try_candidate(a, f"g{c}_{sigma}")

    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def get_solver():
    solver = None
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                print(f"  Solver: {s}")
                break
        except:
            continue
    if solver is None:
        solver = cp.ECOS
        print("  Solver (fallback): ECOS")
    return solver

def lp_fw_optimize(a0, time_limit, solver, verbose=True):
    """
    LP-based Frank-Wolfe with Nesterov acceleration and adaptive cutting planes.
    Optimizes max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a0)
    a = norm_sum1(a0)
    ac = fast_autocorr(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    
    y = a.copy()  # current iterate
    x = a.copy()  # Nesterov extrapolated
    
    # Initialize constraint set with top-K lags
    init_K = min(300, 2*n - 1)
    top_lags = np.argsort(ac)[::-1][:init_K]
    constraint_set = sorted(set(int(k) for k in top_lags))
    
    iter_count = 0
    stall_count = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit:
        # Build constraint matrix
        K = len(constraint_set)
        if K == 0:
            break
        
        # For each constraint k, the coefficient for a[i] is a[k-i] (for valid i)
        # conv_k = sum_i a[i] * a[k-i]
        # So the constraint is: sum_i (a[k-i]) * g[i] <= M
        # where g is the variable we're solving for
        
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_set):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k - idx]
        
        # Solve LP: min M s.t. sum(g)=1, g>=0, coeffs@g <= M
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [cp.sum(g) == 1, coeffs @ g <= M]
        prob = cp.Problem(cp.Minimize(M), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Nesterov update
        t = iter_count + 1
        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        iter_count += 1

        ac_y = fast_autocorr(y)
        M_y = ac_y.max()
        if M_y < best_m - 1e-15:
            best_m = M_y
