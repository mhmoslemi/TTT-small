"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed mirror descent with adaptive log-sum-exp and FFT-based gradient
2. Cyclic coordinate descent with exact 1D minimization
3. Targeted peak-shaving moves with sensitivity analysis
4. LP Frank-Wolfe on active constraints
5. Multi-length exploration with interpolation
Key innovation: exact gradient via np.correlate for the smoothed objective,
combined with coordinate descent that exploits the linearity of conv in each
individual coordinate for fast 1D optimization.

import numpy as np
import time
import cvxpy as cp

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute full linear autocorrelation (np.convolve(a,a)) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def smooth_gradient(a, ac, mu, top_k=200):
    """
    Compute gradient of the smoothed objective mu*log(sum(exp(ac/mu)))
    using np.correlate for efficiency.
    Returns gradient vector of same length as a.
    """
    n = len(a)
    k_sel = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_sel]
    ac_top = ac[top_idx]
    max_ac = ac_top.max()
    exponents = (ac_top - max_ac) / mu
    exponents = np.clip(exponents, -500, 500)
    w_top = np.exp(exponents)
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        w_top = np.ones(k_sel) / k_sel
    else:
        w_top = w_top / w_sum

    w_full = np.zeros(len(ac))
    w_full[top_idx] = w_top

    # grad[j] = 2 * sum_{m=0}^{n-1} w_full[j+m] * a[m]
    # This is np.correlate(w_full, a, mode='valid')
    grad = 2.0 * np.correlate(w_full, a, mode='valid')
    return grad


def smoothed_mirror_descent(a0, time_limit, mu_init=None, mu_end=None,
                            top_k=200, seed=42, verbose=False):
    """
    Nesterov-accelerated mirror descent with adaptive log-sum-exp smoothing.
    """
    n = len(a0)
    if mu_init is None:
        mu_init = 1.0 / (2.0 * n) * 50.0
    if mu_end is None:
        mu_end = 1.0 / (2.0 * n) * 2.0

    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    restart_every = 2000
    total_iters = 100000
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        frac = min(t / total_iters, 1.0)
        mu = mu_init * (mu_end / mu_init) ** frac

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        # Compute autocorrelation and gradient
        ac = fast_autocorr(y)
        M = ac.max()
        grad = smooth_gradient(y, ac, mu, top_k=top_k)

        # Adaptive step size
        lower_bound = 1.0 / (2.0 * n - 1.0)
        gap = M - lower_bound
        gap = max(gap, 1e-10)
        grad_norm_sq = np.dot(grad, grad) + 1e-20
        eta = 2.5 * gap / grad_norm_sq
        eta = min(eta, 3.0 / (t + 1.0))
        eta = max(eta, 1e-12)

        # Mirror descent step (exponential family)
        log_y = np.log(y)
        log_a_new = log_y - eta * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        # Evaluate every 50 steps to save time
        if t % 50 == 0 or t < 20:
            J_curr = 2.0 * n * fast_autocorr(x).max()
            if J_curr < best_J - 1e-14:
                best_J = J_curr
                best_a = x.copy()
                stall = 0
            else:
                stall += 1

        # Restart with perturbation
        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.003 * (stall // restart_every)
            perturb = rng.normal(0, sigma, n) * best_a
            x_r = best_a + perturb
            x_r = np.maximum(x_r, 1e-12)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD t={t} ({elapsed:.0f}s): J={best_J:.8f}, mu={mu:.2e}")

    return best_a, best_J


def coordinate_descent(a0, time_limit, verbose=False, max_sweeps=200):
    """
    Cyclic coordinate descent: for each coordinate, find the 1D optimal value
    using the fact that conv(a,a)[k] is (approximately) linear in a[i].
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute slopes: for each lag k, the coefficient of a[i] in conv(a,a)[k]
            # conv(a,a)[k] involves a[i] when 0 <= k-i < n, i.e., k in [i, i+n-1]
            k_start = i
            k_end = min(2 * n - 2, i + n - 1)

            if k_start > k_end:
                continue

            # Current autocorrelation values for relevant lags
            ac = fast_autocorr(a)
            # Slopes: d(conv(a,a)[k])/da[i] = 2 * a[k-i] for 0 <= k-i < n
            ks = np.arange(k_start, k_end + 1)
            slopes = 2.0 * a[ks - i]

            # Current values
            vals = ac[ks]

            # 1D minimization: min_{x >= 0} max_k (vals_k + slopes_k * (x - a[i]))
            # = min_{x >= 0} max_k ((vals_k - slopes_k * a[i]) + slopes_k * x)
            biases = vals - slopes * a[i]

            # The function f(x) = max_k (biases_k + slopes_k * x)
            # is convex piecewise-linear. Find its minimum for x >= 0.
            # Use bisection on the subgradient.
            # Subgradient at x: {s_k : k in argmax_k(biases_k + slopes_k * x)}
            # If min subgrad > 0: move left. If max subgrad < 0: move right.
            # If 0 is in subgradient: optimum found.

            x_lo = 0.0
            x_hi = max(10.0 * a[i], 1.0 / n * 10.0)

            # Quick check: if the max is already at x=0, don't move
            f_at_0 = np.max(biases)
            # Slopes of the max-achieving terms at x=0
            max_mask_0 = np.abs(biases - f_at_0) < 1e-15
            if np.min(slopes[max_mask_0]) >= 0:
                # All max terms have non-negative slope, so increasing x increases f
                # Check if decreasing helps (but x >= 0, so we stay at 0)
                continue

            # Bisection for 30 iterations
            for _ in range(30):
                x_mid = 0.5 * (x_lo + x_hi)
                f_vals = biases + slopes * x_mid
                f_max = np.max(f_vals)
                max_mask = np.abs(f_vals - f_max) < 1e-14
                s_min = np.min(slopes[max_mask])
                s_max = np.max(slopes[max_mask])

                if s_min >= -1e-14:
                    # 0 is in or to the left of the subgradient, move right
                    x_lo = x_mid
                else:
                    # All subgradient elements are negative, move left
                    x_hi = x_mid

            x_opt = 0.5 * (x_lo + x_hi)
            x_opt = max(x_opt, 0.0)

            if abs(x_opt - a[i]) > 1e-14:
                a[i] = x_opt
                improved = True

        # Renormalize
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        J_curr = 2.0 * n * fast_autocorr(a).max()
        if J_curr < best_J - 1e-14:
            best_J = J_curr
            best_a = a.copy()
        elif not improved:
            break

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep} ({elapsed:.0f}s): J={best_J:.8f}")

    return best_a, best_J


def peak_shaving(a0, time_limit, verbose=False, seed=77):
    """
    Targeted coordinate moves: identify peak contributors and relocate mass
    to positions that don't contribute to high autocorrelation lags.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        n_peaks = min(20, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # Compute sensitivity: how much each position contributes to top peaks
        sensitivity = np.zeros(n)
        for k in top_lags[:10]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contrib = a[idx] * a[j_idx]
            for ii, cc in zip(idx, contrib):
                sensitivity[ii] += cc
            for jj, cc in zip(j_idx, contrib):
                sensitivity[jj] += cc

        # Find high-sensitivity positions with mass
        high_sens = np.where((sensitivity > sensitivity.mean() + 0.5 * sensitivity.std()) & (a > 1e-6))[0]
        low_sens = np.where(sensitivity < sensitivity.mean() - 0.3 * sensitivity.std())[0]
        low_sens = low_sens[low_sens != np.zeros(n, dtype=int)]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            # Random perturbation to escape
            if stall % 20 == 0:
                n_pert = max(3, n // 50)
                idx = rng.choice(n, size=n_pert, replace=False)
                for p in idx:
                    if a[p] > 1e-6:
                        amt = a[p] * rng.uniform(0.02, 0.1)
                        a[p] -= amt
                        dst = int(rng.integers(n))
                        a[dst] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
            continue

        improved = False
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            src = high_sens[int(rng.integers(len(high_sens)))]
            dst = low_sens[int(rng.integers(len(low_sens)))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            fracs = [0.01, 0.03, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6]
            found = False
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = 2.0 * n * fast_autocorr(a_trial).max()
                if J_trial < best_J - 1e-12:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if found:
                stall = 0

        if not improved:
            stall += 1

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave ({elapsed:.0f}s): J={best_J:.8f}, stall={stall}")

    return best_a, best_J


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300):
    """LP Frank-Wolfe with Nesterov acceleration on top-K constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.8f}")

    return best_a, best_m


def make_symmetric_init(n):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_hann_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def make_power_law_init(n, power=0.7):
    x = np.linspace(-1, 1, n)
    a = (1 - x ** 2) ** power + 0.001
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator:
    Phase 1: Smoothed mirror descent from multiple inits
    Phase 2: Coordinate descent polish
    Phase 3: Peak shaving
    Phase 4: LP Frank-Wolfe
    Phase 5: Multi-length exploration
    Phase 6: Final coordination
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.8f}")

    # === Phase 1: Smoothed mirror descent (main workhorse) ===
    print(f"\n=== Phase 1: Smoothed Mirror Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 500.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_hann_init(n0))
        labels.append("hann")

        inits.append(make_power_law_init(n0, power=0.5))
        labels.append("power0.5")

        # Blend parent with hann
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_hann_init(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Remove near-duplicates
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for u in unique_inits:
                if np.sum(np.abs(init - u)) < 0.03:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per = max(30.0, t_phase1 / n_inits)

        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 20)
            if t_avail < 20:
                break
            print(f"  SMD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = smoothed_mirror_descent(init, t_avail, top_k=200,
                                               seed=42 + idx * 7, verbose=False)
            try_update(a_out, f"smd_{lbl}")

    # === Phase 2: Coordinate descent polish ===
    print(f"\n=== Phase 2: Coordinate Descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t2, verbose=True, max_sweeps=500)
        try_update(a_out, f"cd_n{len(a_in)}")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t3, verbose=True, seed=123)
        try_update(a_out, f"peak_n{len(a_in)}")

    # === Phase 4: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP Frank-Wolfe (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K=400)
        try_update(a_out, f"lpfw_n{n_best}")

    # === Phase 5: Second round of SMD + CD on best ===
    print(f"\n=== Phase 5: SMD + CD refinement (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5a = min(remaining * 0.12, 200.0)
    if t5a > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_mirror_descent(a_in, t5a, top_k=100, seed=999)
        try_update(a_out, f"smd2_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    t5b = min(remaining * 0.08, 150.0)
    if t5b > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t5b, verbose=False, max_sweeps=300)
        try_update(a_out, f"cd2_n{len(a_in)}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    candidate_lengths = [n_best + 200, n_best + 400, n_best - 200, n_best - 400]
    candidate_lengths = [l for l in candidate_lengths if 1000 <= l <= 5000]
    # Remove duplicates and sort by proximity to n_best
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_best))

    for new_n in candidate_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 50:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.20, 200.0)
        if t6a > 20:
            a_out, _ = smoothed_mirror_descent(a_long, t6a, top_k=150, seed=555)
            try_update(a_out, f"smd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and best_seq is not None:
            a_in = norm_sum1(best_seq)
            t6b = min(remaining * 0.10, 100.0)
            if t6b > 15:
                a_out, _ = coordinate_descent(a_in, t6b, verbose=False, max_sweeps=200)
                try_update(a_out, f"cd_n{len(a_in)}")

    # === Phase 7: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Peak Shaving (n={n_best}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.3, 200.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_shaving(a_in, t7, verbose=True, seed=456)
        try_update(a_out, f"peak_final_n{n_best}")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final CD (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.5, 200.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t8, verbose=True, max_sweeps=500)
        try_update(a_out, f"cd_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.8f}")

    return [float(x) for x in best_seq]
