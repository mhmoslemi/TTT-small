- If `m+n=k` and `m != n`: the term `a[m]*a[n]` appears twice (once as `a[m]*a[n]` and once as `a[n]*a[m]`). Coefficient in the quadratic form is 2, so Hessian entry is 2.

"""

import numpy as np
import time
from scipy.optimize import minimize_scalar

def fast_conv(a):
    """Fast FFT-based convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = next_power_of_2(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def next_power_of_2(n):
    return 1 << (n - 1).bit_length()

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def objective(a):
    """Compute 2*n*max(b) assuming sum(a)=1."""
    b = fast_conv(a)
    return 2.0 * len(a) * b.max()

def subgradient_at_peak(a, b, top_k=30):
    """Compute subgradient using top-k peak lags."""
    n = len(a)
    M = b.max()
    # Find lags near the peak
    threshold = M * 0.99
    near_peak = np.where(b >= threshold)[0]
    if len(near_peak) > top_k:
        # Take the top-k by value
        top_idx = np.argsort(b)[::-1][:top_k]
    else:
        top_idx = near_peak
    
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        # db[k]/da[j] = 2*a[k-j] if 0<=k-j<n
        valid = np.arange(max(0, k_int - n + 1), min(n, k_int + 1))
        g[valid] += 2.0 * a[k_int - valid]
    g /= len(top_idx)
    return g

def nmd_search(a0, time_limit, top_k=30, c0=3.0, verbose=False):
    """Nesterov Mirror Descent with adaptive parameters."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = objective(a)
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 2000
    rng = np.random.default_rng(42 + n)
    c = c0
    momentum = 0.9
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t + momentum) / (t + 1.0 + momentum)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        b = fast_conv(y)
        M = b.max()
        
        # Adaptive top-k
        threshold = M * (1.0 - 2.0 / (t + 100))  # Start strict, relax
        near_max = np.sum(b >= threshold)
        k_use = min(max(top_k, int(near_max) + 3), min(80, len(b)))
        
        g = subgradient_at_peak(y, b, k_use)
        
        # Step size
        eta = c / (np.sqrt(t) + 1.0)
        
        # Mirror descent update
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-20)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        val = objective(x)
        if val < best_val - 1e-15:
            best_val = val
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart
        if stall_count > max_stall:
            strategy = int(rng.integers(4))
            if strategy == 0:
                x = best_a * (1 + rng.normal(0, 0.01, n))
            elif strategy == 1:
                x = best_a.copy()
                idx = rng.choice(n, size=min(n//5, 500), replace=False)
                x[idx] *= rng.uniform(0.5, 2.0, len(idx))
            elif strategy == 2:
                # Blend with a new random init
                x = 0.7 * best_a + 0.3 * make_init(n, 'raised_cos')
            else:
                x = best_a * (1 + rng.normal(0, 0.005, n))
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            c = max(0.3, c * 0.7)
            max_stall = max(500, int(max_stall * 0.7))
            stall_count = 0
    
    return best_a, best_val

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif kind == 'exponential':
        a = np.exp(3.0 * (x - 0.5))  # Increasing
    elif kind == 'exponential_dec':
        a = np.exp(-3.0 * (x - 0.5))  # Decreasing
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'skewed':
        a = x ** 2  # Quadratic increasing
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def remez_step(a, max_cons=50, r=0.1):
    """One Remez step: solve LP to reduce max autocorrelation."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    
    # Active set: lags near the peak
    threshold = M * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    
    # Linearize: b[k](a') ≈ b[k](a) + grad_b[k](a) · (a' - a)
    # = b[k](a) + sum_j 2*a[k-j]*1[0<=k-j<n] * (a'[j] - a[j])
    # Minimize T s.t. b[k](a) + sum_j 2*a[k-j]*(a'[j]-a[j]) <= T for k in active
    #                a' >= 0, sum(a') = 1, |a'[j] - a[j]| <= r*a[j] (trust region)
    
    # This is an LP with n+1 variables and |active|+n+1 constraints
    # Use scipy for speed (cvxpy might be slow for large n)
    
    from scipy.optimize import linprog
    
    # Variables: a'[0], ..., a'[n-1], T
    # Minimize: T (index n in the variable vector)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active k: sum_j 2*a[k-j]*(a'[j]) - T <= b[k](a) - sum_j 2*a[k-j]*a[j]
    # Note: sum_j 2*a[k-j]*a[j] = b[k](a) (since b[k] = sum_j a[j]*a[k-j], and by symmetry
    #       of the gradient, sum_j 2*a[k-j]*a[j] = 2*b[k]... wait no.
    #       Actually b[k] = sum_j a[j]*a[k-j]. The gradient at a is grad[k,j] = 2*a[k-j].
    #       So grad_b[k](a) · a = sum_j 2*a[k-j]*a[j] = 2 * sum_j a[j]*a[k-j] = 2*b[k].
    #       And grad_b[k](a) · (a' - a) = 2*sum_j a[k-j]*(a'[j]-a[j]) = 2*(b_lin[k](a') - b[k](a))
    #       where b_lin[k](a') = sum_j a[k-j]*a'[j].
    #       So the linearized constraint is: b[k] + 2*(b_lin[k](a') - b[k]) <= T
    #       i.e., 2*b_lin[k](a') - b[k] <= T
    #       i.e., sum_j 2*a[k-j]*a'[j] - T <= b[k]
    
    constraints_eq = []  # Will use A_ub <= b_ub form
    A_rows = []
    b_ub_rows = []
    
    for k in active:
        k_int = int(k)
        row = np.zeros(n + 1)
        for j in range(max(0, k_int - n + 1), min(n, k_int + 1)):
            row[j] = 2.0 * a[k_int - j]
        row[n] = -1.0  # -T
        A_rows.append(row)
        b_ub_rows.append(b[k_int])
    
    # Trust region: -r*a[j] <= a'[j] - a[j] <= r*a[j]
    # a'[j] <= a[j]*(1+r)
    for j in range(n):
        row = np.zeros(n + 1)
        row[j] = 1.0
        A_rows.append(row)
        b_ub_rows.append(a[j] * (1.0 + r))
    
    # a'[j] >= a[j]*(1-r) => -a'[j] <= -a[j]*(1-r)
    for j in range(n):
        row = np.zeros(n + 1)
        row[j] = -1.0
        A_rows.append(row)
        b_ub_rows.append(-max(0, a[j] * (1.0 - r)))
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_ub_rows)
    
    # Equality: sum(a') = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a'[j] >= 0, T >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        a_new = np.maximum(result.x[:n], 0.0)
        s = a_new.sum()
        if s > 1e-10:
            return a_new / s
    return a.copy()

def remez_loop(a0, time_limit, r_init=0.05, verbose=False):
    """Iterative Remez with adaptive trust region."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = objective(a)
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    
    for t in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand = remez_step(a, max_cons=40, r=r)
        
        # Line search
        accepted = False
        for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - frac) * a + frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            val = objective(a_trial)
            if val < best_val - 1e-12:
                a = a_trial
                best_val = val
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.2, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.001, r * 0.6)
            if no_improve > 15:
                # Perturb
                rng = np.random.default_rng(t * 7 + n)
                idx = rng.choice(n, size=min(n//4, 300), replace=False)
                a[idx] *= rng.uniform(0.7, 1.4, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Remez t={t} ({elapsed:.1f}s): val={best_val:.8f}, r={r:.5f}")
    
    return best_a, best_val

def coordinate_polish(a0, time_limit, verbose=False):
    """Coordinate descent: transfer mass between pairs of positions."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_val = objective(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    max_stall = max(30, n // 50)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(50, n // 5)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-12:
                continue
            
            for frac in [0.001, 0.005, 0.01, 0.03, 0.08]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                val = objective(a_trial)
                if val < best_val - 1e-14:
                    a = a_trial
                    best_val = val
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_val

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = norm_sum1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov Mirror Descent with adaptive active-set subgradient
2. LP-based Remez with trust region and line search (scipy HiGHS)
3. Multi-length exploration (interpolating parent to longer sequences)
4. Coordinate descent with smart pair moves for final polish
5. Multi-start from diverse initializations
Key: gradient of b[k] w.r.t. a[j] = 2*a[k-j] if 0<=k-j<n else 0
"""

import numpy as np
import time
from scipy.optimize import linprog

def next_pow2(n):
    return 1 << (n - 1).bit_length()

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_pow2(size)
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def obj_val(a):
    b = fast_conv(a)
    return 2.0 * len(a) * b.max()

def subgrad_topk(a, b, top_k=40):
    n = len(a)
    M = b.max()
    threshold = M * 0.99
    near = np.where(b >= threshold)[0]
    if len(near) > top_k:
        idx = np.argsort(b)[::-1][:top_k]
    else:
        idx = near
    g = np.zeros(n)
    for k in idx:
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[ki - np.arange(lo, hi+1)]
    g /= len(idx)
    return g

def make_init(n, kind='raised_cos'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2)
    elif kind == 'exponential':
        a = np.exp(2.5 * (x - 0.5))
    elif kind == 'skewed':
        a = x ** 1.5
    elif kind == 'flat':
        a = np.ones(n)
    else:
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def remez_lp_step(a, max_cons=50, r=0.08):
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    threshold = M * 0.9995
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    A_rows = []
    b_ub = []
    for k in active:
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        row = np.zeros(n + 1)
        j_arr = np.arange(lo, hi + 1)
        row[j_arr] = 2.0 * a[ki - j_arr]
        row[n] = -1.0
        A_rows.append(row)
        b_ub.append(b[ki])
    # Trust region upper: a'[j] <= a[j]*(1+r)
    for j in range(n):
        row = np.zeros(n + 1)
        row[j] = 1.0
        A_rows.append(row)
        b_ub.append(a[j] * (1.0 + r))
    # Trust region lower: a'[j] >= a[j]*(1-r) => -a'[j] <= -a[j]*(1-r)
    for j in range(n):
        row = np.zeros(n + 1)
        row[j] = -1.0
        A_rows.append(row)
        b_ub.append(-max(0.0, a[j] * (1.0 - r)))
    A_ub = np.array(A_rows)
    b_ub = np.array(b_ub)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            a_new = np.maximum(res.x[:n], 0.0)
            s = a_new.sum()
            if s > 1e-10:
                return a_new / s
    except Exception:
        pass
    return a.copy()

def remez_loop(a0, time_limit, r_init=0.06, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = obj_val(a)
    t0 = time.monotonic()
    r = r_init
    no_imp = 0
    for t in range(1, 5000):
        if time.monotonic() - t0 > time_limit:
            break
        a_cand = remez_lp_step(a, max_cons=45, r=r)
        accepted = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_trial = (1.0 - frac) * a + frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            val = obj_val(a_trial)
            if val < best_val - 1e-13:
                a = a_trial
                best_a = a.copy()
                best_val = val
                accepted = True
                no_imp = 0
                r = min(0.25, r * 1.08)
                break
        if not accepted:
            no_imp += 1
            r = max(0.003, r * 0.6)
            if no_imp > 20:
                rng = np.random.default_rng(t * 13 + n)
                idx = rng.choice(n, size=min(n//5, 300), replace=False)
                a[idx] *= rng.uniform(0.6, 1.5, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_imp = 0
        if verbose and t % 10 == 0:
            el = time.monotonic() - t0
            print(f"    Remez t={t} ({el:.1f}s): val={best_val:.8f}, r={r:.5f}")
    return best_a, best_val

def coord_polish(a0, time_limit, verbose=False):

def interp_len(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = norm_sum1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0.0)
    s = fi.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return fi / s

        t4a = t4 * 0.55
        if t4a > 30:
            ao, _ = nmd_search(al, t4a, top_k=35, c0=3.0, verbose=False)
            try_update(ao, f"NMD_long_{n_long}")
        rem = time_budget - (time.monotonic() - t_global)
        if rem > 30:
            al2 = interp_len(best_seq, n_long)
            ao, _ = remez_loop(al2, rem - 5, r_init=0.04, verbose=False)
            try_update(ao, f"Remez_long_{n_long}")

    # Phase 5: Second length (10% time)
    rem = time_budget - (time.monotonic() - t_global)
    if rem > 80:
        n_long2 = n0 + 1000
        t5 = min(rem * 0.10, 150.0)
        print(f"\n=== P5: Long2 n={n_long2} ({t5:.0f}s) ===")
        al = interp_len(best_seq, n_long2)
        ao, _ = nmd_search(al, t5, top_k=40, c0=3.0, verbose=False)
        try_update(ao, f"NMD_long2_{n_long2}")

    # Phase 6: Final Remez + CD polish (5% time)
    rem = time_budget - (time.monotonic() - t_global)
    nb = len(best_seq) if best_seq is not None else n0
    print(f"\n=== P6: Polish n={nb} ({rem:.0f}s) ===")
    if rem > 20:
        t6a = min(rem * 0.5, 100.0)
        if t6a > 10:
            ao, _ = remez_loop(norm_sum1(best_seq), t6a, r_init=0.02, verbose=False)
            try_update(ao, f"Remez_fin_{nb}")
        rem = time_budget - (time.monotonic() - t_global)
        if rem > 10:
            ao, _ = coord_polish(norm_sum1(best_seq), rem - 3, verbose=False)
            try_update(ao, f"CD_{nb}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, t={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR no valid seq")
        return [1.0]
    return [float(x) for x in best_seq]
