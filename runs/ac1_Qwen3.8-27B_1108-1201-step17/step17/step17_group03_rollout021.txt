        # Hmm this is getting complicated. Let me use a cleaner formulation.
        # ac[k] = sum_{i} a[i] * a[k-i]
        # d/d(a[j]) ac[k] = sum_i [delta(i,j)*a[k-i] + a[i]*delta(k-i,j)]
        # = a[k-j] + a[k-j] = 2*a[k-j]  (when k-j is in [0,n-1] and j != k-j)
        # = a[k-j] = a[j]  (when 2j = k)
        
        # So: grad_a_j of LSE = beta * sum_k softmax[k] * 2*a[k-j] (with boundary)
        #                                      (use *1 instead of *2 when 2j=k)
        
        # This can be computed as a convolution!
        # 2 * conv(softmax, a) gives the gradient (approximately)
        # Actually: sum_k softmax[k] * a[k-j] = conv(softmax, a)[j] (with appropriate indexing)
        
        # Let me just use the FFT approach:
        # grad[j] = 2 * sum_k softmax[k] * a[k-j] (bounded)
        # = 2 * (conv(softmax, reversed_a))[something]
        
        # Actually the simplest: grad = 2 * correlation(softmax, a)
        # where correlation[x][j] = sum_k x[k] * a[k-j]
        
        # Using FFT:
        # cross[j] = sum_k softmax[k] * a[k-j] 
        # This is the cross-correlation of softmax and a
        # = conv(softmax, a_flipped)[j + n - 1] ... this is getting confusing
        
        # Let me just do it with a direct but vectorized approach:
        grad = np.zeros(n)
        # For efficiency, only process lags with significant softmax weight
        sig_k = np.where(softmax > 1e-8)[0]
        for k_int in sig_k:
            w = beta * softmax[k_int]
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            j_start = k_int - (i_end - 1)
            j_end = k_int - i_start + 1
            # grad[i] += w * a[k_int - i] for i in [i_start, i_end)
            # Let j = k_int - i, so j goes from k_int - i_end + 1 to k_int - i_start
            # grad[i] += w * a[j] where i = k_int - j
            idx = np.arange(i_start, i_end)
            jdx = k_int - idx
            grad[idx] += w * a[jdx]
        
        # For the diagonal correction (when 2j = k):
        # We added 2*a[j] but should have added a[j], so subtract a[j] for even k
        for k_int in sig_k:
            if k_int % 2 == 0:
                j = k_int // 2
                if 0 <= j < n:
                    grad[j] -= beta * softmax[k_int] * a[j]
        
        # Project gradient onto tangent space of simplex (subtract mean)
        grad = grad - grad.mean()
        
        # Gradient descent step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        m_new = fast_autocorr(a_new).max()
        if m_new < best_m:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD it={it} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.0f}")
    
    return best_a, best_m

def coordinate_descent_refine(a0, time_limit, verbose=True):
    """Coordinate descent: try moving mass from one position to another."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # Try to reduce peak by moving mass
        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        
        best_improvement = 0
        best_move = None
        
        # Try moving mass from positions that contribute to the peak
        for i in range(i_start, i_end):
            if a[i] < 1e-8:
                continue
            # Moving a[i] to position j changes the autocorrelation
            # Try a few random target positions
            for _ in range(5):
                j = int(rng.integers(n))
                if j == i:
                    continue
                for frac in [0.01, 0.05, 0.1]:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < M_curr - best_improvement:
                        best_improvement = M_curr - M_trial
                        best_move = (i, j, amt)
        
        if best_move is not None:
            i, j, amt = best_move
            a[i] -= amt
            a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.005, 0.02)
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

Intersected with `[0, 2n-2]`: `max(0,j) <= i < min(2n-1, n+j)`.

So the sum is over `i` from `max(0,j)` to `min(2n-2, n+j-1)`.

And the term is `softmax[i] * a[i-j]`.

I want: `sum_k softmax[k] * a[k-j]` over `k` such that `0 <= k-j < n`, i.e., `j <= k < j+n`.
Intersected with `0 <= k < 2n-1`: `max(0,j) <= k < min(2n-1, j+n)`.

These match! Great.

So `grad[j] = 2 * (softmax * a_flipped)[n-1+j]` for j = 0, ..., n-1.

And I can compute this with FFT:
```python
S = np.fft.rfft(softmax, fft_size)
Af = np.fft.rfft(a_flipped, fft_size)
cross = np.fft.irfft(S * Af, fft_size)[:size]
grad = 2.0 * cross[n-1:n-1+n]
```

This is O(n log n) which is much faster than the O(n*K) direct computation.

Now let me also think about the line search. The exact line search for Frank-Wolfe requires evaluating `M(alpha) = max_k [(1-alpha)^2 * ac_a[k] + 2*alpha*(1-alpha)*cross[k] + alpha^2 * ac_dir[k]]` for various alpha.

Each evaluation is O(n) (just a max over 2n-1 values). The cross term `cross = conv(a, direction)` can be precomputed with FFT.

A golden section search with ~50 iterations would take ~50 O(n) operations, which is fast.

But actually, we can be smarter. The function `M(alpha)` is the max of quadratics in alpha. Each quadratic `f_k(alpha) = (1-alpha)^2 * ac_a[k] + 2*alpha*(1-alpha)*cross[k] + alpha^2 * ac_dir[k]` is convex in alpha. The max of convex functions is convex. So we can use a convex optimization approach.

def fast_autocorr(a):

    size = 2 * n - 1

def fast_cross(a, b):

    return np.fft.irfft(A * B, fft_size)[:size]

def build_lp_direction(a, M, solver, top_K=500):
    """Solve LP to find descent direction."""
    n = len(a)
    ac = fast_autocorr(a)
    size = 2 * n - 1
    
    # Select constraints: those with smallest margin
    margins = M - ac
    if top_K >= size:
        active_idx = np.arange(size)
    else:
        # Add some margin to include near-binding constraints
        threshold = 0.005 * M
        active_mask = margins <= threshold
        n_active = active_mask.sum()
        if n_active > top_K:
            active_idx = np.argsort(margins)[:top_K]
        elif n_active < max(50, top_K // 4):
            active_idx = np.argsort(margins)[:max(50, top_K // 2)]
        else:
            active_idx = np.where(active_mask)[0]
    
    K = len(active_idx)
    
    # Build constraint matrix (sparse-friendly)
    T = np.zeros((K, n))
    for row in range(K):
        k_int = int(active_idx[row])
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        T[row, i_start:i_end] = a[k_int - (i_end - 1):k_int - i_start + 1][::-1]
        # T[row, i] = a[k_int - i]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

    t0 = time.monotonic()

    x = a.copy()

    while time.monotonic() - t0 < time_limit:

        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        direction = build_lp_direction(y, M_curr, solver, top_K=top_K)
        if direction is None:

            if direction is None:
                continue

                best_M_tri = M_tri
                best_alpha = alpha

        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:

            x_new = y_new.copy()
        else:

            x_new = y_new + beta_t * (y_new - y)
