    import time

    def fast_autocorr(a):

        A = np.fft.rfft(a, fft_size)
        return np.fft.irfft(A * A, fft_size)[:size]

        if s < 1e-15:
            return np.ones_like(a) / len(a)
        return a / s

            return float(evaluate_sequence(a_list))
        except Exception:

        """Compute log-sum-exp of autocorrelation/T and gradient w.r.t. a."""
        ac = fast_autocorr(a)
        ac_clip = np.maximum(ac, 1e-30)
        scaled = ac_clip / T
        max_s = scaled.max()
        exp_s = np.exp(scaled - max_s)
        sum_exp = exp_s.sum()
        lse_val = max_s + np.log(sum_exp)

        w = exp_s / sum_exp

        W = np.fft.rfft(w, fft_N)
        Af = np.fft.rfft(a_flip, fft_N)
        conv = np.fft.irfft(W * Af, fft_N)
        grad = (2.0 / T) * conv[n - 1 : n - 1 + n]
        
        return lse_val, grad, ac

        n = len(a0)

            y = np.exp(y_log - y_log.max())
            y = y / y.sum()

            lse_val, grad, ac = lse_objective_and_grad(y, T)

            M = ac.max()

                x = np.log(np.maximum(best_a, 1e-15))
                x_prev = x.copy()
                stall = 0
        
        return best_a, best_J

    def coord_descent(a0, time_limit, seed=42):
        """Simple coordinate descent: try moving mass from high-impact to low-impact positions."""
        n = len(a0)
        a = norm_sum1(a0)
        best_a = a.copy()

                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)

                idx = np.arange(i_start, i_end + 1)
                contrib = a[idx] * a[k - idx]
                
                # Move mass from highest contributor
                src = idx[np.argmax(contrib)]
                if a[src] < 1e-10:
                    continue
                
                # Find best destination (lowest impact)
                # Score: how much would adding mass here increase the max autocorr
                impact = np.zeros(n)
                for lag2 in top_lags[:5]:
                    k2 = int(lag2)
                    s2 = max(0, k2 - n + 1)
                    e2 = min(n - 1, k2)
                    if s2 <= e2:
                        idx2 = np.arange(s2, e2 + 1)
                        impact[idx2] += a[k2 - idx2]
                        impact[k2 - idx2] += a[idx2]
                
                # Destination: minimize impact, exclude source region
                impact_safe = impact.copy()
                impact_safe[src] = np.inf
                dst = np.argmin(impact_safe)
                
                # Try different fractions
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial /= s
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-14:
                        a = a_trial
                        best_m = M_trial
                        best_a = a_trial.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            
            if not improved:
                stall += 1
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.01)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        return best_a

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):

        a_list = [float(x) for x in np.asarray(a)]

    except Exception:

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def lse_subgradient(a, T):
    """Compute smoothed subgradient of max(conv) using log-sum-exp."""
    ac = fast_autocorr(a)
    ac_safe = np.maximum(ac, 1e-30)
    scaled = ac_safe / T
    max_s = scaled.max()

    W = np.fft.rfft(w, fft_N)
    Af = np.fft.rfft(a_flip, fft_N)
    conv = np.fft.irfft(W * Af, fft_N)
    cross = conv[n - 1:2 * n - 1]

    grad = 2.0 * cross
    return grad, ac

def nesterov_pg(a0, T_start, T_end, n_steps, eta_scale, verbose=False):
    """Nesterov-accelerated projected gradient with log-sum-exp smoothing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    eta = eta_scale * T_start

    for t in range(1, n_steps + 1):
        T = T_start * (T_end / T_start) ** (t / n_steps)
        T = max(T, T_end)

        beta_t = (t - 2.0) / (t + 1.0)
        if t == 1:
            y = x.copy()
        else:
            y = x + beta_t * (x - x_prev)
            y = project_simplex(y)

        grad, ac = lse_subgradient(y, T)
        M = ac.max()

        x_new = project_simplex(y - eta_t * grad)
        x_prev = x.copy()
        x = x_new

    return best_a, best_m

def coordinate_polish(a0, time_limit, seed=42):

def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss

    """Generate diverse parametric initializations."""
    candidates = []
    x = np.linspace(0.0, 1.0, n)

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        candidates.append((i_arr + 1.0) ** p)
        candidates.append((n - i_arr) ** p)

    for c in [0.3, 0.5, 0.7]:

        candidates.append(np.maximum(np.cos(np.pi * freq * x) ** 2, 0.01))

    for rate in [1, 2, 5]:
        candidates.append(np.exp(-rate * x))
        candidates.append(np.exp(rate * (x - 1.0)))

    if current_best is not None:
        candidates.append(current_best)
        i_pow = (i_arr + 1.0) ** 1.0
        for w in [0.3, 0.5, 0.7]:
            cb = current_best / current_best.sum()
            ip = i_pow / i_pow.sum()
            candidates.append(w * cb + (1 - w) * ip)

    for c in candidates:
        c = norm_sum1(c)
        m = fast_autocorr(c).max()
