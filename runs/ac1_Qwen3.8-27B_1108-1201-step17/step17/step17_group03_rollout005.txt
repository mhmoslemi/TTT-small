def compute_subgradient(a, ac, top_k=80, beta=3.0):
    """Compute weighted subgradient of max autocorrelation."""
    n = len(a)
    ac_max = ac.max()
    if ac_max < 1e-15:
        return np.zeros(n)

    top_idx = np.argsort(ac)[::-1][:top_k]
    top_vals = ac[top_idx]

    # Softmax weights over top peaks
    scale = np.max(top_vals) - np.min(top_vals)
    if scale < 1e-12:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w = np.exp(beta * (top_vals - top_vals.max()) / scale)
        w = np.maximum(w, 0)
        sw = w.sum()
        if sw < 1e-15:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w /= sw

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def mirror_descent(a0, time_limit, top_k=80, beta=3.0, c0=3.0,
                   restart_every=3000, verbose=True, rng_seed=42):
    """Mirror descent with Nesterov acceleration and backtracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov step
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-12)
        sy = y.sum()
        if sy < 1e-12:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        # Backtracking line search in log-space
        eta = gamma / (np.sqrt(t) + 1.0)
        base_m = ac.max()
        accepted = False
        for _ in range(5):
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-15)
            s_new = a_new.sum()
            if s_new < 1e-12:
                break
            a_new = a_new / s_new
            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            else:
                eta *= 0.5
        if not accepted:
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-15)
            s_new = a_new.sum()
            if s_new < 1e-12:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

            eps = 0.02

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   margin_frac=1e-3):
    """LP-based Frank-Wolfe for peak reduction."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 4:
            active_idx = np.argsort(margins)[:top_K * 4]
        elif n_active < 5:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        x_new = y_new + (t - 1.0) / (t + 2.0) * (y_new - y)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            margin_frac = max(margin_frac * 0.9, 1e-5)

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}")

    return best_a, best_m


def make_asymmetric_init(n, skew=0.3):
    """Create an asymmetric, middle-concentrated initial sequence."""
    x = np.linspace(0, 1, n)
    # Skewed beta-like distribution
    peak = 0.5 + skew * (0.3 * np.sin(np.pi * x) ** 0.5)
    a = np.exp(-((x - peak) ** 2) / (0.1 * (1 - abs(x - 0.5)) ** 2 + 0.02))
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_ramp_init(n, power=1.0):
    """Create a power-law ramp (asymmetric)."""
    x = np.linspace(0, 1, n)
    a = (1 - x) ** power
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3600
        best_seq = make_asymmetric_init(n_fb, skew=0.2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Mirror Descent (main optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Mirror Descent (n={n0}) ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t1, top_k=80, beta=3.0, c0=3.5,
                                  restart_every=3000, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "md1")

    # === Phase 2: Coordinate Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coordinate Descent (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t2, verbose=True, rng_seed=77)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd1")

    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True, top_K=200,
                                   margin_frac=2e-3)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp1")

    # === Phase 4: Mirror Descent round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: MD round 2 (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t4, top_k=100, beta=2.5, c0=2.5,
                                  restart_every=2000, verbose=True, rng_seed=123)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "md2")

    # === Phase 5: Coordinate descent round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: CD round 2 (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t5, verbose=True, rng_seed=99)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd2")

    # === Phase 6: Try asymmetric initialization + optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 200:
        print(f"\n=== Phase 6: Asymmetric init exploration (n={n_best}) ===")
        for skew in [0.15, 0.25, 0.4]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 100:
                break
            a_init = make_asymmetric_init(n_best, skew=skew)
            t6 = min(remaining * 0.25, 150.0)
            if t6 > 30:
                a_out, _ = mirror_descent(a_init, t6, top_k=80, beta=3.0, c0=4.0,
                                          restart_every=2000, verbose=False, rng_seed=skew * 100)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"asym_{skew}")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_K=300,
                                   margin_frac=5e-4)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final CD (n={len(best_seq)}) ===")
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent(a_in, t8, verbose=True, rng_seed=555)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = make_asymmetric_init(3600, skew=0.2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
