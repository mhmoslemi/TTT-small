```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2.

Strategy:
- Work with normalized nonnegative sequences.
- Use a continuation method on a smooth approximation of the max:
  (1/beta) * logsumexp(beta * C / S^2), then unwrap the logarithm efficiently.
- Optimize in a log-ratio simplex parametrization to enforce positivity and remove
  the scale invariance.
- Use FFT-based objective/gradient evaluations and L-BFGS.
- Multi-start, interpolate across lengths, and finally polish by zeroing tiny entries.
"""

import numpy as np
from scipy.optimize import minimize
from operator import itemgetter
import time

_current_beta = 1.0
_current_ref = 0
_z_indices = np.array([], dtype=np.intp)
_last_z = None
_last_f = None
_stop_z = None
_deadline = 0.0


def normalize_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if a.size == 0 or s <= 1e-15:
        n = max(2, a.size)
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    if size <= 1:
        return a * a
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def true_J(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if a.size == 0 or s <= 1e-15:
        return np.inf
    C = fast_conv(a)
    return float(2.0 * a.size * C.max() / (s * s))


def set_global_ref(n, ref):
    global _current_ref, _z_indices
    ref = int(ref)
    if ref < 0:
        ref += n
    if ref < 0 or ref >= n:
        ref = 0
    _current_ref = ref
    if n <= 1:
        _z_indices = np.array([], dtype=np.intp)
    else:
        idx = np.arange(n, dtype=np.intp)
        _z_indices = idx[idx != ref]


def a_from_z_global(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.where(np.isfinite(z), z, 0.0)
    n = z.size + 1
    ref = _current_ref
    idx = _z_indices

    m = float(np.max(np.concatenate(([0.0], z))) if z.size else 0.0)
    if not np.isfinite(m):
        m = 0.0

    e_ref = np.exp(-m)
    e_z = np.exp(z - m)
    logD = m + np.log(e_ref + e_z.sum())

    a = np.empty(n, dtype=np.float64)
    log_a_ref = -logD
    if log_a_ref > -745.0:
        a[ref] = np.exp(log_a_ref)
    else:
        a[ref] = 0.0

    if idx.size:
        log_a = z - logD
        log_a = np.maximum(log_a, -745.0)
        a[idx] = np.exp(log_a)

    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a /= s
    return a


def make_z_from_a(a, ref=None):
    a = normalize_a(a)
    n = a.size
    if n <= 1:
        return np.array([], dtype=np.float64)
    if ref is None:
        ref = int(np.argmax(a))
    ref = int(ref)
    if ref < 0 or ref >= n:
        ref = 0
    ref_val = max(float(a[ref]), 1e-300)
    idx = np.arange(n, dtype=np.intp)
    idx = idx[idx != ref]
    vals = np.maximum(a[idx], 1e-300)
    z = np.log(vals) - np.log(ref_val)
    z = np.where(np.isfinite(z), z, 0.0)
    return np.clip(z, -680.0, 680.0)


def fun_jac_z(z):
    global _last_z, _last_f

    z = np.asarray(z, dtype=np.float64).ravel()
    a = a_from_z_global(z)
    n = a.size
    s = float(a.sum())
    if n < 2 or s <= 1e-15:
        _last_z = z.copy()
        _last_f = 1e100
        return 1e100, np.zeros_like(z)

    beta = max(float(_current_beta), 1e-6)

    conv_len = 2 * n - 1
    fft_size = 1 << (3 * n + 5).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:conv_len]

    B = C / (s * s)
    bC = beta * B
    m = float(bC.max())
    if not np.isfinite(m):
        m = 0.0
        bC = np.where(np.isfinite(bC), bC, 0.0)

    e = np.exp(bC - m)
    e_sum = float(e.sum())
    if e_sum <= 1e-15 or not np.isfinite(e_sum):
        e = np.ones(C.size, dtype=np.float64) / float(C.size)
        e_sum = 1.0
    w = e / e_sum

    L = m + np.log(e_sum)
    f = (2.0 * n / beta) * L

    W_C = float(np.dot(w, C))
    W = np.fft.rfft(w, fft_size)

    h1 = np.fft.irfft(np.conj(A) * W, fft_size)[:n]
    h2 = np.fft.irfft(A * W, fft_size)[:n]
    h = h1 + h2
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)

    ga = 2.0 * n * (h / (s * s) - 2.0 * W_C / (s ** 3))
    ga = np.where(np.isfinite(ga), ga, 0.0)

    dot = float(np.dot(a, ga))
    gz = a[_z_indices] * (ga[_z_indices] - dot)
    gz = np.where(np.isfinite(gz), gz, 0.0)

    if not np.isfinite(f):
        f = 1e100
        gz = np.zeros_like(z)

    _last_z = z.copy()
    _last_f = f
    return f, gz


def time_callback(x):
    global _last_z, _stop_z
    x = np.asarray(x, dtype=np.float64)
    if _last_f is not None and np.isfinite(_last_f):
        _last_z = x.copy()
    if time.monotonic() > _deadline:
        _stop_z = x.copy()
        raise TimeoutError("st")


def choose_betas(a):
    a = normalize_a(a)
    C = fast_conv(a)
    cmax = float(C.max())
    if cmax <= 1e-15:
        return [10.0, 50.0, 200.0, 1000.0, 5000.0]

    Ks = (2.0, 10.0, 50.0, 200.0, 1000.0)
    bcap = 2000000.0
    raw = []
    for K in Ks:
        b = max(10.0, K / cmax)
        raw.append(min(bcap, b))

    betas = []
    for b in sorted(raw):
        if not betas or b > betas[-1] * 1.7:
            betas.append(float(b))
    if not betas:
        betas = [10.0]
    return betas[:5]


def optimize_z(a0, time_limit, seed=0, verbose=False):
    global _deadline, _stop_z, _current_beta

    a0 = normalize_a(a0)
    n = a0.size
    if n < 2:
        return a0.copy(), true_J(a0)

    best_a = a0.copy()
    best_val = true_J(best_a)

    ref = int(np.argmax(best_a))
    set_global_ref(n, ref)
    z = make_z_from_a(best_a, ref)

    t0 = time.monotonic()
    betas = choose_betas(best_a)
    m = len(betas)
    last_improved = -10

    for idx, b in enumerate(betas):
        now = time.monotonic()
        remaining_total = time_limit - (now - t0)
        if remaining_total < 5.0:
            break

        cumul = ((idx + 1) ** 1.7) / (m ** 1.7)
        stage_deadline = t0 + time_limit * cumul
        if stage_deadline - now < 3.0:
            continue

        _stop_z = None
        _current_beta = float(b)
        _deadline = min(stage_deadline, t0 + time_limit)

        maxiter = int(max(20, min(6000, (_deadline - time.monotonic()) * 30)))
        try:
            res = minimize(
                fun_jac_z,
                z.copy(),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": maxiter,
                    "maxfun": max(1000, maxiter * 10),
                    "gtol": 1e-7,
                    "ftol": 1e-12,
                    "maxls": 50,
                    "m": 15,
                },
                callback=time_callback,
            )
            new_z = np.asarray(res.x, dtype=np.float64)
            if new_z.size == z.size and np.all(np.isfinite(new_z)):
                z = new_z
        except TimeoutError:
            if _stop_z is not None and np.all(np.isfinite(_stop_z)):
                z = _stop_z.copy()
            elif _last_z is not None and np.all(np.isfinite(_last_z)):
                z = _last_z.copy()
        except Exception:
            pass

        a = a_from_z_global(z)
        if a.size != n:
            a = best_a.copy()
        val = true_J(a)

        if val < best_val - 1e-14:
            best_val = val
            best_a = a.copy()
            last_improved = idx
            if verbose:
                print(f"    [beta={b:.1e}] J={val:.6f} t={time.monotonic()-t0:.1f}s")
        else:
            if idx - last_improved >= 2:
                rng = np.random.default_rng(9000 + int(seed) * 31 + idx * 17)
                base = a[_z_indices]
                scale = 0.035 * np.sqrt(np.maximum(base, 1e-12))
                if scale.size:
                    z = z + rng.normal(0.0, 1.0, z.size) * scale

        if a.size == n and (
            a[ref] < 1e-7
            or (z.size > 0 and (np.max(z) > 600.0 or np.min(z) < -600.0))
            or ref < 0 or ref >= n
        ):
            ref2 = int(np.argmax(a))
            set_global_ref(n, ref2)
            z = make_z_from_a(a, ref2)
            ref = ref2

    a = a_from_z_global(z)
    if a.size == n:
        val = true_J(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return normalize_a(best_a), float(best_val)


def interpolate_seq(a, new_n):
    a = normalize_a(a)
    new_n = max(2, int(new_n))
    if a.size == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    b = np.interp(new_x, old_x, a)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s <= 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    return b / s


def generate_inits(n, seed=1):
    n = max(2, int(n))
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    out = []

    out.append(np.ones(n, dtype=np.float64) / float(n))

    for sig in (0.28, 0.18, 0.10, 0.06):
        a = np.exp(-((x - 0.5) / sig) ** 2)
        out.append(normalize_a(a))

    for amp in (0.35, 0.70, 0.95):
        a = np.maximum(1.0 + amp * np.cos(2.0 * np.pi * x), 1e-5)
        out.append(normalize_a(a))

    a = (1.0 - x) ** 0.65 * (1.0 + 0.6 * np.sin(2.0 * np.pi * x))
    out.append(normalize_a(a))

    a = x ** 0.55 * (1.0 + 0.5 * np.cos(4.0 * np.pi * x))
    out.append(normalize_a(a))

    r = np.abs(rng.normal(0.0, 1.0, n)) + 0.25
    if n > 5:
        kern = np.array([0.05, 0.2, 0.5, 0.2, 0.05], dtype=np.float64)
        kern /= kern.sum()
        r = np.convolve(r, kern, mode="same")
    out.append(normalize_a(r))

    return out


def best_init_for_length(n, parent, seed):
    n = max(2, int(n))
    cands = []
    if parent is not None:
        cands.append(interpolate_seq(parent, n))
    cands.extend(generate_inits(n, seed))

    best = None
    bestv = np.inf
    for a in cands:
        a = normalize_a(a)
        v = true_J(a)
        if v < bestv:
            bestv = v
            best = a.copy()
    if best is None:
        best = np.ones(n, dtype=np.float64) / float(n)
        bestv = true_J(best)
    return best, float(bestv)


def add_candidate(pool, a, label):
    if a is None:
        return
    a = normalize_a(a)
    if a.size < 2 or a.size > 8000:
        return
    v = true_J(a)
    if np.isfinite(v):
        pool.append((a, float(v), label))


def update_best(best_a, best_val, a, label):
    if a is None:
        return best_a, best_val
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size < 2 or a.size > 8000:
        return best_a, best_val
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-12:
        return best_a, best_val
    a = a / s
    v = true_J(a)
    if np.isfinite(v) and v < best_val:
        print(f"  *** NEW BEST [{label}]: J={v:.6f}, n={a.size}")
        return a.copy(), float(v)
    return best_a, best_val


def postprocess(a):
    a = normalize_a(a)
    best_a = a.copy()
    best_val = true_J(best_a)
    amax = float(np.max(best_a)) if best_a.size else 1.0

    thresholds = [
        0.0,
        1e-15,
        1e-12,
        1e-10,
        1e-8,
        1e-6,
        1e-5 * amax,
        1e-4 * amax,
        1e-3 * amax,
    ]

    for th in thresholds:
        b = best_a.copy()
        if th > 0.0:
            cutoff = max(th, 1e-15)
            mask = b < cutoff
            if int(np.count_nonzero(mask)) > 0.25 * b.size:
                continue
            b[mask] = 0.0
        s = b.sum()
        if s <= 1e-12:
            continue
        b = b / s
        v = true_J(b)
        if v < best_val - 1e-14:
            best_val = v
            best_a = b.copy()

    return best_a


def propose_candidate(budget_s=1800.0):
    t_start = time.monotonic()
    time_limit = min(float(budget_s) - 120.0, 1680.0)
    if time_limit < 60.0:
        time_limit = 60.0

    best_a = None
    best_val = np.inf

    parent = None
    raw = globals().get("height_sequence_1", None)
    if raw is not None:
        try:
            p = np.asarray(raw, dtype=np.float64).ravel()
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.maximum(p, 0.0)
            s = p.sum()
            if p.size >= 2 and s > 1e-8:
                parent = p / s
        except Exception:
            parent = None

    pool = []

    if parent is not None:
        add_candidate(pool, parent, "parent")
        rng = np.random.default_rng(42)
        for ns in (0.004, 0.015, 0.04):
            pp = parent * np.exp(rng.normal(0.0, ns, len(parent)))
            add_candidate(pool, pp, "parent_noise_" + repr(ns))

    lengths = set()
    if parent is not None:
        npar = len(parent)
        for f in (0.75, 0.90, 1.05, 1.20, 1.40, 1.60):
            nl = int(round(npar * f))
            if 1500 <= nl <= 8000:
                lengths.add(nl)
    else:
        lengths = {2500, 3200, 4000, 5000, 6000}

    if len(lengths) < 8:
        for nl in (7000, 8000):
            if 1500 <= nl <= 8000:
                lengths.add(nl)

    for nl in sorted(lengths):
        a, _ = best_init_for_length(nl, parent, int(nl) % 997 + 1)
        add_candidate(pool, a, "len_" + str(nl))

    if not pool:
        a, _ = best_init_for_length(3000, parent, 17)
        add_candidate(pool, a, "fallback")

    pool.sort(key=itemgetter(1))

    best_by_len = {}
    best_noise = None
    for a, v, lbl in pool:
        key = len(a)
        if key not in best_by_len or v < best_by_len[key][1]:
            best_by_len[key] = (a, v, lbl)
        if "noise" in lbl and (best_noise is None or v < best_noise[1]):
            best_noise = (a, v, lbl)

    tasks = list(best_by_len.values())
    tasks.sort(key=itemgetter(1))

    if best_noise is not None:
        if len(tasks) < 5:
            tasks.append(best_noise)
        elif best_noise[1] < tasks[-1][1] and best_noise is not tasks[-1]:
            tasks[-1] = best_noise
    tasks = tasks[:5]

    if not tasks:
        a, _ = best_init_for_length(3000, parent, 1)
        tasks = [(a, true_J(a), "still_fallback")]

    print("Initial candidates:")
    for a, v, lbl in tasks:
        print(f"  {lbl}: J={v:.6f}, n={len(a)}")

    phase1_time = min(time_limit * 0.55, 1050.0)
    fracs = (0.35, 0.25, 0.20, 0.15, 0.05)

    for idx, item in enumerate(tasks):
        a, v0, lbl = item
        now = time.monotonic()
        remaining = time_limit - (now - t_start)
        if remaining < 45.0:
            break

        t_alloc = phase1_time * fracs[idx] if idx < len(fracs) else 40.0
        t_alloc = min(t_alloc, remaining - 60.0)
        if t_alloc < 25.0:
            continue

        print(f"\nInitializing task {idx}: {lbl}, initial J={v0:.6f}, time={t_alloc:.0f}s")
        a_out, v_out = optimize_z(a, t_alloc, seed=100 + idx, verbose=(idx == 0))
        best_a, best_val = update_best(best_a, best_val, a_out, lbl)

    if best_a is not None:
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining > 150.0:
            nbest = len(best_a)
            variants = set()
            for f in (0.80, 0.90, 1.10, 1.25, 1.45):
                nl = int(round(nbest * f))
                if 1500 <= nl <= 8000 and abs(nl - nbest) > 20:
                    variants.add(nl)
            for nl in (3200, 4000, 5000, 6500):
                if 1500 <= nl <= 8000 and abs(nl - nbest) > 20:
                    variants.add(nl)

            variants = sorted(variants)[:5]
            if variants:
                print("\nLength variants:")
                for i, nl in enumerate(variants):
                    now = time.monotonic()
                    rem = time_limit - (now - t_start)
                    if rem < 45.0:
                        break
                    a = interpolate_seq(best_a, nl)
                    t_alloc = max(20.0, min(rem - 60.0, (rem - 60.0) / max(1, len(variants) - i)))
                    print(f"  variant n={nl}, time={t_alloc:.0f}s")
                    a_out, _ = optimize_z(a, t_alloc, seed=2000 + nl, verbose=False)
                    best_a, best_val = update_best(best_a, best_val, a_out, "variant_" + str(nl))

    if best_a is not None:
        rng = np.random.default_rng(1937)
        for k in range(4):
            now = time.monotonic()
            rem = time_limit - (now - t_start)
            if rem < 45.0:
                break
            noise = 0.003 + 0.004 * k
            a = best_a * np.exp(rng.normal(0.0, noise, len(best_a)))
            t_alloc = min(rem - 30.0, 140.0)
            if t_alloc < 20.0:
                break
            print(f"\nFinal perturbation {k}, noise={noise:.4f}, time={t_alloc:.0f}s")
            a_out, _ = optimize_z(a, t_alloc, seed=5000 + k, verbose=False)
            best_a, best_val = update_best(best_a, best_val, a_out, "final_" + str(k))

    if best_a is None:
        best_a = np.ones(3000, dtype=np.float64) / 3000.0

    best_a = postprocess(best_a)

    if best_a.size > 8000:
        best_a = normalize_a(best_a[:8000])

    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s <= 1e-8:
        best_a = np.ones(3000, dtype=np.float64) / 3000.0
    else:
        best_a = best_a / s

    eval_func = globals().get("evaluate_sequence", None)
    if eval_func is not None:
        try:
            exact = eval_func([float(x) for x in best_a])
            print(f"Exact evaluation of proposed sequence: {exact:.6f}")
        except Exception:
            exact = np.inf
        if not np.isfinite(exact):
            if parent is not None:
                best_a = normalize_a(parent)
            else:
                best_a = np.ones(3000, dtype=np.float64) / 3000.0

    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s <= 0.01:
        best_a = np.ones(3000, dtype=np.float64) / 3000.0

    return [float(x) for x in best_a]
```
