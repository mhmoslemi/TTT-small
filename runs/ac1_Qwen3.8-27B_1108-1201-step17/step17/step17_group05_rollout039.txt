```python
    grad = 2.0 * conv_result[n-1 : 2*n-1]
    
    return grad, ac.max()
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated mirror descent
with adaptive soft-max subgradient on the probability simplex.
Key innovations: FFT-based O(n log n) subgradient, Nesterov acceleration,
adaptive sharpening, multi-scale length search, and targeted peeling.
"""

import numpy as np
import time

def fast_conv_self(a):
    """Compute conv(a, a) = full linear self-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_soft_subgrad(a, beta):
    """
    Compute the soft-max subgradient of max_k conv(a,a)[k].
    Returns (gradient, max_value).
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Self-convolution
    A = np.fft.rfft(a, fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]
    M = ac.max()
    
    # Soft-max weights
    shifted = ac - M
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(size) / size
    else:
        w = w / w_sum
    
    # Gradient: g[i] = 2 * sum_m w[m+i] * a[m]
    # = 2 * (a_rev * w)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_len = n + size - 1  # 3n-2
    fft_sg = 1 << (conv_len - 1).bit_length()
    AR = np.fft.rfft(a_rev, fft_sg)
    W = np.fft.rfft(w, fft_sg)
    conv_result = np.fft.irfft(AR * W, fft_sg)
    grad = 2.0 * conv_result[n - 1 : 2 * n - 1]
    
    return grad, M

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def mirror_descent_nesterov(a0, time_limit, beta_start=5.0, beta_end=30.0,
                             c_step=10.0, momentum_start=0.0, momentum_end=0.9,
                             restart_after=3000, seed=42, verbose=True):
    """
    Nesterov-accelerated mirror descent on simplex.
    """
    n = len(a0)
    a = normalize(a0)
    a_prev = a.copy()
    y = a.copy()
    
    best_a = a.copy()
    best_m = fast_conv_self(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    iteration = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        elapsed = time.monotonic() - t0
        
        # Adaptive parameters
        progress = min(1.0, iteration / (restart_after * 3))
        beta = beta_start + (beta_end - beta_start) * progress
        momentum = momentum_start + (momentum_end - momentum_start) * progress
        eta = c_step / (np.sqrt(iteration) + 1.0)
        
        # Nesterov extrapolation
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Compute subgradient
        g, M_curr = compute_soft_subgrad(y, beta)
        
        # Mirror descent step
        # a_new[i] = y[i] * exp(-eta * g[i]) / Z
        log_step = -eta * g
        log_step -= log_step.max()  # numerical stability
        a_new = y * np.exp(log_step)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        a_prev = a
        a = a_new
        
        # Evaluate
        M_new = fast_conv_self(a).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart logic
        if stall > restart_after:
            stall = 0
            # Perturb around best
            noise_scale = max(0.001, 0.02 * (1 - 0.5 * min(1.0, stall / restart_after)))
            noise = rng.normal(0, noise_scale, n)
            a_pert = best_a * (1 + noise)
            a = normalize(a_pert)
            a_prev = a.copy()
            
            if verbose:
                J = 2.0 * n * best_m
                print(f"    MD restart at iter {iteration} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")
        
        if verbose and iteration % 5000 == 0:
            J = 2.0 * n * best_m
            print(f"    MD iter {iteration} ({elapsed:.0f}s): J={J:.6f}, stall={stall}, beta={beta:.1f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True, seed=0):
    """
    Fine-grained coordinate descent polish.
    Tries to move small amounts of mass between pairs of positions.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv_self(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv_self(a)
        M_curr = ac.max()
        k_star = int(np.argmax(ac))
        
        # Find the most contributing pairs at the peak lag
        # conv_k[a][i] = a[i] * a[k-i]
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]
        
        # Sort by contribution
        order = np.argsort(contrib)[::-1]
        top_idx = idx[order[:10]]
        
        improvement = False
        for src in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Try moving a fraction to a random low-contribution position
            for _ in range(5):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                
                for frac in [0.01, 0.005, 0.02, 0.003, 0.001]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    trial = a.copy()
                    trial[src] -= amt
                    trial[dst] += amt
                    trial = normalize(trial)
                    M_trial = fast_conv_self(trial).max()
                    if M_trial < best_m - 1e-14:
                        a = trial
                        best_m = M_trial
                        best_a = a.copy()
                        improvement = True
                        stall = 0
                        break
                if improvement:
                    break
            if improvement:
                break
        
        if not improvement:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 50 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.01)
                    a[i] -= amt
                    a[j] += amt
                    a = normalize(a)
                    M_new = fast_conv_self(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def make_init(n, seed=0, method="smooth"):
    """Create various initializations."""
    rng = np.random.default_rng(seed)
    if method == "uniform":
        return np.ones(n) / n
    elif method == "smooth":
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for f in range(1, 8):
            a += rng.normal(0, 0.05) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2*np.pi))
        return normalize(a)
    elif method == "bimodal":
        # Two peaks
        centers = [n // 4, 3 * n // 4]
        sigma = n // 20
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-0.5 * (np.arange(n) - c) ** 2 / sigma ** 2)
        return normalize(a)
    elif method == "quadratic_residue":
        # Based on quadratic residues
        p = n  # Use n as the modulus (not necessarily prime)
        qr = set()
        for x in range(p):
            qr.add((x * x) % p)
        a = np.zeros(n)
        for r in qr:
            if r < n:
                a[r] = 1.0
        return normalize(a)
    elif method == "thue_morse":
        # Thue-Morse based
        tm = np.zeros(n)
        for i in range(n):
            # Popcount parity
            c = bin(i).count('1')
            tm[i] = 1.0 if c % 2 == 0 else 0.5
        return normalize(tm)
    else:
        return np.ones(n) / n
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated mirror descent
with adaptive soft-max subgradient, Nesterov momentum, targeted peak
equalization, and multi-length search. Uses FFT for O(n log n) subgradients.

import numpy as np
import time


def fast_conv_self(a):
    """Compute full linear self-convolution conv(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_soft_subgrad(a, beta):
    """
    Compute soft-max subgradient of max_k conv(a,a)[k].
    Returns (gradient of length n, max convolution value).
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]
    M = ac.max()

    shifted = ac - M
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(size) / size
    else:
        w = w / w_sum

    # grad[i] = 2 * sum_m w[m+i] * a[m] = 2*(a_rev * w)[n-1:2n-1]
    a_rev = a[::-1]
    conv_len = n + size - 1
    fft_sg = 1 << (conv_len - 1).bit_length()
    AR = np.fft.rfft(a_rev, fft_sg)
    W = np.fft.rfft(w, fft_sg)
    conv_result = np.fft.irfft(AR * W, fft_sg)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]

    return grad, M


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def make_smooth_init(n, seed=0, n_freqs=6, amplitude=0.06):
    """Smooth random initialization via Fourier series."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freqs + 1):
        coeff = rng.normal(0, amplitude)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    return normalize(a)


def make_symmetric_init(n, seed=0):
    """Symmetric smooth initialization."""
    a = make_smooth_init(n, seed=seed)
    a_sym = 0.5 * (a + a[::-1])

                    trial = a.copy()
                    trial[partner] -= amt
                    trial[dst] += amt
                    trial = normalize(trial)
                    M_trial = fast_conv_self(trial).max()
                    if M_trial < best_m - 1e-14:
                        a = trial
                        best_m = M_trial
                        best_a = a.copy()
                        improvement_found = True
                        stall = 0
                        break
                if improvement_found:
                    break

        if not improvement_found:
            stall += 1
            # Periodic random perturbation
            if stall % 30 == 0:
                n_move = int(rng.integers(1, 4))
                for _ in range(n_move):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.01)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                a = normalize(a)
                M_new = fast_conv_self(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakEq ({elapsed:.0f}s): J={J:.6f}, stall={stall}, k*={k_star}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, seed=0):
    """
    Pairs-based coordinate descent: try moving mass between pairs of
    positions to reduce the peak convolution.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv_self(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv_self(a)
        k_star = int(np.argmax(ac))

        # Get top contributing pairs at peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        products = a[idx] * a[k_star - idx]
        order = np.argsort(products)[::-1]
        top_i = idx[order[:5]]

        improvement = False
        for src in top_i:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue

            # Try moving to several random destinations
            for _ in range(10):
                dst = int(rng.integers(n))
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.003, 0.008]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    trial = a.copy()
                    trial[src] -= amt
                    trial[dst] += amt
                    trial = normalize(trial)
                    M_trial = fast_conv_self(trial).max()
                    if M_trial < best_m - 1e-14:
                        a = trial
                        best_m = M_trial
                        best_a = a.copy()
                        improvement = True
                        stall = 0
                        break
                if improvement:
                    break
            if improvement:
                break

        if not improvement:
            stall += 1
            if stall % 50 == 0:
                # Random pair move
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.001, 0.008)
                    a[i] -= amt
                    a[j] += amt
                    a = normalize(a)
                    M_new = fast_conv_self(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator. Multi-phase optimization combining
    mirror descent, peak equalization, coordinate polish, and
    length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_list = [float(x) for x in a]
        try:
            sc = evaluate_sequence(a_list)
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_list = [float(x) for x in p]
            p_score = evaluate_sequence(p_list)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base, seed=1)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Mirror descent on parent length with multiple inits ===
    print(f"\n=== Phase 1: Mirror Descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 550.0)

    if t_phase1 > 60:
        # Initialize with parent + diverse inits
        inits_list = []
        labels_list = []

        inits_list.append(normalize(best_seq))
        labels_list.append("parent")

        inits_list.append(make_symmetric_init(n0, seed=42))
        labels_list.append("sym")

        inits_list.append(make_smooth_init(n0, seed=77, n_freqs=4, amplitude=0.04))
        labels_list.append("smooth_low")

        t_per_init = t_phase1 / len(inits_list)

        for ii, (init, lbl) in enumerate(zip(inits_list, labels_list)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_avail = min(t_per_init, remaining - 120)
            if t_avail < 20:
                break
            print(f"  MD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = mirror_descent_optimize(
                init, t_avail,
                beta_start=5.0, beta_end=25.0,
                c_step=8.0, mom_start=0.0, mom_end=0.85,
                restart_after=2000, seed=100 + ii,
                verbose=False
            )
            _try_update(a_out, f"md_{lbl}")

    # === Phase 2: Extended MD on best with sharper beta ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MD Sharp (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_in = normalize(best_seq)
        a_out, _ = mirror_descent_optimize(
            a_in, t2,
            beta_start=15.0, beta_end=50.0,
            c_step=4.0, mom_start=0.4, mom_end=0.9,
            restart_after=3000, seed=500,
            verbose=True
        )
        _try_update(a_out, "md_sharp")

    # === Phase 3: Peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak Equalization ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = normalize(best_seq)
        a_out, _ = peak_equalization(a_in, t3, verbose=True, seed=42)
        _try_update(a_out, "peak_eq")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coordinate Polish ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=True, seed=123)
        _try_update(a_out, "coord")

    # === Phase 5: Another MD round with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: MD Round 2 ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_in = normalize(best_seq)
        a_out, _ = mirror_descent_optimize(
            a_in, t5,
            beta_start=8.0, beta_end=35.0,
            c_step=6.0, mom_start=0.2, mom_end=0.9,
            restart_after=2500, seed=888,
            verbose=True
        )
        _try_update(a_out, "md_r2")

    # === Phase 6: Try a different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    alt_lengths = []
    for delta in [-400, 400, 800]:
        alt_n = n_best + delta
        if 1500 <= alt_n <= 5000:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n=== Phase 6: Alt Length n={alt_n} ===")

        # Interpolate best to new length
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, alt_n)
        a_interp = np.interp(x_new, x_old, normalize(best_seq))
        a_interp = normalize(a_interp)

        t6 = min(remaining * 0.4, 150.0)
        if t6 > 20:
            a_out, _ = mirror_descent_optimize(
                a_interp, t6,
                beta_start=6.0, beta_end=25.0,
                c_step=7.0, mom_start=0.0, mom_end=0.8,
                restart_after=2000, seed=777,
                verbose=False
            )
            _try_update(a_out, f"md_n{alt_n}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final Polish ===")
    t7 = min(remaining * 0.4, 100.0)
    if t7 > 10:
        a_in = normalize(best_seq)
        a_out, _ = peak_equalization(a_in, t7, verbose=False, seed=333)
        _try_update(a_out, "final_peakeq")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7b = min(remaining * 0.7, 60.0)
        if t7b > 10:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t7b, verbose=False, seed=444)
            _try_update(a_out, "final_coord")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        best_seq = make_symmetric_init(3200, seed=1)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
