def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def smoothed_max_descent(a0, time_limit, tau_init=0.05, top_k=300, verbose=True):

        grad = compute_gradient_smoothmax(a, tau=tau, top_k=top_k)
        
        # Projected gradient step (stay in simplex)

        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - alpha * grad
        log_a = log_a - log_a.max()
        a_new = np.exp(log_a)

        if m_new < best_m - 1e-15:

            # Small improvement, accept
            a = a_new
            stall += 1
        else:
            a = a_new  # Still move (momentum-like)
            stall += 1
        
        a = a_new
        
        # Adaptive parameters
        if stall > max_stall:
            # Restart with perturbation
            stall = 0
            tau = max(tau * 0.7, 0.005)
            alpha = min(alpha * 1.3, 1.0)

            a = np.maximum(a, 1e-10)

        # Slowly decrease tau (annealing)

        if verbose and stall % 50 == 0:

            J = 2.0 * n * best_m
            print(f"    SM t=(elapsed:.0f}s: J={J:.6f}, tau={tau:.4f}, stall={stall}")

def peak_crush(a0, time_limit, verbose=True, rng_seed=123):

        total_contrib = contrib.sum()

            stall += 1
            continue

            best_move = None

                    if dst == src:

                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    LP Frank-Wolfe with adaptive margin and constraint selection.
    Minimizes max_k conv_k(a) over the simplex.

        margins = M_curr - ac
        threshold = margin_frac * M_curr

        n_active = int(active_mask.sum())
        
        # Cap the number of constraints for LP efficiency
        max_constraints = 600
        if n_active > max_constraints:
            # Use only the top max_constraints peaks
            active_idx = np.argsort(margins)[:max_constraints]
        elif n_active < 10:
            # Always include at least some constraints
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        T_sel = np.zeros((K, n))

        b_val = np.maximum(b_val, 0.0)

        if b_sum < 1e-15:

        g_dir = b_val / b_sum
        
        # FW step with step size
        # Use diminishing step: alpha_t = 2/(t+2) or constant
        alpha_t = alpha_lr * 2.0 / (np.sqrt(t) + 1.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new
        
        # Update best
        my = fast_autocorr(y).max()
        if my < best_m - 1e-15:
            best_m = my
            best_a = y.copy()
            stall = 0
            # Tighten margin on improvement
            margin_frac = max(margin_frac * 0.97, 5e-5)
            # Increase top_K slightly
            top_K = min(top_K + 5, 500)
        else:
            stall += 1
        
        # Restart if stalled
        if stall > max_stall:
            stall = 0
            # Relax margin
            margin_frac = min(margin_frac * 1.5, 5e-3)
            top_K = max(top_K - 50, 100)
            # Perturb from best
            pert_scale = 0.02
            pert = rng.normal(0, pert_scale, n) * best_a
            y = best_a + pert
            y = np.maximum(y, 1e-10)
            y = y / y.sum()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}, stall={stall}")
    
    return best_a, best_m


def smoothmax_descent(a0, time_limit, verbose=True):
    """
    Optimizes using smoothed max (log-sum-exp) of the autocorrelation.
    Uses exponential family (mirror) descent on the simplex.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)
    
    tau = 0.05  # Temperature for log-sum-exp
    eta = 0.5   # Step size
    top_k = 150
    
    stall = 0
    max_stall = 100
    momentum = 0.9
    vel = np.zeros(n)
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        # Compute soft-max weights
        ac_shifted = ac - ac.max()
        weights = np.exp(ac_shifted / tau)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones_like(ac) / len(ac)
        else:
            weights = weights / w_sum
        
        # Focus on top peaks
        top_idx = np.argsort(ac)[::-1][:top_k]
        
        # Compute gradient
        grad = np.zeros(n)
        for k in top_idx:
            w = weights[int(k)]
            if w < 1e-10:
                continue
            k_i = int(k)
            m_start = max(0, k_i - n + 1)
            m_end = min(n, k_i + 1)
            if m_start >= m_end:
                continue
            m_idx = np.arange(m_start, m_end)
            j_idx = k_i - m_idx
            grad[m_idx] += 2.0 * w * a[j_idx]
        
        # Mirror descent step with momentum
        vel = momentum * vel + (1 - momentum) * grad
        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - eta * vel
        log_a = log_a - log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a = a_new
        
        m_new = fast_autocorr(a).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive temperature
        tau = max(tau * 0.998, 0.01)
        eta = max(eta * 0.999, 0.05)
        
        if stall > max_stall:
            stall = 0
            # Restart
            pert = rng.normal(0, 0.015, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            vel = np.zeros(n)
            tau = 0.05
            eta = 0.3
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SM t={t} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.4f}, stall={stall}")
    
    return best_a, best_m


def peak_crush(a0, time_limit, verbose=True, seed=42):

def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score
