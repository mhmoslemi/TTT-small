def fast_autocorr(a):
    """Compute full conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the objective using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]

def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    # Euclidean projection onto simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def make_exp_ramp(n, rate, seed=0):
    """Exponential ramp: a[i] proportional to exp(rate * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(rate * i)
    return norm_sum1(a)


def make_cosine_taper(n, seed=0):
    """Raised cosine taper."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 1e-10)
    return norm_sum1(a)


def make_smooth_asym(n, seed=0, freqs=None):
    """Smooth asymmetric initialization."""
    rng = np.random.default_rng(seed)
    i = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    if freqs is None:
        freqs = [1, 2, 3, 5]
    for f in freqs:
        coeff = rng.normal(0, 0.15)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * i + phase)
    a += 0.1 * rng.normal(0, 1, n)
    return norm_sum1(a)


def make_waterfall(n, alpha=1.0):
    """Waterfall shape: a[i] = (n-i)^alpha / sum."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** alpha
    return norm_sum1(a)


def make_ramp_up(n, alpha=1.0):
    """Ramp up shape: a[i] = (i+1)^alpha / sum."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha
    return norm_sum1(a)


def subgradient_optimize(a0, time_limit, verbose=False, top_k=20, c0=3.0, seed=42):
    """
    Subgradient descent on the simplex with active-set focusing and backtracking.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    c = fast_autocorr(a)
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    gamma = c0
    stall = 0
    max_stall = 3000
    eta = gamma

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        M_curr = c.max()

        # Find active set
        k_use = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_use]
        # Include only lags within 1% of max
        threshold = M_curr * 0.99
        active = [idx for idx in top_idx if c[idx] >= threshold]
        if not active:
            active = [top_idx[0]]

        # Compute subgradient: average gradient over active lags
        g = np.zeros(n)
        for k in active:
            ki = int(k)
            # grad_k[j] = 2 * a[ki - j] if 0 <= ki - j < n
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                g[js] += 2.0 * a[ki - js]
        g /= len(active)

        # Backtracking line search
        alpha_step = eta
        improved = False
        for _ in range(30):
            a_new = a - alpha_step * g
            a_new = project_simplex(a_new)
            c_new = fast_autocorr(a_new)
            M_new = c_new.max()
            if M_new < M_curr - 1e-15:
                a = a_new
                improved = True
                break
            alpha_step *= 0.7

        if not improved:
            # Try random perturbation
            pert = rng.normal(0, 0.01 * eta, n)
            a_new = project_simplex(a + pert)
            c_new = fast_autocorr(a_new)
            if c_new.max() < M_curr:
                a = a_new
            else:
                stall += 1
                if stall > max_stall:
                    break
                # Decrease step size
                eta *= 0.9
                if eta < 1e-8:
                    break
        else:
            stall = 0
            # Slowly increase step size
            eta = min(eta * 1.001, 5.0)

        c = fast_autocorr(a)
        M_curr = c.max()
        if M_curr < best_m:
            best_m = M_curr
            best_a = a.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Subgrad ({elapsed:.0f}s): J={J:.6f}, top_k={top_k}")

def partner_coord_descent(a0, time_limit, verbose=False, seed=42, top_k=15):
    """
    Partner-based coordinate descent.
    Move mass from positions with heavy 'partner' at peak lags to positions
    with light partners.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    c = fast_autocorr(a)
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500
    move_count = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        M_curr = c.max()

        # Find top active lags
        k_use = min(top_k, len(c))
        top_lags = np.argsort(c)[::-1][:k_use]
        threshold = M_curr * 0.995
        active_lags = [int(idx) for idx in top_lags if c[idx] >= threshold]
        if not active_lags:
            active_lags = [int(top_lags[0])]

        # Compute partner sensitivity for each position
        # sensit[j] = max over active lags k of a[k-j] (if valid)
        sensit = np.zeros(n)
        for k in active_lags:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                partners = a[k - js]
                sensit[js] = np.maximum(sensit[js], partners)

        # Also compute "global" sensitivity: max over ALL lags where a has mass
        # This helps avoid creating new peaks
        global_sens = np.zeros(n)
        nonzero_lags = np.where(c > M_curr * 0.5)[0]
        for k in nonzero_lags[:min(50, len(nonzero_lags))]:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                partners = a[ki - js]
                global_sens[js] = np.maximum(global_sens[js], partners)

        # Find best (source, dest) pair
        # Source: high sensit, has mass
        # Dest: low global_sens, far from source
        has_mass = a > best_m * 0.01  # Only consider positions with meaningful mass
        if not np.any(has_mass):
            stall += 1
            continue

        # Score sources by sensitivity * mass
        src_scores = sensit * a
        # Score destinations by inverse global sensitivity
        dst_scores = -global_sens  # We want low global sensitivity

        # Try top source-destination pairs
        n_try = min(20, n)
        src_candidates = np.where(has_mass)[0]
        if len(src_candidates) == 0:
            stall += 1
            continue

        # Sort sources by score
        src_order = src_candidates[np.argsort(-src_scores[src_candidates])][:min(10, len(src_candidates))]

        best_gain = 0
        best_move = None

        for s in src_order:
            s = int(s)
            if a[s] < 1e-12:
                continue
            # Find best destination
            # Dest should have low global_sens and be different from s
            valid_dst = np.where((global_sens < global_sens.mean()) & (np.arange(n) != s))[0]
            if len(valid_dst) == 0:
                valid_dst = np.where(global_sens < global_sens.mean())[0]
            if len(valid_dst) == 0:
                valid_dst = np.arange(n)

            # Prefer destinations far from source
            dist = np.abs(valid_dst.astype(float) - s)
            combined_score = global_sens[valid_dst] * (1.0 + 0.1 * dist / n)
            d = valid_dst[np.argmin(combined_score)]
            d = int(d)

            if d == s:
                continue

            # Try transfer
            for frac in [0.05, 0.02, 0.01, 0.08, 0.03, 0.015]:
                amt = a[s] * frac
                if amt < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[s] -= amt
                a_trial[d] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                gain = M_curr - M_trial
                if gain > best_gain:
                    best_gain = gain
                    best_move = (s, d, amt, a_trial)

        if best_move is not None and best_gain > 1e-14:
            s, d, amt, a_new = best_move
            a = a_new
            move_count += 1
            stall = 0
            if verbose and move_count % 100 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, moves={move_count}")
        else:
            stall += 1
            # Random kicks to escape local minima
            if stall % 20 == 0:
                n_kicks = min(5, n // 10)
                for _ in range(n_kicks):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.01, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}, moves={move_count}")

    return best_a, best_m

            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_m - 1e-15:
                a = a_trial
                best_m = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                # Random micro-perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * 0.005
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Fine-tune ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def softmax_subgradient(a, beta=5.0, top_k=50):
    """
    Compute smooth approximation of subgradient using soft-max over top-k lags.
    """
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    shifted = top_vals - top_vals.max()
    w = np.exp(beta * shifted)
    w = w / w.sum()

    # Build weight vector
    w_full = np.zeros(len(c))
    for wi, idx in zip(w, top_idx):
        w_full[int(idx)] = wi

    # Gradient: g[j] = sum_k w_k * 2 * a[k-j] (valid)
    # Use FFT: g = 2 * convolve(a_rev, w_full) [n-1 : 2n-1]
    total_len = n + len(w_full) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_pad = np.zeros(fft_size)
    w_pad[:len(w_full)] = w_full
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_pad), fft_size)
    g = 2.0 * conv[n - 1:2 * n - 1]
    return g, c.max()


def accelerated_subgradient(a0, time_limit, verbose=False, beta_init=6.0,
                            beta_min=2.0, top_k=40, c0=4.0, seed=42):
    """
    Nesterov-accelerated subgradient with soft-max smoothing and adaptive beta.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    c = fast_autocorr(a)
    best_m = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    beta = beta_init
    iters = 0
    restart_every = 4000

    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Nesterov extrapolation
        t_beta = (iters - 1.0) / (iters + 2.0)
        y = x + t_beta * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute soft subgradient
        g, M_curr = softmax_subgradient(y, beta=beta, top_k=top_k)

        # Step size
        eta = gamma / (np.sqrt(iters) + 1.0)

        # Gradient step with projection
        a_new = y - eta * g
        a_new = project_simplex(a_new)

        x_prev = x.copy()
        x = a_new

        # Evaluate
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive beta
        beta = max(beta_min, beta_init - 0.0005 * iters)

        # Restart if stalled
        if stall > restart_every:
            stall = 0
            # Perturb around best
            scale = 0.03
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    Accel-Subgrad restart ({elapsed:.0f}s): J={J:.6f}")

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Accel-Subgrad iter={iters} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

        if s < 0.01:

        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
