        #   This is a function of the vector a. The partial derivative with respect to a[i] is:

        #   Take k-j = i, i.e., j = k-i: term is a[k-i] * a[i], derivative w.r.t. a[i] is a[k-i]

        # But if k-i == i-k, i.e., k=0 and i=... no wait.

        # In that case, both terms equal a[0], and the derivative is 2*a[0].
        # But wait, that's correct because conv_i = sum_j a[j]*a[i-j], and the term j=i gives a[i]*a[0],
        # and the term j=0 gives a[0]*a[i]. So a[i] appears twice: once as the first factor (j=i) 
        # and once as the second factor (j=0). So the derivative IS 2*a[0] when k=i.
        # 
        # More generally, d(conv_k)/d(a[i]) = a[k-i] * [0<=k-i<n] + a[i-k] * [0<=i-k<n]
        # This counts both occurrences of a[i] in the sum.

    # More efficient computation:
    g = np.zeros(n)
    for k in active:
        k = int(k)
        # For each i, d(conv_k)/d(a[i]) = a[k-i] (if valid) + a[i-k] (if valid)
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        # a[i-k] is valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
        
        # Contribution from a[k-i]: valid for i in [max(0, k-n+1), min(n-1, k)]
        i_start1 = max(0, k - n + 1)
        i_end1 = min(n - 1, k)
        if i_start1 <= i_end1:
            idx = np.arange(i_start1, i_end1 + 1)
            g[idx] += a[k - idx]
        
        # Contribution from a[i-k]: valid for i in [max(0, k), min(n-1, k+n-1)]
        i_start2 = max(0, k)
        i_end2 = min(n - 1, k + n - 1)
        if i_start2 <= i_end2:
            idx = np.arange(i_start2, i_end2 + 1)
            g[idx] += a[idx - k]
    
    # Normalize: uniform convex combination of subgradients is a valid subgradient
    g /= len(active)
    return g


def mirror_descent_v2(a0, time_limit, threshold_frac=0.995, max_lags=150, 
                      c0=3.0, restart_every=5000, verbose=False, seed=42):
    """
    Improved mirror descent with exact subgradient and adaptive parameters.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_c = c0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute subgradient
        conv = fast_conv(y)
        g = subgradient_max_conv(y, conv, threshold_frac=threshold_frac, max_lags=max_lags)

        # Step size
        eta = cur_c / (np.sqrt(t) + 1.0)

        # Mirror step (KL projection)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        # Track best
        val = eval_J(a_new)
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive restart
        if stall > restart_every:
            stall = 0
            # Try different perturbation strategies
            strategy = rng.integers(3)
            if strategy == 0:
                # Gaussian perturbation
                scale = rng.uniform(0.01, 0.05)
                pert = rng.normal(0, scale, n) * best_a
                x_r = best_a + pert
            elif strategy == 1:
                # Fourier perturbation
                pert = np.zeros(n)
                n_modes = rng.integers(2, 8)
                for m in range(n_modes):
                    freq = rng.integers(1, n // 4 + 1)
                    amp = rng.uniform(0.005, 0.02)
                    phase = rng.uniform(0, 2 * np.pi)
                    pert += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                x_r = best_a + pert
            else:
                # Local annealing: randomize a small fraction
                frac = rng.uniform(0.05, 0.2)
                mask = rng.random(n) < frac
                x_r = best_a.copy()
                x_r[mask] *= rng.uniform(0.5, 1.5, mask.sum())
            
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            cur_c = max(cur_c * 0.85, 0.5)
            if verbose:
                print(f"    MD restart t={t}: J={best_val:.6f}, c={cur_c:.2f}")

        if verbose and t % 5000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}")

    return best_a, best_val


def fourier_perturb_search(a0, time_limit, n_trials=50, verbose=False, seed=7):
    """
    Search by adding small Fourier perturbations to the current best.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        # Generate random Fourier perturbation
        n_modes = rng.integers(1, 12)
        pert = np.zeros(n)
        for m in range(n_modes):
            freq = rng.integers(1, n // 3 + 1)
            amp = rng.uniform(0.001, 0.01)
            phase = rng.uniform(0, 2 * np.pi)
            pert += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        
        a_new = best_a + pert
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new /= s_new
        val = eval_J(a_new)
        if val < best_val - 1e-15:
            best_val = val
            best_a = a_new.copy()
            if verbose:
                print(f"    Fourier trial {trial}: J={best_val:.6f}")

    return best_a, best_val


def pairwise_cd(a0, time_limit, verbose=False, seed=99):
    """
    Pairwise coordinate descent: move mass between pairs of positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    # Precompute: for each pair (i, j), moving delta from i to j changes conv by:
    # For each k: conv_k changes by a[j]*a[k-i] - a[i]*a[k-j] + ... (complex)
    # So we just evaluate directly.
    
    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        M = conv.max()
        
        # Find the "hottest" positions (those contributing most to the max)
        k_max = int(np.argmax(conv))
        
        # Positions that contribute to conv[k_max]
        i_min = max(0, k_max - n + 1)
        i_max = min(n - 1, k_max)
        
        if i_min > i_max:
            stall += 1
            continue
        
        # Compute "pressure" at each position
        pressure = np.zeros(n)
        for i in range(i_min, i_max + 1):
            j = k_max - i
            pressure[i] += a[j]  # contribution to peak from position i
        
        # Try moving mass from high-pressure to low-pressure positions
        # Pick random pairs
        n_pairs = min(200, n // 5)
        improved = False
        for _ in range(n_pairs):
            # Source: weighted by pressure * mass
            if pressure.max() > 0 and a.max() > 0:
                weights = pressure * a
                if weights.sum() > 0:
                    src = int(rng.choice(n, p=weights / weights.sum()))
                else:
                    src = int(np.argmax(pressure * a))
            else:
                src = int(rng.integers(n))
            
            if a[src] < 1e-10:
                continue
            
            # Destination: low pressure, low mass
            dst_weights = np.maximum(0, 1 - pressure / max(pressure.max(), 1e-10)) * (1 - a / max(a.max(), 1e-10))
            dst_weights[src] = 0
            if dst_weights.sum() < 1e-20:
                dst = int(rng.integers(n))
                if dst == src:
                    continue
            else:
                dst = int(rng.choice(n, p=dst_weights / dst_weights.sum()))
            
            # Try different fractions
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                a_t /= a_t.sum()
                val = eval_J(a_t)
                if val < best_val - 1e-15:
                    a = a_t
                    best_val = val
                    best_a = a.copy()
                    improved = True
                    if verbose:
                        print(f"    CD: J={best_val:.6f}")
                    break
        
        if not improved:
            stall += 1
            # Perturb to escape
            if stall % 20 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    val = eval_J(a)
                    if val < best_val:
                        best_val = val
                        best_a = a.copy()
                        stall = 0

    return best_a, best_val


def scipy_polish(a0, time_limit, verbose=False):
    """
    Use scipy.optimize for local polish.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_val = eval_J(a)
    t0 = time.monotonic()

    def objective(x):
        a_x = np.maximum(x, 0.0)
        s_x = a_x.sum()
        if s_x < 1e-10:
            return 1e10
        a_x = a_x / s_x
        return eval_J(a_x)

    # Try Nelder-Mead on a subspace (too many dimensions for full NM)
    # Instead, try L-BFGS-B with a smoothed objective
    n_active = min(n, 200)  # only optimize a subset of coordinates
    # Choose the most "important" coordinates (highest mass)
    active_idx = np.argsort(-a)[:n_active]
    
    def obj_smooth(x):
        a_x = a.copy()
        a_x[active_idx] = np.maximum(x, 1e-15)
        s_x = a_x.sum()
        if s_x < 1e-10:
            return 1e10
        a_x = a_x / s_x
        return eval_J(a_x)
    
    x0 = a[active_idx].copy()
    
    try:
        res = scipy_minimize(obj_smooth, x0, method='L-BFGS-B', 
                            options={'maxiter': 100, 'maxfun': 200, 'disp': False})
        if time.monotonic() - t0 > time_limit * 0.7:
            return a.copy(), best_val
        
        a_new = a.copy()
        a_new[active_idx] = np.maximum(res.x, 1e-15)
        a_new = a_new / a_new.sum()
        val_new = eval_J(a_new)
        if val_new < best_val:
            best_val = val_new
            a = a_new
            if verbose:
                print(f"    SciPy polish: J={best_val:.6f}")
    except Exception:
        pass

    return a, best_val


def make_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_log_ramp(n):
    """Logarithmic ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = np.log(x + 1)
    return a / a.sum()


def make_exponential(n, rate=1.0):
    """Exponential ramp."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(rate * x / n)
    return a / a.sum()


def make_bimodal(n, sep=0.7):
    """Bimodal."""
    x = np.linspace(0, 1, n)
    a = np.exp(-50 * (x - 0.15) ** 2) + np.exp(-50 * (x - (1 - 0.15)) ** 2) + 0.01
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s

            parent = p.copy()

    except Exception:
        pass

    def try_update(a, label):

            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3200
        best_a = make_ramp(n0, 1.0)
        best_score = eval_J(best_a)

    print(f"\n=== Phase 1: MD Multi-Init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)

    # Deduplicate
    seen = set()
    u_inits = []

    for init, lbl in zip(inits, labels):

    inits = u_inits
