def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4200
        best_seq = make_sine_window(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}, budget={time_budget:.0f}s")

    # === Phase 1: Soft-max mirror descent on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: SoftMD on parent (n={n0}) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = softmax_md(a_in, t1, beta_start=30.0, beta_end=5000.0,
                              c0=8.0, verbose=True, seed=42)
        try_update(a_out, f"softmd_parent_n{n0}")

    # === Phase 2: SoftMD with different beta schedule ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: SoftMD slow anneal (n={n0}) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = softmax_md(a_in, t2, beta_start=10.0, beta_end=2000.0,
                              c0=6.0, verbose=True, seed=123)
        try_update(a_out, f"softmd_slow_n{n0}")

    # === Phase 3: Frequency-domain flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: FreqFlatten (n={n0}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = freq_flatten(a_in, t3, iters=3000, eta0=0.4, verbose=True)
        try_update(a_out, f"freqflat_n{n0}")
        # Polish after freq flatten
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.08, 100.0)
        if t3b > 15:
            a_in2 = norm_sum1(best_seq)
            a_out2, _ = softmax_md(a_in2, t3b, beta_start=100.0, beta_end=3000.0,
                                   c0=5.0, verbose=False, seed=77)
            try_update(a_out2, f"softmd_postfreq_n{n0}")

    # === Phase 4: Try alternative initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Alt inits (n={n0}) ===")
    alts = [
        (make_cosine_taper(n0, power=1.5), "cos1.5"),
        (make_asymmetric(n0, skew=0.2), "asym0.2"),
        (make_asymmetric(n0, skew=0.4), "asym0.4"),
        (make_random_smooth(n0, seed=5, n_freq=8), "rand5"),
        (make_random_smooth(n0, seed=99, n_freq=12), "rand99"),
    ]
    t_per = max(15.0, remaining * 0.15 / len(alts))
    for init, lbl in alts:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 20:
            break
        t_avail = min(t_per, remaining - 10)
        if t_avail < 15:
            break
        print(f"  SoftMD '{lbl}' ({t_avail:.0f}s)...")
        a_out, _ = softmax_md(init, t_avail, beta_start=50.0, beta_end=4000.0,
                              c0=7.0, verbose=False, seed=7)
        try_update(a_out, f"softmd_{lbl}_n{n0}")

    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: CoordPolish (n={n_best}) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coord_polish(a_in, t5, verbose=True, seed=42)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 6: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Longer seq ===")
    for new_n in [n_best + 800, n_best + 1500]:
        if new_n > 7000:
            break
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        a_long = interpolate_seq(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")
        t6 = min(remaining * 0.25, 200.0)
        if t6 > 20:
            a_out, _ = softmax_md(a_long, t6, beta_start=20.0, beta_end=3000.0,
                                  c0=7.0, verbose=True, seed=55)
            try_update(a_out, f"softmd_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.15, 80.0)
            if t6b > 10:
                a_out2, _ = freq_flatten(norm_sum1(best_seq), t6b, iters=1000, eta0=0.3, verbose=False)
                a_long2 = interpolate_seq(a_out2, new_n)
                a_out3, _ = softmax_md(a_long2, 30, beta_start=100.0, beta_end=3000.0,
                                       c0=6.0, verbose=False, seed=88)
                try_update(a_out3, f"softmd_freq_n{new_n}")

    # === Phase 7: Try shorter sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 50 and n_best > 3000:
        print(f"\n=== Phase 7: Shorter seq ===")
        new_n = max(2500, n_best - 1000)
        a_short = interpolate_seq(best_seq, new_n)
        try_update(a_short, f"interp_short_n{new_n}")
        t7 = min(remaining * 0.3, 150.0)
        if t7 > 20:
            a_out, _ = softmax_md(a_short, t7, beta_start=30.0, beta_end=2000.0,
                                  c0=7.0, verbose=True, seed=33)
            try_update(a_out, f"softmd_short_n{new_n}")

    # === Phase 8: Final intensive soft-max on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final SoftMD (n={n_best}) ===")
    t8 = min(remaining * 0.35, 200.0)
    if t8 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = softmax_md(a_in, t8, beta_start=200.0, beta_end=10000.0,
                              c0=4.0, verbose=True, seed=2024)
        try_update(a_out, f"softmd_final_n{n_best}")

    # === Phase 9: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final CoordPolish (n={n_best}) ===")
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coord_polish(a_in, t9, verbose=True, seed=777)
            try_update(a_out, f"coord_final_n{n_best}")

    # === Phase 10: One more freq flatten if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 10: Final FreqFlatten (n={n_best}) ===")
        t10 = min(remaining * 0.5, 100.0)
        if t10 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = freq_flatten(a_in, t10, iters=2000, eta0=0.3, verbose=True, seed=5)
            try_update(a_out, f"freqflat_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4200
        best_seq = make_sine_window(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
