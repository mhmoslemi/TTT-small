```python
def simplex_project(z):
    """Project z onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(z)
    if n == 0:
        return z
    # Algorithm from Duchi et al. 2008
    if z.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0)
```

```python
    t0 = time.monotonic()

    mu = 0.9

        momentum = mu * momentum + (1 - mu) * grad
        a_new = a - lr_current * momentum

        if M_new < best_m:

            lr_current *= 0.5
        else:

        if verbose and t % 100 == 0:

            J = 2.0 * n * best_m
            print(f"  SGD t={t} ({elapsed:.0f}s): J={J:.6f}, M={M_curr:.6f}")
    
    return best_a, best_m
```

```python
_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])

            if prob.status in ('optimal', 'optimal_inaccurate'):

    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def simplex_project(z):
    """Project z onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(z)
    if n == 0:
        return z
    z = np.asarray(z, dtype=np.float64)
    if z.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0)


def compute_smoothed_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*conv[k])) at point a."""
    n = len(a)
    ac = fast_autocorr(a)
    # Softmax weights
    shifted = ac * beta - np.max(ac * beta)
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p = p / p_sum
    # grad[m] = 2 * sum_{j=0}^{n-1} a[j] * p[j+m]
    # Use FFT: conv of a_flipped with p, then extract the right part
    a_flipped = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    AF = np.fft.rfft(a_flipped, fft_size)
    PF = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(AF * PF, fft_size)[:3 * n - 2]
    # (a_flipped * p)[k] = sum_l a[l] * p[k - n + 1 + l]
    # We want grad[m] = 2 * sum_l a[l] * p[m + l] = 2 * (a_flipped * p)[m + n - 1]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]

def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / width) ** 2) + np.exp(-((x - 0.85) / width) ** 2) + 0.005
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def lp_frank_wolfe_v2(a0, time_limit, solver, verbose=True, top_K=None):
    """Improved LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    
    if top_K is not None:
        K_sel = min(top_K, 2 * n - 1)
    else:
        K_sel = min(300, 2 * n - 1)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov
        beta_t = (t - 1.0) / (t + 2.0)
        y = x + beta_t * (x - y)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select binding constraints
        if K_sel >= 2 * n - 1:
            binding_idx = np.arange(2 * n - 1)
        else:
            # Use top-K peaks
            binding_idx = np.argsort(ac)[::-1][:K_sel]

        # Build constraint matrix
        T_sel = np.zeros((len(binding_idx), n))
        for ii, k in enumerate(binding_idx):
            k_int = int(k)
            j_min = max(0, k_int - n + 1)
            j_max = min(n - 1, k_int)
            if j_min <= j_max:
                T_sel[ii, j_min:j_max+1] = y[j_min:j_max+1]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(len(binding_idx))]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y_new / sy if sy > 1e-15 else y.copy()

        x = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        # Adaptively increase K
        if t % 50 == 0 and K_sel < 2 * n - 1:
            K_sel = min(K_sel + 50, 2 * n - 1)

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")

    return best_a, best_m


def smoothed_gd(a0, time_limit, verbose=True, beta_init=20.0, beta_final=200.0, 
                lr_init=0.1, rng_seed=42):
    """FFT-based smoothed gradient descent with adaptive temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    momentum = np.zeros(n)
    mu = 0.8
    lr = lr_init
    stall = 0
    
    total_iters = 200000  # will be limited by time
    
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta: start low, increase over time
        progress = min(t / total_iters, 1.0)
        beta = beta_init + (beta_final - beta_init) * progress ** 2
        
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Compute smoothed gradient
        grad = compute_smoothed_grad(a, beta)
        
        # Add a small L2 regularization to prevent concentration
        # This encourages spreading mass
        reg = 0.0  # Keep reg=0 for now
        
        # Momentum update
        momentum = mu * momentum + (1 - mu) * (grad + reg)
        
        # Step
        a_new = a - lr * momentum
        a_new = simplex_project(a_new)
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Adaptive learning rate
        if stall > 200:
            lr *= 0.8
            mu = min(mu + 0.01, 0.95)
            if stall > 1000:
                # Restart with perturbation
                if lr < 0.001:
                    pert = rng.normal(0, 0.02, n) * best_a
                    a = best_a + pert
                    a = simplex_project(a)
                    lr = lr_init * 0.5
                    mu = 0.8
                    stall = 0
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SGD t={t} ({elapsed:.0f}s): J={J:.6f}, lr={lr:.5f}, beta={beta:.1f}")

    return best_a, best_m


def mirror_descent(a0, time_limit, verbose=True, beta=50.0, c0=3.0, 
                   restart_every=5000, rng_seed=42):
    """Mirror descent (multiplicative weight update) with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        ac = fast_autocorr(y)
        grad = compute_smoothed_grad(y, beta)

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x = x_r / x_r.sum()
            x_prev = x.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descend_v2(a0, time_limit, verbose=True, rng_seed=123):
    """Coordinate descent with peak-aware move selection and line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100
    n_top = 10

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        # Build sensitivity map
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:6]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination: not in peak region, low sensitivity
            mask = ~in_peak
            if mask.sum() < n // 10:
                mask = np.ones(n, dtype=bool)
                mask[src] = False
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
                candidates = candidates[candidates != src]
            if len(candidates) == 0:
                continue
            cand_sens = sens[candidates]
            dst = candidates[np.argmin(cand_sens)]

            # Try multiple fractions with bisection-like search
            found = False
            for frac in [0.005, 0.01, 0.02, 0.002, 0.03, 0.05, 0.08, 0.12]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if not found and stall > 30:
                # Random kick
                j = int(rng.integers(n))
                if j != src:
                    frac = rng.uniform(0.003, 0.02)
                    amt = src_val * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Multi-point random perturbation
            n_pert = min(3, n)
            for _ in range(n_pert):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if verbose and stall % 30 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m
```

    try:
        p = np
