def fast_autocorr(a):
    """Compute conv(a, a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Compute conv(a, b) using FFT."""
    n1 = len(a)

    return np.fft.irfft(A * B, fft_size)[:size]

        vals = ac[active]
        top_idx = np.argsort(vals)[::-1][:num_active]
        active = active[top_idx]
    
    # Build subgradient matrix S where S[:, j] = 2 * (d conv_j / da)
    # d conv_k / da_i = a[i+k] + a[i-k] (for valid indices, with boundaries)
    # conv_k = sum_t a[t] * a[t+k] (for valid t, t+k in [0, n-1])
    # d conv_k / da_i = a[i+k] (if i+k in [0,n-1]) + a[i-k] (if i-k in [0,n-1])
    # Note: when k=0, d conv_0 / da_i = 2*a[i]
    
    n_active = len(active)
    S = np.zeros((n_active, n))
    for idx, k in enumerate(active):
        k = int(k)
        # For each i, d conv_k / da_i:
        # conv_k = sum_{t} a[t]*a[t+k] where 0<=t<n and 0<=t+k<n
        # = sum_{t} a[t]*a[t+k] over valid range
        # d/d a_i: appears when t=i (gives a[i+k] if valid) or t+k=i i.e. t=i-k (gives a[i-k] if valid)
        # But if k=0, both terms give a[i], so total is 2*a[i]
        
        # a[i+k] for i in range where i+k < n, i.e., i < n-k
        if k >= 0:
            # term 1: a[i+k] for i in [0, n-k-1]
            S[idx, 0:n-k] += a[k:n]  # a[i+k] for i=0..n-k-1
            # term 2: a[i-k] for i in [k, n-1]
            S[idx, k:n] += a[0:n-k]  # a[i-k] for i=k..n-1
        else:
            k_neg = -k
            # term 1: a[i+k] = a[i-k_neg] for i in [k_neg, n-1]
            S[idx, k_neg:n] += a[0:n-k_neg]
            # term 2: a[i-k] = a[i+k_neg] for i in [0, n-k_neg-1]
            S[idx, 0:n-k_neg] += a[k_neg:n]
    
    S *= 2.0  # The subgradient is 2 * (d conv_k / da) summed with weights
    
    # Solve: min ||S^T lambda||^2 s.t. sum lambda = 1, lambda >= 0
    # This is: min lambda^T (S S^T) lambda s.t. 1^T lambda = 1, lambda >= 0
    # For small n_active, solve via KKT or simple projection
    
    if n_active == 1:
        lam = np.array([1.0])
    else:
        # For small problems, use scipy or manual QP
        try:
            from scipy.optimize import minimize
            A_mat = S @ S.T
            b_vec = S @ S @ np.ones(n_active) / 2
            # min lambda^T A_mat lambda - b_vec^T lambda s.t. sum=1, lambda>=0
            # Actually: min ||S^T lambda||^2 = min lambda^T (S S^T) lambda
            # So A_mat = S S^T, and we minimize lambda^T A_mat lambda s.t. sum=1, lam>=0
            
            def obj(lam):
                return lam @ A_mat @ lam
            
            def grad(lam):
                return 2 * A_mat @ lam
            
            cons = [{'type': 'eq', 'fun': lambda l: l.sum() - 1, 'jac': lambda l: np.ones(n_active)}]
            bounds = [(0, 1)] * n_active
            lam0 = np.ones(n_active) / n_active
            res = minimize(obj, lam0, jac=grad, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 100, 'ftol': 1e-12})
            lam = res.x
            lam = np.maximum(lam, 0)
            lam /= lam.sum()
        except Exception:
            # Fallback: uniform weights
            lam = np.ones(n_active) / n_active
    
    # Subgradient = S^T @ lambda
    grad = S.T @ lam
    return grad, ac, M


def subgradient_step(a, lr, momentum_state=None, beta=0.9):
    """
    One step of Nesterov-accelerated mirror descent (exponential family / simplex).
    """
    n = len(a)
    a = np.maximum(a, 1e-30)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        return a, None
    
    a = a / s
    
    if momentum_state is None:
        y = a.copy()
        m = np.zeros(n)
    else:
        m, y_prev = momentum_state
        y = a + beta * (a - y_prev)
        y = np.maximum(y, 1e-30)
        y = y / y.sum()
    
    # Compute subgradient at y
    g, ac, M = compute_subgradient_min_norm(y)
    
    # Mirror descent update: a_new_i = y_i * exp(-lr * g_i)
    # For the simplex (exponential family), this is:
    # a_new = softmax(log(y) - lr * g)
    log_y = np.log(np.maximum(y, 1e-30))
    log_new = log_y - lr * g
    log_new -= log_new.max()
    a_new = np.exp(log_new)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    
    new_momentum = (a_new, a)
    return a_new, new_momentum


def nesterov_mirror_descent(a0, time_limit, verbose=True, lr_init=0.5,
                            beta=0.9, decay=0.9995, seed=42):
    """
    Nesterov-accelerated mirror descent on the simplex with explicit subgradient.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    momentum_state = None
    lr = lr_init
    iter_count = 0
    stall = 0
    max_stall = 5000
    
    while time.monotonic() - t0 < time_limit:
        a, momentum_state = subgradient_step(a, lr, momentum_state, beta)
        iter_count += 1
        lr *= decay
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            # Restart
            stall = 0
            momentum_state = None
            lr = max(lr_init * 0.5, 0.1)
            # Perturb
            rng = np.random.default_rng(hash((iter_count, seed)) % (2**31))
            pert = rng.normal(0, 0.02, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            if verbose:
                print(f"    NMD restart at iter {iter_count}, J={2*n*best_M:.6f}")
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD iter={iter_count} ({elapsed:.0f}s): J={2*n*best_M:.6f}, lr={lr:.4f}")
    
    return best_a, best_M


def pairwise_coordinate_descent(a0, time_limit, verbose=True, seed=42,
                                num_pairs_per_sweep=200):
    """
    Coordinate descent on pairs: for each pair (i,j), find optimal transfer
    amount by 1D line search on the max autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 50
    total_improvements = 0
    
    # Precompute autocorrelation for tracking
    ac = fast_autocorr(a)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Sample random pairs
        pairs = []
        for _ in range(num_pairs_per_sweep):
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10 and a[j] > 1e-12:
                pairs.append((i, j))
        
        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try transferring mass from i to j (and vice versa)
            # a[i] -> a[i] - t, a[j] -> a[j] + t
            # Find t that minimizes max autocorrelation
            # Use ternary search on t in [-a[i]*0.5, a[j]*0.5 + a[i]*0.5]
            
            best_t = 0.0
            best_trial_M = ac.max()
            
            # Quick check: evaluate at a few candidate t values
            c1 = a[i]
            c2 = a[j]
            
            # Binary/ternary search for optimal t
            lo = -min(c1, 0.1 * n) * 0.8
            hi = min(c2 + c1, 0.1 * n) * 0.8
            
            # Actually, t must keep both a[i]-t >= 0 and a[j]+t >= 0
            # So t in [-a[j], a[i]]
            lo = -a[j] * 0.9
            hi = a[i] * 0.9
            
            # Sample a few points
            candidates = np.linspace(lo, hi, 20)
            best_trial_M = ac.max()
            
            for t_val in candidates:
                if t_val == 0:
                    continue
                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                if a_trial[i] < 0 or a_trial[j] < 0:
                    continue
                # Compute new max autocorrelation efficiently
                # Only lags k where a[i] or a[j] participate are affected
                # Conv change at lag k: 
                #   delta_k = -t*a[i+k] - t*a[i-k] + t*a[j+k] + t*a[j-k] + (cross terms)
                # Simpler: just compute full autocorr for small number of candidates
                ac_trial = fast_autocorr(a_trial)
                M_trial = ac_trial.max()
                if M_trial < best_trial_M - 1e-15:
                    best_trial_M = M_trial
                    best_t = t_val
            
            if best_t != 0 and best_trial_M < best_M - 1e-14:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                ac = fast_autocorr(a)
                best_M = ac.max()
                best_a = a.copy()
                improved = True
                total_improvements += 1
        
        if not improved:
            stall += 1
            if stall >= max_stall:
                # Random perturbation
                for _ in range(10):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-8:
                        t = rng.uniform(-0.03, 0.03) * a[i]
                        a[i] -= t
                        a[j] += t
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        ac = fast_autocorr(a)
                        if ac.max() < best_M:
                            best_M = ac.max()
                            best_a = a.copy()
                            stall = 0
                            break
                        else:
                            a = best_a.copy()
                            ac = fast_autocorr(a)
                if stall >= max_stall * 3:
                    if verbose:
                        print(f"    CoordDescent: stalled at J={2*n*best_M:.6f}")
                    break
        
        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent ({elapsed:.0f}s): J={2*n*best_M:.6f}, improvements={total_improvements}")
    
    return best_a, best_M


def spectral_polish(a0, time_limit, verbose=True):
    """
    Spectral domain optimization: adjust the DFT phase to flatten autocorrelation
    while maintaining non-negativity. Uses iterative projection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Work in frequency domain
    # A = DFT(a), autocorr = IDFT(|A|^2)
    # We want to minimize max(IDFT(|A|^2)) while a = IDFT(A) >= 0, sum(a) = 1
    
    # Approach: alternate between:
    # 1. In frequency domain: adjust |A|^2 to be more flat (reducing peak of IDFT)
    # 2. In time domain: project back to non-negative simplex
    
    # Actually, a simpler approach: use the DFT to identify which lags are "hot"
    # and construct a direction that specifically reduces those lags.
    
    iter_count = 0
    streak = 0
    while time.monotonic() - t0 < time_limit and streak < 3000:
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            streak = 0
        else:
            streak += 1
        
        # Find the peak lag(s)
        peak_lags = np.where(ac >= M * (1 - 1e-6))[0]
        
        # Construct a "pressure" vector that pushes mass away from contributing positions
        # For lag k, the autocorrelation is sum_i a[i]*a[i+k]
        # To reduce it, we want to reduce a[i]*a[i+k] for large products
        pressure = np.zeros(n)
        for k in peak_lags[:10]:
            k = int(k)
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                partner = k - i
                if 0 <= partner < n:
                    pressure[i] += a[partner]
        
        # Normalize pressure
        p_sum = pressure.sum()
        if p_sum < 1e-15:
            break
        pressure = pressure / p_sum
        
        # Move: reduce mass where pressure is high, increase where low
        # Use a smooth version: a_new = a * (1 - alpha * (pressure - pressure.mean()))
        alpha = 0.01
        a_new = a * (1.0 - alpha * pressure * n / (n * pressure.mean() + 1e-15))
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Also try the opposite direction (concentrate where pressure is low)
        alpha2 = 0.005
        a_new2 = a * (1.0 + alpha2 * (1.0 - pressure) * n / (n * (1-pressure).mean() + 1e-15))
        a_new2 = np.maximum(a_new2, 0)
        s2 = a_new2.sum()
        if s2 > 1e-15:
            a_new2 = a_new2 / s2
            M2 = fast_autocorr(a_new2).max()
            M1 = fast_autocorr(a_new).max()
            if M2 < M1:
                a = a_new2
            else:
                a = a_new
        else:
            a = a_new
        
        iter_count += 1
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral ({elapsed:.0f}s): J={2*n*best_M:.6f}")
    
    return best_a, best_M


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def smooth_init(n, seed=0, num_freqs=5):
    """Create a smooth non-negative initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, num_freqs + 1):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def structured_init(n, seed=0):
    """
    Create initialization based on a windowed m-sequence pattern.
    Uses the structure of Golay sequences for good autocorrelation.
    """
    rng = np.random.default_rng(seed)
    # Start with a smooth base
    a = smooth_init(n, seed=seed, num_freqs=3)
    # Add some structure: modulate with a slow cosine
    x = np.linspace(0, 1, n)
    mod = 1.0 + 0.3 * np.cos(2 * np.pi * 2 * x + rng.uniform(0, 2 * np.pi))
    a = a * mod
    a = np.maximum(a, 0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator. Multi-phase optimization:
    1. Nesterov mirror descent from multiple initializations
    2. Pairwise coordinate descent polish
    3. Spectral refinement
    4. Multi-length exploration
    5. Final coordinate descent
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 3200
        best_seq = structured_init(n_base, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Nesterov Mirror Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov MD ({remaining:.0f}s available) ===")
    t_phase1 = min(remaining * 0.35, 550.0)
    
    if t_phase1 > 60:
        inits = []
        labels = []
        
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        
        # Structured init
        inits.append(structured_init(n0, seed=5))
        labels.append("structured")
        
        # Smooth init with different seed
        inits.append(smooth_init(n0, seed=23, num_freqs=7))
        labels.append("smooth")
        
        # Blend
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * structured_init(n0, seed=11)
            inits.append(norm_sum1(blend))
            labels.append("blend")
        
        # Remove near-duplicates
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for existing in unique_inits:
                if np.abs(np.corrcoef(init, existing)[0, 1]) > 0.999:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)
        
        t_per = max(25.0, t_phase1 / len(unique_inits))
        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 20:
                break
            print(f"  NMD '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = nesterov_mirror_descent(init, t_avail, verbose=False,
                                                lr_init=0.3, beta=0.85,
                                                decay=0.9997, seed=42 + idx)
            try_update(a_out, f"nmd_{lbl}")
    
    # === Phase 2: Coordinate Descent Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coordinate Descent (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_coordinate_descent(a_in, t2, verbose=True, seed=77,
                                                num_pairs_per_sweep=300)
        try_update(a_out, f"coord_n{len(best_seq)}")
    
    # === Phase 3: Spectral Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral Polish ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_polish(a_in, t3, verbose=True)
        try_update(a_out, f"spectral_n{len(best_seq)}")
    
    # === Phase 4: Second NMD round on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: NMD Round 2 ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_mirror_descent(a_in, t4, verbose=False,
                                            lr_init=0.15, beta=0.8,
                                            decay=0.9998, seed=999)
        try_update(a_out, f"nmd_r2_n{len(best_seq)}")
    
    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = [n_best - 400, n_best + 400, n_best + 800, n_best - 800]
    candidate_lengths = [l for l in candidate_lengths if 1500 <= l <= 5000]
    
    for new_n in candidate_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 5: Length n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)
        
        t5a = min(remaining * 0.3, 150.0)
        if t5a > 20:
            a_out, _ = nesterov_mirror_descent(a_long, t5a, verbose=False,
                                                lr_init=0.2, beta=0.85,
                                                decay=0.9997, seed=555)
            try_update(a_out, f"nmd_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.3, 120.0)
            if t5b > 15:
                a_out, _ = pairwise_coordinate_descent(a_long, t5b, verbose=False,
                                                        seed=333, num_pairs_per_sweep=200)
                try_update(a_out, f"coord_n{new_n}")
    
    # === Phase 6: Final coordinate descent on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Coord (n={n_best}) ===")
    if remaining > 30:
        t6 = min(remaining * 0.5, 200.0)
        if t6 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = pairwise_coordinate_descent(a_in, t6, verbose=True, seed=1234,
                                                    num_pairs_per_sweep=400)
            try_update(a_out, f"coord_final_n{len(best_seq)}")
    
    # === Phase 7: Final spectral + short NMD ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 7: Final Polish ===")
        a_in = norm_sum1(best_seq)
        
        t7a = min(remaining * 0.4, 100.0)
        if t7a > 15:
            a_out, _ = spectral_polish(a_in, t7a, verbose=False)
            try_update(a_out, f"spectral_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            a_in = norm_sum1(best_seq)
            t7b = min(remaining * 0.5, 80.0)
            if t7b > 10:
                a_out, _ = nesterov_mirror_descent(a_in, t7b, verbose=False,
                                                    lr_init=0.05, beta=0.7,
                                                    decay=0.9999, seed=7777)
                try_update(a_out, f"nmd_final")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = smooth_init(n_fb, seed=1)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
