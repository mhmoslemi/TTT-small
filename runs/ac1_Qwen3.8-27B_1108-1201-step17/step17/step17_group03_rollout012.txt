    # (a*a)[k] = sum_i a[i]*a[k-i], grad w.r.t. a[i] = a[k-i] (for valid k-i)

    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        valid = (j_idx >= 0) & (j_idx < n)
        idx = idx[valid]
        j_idx = j_idx[valid]
        if len(idx) > 0:
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad

def amd_optimize(a0, time_limit, verbose=True, top_k=50, beta_init=10.0, 
                 beta_decay=0.999, c0=4.0, restart_every=3000):
    """Nesterov accelerated mirror descent with backtracking."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_beta = beta_init
    cur_top_k = top_k
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        
        # Compute subgradient
        g = compute_smooth_subgradient(y, ac, n, beta=cur_beta, top_k=cur_top_k)
        
        # Backtracking line search
        eta = gamma / (np.sqrt(t) + 1.0)
        base_m = M_curr
        accepted = False
        for _ in range(8):
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            else:
                eta *= 0.5
        
        if not accepted:
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Decay beta to sharpen the max approximation
        cur_beta *= beta_decay
        if cur_beta > 500:
            cur_beta = 500
        
        # Restart on stall
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_beta = max(cur_beta * 0.5, 5.0)
            cur_top_k = min(cur_top_k + 10, 80)
            if verbose:
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t}: J={J:.6f}")
        
        if verbose and t % 50000 == 0:
            J = 2.0 * n * best_m
            print(f"    AMD t={t}: J={J:.6f}, beta={cur_beta:.1f}, stall={stall_count}")
    
    return best_a, best_m


def pairwise_swap_optimize(a0, time_limit, verbose=True, rng_seed=123):
    """Targeted pairwise swap coordinate descent."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        
        # Find top few peaks
        top_peaks = np.argsort(ac)[::-1][:5]
        
        improved = False
        for peak_k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(peak_k)

            i_start = max(0, peak_k - n + 1)
            i_end = min(n, peak_k + 1)
            idx = np.arange(i_start, i_end)
            j_idx = peak_k - idx
            valid = (j_idx >= 0) & (j_idx < n)
            idx = idx[valid]
            j_idx = j_idx[valid]
            contrib = a[idx] * a[j_idx]
            total = contrib.sum()
            
            if total < 1e-20:
                continue
            
            # Try to reduce the top contributors
            n_top = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]
            
            for li in top_local:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                best_new_m = M_curr
                best_move = None
                
                # Try different fractions of the move
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    # Try random destinations
                    for _ in range(20):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_new_m:
                            best_new_m = M_trial
                            best_move = (src, dst, amt)

                    src_i, dst_i, amt = best_move
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)

def symmetrize(a):
    """Make sequence symmetric: a[i] = (a[i] + a[n-1-i])/2."""
    n = len(a)
    result = (a + a[::-1]) / 2.0
    return result / result.sum()


def freq_perturb(a, strength=0.01, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

        return best_seq, best_score
