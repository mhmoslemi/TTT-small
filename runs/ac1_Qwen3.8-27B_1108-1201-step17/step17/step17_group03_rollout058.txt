    warm_b = None
    stall = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_conv(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:min(top_K, len(ac))]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)
        T_sel = np.zeros((K, n))
        rhs = np.zeros(K)
        for row, k_int in enumerate(active_idx):
            ki = int(k_int)
            i_start = max(0, ki - n + 1)
            i_end = min(n, ki + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[ki - i]
            rhs[row] = M_curr + ac[ki]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= rhs]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stall += max_iters_excess
            break

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            stall += 1
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_conv(y).max()
        improved = False
        if my < best_m:
            best_m = my
            best_a = y.copy()
            improved = True
            stall = 0
            margin_frac = max(margin_frac * 0.85, 2e-5)

        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            improved = True
            stall = 0

        if not improved:
            stall += 1
            if stall > 30:
                margin_frac = min(margin_frac * 1.3, 5e-2)
                stall = 0

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP t={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")

    return best_a, best_m


def peak_crush(a0, time_limit, verbose=True, rng_seed=77):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        ac = fast_conv(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total_c = contrib.sum()
        if total_c < 1e-20:
            stall += 1
            continue

        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]
        improved = False

        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_trial_m = M_curr
            best_move = None

            for frac in [0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                for _ in range(40):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_trial_m:
                        best_trial_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.015)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 40 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PC ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def freq_perturb(a, strength=0.02, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'beta':
        x = np.linspace(0.02, 0.98, n)
        return norm_sum1(x ** 1.5 * (1 - x) ** 0.8)
    elif family == 'skew':
        x = np.linspace(0, 1, n)
        return norm_sum1((x ** 0.7) * (1 - x) ** 1.2)
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = norm_sum1(p)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
            best_seq = a_norm.copy()
            best_score = sc

    # Phase 1: Mirror descent (fast, many iterations)
    print(f"\n=== Phase 1: Mirror Descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 400.0)
    if t1 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent(a_in, t1, top_k=70, beta=3.5, c0=3.5,
                                  restart_every=3000, verbose=True, rng_seed=42)
        try_update(a_out, "md1")

        # Second round with different params
        remaining = time_budget - (time.monotonic() - t_global)
        t1b = min(remaining * 0.10, 120.0)
        if t1b > 20:
            a_in2 = norm_sum1(best_seq)
            a_out2, _ = mirror_descent(a_in2, t1b, top_k=50, beta=5.0, c0=5.0,
                                       restart_every=2000, verbose=False, rng_seed=123)
            try_update(a_out2, "md2")

    # Phase 2: LP Frank-Wolfe (main optimizer)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_cur}) ===")
    t2 = min(remaining * 0.25, 380.0)
    if t2 > 50 and n_cur <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=200,
                                   margin_frac=3e-3)
        try_update(a_out, "lp_main")

    # Phase 3: Peak-crush coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Peak-crush (n={n_cur}) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_crush(a_in, t3, verbose=True, rng_seed=42)
        try_update(a_out, "pc1")

    # Phase 4: Second LP with tighter margins
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP-FW tight (n={n_cur}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 40 and n_cur <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K=300,
                                   margin_frac=5e-4)
        try_update(a_out, "lp_tight")

    # Phase 5: Try longer sequences
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    longer = sorted(set([n_cur + 500, n_cur + 1000, n_cur + 1800]))
    longer = [l for l in longer if l <= 5500 and remaining > 250]

    for new_n in longer[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print(f"\n=== Phase 5: Extend to n={new_n} ===")
        a_long = interpolate_seq(best_seq, new_n)

        t5a = min(remaining * 0.18, 180.0)
        if t5a > 30:
            a_out, _ = mirror_descent(a_long, t5a, top_k=60, beta=4.0, c0=3.5,
                                       restart_every=2500, verbose=False, rng_seed=99)
            try_update(a_out, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100 and new_n <= 4500:
            t5b = min(remaining * 0.15, 130.0)
            if t5b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t5b, solver, verbose=True,
                                           top_K=180, margin_frac=2e-3)
                try_update(a_out, f"lp_n{new_n}")

    # Phase 6: Frequency perturbation + re-optimize
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Freq-diversify (n={n_cur}) ===")
    rng = np.random.default_rng(200)
    t6_per = min(remaining * 0.06, 80.0)
    if t6_per > 25:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t6_per + 20:
                break
            a_pert = freq_perturb(best_seq, strength=0.015, rng=rng)
            a_out, _ = mirror_descent(a_pert, t6_per, top_k=50, beta=5.0, c0=4.5,
                                       restart_every=1500, verbose=False, rng_seed=trial)
            try_update(a_out, f"freq{trial}")

    # Phase 7: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_cur}) ===")
    t7 = min(remaining * 0.12, 150.0)
    if t7 > 30 and n_cur <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_K=350,
                                   margin_frac=1e-4)
        try_update(a_out, "lp_final")

    # Phase 8: Final peak-crush
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final peak-crush (n={n_cur}) ===")
    if remaining > 25:
        t8 = min(remaining * 0.20, 120.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_crush(a_in, t8, verbose=True, rng_seed=555)
            try_update(a_out, "pc_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = make_parametric(3200, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining")

    return [float(x) for x in best_seq]
