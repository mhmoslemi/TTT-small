    """Compute full linear autocorrelation using FFT."""

    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * np.max(ac)


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def smooth_subgradient(a, mu):
    """
    Compute smooth (log-sum-exp) subgradient of max(conv(a,a)) using FFT cross-correlation.
    Returns (gradient, max_autocorr, softmax_weights).
    O(n log n) per call.
    """
    n = len(a)
    ac = fast_autocorr(a)
    ac_max = np.max(ac)
    
    # Compute softmax weights with temperature mu
    shifted = (ac - ac_max) / mu
    # Clip to prevent overflow
    shifted = np.clip(shifted, -50.0, 50.0)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros_like(ac)
        weights[int(np.argmax(ac))] = 1.0
    else:
        weights /= w_sum
    
    # Compute gradient via cross-correlation:
    # g_j = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
    # This is the cross-correlation of a and w, computed via FFT
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(weights, N)
    corr = np.fft.irfft(np.conj(A) * W, N)[:n]
    grad = 2.0 * corr
    
    return grad, ac_max, weights


def nesterov_smooth(a0, time_limit, mu_init=2.0, mu_final=0.005, gamma=0.9995,
                    c=3.0, restart_every=5000, seed=42, verbose=False):
    """
    Nesterov-accelerated mirror descent with smooth subgradient and mu-continuation.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    mu = mu_init
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    stall = 0
    iteration = 0
    
    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break
        iteration += 1
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s
        
        # Compute smooth subgradient
        g, M, w = smooth_subgradient(y, mu)
        
        # Polyak step size with cap
        cur_val = 2.0 * n * M
        gap = cur_val - lower_bound
        if gap < 1e-10:
            gap = 1e-10
        g_norm_sq = np.dot(g, g) + 1e-20
        eta = c * gap / g_norm_sq
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)
        
        # Mirror descent (exponential) step
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Decrease mu (continuation)
        mu = max(mu * gamma, mu_final)
        
        # Restart with adaptation
        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.005 * min(stall // restart_every, 5)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    NS restart ({elapsed:.0f}s): J={best_J:.6f}, mu={mu:.4f}")
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NS t={t} ({elapsed:.0f}s): J={best_J:.6f}, mu={mu:.4f}")
    
    return best_a, best_J


def peak_reduction_moves(a0, time_limit, verbose=False, seed=77):
    """
    Targeted coordinate moves: identify peak contributors and relocate mass to safe zones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    stall = 0
    max_stall = 800
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = np.max(ac)
        
        # Get top peaks
        n_peaks = min(20, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]
        
        # Compute sensitivity: for each position, how much does it contribute to peaks?
        sensitivity = np.zeros(n)
        for k in top_lags[:10]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contribution = a[idx] * a[j_idx]
            for ii, jj, cc in zip(idx, j_idx, contribution):
                sensitivity[ii] += cc
                sensitivity[jj] += cc
        
        mean_s = sensitivity.mean()
        std_s = sensitivity.std()
        if std_s < 1e-20:
            stall += 1
            continue
        
        high_sens = np.where((sensitivity > mean_s + std_s) & (a > 1e-6))[0]
        low_sens = np.where((sensitivity < mean_s - 0.5 * std_s))[0]
        
        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            # Try random moves as escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-6:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
                    J_new = compute_J(a)
                    if J_new < best_J:
                        best_J = J_new
                        best_a = a.copy()
                        stall = 0
            continue
        
        improved = False
        
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            
            src = high_sens[int(rng.integers(len(high_sens)))]
            dst = low_sens[int(rng.integers(len(low_sens)))]
            if src == dst:
                continue
            src_val = a[src]
            if src_val < 1e-8:
                continue
            
            fracs = [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.15]
            found = False
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break
            
            if found:
                stall = 0
        
        if not improved:
            stall += 1
            if stall % 30 == 0:
                # Larger random perturbation to escape
                n_pert = max(10, n // 50)
                perturb_positions = rng.choice(n, size=n_pert, replace=False)
                for p in perturb_positions:
                    if a[p] > 1e-6:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a[p] * frac
                        a[p] -= amt
                        dst = int(rng.integers(n))
                        a[dst] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
        
        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=500):
    """LP Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    prev_y = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build constraint matrix for top-K constraints
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y
        
        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        
        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        prev_y = y
        y = y_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        # Also check the Nesterov-extrapolated point
        if t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - prev_y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx > 1e-15:
                x_new = x_new / sx
                mx = fast_autocorr(x_new).max()
                if mx < best_m:
                    best_m = mx
                    best_a = x_new.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_symmetric_init(n):
    """Create a symmetric initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.15):
    """Gaussian bumps at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.005
    return norm_sum1(a)


def make_center_gaussian(n, sigma=0.2):
    """Gaussian bump at center."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) / sigma) ** 2) + 0.005
    return norm_sum1(a)


def make_u_shape(n):
    """U-shaped (high at edges, low in middle)."""
    x = np.linspace(0, 1, n)
    a = (x - 0.5) ** 2 + 0.1
    return norm_sum1(a)

            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()

            parent = p.copy()

        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
