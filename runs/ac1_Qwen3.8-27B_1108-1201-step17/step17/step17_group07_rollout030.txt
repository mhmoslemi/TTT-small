def smooth_objective_and_grad(a, t):
    """Compute smooth approximation of max(conv) and its gradient.
    F_t(a) = (1/t) * logsumexp(t * conv(a,a))
    Returns (F_t, grad) where grad = dF_t/da
    """
    n = len(a)
    b = fast_conv(a)  # length 2n-1
    tb = t * b
    max_tb = tb.max()
    # logsumexp for numerical stability
    log_sum_exp = max_tb + np.log(np.sum(np.exp(tb - max_tb)))
    F_t = log_sum_exp / t
    
    # softmax weights
    p = np.exp(tb - max_tb)
    p /= p.sum()
    
    # Weighted autocorrelation: q[k] = p[k] * b[k]
    q = p * b
    
    # Gradient: dF_t/da[j] = sum_k q[k] * (2*a[k-j] if 0<=k-j<n else 0) / t
    # This is 2/t * (q * a_flipped)[something] ... 
    # Actually: sum_k q[k] * 2*a[k-j] for k where 0 <= k-j < n
    # = 2 * sum_{m=0}^{n-1} q[m+j] * a[m] (where m = k-j, k = m+j)
    # This is a cross-correlation of q and a
    # Can compute with FFT: ifft(FFT(q) * conj(FFT(a))[::-1]) ... 
    # Or more simply: it's the inner product of a with a shifted version of q
    
    # Let me compute it directly using the structure:
    # grad[j] = (2/t) * sum_{m=0}^{n-1} a[m] * q[m+j]
    # This is a "sliding dot product" - can be computed with FFT
    # Pad q to length 2n-1 (it already is), pad a to length 2n-1
    # The sliding dot product is a cross-correlation
    
    a_padded = np.zeros(2*n-1)
    a_padded[:n] = a
    q_padded = np.zeros(2*n-1)
    q_padded[:2*n-1] = q  # q is already length 2n-1
    
    # Cross-correlation: (a_padded * q_padded)[n-1+j] for j=0..n-1
    # Actually, sum_m a[m] * q[m+j] = convolve(a_reversed, q)[n-1+j]
    # Or: it's the correlation of a with q, evaluated at lag j
    
    # Using FFT: corr[j] = sum_m a[m] * q[m+j]
    # = ifft(FFT(a_padded) * conj(FFT(q_padded))) ... no, that's circular correlation
    
    # Let me just use the direct formula with FFT:
    # Define r[j] = sum_{m=0}^{n-1} a[m] * q[m+j]
    # This is a "cross-correlation" that can be computed as:
    # r = ifft(fft(a, N) * conj(fft(q, N))) for appropriate N
    # But we need LINEAR correlation, not circular
    
    # Actually, r[j] = sum_m a[m] * q[m+j] for j = 0, ..., n-1
    # This is the (n-1-j) lag of the cross-correlation of a and q
    # Or equivalently, it's a "sliding window" inner product
    
    # Simplest: use FFT-based convolution
    # r[j] = (a_reversed * q)[n-1+j] where * denotes convolution
    a_rev = a[::-1]
    # conv(a_rev, q)[n-1+j] = sum_i a_rev[i] * q[n-1+j-i]
    # = sum_i a[n-1-i] * q[n-1+j-i]
    # Let m = n-1-i: sum_m a[m] * q[j+m]
    # Yes! That's exactly r[j].
    
    N = 1 << (3*n-2).bit_length()  # enough for linear conv of length n and 2n-1
    A_fft = np.fft.rfft(a_rev, N)
    Q_fft = np.fft.rfft(q, N)
    r = np.fft.irfft(A_fft * Q_fft, N)[:2*n-1]
    grad = 2.0 * r[n-1:n-1+n] / t
    
    return F_t, grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full FFT-based gradient computation for Nesterov mirror descent
2. L-BFGS with logsumexp smooth objective and temperature annealing
3. LP-based Remez method for constraint polishing
4. Asymmetric initializations (the optimal solution is asymmetric)
5. Multi-length exploration
Key insight: The optimal non-negative sequence is ASYMMETRIC, allowing
the center lag conv[n-1] = sum a[i]*a[n-1-i] to be much smaller than
sum a[i]^2, reducing the peak-to-average ratio.

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))

def norm_sum1(a):

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_full_grad_smooth(a, t):
    """Compute gradient of (1/t)*logsumexp(t*conv(a,a)) w.r.t. a.
    Returns (F_t, grad) where grad has same shape as a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    b = fast_conv(a)  # length 2n-1
    tb = t * b
    max_tb = tb.max()
    exp_tb = np.exp(tb - max_tb)
    sum_exp = exp_tb.sum()
    log_sum_exp = max_tb + np.log(sum_exp)
    F_t = log_sum_exp / t
    
    # Softmax weights
    p = exp_tb / sum_exp  # length 2n-1
    
    # q[k] = p[k] * b[k]
    q = p * b  # length 2n-1
    
    # grad[j] = (2/t) * sum_{m=0}^{n-1} a[m] * q[m+j]
    # = (2/t) * conv(a_rev, q)[n-1+j]
    a_rev = a[::-1]
    # Linear conv of a_rev (length n) and q (length 2n-1) -> length 3n-2
    N_fft = 1 << (3 * n - 1).bit_length()
    A_fft = np.fft.rfft(a_rev, N_fft)
    Q_fft = np.fft.rfft(q, N_fft)
    r = np.fft.irfft(A_fft * Q_fft, N_fft)[:3*n-2]
    grad = 2.0 * r[n-1:n+n-1] / t
    
    return F_t, grad

def nmd_full_grad(a0, time_limit, t=50.0, lr_init=0.1, verbose=False):
    """Nesterov mirror descent with full FFT-based gradient of smooth objective."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_val = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Precompute FFT structure for speed
    N_fft = 1 << (3 * n - 1).bit_length()
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 2000
    rng = np.random.default_rng(42 + n)
    lr = lr_init
    
    for t_iter in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t_iter - 1.0) / (t_iter + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute gradient
        b = fast_conv(y)
        tb = t * b
        max_tb = tb.max()
        exp_tb = np.exp(tb - max_tb)
        sum_exp = exp_tb.sum()
        p = exp_tb / sum_exp
        q = p * b
        
        a_rev = y[::-1]
        A_fft = np.fft.rfft(a_rev, N_fft)
        Q_fft = np.fft.rfft(q, N_fft)
        r = np.fft.irfft(A_fft * Q_fft, N_fft)[:3*n-2]
        grad = 2.0 * r[n-1:n+n-1] / t
        
        # Mirror descent step
        eta = lr / (np.sqrt(t_iter) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_val - 1e-18:
            best_val = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(300, int(max_stall * 0.7))
            strategy = int(rng.integers(3))
            if strategy == 0:
                pert = rng.normal(0, 0.03, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n//4, 500), replace=False)
                x_r[idx] *= rng.uniform(0.5, 1.5, len(idx))
            else:
                x_r = best_a * (1 + rng.normal(0, 0.01, n))
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            lr = max(0.01, lr * 0.7)
        
        if verbose and t_iter % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_val
            print(f"    NMD-FG t={t_iter} ({elapsed:.1f}s): J={J:.8f}, stall={stall_count}")
    
    return best_a, best_val

def lbfgs_smooth(a0, time_limit, t_schedule=None, verbose=False):
    """L-BFGS on smooth objective with temperature annealing.
    Uses log parameterization: a = softmax(z).
    """
    n = len(a0)
    a = norm_sum1(a0)
    z = np.log(a + 1e-30)
    best_a = a.copy()
    best_val = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    if t_schedule is None:
        t_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000]
    
    for t_curr in t_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        t_budget = time_limit - (time.monotonic() - t0)
        if t_budget < 5:
            break
        
        def objective(z_vec):
            a_vec = np.exp(z_vec - z_vec.max())
            a_vec = a_vec / a_vec.sum()
            b = fast_conv(a_vec)
            tb = t_curr * b
            max_tb = tb.max()
            return max_tb + np.log(np.sum(np.exp(tb - max_tb))) / t_curr
        
        def gradient(z_vec):
            a_vec = np.exp(z_vec - z_vec.max())
            a_vec = a_vec / a_vec.sum()
            F_t, grad_a = compute_full_grad_smooth(a_vec, t_curr)
            # Convert to z-gradient: dF/dz[i] = a[i] * (grad_a[i] - sum_j grad_a[j]*a[j])
            mean_grad = np.dot(grad_a, a_vec)
            grad_z = a_vec * (grad_a - mean_grad)
            return grad_z
        
        result = minimize(objective, z, method='L-BFGS-B', 
                         jac=gradient, 
                         options={'maxiter': 200, 'maxfun': 5000, 
                                  'gtol': 1e-8, 'ftol': 1e-12})
        
        if result.success or (result.fun < objective(z)):
            z = result.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            M_new = fast_conv(a_new).max()
            if M_new < best_val - 1e-18:
                best_val = M_new
                best_a = a_new.copy()
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    L-BFGS t={t_curr} ({elapsed:.1f}s): J={2*n*best_val:.8f}")
    
    return best_a, best_val

def lp_remez_single(a, max_cons=30, r=0.1, solver=None):
    """Single LP Remez step."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_lp_method(a0, time_limit, solver=None, r_init=0.08, verbose=False):
    """LP-based Remez with adaptive trust region."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(777)
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand = lp_remez_single(a, max_cons=40, r=r, solver=solver)
        
        accepted = False
        for t_frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-18:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.3, r * 1.05)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.6)
            if no_improve > 15:
                idx = rng.choice(n, size=min(n//5, 400), replace=False)
                a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with random pair moves."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(20, n // 100)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def make_init(n, kind='asym_gauss'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'asym_gauss':
        a = np.exp(-5.0 * (x - 0.35) ** 2 / 0.08)
    elif kind == 'asym_gauss_r':
        a = np.exp(-5.0 * (x - 0.65) ** 2 / 0.08)
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.1)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'asym_exp':
        a = np.exp(-3.0 * x * (1 - x) * 2) * np.exp(-2.0 * (x - 0.4)**2 / 0.05)
    elif kind == 'flat':
        a = np.ones(n)
    else:
        a = np.exp(-5.0 * (x - 0.4) ** 2 / 0.08)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
