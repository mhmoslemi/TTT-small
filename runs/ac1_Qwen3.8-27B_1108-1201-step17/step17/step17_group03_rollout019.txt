def fast_conv(a):

    size = 2 * n - 1

def compute_exact_subgradient(a, conv, top_k=30, beta_power=3.0):
    """Compute soft subgradient of max(conv) w.r.t. a using top peaks."""
    n = len(a)
    conv_len = len(conv)
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    vals = np.maximum(conv[top_idx], 1e-30)
    w = vals ** beta_power
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   margin_frac=1e-3, use_nesterov=True):
    """LP Frank-Wolfe with adaptive margins and Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_b = None

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_conv(y)
        M_curr = conv.max()
        if M_curr < 1e-16:
            break

        # Adaptive constraint selection
        margins = M_curr - conv
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 3:
            active_idx = np.argsort(margins)[:top_K * 3]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)

        # Build constraint matrix
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, int(k_int) - n + 1)
            i_end = min(n, int(k_int) + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            T_sel[row, idx] = y[int(k_int) - idx]

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(0.0), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Frank-Wolfe step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        # Track best
        my = fast_conv(y).max()
        if my < best_m * (1 - 1e-10):
            best_m = my
            best_a = y.copy()
            margin_frac = max(margin_frac * 0.92, 5e-6)

        mx = fast_conv(x).max()
        if mx < best_m * (1 - 1e-10):
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:

            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")

    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=42):
    """Targeted coordinate descent: move mass from high-contributing positions to reduce peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()

        i_end = min(n, peak_k + 1)

        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 1
            continue

        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        for li in top_local:

                break
            src = int(idx[li])

            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

                    dst = int(rng.integers(n))

                    if s_t < 1e-15:

def freq_perturb(a, strength=0.01, rng=None):
    """Perturb phase of DFT to escape local minima."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'beta_3_3':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 3 * (1 - x) ** 3)
    elif family == 'raised_cos':
        x = np.linspace(0, 1, n)
        raise_cos = 0.5 * (1 + np.cos(2 * np.pi * x))
        return norm_sum1(raise_cos + 0.01)
    elif family == 'gaussian':
        x = np.linspace(0, 1, n)
        center = 0.5
        sigma = 0.2
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return norm_sum1(g)
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)

    rng_init = np.random.default_rng(7)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        families = ['sine', 'cosine', 'beta_2_2', 'beta_3_3', 'raised_cos', 'gaussian']
        for fam in families:
            a_init = make_parametric(n0, fam)
            # Make palindromic
            a_pal = make_palindromic(a_init)
            a_pal = norm_sum1(a_pal)
            sc = evaluate_safe(a_pal)
            if np.isfinite(sc) and sc < best_score:
                print(f"  Better init [{fam}]: {sc:.6f}")
                best_seq = a_pal.copy()
                best_score = sc
    # Also try random
    a_rand = rng_init.dirichlet(np.ones(n0))
    a_rand_pal = make_palindromic(a_rand)
    a_rand_pal = norm_sum1(a_rand_pal)
    sc = evaluate_safe(a_rand_pal)
    if np.isfinite(sc) and sc < best_score:
        print(f"  Better init [random]: {sc:.6f}")
        best_seq = a_rand_pal.copy()
        best_score = sc

    # === Phase 2: Mirror Descent warmup ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror Descent (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(a_in, t2, top_k=60, beta=4.0, c0=3.5,
                                           restart_every=3000, verbose=True, use_backtrack=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "md_warmup")

    # === Phase 3: LP Frank-Wolfe (main optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW main (n={n_best}) ===")
    t3 = min(remaining * 0.30, 500.0)
    if t3 > 50 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True, top_K=300,
                                   margin_frac=2e-3)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 4: Coordinate Descent Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: CD Polish (n={n_best}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_polish(a_in, t4, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd1")

    # === Phase 5: Second LP with tighter margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW tight (n={n_best}) ===")
    t5 = min(remaining * 0.18, 280.0)
    if t5 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True, top_K=400,
                                   margin_frac=3e-4)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 6: Frequency perturb + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Freq-diversify (n={n_best}) ===")
    rng = np.random.default_rng(99)
    t6_total = min(remaining * 0.10, 150.0)
    if t6_total > 60:
        t6_per = t6_total / 3
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t6_per + 20:
                break
            a_pert = freq_perturb(best_seq, strength=0.02, rng=rng)
            a_out, _ = mirror_descent_optimize(a_pert, t6_per, top_k=50, beta=5.0, c0=4.0,
                                               restart_every=2000, verbose=False, use_backtrack=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 7: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration (base n={n_best}) ===")
    candidate_lengths = sorted(set([
        n_best - 200, n_best - 400,
        n_best + 200, n_best + 400, n_best + 600, n_best + 1000
    ]))
    candidate_lengths = [l for l in candidate_lengths if 2000 <= l <= 5500]

    for new_n in candidate_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        a_long_pal = make_palindromic(a_long)
        a_long_pal = norm_sum1(a_long_pal)

        t7a = min(remaining * 0.15, 150.0)
        if t7a > 40:
            a_out, _ = mirror_descent_optimize(a_long_pal, t7a, top_k=60, beta=4.0, c0=3.0,
                                               restart_every=2500, verbose=False, use_backtrack=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"md_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 4000:
            t7b = min(remaining * 0.10, 100.0)
            if t7b > 30:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False, top_K=200,
                                           margin_frac=2e-3)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

    # === Phase 8: Final LP on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    t8 = min(remaining * 0.12, 150.0)
    if t8 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=500,
                                   margin_frac=1e-4)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final CD (n={n_best}) ===")
    if remaining > 40:
        t9 = min(remaining * 0.20, 120.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_descent_polish(a_in, t9, verbose=True, rng_seed=777)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "cd_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Safety
    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
