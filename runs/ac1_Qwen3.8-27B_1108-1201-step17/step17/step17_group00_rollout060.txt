"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max smoothed gradient descent with Adam
2. Adaptive temperature annealing
3. Targeted peak reduction
4. Multi-scale refinement and multi-length exploration
"""

import numpy as np
import time
import math

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def soft_max_value_and_weights(c, T):
    """Compute soft-max value and weights for the peak."""
    c_max = c.max()
    shifted = (c - c_max) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    val = c_max + T * np.log(exp_vals.sum())
    return val, weights

def compute_gradient_top_lags(a, weights, top_K):
    """Compute gradient using only top K lags for efficiency."""
    n = len(a)
    c_len = 2 * n - 1
    # Get top K lags by weight
    top_idx = np.argsort(weights)[::-1][:top_K]
    top_w = weights[top_idx]
    # Renormalize
    top_w = top_w / top_w.sum()
    
    grad = np.zeros(n)
    for k, w in zip(top_idx, top_w):
        k = int(k)
        # Contribution: w * (a[k-j] + a[j-k]) for each j
        # a[k-j] is valid for max(0, k-n+1) <= j <= min(k, n-1)
        j_start1 = max(0, k - n + 1)
        j_end1 = min(k, n - 1)
        if j_start1 <= j_end1:
            # a[k-j] for j in [j_start1, j_end1]
            indices = np.arange(j_start1, j_end1 + 1)
            a_idx = k - indices
            grad[indices] += w * a[a_idx]
        
        # a[j-k] is valid for max(k, 0) <= j <= min(k+n-1, n-1)
        j_start2 = max(k, 0)
        j_end2 = min(k + n - 1, n - 1)
        if j_start2 <= j_end2:
            indices = np.arange(j_start2, j_end2 + 1)
            a_idx = indices - k
            grad[indices] += w * a[a_idx]
    
    return grad

def adam_soft_max_optimize(a0, time_limit, T_init, T_min, top_K, lr, momentum_betas, restart_every, verbose=True):
    """Optimize using soft-max gradient with Adam."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    c = fast_autocorr(a)
    best_J = 2.0 * n * c.max()
    t0 = time.monotonic()
    
    beta1, beta2 = momentum_betas
    m = np.zeros(n)
    v = np.zeros(n)
    T = T_init
    t = 0
    stall = 0
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Adaptive temperature: decrease over time
        T = max(T_min, T_init * (T_min / T_init) ** (t / 5000.0))
        
        c = fast_autocorr(a)
        S = a.sum()
        sm_val, weights = soft_max_value_and_weights(c, T)
        
        # Compute gradient of sm_val w.r.t. a
        grad_sm = compute_gradient_top_lags(a, weights, top_K)
        
        # Gradient of J = 2n * sm / S^2
        # dJ/da[j] = 2n * grad_sm[j] / S^2 - 2 * (2n * sm / S^2) * a[j] / S
        #         = 2n * grad_sm[j] / S^2 - 4n * sm * a[j] / S^3
        grad_J = 2.0 * n * grad_sm / (S * S) - 4.0 * n * sm_val * a / (S * S * S)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_J
        v = beta2 * v + (1 - beta2) * (grad_J ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr_t = lr / (1 + 0.1 * t / 1000)  # decay
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        c_new = fast_autocorr(a)
        J_new = 2.0 * n * c_new.max()
        
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            stall = 0
            # Perturb and restart
            pert = rng.normal(0, 0.01, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / a_r.sum()
            a = a_r
            m = np.zeros(n)
            v = np.zeros(n)
            T = T_init  # Reset temperature
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Adam-SM restart t={t} ({elapsed:.0f}s): J={best_J:.6f}")
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam-SM t={t} ({elapsed:.0f}s): J={best_J:.6f}, T={T:.4f}")
    
    return best_a, best_J

3. Find the "bottleneck" pairs (i, k*-i) that contribute most to the peak
4. For each such pair, consider moving mass from a[i] (or a[k*-i]) to a position j that:
   - Reduces c[k*] the most
   - Increases the second-highest c[k] the least
5. This is a 2-opt style move

Actually, let me think about this more carefully. The constraint is that we need to maintain sum(a) = 1 and a >= 0.

A "move" is: reduce a[i] by δ and increase a[j] by δ (or reduce a[i] by δ1 and a[k] by δ2, increase a[j] by δ1 and a[l] by δ2, etc.)

The effect on c[m] is:
Δc[m] = -(a[m-i] + a[i-m]) * δ + (a[m-j] + a[j-m]) * δ + higher order terms

(where a[x] = 0 if x not in [0,n-1])

So for a move from i to j:
Δc[m] ≈ δ * (a[m-j] + a[j-m] - a[m-i] - a[i-m])

We want to reduce c[k*] (the peak) without increasing any other c[m] above the current second-highest.

This gives us a linear constraint on δ for each m:
δ * (a[m-j] + a[j-m] - a[m-i] - a[i-m]) <= threshold_m

where threshold_m = second_max - c[m] (or some margin).

The maximum δ is the minimum over all m of threshold_m / (a[m-j] + a[j-m] - a[m-i] - a[i-m]) (for positive denominators).

This is a nice structure! We can:
1. For each candidate (i, j) pair, compute the maximum feasible δ
2. Pick the (i, j) pair that gives the best improvement
3. Execute the move

But enumerating all (i, j) pairs is O(n^2), which is too slow for n=5000.

Instead, we can:
1. Identify the top contributors to the peak: (i, k*-i) pairs with large a[i]*a[k*-i]
2. For each such i, find the best j to move to
3. The best j is one that minimizes the increase in c[m] for all m

    a = _norm_sum1(a0)
    best_a = a.copy()
    best_J = 2.0 * n * _fast_autocorr(a).max()
    t0 = time.monotonic()
    b1, b2 = 0.85, 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    T = T_init
    t = 0
    stall = 0
    rng = np.random.default_rng(77)

    while time.monotonic() - t0 < time_limit:
        t += 1
        frac = t / max(1, int(time_limit * 2.0))
        T = T_init * (T_min / T_init) ** min(frac, 1.0)

        c = _fast_autocorr(a)
        S = a.sum()
        c_max = c.max()
        shifted = np.clip((c - c_max) / T, -40.0, 40.0)
        ew = np.exp(shifted)
        w = ew / ew.sum()
        sm_val = c_max + T * np.log(ew.sum())

        g_sm = _compute_gradient_peaks(a, w, top_K)
        grad_J = 2.0 * n * g_sm / (S * S) - 4.0 * n * sm_val * a / (S * S * S)

        m = b1 * m + (1.0 - b1) * grad_J
        v = b2 * v + (1.0 - b2) * (grad_J * grad_J)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        lr_t = lr * 1.0 / (1.0 + 0.05 * t / 500.0)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        a = a_new

        c_new = _fast_autocorr(a)
        J_new = 2.0 * n * c_new.max()
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.015, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / a_r.sum()
            a = a_r
            m = np.zeros(n)
            v = np.zeros(n)
            T = T_init

        if t % 500 == 0:
            print(f"    Adam t={t} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, T={T:.4f}")

    return best_a, best_J


def _blame_update(a0, time_limit, epochs=300):
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_J = 2.0 * n * _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for ep in range(epochs):
        if time.monotonic() - t0 > time_limit:
            break
        c = _fast_autocorr(a)
        thr = c.max() * 0.95
        active_mask = c > thr
        if not np.any(active_mask):
            active_mask = np.zeros(len(c), dtype=bool)
            active_mask[int(np.argmax(c))] = True
        excess = np.where(active_mask, c - thr, 0.0)
        blame = np.zeros(n)
        for k in np.where(active_mask)[0]:
            ki = int(k)
            j_s = max(0, ki - n + 1)
            j_e = min(n - 1, ki)
            if j_s > j_e:
                continue
            idx = np.arange(j_s, j_e + 1)
            a1 = a[idx]
            a2 = a[ki - idx]
            blame[idx] += excess[k] * (a1 + a2)
        b_max = blame.max()
        if b_max < 1e-20:
            break
        eta = 0.3 / b_max
        log_a = np.log(np.maximum(a, 1e-30)) - eta * blame
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        a = a_new
        c_new = _fast_autocorr(a)
        J_new = 2.0 * n * c_new.max()
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a.copy()
        else:
            pert = rng.normal(0, 0.005, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 0.0)
            a_r = a_r / a_r.sum()
            a = a_r

        if ep % 50 == 0:
            print(f"    Blame ep={ep} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def _parametric_init(n):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = np.ones(n) / n
    best_val = 2.0 * n * _fast_autocorr(best_a).max()

    def try_update(a, label):
        nonlocal best_a, best_val
        a = _norm_sum1(a)
        val = 2.0 * n * _fast_autocorr(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"    Parametric [{label}]: J={val:.6f}")

    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        try_update((i_arr + 1.0) ** p, f"pow{p}")
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        try_update((n - i_arr) ** p, f"ipow{p}")
    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.5, 1.0, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            try_update(a, f"beta{alpha}_{beta}")
    for c0 in [0.3, 0.5, 0.7]:
        for sig in [0.1, 0.2, 0.3, 0.5]:
            try_update(np.exp(-((x - c0) ** 2) / (2 * sig ** 2)), f"gc{c0}_{sig}")
    for c0 in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20, 50]:
            try_update(1.0 / (1.0 + np.exp(-s * (x - c0))), f"sg{c0}_{s}")
    try_update(np.sin(np.pi * x) ** 2, "sine")
    try_update(np.abs(np.sin(np.pi * x)), "abs_sine")
    for rate in [1, 2, 5, 10]:
        try_update(np.exp(-rate * x), f"er{rate}")
        try_update(np.exp(rate * x), f"en{rate}")

    print(f"  Parametric done: J={best_val:.6f}")
    return best_a, best_val


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = _get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = _evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = _norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = _norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={time.monotonic()-t_global:.0f}s)")

    # Phase 0: Quick parametric (only if no parent or parent is bad)
    remaining = time_budget - (time.monotonic() - t_global)
    n0 = len(best_seq)
    if best_score > 1.55 and remaining > 30:
        a_p, v_p = _parametric_init(n0)
        check_update(a_p, "parametric")

    # Phase 1: LP-FW main
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-FW (n={n_best}) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 40 and n_best <= 6000:
        a_out, _ = _lp_fw(_norm_sum1(best_seq), t1, solver, init_K=500, max_K=4000, add_n=50)
        check_update(a_out, f"lp1_n{n_best}")

    # Phase 2: Soft-peak Adam
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Adam (n={n_best}) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, _ = _soft_peak_adam(_norm_sum1(best_seq), t2, T_init=0.08, T_min=0.008,
                                   top_K=60, lr=0.02, restart_every=8000)
        check_update(a_out, f"adam_n{n_best}")

    # Phase 3: Blame update
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Blame (n={n_best}) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 20:
        a_out, _ = _blame_update(_norm_sum1(best_seq), t3, epochs=200)
        check_update(a_out, f"blame_n{n_best}")

    # Phase 4: LP-FW again (refined)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW refined (n={n_best}) ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 40 and n_best <= 6000:
        a_out, _ = _lp_fw(_norm_sum1(best_seq), t4, solver, init_K=600, max_K=5000, add_n=60)
        check_update(a_out, f"lp2_n{n_best}")

    # Phase 5: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length ===")
    deltas = [500, 1000, -500, 1500]
    tried = set()
    for d in deltas:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        new_n = n_best + d
        if new_n < 1000 or new_n > 7000 or abs(d) in tried:
            continue
        tried.add(abs(d))
        a_long = _interpolate_to_length(best_seq, new_n)
        sc = _evaluate_safe(a_long)
        print(f"  n={new_n}: score after interp = {sc:.6f}")

        t5a = min(remaining * 0.30, 150.0)
        if t5a > 30:
            a_out, _ = _soft_peak_adam(a_long, t5a, T_init=0.06, T_min=0.006,
                                       top_K=50, lr=0.015, restart_every=5000)
            check_update(a_out, f"adam_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 5000:
            t5b = min(remaining * 0.35, 150.0)
            if t5b > 30:
                a_out, _ = _lp_fw(_norm_sum1(best_seq) if len(_norm_sum1(best_seq)) == new_n else a_long,
                                  t5b, solver, init_K=400, max_K=3000, add_n=40)
                check_update(a_out, f"lp_n{new_n}")

    # Phase 6: Final Adam polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final Adam (n={n_best}) ===")
    t6 = min(remaining * 0.40, 200.0)
    if t6 > 20:
        a_out, _ = _soft_peak_adam(_norm_sum1(best_seq), t6, T_init=0.04, T_min=0.003,
                                   top_K=80, lr=0.01, restart_every=6000)
        check_update(a_out, f"adam_final_n{n_best}")

    # Phase 7: Final blame
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Blame (n={n_best}) ===")
    t7 = min(remaining * 0.40, 150.0)
    if t7 > 15:
        a_out, _ = _blame_update(_norm_sum1(best_seq), t7, epochs=300)
        check_update(a_out, f"blame_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = _norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = _evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
