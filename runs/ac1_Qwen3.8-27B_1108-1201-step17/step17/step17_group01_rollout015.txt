def fast_autocorr(a):
    """Compute full autocorrelation (convolution of a with itself) via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def norm_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interp_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        y = a + beta_t * (a - a_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # Compute autocorrelation
        c = fast_autocorr(y)

        # Softmax weights (numerically stable)
        c_max = c.max()
        p = np.exp((c - c_max) / mu)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(len(c)) / len(c)
        else:
            p = p / p_sum

        # Compute gradient
        g = smoothed_grad(y, p)
        g = g - g.mean()  # Center to handle the simplex constraint better

        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror descent step (exponential update)
        x_new = np.log(y + 1e-15) - eta * g
        x_new -= x_new.max()
        a_raw = np.exp(x_new)
        a_new = a_raw / a_raw.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        a_prev = a
        a = a_new

        # Check improvement
        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Decay smoothing
        mu = max(mu * mu_decay, mu_min)

        # Restart if stalled
        if stall > restart_every:
            stall = 0
            # Perturb from best
            strength = 0.05
            pert = rng.normal(0, strength, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            a_prev = a_r.copy()
            x = np.log(a + 1e-15)
            v = np.zeros(n)
            mu = max(mu, 0.3)  # Re-smooth after restart
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    SmoothND restart t={t} ({elapsed:.0f}s): J={J:.6f}, mu={mu:.4f}")

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SmoothND t={t} ({elapsed:.0f}s): J={J:.6f}, mu={mu:.4f}, stall={stall}")

    return best_a, best_m


def get_lp_solver():
    """Get a working LP solver."""
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def lp_refine(a0, time_limit, solver, verbose=False, top_K=200, max_iters=5000):
    """
    LP-based Frank-Wolfe refinement.
    Solves LP relaxation with adaptive constraint selection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    import cvxpy as cp

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()
    
    # Precompute T matrix structure (banded)
    # T[k, i] = 1 if k-i is in [0, n-1], representing c_k = sum_i T[k,i] * a[i] * a[k-i]
    # For the linearization, we fix one factor: T_lin[k, i] = a[k-i] if valid
    # This changes with y, so we recompute each iteration.

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Build linearized T at current y
        # T[k, i] = y[k-i] if 0 <= k-i < n, else 0
        # For the LP: maximize sum(b) s.t. T_sel @ b <= M * (sum b)
        # (homogeneous formulation)
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Select binding constraints
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        # Build T_sel @ y for the constraints
        # T_sel[k, i] = y[k-i] if 0 <= k-i < n
        T_sel = np.zeros((K_sel, n))
        for idx_i, k in enumerate(binding_idx):
            k_int = int(k)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    T_sel[idx_i, i] = y[j]

        # LP: maximize sum(b) s.t. T_sel @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Frank-Wolfe step with Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def make_init_power(n, power=1.0, center=False):
    x = np.arange(1, n + 1, dtype=np.float64)
    if center:
        x = np.abs(x - (n + 1) / 2.0) + 1.0
        a = x ** (-power)
    else:
        a = x ** power
    return norm_sum1(a)


def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_init_sinc(n, bandwidth=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bandwidth
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_init_humps(n, n_humps=3, spread=0.15):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_humps):
        center = (i + 1) / (n_humps + 1)
        a += np.exp(-((x - center) ** 2) / (2 * spread ** 2))
    a += 0.01
    return norm_sum1(a)


def make_init_random(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def coordinate_polish(a0, time_limit, verbose=False, seed=42):
    """
    Coordinate-wise mass transport for final polish.
    Moves mass from high-contributing positions to low-cost positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:15]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = int(idx[np.argmax(contrib)])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best destination (low contribution to top peaks)
            costs = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                js2 = max(0, k2i - n + 1)
                je2 = min(n - 1, k2i)
                idx2 = np.arange(js2, je2 + 1)
                costs[idx2] += a[k2i - idx2]

            # Try moving to a few candidates
            candidates = np.argsort(costs + a * 10)[:max(5, n // 10)]
            for dst in candidates[:5]:
                dst_int = int(dst)
                if dst_int == src:
                    continue
                for frac in [0.05, 0.03, 0.01, 0.08, 0.12]:
                    amt = src_val * frac
                    if amt < 1e-14:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst_int] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break

            if improved:
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Uses smoothed Nesterov mirror descent + LP refinement + coordinate polish.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    rng = np.random.default_rng(42)

    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3000
        best_seq = make_init_sinc(n_base, bandwidth=5.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting n={n0}, score={best_score:.6f}")

    # === Phase 1: Smoothed Nesterov on diverse inits ===
    print(f"\n=== Phase 1: Smoothed Nesterov (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.35, 600.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_init_power(n0, power=0.5))
        labels.append("sqrt_ramp")
        inits.append(make_init_power(n0, power=1.0))
        labels.append("linear")
        inits.append(make_init_sinc(n0, bandwidth=4.0))
        labels.append("sinc4")
        inits.append(make_init_sinc(n0, bandwidth=8.0))
        labels.append("sinc8")
        inits.append(make_init_humps(n0, n_humps=3, spread=0.15))
        labels.append("3hump")
        inits.append(make_init_random(n0, seed=7, n_freq=6))
        labels.append("smooth1")
        inits.append(make_init_power(n0, power=1.0, center=True))
        labels.append("center_inv")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 50)], 5))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(15.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 10:
                break
            print(f"  SmoothND '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = smoothed_nesterov(init, t_avail, mu_init=0.3, mu_min=0.01,
                                          c0=3.0, verbose=False, seed=42 + idx)
            try_update(a_out, f"smooth_{lbl}")

    # === Phase 2: LP refinement on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP Refinement (n={n_best}) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30 and n_best <= 5000:
        solver = get_lp_solver()
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_refine(a_in, t2, solver, verbose=False, top_K=300, max_iters=5000)
        try_update(a_out, f"lp_n{n_best}")

    # === Phase 3: More Smoothed Nesterov (longer, from best) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smoothed Nesterov long (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = smoothed_nesterov(a_in, t3, mu_init=0.2, mu_min=0.005,
                                      c0=2.5, verbose=False, seed=123)
        try_update(a_out, f"smooth_long_n{n_best}")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate polish (n={n_best}) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t4, verbose=False, seed=99)
        try_update(a_out, f"coord_n{n_best}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length search ===")

    for new_n in [n_best + 400, n_best + 800]:
        remaining = time_budget - (time.monotonic() - t_global)

        a_long = interp_seq(best_seq, new_n)
        try_update(a_long, f"interp_n{new_n}")

        t5a = min(remaining * 0.25, 150.0)
        if t5a > 20:
            a_out, _ = smoothed_nesterov(a_long, t5a, mu_init=0.3, mu_min=0.01,
                                          c0=3.0, verbose=False, seed=77)
            try_update(a_out, f"smooth_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t5b = min(remaining * 0.20, 100.0)
            if t5b > 15:
                a_out, _ = coordinate_polish(a_out if 'a_out' in dir() else a_long, t5b, seed=55)
                try_update(a_out, f"coord_n{new_n}")

    # Try slightly shorter
    for new_n in [max(2000, n_best - 500), max(1500, n_best - 1000)]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60 or new_n < 1000:

        print(f"  Trying n={new_n}...")
        a_short = interp_seq(best_seq, new_n)
        try_update(a_short, f"interp_short_n{new_n}")

        t5c = min(remaining * 0.30, 120.0)
        if t5c > 20:
            a_out, _ = smoothed_nesterov(a_short, t5c, mu_init=0.3
