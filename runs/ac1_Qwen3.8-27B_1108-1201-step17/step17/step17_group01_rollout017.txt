Let me also think about the spectral approach more carefully.

If $A(\omega) = \sum_i a_i e^{-i\omega i}$, then the autocorrelation is:
$c_k = \frac{1}{2\pi} \int_0^{2\pi} |A(\omega)|^2 e^{i\omega k} d\omega$

So $c$ is the inverse Fourier transform of $|A(\omega)|^2$. To make $c$ flat, we need $|A(\omega)|^2$ to be flat (constant), which means $|A(\omega)|$ should be constant. This is the "flat spectrum" condition.

But we also need $a_i \geq 0$, which constrains what spectra are achievable.

The approach:
1. Start with a valid $a \geq 0$, $\sum a = 1$
2. Compute $|A(\omega)|$
3. Modify $|A(\omega)|$ to be flatter (e.g., $|A|_{new} = |A|^{1-\alpha}$ for some $\alpha > 0$)
4. Invert to get new $a$ (using the original phase)
5. Project to $a \geq 0$ (set negative values to 0)
6. Renormalize
7. Repeat

This is essentially the Gerchberg-Saxton algorithm adapted for this problem.

The challenge is that step 5 (projection to non-negativity) can destroy the flatness of the spectrum. But iterating should help.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based adaptive mirror descent with softmax-smoothed subgradient
2. Spectral domain iterative flattening (Gerchberg-Saxton variant)
3. Multi-start with diverse initializations (power-law, cosine, random smooth)
4. Adaptive length exploration
5. Coordinate descent polish
"""

import numpy as np
import time

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _compute_grad(a, w, fft_size):
    """Compute 2 * cross_corr(w, a) where w has length 2n-1, a has length n."""
    n = len(a)
    w_pad = np.zeros(fft_size)
    w_pad[:2*n-1] = w
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    W = np.fft.rfft(w_pad)
    A = np.fft.rfft(a_pad)
    R = np.fft.irfft(W * np.conj(A), fft_size)
    return 2.0 * R[:n]

def _norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _mirror_descent(a0, time_limit, beta0=30.0, beta_min=5.0, c0=3.0, 
                    restart_every=5000, verbose=False, seed=42):
    n = len(a0)
    a = _norm_sum1(a0)
    fft_size = 1 << (3 * n - 2).bit_length()
    
    best_M = _fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    beta = beta0
    stall = 0
    it = 0
    momentum = 0.0
    v = np.zeros(n)  # for momentum
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = _fast_autocorr(a)
        M = c.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            stall = 0
            beta = min(beta * 1.5, beta0)
            pert = rng.normal(0, 0.02, n) * best_a
            a = best_a + pert
            a = _norm_sum1(a)
            v = np.zeros(n)
            continue
        
        # Softmax weights
        c_shift = c - M
        log_w = beta * c_shift
        lw_max = log_w.max()
        w = np.exp(log_w - lw_max)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2*n-1) / (2*n-1)
        else:
            w = w / w_sum
        
        # Gradient
        g = _compute_grad(a, w, fft_size)
        
        # Adaptive step size
        g_norm = np.abs(g).max()
        if g_norm < 1e-15:
            break
        eta = c0 / (g_norm * (1.0 + 0.5 * np.sqrt(it / 100.0)))
        eta = min(eta, 5.0)
        
        # Mirror descent with momentum
        v = momentum * v + (1 - momentum) * g
        log_update = -eta * v
        log_update -= log_update.mean()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
        beta = max(beta_min, beta * 0.9999)
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    MD it={it} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}, eta={eta:.4f}")
    
    return best_a, 2.0 * n * best_M
