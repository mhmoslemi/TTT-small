def fast_conv(a):

def exact_subgradient(a, b, tol=0.999):
    """Compute exact subgradient of max(b) at point a.
    Returns average of gradients at all near-max lags."""
    n = len(a)
    M = b.max()
    if M < 1e-30:
        return np.ones(n) / n
    active = np.where(b >= M * tol)[0]
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        # grad b[k]/a[l] = 2*a[k-l] if 0 <= k-l < n
        valid = np.arange(max(0, k_int - n + 1), min(n, k_int + 1))
        for l in valid:
            g[l] += 2.0 * a[k_int - l]
    g /= len(active)
    return g


def exact_subgradient_fast(a, b, tol=0.999):
    """Vectorized exact subgradient computation."""
    n = len(a)
    M = b.max()
    if M < 1e-30:
        return np.ones(n) / n
    active = np.where(b >= M * tol)[0]
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        l_start = max(0, k_int - n + 1)
        l_end = min(n - 1, k_int)
        if l_start <= l_end:
            g[l_start:l_end+1] += 2.0 * a[k_int - (l_start + np.arange(l_end - l_start + 1))]
    g /= len(active)
    return g


def spectral_init(n, n_iters=200, seed=42):
    """Gerchberg-Saxton style initialization for flat power spectrum."""
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.5, 1.5, n)
    a = a / a.sum()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        # Frequency domain: flatten magnitude
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        target_mag = np.mean(mag)
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        # Keep DC component
        A_new[0] = A[0]
        
        # Time domain: project to non-negative, sum=1
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check convergence
        if it > 10:
            b = fast_conv(a_new)
            b_curr = fast_conv(a)
            if abs(b.max() - b_curr.max()) < 1e-12:
                break
        a = a_new
    
    return a

def lp_remez_single(a, max_cons=30, r=0.1, solver=None):
    """Single LP Remez step with two-sided trust region."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    # Two-sided trust region
    lower = a * (1.0 - r)
    upper = a * (1.0 + r)
    constraints.append(a_var >= lower)
    constraints.append(a_var <= upper)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.1)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = (1.0 - (2.0 * x - 1.0) ** 4)
    elif kind == 'beta':
        # Beta distribution shape
        a = (x * (1 - x)) ** 0.5
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    while time.monotonic() - t0 < time_limit:
        it += 1
        
        beta_t = (it - 1.0) / (it + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        ac = fast_conv(y)
        M = ac.max()
        
        # Top-k subgradient
        k_max = min(top_k, len(ac))
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
        top_idx = np.sort(top_idx)
        
        g = np.zeros(n)
        for k in top_idx:
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n - 1, k_int)
            if j_start <= j_end:
                idx = np.arange(j_start, j_end + 1)
                g[idx] += 2.0 * y[k_int - idx]
        g /= k_max
        
        eta = c / (np.sqrt(it) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stallCount = stall_count + 1
            stall_count = stallCount
        
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, int(max_stall * 0.6))
            strategy = int(rng.integers(3))
            if strategy == 0:
                pert = rng.normal(0, 0.02, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n//4, 600), replace=False)
                x_r[idx] *= rng.uniform(0.5, 1.5, len(idx))
            else:
                x_r = best_a * (1 + rng.normal(0, 0.005, n))
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.5, c * 0.6)
    
    return best_a, best_m
