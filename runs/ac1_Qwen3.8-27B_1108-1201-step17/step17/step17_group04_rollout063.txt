"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated Nesterov mirror descent with annealed soft subgradient
2. Efficient subgradient via cross-correlation structure (shift-reverse of a)
3. Aggressive peak-targeting mass relocation moves
4. Coordinate descent polishing
5. Multi-length exploration with interpolation
Key insight: the subgradient of conv_k w.r.t. a is 2*a reversed-and-shifted,
enabling O(n log n) gradient computation via FFT cross-correlation.
"""

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Clamp to non-negative and normalize to sum 1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def soft_subgradient(a, c, top_K, beta):
    """
    Compute soft subgradient of max(conv(a,a)) using FFT cross-correlation.
    subgrad[m] = 2 * sum_k w_k * a[k-m] (valid indices)
    This equals 2 * cross_correlation(w_full, a)[m].
    """
    n = len(a)
    size = 2 * n - 1
    # For cross-correlation of length `size` with length `n`, need fft_size >= size + n - 1
    fft_size = 1 << (size + n - 2).bit_length()

    top_idx = np.argsort(c)[::-1][:top_K].astype(int)
    vals = np.maximum(c[top_idx], 1e-30)
    w = vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(top_K) / top_K
    else:
        w = w / w_sum

    w_full = np.zeros(size)
    w_full[top_idx] = 2.0 * w

    W = np.fft.rfft(w_full, fft_size)
    A = np.fft.rfft(a, fft_size)
    grad = np.fft.irfft(W * np.conj(A), fft_size)[:n]

    return grad


def exact_subgradient(a, c):
    """Exact subgradient at the single peak lag."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size + n - 2).bit_length()

    k = int(np.argmax(c))
    w_full = np.zeros(size)
    w_full[k] = 2.0

    W = np.fft.rfft(w_full, fft_size)
    A = np.fft.rfft(a, fft_size)
    grad = np.fft.irfft(W * np.conj(A), fft_size)[:n]

    return grad


def nesterov_mirror_descent(a0, time_limit, seed=42):
    """
    Nesterov-accelerated mirror descent in log-space with adaptive
    soft subgradient and annealed temperature.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Nesterov state
    x = a.copy()
    x_prev = a.copy()

    # Hyperparameters
    top_K = 20
    beta = 4.0
    c_param = 4.0
    exact_frac = 0.3
    restart_every = 3000
    stall = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if iteration > 1:
            beta_t = (iteration - 1.0) / (iteration + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-15:
            y = x.copy()
        else:
            y = y / s_y

        # Compute convolution and subgradient
        c = fast_conv(y)
        M = c.max()

        # Adaptive: mix exact and soft subgradient
        if rng.random() < exact_frac:
            g = exact_subgradient(y, c)
        else:
            g = soft_subgradient(y, c, top_K, beta)

        # Adaptive step size: Polyak-like
        lower_bound = 1.0 / (2.0 * n - 1.0)
        gap = M - lower_bound
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            eta = 1e-8
        else:
            eta = c_param * gap / g_norm_sq
        eta = min(eta, 2.0 / (iteration + 1.0))
        eta = max(eta, 1e-10)

        # Mirror descent step (exponential family)
        log_y = np.log(np.maximum(y, 1e-30))
        log_new = log_y - eta * g
        max_log = log_new.max()
        a_new = np.exp(log_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with adaptation
        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.005 * min(stall // restart_every, 5)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            top_K = min(top_K + 5, 60)
            beta = max(beta - 0.2, 2.0)
            c_param = min(c_param + 0.5, 10.0)
            exact_frac = min(exact_frac + 0.05, 0.6)

    return best_a, best_J


def peak_targeting_moves(a0, time_limit, seed=77):
    """
    Direct peak-targeting: identify the peak lag, compute its gradient,
    and relocate mass from high-gradient positions to low-gradient positions.
    Uses multiple trial step sizes and accepts only improving moves.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(a)
        k_peak = int(np.argmax(c))

        # Gradient at peak: grad[m] = 2*a[k_peak - m]
        m_start = max(0, k_peak - n + 1)
        m_end = min(n, k_peak + 1)
        grad = np.zeros(n)
        for m in range(m_start, m_end):
            j = k_peak - m
            if 0 <= j < n:
                grad[m] = 2.0 * a[j]

        # Positions with high gradient (contribute to peak) vs low gradient
        grad_positive = grad[grad > 0]
        if len(grad_positive) == 0:
            stall += 10
            continue
        threshold = np.median(grad_positive)
        high_idx = np.where((grad > threshold) & (a > 1e-8))[0]
        low_idx = np.where((grad < threshold * 0.5) & (a >= 0))[0]

        if len(high_idx) == 0 or len(low_idx) == 0:
            # Try random moves
            if rng.random() < 0.1:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * 0.05
                    a[i] -= amt
                    a[j] += amt
                    a = normalize(a)
                    J_new = compute_J(a)
                    if J_new < best_J:
                        best_J = J_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a = best_a.copy()
                        stall += 1
            else:
                stall += 1
            continue

        improved = False
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            src = high_idx[int(rng.integers(len(high_idx)))]
            dst = low_idx[int(rng.integers(len(low_idx)))]
            if src == dst or a[src] < 1e-8:
                continue

            # Try several transfer fractions
            fracs = [0.001, 0.003, 0.01, 0.03, 0.05, 0.1, 0.2, 0.35]
            found = False
            for frac in fracs:
                amt = a[src] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = normalize(a_trial)
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if found:
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Larger perturbation to escape
                n_pert = max(10, n // 200)
                pert_positions = rng.choice(n, size=n_pert, replace=False)
                for p in pert_positions:
                    if a[p] > 1e-8:
                        frac = rng.uniform(0.01, 0.15)
                        amt = a[p] * frac
                        a[p] -= amt
                        dst = int(rng.integers(n))
                        a[dst] += amt
                a = normalize(a)
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a = best_a.copy()

def coordinate_descent_polish(a0, time_limit, seed=200):
    """
    Coordinate descent: for each coordinate, try scaling it by various
    factors and keep the best. Cycles through coordinates until time runs out.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    factors = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]
    # For efficiency, pre-compute the "rest" contribution to each conv lag
    # But for simplicity, just use full conv computation (FFT is fast enough)

    order = list(range(n))
    rng.shuffle(order)
    pos = 0
    no_improve = 0
    max_no_improve = 3 * n

    while time.monotonic() - t0 < time_limit:
        i = order[pos % n]
        pos += 1

        if a[i] < 1e-12:
            # Try small positive values
            for val in [1e-6, 1e-4, 1e-3]:
                a_trial = a.copy()
                a_trial[i] = val
                a_trial = normalize(a_trial)
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1
            continue

        # Save original value
        orig_val = a[i]
        best_local = orig_val
        best_local_J = best_J

        for fac in factors:
            a_trial = a.copy()
            a_trial[i] = orig_val * fac
            a_trial = normalize(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_local_J - 1e-14:
                best_local_J = J_trial
                best_local = orig_val * fac

        if best_local != orig_val:
            a[i] = best_local
            a = normalize(a)
            best_J = best_local_J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    return best_a, best_J


def sinusoidal_restart(a0, rng, n_freqs=3):
    """Create a perturbed version with sinusoidal structure."""
    n = len(a0)
    x = np.arange(n)
    pert = np.zeros(n)
    for _ in range(n_freqs):
        f = rng.uniform(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.05)
        pert += amp * np.cos(2 * np.pi * f * x / n + phase)
    a_new = a0 + pert * a0
    a_new = normalize(a_new)
    return a_new


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Phase 1: Nesterov mirror descent (55% time)
    Phase 2: Peak targeting moves (20% time)
    Phase 3: Coordinate descent polish (15% time)
    Phase 4: Length exploration + final SGD (10% time)
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
    except Exception:
        pass

    if parent is not None:
        a0 = normalize(parent)
    else:
        n_init = 3200
        a0 = np.ones(n_init) / n_init

    n0 = len(a0)
    initial_score = evaluate_safe(a0)
    print(f"Initial: n={n0}, score={initial_score:.6f}")

    best_seq = a0.copy()
    best_score = initial_score

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Nesterov Mirror Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.55, 900.0)
    print(f"\n=== Phase 1: Nesterov MD (n={n0}, t={t_phase1:.0f}s) ===")

    if t_phase1 > 30:
        # Run multiple seeds to explore different basins
        n_runs = 3
        t_per_run = t_phase1 / n_runs
        for run in range(n_runs):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            t_avail = min(t_per_run, remaining - 10)
            if t_avail < 15:
                break

            if run == 0:
                init = a0.copy()
            elif run == 1:
                rng_tmp = np.random.default_rng(run + 100)
                init = sinusoidal_restart(a0, rng_tmp, n_freqs=3)
            else:
                rng_tmp = np.random.default_rng(run + 200)
                # Blend with slight noise
                noise = rng_tmp.normal(0, 0.01, n0)
                init = normalize(a0 + noise * a0)

            print(f"  Run {run+1}/{n_runs} (t={t_avail:.0f}s)")
            a_out, J_out = nesterov_mirror_descent(init, t_avail, seed=42 + run * 10)
            try_update(a_out, f"nesterov_run{run+1}")

    # === Phase 2: Peak Targeting ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Peak targeting (n={n_best}, remaining={remaining:.0f}s) ===")
    t_phase2 = min(remaining * 0.22, 300.0)
    if t_phase2 > 20:
        a_in = normalize(best_seq)
        a_out, J_out = peak_targeting_moves(a_in, t_phase2, seed=77)
        try_update(a_out, "peak_target")

        # Second round with different seed
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t_phase2b = min(remaining * 0.1, 100.0)
            if t_phase2b > 15:
                a_in = normalize(best_seq)
                a_out, J_out = peak_targeting_moves(a_in, t_phase2b, seed=321)
                try_update(a_out, "peak_target2")

    # === Phase 3: Coordinate Descent Polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent (n={n_best}, remaining={remaining:.0f}s) ===")
    t_phase3 = min(remaining * 0.18, 250.0)
    if t_phase3 > 15:
        a_in = normalize(best_seq)
        a_out, J_out = coordinate_descent_polish(a_in, t_phase3, seed=200)
        try_update(a_out, "coord_descent")

    # === Phase 4: Additional optimization + length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Final optimization (remaining={remaining:.0f}s) ===")

    # Try another SGD round on best
    if remaining > 40:
        t_sgd = min(remaining * 0.3, 200.0)
        if t_sgd > 15:
            a_in = normalize(best_seq)
            a_out, _ = nesterov_mirror_descent(a_in, t_sgd, seed=999)
            try_update(a_out, f"sgd_final_n{n_best}")

    # Try slightly different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        for delta_n in [200, -200, 400]:
            new_n = n_best + delta_n
            if new_n < 500 or new_n > 8000:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            a_interp = interpolate_sequence(best_seq, new_n)
            t_try = min(remaining * 0.25, 100.0)
            if t_try > 15:
                a_out, _ = nesterov_mirror_descent(a_interp, t_try, seed=555)
                try_update(a_out, f"len{n_new_n}")

    # Final peak targeting if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t_final = remaining * 0.6
        if t_final > 15:
            a_in = normalize(best_seq)
            a_out, _ = peak_targeting_moves(a_in, t_final, seed=1234)
            try_update(a_out, "peak_final")

    # Final coord descent if time remains
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_best = len(best_seq)
        t_cd = remaining * 0.7
        if t_cd > 10:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_descent_polish(a_in, t_cd, seed=456)
            try_update(a_out, "coord_final")

    # Final validation
    final_score = evaluate_safe(best_seq)
    if np.isfinite(final_score):
        best_score = final_score
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")

    return [float(x) for x in best_seq]
