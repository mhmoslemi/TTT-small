def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_T(y):
    n = len(y)
    rows = np.arange(2 * n - 1)[:, None]
    cols = np.arange(n)[None, :]
    idx = rows - cols
    valid = (idx >= 0) & (idx < n)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    T[valid] = y[idx[valid]]
    return T


def lp_solve_direction(y, M, solver, max_constraints=None):
    n = len(y)
    n_total = 2 * n - 1
    if max_constraints is None or max_constraints >= n_total:
        T = build_T(y)
        M_vec = M * np.ones(T.shape[0])
    else:
        ac = fast_autocorr(y)
        top_k = np.argsort(ac)[::-1][:max_constraints]
        T = build_T(y)
        T = T[top_k, :]
        M_vec = M * np.ones(len(top_k))
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M_vec])
    try:
        t_start = time.monotonic()
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    bs = b_val.sum()
    if bs < 1e-15:
        return None
    return b_val / bs

def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.1) / width) ** 2) + np.exp(-((x - 0.9) / width) ** 2) + 0.001
    return norm_sum1(a)


def make_ring(n, edge_width=0.08, center_floor=0.2):
    x = np.linspace(0, 1, n)
    dist = np.minimum(x, 1.0 - x)
    a = np.exp(-dist / edge_width) + center_floor
    return norm_sum1(a)


def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            a_out = accelerated_fw(inits[idx], t_avail, solver, verbose=False)
            try_update_state(state, a_out, "fw_" + lbl, t_global)

    # Phase 3: Symmetric FW
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['best_seq'])
    print(f"\n=== Phase 3: Symmetric FW (n={n_best}) ===")
    t_p3 = min(remaining * 0.10, 120.0)
    if t_p3 > 20:
        a_in = norm_sum1(state['best_seq'])
        a_out = fw_symmetric(a_in, t_p3, solver, verbose=False)
        try_update_state(state, a_out, "fw_sym", t_global)

    # Phase 4: Local search
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['best_seq'])
    print(f"\n=== Phase 4: Local search (n={n_best}) ===")
    t_p4 = min(remaining * 0.15, 200.0)
    if t_p4 > 20:
        a_in = norm_sum1(state['best_seq'])
        a_out = local_search(a_in, t_p4, verbose=True, rng_seed=42)
        try_update_state(state, a_out, "ls_main", t_global)

    # Phase 5: FW round 2 on updated best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['best_seq'])
    print(f"\n=== Phase 5: FW round 2 (n={n_best}) ===")
    t_p5 = min(remaining * 0.12, 150.0)
    if t_p5 > 20:
        a_in = norm_sum1(state['best_seq'])
        a_out = accelerated_fw(a_in, t_p5, solver, verbose=False)
        try_update_state(state, a_out, "fw_r2", t_global)

    # Phase 6: Local search 2
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Local search 2 ===")
    t_p6 = min(remaining * 0.08, 100.0)
    if t_p6 > 15:
        a_in = norm_sum1(state['best_seq'])
        a_out = local_search(a_in, t_p6, verbose=False, rng_seed=999)
        try_update_state(state, a_out, "ls2", t_global)

    # Phase 7: Length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['best_seq'])
    print(f"\n=== Phase 7: Length exploration ===")
    trial_lengths = []
    for delta in [500, 1000, -500, 1500, 2000, -1000]:
        new_n = n_best + delta
        if 2000 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if l != n_best]
    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(state['best_seq'], new_n)
        t7a = min(remaining * 0.3, 120.0)
        if t7a > 30:
            a_out = accelerated_fw(a_long, t7a, solver, verbose=False)
            try_update_state(state, a_out, "fw_n" + str(new_n), t_global)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t7b = min(remaining * 0.25, 60.0)
            if t7b > 15:
                a_out = local_search(norm_sum1(state['best_seq']), t7b, verbose=False, rng_seed=new_n % 10000)
                try_update_state(state, a_out, "ls_n" + str(new_n), t_global)

    # Phase 8: Final local search
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish ===")
    if remaining > 20:
        t_p8 = min(remaining * 0.6, 80.0)
        if t_p8 > 10:
            a_in = norm_sum1(state['best_seq'])
            a_out = local_search(a_in, t_p8, verbose=False, rng_seed=12345)
            try_update_state(state, a_out, "final_ls", t_global)

    if state['best_seq'] is not None:
        fs = evaluate_safe(state['best_seq'])
        if np.isfinite(fs):
            state['best_score'] = fs
        print(f"\nFINAL: score={state['best_score']:.6f}, n={len(state['best_seq'])}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3500
        state['best_seq'] = make_u_shape(n_fb, 2.0)
        state['best_score'] = evaluate_safe(state['best_seq'])
        print(f"Fallback: score={state['best_score']:.6f}")

    return [float(xi) for xi in state['best_seq']]
