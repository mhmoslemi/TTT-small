    if s < 1e-15:

    return a / s

import numpy as np

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                return s
        except Exception:

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_exact_subgradient(y, threshold=0.02):
    """
    Compute subgradient of max_k b_k(y) using ALL lags within threshold of max.
    Returns (gradient, max_value, num_active).
    """
    n = len(y)
    b = fast_conv(y)
    M = b.max()
    active = np.where(b >= M * (1.0 - threshold))[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= len(active)
    return g, M, len(active)

def compute_topk_subgradient(y, ac, top_k):
    """Compute subgradient using top-k lags. O(n*k)."""
    n = len(y)
    k_max = min(top_k, len(ac))
    if k_max == len(ac):
        top_idx = np.arange(len(ac))
    else:
        top_idx = np.argpartition(ac, -k_max)[-k_max:]
    top_idx = np.sort(top_idx)
    g = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= k_max
    return g

def nesterov_mirror_descent_v2(a0, time_limit, verbose=False, use_exact=True,
                                c0=3.0, threshold_init=0.02):
    """
    Nesterov mirror descent with exact/adaptive subgradient and backtracking.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 3000
    rng = np.random.default_rng(777 + n)
    c = c0
    threshold = threshold_init
    step_mult = 1.0
    no_decrease_count = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        if use_exact:
            g, M_cur, n_active = compute_exact_subgradient(y, threshold)
            # Adaptive threshold
            if n_active < 10:
                threshold = max(0.001, threshold * 0.9)
            elif n_active > 300:
                threshold = min(0.5, threshold * 1.15)
        else:
            ac = fast_conv(y)
            M_cur = ac.max()
            near_max = np.sum(ac >= M_cur * 0.99)
            k_use = min(max(30, int(near_max) + 10), 100)
            g = compute_topk_subgradient(y, ac, k_use)
            n_active = k_use

        eta = c * step_mult / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        M_new = fast_conv(a_new).max()

        # Backtracking: revert if significant increase
        if M_new > best_m * 1.001:
            step_mult *= 0.8
            a_new = x.copy()
            M_new = fast_conv(x).max()
            no_decrease_count += 1
        else:
            no_decrease_count = 0
            if M_new < best_m * 0.999:
                step_mult = min(step_mult * 1.03, 3.0)

        x_prev = x
        x = a_new

        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(400, int(max_stall * 0.65))
            strategy = int(rng.integers(5))
            if strategy == 0:
                pert = rng.normal(0, 0.015, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n // 5, 400), replace=False)
                x_r[idx] *= rng.uniform(0.4, 1.6, len(idx))
            elif strategy == 2:
                x_r = best_a * (1 + rng.normal(0, 0.004, n))
            elif strategy == 3:
                r_dir = rng.dirichlet(np.ones(n) * 0.5)
                x_r = 0.9 * best_a + 0.1 * r_dir
            else:
                # Shift the shape
                shift = int(rng.integers(1, max(2, n // 20)))
                x_r = best_a.copy()
                x_r[shift:] = best_a[:-shift] * 0.95
                x_r[:shift] = best_a[:shift] * 0.9
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.3, c * 0.6)
            step_mult = 1.0

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD2 t={t} ({elapsed:.1f}s): J={J:.8f}, n_act={n_active}, c={c:.3f}, sm={step_mult:.3f}")

    return best_a, best_m

def lp_remez_step(a, max_cons=50, r=0.1, solver=None):
    """Single LP Remez step with linearized constraints."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = active[np.argsort(b[active])[::-1][:max_cons]]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])

    constraints.append(a_var <= a * (1.0 + r))
    constraints.append(a_var >= a * max(0.0, 1.0 - r))

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_lp_method_v2(a0, time_limit, solver=None, r_init=0.12, verbose=False):
    """LP-based Remez with adaptive trust region, line search, and oscillation handling."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(555 + n)
    history = [best_m]

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        a_cand = lp_remez_step(a, max_cons=50, r=r, solver=solver)

        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                history.append(m_trial)
                no_improve = 0
                r = min(0.4, r * 1.15)
                break

        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.65)
            if no_improve > 15:
                # Escalation: perturb current point
                idx = rng.choice(n, size=min(n // 4, 500), replace=False)
                a[idx] *= rng.uniform(0.6, 1.4, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = min(r * 2, 0.3)
                # Also try from best
                if rng.random() < 0.3:
                    a = best_a * (1 + rng.normal(0, 0.01, n))
                    a = np.maximum(a, 1e-10)
                    a = a / a.sum()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez2 t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, no_imp={no_improve}")

    return best_a, best_m

def coordinate_descent_v2(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent with pair moves and adaptive step sizes."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 80)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue

            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = np.maximum(1.0 - (2.0 * x - 1.0) ** 4, 0.0)
    elif kind == 'sech':
        a = 1.0 / np.cosh(4.0 * (x - 0.5))
    elif kind == 'beta_like':
        a = x * (1 - x) + 0.01
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
