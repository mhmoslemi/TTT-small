import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_corr(w, p):
    """Compute cross-correlation q_m = sum_j w[m+j] * p[j] for m=0..n-1."""
    n = len(p)
    # w has length 2n-1, p has length n
    # q_m = sum_{j=0}^{n-1} w[m+j] * p[j]
    # This is cross_correlation(w, p)[m] = sum_k w[k+m] * p[k]
    L = 1 << ((len(w) + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    # cross_corr = ifft(conj(W) * P)
    result = np.fft.irfft(np.conj(W) * P, L)
    return result[:n]

def compute_gradient_p(b, p, beta):
    """Compute gradient of logsumexp_beta(b) w.r.t. p.
    
    Returns 2 * cross_corr(w, p) where w = softmax_beta(b).
    """
    # Softmax weights
    beta_b = beta * b
    beta_b_max = beta_b.max()
    exp_beta_b = np.exp(beta_b - beta_b_max)
    w = exp_beta_b / exp_beta_b.sum()
    
    # Gradient = 2 * cross_corr(w, p)
    n = len(p)
    g = 2.0 * fast_cross_corr(w, p)
    return g

def log_softmax_obj_and_grad(f, beta):
    """Compute objective and gradient for log-parameterized softmax.
    
    f: log-parameters (unconstrained)
    Returns: (objective_value, gradient_w.r.t.f)
    """
    # p = softmax(f)
    f_shift = f - f.max()
    exp_f = np.exp(f_shift)
    p = exp_f / exp_f.sum()
    n = len(p)
    
    # Autocorrelation
    b = fast_autocorr(p)
    
    # Objective: 2n * logsumexp_beta(b) / beta (this approximates 2n * max(b))
    # Actually: J_beta = 2n * (1/beta) * log(sum exp(beta * b_k))
    # = 2n * (1/beta) * (beta * max(b) + log(sum exp(beta*(b_k - max(b)))))
    # ≈ 2n * max(b) for large beta
    beta_b = beta * b
    beta_b_max = beta_b.max()
    logsumexp = beta_b_max / beta + np.log(np.exp(beta_b - beta_b_max).sum()) / beta
    
    obj = 2.0 * n * logsumexp
    
    # Gradient w.r.t. p
    g_p = compute_gradient_p(b, p, beta)
    
    # Gradient w.r.t. f: grad_f = p * (g_p - dot(g_p, p))
    grad_f = p * (g_p - np.dot(g_p, p))
    
    return obj, grad_f

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def amd_softmax_optimize(p0, time_limit, beta_init=50, beta_max=2000, max_iters=10000, verbose=False):
    """Optimize using Nesterov momentum with softmax-approximated objective."""
    n = len(p0)
    t0 = time.monotonic()
    
    # Log parameterization
    p0_safe = np.maximum(p0, 1e-15)
    p0_safe = p0_safe / p0_safe.sum()
    f = np.log(p0_safe)
    
    f_prev = f.copy()
    f_curr = f.copy()
    
    best_p = p0_safe.copy()
    best_m = fast_autocorr(p0_safe).max()
    best_obj = 2.0 * n * best_m
    
    # Adaptive learning rate
    lr = 0.1
    best_lr = lr
    lr_decay = 0.999
    lr_min = 1e-6
    
    beta = beta_init
    beta_growth = 1.001
    
    stall = 0
    max_stall = 200
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov step
        f_pred = f_curr + (it / (it + 2.0)) * (f_curr - f_prev)
        
        # Compute objective and gradient at f_pred
        obj, grad_f = log_softmax_obj_and_grad(f_pred, beta)
        
        # Gradient descent step
        f_curr = f_pred - lr * grad_f
        
        # Nesterov update
        f_prev, f_curr = f_curr, f_curr + ((it + 1) / (it + 3.0)) * (f_curr - f_pred)
        
        # Get p and evaluate
        p_curr = np.exp(f_curr - f_curr.max())
        p_curr = p_curr / p_curr.sum()
        
        M_curr = fast_autocorr(p_curr).max()
        if M_curr < best_m - 1e-12:
            best_m = M_curr
            best_p = p_curr.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive learning rate
        if stall > 20:
            lr *= lr_decay
            if lr < lr_min:
                lr = best_lr * 0.5
        if stall > max_stall:
            # Restart
            f = np.log(np.maximum(best_p, 1e-15))
            f_prev = f.copy()
            f_curr = f.copy()
            lr = best_lr
            stall = 0
            beta = max(beta_init, beta * 0.8)
        
        # Gradually increase beta
        beta = min(beta * beta_growth, beta_max)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AMD-softmax it={it} ({elapsed:.1f}s): obj={best_obj:.6f}, beta={beta:.1f}, lr={lr:.6f}")
    
    return best_p, 2.0 * n * best_m

def lbfgs_optimize(p0, time_limit, beta=100, max_iter=500, verbose=False):
    """Optimize using L-BFGS on log-parameterization."""
    t0 = time.monotonic()
    n = len(p0)
    
    p0_safe = np.maximum(p0, 1e-15)
    p0_safe = p0_safe / p0_safe.sum()
    f0 = np.log(p0_safe)
    
    best_p = p0_safe.copy()
    best_m = fast_autocorr(p0_safe).max()
    
    def objective(f):
        obj, grad = log_softmax_obj_and_grad(f, beta)
        return obj, grad
    
    try:
        result = scipy_minimize(objective, f0, method='L-BFGS-B',
                                options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8,
                                        'maxfun': max_iter * 3})
        
        f_opt = result.x
        p_opt = np.exp(f_opt - f_opt.max())
        p_opt = p_opt / p_opt.sum()
        
        M_opt = fast_autocorr(p_opt).max()
        if M_opt < best_m:
            best_m = M_opt
            best_p = p_opt.copy()
            
    except Exception as e:
        if verbose:
            print(f"    L-BFGS error: {e}")
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    L-BFGS ({elapsed:.1f}s): score={2*n*best_m:.6f}")
    
    return best_p, 2.0 * n * best_m

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init_power_law(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return norm_sum1(a)

def make_init_inverse_power(n, power=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** (-power)
    return norm_sum1(a)

def make_init_parabola(n, center=0.5, width=1.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    return norm_sum1(a)

def make_init_ramp(n, asym=1.0):
    """Asymmetric ramp: a[i] = (i/n)^asym"""
    x = np.linspace(0, 1, n)
    a = x ** asym + 0.01
    return norm_sum1(a)

def make_init_cosine(n, mode=1, offset=0.0):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5 + offset)) ** 2
    a = np.maximum(a, 0.01)
    return norm_sum1(a)

def frequency_shaping(p, time_limit, n_iters=500, verbose=False):
    """Iteratively shape the frequency response to flatten the autocorrelation."""
    n = len(p)
    a = p.copy()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2  # Power spectrum = |A|^2
        P_mean = P.mean()
        
        # Compress dynamic range: P_new = P_mean * (P/P_mean)^(1-alpha)
        alpha = 0.3 * (1 + 0.5 * it / max(1, n_iters))
        alpha = min(alpha, 0.95)
        P_new = P_mean * (P / P_mean) ** (1 - alpha)
        
        # Update: keep phase, adjust magnitude
        phase = np.angle(A)
        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase)
        
        a_candidate = np.fft.irfft(A
