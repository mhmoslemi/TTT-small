"""
Minimization of 2*n*max(conv(a,a))/(sum(a))^2 for non-negative step functions.

Strategy (different from the mirror-descent/coord-descent SOTA):
  1. L-BFGS-B quasi-Newton optimization in log-space (softmax parameterization)
     of the smooth log-sum-exp (soft-max) approximation of the autocorrelation
     peak, with an annealed temperature beta. Quasi-Newton superlinear
     convergence makes this far more effective per unit time than subgradient
     mirror descent for polishing an already good point.
  2. Light Frank-Wolfe polishing: an LP finds a feasible direction that keeps
     the cross-convolution bounded, followed by an exact bounded line search.
  3. Length sweeping / interpolation to exploit nearby sequence lengths.
  4. Perturb + re-optimize restarts to escape shallow local minima.

All helpers are top-level with no closures or lambdas (scipy receives extra
arguments through its `args` mechanism). The true evaluator value
2*n*max(conv(a,a))/sum(a)^2 is what is tracked; the soft-max value is used
only to compute gradients/Hessian information for L-BFGS.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize, minimize_scalar


# --------------------------------------------------------------------------- #
# Core numeric building blocks                                                 #
# --------------------------------------------------------------------------- #
def fast_autocorr(a):
    """Full linear autocorrelation conv(a, a) via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_nonneg(a):
    """Clip to >=0 and normalize to sum 1 (robust)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def softmax(x):
    """Numerically stable softmax."""
    x = np.asarray(x, dtype=np.float64)
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x, dtype=np.float64) / len(x)
    return e / s


def initial_theta(a):
    """Log-space parameter for a (avoids log(0))."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-15)
    return np.log(a / a.sum())


# --------------------------------------------------------------------------- #
# Smooth soft-max of the autocorrelation peak + gradient w.r.t. a              #
# --------------------------------------------------------------------------- #
def smooth_max_val(a, beta):
    c = fast_autocorr(a)
    bc = beta * c
    m = bc.max()
    return (m + np.log(np.exp(bc - m).sum())) / beta, c.max()


def grad_smooth_max_wrt_a(a, beta):
    """
    Gradient of (1/beta)*log sum_k exp(beta*c[k]) with respect to a.
    c[k] = sum_j a[j] a[k-j]; dc[k]/da[i] = 2 a[k-i] (if valid).
    => grad_a[i] = 2 * cross_correlation(a, p)[i],  p[k]=softmax(beta*c)[k].
    """
    c = fast_autocorr(a)
    bc = beta * c
    m = bc.max()
    exp_bc = np.exp(bc - m)
    p = exp_bc / exp_bc.sum()
    n = len(a)
    p_rev = p[::-1]
    L = n + len(p) - 1
    fsize = 1 << (L - 1).bit_length()
    cc = np.fft.irfft(np.fft.rfft(a, fsize) * np.fft.rfft(p_rev, fsize), fsize)
    grad_a = 2.0 * cc[len(p) - 1:len(p) - 1 + n]
    return grad_a


# --------------------------------------------------------------------------- #
# L-BFGS in theta-space (a = softmax(theta))                                   #
# --------------------------------------------------------------------------- #
def lbfgs_objective(theta, beta):
    a = softmax(theta)
    v, _ = smooth_max_val(a, beta)
    return v


def lbfgs_jacobian(theta, beta):
    a = softmax(theta)
    grad_a = grad_smooth_max_wrt_a(a, beta)
    # d(softmax)/dtheta is outer: dF/dtheta = a*(grad_a - <a,grad_a>)
    mean_g = np.dot(a, grad_a)
    return a * (grad_a - mean_g)


def true_score(a, n):
    """True objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    return 2.0 * n * fast_autocorr(a).max()


def lbfgs_optimize(a_init, time_limit, beta_schedule, verbose=True, tag=""):
    """Annealed-temperature L-BFGS-B. Returns (best_a, best_true_score)."""
    a = norm_nonneg(a_init)
    n = len(a)
    theta = initial_theta(a)
    best_a = a.copy()
    best_true = true_score(a, n)
    t0 = time.monotonic()

    for beta in beta_schedule:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break
        try:
            res = minimize(
                fun=lbfgs_objective,
                x0=theta,
                jac=lbfgs_jacobian,
                args=(beta,),
                method="L-BFGS-B",
                options={"maxiter": 400, "gtol": 1e-9,
                         "maxfun": 1200, "maxls": 40, "iprint": -1},
            )
            if res.x is not None:
                a_new = softmax(res.x)
                a_new = norm_nonneg(a_new)
                sc = true_score(a_new, n)
                if sc < best_true:
                    best_true = sc
                    best_a = a_new.copy()
                theta = res.x
                if verbose:
                    print("    [{} beta={:4.0f}] true J={:.6f}  t={:.0f}s".format(
                        tag, beta, sc, time.monotonic() - t0))
        except Exception as e:
            if verbose:
                print("    [{} beta={:4.0f}] L-BFGS error: {}".format(tag, beta, e))

    if verbose:
        print("    [{}] L-BFGS done: best true J={:.6f}  t={:.0f}s".format(
            tag, best_true, time.monotonic() - t0))
    return best_a, best_true


# --------------------------------------------------------------------------- #
# Frank-Wolfe polishing                                                        #
# --------------------------------------------------------------------------- #
def fw_alpha_fun(alpha, a, d):
    """Objective for the FW line search: peak of conv after the alpha step."""
    a_try = (1.0 - alpha) * a + alpha * d
    a_try = np.maximum(a_try, 0.0)
    s = a_try.sum()
    if s < 1e-15:
        return 1e30
    a_try = a_try / s
    return fast_autocorr(a_try).max()


def lp_fw_step(a, M, top_K, solver):
    """
    Solve max sum(b) s.t. conv(a,b)[k] <= M (for the top_K lags), b>=0,
    sum(b)=1. Returns normalized direction b or None.
    """
    n = len(a)
    c = fast_autocorr(a)
    if top_K and top_K < 2 * n - 1:
        top_idx = np.argsort(c)[::-1][:top_K]
    else:
        top_idx = np.arange(2 * n - 1)
    K = len(top_idx)
    T = np.zeros((K, n))
    for i, m in enumerate(top_idx):
        m_int = int(m)
        js = np.arange(max(0, m_int - n + 1), min(n - 1, m_int) + 1)
        T[i, js] = a[m_int - js]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M * np.ones(K), cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    solved = False
    for s in [solver] if solver is not None else [cp.ECOS]:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                solved = True
                break
        except Exception:
            continue
    if not solved:
        for s in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            try:
                prob.solve(solver=s, verbose=False)
                if prob.status in ("optimal", "optimal_inaccurate"):
                    solved = True
                    break
            except Exception:
                continue
    if not solved or b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-12:
        return None
    return b_val / b_sum


def fw_optimize(a_init, time_limit, top_K, max_iter, verbose=True, tag="",
                solver=None):
    """Frank-Wolfe with exact bounded line search. Returns (best_a, best_J)."""
    a = norm_nonneg(a_init)
    n = len(a)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(a).max()
        direction = lp_fw_step(a, M_curr, top_K, solver)
        if direction is None:
            break
        ls = minimize_scalar(fw_alpha_fun, args=(a, direction),
                             bounds=(0.0, 1.0), method="bounded",
                             options={"xatol": 1e-9})
        alpha = float(ls.x)
        a_new = (1.0 - alpha) * a + alpha * direction
        a_new = norm_nonneg(a_new)
        M_new = fast_autocorr(a_new).max()
        a = a_new
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        if verbose and (it % 5 == 0 or it == max_iter - 1):
            print("    [{} FW it={:2d}] alpha={:.4f} J={:.6f} t={:.0f}s".format(
                tag, it, alpha, 2.0 * n * M_new, time.monotonic() - t0))

    if verbose:
        print("    [{} FW done] best J={:.6f} t={:.0f}s".format(
            tag, 2.0 * n * best_M, time.monotonic() - t0))
    return best_a, 2.0 * n * best_M

def get_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                return s
        except Exception:
            continue
    return cp.ECOS


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float("inf")


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = norm_nonneg(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return norm_nonneg(f_interp)


def smooth_init(n, seed):
    """Smooth non-negative Fourier-style init."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = np.ones(n) * 0.5
    for f in [1, 2, 3, 4, 5, 6, 7]:
        a += rng.normal(0, 0.08) * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    return norm_nonneg(a)
