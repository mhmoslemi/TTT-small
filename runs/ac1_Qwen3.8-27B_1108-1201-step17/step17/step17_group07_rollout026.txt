Key insight: the gradient of c[k] w.r.t. a[j] is 2*a[k-j] for valid j,
making it a simple shift-and-scale. Power-law sequences with p>1 reduce
the central lag below 1/n, enabling J < 2.
"""

import numpy as np
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    try:
        import cvxpy as cp
        for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                x = cp.Variable(10, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    _global_solver = (cp, s)
                    return _global_solver
            except Exception:
                continue
        _global_solver = (cp, cp.SCS)
        return _global_solver
    except ImportError:
        _global_solver = None
        return None

def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size, n=fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def compute_objective(a):
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max() / (a.sum() ** 2)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_power_law(n, p, reverse=False):
    """a[i] proportional to (i+1)^p or (n-i)^p"""
    if reverse:
        x = np.arange(n - 1, -1, -1, dtype=np.float64) + 1.0
    else:
        x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_exp_law(n, t, reverse=False):
    x = np.linspace(0.0, 1.0, n)
    if reverse:
        x = 1.0 - x
    a = np.exp(t * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_poly(n, coeffs):
    x = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for c, p in coeffs:
        a += c * x ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def compute_subgradient(a, k, n):
    """Subgradient of max(c) at the given set of active lags k. Returns gradient vector."""
    g = np.zeros(n)
    for ki in k:
        ki = int(ki)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * a[ki - idx]
    g /= len(k)
    return g

def nesterov_mirror_descent(a0, time_limit, top_k=30, c0=3.0, verbose=False):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 2000
    rng = np.random.default_rng(42 + n)
    c = c0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        ac = fast_conv(y)
        M = ac.max()
        
        # Use all lags that are >= 99% of max
        threshold = M * 0.99
        active = np.where(ac >= threshold)[0]
        if len(active) > top_k:
            active = np.argsort(ac)[::-1][:top_k]
        
        g = compute_subgradient(y, active, n)

        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-18:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(300, int(max_stall * 0.7))
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.015, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                idx = rng.choice(n, size=min(n // 3, 500), replace=False)
                x_r[idx] *= rng.uniform(0.6, 1.4, len(idx))
            elif strategy == 2:
                x_r = best_a * (1 + rng.normal(0, 0.005, n))
            else:
                # Skew perturbation
                x_r = best_a.copy()
                skew = rng.uniform(-0.1, 0.1)
                pos = np.linspace(0, 1, n)
                x_r *= (1 + skew * (2 * pos - 1))
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.3, c * 0.65)

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall_count}, c={c:.3f}")

    return best_a, best_m

def lp_remez_step(a, active_lags, r, solver_info):
    """Single LP Remez step with given active lags and trust region radius r."""
    if solver_info is None:
        return a.copy()
    cp, solver = solver_info
    n = len(a)
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1]

    for ki in active_lags:
        ki = int(ki)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[ki - idx]
            # Linearized constraint: sum coeffs * a_var[idx] <= M_var
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) <= M_var)

    # Trust region
    constraints.append(a_var <= a * (1.0 + r))

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return a.copy()

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_lp_method(a0, time_limit, solver_info=None, r_init=0.1, verbose=False):
    """LP-based Remez with adaptive trust region."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(777)

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(a)
        M = c.max()
        
        # Find active set: all lags within 99.95% of max
        threshold = M * 0.9995
        active = np.where(c >= threshold)[0]
        if len(active) > 80:
            active = np.argsort(c)[::-1][:80]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        a_cand = lp_remez_step(a, active, r, solver_info)

        # Line search
        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-20:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.5, r * 1.05)
                break

        if not accepted:
            no_improve += 1
            r = max(0.002, r * 0.6)
            if no_improve > 15:
                idx = rng.choice(n, size=min(n // 4, 300), replace=False)
                a[idx] *= rng.uniform(0.7, 1.3, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                r = r_init * 0.5

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, |active|={len(active)}")

    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with random pair moves and bisection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 80)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue

            for frac in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15, 0.3]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            # Periodic perturbation to escape local minima
            if stall % 5 == 0:
                idx = rng.choice(n, size=min(n // 10, 200), replace=False)
                a[idx] *= rng.uniform(0.8, 1.2, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                m = fast_conv(a).max()
                if m < best_m:
                    best_m = m
                    best_a = a.copy()

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            return

        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()

    # Load parent

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.8f}")
