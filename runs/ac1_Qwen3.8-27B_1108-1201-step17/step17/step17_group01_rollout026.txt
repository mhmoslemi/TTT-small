def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fft_subgradient(a, ac, temperature=0.01):
    """
    Compute subgradient of max_k (a*a)_k using log-sum-exp smoothing.
    Key insight: d/d a[m] of (a*a)_k = 2*a[k-m] if 0<=k-m<n, else 0.
    So grad[m] = 2 * sum_k w[k] * a[k-m] * I(0<=k-m<n)
               = 2 * cross_correlation(w, a)[m]
    This can be computed in O(n log n) via FFT.
    """
    n = len(a)
    ac_max = ac.max()
    if ac_max < 1e-30:
        return np.zeros(n)
    
    shifted = (ac - ac_max) / temperature
    shifted = np.maximum(shifted, -60.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        top_k = min(20, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        w = np.zeros(len(ac))
        w[top_idx] = 1.0 / top_k
    else:
        w = w / w_sum
    
    # Compute grad[m] = 2 * sum_k w[k] * a[k-m] * I(0<=k-m<n)
    # Let j = k-m, so k = m+j, j in [0, n-1]:
    # grad[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    # This is the cross-correlation of w (length 2n-1) with a (length n) at lags 0..n-1.
    
    N = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    # Cross-correlation: c[m] = sum_j w[m+j] * a[j]
    # c = ifft(W * conj(A))
    cross = np.fft.irfft(W * np.conj(A), N)[:n]
    grad = 2.0 * cross
    return grad

def fft_subgradient_topk(a, ac, top_k=100, beta=3.0):
    """
    Alternative: use only top_k peaks with power weighting, but via FFT.
    Faster when top_k << 2n-1.
    """
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w_top = top_vals ** beta
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        w_top = np.ones(k_max) / k_max
    else:
        w_top = w_top / w_sum
    
    # Build sparse weight vector and use FFT
    w = np.zeros(2 * n - 1)
    for wi, k in zip(w_top, top_idx):
        w[int(k)] = wi
    
    N = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(W * np.conj(A), N)[:n]
    grad = 2.0 * cross
    return grad

def amd_optimize_v2(a0, time_limit, verbose=False, seed=42):
    """
    Improved Nesterov-accelerated mirror descent with:
    - Adaptive temperature (starts hot, cools down)
    - FFT-based subgradient
    - Spectral restarts (add random sinusoid)
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    
    # Adaptive parameters
    gamma = 5.0
    restart_count = 0
    restart_every = 3000
    cur_temp = 0.05  # start hot
    stall_count = 0
    max_stall_before_restart = 5000
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute autocorrelation and subgradient
        ac = fast_autocorr(y)
        
        # Adaptive temperature: start hot, cool down
        if t < 10000:
            cur_temp = 0.1 * np.exp(-t / 5000.0) + 0.005
        else:
            cur_temp = max(0.003, 0.005 * np.exp(-(t - 10000) / 20000.0))
        
        g = fft_subgradient(y, ac, temperature=cur_temp)
        
        # Step size
        eta = gamma / (np.sqrt(t) + 1.0)
        
        # Mirror (exponential) update
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart strategy
        if stall_count > max_stall_before_restart:
            stall_count = 0
            restart_count += 1
            # Spectral perturbation: add random low-freq sinusoid
            n_pert = rng.integers(2, 8)
            perm = best_a.copy()
            freq = rng.uniform(1, 15)
            phase = rng.uniform(0, 2 * np.pi)
            x_axis = np.arange(n)
            pert = np.cos(2 * np.pi * freq * x_axis / n + phase)
            pert = (pert - pert.min()) / (pert.max() - pert.min() + 1e-10)
            strength = min(0.15, 0.03 + 0.02 * restart_count)
            proba = best_a + strength * pert * best_a.std()
            proba = np.maximum(proba, 1e-10)
            proba = proba / proba.sum()
            x = proba
            x_prev = proba.copy()
            gamma = 5.0 * (1.0 + 0.1 * min(restart_count, 10))
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD2 restart {restart_count} (t={elapsed:.0f}s): J={J:.6f}, temp={cur_temp:.4f}")
    
    return best_a, best_m

def spectral_shaping(a0, time_limit, alpha=0.7, n_iters=500, verbose=False):
    """
    Low-pass spectral shaping: concentrate the power spectrum
    to make the autocorrelation broader (lower peak).
    
    Strategy: multiply the spectrum by a low-pass filter, then
    re-impose non-negativity.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.rfft(a, fft_size)
        freqs = np.fft.rfftfreq(fft_size)
        
        # Low-pass filter: emphasize low frequencies
        # Use a smooth roll-off
        cutoff = 5.0 / n  # in cycles per sample
        lp = 1.0 / (1.0 + (np.abs(freqs) / cutoff) ** 4)
        
        # Applied shaping
        A_shaped = A * (lp ** (alpha * 0.1))
        
        a_new = np.fft.irfft(A_shaped, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend
            blend = 0.5
            a = (1 - blend) * a + blend * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=42):
    """
    Coordinate descent: for each position, find the optimal value
    that minimizes the peak autocorrelation.
    Exploits the bilinear structure: changing a[i] by delta changes
    (a*a)[k] by 2*delta*a[k-i] (for valid k-i).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 10
    
    # Precompute: for each i, the affected autocorrelation indices
    # (a*a)[k] is affected by a[i] when 0 <= k-i < n, i.e., k in [i, i+n-1]
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        stall += 1
        any_improve = False
        ac = fast_autocorr(a)
        
        # Process coordinates in random order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # For coordinate i, (a*a)[k] changes by 2*delta*a[k-i] for k in [i, min(i+n-1, 2n-2)]
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            
            # Current max
            cur_max = ac.max()
            cur_max_idx = ac.argmax()
            
            # Try different values
            x0 = a[i]
            if x0 < 1e-15:
                # Try small positive values
                trials = [1e-6, 1e-5, 1e-4, x0 + 1e-5]
            else:
                trials = [0.0, x0 * 0.3, x0 * 0.6, x0 * 1.4, x0 * 2.0]
            
            best_trial = x0
            best_trial_max = cur_max
            
            for trial in trials:
                if abs(trial - x0) < 1e-15:
                    continue
                # Compute change in autocorrelation
                delta = trial - x0
                if delta == 0:
                    continue
                
                # Change in (a*a)[k] for k in [k_start, k_end-1]:
                # d(a*a)[k] = 2 * delta * a[k-i] (linear approximation)
                # Plus quadratic term: delta^2 if k-i == i (i.e., k == 2i)
                k_idx = np.arange(k_start, k_end)
                j_idx = k_idx - i  # = 0, 1, ..., n-1 (clipped)
                j_idx = np.maximum(j_idx, 0)
                j_idx = np.minimum(j_idx, n - 1)
                
                # Linear change
                change = 2.0 * delta * a[j_idx]
                # Quadratic correction for k = 2i
                if 2 * i < 2 * n - 1:
                    q_idx = 2 * i - k_start
                    if 0 <= q_idx < len(change):
                        change[q_idx] += delta * delta
                
                new_ac_approx = ac.copy()
                new_ac_approx[k_start:k_end] += change
                
                # Also need to check: if delta is negative and a[i]+delta < 0, skip
                if trial < -1e-15:
                    continue
                
                max_val = new_ac_approx.max()
                if max_val < best_trial_max - 1e-14:
                    best_trial_max = max_val
                    best_trial = trial
            
            if best_trial != x0:
                a[i] = max(0, best_trial)
                any_improve = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = a.copy()
            stall = 0
        
        if verbose and stall % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CD polish ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def make_cosine_init(n, n_harmonics=5, seed=42):
    """Create initialization as sum of cosines (spectrally concentrated)."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for j in range(1, n_harmonics + 1):
        freq = rng.uniform(0.5, 3.0)
        amp = rng.uniform(0.1, 0.4) / j
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * x / n + phase)
    a = np.maximum(a, 0.01)
    return norm_sum1(a)

def make_raised_cosine(n, order=1):
    """a_i = 1 + cos(2*pi*order*i/n) = 2*cos^2(pi*order*i/n)"""
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + np.cos(2 * np.pi * order * x / n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def make_smooth_bump(n, center_frac=0.5, width=0.3):
    """Gaussian bump."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((x - center_frac) ** 2) / (2 * width ** 2))
    return norm_sum1(a)

def make_tapered(n, power=2.0):
    """Tapered ends: a_i proportional to (i+1)^(power) * (n-i)^(power) / n^(2*power)"""
    x = np.arange(1, n + 1, dtype=np.float64)
    y = np.arange(n, 0, -1, dtype=np.float64)
    a = (x ** power) * (y ** power)
    a = a / a.max()
    return norm_sum1(a)

def multi_phase_optimize(a0, time_limit, verbose=False):
    """
    Multi-phase optimization:
    Phase A: Spectral shaping (concentrate spectrum)
    Phase B: AMD with FFT subgradient
    Phase C: Coordinate descent polish
    Phase D: AMD again with lower temperature
    Phase E: Final coordinate descent
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    def try_update(a, label):
        nonlocal best_a, best_m
        if a is None:
            return
        a = norm_sum1(a)
        m = fast_autocorr(a).max()
        if m < best_m - 1e-14:
            best_m = m
            best_a = a.copy()
            if verbose:
                J = 2.0 * n * best_m
                print(f"      [{label}] J={J:.6f}")
    
    # Phase A: Spectral shaping
    elapsed = time.monotonic() - t0
    t_a = min((time_limit - elapsed) * 0.15, 60.0)
    if t_a > 5:
        if verbose:
            print(f"    Phase A: Spectral shaping ({t_a:.0f}s)")
        a_out, _ = spectral_shaping(a, t_a, alpha=0.7, n_iters=200, verbose=False)
        try_update(a_out, "spectral")
    
    # Phase B: AMD
    elapsed = time.monotonic() - t0
    t_b = min((time_limit - elapsed) * 0.45, 400.0)
    if t_b > 10:
        if verbose:
            print(f"    Phase B: AMD ({t_b:.0f}s)")
        a_in = best_a if fast_autocorr(best_a).max() < fast_autocorr(a).max() else a
        a_out, _ = amd_optimize_v2(a_in, t_b, verbose=False, seed=42)
        try_update(a_out, "amd")
    
    # Phase C: Coordinate descent
    elapsed = time.monotonic() - t0
    t_c = min((time_limit - elapsed) * 0.15, 80.0)
    if t_c > 5:
        if verbose:
            print(f"    Phase C: CD polish ({t_c:.0f}s)")
        a_out, _ = coordinate_descent_polish(best_a, t_c, verbose=False, rng_seed=42)
        try_update(a_out, "cd")
    
    # Phase D: AMD with different seed
    elapsed = time.monotonic() - t0
    t_d = min((time_limit - elapsed) * 0.20, 150.0)
    if t_d > 10:
        if verbose:
            print(f"    Phase D: AMD2 ({t_d:.0f}s)")
        a_out, _ = amd_optimize_v2(best_a, t_d, verbose=False, seed=123)
        try_update(a_out, "amd2")
    
    # Phase E: Final CD
    elapsed = time.monotonic() - t0
    t_e = min((time_limit - elapsed) * 0.5, 60.0)
    if t_e > 5:
        if verbose:
            print(f"    Phase E: CD final ({t_e:.0f}s)")
        a_out, _ = coordinate_descent_polish(best_a, t_e, verbose=False, rng_seed=777)
        try_update(a_out, "cd_final")
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator.
    Strategy:
    1. Load parent sequence if available.
    2. Try multiple initializations at several lengths.
    3. For each, run multi-phase optimization.
    4. Track global best and return.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update_seq(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            parent_norm = norm_sum1(parent)
            try_update_seq(parent_norm, "parent")
    except Exception as e:
        print(f"No parent available: {e}")
    
    n_parent = len(parent) if parent is not None else 4200
    print(f"Target: 1.5027, current best: {best_score:.6f}, n={n_parent}")
    
    # Determine lengths to try
    # Focus on lengths near the parent, plus a few others
    lengths = set()
    lengths.add(n_parent)
    lengths.add(n_parent + 200)
    lengths.add(n_parent + 500)
    lengths.add(n_parent - 200)
    if n_parent > 3000:
        lengths.add(3000)
    if n_parent > 5000:
        lengths.add(5000)
    
    lengths = sorted([l for l in lengths if 1000 <= l <= 6000])
    print(f"Lengths to try: {lengths}")
    
    # Phase 1: Multi-init optimization at parent length
    n_base = n_parent
    remaining = time_budget - (time.monotonic() - t_global)
    
    print(f"\n=== Phase 1: Multi-init at n={n_base} ===")
    t_phase1 = min(remaining * 0.40, 600.0)
    
    if t_phase1 > 60:
        # Generate initializations
        inits = []
        labels = []
        
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        
        # Cosine-based inits (spectrally concentrated)
        for h in [3, 5, 8]:
            for sd in [42, 123, 456]:
                inits.append(make_cosine_init(n_base, n_harmonics=h, seed=sd))
                labels.append(f"cos{h}_{sd}")
        
        # Raised cosine
        for order in [1, 2, 3]:
            inits.append(make_raised_cosine(n_base, order=order))
            labels.append(f"rased_cos{order}")
        
        # Tapered
        for pw in [1.0, 2.0, 3.0, 5.0]:
            inits.append(make_tapered(n_base, power=pw))
            labels.append(f"taper{pw}")
        
        # Smooth bumps
        for wc in [0.15, 0.25, 0.4]:
            inits.append(make_smooth_bump(n_base, width=wc))
            labels.append(f"bump_{wc}")
        
        # Deduplicate (check similarity)
        seen_sigs = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            # Use a coarse fingerprint
            step = max(1, n_base // 100)
            sig = tuple(np.round(init[::step], 4))
            if sig not in seen_sigs:
                seen_sigs.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        
        inits = unique_inits
        labels = unique_labels
        print(f"  {len(inits)} unique inits")
        
        # Allocate time per init
        n_inits = len(inits)
        t_per = t_phase1 / n_inits
        t_per = max(8.0, min(t_per, 60.0))
        
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if time_budget - elapsed_so_far < 60:
                break
            t_avail = min(t_per, time_budget - 60 - elapsed_so_far)
            if t_avail < 8:
                break
            
            # Quick eval of init
            init_score = evaluate_safe(init)
            # Skip if init is already worse than best by a lot
            if init_score > best_score * 1.5 and idx > 0:
                continue
            
            print(f"  [{lbl}] t_avail={t_avail:.0f}s, init_score={init_score:.4f}")
            
            a_out, _ = multi_phase_optimize(init, t_avail, verbose=False)
            try_update_seq(a_out, f"p1_{lbl}")
    
    # Phase 2: Refine best at parent length with more time
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Refine best (n={len(best_seq)}) ===")
    t_phase2 = min(remaining * 0.25, 350.0)
    
    if t_phase2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = multi_phase_optimize(a_in, t_phase2, verbose=True)
        try_update_seq(a_out, "phase2_refine")
        
        # Additional AMD with different parameters
        remaining = time_budget - (time.monotonic() - t_global)
        t2b = min(remaining * 0.15, 150.0)
        if t2b > 15:
            a_out, _ = amd_optimize_v2(norm_sum1(best_seq), t2b, verbose=True, seed=999)
            try_update_seq(a_out, "phase2_amd2")
    
    # Phase 3: Try other lengths
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Other lengths ===")
    
    n_best = len(best_seq)
    other_lengths = [l for l in lengths if l != n_best]
    other_lengths.sort(key=lambda l: abs(l - n_best))
    
    for new_n in other_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        try_update_seq(a_long, f"interp_n{new_n}")
        
        t_avail = min(remaining * 0.4, 200.0)
        if t_avail > 15:
            a_out, _ = multi_phase_optimize(a_long, t_avail, verbose=False)
            try_update_seq(a_out, f"opt_n{new_n}")
    
    # Phase 4: Heavy coordinate descent on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Heavy CD (n={n_best}) ===")
    t_phase4 = min(remaining * 0.20, 120.0)
    if t_phase4 > 10:
        a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t_phase4, verbose=True, rng_seed=2024)
        try_update_seq(a_out, "heavy_cd")
    
    # Phase 5: Final AMD push
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final AMD (n={n_best}) ===")
    t_phase5 = min(remaining * 0.50, 200.0)
    if t_phase5 > 10:
        a_out, _ = amd_optimize_v2(norm_sum1(best_seq), t_phase5, verbose=True, seed=31415)
        try_update_seq(a_out, "final_amd")
    
    # Phase 6: Final CD
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        print(f"\n=== Phase 6: Final CD ===")
        t_phase6 = min(remaining * 0.7, 60.0)
        if t_phase6 > 5:
            a_out, _ = coordinate_descent_polish(norm_sum1(best_seq), t_phase6, verbose=True, rng_seed=27182)
            try_update_seq(a_out, "final_cd")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3000
        best_seq = np.sin(np.pi * np.linspace(0, 1, n_fb)) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
