import numpy as np

from scipy import sparse
from scipy.optimize import linprog


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_T_rows(a, indices):
    """Build sparse rows of the convolution matrix for given lag indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(indices):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            for i in range(max(0, 0), min(n, max(1, k + 1))):
                pass
            continue
        for i in range(i_start, i_end + 1):
            rows.append(row_idx)
            cols.append(i)
            vals.append(a[k - i])
    T = sparse.csc_matrix((vals, (rows, cols)), shape=(len(indices), n))
    return T


def solve_lp_direction(y, top_k=2000, lp_timeout=15.0):
    """Solve max sum(b) s.t. conv(y, b)[k] <= M for top-k lags, b >= 0."""
    n = len(y)
    ac = fast_autocorr(y)
    M = ac.max()
    if M < 1e-16:
        return None

    n_total = 2 * n - 1
    if top_k is not None and top_k < n_total:
        active_idx = np.argsort(ac)[::-1][:top_k]
    else:
        active_idx = np.arange(n_total)

    T = build_T_rows(y, active_idx)

    c_obj = -np.ones(n)
    A_ub = sparse.vstack([T, -sparse.eye(n, format='csr')], format='csc')
    b_ub = np.concatenate([np.full(len(active_idx), M), np.zeros(n)])

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs',
                      options={'presolve': True, 'time_limit': lp_timeout})
        if res.success:
            b = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
            s = b.sum()
            if s > 1e-15:
                return b / s
    except Exception:
        pass

    return None


def lp_fw_optimize(a0, time_limit, top_k=2000, verbose=True):

    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()

    x = a.copy()
    it = 0

        it += 1

            break

        b_dir = solve_lp_direction(y, top_k=top_k)

        if b_dir is None:

            log_y = np.log(np.maximum(y, 1e-30))
            log_new = log_y - eta * g
            log_new -= log_new.max()
            b_dir = np.exp(log_new)
            s = b_dir.sum()

                b_dir = y.copy()
            else:
                b_dir = b_dir / s

        # Line search
        base_obj = fast_autocorr(y).max()
        best_alpha = 2.0 / (it + 2.0)
        best_trial_obj = base_obj
        alphas = [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02]
        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * b_dir
            trial = np.maximum(trial, 0.0)
            s_t = trial.sum()
            if s_t < 1e-15:
                continue
            trial = trial / s_t
            obj = fast_autocorr(trial).max()
            if obj < best_trial_obj - 1e-18:
                best_trial_obj = obj
                best_alpha = alpha

        alpha = best_alpha
        y_new = (1.0 - alpha) * y + alpha * b_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()

            y_new = y.copy()

            y_new = y_new / sy

            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW it={it} ({elapsed:.0f}s): J={J:.6f}, alpha={alpha:.4f}")

    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=80, beta=4.0, c0=3.5,

        log_y = np.log(np.maximum(y, 1e-30))

            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m

def freq_perturb(a, strength=0.02, rng_seed=99):
    """Perturb phase in frequency domain to diversify."""
    rng = np.random.default_rng(rng_seed)
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_parametric(n, family='sine'):
    """Generate a parametric initial sequence."""
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x / 2) ** 2)
    elif family == 'beta':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x * (1 - x))
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)

    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP top-4000 (n={n_best}) ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 40 and n_best <= 5000:
        a_out, _ = lp_fw_optimize(norm_sum1(best_seq), t7, top_k=4000, verbose=True)
        best_seq, best_score = update_best(a_out, best_score, t_global, "lp_final")

    # === Phase 8: Final peak-crush ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final peak-crush (n={n_best}) ===")
    if remaining > 20:
        t8 = min(remaining * 0.35, 150.0)
        if t8 > 15:
            a_out, _ = peak_crush(norm_sum1(best_seq), t8, verbose=True, rng_seed=777)
            best_seq, best_score = update_best(a_out, best_score, t_global, "pc_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")

        # One last quick peak-crush with remaining time
        rem = float(budget_s) - (time.monotonic() - t_global)
        if rem > 30:
            print(f"  Extra polish time: {rem:.0f}s")
            a_out, _ = peak_crush(norm_sum1(best_seq), max(rem - 15, 10.0),
                                  verbose=False, rng_seed=1234)
            best_seq, best_score = update_best(a_out, best_score, t_global, "polish")
            fs2 = evaluate_safe(best_seq)
            if np.isfinite(fs2):
                best_score = fs2
            print(f"  After polish: score={best_score:.6f}")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
