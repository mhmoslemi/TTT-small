Key: exact subgradient via all near-max lags, FFT-based convolution,
   aggressive Remez with 60 constraints, and multi-n search.
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_exact_subgrad(y, ac, threshold_frac=0.995):
    """Compute exact subgradient using all lags where ac[k] >= threshold_frac * max."""
    n = len(y)
    M = ac.max()
    if M < 1e-15:
        return np.zeros(n)
    threshold = M * threshold_frac
    active = np.where(ac >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(ac))])
    
    g = np.zeros(n)
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * y[k_int - idx]
    g /= len(active)
    return g

def nesterov_mirror_descent(a0, time_limit, top_k=0, c0=3.0, threshold_frac=0.995, verbose=False):
    """Nesterov mirror descent. If top_k=0, use exact subgradient (all near-max lags)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    max_stall = 5000
    rng = np.random.default_rng(12345 + n)
    c = c0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        ac = fast_conv(y)
        M = ac.max()
        
        if top_k > 0 and top_k < n:
            # Top-k subgradient
            k_max = min(top_k, len(ac))
            top_idx = np.argpartition(ac, -k_max)[-k_max:]
            g = np.zeros(n)
            for k in top_idx:
                k_int = int(k)
                j_start = max(0, k_int - n + 1)
                j_end = min(n - 1, k_int)
                if j_start <= j_end:
                    idx = np.arange(j_start, j_end + 1)
                    g[idx] += 2.0 * y[k_int - idx]
            g /= k_max
        else:
            # Exact subgradient
            g = compute_exact_subgrad(y, ac, threshold_frac)
        
        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-20:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            stall_count = 0
            max_stall = max(500, int(max_stall * 0.65))
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.015, n)
                x_r = best_a * (1 + pert)
            elif strategy == 1:
                x_r = best_a.copy()
                n_pert = min(n // 5, 500)
                idx = rng.choice(n, size=n_pert, replace=False)
                x_r[idx] *= rng.uniform(0.4, 1.6, len(idx))
            elif strategy == 2:
                x_r = best_a * (1 + rng.normal(0, 0.005, n))
            else:
                # Move mass from high-gradient to low-gradient regions
                grad = compute_exact_subgrad(best_a, fast_conv(best_a), 0.99)
                x_r = best_a.copy()
                top_half = np.argsort(grad)[:n//4]
                bot_half = np.argsort(grad)[-n//4:]
                amt = x_r[top_half].sum() * 0.05
                x_r[top_half] -= amt / len(top_half)
                x_r[bot_half] += amt / len(bot_half)
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r
            c = max(0.3, c * 0.55)
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    NMD t={t} ({elapsed:.1f}s): J={J:.8f}, stall={stall_count}, c={c:.3f}")
    
    return best_a, best_m

def lp_remez_single(a, max_cons=60, r=0.1, solver=None, threshold_frac=0.999):
    """Single LP Remez step with box trust region."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    threshold = max_b * threshold_frac
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = active[np.argsort(b[active])[::-1][:max_cons]]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    # Box trust region: a*(1-r) <= a_var <= a*(1+r)
    lower = np.maximum(a * (1.0 - r), 0.0)
    constraints.append(a_var >= lower)
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return a.copy()
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy()
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def remez_lp_method(a0, time_limit, solver=None, r_init=0.08, verbose=False, max_cons=60):
    """LP-based Remez with adaptive trust region and aggressive line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    r = r_init
    no_improve = 0
    rng = np.random.default_rng(777 + n)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_cand = lp_remez_single(a, max_cons=max_cons, r=r, solver=solver, threshold_frac=0.999)
        
        # Aggressive line search
        accepted = False
        for t_frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_trial = (1 - t_frac) * a + t_frac * a_cand
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-20:
                a = a_trial
                best_m = m_trial
                best_a = a.copy()
                accepted = True
                no_improve = 0
                r = min(0.35, r * 1.08)
                break
        
        if not accepted:
            no_improve += 1
            r = max(0.003, r * 0.65)
            if no_improve > 15:
                # Perturb to escape local optimum
                idx = rng.choice(n, size=min(n//4, 600), replace=False)
                a[idx] *= rng.uniform(0.5, 1.8, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Remez t={t} ({elapsed:.1f}s): J={J:.8f}, r={r:.4f}, no_imp={no_improve}")
    
    return best_a, best_m

def coordinate_descent_polish(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent with adaptive pair moves and bisection."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = max(30, n // 80)
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        improved = False
        n_pairs = min(200, n // 2)
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-14:
                continue
            
            # Try multiple fractions
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-18:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='raised_cos'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-6.0 * (x - 0.5) ** 2 / 0.08)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'u_shape':
        a = 0.3 + 0.7 * (2.0 * x - 1.0) ** 2
    elif kind == 'quartic':
        a = 1.0 - (2.0 * x - 1.0) ** 4
    elif kind == 'sine_hump':
        a = np.sin(np.pi * x) ** 1.5
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def heavy_average_nmd(a0, time_limit, c0=2.0, tau=100):
    """Nesterov mirror descent with heavy averaging for better convergence."""
    n = len(a0)
    a = norm_sum1(a0)
    x = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    # Store iterates for averaging
    store_interval = max(1, tau)
    stored = []
    c = c0
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        ac = fast_conv(y)
        g = compute_exact_subgrad(y, ac, 0.995)
        
        eta = c / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_conv(x).max()
        if M_new < best_m - 1e-20:
            best_m = M_new
            best_a = x.copy()
        
        # Heavy averaging: every tau steps, compute the average and check
        if t % store_interval == 0:
            ba = best_a / (best_a.sum() + 1e-30)
            # Occasionally try averaging recent iterates
            if len(stored) >= tau:
                avg_a = np.mean(stored[-tau:], axis=0)
                avg_a = np.maximum(avg_a, 0)
                avg_a = avg_a / (avg_a.sum() + 1e-30)
                m_avg = fast_conv(avg_a).max()
                if m_avg < best_m:
                    best_m = m_avg
                    best_a = avg_a.copy()
        
        stored.append(x.copy())
        if len(stored) > 3 * tau:
            stored = stored[-2 * tau:]
        
        # Adaptive restart
        if t % 5000 == 0:
            c = max(0.2, c * 0.7)
    
    return best_a, best_m

def propose_candidate(budget_s=1800.0):

    t_global = time.monotonic()

    best_seq = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        pass
    
    if parent is None:
        parent = make_init(2400, 'raised_cos')
        best_seq = parent.copy()

        print(f"Default init: n={len(parent)}, score={best_score:.8f}")
    
    # === Phase 1: NMD on parent at parent's length (30% time) ===
    n_parent = len(parent)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: NMD n={n_parent} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_mirror_descent(parent, t1, top_k=0, c0=4.0,

    # === Phase 2: Remez on best (20% time) ===

    remaining = time_budget - (time.monotonic() - t_global)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: Final push n={n_best} (budget={remaining:.0f}s) ===")
        
        t7a = min(remaining * 0.4, 120.0)
        if t7a > 15:
            a_out, _ = remez_lp_method(norm_sum1(best_seq), t7a, solver=solver,
                                         r_init=0.04, verbose=False, max_cons=70)
            try_update(a_out, f"Remez_final_n{n_best}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t7b = min(remaining * 0.4, 100.0)

        return [1.0]
    
    return [float(x) for x in best_seq]
