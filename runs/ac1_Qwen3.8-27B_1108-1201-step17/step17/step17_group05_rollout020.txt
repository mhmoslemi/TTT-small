def _fast_autocorr(a):
    """Compute full self-convolution using FFT. Length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _fast_conv(a, b):
    """Compute full linear convolution using FFT."""
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = np.fft.rfft(np.asarray(b, dtype=np.float64), fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def _norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _evaluate_fast(a):
    """Fast evaluation without calling evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _fast_autocorr(a)
    M = c.max()
    return float(2.0 * n * M / (s * s))


def _make_smooth_init(n, seed=0, freqs=None):
    """Create a smooth non-negative initialization via Fourier series."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    if freqs is None:
        freqs = [1, 2, 3, 4, 5, 6, 7]
    for f in freqs:
        coeff = rng.normal(0, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return _norm_sum1(a)


def _make_symmetric_init(n, seed=0):
    """Create a symmetric smooth initialization."""
    a = _make_smooth_init(n, seed=seed)
    a_sym = 0.5 * (a + a[::-1])

def _peeling_optimize(a0, time_limit, verbose=True, max_peels=5000):
    """
    Aggressive peeling-based coordinate descent.
    At each step: identify peak-contributing pairs, move mass to quiet positions.
    Uses line search for optimal transfer amount.
    """
    n = len(a0)
    a = _norm_sum1(a0).copy()
    best_a = a.copy()
    c = _fast_autocorr(a)
    best_M = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    peels = 0
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall and peels < max_peels:
        c = _fast_autocorr(a)
        M_curr = c.max()

        # Find top peak lags
        n_top = min(20, len(c))
        top_lags = np.argsort(c)[::-1][:n_top]

        # Compute per-position sensitivity (how much each position contributes to peaks)
        sens = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            sens[idx] += contrib

        # Find best source (high contribution to peak)
        if sens.max() < 1e-20:
            stall += 1
            continue

        src = int(np.argmax(sens))
        src_val = a[src]
        if src_val < 1e-12:
            stall += 1
            continue

        # Find best destination: low sensitivity, far from source
        dist = np.abs(np.arange(n) - src).astype(float)
        dist[src] = 1e10
        score = sens / (1.0 + dist / n)
        # Avoid moving to positions too close
        mask = dist > n // 10
        if mask.sum() > 0:
            candidates = np.where(mask)[0]
            best_dst = candidates[np.argmin(score[candidates])]
        else:
            candidates = np.arange(n)
            candidates = candidates[candidates != src]
            best_dst = candidates[np.argmin(score[candidates])]

        # Line search: try multiple fractions
        improved = False
        for frac in [0.05, 0.03, 0.02, 0.08, 0.01, 0.1, 0.015, 0.04]:
            amt = src_val * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[best_dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            M_trial = _fast_autocorr(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            peels += 1
            stall = 0
        else:
            stall += 1
            # Random perturbation every 10 stalls
            if stall % 10 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.01, 0.05)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = _fast_autocorr(a).max()
                        if M_new < best_M:
                            best_M = M_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and peels % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Peel ({elapsed:.0f}s): J={J:.6f}, peels={peels}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    Peel final ({elapsed:.0f}s): J={J:.6f}, peels={peels}")
    return best_a, best_M


def _lp_frank_wolfe(a0, time_limit, solver, verbose=True, init_K=100, max_K=None):

    LP Frank-Wolfe with adaptive constraint management and line search.
    Starts with top-K constraints, adds violated ones each iteration.
    Uses line search for the FW step.
    """

    a = _norm_sum1(a0).copy()
    best_a = a.copy()
    c = _fast_autocorr(a)
    best_M = c.max()
    t0 = time.monotonic()

        max_K = 2 * n - 1

    y = a.copy()
    x = a.copy()
    active_constraints = None  # Will be set on first iteration
    iters = 0

    for t in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break

        iters += 1
        cy = _fast_autocorr(y)
        My = cy.max()
        if My < 1e-16:
            break

        # Determine constraint set
        if active_constraints is None or Iters % 20 == 1:

            K = min(init_K + iters // 10, max_K)
            K = min(K, 2 * n - 1)
            top_idx = np.argsort(cy)[::-1][:K]
            active_constraints = top_idx
        else:
            # Add violated constraints from previous LP solution
            # (handled below after LP solve)

        K = len(active_constraints)

        # Build constraint matrix T: T[m, j] = y[m-j] if valid
        T = np.zeros((K, n))
        for col_idx, m in enumerate(active_constraints):
            mi = int(m)
            js = max(0, mi - n + 1)
            je = min(n - 1, mi)
            js_arr = np.arange(js, je + 1)
            T[col_idx, js_arr] = y[mi - js_arr]

        # Solve LP: maximize sum(b) subject to T @ b <= My, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= My * np.ones(K)]
        objective = cp.Maximize(cp.sum(b_var))
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            b_val = np.ones(n) / n
            b_sum = 1.0
        g_dir = b_val / b_sum

        # Check for violated constraints and add them
        cg = _fast_conv(y, g_dir)
        violated = np.where(cg > My * (1 + 1e-6))[0]
        violated = violated[~np.isin(violated, active_constraints)]
        if len(violated) > 0:
            # Add up to 20 most violated
            viol_vals = cg[violated]
            add_idx = violated[np.argsort(viol_vals)[::-1]][:20]

            active_constraints = np.unique(active_constraints)
            if len(active_constraints) > max_K:
                # Keep only top max_K by constraint value
                vals = cg[active_constraints]
                keep = np.argsort(vals)[::-1][:max_K]
                active_constraints = active_constraints[keep]

        # Line search for step size
        best_alpha = 0.0
        best_obj = My
        alphas = [2.0 / (t + 2.0)]
        # Add line search candidates
        base_alpha = 2.0 / (t + 2.0)
        for mult in [0.25, 0.5, 1.0, 1.5, 2.0, 0.1]:
            alphas.append(base_alpha * mult)
        alphas = [al for al in alphas if 0 < al <= 1.0]

        for alpha in sorted(set(alphas), reverse=True):
            y_trial = (1 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            sy = y_trial.sum()
            if sy < 1e-15:
                continue
            y_trial = y_trial / sy
            M_trial = _fast_autocorr(y_trial).max()
            if M_trial < best_obj:
                best_obj = M_trial
                best_alpha = alpha

        if best_alpha == 0.0:
            # No improvement found, try small random step
            best_alpha = 0.01

        y_new = (1 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Nesterov extrapolation
        if t >= 3:

            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        else:
            x_new = g_dir.copy()

        y = y_new
        x = x_new

        my = _fast_autocorr(y).max()
        if my < best_M:
            best_M = my
            best_a = y.copy()

        mx = _fast_autocorr(x).max()
        if mx < best_M:
            best_M = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={len(active_constraints)}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    LP-FW final ({elapsed:.0f}s): J={J:.6f}, iters={iters}")
    return best_a, best_M


def _softmax_adam(a0, time_limit, verbose=True, seed=0):
    """
    Soft-max Adam optimization with beta annealing.
    Uses top-k focused subgradient for efficiency.
    """
    n = len(a0)
    a = _norm_sum1(a0).copy()
    best_a = a.copy()
    c = _fast_autocorr(a)
    best_M = c.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Log-parameterization: a_i = exp(x_i) / sum(exp(x_j))
    x = np.log(np.maximum(a, 1e-15))
    x = x - x.mean()  # Center for numerical stability

    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.02

    # Beta schedule for soft-max
    beta_start = 4.0
    beta_end = 30.0
    total_steps_est = max(1000, int(time_limit / 0.02))  # Estimate

    it = 0
    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        it += 1

        # Current softmax beta (annealing)
        progress = min(1.0, it / total_steps_est)
        cur_beta = beta_start + (beta_end - beta_start) * progress

        # Learning rate schedule (cosine)
        cur_lr = lr * 0.5 * (1 + np.cos(np.pi * progress))

        # Compute soft-max gradient
        c = _fast_autocorr(a)
        # Top-k focus
        k_frac = max(0.1, 0.5 - 0.4 * progress)
        k_top = max(10, int(k_frac * len(c)))
        top_idx = np.argsort(c)[::-1][:k_top]
        top_vals = c[top_idx]

        # Build focused weights
        shifted = top_vals - top_vals.max()
        w_top = np.exp(cur_beta * shifted)
        w_top = w_top / w_top.sum()

        # Build full weight vector
        w = np.zeros(len(c))
        for wi, idx in zip(w_top, top_idx):
            w[int(idx)] = wi

        # Gradient via FFT
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        w_padded = np.zeros(fft_size)
        w_padded[:len(w)] = w
        a_rev_padded = np.zeros(fft_size)
        a_rev_padded[:n] = a[::-1]
        conv = np.fft.irfft(np.fft.rfft(w_padded) * np.fft.rfft(a_rev_padded), fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]

        # Project gradient to tangent space of simplex
        grad = grad - np.mean(grad)

        # Adam update
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * (grad * grad)
        m_hat = m_state / (1 - beta1 ** it)
        v_hat = v_state / (1 - beta2 ** it)
        x = x - cur_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Convert back to simplex
        x = x - x.max()
        a_new = np.exp(x)
        a_new = a_new / a_new.sum()
        a = a_new

        # Evaluate
        M_new = _fast_autocorr(a).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Periodic restart
        if stall > 1000 and it > 500:
            stall = 0
            # Perturb around best
            pert = rng.normal(0, 0.02, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-15)
            a_r = a_r / a_r.sum()
            a = a_r
            x = np.log(np.maximum(a, 1e-15))
            x = x - x.mean()
            m_state = np.zeros(n)
            v_state = np.zeros(n)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    Adam iter={it} ({elapsed:.0f}s): J={J:.6f}, beta={cur_beta:.1f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_M
        print(f"    Adam final ({elapsed:.0f}s): J={J:.6f}, iters={it}")
    return best_a, best_M


def _interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

    if t3 > 20:

        a_out, _ = _peeling_optimize(a_in, t3, verbose=True, max_peels=2000)
        try_update(a_out, "peel2")

    # === Phase 4: Soft-max Adam ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Soft-max Adam (n={n_best}) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _softmax_adam(a_in, t4, verbose=True, seed=42)
        try_update(a_out, "adam1")

    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW Round 2 (n={n_best}) ===")
    t5 = min(remaining * 0.20, 350.0)
    if t5 > 60 and n_best <= 5000:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _lp_frank_wolfe(a_in, t5, solver, verbose=True, init_K=120, max_K=1200)
        try_update(a_out, "lp2")

    # === Phase 6: Adam round 2 with different seed ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Adam Round 2 ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _softmax_adam(a_in, t6, verbose=True, seed=123)
        try_update(a_out, "adam2")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidate_lengths = [n_best - 100, n_best + 100, n_best - 200, n_best + 200]
    candidate_lengths = [l for l in candidate_lengths if 2000 <= l <= 5000]

    for new_n in candidate_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n=== Phase 7: Length n={new_n} ===")
        a_interp = _interpolate_sequence(_norm_sum1(best_seq), new_n)

        t7a = min(remaining * 0.25, 120.0)
        if t7a > 20:
            a_out, _ = _peeling_optimize(a_interp, t7a, verbose=False, max_peels=1000)
            try_update(a_out, f"peel_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 3500:
            t7b = min(remaining * 0.30, 120.0)
            if t7b > 20:
                a_in = _norm_sum1(best_seq) if len(best_seq) == new_n else a_interp
                a_out, _ = _lp_frank_wolfe(a_in, t7b, solver, verbose=False, init_K=80, max_K=600)
                try_update(a_out, f"lp_n{new_n}")

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 4000:
        t8 = min(remaining * 0.5, 200.0)
        if t8 > 20:
            a_in = _norm_sum1(best_seq)
            a_out, _ = _lp_frank_wolfe(a_in, t8, solver, verbose=True, init_K=150, max_K=1500)
            try_update(a_out, "lp_final")

    # === Phase 9: Final peeling polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final Peeling (n={n_best}) ===")
    if remaining > 20:
        t9 = min(remaining * 0.6, 100.0)
        if t9 > 10:
            a_in = _norm_sum1(best_seq)
            a_out, _ = _peeling_optimize(a_in, t9, verbose=False, max_peels=1000)
            try_update(a_out, "peel_final")

    # === Final validation ===
    if best_seq is not None:
        fs = _evaluate_fast(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = _make_symmetric_init(n_fb, seed=1)
        best_score = _evaluate_fast(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
