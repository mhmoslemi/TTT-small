        top_eps = np.where(ac > M - eps)[0]

        n_active_actual = len(top_eps)
        if n_active_actual == 0:
            break

        # But we want to minimize the max, so we want to move in the -subgradient direction
        # projected onto the tangent space of the simplex (sum(d) = 0)
        
        # Compute subgradient
        weights = np.ones(n_active_actual) / n_active_actual
        grad = np.zeros(n)
        for wi, k in zip(weights, top_eps):
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            grad[idx] += 2.0 * wi * a[partner]
        
        # Project gradient onto tangent space (remove mean)
        grad_tangent = grad - grad.mean()
        
        # Step size: use backtracking
        alpha = 0.01
        for _ in range(50):
            a_new = a - alpha * grad_tangent
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                alpha /= 2
                continue
            a_new = a_new / s
            M_new = fast_autocorr(a_new).max()
            if M_new < M:
                a = a_new
                break
            alpha /= 2
        
    return a

Let me implement the active-set QP approach:

Given active peaks k_1, ..., k_K, I want to minimize:
f(a) = max_{i=1..K} c_{k_i}(a)

subject to sum(a) = 1, a >= 0.

At the optimum of this relaxed problem, all K peaks should be equal (if the solution is interior). So I can solve:
minimize M
subject to:
  c_{k_i}(a) <= M for i = 1..K
  sum(a) = 1
  a >= 0

This is a QCQP. Each constraint c_{k_i}(a) <= M is:
sum_{j: 0<=k_i-j<n} a[j] * a[k_i-j] <= M

This is a quadratic constraint. If I linearize around the current point a:
c_{k_i}(a + d) ≈ c_{k_i}(a) + 2 * sum_j a[k_i-j] * d[j] (for valid j)

Then the linearized problem is:
minimize M
subject to:
  c_{k_i}(a) + 2 * sum_j a[k_i-j] * d[j] <= M for i = 1..K
  sum(d) = 0
  a + d >= 0

This is a linear program in (d, M)! I can solve it with cvxpy.

Then I take an appropriate step in direction d (possibly damped) and iterate.

This is essentially a linearization-based method (like the cutter method in global optimization, or the linearization of the trust region method).

import numpy as np
import time
import cvxpy as cp

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def active_set_lp(a0, solver, time_limit=300, K_max=30, verbose=True):
    """
    Key new method: Linearize the top-K peak constraints around current point,
    solve LP for optimal direction, take damped step.
    Iteratively increases K as peaks become more balanced.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    K = 5  # Start with fewer peaks, increase over time
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < 10000:
        it += 1
        ac = fast_autocorr(a)
        M = ac.max()

        # Gradually increase K
        if it % 30 == 0 and K < K_max:
            K = min(K + 2, K_max)

        # Get active peaks
        active_k = np.argsort(ac)[::-1][:K]
        peak_vals = ac[active_k]
        spread = peak_vals[0] - peak_vals[-1]

        # Set up LP: minimize M_new subject to linearized constraints
        d = cp.Variable(n)
        M_var = cp.Variable()

        constraints = [
            cp.sum(d) == 0,
            d >= -a * 0.5 + 1e-12,  # a + d >= 0.5*a (relaxed)
            d <= a * 0.5,            # limit step size
        ]

        # Linearized peak constraints: c_k(a) + grad_c_k(a) @ d <= M_var
        for k in active_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            # grad of c_k at a: vector where position j gets 2*a[k-j] if valid
            lin_grad = np.zeros(n)
            lin_grad[idx] = 2.0 * a[partner]
            constraints.append(ac[ki] + lin_grad @ d <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        d_val = np.asarray(d.value, dtype=np.float64)
        if not np.all(np.isfinite(d_val)):
            continue

        # Line search: try multiple step sizes
        best_M_trial = M
        best_t = 0.0
        for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02]:
            a_trial = a + t * d_val
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:

            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_t = t

        if best_t > 0:
            a = a + best_t * d_val
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m - 1e-14:
                best_m = M_new
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    elapsed = time.monotonic() - t0
                    J = 2.0 * n * best_m
                    print(f"    ASLP it={it} ({elapsed:.0f}s): J={J:.6f}, K={K}, spread={spread:.2e}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # If stuck, perturb
        if no_improve > 40:
            no_improve = 0
            rng = np.random.default_rng(it * 7919)
            scale = 0.005
            pert = rng.normal(0, scale, n) * a
            a_new = a + pert
            a_new = np.maximum(a_new, 0.0)

                M_new = fast_autocorr(a_new).max()

def smooth_lse_mirror_descent(a0, time_limit=300, beta_start=200, beta_end=2000,
                              verbose=True, rng_seed=42):
    """
    Smooth optimization using log-sum-exp approximation with Nesterov acceleration.
    Uses exponential weights (mirror map) for simplex constraint.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = np.log(a + 1e-30)  # work in log space
    x_prev = x.copy()
    it = 0

    total_iters_est = 100000
    for t in range(1, total_iters_est + 1):
        if time.monotonic() - t0 > time_limit:
            break
        it = t

        # Anneal beta
        frac = min(1.0, t / (total_iters_est * 0.8))
        beta = beta_start + (beta_end - beta_start) * frac

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y_log = x + beta_t * (x - x_prev)

        # Convert to probability
        y_log_shift = y_log - y_log.max()
        y = np.exp(y_log_shift)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        # Soft subgradient
        max_ac = ac.max()
        shifted = beta * (ac - max_ac)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -50, 50)
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / exp_shifted.sum()

        # Compute gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] (valid range)
        grad = np.zeros(n)
        # Use FFT for efficiency
        # grad[j] = 2 * sum_{k: max(0,j) <= k <= min(2n-2, j+n-1)} weights[k] * a[k-j]
        # This is cross-correlation of weights and a
        # Use direct computation for correctness, FFT if n is large
        if n <= 5000:
            for k in range(len(ac)):
                if weights[k] < 1e-12:
                    continue
                ki = int(k)
                s = max(0, ki - n + 1)
                e = min(n - 1, ki)
                idx = np.arange(s, e + 1)
                partner = ki - idx
                grad[idx] += 2.0 * weights[k] * y[partner]

        # Mirror descent step
        eta = 1.0 / (np.sqrt(t) + 1.0) * 2.0
        x_new = y_log - eta * grad
        x_new = x_new - x_new.max()  # normalization

        x_prev = x
        x = x_new

        # Convert back
        a_new = np.exp(x)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sa

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()

        # Nesterov extrapolation
        x = x + beta_t * (x - x_prev)
        x = x - x.max()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE-MD t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.0f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    LSE-MD final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def fine_coordinate_descent(a0, time_limit=200, verbose=True, rng_seed=123):
    """
    Very fine coordinate descent with exact line search.
    Focuses on the single most responsible pair for each top peak.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500
    n_top = 10

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            # Find most responsible pair
            best_i_local = np.argmax(contrib)
            src = idx[best_i_local]
            src_partner = partner[best_i_local]

            # Try moving mass from src to various destinations
            # Build sensitivity: how much does each position contribute to top peaks?
            sens = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                idx2 = np.arange(s2, e2 + 1)
                sens[idx2] += a[k2i - idx2]

            # Find low-sensitivity destinations
            exclude = {src, src_partner}
            low_sens = np.argsort(sens)[:n // 3]
            candidates = [d for d in low_sens if d not in exclude][:20]
            if not candidates:
                candidates = [d for d in range(n) if d not in exclude][:20]

            found = False
            for dst in candidates[:8]:
                # Line search on fraction
                lo, hi = 0.0, min(a[src], 0.3 * a[src])
                if hi < 1e-12:
                    continue
                best_frac = 0
                best_M_trial = M_curr
                for _ in range(20):
                    mid = (lo + hi) / 2
                    a_trial = a.copy()
                    a_trial[src] -= mid
                    a_trial[dst] += mid
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_M_trial:
                        best_M_trial = M_trial
                        best_frac = mid
                        lo = mid
                    else:
                        hi = mid

                if best_frac > 1e-10 and best_M_trial < best_m - 1e-14:
                    a[src] -= best_frac
                    a[dst] += best_frac
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    best_m = best_M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if found:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall > 50:
                j = int(rng.integers(n))
                k = int(rng.integers(n))
                if j != k and a[j] > 1e-10:
                    frac = rng.uniform(0.002, 0.01)
                    amt = a[j] * frac
                    a[j] -= amt
                    a[k] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Coord final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def peak_balancing(a0, time_limit=200, verbose=True, rng_seed=55):
    """
    Water-filling approach: iteratively identify which peak is highest,
    then find the optimal redistribution to reduce it while keeping others bounded.
    Uses a simple gradientprojection with adaptive step.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    peak_history = []
    stalld = 0

    while time.monotonic() - t0 < time_limit and stalld < 300:
        ac = fast_autocorr(a)
        M = ac.max()
        peak_idx = np.argmax(ac)

        # Track peak history to detect cycling
        peak_history.append(peak_idx)
        if len(peak_history) > 100:
            peak_history.pop(0)
            # If cycling among same peaks, increase perturbation
            if len(set(peak_history[-20:])) <= 5:
                stalld += 5

        # Compute subgradient of the max peak
        # At argmax, subgradient is the gradient of that peak
        ki = int(peak_idx)
        s = max(0, ki - n + 1)
        e = min(n - 1, ki)
        idx = np.arange(s, e + 1)
        partner = ki - idx
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[partner]

        # Check if other peaks are close (within epsilon)
        eps_rel = 1e-4
        close_peaks = np.where(ac > M * (1 - eps_rel))[0]
        if len(close_peaks) > 1:
            # Average gradient over close peaks
            grad = np.zeros(n)
            for k2 in close_peaks[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                idx2 = np.arange(s2, e2 + 1)
                grad[idx2] += 2.0 * a[k2i - idx2]
            grad /= len(close_peaks[:10])

        # Project gradient onto tangent space (sum = 0)
        grad_t = grad - grad.mean()

        # Adaptive step size with backtracking
        alpha = 0.05
        improved_step = False
        for _ in range(30):
            a_trial = a - alpha * grad_t
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                alpha *= 0.5
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < M - 1e-14:
                a = a_trial
                improved_step = True
                break
            alpha *= 0.7

        if improved_step:
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                stalld = 0
        else:
            stalld += 1
            # Perturb to escape
            if stalld % 20 == 0:
                scale = min(0.02, 0.005 * stalld / 20)
                pert = rng.normal(0, scale, n) * a
                a_new = a + pert
                a_new = np.maximum(a_new, 0.0)
                if a_new.sum() > 1e-15:
                    a_new = a_new / a_new.sum()
                    if fast_autocorr(a_new).max() < best_m:
                        best_m = fast_autocorr(a_new).max()
                        best_a = a_new.copy()
                    a = a_new

        if verbose and (stalld % 50 == 0 or stalld == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakBal ({elapsed:.0f}s): J={J:.6f}, stall={stalld}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakBal final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.12) / width) ** 2) + np.exp(-((x - 0.88) / width) ** 2) + 0.003
    return norm_sum1(a)


def spectral_init(n, n_modes=30, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.12 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator:
    Phase 1: Active-set LP (new method, 500s)
    Phase 2: Smooth LSE mirror descent (350s)
    Phase 3: Fine coordinate descent (250s)
    Phase 4: Peak balancing (200s)
    Phase 5: Multi-length exploration (200s)
    Phase 6: Final active-set LP + coord (200s)
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = make_u_shape(n_base, 1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Active-set LP (main new method) ===
    elapsed_so_far = time.monotonic() - t_global
    remaining = time_budget - elapsed_so_far
    print(f"\n=== Phase 1: Active-Set LP (n={n0}, remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.28, 500.0)
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = active_set_lp(a_in, solver, time_limit=t1, K_max=30, verbose=True)
        try_update(a_out, "aslp_main")

    # === Phase 2: Smooth LSE mirror descent ===
    elapsed_so_far = time.monotonic() - t_global
    remaining = time_budget - elapsed_so_far
    print(f"\n=== Phase 2: Smooth LSE-MD (n={len(best_seq)}, remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = smooth_lse_mirror_descent(a_in, time_limit=t2, beta_start=100,
                                             beta_end=1500, verbose=True, rng_seed=42)
        try_update(a_out, "lse_md")

    # === Phase 3: Fine coordinate descent ===
    elapsed_so_far = time.monotonic() - t_global
    remaining = time_budget - elapsed_so_far
    print(f"\n=== Phase 3: Fine Coord Descent (n={len(best_seq)}, remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.14, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = fine_coordinate_descent(a_in, time_limit=t3, verbose=True, rng_seed=123)
        try_update(a_out, "coord_fine")

    # === Phase 4: Peak balancing ===
    elapsed_so_far = time.monotonic() - t_global
    remaining = time_budget - elapsed_so_far
    print(f"\n=== Phase 4: Peak Balancing (n={len(best_seq)}, remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_balancing(a_in, time_limit=t4, verbose=True, rng_seed=55)
        try_update(a_out, "peakbal")

    # === Phase 5: Multi-length exploration ===
    elapsed_so_far = time.monotonic() - t_global
    remaining = time_budget - elapsed_so_far
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Multi-length (n_best={n_best}, remaining={remaining:.0f}s) ===")

    trial_lengths = []
    for delta in [300, -300, 600, -600]:
        new_n = n_best + delta
        if 2000 <= new_n <= 6000 and new_n != n_best:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 100]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.40, 200.0)
        if t5a > 25:
            a_out, _ = active_set_lp(a_long, solver, time_limit=t5a, K_max=25, verbose=False)
            try_update(a_out, f"aslp_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            t5b = min(remaining * 0.35, 100.0)
            if t5b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = fine_coordinate_descent(a_in, time_limit=t5b, verbose=False, rng_seed=new_n)
                try_update(a_out, f"coord_n{new_n}")

    # === Phase 6: Final active-set LP + coordinate ===
    elapsed_so_far = time.monotonic() - t_global
    remaining = time_budget - elapsed_so_far
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish (n={n_best}, remaining={remaining:.0f}s) ===")

    if remaining > 40 and n_best <= 6000:
        t6a = min(remaining * 0.45, 180.0)
        if t6a > 25:
            a_in = norm_sum1(best_seq)
            a_out, _ = active_set_lp(a_in, solver, time_limit=t6a, K_max=35, verbose=True)
            try_update(a_out, "aslp_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t6b = min(remaining * 0.5, 100.0)
        if t6b > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = fine_coordinate_descent(a_in, time_limit=t6b, verbose=False, rng_seed=999)
            try_update(a_out, "coord_final")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t6c = min(remaining * 0.6, 60.0)
        if t6c > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_balancing(a_in, time_limit=t6c, verbose=False, rng_seed=777)
            try_update(a_out, "peakbal_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_u_shape(n_fb, 1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
