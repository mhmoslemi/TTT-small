import numpy as np
import time


def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = fast_conv(a)
    S = a.sum()
    return 2.0 * n * conv.max() / (S * S)


def compute_grad_conv(a, beta, top_k=300):
    """
    Compute gradient of LSE_beta(conv(a,a)) w.r.t. a.
    Uses FFT-based cross-correlation for efficiency.
    Returns (gradient, conv, effective_beta).
    """
    n = len(a)
    conv = fast_conv(a)
    conv_len = len(conv)
    
    # Select top-k lags
    k_sel = min(top_k, conv_len)
    top_idx = np.argpartition(conv, -k_sel)[-k_sel:]
    top_vals = conv[top_idx]
    
    # Softmax weights (Gibbs distribution)
    shifted = top_vals - top_vals.max()
    p = np.exp(beta * shifted)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(k_sel) / k_sel
    else:
        p = p / p_sum
    
    # Gradient via cross-correlation: grad[j] = 2 * sum_k p[k] * a[k-j]
    # = 2 * (a cross-corr P)[j] where P[k] = p for k in top_idx

    A_hat = np.fft.rfft(a, fft_size)
    P = np.zeros(fft_size)
    P[top_idx] = p
    P_hat = np.fft.rfft(P, fft_size)
    cross = np.fft.irfft(np.conj(A_hat) * P_hat, fft_size)
    grad = 2.0 * cross[:n]
    
    return grad, conv

        # Identify peak lags
        n_peaks = min(30, len(conv))
        peak_idx = np.argsort(conv)[::-1][:n_peaks]
        
        # Compute "pressure" at each position
        # Position i is "pressured" if it participates in many peak lags with large partners
        pressure = np.zeros(n)
        for k in peak_idx[:20]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            idx = np.arange(i_s, i_e + 1)
            partners = k_i - idx
            pressure[idx] += a[partners]
        
        # Find best (src, dst) pair to swap
        # Source: high pressure AND has significant mass
        # Dest: low pressure
        pressure_norm = pressure / (pressure.max() + 1e-15)
        
        # Score sources: want high pressure * mass
        src_score = pressure * a
        # Score destinations: want low pressure
        dst_score = -pressure_norm
        
        # Try top few sources
        top_src = np.argsort(src_score)[::-1][:5]
        top_dst = np.argsort(dst_score)[:10]
        
        improved = False
        for s in top_src:
            if a[s] < 1e-10:
                continue
            for d in top_dst:
                if d == s:
                    continue
                # Try transferring some mass
                for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                    amt = a[s] * frac
                    a_t = a.copy()
                    a_t[s] -= amt
                    a_t[d] += amt
                    a_t = np.maximum(a_t, 0.0)
                    a_t = a_t / a_t.sum()
                    J_t = eval_J(a_t)
                    if J_t < best_J - 1e-12:
                        a = a_t
                        best_J = J_t
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.005, 0.1)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    J_new = eval_J(a)
                    if J_new < best_J:
                        best_J = J_new
                        best_a = a.copy()
                        stall = 0
                        break
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J


def make_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_gaussian(n, center=0.5, width=0.2):
    """Gaussian bump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2) + 1e-6
    return a / a.sum()


def make_bimodal(n, w=0.15):
    """Two Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.2) / w) ** 2) + np.exp(-((x - 0.8) / w) ** 2) + 1e-6
    return a / a.sum()


def make_cosh(n):
    """Cosh-shaped (decreases from center)."""
    x = np.linspace(-1, 1, n)
    a = 1.0 / np.cosh(3.0 * x) + 1e-6
    return a / a.sum()


def make_power_cos(n, p=1.0):
    """Power law times cosine window."""
    x = np.linspace(0, 1, n)
    a = (x ** p) * (0.5 * (1 + np.cos(np.pi * x))) + 1e-8
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_a(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass
    
    best_a = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3000
        best_a = make_ramp(n0, 1.0)
        best_score = eval_J(best_a)
        print(f"Initial: n={len(best_a)}, J={best_score:.6f}")
    
    # === Phase 1: Multi-init momentum GD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-init (budget={remaining:.0f}s) ===")
    
    n0 = len(best_a)
    inits = []
    labels = []
    
    if parent is not None:
        inits.append(parent)
        labels.append("parent")
    inits.append(make_ramp(n0, 0.5))
    labels.append("ramp05")
    inits.append(make_ramp(n0, 1.0))
    labels.append("ramp10")
    inits.append(make_ramp(n0, 1.5))
    labels.append("ramp15")
    inits.append(make_cosh(n0))
    labels.append("cosh")
    inits.append(make_bimodal(n0))
    labels.append("bimodal")
    inits.append(make_power_cos(n0, 1.0))
    labels.append("powcos")
    
    # Blend with parent
    if parent is not None:
        for pfrac in [0.3, 0.5, 0.7]:
            blend = pfrac * parent + (1 - pfrac) * make_ramp(n0, 1.0)
            inits.append(blend / blend.sum())
            labels.append(f"blend{int(pfrac*10)}")
    
    # Deduplicate
    seen = set()
    u_inits, u_labels = [], []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n0//80)], 2))
        if sig not in seen:
            seen.add(sig)
            u_inits.append(init)
            u_labels.append(lbl)
    inits, labels = u_inits, u_labels
    
    t_per = max(10.0, remaining * 0.35 / len(inits))
    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 60:
            break
        t_avail = min(t_per, time_budget - 60 - elapsed)
        if t_avail < 8:
            break
        print(f"  GD [{lbl}] ({t_avail:.0f}s)...")
        a_out, _ = momentum_gd(init, t_avail, beta_start=10, beta_end=200,
                                momentum=0.9, lr_start=0.5, lr_decay=0.9998,
                                top_k=300, seed=100+idx)
        try_update(a_out, f"gd_{lbl}")
    
    # === Phase 2: Extended GD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Extended GD ({remaining:.0f}s remaining) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_out, _ = momentum_gd(best_a.copy(), t2, beta_start=50, beta_end=500,
                                momentum=0.92, lr_start=0.3, lr_decay=0.9995,
                                top_k=400, seed=777, verbose=True)
        try_update(a_out, "gd_ext")
    
    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord Descent ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 15:
        a_out, _ = coordinate_descent(best_a.copy(), t3, seed=42, verbose=True)
        try_update(a_out, "cd1")
    
    # === Phase 4: GD after CD ===
    remaining = time_budget - (time.monotonic() - t_global)

        a_out, _ = momentum_gd(best_a.copy(), t4, beta_start=100, beta_end=1000,

    # === Phase 5: Length sweeping ===

    lengths = []
    for d in [-500, -300, -200, -100, 100, 200, 300, 500, 700, 1000]:
        nl = n_best + d

            lengths.append(nl)

            lengths.append(nl)

            break
        if abs(nl - n_best) <= 50:

        a_int = interpolate_a(best_a, nl)

                                    momentum=0.9, lr_start=0.4, lr_decay=0.9997,

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 6: Final Polish (n={n_cur}) ===")
    
    # GD with high beta

    if t6 > 15:

    t7 = min(remaining * 0.30, 150.0)

        a_out, _ = coordinate_descent(best_a.copy(), t7, seed=314, verbose=True)
        try_update(a_out, "polish_cd")

    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.30, 150.0)
    if t8 > 10:
        a_out, _ = momentum_gd(best_a.copy(), t8, beta_start=500, beta_end=5000,
                                momentum=0.85
