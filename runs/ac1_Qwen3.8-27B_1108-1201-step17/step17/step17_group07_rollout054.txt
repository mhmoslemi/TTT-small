"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp mirror descent with exact gradient (soft-max approximation of max)
2. Annealed tau for progressive sharpening
3. LP-based Remez for final polish
4. Top-k Nesterov for initial fast convergence
5. Multi-length exploration
Key: gradient of lse(tau*b) w.r.t. a[j] = 2 * (a ⊛ p)[j] where p = softmax(tau*b),
computed via np.correlate in O(n^2) per step (~10ms for n=2600).
"""

import numpy as np
import cvxpy as cp
import time

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    return _global_solver

def fast_conv(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def lse_mirror_descent(a0, time_limit, tau_start=1.0, tau_max=500.0, 
                        tau_anneal_rate=500.0, c0=1.0, verbose=False):
    """Mirror descent with log-sum-exp objective and annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 5000
    rng = np.random.default_rng(12345 + n)
    
    for t in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        tau = min(tau_max, tau_start + t / tau_anneal_rate)
        
        b = fast_conv(a)
        M = b.max()
        
        # Softmax weights (numerically stable)
        tau_b = tau * b
        tau_b_shifted = tau_b - tau_b.max()
        p = np.exp(tau_b_shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros_like(b)
            p[np.argmax(b)] = 1.0
        else:
            p = p / p_sum
        
        # Gradient: g[j] = 2 * sum_m a[m] * p[m+j]
        g = 2.0 * np.correlate(p, a, mode='valid')
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t + 1.0) + 1.0)
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        if stall > max_stall:
            # Restart with perturbation
            strategy = int(rng.integers(4))
            if strategy == 0:
                pert = rng.normal(0, 0.01, n)
                a_r = best_a * (1 + pert)
            elif strategy == 1:
                a_r = best_a.copy()
                idx = rng.choice(n, size=min(n//4, 500), replace=False)
                a_r[idx] *= rng.uniform(0.7, 1.3, len(idx))
            elif strategy == 2:
                a_r = best_a * (1 + rng.normal(0, 0.003, n))
            else:
                # Localized perturbation
                center = int(rng.integers(n // 4, 3 * n // 4))
                width = max(5, n // 40)
                idx = np.arange(max(0, center - width), min(n, center + width))
                a_r = best_a.copy()
                a_r[idx] *= rng.uniform(0.8, 1.2, len(idx))
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r
            stall = 0
            max_stall = max(1000, int(max_stall * 0.7))
            # Reset tau to restart annealing
            tau_anneal_rate = max(100.0, tau_anneal_rate * 0.8)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE-MD t={t} ({elapsed:.1f}s): tau={tau:.1f}, M={best_m:.10f}, J={J:.8f}, stall={stall}")
    
    return best_a, best_m
