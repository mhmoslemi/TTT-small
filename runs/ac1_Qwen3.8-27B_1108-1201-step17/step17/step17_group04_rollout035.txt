                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-9:
                        delta = rng.uniform(0.005, 0.05) * a[i]
                        a[i] -= delta
                        a[j] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()

        if verbose and stall % 50 == 0:
            print(f"    CoordDesc ({time.monotonic()-t0:.0f}s): J={best_J:.7f} stall={stall}")

    return best_a, best_J


def structured_perturb_optimize(a0, time_limit, verbose=False, seed=100):
    """Apply smooth frequency-based perturbations and re-optimize."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    freqs = [1, 2, 3, 4, 5, 7, 11, 13, 17, 23]

    for freq in freqs:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        x = np.arange(n)
        h_cos = np.cos(2.0 * np.pi * freq * x / n)
        h_sin = np.sin(2.0 * np.pi * freq * x / n)
        h_cos = h_cos - h_cos.mean()
        h_sin = h_sin - h_sin.mean()

        for h in [h_cos, h_sin]:
            if time.monotonic() - t0 > time_limit:
                break

            best_eps = 0.0
            best_trial = best_J
            eps_candidates = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5,
                              -0.01, -0.02, -0.05, -0.1, -0.15, -0.2, -0.3, -0.4, -0.5]

            for eps in eps_candidates:
                a_pert = a * (1.0 + eps * h)
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s < 1e-15:
                    continue
                a_pert = a_pert / s
                J_trial = compute_J(a_pert)
                if J_trial < best_trial:
                    best_trial = J_trial
                    best_eps = eps

            if best_eps != 0.0 and best_trial < best_J - 1e-14:
                h_use = h_cos if best_eps in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5] else h_sin
                if best_eps < 0:
                    for e in eps_candidates:
                        if e == best_eps:
                            break
                    # Figure out which h was used
                    # Just recompute
                    a_pert = a * (1.0 + best_eps * h)
                    a_pert = np.maximum(a_pert, 0.0)
                    a = a_pert / a_pert.sum()
                    best_J = best_trial
                    best_a = a.copy()

                # Re-optimize with L-BFGS-B
                t_avail = min(remaining * 0.3, 60.0)
                if t_avail > 10:
                    a_opt, J_opt = lbfgs_smooth_max(a, t_avail,
                                                     lam_schedule=[50, 100, 300, 800],
                                                     verbose=False)
                    if J_opt < best_J:
                        best_J = J_opt
                        best_a = a_opt.copy()

        if verbose:
            print(f"    StructPert freq={freq} ({time.monotonic()-t0:.0f}s): J={best_J:.7f}")

    return best_a, best_J


def make_symmetric_init(n):
    """U-shaped symmetric initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.05
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_power_law_init(n, power=0.7):
    """Power law from edges."""
    x = np.linspace(-1, 1, n)
    a = (1 - x ** 2) ** power + 0.005
    return norm_sum1(a)


def make_gaussian_edges(n, sigma=0.12):
    """Gaussian bumps at edges."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.03) / sigma) ** 2) + np.exp(-((x - 0.97) / sigma) ** 2) + 0.01
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def gradient_refinement(a0, time_limit, verbose=False, seed=777):
    """
    Custom gradient-based refinement using the smooth max with adaptive step size.
    Faster than L-BFGS-B for very high lambda (near-smooth regime).
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    lam = 500.0
    step = 0.05
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        max_ac = ac.max()
        shifted = lam * (ac - max_ac)
        shifted = np.clip(shifted, -700.0, 700.0)
        p = np.exp(shifted)
        p = p / p.sum()
        grad = 2.0 * np.correlate(p, a, mode='valid')

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        # Mirror descent step (exponential)
        log_a = np.log(np.maximum(a, 1e-30)) - step * grad
        log_a = log_a - log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            # Reduce step on improvement
            step *= 0.95
        else:
            stall += 1
            step *= 1.02
            step = min(step, 0.15)

            if stall > 500:
                # Restart with perturbation
                pert = rng.normal(0, 0.01, n) * best_a
                a = best_a + pert
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                stall = 0
                step = 0.05
                # Increase lambda for sharper gradient
                lam = min(lam * 1.5, 5000.0)

        if verbose and t % 5000 == 0:
            print(f"    GradRefl t={t} ({time.monotonic()-t0:.0f}s): J={best_J:.7f} lam={lam:.0f}")

    return best_a, best_J


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining smooth-max L-BFGS-B, coordinate descent,
    structured perturbations, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.7f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.7f}")

    # === Phase 1: L-BFGS-B smooth max from parent ===
    print(f"\n=== Phase 1: L-BFGS-B smooth max (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = lbfgs_smooth_max(a_in, t1,
                                     lam_schedule=[15, 30, 60, 120, 250, 500, 1000],
                                     verbose=True)
        try_update(a_out, "lbfgs_parent")

    # === Phase 2: L-BFGS-B from alternative inits ===
    print(f"\n=== Phase 2: L-BFGS-B alternative inits ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2_total = min(remaining * 0.15, 250.0)
    if t2_total > 40:
        inits = []
        labels = []
        inits.append(make_symmetric_init(n0))
        labels.append("sym")
        inits.append(make_power_law_init(n0, power=0.5))
        labels.append("pow05")
        inits.append(make_gaussian_edges(n0, sigma=0.10))
        labels.append("gauss")
        if parent is not None:
            blend = 0.7 * norm_sum1(parent) + 0.3 * make_symmetric_init(n0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        t_per = t2_total / len(inits)
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 20)
            if t_avail < 15:
                break
            print(f"  L-BFGS init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = lbfgs_smooth_max(init, t_avail,
                                         lam_schedule=[20, 50, 100, 300, 800],
                                         verbose=False)
            try_update(a_out, f"lbfgs_{lbl}")

    # === Phase 3: Pairwise coordinate descent ===
    print(f"\n=== Phase 3: Pairwise coord descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_coord_desc(a_in, t3, verbose=True, seed=42)
        try_update(a_out, "coorddesc1")

    # === Phase 4: Structured perturbation ===
    print(f"\n=== Phase 4: Structured perturbation (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = structured_perturb_optimize(a_in, t4, verbose=True, seed=100)
        try_update(a_out, "structpert")

    # === Phase 5: Gradient refinement ===
    print(f"\n=== Phase 5: Gradient refinement (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = gradient_refinement(a_in, t5, verbose=True, seed=777)
        try_update(a_out, "gradrefl")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    candidates_n = [n_best - 200, n_best + 200, n_best + 400, n_best + 800]
    candidates_n = [c for c in candidates_n if 1000 <= c <= 6000]

    for new_n in candidates_n[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.25, 150.0)
        if t6a > 20:
            a_out, _ = lbfgs_smooth_max(a_interp, t6a,
                                         lam_schedule=[20, 50, 100, 300],
                                         verbose=False)
            try_update(a_out, f"lbfgs_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = pairwise_coord_desc(a_in, min(remaining * 0.15, 60.0),
                                           verbose=False, seed=555)
            try_update(a_out, f"coord_n{len(a_in)}")

    # === Phase 7: Final L-BFGS-B polish ===
    print(f"\n=== Phase 7: Final L-BFGS-B polish (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.25, 250.0)
    if t7 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = lbfgs_smooth_max(a_in, t7,
                                     lam_schedule=[100, 300, 600, 1200, 2000],
                                     verbose=True)
        try_update(a_out, "lbfgs_final")

    # === Phase 8: Final coordinate descent ===
    print(f"\n=== Phase 8: Final coord descent (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = pairwise_coord_desc(a_in, t8, verbose=True, seed=1234)
        try_update(a_out, "coorddesc_final")

    # === Phase 9: One more structured perturbation if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 9: Final structured perturbation ===")
        t9 = min(remaining * 0.5, 100.0)
        a_in = norm_sum1(best_seq)
        a_out, _ = structured_perturb_optimize(a_in, t9, verbose=False, seed=2000)
        try_update(a_out, "structpert_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.7f}")

    return [float(x) for x in best_seq]
