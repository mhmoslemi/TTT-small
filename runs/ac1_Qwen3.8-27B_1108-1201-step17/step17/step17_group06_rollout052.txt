"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth gradient with adaptive soft-max temperature
2. Adam optimizer in log-softmax parameterization
3. Coordinate descent polish with pressure-based transfers
4. Multi-strategy initialization and length exploration
"""

import numpy as np
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_conv(a_norm)
    return 2.0 * n * conv.max()


def compute_grad_fft(a, top_k=200, beta_sm=50.0):
    """Compute gradient of J w.r.t. a using FFT-based O(n log n) method."""
    n = len(a)
    s = a.sum()
    a_norm = a / s
    conv = fast_conv(a_norm)
    conv_len = 2 * n - 1

    k_sel = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_sel]
    top_vals = conv[top_idx]

    # Soft-max weights
    c = top_vals.max()
    exp_c = np.exp(beta_sm * (top_vals - c))
    w = exp_c / exp_c.sum()

    # Create sparse weight vector and compute gradient via FFT
    w_full = np.zeros(conv_len)
    w_full[top_idx] = w
    a_rev = a_norm[::-1]

    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    Ar = np.fft.rfft(a_rev, fft_size)
    conv_wa = np.fft.irfft(W * Ar, fft_size)

    g_conv = 2.0 * conv_wa[n - 1:2 * n - 1]

    max_conv = float((w * top_vals).sum())
    g = 2.0 * n * (g_conv * s - 2.0 * max_conv * a) / (s ** 3)

    return g, 2.0 * n * max_conv / (s ** 2)


def adam_optimize(a0, time_limit, lr=0.05, beta1=0.9, beta2=0.999,
                  top_k_init=50, top_k_max=300, beta_sm_init=200.0,
                  beta_sm_min=10.0, verbose=False, seed=42):
    """Adam optimization with softmax parameterization and FFT gradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()

    b = np.log(a)
    b = b - b.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    stall = 0
    max_stall = 3000
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed_frac = min(1.0, t / 20000.0)

        cur_top_k = int(top_k_init + (top_k_max - top_k_init) * elapsed_frac)
        cur_beta_sm = max(beta_sm_min, beta_sm_init * (1.0 - 0.8 * elapsed_frac))
        lr_t = lr / (1.0 + 0.005 * t)

        g_a, J_val = compute_grad_fft(a, top_k=cur_top_k, beta_sm=cur_beta_sm)

        # Transform to b-space
        mu = float(np.dot(g_a, a))
        g_b = a * (g_a - mu)

        # Gradient clipping
        g_norm = np.linalg.norm(g_b)
        if g_norm > 50.0:
            g_b = g_b / g_norm * 50.0

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_b
        v = beta2 * v + (1.0 - beta2) * (g_b * g_b)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        b -= lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Convert back
        b_shifted = b - b.max()
        a = np.exp(b_shifted)
        a = a / a.sum()

        if J_val < best_val - 1e-12:
            best_val = J_val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            stall = 0
            scale = rng.uniform(1.0, 3.0)
            b = np.log(np.maximum(best_a, 1e-15))
            b += rng.normal(0, scale, n)
            b = b - b.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0

        if verbose and t % 1000 == 0:
            print(f"    Adam t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}, "
                  f"bsm={cur_beta_sm:.1f}, k={cur_top_k}")

    return best_a, best_val


def coord_descent(a0, time_limit, verbose=False, seed=42):
    """Coordinate descent with pressure-based mass transfer."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        conv = fast_conv(a)
        M = conv.max()
        top_lags = np.argsort(conv)[::-1][:min(30, 2 * n - 1)]

        pressure = np.zeros(n)
        for k in top_lags[:12]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                pressure[idx] += a[idx] * a[k_i - idx]

        valid = pressure * a > 1e-20
        if not valid.any():
            stall += 1
            break
        src_scores = np.where(valid, pressure * a, 0.0)
        src = int(np.argmax(src_scores))
        if a[src] < 1e-12:
            stall += 1
            continue

        dest_scores = np.zeros(n)
        for k in top_lags[:6]:
            k_i = int(k)
            i_s = max(0, k_i - n + 1)
            i_e = min(n - 1, k_i)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                dest_scores[idx] += a[k_i - idx]
        dest_scores[src] = np.inf
        dest_scores += 0.1 * pressure
        dst = int(np.argmin(dest_scores))

        improved = False
        for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J(a_t)
            if J_t < best_J - 1e-14:
                a = a_t
                best_J = J_t
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        amt = a[i] * rng.uniform(0.005, 0.05)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

    return best_a, best_J


def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=80.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.1) ** 2) + np.exp(-width * (x - 0.9) ** 2) + 0.005
    return a / a.sum()


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 12):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def make_exponential(n, rate=1.0):
    x = np.linspace(0, 1, n)
    a = np.exp(rate * x)
    return a / a.sum()


def make_gaussian_ramp(n, center=0.3, width=0.4):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    return a / a.sum()

def compute_grad_fft_full(a, top_k=200, beta_sm=50.0):
    n = len(a)
    s = a.sum()
    a_norm = a / s
    
    # One FFT for a_norm
    conv_size = 2 * n - 1
    fft_size1 = 1 << (conv_size - 1).bit_length()
    A = np.fft.rfft(a_norm, fft_size1)
    conv = np.fft.irfft(A * A, fft_size1)[:conv_size]
    
    # Top-K selection
    k_sel = min(top_k, conv_size)
    top_idx = np.argsort(conv)[::-1][:k_sel]
    top_vals = conv[top_idx]
    
    # Soft weights
    c = top_vals.max()
    exp_c = np.exp(beta_sm * (top_vals - c))
    w = exp_c / exp_c.sum()
    
    # Gradient via FFT
    w_full = np.zeros(conv_size)
    w_full[top_idx] = w
    a_rev = a_norm[::-1]
    
    fft_size2 = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w_full, fft_size2)
    Ar = np.fft.rfft(a_rev, fft_size2)
    conv_wa = np.fft.irfft(W * Ar, fft_size2)
    
    g_conv = 2.0 * conv_wa[n - 1:2 * n - 1]
    
    max_conv = float((w * top_vals).sum())
    g = 2.0 * n * (g_conv * s - 2.0 * max_conv * a) / (s ** 3)
    
    return g, 2.0 * n * max_conv / (s ** 2)
