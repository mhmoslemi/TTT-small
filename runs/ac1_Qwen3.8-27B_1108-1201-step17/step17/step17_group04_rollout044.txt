3. Targeted peak-deflation coordinate moves

import numpy as np
import time
import cvxpy as cp

# --- Global solver setup ---
_global_solver = None

        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:

            prob.solve(solver=s, verbose=False)

        except Exception:

    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")

def fast_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective J = 2*n*max(c) assuming sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max()


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def subgradient_at_peak(a, c, top_k=1, beta=2.0):
    """Compute subgradient of max of top-k convolution values."""
    n = len(a)
    c_len = len(c)
    k_use = min(top_k, c_len)
    top_idx = np.argsort(c)[::-1][:k_use]
    
    if top_k == 1:
        k_max = int(top_idx[0])
        grad = np.zeros(n)
        i_start = max(0, k_max - n + 1)
        i_end = min(n, k_max + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_max - idx
        grad[idx] = a[j_idx]
        return grad
    else:
        # Soft weighting
        top_vals = np.maximum(c[top_idx], 1e-30)
        w = top_vals ** beta
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        return grad


def nesterov_mirror(a0, time_limit, verbose=False, seed=42,
                     top_k_start=1, top_k_end=20, beta=2.5,
                     lower_bound=1.0, restart_every=2000):
    """
    Nesterov-accelerated mirror descent (KL/exponential) for minimizing max(conv(a,a)).
    Uses natural non-negativity handling via log-barrier.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-30)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s
        
        c = fast_conv(y)
        M = c.max()
        
        # Adaptive top_k: start small, increase over time
        progress = min(1.0, t / 10000)
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        
        g = subgradient_at_peak(y, c, top_k=cur_top_k, beta=beta)
        
        # Polyak step size: (current - best) / ||g||^2
        gap = M - lower_bound / (2.0 * n)  # lower bound for max(c) when sum=1
        gap = max(gap, 1e-10)
        g_norm_sq = np.dot(g, g) + 1e-20
        eta = gap / g_norm_sq
        eta = min(eta, 3.0 / (t + 1.0))
        eta = max(eta, 1e-10)
        
        # Mirror descent step (exponential)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > restart_every:
            stall = 0
            sigma = 0.02
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Mirror t={t} ({elapsed:.0f}s): J={best_J:.6f}")
    
    return best_a, best_J


def lp_frank_wolfe_fast(a0, time_limit, solver, verbose=False, top_K=30):
    """
    Fast LP Frank-Wolfe: solve many small LPs with few constraints.
    Each LP maximizes sum(b) subject to T_sel @ b <= M_sel, b >= 0.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_fw = a.copy()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current convolution
        c = fast_conv(y)
        M_curr = c.max()
        if M_curr < 1e-16:
            break
        
        # Select top-K constraints
        margins = M_curr - c
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        
        # Build constraint matrix (sparse: each row has at most n non-zeros)
        T_sel = np.zeros((K_sel, n))
        for row, k in enumerate(binding_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            T_sel[row, i_start:i_end] = y[i_start:i_end]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x_fw = x_new
        
        # Check both y and x_fw
        for check in [y, x_fw]:
            m_check = fast_conv(check).max()
            if m_check < best_m:
                best_m = m_check
                best_a = check.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m


def deflation_moves(a0, time_limit, verbose=False, seed=111):
    """
    Targeted deflation: for each peak lag, compute which pairs contribute most,
    and move mass from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 2000:
        c = fast_conv(a)
        M = c.max()
        
        # Get top peaks
        n_peaks = min(10, len(c))
        top_lags = np.argsort(c)[::-1][:n_peaks]
        
        # Compute contribution map for the top peak
        k_star = int(top_lags[0])
        i_start = max(0, k_star - n + 1)
        i_end = min(n, k_star + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_star - idx
        contrib = a[idx] * a[j_idx]
        
        # Find positions with high contribution
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            stall += 1
            break
        
        # Sensitivity: how much does each position contribute to peaks?
        sensitivity = np.zeros(n)
        for k in top_lags[:5]:
            k_int = int(k)
            i_s = max(0, k_int - n + 1)
            i_e = min(n, k_int + 1)
            ii = np.arange(i_s, i_e)
            jj = k_int - ii
            cc = a[ii] * a[jj]
            for p in range(len(ii)):
                sensitivity[ii[p]] += cc[p]
                sensitivity[jj[p]] += cc[p]
        
        # Find "safe" destinations (low sensitivity, low current value)
        mean_sens = sensitivity.mean()
        safe_mask = (sensitivity < 0.5 * mean_sens) & (a < 1.0 / n)
        safe_positions = np.where(safe_mask)[0]
        
        if len(safe_positions) == 0:
            safe_mask = a < 1.0 / (2 * n)
            safe_positions = np.where(safe_mask)[0]
        
        if len(safe_positions) == 0:
            stall += 1
            continue
        
        improved = False
        
        # Try moving mass from high-contrib positions to safe positions
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a source from high-contribution positions
            if len(idx) > 0 and contrib.max() > 0:
                src_candidates = np.where(contrib > contrib.mean())[0]
                if len(src_candidates) == 0:
                    src_candidates = np.arange(len(idx))
                src_local = src_candidates[rng.integers(len(src_candidates))]
                src = idx[src_local]
            else:
                src = int(rng.integers(n))
            
            if a[src] < 1e-10:
                continue
            
            dst = safe_positions[rng.integers(len(safe_positions))]
            if src == dst:
                continue
            
            # Try several transfer fractions
            fracs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
            best_frac = None
            best_trial_J = best_J
            
            for frac in fracs:
                amt = a[src] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_frac = a_trial.copy()
            
            if best_frac is not None:
                a = best_frac
                if best_trial_J < best_J:
                    best_J = best_trial_J
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            stall += 1
            # Escape: random perturbation
            if stall % 20 == 0:
                n_pert = max(3, n // 200)
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * rng.uniform(0.01, 0.05)
                        a[i] -= amt
                        a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Deflate ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Deflate final ({elapsed:.0f}s): J={best_J:.6f}")
    return best_a, best_J


def coordinate_descent(a0, time_limit, verbose=False, seed=55):
    """
    Coordinate descent: try adjusting individual coordinates to reduce the peak.
    Uses the fact that c[k] is quadratic in a, so for fixed other coordinates,
    optimizing one coordinate is a 1D problem.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # If we change a[i] to a[i] + delta, how does c change?
    # c_new[k] = c[k] + delta*(a[k-i] + a[i]) + delta^2 * (1 if k-2i is valid...)

    # If we change a[i] by delta:
    #   c_new[k] = c[k] + delta * a[k-i] + delta * a[k-i] ... 
    # Wait, let's be more careful.
    # c[k] = sum_j a[j]*a[k-j]
    # After changing a[i] -> a[i]+delta:
    # c_new[k] = sum_j (a[j]+delta*1{j==i})*(a[k-j]+delta*1{k-j==i})

    #   or c[k] + 2*delta*a[i] + delta^2 (if k-i == i, i.e., k=2i)
    # More precisely:
    # c_new[k] = c[k] + delta * (a[k-i] + a[k-i]) + delta^2 * (1 if k-2i in [0,n-1] and the two terms both hit i)
    # Hmm, let me just do it numerically.
    
    # For coordinate descent, we'll try: for coordinate i, find the best value a[i]
    # that minimizes max(c), keeping all other coordinates fixed.
    
    # This is expensive to do exactly, so we'll use a heuristic:
    # For each i, the "cost" of a[i] is proportional to how much it contributes to peaks.
    # We want to reduce a[i] if it contributes to peaks, increase it otherwise.
    
    stall = 0
    max_stall = 1000
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M = c.max()
        peak_lags = np.where(c >= M * 0.95)[0]
        
        # Compute "pressure" on each coordinate
        pressure = np.zeros(n)
        for k in peak_lags[:5]:
            k_int = int(k)
            i_s = max(0, k_int - n + 1)
            i_e = min(n, k_int + 1)
            ii = np.arange(i_s, i_e)
            jj = k_int - ii
            # pressure on ii and jj
            for p in range(len(ii)):
                pressure[ii[p]] += a[jj[p]]
                pressure[jj[p]] += a[ii[p]]
        
        # Try to reduce high-pressure coordinates
        # and increase low-pressure coordinates
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a high-pressure source
            if pressure.max() > 0:
                p_norm = pressure / pressure.max()
                # Sample source proportional to pressure
                src = int(rng.choice(n, p=p_norm))
            else:
                src = int(rng.integers(n))
            
            # Pick a low-pressure destination
            p_dest = np.maximum(1.0 - pressure, 0)
            if p_dest.sum() < 1e-15:
                p_dest = np.ones(n)
            p_dest = p_dest / p_dest.sum()
            dst = int(rng.choice(n, p=p_dest))
            
            if src == dst or a[src] < 1e-10:
                continue
            
            # Binary search for the best transfer amount
            lo = 0.0
            hi = a[src] * 0.5
            for _ in range(8):
                mid = (lo + hi) / 2
                a_trial = a.copy()
                a_trial[src] -= mid
                a_trial[dst] += mid
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    lo = mid  # can transfer more
                else:
                    hi = mid
            
            if a is not best_a:  # if we improved
                stall = 0
        
        # Check if we improved in this round
        if best_a is not a:
            stall = 0
        else:
            stall += 1
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordDesc final ({elapsed:.0f}s): J={best_J:.6f}")
    return best_a, best_J


def make_init_symmetric(n):
    x = np.linspace(0, 1, n)
    a = (1 + np.cos(2 * np.pi * x)) / 2 + 0.01
    return normalize(a)


def make_init_power(n, power=0.5):
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** power + 0.001
    return normalize(a)


def make_init_bimodal(n, sigma=0.1):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.1) / sigma)**2) + np.exp(-((x - 0.9) / sigma)**2) + 0.005
    return normalize(a)


def make_init_random(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    return normalize(a)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
