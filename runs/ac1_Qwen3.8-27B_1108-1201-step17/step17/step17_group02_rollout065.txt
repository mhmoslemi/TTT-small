"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 with a multi-phase search:
  Phase 1: Nesterov-accelerated mirror descent (AMD) from diverse inits
  Phase 2: LP-based Frank-Wolfe (PDLP/Clarabel) with adaptive constraint set
  Phase 3: Targeted peak-pair reduction
  Phase 4: Multi-length exploration via interpolation
  Phase 5: Final LP + coordinate/peak polish

Improvements over prior code:
  - Vectorized (per-row fancy-index) LP constraint construction (fast)
  - Adaptive K growth tuned for the parent length
  - Extra compute budget on LP-FW (the strongest method)
  - Dual y/x tracking for improvement
  - Low-sensitivity destination selection in peak reduction

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        vals = [float(x) for x in a_arr]
        sc = float(evaluate_sequence(vals))
        return sc
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def obj_from_M(n, M, s=1.0):
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * M / (s * s)


def get_solver():
    if hasattr(get_solver, '_result'):
        return get_solver._result
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                get_solver._result = s
                print("  LP solver: %s" % s)
                return s
        except Exception:
            continue
    get_solver._result = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return cp.ECOS

def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_inv_u(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * (1.0 - x ** 2)
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = (np.exp(-((x - 0.15) / width) ** 2)
         + np.exp(-((x - 0.85) / width) ** 2) + 0.01)
    return norm_sum1(a)


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _softmax_subgrad(y, a, n, top_k, beta):
    ac = fast_autocorr(y)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        s1 = max(0, ki - n + 1)
        e1 = min(n, ki + 1)
        idx = np.arange(s1, e1)
        grad[idx] += 2.0 * wi * a[ki - idx]
    return grad, ac


def amd_optimize(a0, time_limit, top_k=60, beta=6.0, c0=4.0,
                 restart_every=5000, verbose=False, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 40000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        g, ac = _softmax_subgrad(y, a, n, cur_top_k, cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_beta = max(cur_beta - 1.0, 2.0)
            if verbose:
                print("    AMD restart J=%.6f" % obj_from_M(n, best_m))

        if verbose and t % 20000 == 0:
            print("    AMD t=%d (%ds) J=%.6f" %
                  (t, int(time.monotonic() - t0), obj_from_M(n, best_m)))

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False,
                   initial_K=None, max_K=None, K_growth=60, K_interval=15):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    total_lags = 2 * n - 1
    if initial_K is None:
        initial_K = min(250, total_lags)
    if max_K is None:
        max_K = min(1600, total_lags)
    K_sel = initial_K

    y = a.copy()
    x = a.copy()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        if K_sel >= total_lags:
            binding_idx = np.arange(total_lags)
        else:
            binding_idx = np.argsort(M_curr - ac)[:K_sel]

        K = len(binding_idx)
        T_sel = np.zeros((K, n))
        for row in range(K):
            ki = int(binding_idx[row])
            js = max(0, ki - n + 1)
            je = min(n, ki + 1)
            if js < je:
                cols = np.arange(js, je)
                T_sel[row, cols] = y[ki - cols]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 10 == 0:
            print("    LP iter=%d (%ds) J=%.6f K=%d" %
                  (t, int(time.monotonic() - t0), obj_from_M(n, best_m), K_sel))

        if K_sel < max_K and t % K_interval == 0:
            K_sel = min(K_sel + K_growth, max_K)

    return best_a, best_m


def peak_reduction(a0, time_limit, verbose=False, rng_seed=77):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 400
    n_peaks = 8

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_peaks]

        sens = np.zeros(n)
        for k in top_k:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            sens[idx] += a[ki - idx]

        improved = False
        for k in top_k[:3]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n, ki + 1)
            idx = np.arange(s, e)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            best_local = np.argmax(contrib)
            src = idx[best_local]
            src_partner = ki - idx[best_local]
            reduce_idx = src if sens[src] >= sens[src_partner] else src_partner
            if a[reduce_idx] < 1e-12:
                continue

            exclude = {int(reduce_idx)}
            candidates = np.array([d for d in range(n) if d not in exclude])
            order = candidates[np.argsort(sens[candidates])][:25]

            for dst in order:
                if time.monotonic() - t0 > time_limit:
                    break
                for frac in (0.01, 0.005, 0.02, 0.05, 0.03):
                    amt = a[reduce_idx] * frac
                    if amt < 1e-15:
                        break
                    at = a.copy()
                    at[reduce_idx] -= amt
                    at[dst] += amt
                    at = np.maximum(at, 0.0)
                    at = at / at.sum()
                    Mt = fast_autocorr(at).max()
                    if Mt < best_m - 1e-14:
                        a = at
                        best_m = Mt
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            for attempt in range(6):
                hi = np.argsort(sens)[::-1][:min(15, n)]
                ridx = int(hi[attempt % len(hi)])
                if a[ridx] < 1e-12:
                    continue
                dst = int(rng.integers(n))
                if dst == ridx:
                    continue
                frac = rng.uniform(0.005, 0.03)
                amt = a[ridx] * frac
                at = a.copy()
                at[ridx] -= amt
                at[dst] += amt
                at = np.maximum(at, 0.0)
                at = at / at.sum()
                Mt = fast_autocorr(at).max()
                if Mt < best_m - 1e-14:
                    a = at
                    best_m = Mt
                    best_a = a.copy()
                    stall = 0
                    break

        if verbose and stall % 80 == 0:
            print("    Peak (%ds) J=%.6f stall=%d" %
                  (int(time.monotonic() - t0), obj_from_M(n, best_m), stall))

    return best_a, best_m

    trial_lengths = []
    for delta in (700, 1200, -600, 1800):
        nn = cur_n + delta
        if 600 <= nn <= 6000:
            trial_lengths.append(nn)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 90:
            break
        print("  try n=%d" % new_n)
        a_long = interpolate_sequence(best_seq, new_n)
        t4a = min(remaining * 0.4, 200.0)
        if t4a > 30:
            a_out, _ = amd_optimize(a_long, t4a, top_k=70, beta=6.0, c0=4.0,
                                    restart_every=5000, verbose=False,
                                    rng_seed=new_n % 997)
            try_update(a_out, "amd_n%d" % new_n)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 4000:
            t4b = min(remaining * 0.4, 140.0)
            if t4b > 25:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t4b, solver, verbose=False,
                                          initial_K=200,
                                          max_K=min(1200, 2 * new_n - 1),
                                          K_growth=60, K_interval=15)
                try_update(a_out, "lp_n%d" % new_n)

    # ---- Phase 5: final LP + polish on best ----
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 5: final polish (remaining=%.0fs) ===" % remaining)
    if remaining > 40 and len(best_seq) <= 5000:
        t5a = min(remaining * 0.5, 250.0)
        if t5a > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t5a, solver, verbose=True,
                                      initial_K=400,
                                      max_K=min(2000, 2 * len(a_in) - 1),
                                      K_growth=100, K_interval=15)
            try_update(a_out, "lp_final")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        t5b = min(remaining * 0.5, 120.0)
        if t5b > 15:
            a_out, _ = peak_reduction(norm_sum1(best_seq), t5b, verbose=False, rng_seed=999)
            try_update(a_out, "peak_final")
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t5c = min(remaining * 0.5, 60.0)
        if t5c > 10:
            a_out, _ = coordinate_polish(norm_sum1(best_seq), t5c, verbose=False, rng_seed=789)
            try_update(a_out, "coord_final")

    # ---- final validation ----
    if best_seq is None:
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1((np.sin(np.pi * x) ** 2 + 0.3).clip(0, None))
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs
    print("\nFINAL: score=%.6f n=%d time=%.0fs" %
          (best_score, len(best_seq), time.monotonic() - t_global))
    return [float(xv) for xv in best_seq]
