def simplex_project(v):
    """Project v onto the probability simplex (x >= 0, sum x = 1)."""
    n = len(v)
    # Bisection on the Lagrange multiplier
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)
    rho = np.arange(1, n+1)
    cond = sorted_v - cssv/rho > 0
    rho_star = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho_star-1] - 1.0) / rho_star
    return np.maximum(v - theta, 0.0)

def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - cssv / rho > 0
    if not cond.any():
        return np.ones(n) / n
    rho_star = int(np.max(np.where(cond)[0])) + 1
    theta = (cssv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(v - theta, 0.0)

                     n_anneal_steps=2000, verbose=True):

    n = len(a0)
    x = norm_sum1(a0).copy()
    z = x.copy()
    t = 1.0

    best_true_obj = fast_autocorr(x).max()
    t0 = time.monotonic()
    
    L = 4.0 * n  # Lipschitz constant estimate (rough upper bound)

        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        progress = min(1.0, k / n_anneal_steps)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient at z
        g = smooth_max_grad_a(z, beta)
        
        # FISTA update
        y = z + ((t - 1) / (t + 2)) * (z - x)
        y = np.maximum(y, 1e-30)
        s = y.sum()
        if s < 1e-15:
            y = z.copy()
        else:
            y = y / s
        
        # Gradient step
        x_new = y - g / L
        x_new = simplex_project(x_new)
        
        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t * t))
        
        # Backtracking line search (simple version)
        obj_y = smooth_max_obj(y, beta)
        obj_xnew = smooth_max_obj(x_new, beta)
        if obj_xnew > obj_y + 1e-15:
            # Step was too large, try smaller
            for _ in range(10):
                L *= 2.0
                x_new = y - g / L
                x_new = simplex_project(x_new)
                obj_xnew = smooth_max_obj(x_new, beta)
                if obj_xnew <= obj_y:
                    break
        
        x = x_new
        z = x
        t = t_new
        
        # Track best
        if k % 5 == 0:
            true_obj = fast_autocorr(x).max()
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = x.copy()
        
        if verbose and k % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_true_obj
            print(f"    FISTA k={k} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_true_obj
