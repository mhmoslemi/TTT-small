"""
Minimizes  2*n*max(conv(a,a)) / (sum(a))^2

Three-phase strategy (deliberately different from the prior SOTA):
  1. Projected subgradient descent on the simplex using the EXACT max of the
     autocorrelation (no soft approximation).  The subgradient of max_k b_k is
     the averaged gradient over all (near-)maximal lags, which is just a shifted
     copy of a.  Wrapped in Nesterov extrapolation + backtracking line search
     + periodic perturb-restart to escape flat local minima.
  2. Active-set LP refinement: the lags that sit at the peak define binding
     first-order constraints; an LP picks the best small decrement direction.
  3. Random pair-swap local search finishing polish.

All helpers are top-level (no closures / no lambdas).  Uses FFT for convolutions.
"""

import numpy as np
import time
import cvxpy as cp


# ------------------------- core utilities -------------------------

def fast_autocorr(a):
    """conv(a, a) via FFT, returned as 2n-1 real values, clipped >= 0."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    out += 0.0
    np.maximum(out, 0.0, out=out)
    return out


def objective_moment(a):
    """max of conv(a,a) for a zero-padded/normalized vector (we pass sum=1)."""
    return float(fast_autocorr(a).max())


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float("inf")


def proj_simplex(v):
    """Euclidean projection of v onto {x>=0, sum x = 1} (1D Medik-Sort)."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    ok = u * idx > (cssv - 1.0)
    rho = int(np.where(ok)[0][-1]) if np.any(ok) else 0
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgrad_for_lag(a, k):
    """d/da of conv(a,a)[k]  ==  2 * a[k-i] over valid i, zero elsewhere."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    k = int(k)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        g[i_start:i_end + 1] = 2.0 * a[k - i_end:k - i_start - 1:-1]
    return g


def max_lag_subgrad(a, tol_rel=1e-4):
    """Averaged subgradient over all lags within tol_rel of the peak."""
    a = np.asarray(a, dtype=np.float64)
    b = fast_autocorr(a)
    M = float(b.max())
    thresh = M * (1.0 - tol_rel)
    idx = np.where(b >= thresh)[0]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(b))])
    g = np.zeros(len(a), dtype=np.float64)
    for k in idx:
        g += subgrad_for_lag(a, int(k))
    g /= float(len(idx))
    # subtract mean so step is sum-neutral before projection (helps line search)
    return g, M, len(idx)


# ------------------------- initializers -------------------------

def make_init(n, seed=0, kind="smooth"):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "cosine":
        a = 0.6 + 0.4 * np.cos(2.0 * np.pi * x)
    elif kind == "edge":
        a = np.ones(n)
        a[: n // 12] = 0.25
        a[-(n // 12):] = 0.25
    else:  # smooth multi-harmonic
        a = np.ones(n) * 0.5
        for f in range(1, 9):
            a = a + rng.normal(0, 0.04) * np.sin(2.0 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.0)
    return proj_simplex(a)


def symmetrize(a):
    a = np.asarray(a, dtype=np.float64)
    return proj_simplex(0.5 * (a + a[::-1]))


# ------------------------- phase 1: subgradient -------------------------

def sgd_optimize(a0, time_limit, seed=0, verbose=True):
    n = len(a0)
    a = proj_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    iters = 0
    stall = 0
    best_eta = 1.0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        beta_t = (iters - 1.0) / (iters + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-14)
        sy = y.sum()
        y = y / sy if sy > 1e-14 else x.copy()

        g, M, nlag = max_lag_subgrad(y, tol_rel=1e-4 * min(8, iters))
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-30:
            break

        eta = best_eta
        chosen = None
        for _ in range(24):
            cand = proj_simplex(y - eta * g)
            Mc = fast_autocorr(cand).max()
            if Mc < M:
                chosen = cand
                chosen_M = Mc
                best_eta = eta
                break
            eta *= 0.6
            best_eta = min(best_eta, eta * 1.6)

        if chosen is None:
            chosen = proj_simplex(y - eta * g)
            chosen_M = fast_autocorr(chosen).max()

        x_prev = x
        x = chosen

        if chosen_M < best_M - 1e-15:
            best_M = chosen_M
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall >= 1200:
            # perturb-restart around the best to escape a flat basin
            scale = 0.004 + 0.004 * (rng.random() < 0.3)
            xr = best_a + rng.normal(0, scale, n)
            xr = np.maximum(xr, 0.0)
            xr = xr / xr.sum()
            x = xr
            x_prev = xr.copy()
            stall = 0

        if verbose and iters % 700 == 0:
            J = 2.0 * n * best_M
            print("    SGD iter=%d (%.0fs) J=%.6f active=%d eta=%.3e" %
                  (iters, time.monotonic() - t0, J, nlag, best_eta))

    return best_a, best_M

_LP_SOLVER = None


def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    for s in ["CLARABEL", "ECOS", "SCS", "PDLP"]:
        try:
            v = cp.Variable(6)
            pr = cp.Problem(cp.Minimize(cp.sum(v)), [cp.sum(v) == 1, v >= 0])
            pr.solve(solver=s, verbose=False)
            if pr.status in ("optimal", "optimal_inaccurate"):
                _LP_SOLVER = s
                print("  LP solver ->", s)
                return s
        except Exception:
            continue
    _LP_SOLVER = "ECOS"
    print("  LP solver -> ECOS (fallback)")
    return _LP_SOLVER


def lp_refine(a0, solver, time_limit, verbose=True):
    n = len(a0)
    a = proj_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    iters = 0
    stall = 0
    rho = 0.008

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b = fast_autocorr(a)
        M = float(b.max())
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 250:
            break

        thresh = M * (1.0 - 1e-4)
        active = np.where(b >= thresh)[0]
        if len(active) > 60:
            active = active[np.argsort(-b[active])][:60]

        delta = cp.Variable(n)
        tv = cp.Variable()
        cons = [
            cp.sum(delta) == 0,
            delta >= -a + 1e-12,
            delta <= rho,
            -delta <= rho,
        ]
        for k in active:
            ki = int(k)
            gk = subgrad_for_lag(a, ki)
            cons.append(gk @ delta <= tv - b[ki])

        pr = cp.Problem(cp.Minimize(tv), cons)
        try:
            pr.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if pr.status not in ("optimal", "optimal_inaccurate"):
            continue
        dv = np.asarray(delta.value, dtype=np.float64)
        dv = np.clip(dv, -a + 1e-12, 1e9)
        dv = np.maximum(dv, -a + 1e-12)
        dv -= dv.mean()

        step_taken = False
        for frac in [1.0, 0.5, 0.2, 0.05]:
            trial = a + frac * dv
            if trial.min() < 0.0 or trial.max() < 0.0:
                continue
            trial = np.maximum(trial, 0.0)
            sts = trial.sum()
            if sts < 1e-12:
                continue
            trial = trial / sts
            Mt = fast_autocorr(trial).max()
            if Mt < M - 1e-15:
                a = trial
                step_taken = True
                break
        if not step_taken:
            stall += 50

        if verbose and iters % 15 == 0:
            print("    LP iter=%d (%.0fs) J=%.6f active=%d" %
                  (iters, time.monotonic() - t0, 2.0 * n * best_M, len(active)))

    return best_a, best_M


# ------------------------- phase 3: local search -------------------------

def local_search(a0, time_limit, seed=0, verbose=True):
    n = len(a0)
    a = proj_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    iters = 0
    stall = 0
    scales = (1e-3, 2e-4, 5e-4, 1e-2, 3e-4, 1e-4)

    while time.monotonic() - t0 < time_limit and stall < 8000:
        iters += 1
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        base = min(a[i], a[j])
        if base < 1e-9:
            stall += 1
            continue
        best_d = 0.0
        best_lm = best_M
        for sc in scales:
            for sgn in (1.0, -1.0):
                d = sgn * sc * base
                if a[i] - d < 0 or a[j] + d < 0:
                    continue
                a[i] -= d
                a[j] += d
                lm = fast_autocorr(a).max()
                if lm < best_lm:
                    best_lm = lm
                    best_d = d
                    best_keep = a.copy()
                a[i] += d
                a[j] -= d
        if best_d != 0.0:
            a = best_keep
            if best_lm < best_M - 1e-15:
                best_M = best_lm
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if verbose and iters % 600 == 0:
            print("    LS iter=%d (%.0fs) J=%.6f" %
                  (iters, time.monotonic() - t0, 2.0 * n * best_M))

    return best_a, best_M


# ------------------------- candidate bookkeeping (no closure) -------------------------

def record_if_better(best_seq, best_score, a, label, t0):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        print("  *** NEW BEST [%s]: %.6f (n=%d, t=%.0fs)" % (label, sc, len(a), time.monotonic() - t0))
        return a.copy(), sc
    return best_seq, best_score


def try_each_init(best_seq, best_score, n, t0, time_budget, seed):
    """Give a couple of diverse structured inits a short SGD shot."""
    for kind, sd in (("cosine", seed + 1), ("edge", seed + 2), ("smooth", seed + 3)):
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 90:
            break
        t = min(remaining * 0.2, 240.0)
        if t < 30:
            break
        init = make_init(n, seed=sd, kind=kind)
        out, _ = sgd_optimize(init, t, seed=sd, verbose=False)
        best_seq, best_score = record_if_better(best_seq, best_score, out, "init_" + kind, t0)
    return best_seq, best_score
