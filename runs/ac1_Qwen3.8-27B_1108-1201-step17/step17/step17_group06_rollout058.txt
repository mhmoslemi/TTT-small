def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def cross_correlate(w, a):
    """Compute cross-correlation: result[i] = sum_k w[k]*a[k-i] for valid k,i.
    w has length 2n-1, a has length n, result has length n.
    """
    n = len(a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    Wf = np.fft.fft(w, N)
    Af = np.fft.fft(a, N)
    corr = np.fft.ifft(Wf * np.conj(Af)).real[:n]
    return corr


def compute_lse_grad(a, beta):
    """Compute f_beta(a) = beta*logsumexp(conv/beta) and gradient w.r.t. a.
    Returns (f_beta_value, grad_a).

    corr = cross_correlate(w, a)

        # Compute soft subgradient
        conv = fast_conv(y)
        top_idx = np.argsort(conv)[::-1][:top_k]
        top_vals = np.maximum(conv[top_idx], 1e-30)
        w = top_vals ** beta_w
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w = w / w_sum

        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                g[idx] += 2.0 * wi * y[k_int - idx]

        # Step size
        eta = c0 / (np.sqrt(t) + 5.0)

        # Mirror step (projected gradient in log-space = multiplicative update)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-30)
        x /= x.sum()

        # Track best
        val = eval_J(a_new)
        if val < best_J - 1e-15:
            best_J = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Restart logic
        if stall > 2000:
            stall = 0
            scale = rng.uniform(0.003, 0.02)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            c0 = max(c0 * 0.85, 0.5)

        if verbose and t % 5000 == 0:
            print(f"    SG t={t} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=80.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.1) ** 2) + np.exp(-width * (x - 0.9) ** 2) + 0.005
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 12):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def cosine_taper(n, alpha=0.5):
    """Sequence with cosine taper at edges."""
    a = np.ones(n)
    m = max(1, int(n * alpha))
    for i in range(min(m, n)):
        taper = 0.5 * (1 + np.cos(np.pi * i / m))
        a[i] = taper
        a[n - 1 - i] = taper
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_gaussian_center(n, sigma_frac=0.3):
    """Gaussian centered in the middle."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-x ** 2 / (2 * sigma_frac ** 2))
    return a / a.sum()


def make_edges(n, power=2.0):
    """Mass concentrated at edges."""
    x = np.linspace(0, 1, n)
    a = x ** power + (1 - x) ** power + 0.01
    return a / a.sum()


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver combining multiple strategies."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize best
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3000
        best_a = make_power_ramp(n0, 1.0)
        best_score = eval_J(best_a)
        print(f"Initial: n={len(best_a)}, J={best_score:.6f}")

    n0 = len(best_a)
    print(f"Starting optimization: n={n0}, J={best_score:.6f}, budget={time_budget:.0f}s")

    # === Phase 1: LSE continuation on parent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE Continuation on Parent (n={n0}) ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_out, J_out = continuation_lse(best_a.copy(), t1,
                                        beta_schedule=[0.01, 0.05, 0.2, 1.0, 5.0, 20.0, 100.0, 500.0])
        try_update(a_out, "lse_parent")

    # === Phase 2: Multi-start LSE continuation ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Multi-Start LSE (n={n0}) ===")

    inits = []
    labels = []
    inits.append(make_power_ramp(n0, 0.3))
    labels.append("ramp03")
    inits.append(make_power_ramp(n0, 0.7))
    labels.append("ramp07")
    inits.append(make_power_ramp(n0, 1.5))
    labels.append("ramp15")
    inits.append(make_bimodal(n0))
    labels.append("bi")
    inits.append(make_smooth_random(n0, seed=3))
    labels.append("sm3")
    inits.append(make_smooth_random(n0, seed=42))
    labels.append("sm42")
    inits.append(cosine_taper(n0, 0.3))
    labels.append("cos03")
    inits.append(cosine_taper(n0, 0.6))
    labels.append("cos06")
    inits.append(make_gaussian_center(n0, 0.3))
    labels.append("gauss")
    inits.append(make_edges(n0, 3.0))
    labels.append("edges")

    # Blends with parent
    if parent is not None:
        for frac in [0.3, 0.5, 0.7]:
            blend = frac * parent + (1 - frac) * make_power_ramp(n0, 1.0)
            inits.append(blend / blend.sum())
            labels.append(f"bl{int(frac*10)}")

    # Deduplicate
    seen = set()
    u_inits = []
    u_labels = []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n0 // 50)], 2))
        if sig not in seen:
            seen.add(sig)
            u_inits.append(init)
            u_labels.append(lbl)

    t_per = max(8.0, remaining * 0.30 / len(u_inits))
    for idx, (init, lbl) in enumerate(zip(u_inits, u_labels)):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_avail = min(t_per, remaining - 20)
        if t_avail < 8:
            break
        print(f"  LSE [{lbl}] ({t_avail:.0f}s)...")
        a_out, _ = continuation_lse(init, t_avail,
                                     beta_schedule=[0.05, 0.5, 5.0, 50.0])
        try_update(a_out, f"lse_{lbl}")

    # === Phase 3: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Subgradient Polish (n={len(best_a)}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_out, _ = subgradient_polish(best_a.copy(), t3, top_k=60, beta_w=6.0)
        try_update(a_out, "sg_polish")

    # === Phase 4: High-beta LSE on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: High-Beta LSE (n={len(best_a)}) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 20:
        a_out, _ = continuation_lse(best_a.copy(), t4,
                                     beta_schedule=[20.0, 100.0, 500.0, 2000.0, 10000.0])
        try_update(a_out, "high_beta")

    # === Phase 5: p-norm optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: p-norm Opt (n={len(best_a)}) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        for p_val in [50, 200, 1000]:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            t_avail = min(remaining * 0.3, 40.0)
            if t_avail < 8:
                break
            print(f"  p={p_val} ({t_avail:.0f}s)...")
            a_out, _, _ = optimize_pnorm(best_a.copy(), p_val, t_avail, max_iter=100)
            try_update(a_out, f"p{p_val}")

    # === Phase 6: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length Sweep ===")
    n_best = len(best_a)
    lengths_to_try = []
    for d in [-500, -300, -100, 100, 300, 500, 800, 1200]:
        nl = n_best + d
        if 1500 <= nl <= 7000:
            lengths_to_try.append(nl)

    for nl in sorted(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if abs(nl - n_best) <= 50:
            continue
        print(f"  Length: n={nl}...")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.30, 100.0)
        if t_md > 15:
            a_out, _ = continuation_lse(a_int, t_md,
                                         beta_schedule=[0.1, 1.0, 10.0, 100.0])
            try_update(a_out, f"len{nl}")

    # === Phase 7: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 7: Final Polish (n={n_cur}) ===")

    # LSE with very high beta
    t7a = min(remaining * 0.30, 200.0)
    if t7a > 15:
        a_out, _ = continuation_lse(best_a.copy(), t7a,
                                     beta_schedule=[50.0, 500.0, 5000.0, 50000.0])
        try_update(a_out, "final_lse")

    # Subgradient
    remaining = time_budget - (time.monotonic() - t_global)
    t7b = min(remaining * 0.30, 150.0)
    if t7b > 15:
        a_out, _ = subgradient_polish(best_a.copy(), t7b, top_k=80, beta_w=4.0)
        try_update(a_out, "final_sg")

    # One more LSE
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7c = min(remaining * 0.5, 150.0)
        if t7c > 10:
            a_out, _ = continuation_lse(best_a.copy(), t7c,
                                         beta_schedule=[100.0, 1000.0, 10000.0])
            try_update(a_out, "extra_lse")

    # Final output
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        result = [float(x) for x in best_a]
        return result
    else:
        n_fb = 3000
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
