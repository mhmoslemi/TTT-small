```python
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
```

```python
def hankel_mv(w, x):
    """Compute W @ x where W[i,j] = w[i+j], in O(n log n)."""
    n = len(x)
    # b = x reversed
    b = x[::-1]
    # Cross-correlation: (w ⊛ b)_m = sum_k w[m-k] * b[k]
    # We need (w ⊛ b)[n-1 : 2n-1]
    size = len(w) + n - 1  # = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    Wf = np.fft.rfft(w, fft_size)
    Bf = np.fft.rfft(b, fft_size)
    # Cross-correlation via FFT: ifft(Wf * conj(Bf))
    corr = np.fft.irfft(Wf * np.conj(Bf), fft_size)
    return corr[n-1:n-1+n]
```

```python
def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def hankel_mv(w, x):
    """Compute W @ x where W[i,j] = w[i+j], using FFT cross-correlation trick. O(n log n)."""
    n = len(x)
    b = x[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    Wf = np.fft.rfft(w, fft_size)
    Bf = np.fft.rfft(b, fft_size)
    corr = np.fft.irfft(Wf * np.conj(Bf), fft_size)
    return corr[n - 1:n - 1 + n]

def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)

def mm_projected_gradient(a0, time_limit, beta_start, beta_end, n_inner, verbose=True):
    """MM algorithm with projected gradient inner loop and temperature annealing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    total_iters = 5000
    x_mom = a.copy()
    prev_grad_norm = None

    for outer_t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal temperature
        frac = outer_t / total_iters
        beta = beta_start * (beta_end / beta_start) ** frac

        ac = fast_autocorr(a)
        ac_clipped = np.maximum(ac, 1e-30)
        w = ac_clipped ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = w / w_sum

        # Inner loop: projected gradient on a^T W a

        lr = 1.0 / (2.0 * np.max(np.abs(hankel_mv(w, a_inner))))
        if lr < 1e-10:

            grad = 2.0 * hankel_mv(w, a_inner)
            a_trial = a_inner - lr * grad
            a_trial = project_simplex(a_trial)
            # Simple line search (accept if objective decreases)

            obj_old = np.dot(w, fast_autocorr(a_inner))

                a_inner = a_trial
            else:
                lr *= 0.5
                if lr < 1e-8:

        # Nesterov momentum on outer loop

            x_mom = a_inner.copy()
        else:
            beta_mom = (outer_t - 1) / (outer_t + 1.5)

            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = project_simplex(y_new)

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = project_simplex(x_new)

        y = y_new
        x = x_new

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-CW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-CW done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def coordinate_polish(a0, time_limit, rng_seed, verbose=True):
    """Coordinate descent: move mass from peak-contributing positions to low-impact ones."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 600

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        improved = False
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score all positions by their contribution to top lags
            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            cand_scores = scores[candidates]
            dst = candidates[np.argmin(cand_scores)]

            if dst == src:
                continue

            found = False
            for frac in [0.01, 0.02, 0.04, 0.07, 0.12, 0.18]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = project_simplex(a_trial)
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 100:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = project_simplex(a)
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.025)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = project_simplex(a)
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 150 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m

def parametric_search(n, time_limit, best_init, best_val):
    """Search over parametric families for a good initial sequence."""
    t0 = time.monotonic()
    i_arr = np.arange(n, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    best_a = best_init.copy()
    best_score = best_val

    def try_update_local(a, label):
        nonlocal best_a, best_score
        a_n = norm_sum1(a)
        ac = fast_autocorr(a_n)
        val = 2.0 * n * ac.max()
        if val < best_score:
            best_score = val
            best_a = a_n.copy()
            elapsed = time.monotonic() - t0
            print(f"    Param [{label}]: J={val:.6f} (t={elapsed:.0f}s)")

    for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        a = (i_arr + 1.0) ** p
        try_update_local(a, f"pow_{p}")

    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        a = (n - i_arr) ** p
        try_update_local(a, f"ipow_{p}")

    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.5, 1.0, 2.0]:
            a = np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta
            try_update_local(a, f"beta_{alpha}_{beta}")

    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3, 0.5]:
            a = np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma))
            try_update_local(a, f"gc{c:0.1f}s{sigma:0.1f}")

    for c1 in [0.2, 0.3]:
        for c2 in [0.7, 0.8]:
            for sigma in [0.05, 0.1, 0.15]:
                a = np.exp(-((x - c1) ** 2) / (2.0 * sigma * sigma)) + np.exp(-((x - c2) ** 2) / (2.0 * sigma * sigma))
                try_update_local(a, f"bi{c1:0.1f}_{c2:0.1f}_{sigma:0.2f}")

    for c in [0.3, 0.5, 0.7]:
        for s in [5, 10, 20, 50]:
            a = 1.0 / (1.0 + np.exp(-s * (x - c)))
            try_update_local(a, f"sg{c:0.1f}_{s}")

    for rate in [1, 2, 5, 10]:
        a = np.exp(-rate * x)
        try_update_local(a, f"er{rate}")

    if time.monotonic() - t0 > time_limit * 0.8:
        break

    elapsed = time.monotonic() - t0
    print(f"  Parametric done ({elapsed:.0f}s): best J={best_score:.6f}")
    return best_a, best_score

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with MM, LP-FW, and coordinate descent."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        i_arr = np.arange(n_base, dtype=np.float64)
        # Try a few parametric inits
        candidates = [
            np.sin(np.pi * x) ** 2,
            (i_arr + 1) ** 1.0,
            (i_arr + 1) ** 0.5 * (n_base - i_arr) ** 0.5,
        ]
        for c in candidates:
            c_n = norm_sum1(c)
            sc = 2.0 * n_base * fast_autocorr(c_n).max()
            if sc < best_score:
                best_score = sc
                best_seq = c_n
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    def check_and_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        ac = fast_autocorr(a_norm)
        sc = 2.0 * len(a_norm) * ac.max()
        # Also verify with actual evaluate
        sc_actual = evaluate_safe(a_norm)
        if np.isfinite(sc_actual) and sc_actual < best_score:
            best_score = sc_actual
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {best_score:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")
        elif np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {best_score:.6f} (n={len(a_norm)}, t={elapsed:.0f}s)")

    # === Phase 1: Brief parametric search ===
    print(f"\n=== Phase 1: Parametric search (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 50.0)
    if t1 > 10:
        a_param, val_param = parametric_search(n0, t1, best_seq, 2.0 * n0 * fast_autocorr(best_seq).max())
        check_and_update(a_param, "parametric")

    # === Phase 2: MM algorithm (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: MM with projected gradient (n={n_best}) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mm_projected_gradient(a_in, t2, beta_start=8.0, beta_end=2.0,
                                         n_inner=8, verbose=True)
        check_and_update(a_out, f"mm_n{n_best}")

    # === Phase 3: Nesterov subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Nesterov subgradient (n={n_best}) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_subgradient(a_in, t3, top_k=80, beta=5.0, c0=3.0, verbose=False)
        check_and_update(a_out, f"nsg_n{n_best}")

    # === Phase 4: LP-FW (aggressive) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW aggressive (n={n_best}) ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 30 and n_best <= 6000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, initial_K=400, max_K=5000,
                                   add_per_iter=50, verbose=True)
        check_and_update(a_out, f"lp_n{n_best}")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate descent (n={n_best}) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, rng_seed=77, verbose=True)
        check_and_update(a_out, f"coord_n{n_best}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = [n_best + 500, n_best + 1000, n_best + 1500]
    longer_lengths = [l for l in longer_lengths if l <= 7000]

    for new_n in longer_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n=== Phase 6: Scale to n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t6a = min(remaining * 0.35, 150.0)
        if t6a > 30:
            a_out, _ = mm_projected_gradient(a_long, t6a, beta_start=6.0, beta_end=2.5,
                                             n_inner=6, verbose=False)
            check_and_update(a_out, f"mm_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 5500:
            t6b = min(remaining * 0.40, 180.0)
            if t6b > 30:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, initial_K=300,
                                           max_K=4000, add_per_iter=40, verbose=True)
                check_and_update(a_out, f"lp_n{new_n}")

    # === Phase 7: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}) ===")
    if remaining > 60 and n_best <= 6000:
        t7 = min(remaining * 0.40, 250.0)
        if t7 > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t7, solver, initial_K=500,
                                       max_K=6000, add_per_iter=60, verbose=True)
            check_and_update(a_out, f"lp_final_n{n_best}")

    # === Phase 8: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord polish (n={n_best}) ===")
    if remaining > 30:
        t8 = min(remaining * 0.50, 120.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, rng_seed=314, verbose=True)
            check_and_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
