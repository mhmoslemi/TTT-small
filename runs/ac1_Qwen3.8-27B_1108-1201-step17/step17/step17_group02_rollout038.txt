    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_init(n, n_modes=20, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(freq))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def make_bimodal(n, width=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.2) / width) ** 2) + np.exp(-((x - 0.8) / width) ** 2) + 0.003
    return norm_sum1(a)


def make_skew(n, skew_power=1.5):
    x = np.linspace(0, 1, n)
    a = x ** skew_power + (1 - x) ** (2 * skew_power) + 0.01
    return norm_sum1(a)


def flatten_sequence(a, amount=0.3):
    """Compress dynamic range: a_new = (1-amount)*a + amount*mean(a)."""
    a = norm_sum1(a)
    mean_a = np.ones_like(a) / len(a)
    a_new = (1 - amount) * a + amount * mean_a
    return norm_sum1(a_new)


def amplified_sequence(a, amount=0.3):
    """Increase dynamic range: boost peaks, suppress valleys."""
    a = norm_sum1(a)
    a_max = a.max()
    a_min = a.min()
    if a_max - a_min < 1e-15:
        return a
    a_norm = (a - a_min) / (a_max - a_min)
    a_new = a_min + (a_max - a_min) * (a_norm ** (1.0 / (1 + amount)))
    return norm_sum1(a_new)


def mirror_descent_optimize(a0, time_limit, top_k_init=15, beta_init=3.0,
                            alpha_init=0.7, c0=5.0, restart_every=5000,
                            use_var_grad=True, verbose=False, rng_seed=42):
    """Nesterov mirror descent with combined peak+variance gradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k_init
    cur_beta = beta_init
    cur_alpha = alpha_init
    iter_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Compute gradients
        ac = fast_autocorr(y)
        if use_var_grad:
            g = combined_gradient(y, ac, top_k=cur_top_k, beta=cur_beta, alpha=cur_alpha)
        else:
            g = peak_gradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        # Mirror descent update
        eta = gamma / (np.sqrt(t) + 1.0)
        # Clip gradient to avoid numerical issues
        g_max = np.max(np.abs(g))
        if g_max > 1.0:
            g = g / g_max * 1.0
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new
        iter_count += 1

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive parameter adjustment
        if stall_count > restart_every // 4:
            cur_beta = max(cur_beta * 0.95, 1.5)
            cur_top_k = min(cur_top_k + 3, 200)
            if use_var_grad:
                cur_alpha = max(cur_alpha * 0.97, 0.2)

        if stall_count > restart_every:
            stall_count = 0
            # Structured perturbation: smooth sinusoidal perturbation
            n_pert = int(rng.integers(3, min(20, n // 10) + 1))
            freqs = rng.choice(np.arange(1, n // 2 + 1), size=n_pert, replace=False)
            phases = rng.uniform(0, 2 * np.pi, n_pert)
            amps = rng.uniform(0.01, 0.05, n_pert)
            pert = np.zeros(n)
            for f, p, a_p in zip(freqs, phases, amps):
                pert += a_p * np.sin(2 * np.pi * f * np.arange(n) / n + p)
            x_r = best_a + pert * best_a
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 200)
            cur_alpha = max(cur_alpha - 0.1, 0.2)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"      MD restart t={t} ({elapsed:.0f}s): J={J:.6f}, top_k={cur_top_k}, alpha={cur_alpha:.2f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      MD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Targeted coordinate descent: move mass from peak-contributing positions to cold positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 100
    n_top = 10

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        mean_ac = ac.mean()

        # Build sensitivity map
        sens = np.zeros(n)
        cold_score = np.zeros(n)
        for k in top_k[:6]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            sens[idx] += a[partner]
            # Cold score: how much does position contribute to sub-average lags?
            for i in idx:
                ki2 = int(i)
                for j in partner:
                    pass  # Simplified: just use sens

        # Cold positions: low sensitivity to peaks
        cold_threshold = np.percentile(sens, 30)
        cold_positions = np.where(sens <= cold_threshold)[0]
        if len(cold_positions) == 0:
            cold_positions = np.arange(n)

        improved = False
        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            # Top 3 most responsible pairs
            order = np.argsort(contrib)[::-1][:3]
            for oi in order:
                if time.monotonic() - t0 > time_limit:
                    break
                src = idx[oi]
                src_partner = partner[oi]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Choose which to reduce
                imp_src = sens[src]
                imp_partner = sens[src_partner]
                if imp_src >= imp_partner:
                    reduce_idx = src
                else:
                    reduce_idx = src_partner

                if a[reduce_idx] < 1e-12:
                    continue

                # Best destination: coldest position
                exclude = {src, src_partner}
                valid_cold = [c for c in cold_positions if c not in exclude]
                if not valid_cold:
                    valid_cold = [c for c in range(n) if c not in exclude]
                if not valid_cold:
                    continue
                dst = valid_cold[np.argmin(sens[valid_cold])]

                found = False
                for frac in [0.005, 0.01, 0.02, 0.002, 0.03, 0.05]:
                    amt = a[reduce_idx] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[reduce_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        found = True
                        break

                if found:
                    break

        if not improved:
            stall += 1
            # Random perturbation
            if stall > 20:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"      Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Primary MD optimization on parent ===
    print(f"\n=== Phase 1: MD with variance gradient (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.30, 450.0)
    if t_phase1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(
            a_in, t_phase1, top_k_init=20, beta_init=3.0, alpha_init=0.6,
            c0=5.0, restart_every=4000, use_var_grad=True,
            verbose=True, rng_seed=42
        )
        try_update(a_out, "md_parent")

    # === Phase 2: MD with more aggressive peak focus ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MD peak-focused (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t_phase2 = min(remaining * 0.20, 300.0)
    if t_phase2 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(
            a_in, t_phase2, top_k_init=10, beta_init=6.0, alpha_init=0.9,
            c0=4.0, restart_every=6000, use_var_grad=True,
            verbose=False, rng_seed=100
        )
        try_update(a_out, "md_peak")

    # === Phase 3: MD from diverse initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Multi-init MD (n={n0}) ===")
    t_phase3 = min(remaining * 0.20, 300.0)
    if t_phase3 > 60:
        inits = []
        labels = []

        # Flattened parent
        if parent is not None:
            inits.append(flatten_sequence(parent, 0.2))
            labels.append("flatten0.2")
            inits.append(flatten_sequence(parent, 0.4))
            labels.append("flatten0.4")
            inits.append(amplified_sequence(parent, 0.3))
            labels.append("amp0.3")

        # U-shape variants
        inits.append(make_u_shape(n0, 1.0))
        labels.append("u1.0")
        inits.append(make_u_shape(n0, 3.0))
        labels.append("u3.0")

        # Skew
        inits.append(make_skew(n0, 1.5))
        labels.append("skew1.5")
        inits.append(make_skew(n0, 2.5))
        labels.append("skew2.5")

        # Bimodal
        inits.append(make_bimodal(n0, 0.2))
        labels.append("bimodal0.2")

        # Spectral
        inits.append(spectral_init(n0, 20, seed=77))
        labels.append("spectral77")

        # Blend parent with u-shape
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_u_shape(n0, 2.0)
            inits.append(norm_sum1(blend))
            labels.append("blend")

        # Deduplicate
        seen = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 4))
            if sig not in seen:
                seen.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels

        t_per = max(20.0, t_phase3 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  MD init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = mirror_descent_optimize(
                init, t_avail, top_k_init=15, beta_init=4.0, alpha_init=0.5,
                c0=5.0, restart_every=3000, use_var_grad=True,
                verbose=False, rng_seed=200 + idx
            )
            try_update(a_out, f"md_{lbl}")

    # === Phase 4: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coordinate polish (n={n_best}) ===")
    t_phase4 = min(remaining * 0.12, 180.0)
    if t_phase4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t_phase4, verbose=True, rng_seed=555)
        try_update(a_out, "coord")

    # === Phase 5: Second MD pass with updated parameters ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: MD round 2 (n={n_best}) ===")
    t_phase5 = min(remaining * 0.12, 180.0)
    if t_phase5 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = mirror_descent_optimize(
            a_in, t_phase5, top_k_init=40, beta_init=5.0, alpha_init=0.4,
            c0=3.0, restart_every=8000, use_var_grad=True,
            verbose=False, rng_seed=300
        )
        try_update(a_out, "md_r2")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    trial_lengths = []
    for delta in [600, -600, 1200, -1200, 2000, -2000]:
        new_n = n_best + delta
        if 1000 <= new_n <= 8000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 80]

    for new_n in trial_lengths[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n} (remaining={remaining:.0f}s)...")
        a_long = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.30, 200.0)
        if t6a > 30:
            a_out, _ = mirror_descent_optimize(
                a_long, t6a, top_k_init=20, beta_init=4.0, alpha_init=0.5,
                c0=5.0, restart_every=4000, use_var_grad=True,
                verbose=False, rng_seed=new_n % 1000
            )
            try_update(a_out, f"md_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.30, 100.0)
            if t6b > 15:
                a_in = interpolate_sequence(best_seq, new_n)
                a_out, _ = coordinate_polish(a_in, t6b, verbose=False, rng_seed=new_n % 500)
                try_update(a_out, f"coord_n{new_n}")

    # === Phase 7: Final intensive MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final MD (n={n_best}) ===")
    if remaining > 30:
        t7 = min(remaining * 0.40, 200.0)
        if t7 > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = mirror_descent_optimize(
                a_in, t7, top_k_init=50, beta_init=4.0, alpha_init=0.35,
                c0=2.5, restart_every=10000, use_var_grad=True,
                verbose=True, rng_seed=999
            )
            try_update(a_out, "md_final")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 8: Final coord polish (n={n_best}) ===")
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = coordinate_polish(a_in, t8, verbose=False, rng_seed=777)
            try_update(a_out, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
