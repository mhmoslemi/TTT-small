            cand_score = _objective(rot_candidate)
            if cand_score < best_score - 5e-4:
                best_score = cand_score
                a_current = rot_candidate / np.sum(rot_candidate)
                new_conv = np.convolve(a_current, a_current, mode='full')
                if np.max(new_conv) < best_all_conv:
                    best_all_conv = np.max(new_conv)
                return rand_rot, cand_score
            stall_count += 1
            if stall_count >= stall_limit:
                break
        return test_type, best_score
    
def de_optimize(initial_sequence, time_limit=120, n_particles=100, max_velocity=0.9, crossover=0.7):
    """Differential evolution optimizer with early termination."""
    n = len(initial_sequence)
    # Check for validity
    total = np.sum(initial_sequence)
    if total < 1e-10:
        return initial_sequence, evaluate_sequence(initial_sequence)
    bounds = np.array([[0, 1000] * n], dtype=float).reshape(-1, 2)
    # Define objective function
    def objective(x):
        scores = np.zeros(population_size)
        scores = [evaluate_safe(x[i*n:(i+1)*n]) for i in range(population_size)]
        return scores
    # Set up evolution home
    curr = np.array([initial_sequence] * population_size, dtype=float)
    next_x01 = np.random.rand(population_size, n)
    velocities = (np.random.rand(population_size, n) - 0.5) * max_velocity * (bounds[:, 1] - bounds[:, 0]) / np.sqrt(time_limit)
    # Clash protocol
    for _ in range(generations):
        # Ensure no infinite loops from bad initialization
        curr_valid = np.zeros(population_size)
        for i in range(population_size):
            a_seq = curr[i]
            for j in range(n):
                if a_seq[j] < 0 or a_seq[j] > 1000:
                    curr_valid[i] = float('inf')
                else:
                    try:
                        score = evaluate_safe(a_seq)
                        curr_valid[i] = score
                    except:
                        curr_valid[i] = float('inf')
        # Check time expiration
        elapsed = time.time() - start_time
        if elapsed > time_limit * (estop_multiplier):
            break
        # Apply crossover
        pos = crossover
        best_index = np.argmax(curr_valid)
        # Maintain constraints
        for i in range(population_size):
            idxs = np.random.choice(range(n_particles), size=n)
            dx = next_x01[idxs] - curr[idxs]
            curr[i] = curr[i] * next_x01[idxs]
            curr[i][idxs] += np.exp(-j) * dx
        # Renormalization and protection
        for i in range(population_size):
            alphas = np.maximum(curr[i], 0.0) / np.sum(curr[i])
            curr[i] = alphas.copy()
    return curr[best_index], curr_valid[best_index]
    
def interpolate_sequence(parent_seq, reduction_factor=0.7):
    """Interpolate the parent sequence for reduced length."""
    n_original = len(parent_seq)
    new_len = int(n_original * reduction_factor)
    new_len = max(1, new_len)  # Ensuring positive length
    # Create a grid of points for the new sequence
    new_x = np.linspace(0, 1, new_len + 1)[:-1] if parent_seq else np.linspace(0, 1, 2)[1:]
    # Use the original sequence as a basis for shape preservation
    base_vec = np.ones(new_len) / new_len if not parent_seq else parent_seq[:new_len]
    return base_vec
