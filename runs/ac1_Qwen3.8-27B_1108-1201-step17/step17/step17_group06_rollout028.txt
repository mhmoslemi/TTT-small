```python
def compute_grad_fft(w, a):
    """Compute g[j] = sum_k w[k] * a[|k-j|] using FFT."""
    n = len(a)
    # Build s_ext (even extension of a), length 2n-1
    # s_ext[i] = a[|i - (n-1)|] for i in [0, 2n-2]
    # i.e., s_ext = [a[n-1], a[n-2], ..., a[1], a[0], a[1], ..., a[n-1]]
    s_ext = np.concatenate([a[::-1], a[1:]])
    
    # convolve w (length 2n-1) with s_ext (length 2n-1)
    # result has length 4n-3
    conv_size = 4 * n - 3
    fft_size = 1 << (conv_size - 1).bit_length()
    
    W = np.fft.rfft(w, fft_size)
    S = np.fft.rfft(s_ext, fft_size)
    g_full = np.fft.irfft(W * S, fft_size)[:conv_size]
    
    # g[j] = g_full[j + n - 1] for j in [0, n-1]
    return g_full[n-1 : 2*n - 1]
```

```python
def adam_smooth(a0, time_limit, t_init=0.5, t_min=0.01, lr=0.1, decay=0.9995, seed=42):
    """Adam optimizer on log-parameters with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(a0, 1e-10)
    a = a / a.sum()
    theta = np.log(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    t_sm = t_init
    best_a = a.copy()
    best_J = eval_J(a)
    stall = 0
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        t_sm = max(t_min, t_init * (decay ** step))
        
        # Compute weights for soft-max
        conv = fast_conv(a)
        conv_max = conv.max()
        shifted = (conv - conv_max) / t_sm
        max_shifted = shifted.max()
        exp_s = np.exp(shifted - max_shifted)
        w = exp_s / exp_s.sum()
        
        # Compute gradient using FFT
        g_a = compute_grad_fft(w, a)
        
        # Convert to gradient w.r.t. theta
        a_dot_g = np.dot(a, g_a)
        g_theta = a * (g_a - a_dot_g)
        
        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * g_theta**2
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        
        # Adaptive learning rate
        lr_t = lr / (1 + 0.001 * step)
        theta -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Renormalize for stability
        theta -= theta.max()
        theta = np.clip(theta, -50, 0)
        
        a = np.exp(theta)
        a_sum = a.sum()
        if a_sum < 1e-10:
            a = best_a.copy()
            theta = np.log(a)
            m = np.zeros(n)
            v = np.zeros(n)
            continue
        a /= a_sum
        
        J = eval_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > 2000:
            stall = 0
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * a
            a_new = np.maximum(a + pert, 1e-10)
            a_new /= a_new.sum()
            J_new = eval_J(a_new)
            if J_new < best_J * 1.1:  # allow some degradation
                a = a_new
                theta = np.log(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_sm = max(t_sm * 2, t_min * 5)
            else:
                a = best_a.copy()
                theta = np.log(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_sm = max(t_sm * 2, t_min * 5)
        
        if step % 5000 == 0:
            print(f"    Adam step={step}, t_sm={t_sm:.4f}, J={best_J:.6f}")
    
    return best_a, best_J
```
