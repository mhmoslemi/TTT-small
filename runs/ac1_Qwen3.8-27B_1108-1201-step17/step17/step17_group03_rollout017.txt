def fast_autocorr(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def make_hann(n):
    x = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    return norm_sum1(w)


def make_blackman(n):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient(a, ac, top_k=100, temperature=3.0):
    """Compute a soft subgradient of max(ac) using softmax weighting."""
    n = len(a)
    # Use softmax weighting on the top values
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    
    # Softmax weighting (scaled by temperature)
    w = np.exp(temperature * (top_vals - top_vals[0]))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def exp_map(y, grad, eta):
    """Exponential map: mirror descent step"""
    log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                    margin_frac=1e-3, use_line_search=True):
    """LP Frank-Wolfe with adaptive active set and optional line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()  # Current iterate
    x = a.copy()  # Nesterov momentum point
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-18:
            break
        
        # Compute active set
        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()
        
        if n_active > top_K:
            active_idx = np.argsort(margins)[:top_K]
        elif n_active < 3:
            active_idx = np.argsort(margins)[:max(10, top_K // 10)]
        else:
            active_idx = np.where(active_mask)[0]
        
        K = len(active_idx)
        
        # Build constraint matrix T: T[k, i] = y[k-i] if valid
        # (y * b)_k = sum_i y[k-i] * b_i
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe step with Nesterov acceleration
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        # Optional line search along the direction
        if use_line_search:
            # Try a few step sizes
            best_alpha_ls = alpha
            best_m_ls = fast_autocorr(y_new).max()
            for ls_alpha in [alpha * 0.5, alpha, alpha * 1.5, alpha * 2.0, alpha * 3.0]:
                if ls_alpha < 0 or ls_alpha > 1:
                    continue
                cand = (1.0 - ls_alpha) * y + ls_alpha * g_dir
                cand = np.maximum(cand, 0.0)
                sc = cand.sum()
                if sc < 1e-15:
                    continue
                cand = cand / sc
                m_cand = fast_autocorr(cand).max()
                if m_cand < best_m_ls:
                    best_m_ls = m_cand
                    best_alpha_ls = ls_alpha
                    y_new = cand
        
        y = y_new
        x = x_new
        
        # Update best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            # Adaptively tighten margin
            margin_frac = max(margin_frac * 0.95, 1e-5)
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:

            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")
    
    return best_a, best_m


def peak_crush(a0, time_limit, verbose=True, rng_seed=42):
    """Fast coordinate descent that focuses on reducing the maximum autocorrelation peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))
        
        # Find all lags near the maximum
        near_max = ac > 0.95 * M_curr
        peak_lags = np.where(near_max)[0]
        
        improved = False
        
        # For each peak lag, try to reduce it by redistributing mass
        for pl in peak_lags[:5]:  # Limit to top 5 peak lags
            if time.monotonic() - t0 > time_limit:
                break
            
            i_start = max(0, pl - n + 1)
            i_end = min(n, pl + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[pl - idx]
            total_contrib = contrib.sum()
            
            if total_contrib < 1e-20:
                continue
            
            # Find the top contributors
            n_top = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]
            
            for li in top_local:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                # Try moving a fraction of a[src] to various destinations
                best_improvement = M_curr
                best_move = None
                
                # Try different fractions
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    
                    # Try random destinations and some structured ones
                    n_trials = 40
                    for _ in range(n_trials):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_improvement:
                            best_improvement = M_trial
                            best_move = (src, dst, amt)
                
                if best_move is not None:
                    src_i, dst_i, amt = best_move
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)

def amd_optimize(a0, time_limit, top_k=50, beta=4.0, c0=4.0,
                 restart_every=5000, verbose=True, rng_seed=42):
    """Nesterov mirror descent with backtracking line search."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=top_k, temperature=beta)
        
        # Backtracking line search
        eta = gamma / (t + 1.0)
        base_m = fast_autocorr(y).max()
        accepted = False
        for _ in range(8):
            a_new = exp_map(y, g, eta)
            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            else:
                eta *= 0.5
        if not accepted:
            a_new = exp_map(y, g, eta)
            x_prev = x
            x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")
    
    return best_a, best_m

        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_subgradient(a, ac, top_k=80, temperature=4.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    w = np.exp(temperature * (top_vals - top_vals[0]))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def exp_map(y, grad, eta):
    log_a = np.log(np.maximum(y, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return y.copy()
    return a_new / s


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=300,
                    margin_frac=2e-3, use_line_search=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()
    cur_K = top_K
    cur_margin = margin_frac

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-18:
            break

        margins = M_curr - ac
        threshold = cur_margin * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > cur_K:
            active_idx = np.argsort(margins)[:cur_K]
        elif n_active < 5:
            active_idx = np.argsort(margins)[:max(10, cur_K // 5)]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if use_line_search:
            base_m = fast_autocorr(y).max()
            best_ls_m = fast_autocorr(y_new).max()
            best_ls_alpha = alpha
            for ls_a in [alpha * 0.5, alpha * 0.75, alpha * 1.5, alpha * 2.0, alpha * 3.0, alpha * 5.0]:
                if ls_a <= 0 or ls_a > 1.0:
                    continue
                cand = (1.0 - ls_a) * y + ls_a * g_dir
                cand = np.maximum(cand, 0.0)
                sc = cand.sum()
                if sc < 1e-15:
                    continue
                cand = cand / sc
                m_cand = fast_autocorr(cand).max()
                if m_cand < best_ls_m - 1e-16:
                    best_ls_m = m_cand
                    best_ls_alpha = ls_a
                    y_new = cand

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        improved = False
        if my < best_m:
            best_m = my
            best_a = y.copy()
            improved = True

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
            improved = True

        if improved:
            cur_margin = max(cur_margin * 0.97, 1e-5)
            if cur_K < 600:
                cur_K = min(cur_K + 5, 600)

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={cur_margin:.2e}")

    return best_a, best_m


def peak_crush(a0, time_limit, verbose=True, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 250

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        near_max = ac > 0.97 * M_curr
        peak_lags = np.where(near_max)[0]
        improved = False

        for pl in peak_lags[:4]:
            if time.monotonic() - t0 > time_limit:
                break
            pl_int = int(pl)
            i_start = max(0, pl_int - n + 1)
            i_end = min(n, pl_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[pl_int - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-20:
                continue
            n_top = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]

            for li in top_local:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                best_improvement = M_curr
                best_move = None
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    for _ in range(50):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_improvement:
                            best_improvement = M_trial
                            best_move = (src, dst, amt)

                if best_move is not None:
                    src_i, dst_i, amt = best_move
                    a[src_i] -= amt
                    a[dst_i] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakCrush final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=60, beta=4.0, c0=4.0,
                 restart_every=3000, verbose=True, rng_seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=top_k, temperature=beta)
        eta = gamma / (t + 1.0)
        base_m = fast_autocorr(y).max()
        accepted = False
        for _ in range(8):
            a_new = exp_map(y, g, eta)
            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            else:
                eta *= 0.5
        if not accepted:
            a_new = exp_map(y, g, eta)
            x_prev = x
            x = a_new
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")
    return best_a, best_m


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # Multi-shape initialization
    print("\n=== Phase 0: Multi-shape initialization ===")
    n_base = 3600

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)

    shapes_to_try = [
        ('hann', make_hann(n_base)),
        ('blackman', make_blackman(n_base)),
        ('sin1', make_sin_power(n_base, 1.0)),
        ('sin1.5', make_sin_power(n_base, 1.5)),
        ('sin2', make_sin_power(n_base, 2.0)),
        ('sin3', make_sin_power(n_base, 3.0)),
        ('gauss0.25', make_gauss(n_base, 0.25)),
        ('gauss0.35', make_gauss(n_base, 0.35)),
    ]

    for name, init in shapes_to_try:
        sc = evaluate_safe(init)
        print(f"  Shape {name}: J={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = init.copy()

    if parent is not None:
        p_norm = norm_sum1(parent)
        ps = evaluate_safe(p_norm)
        if ps < best_score:
            best_score = ps
            best_seq = p_norm.copy()
            print(f"  Parent remains best: J={ps:.6f}")

    print(f"  Best initial: J={best_score:.6f}, n={len(best_seq)}")
    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 1: AMD on best initial
    print(f"\n=== Phase 1: AMD ({len(best_seq)}) ===")
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 20:
        a_out, _ = amd_optimize(best_seq, t1, top_k=60, beta=4.0, c0=3.5,
                                restart_every=3000, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd1")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 2: LP Frank-Wolfe main
    print(f"\n=== Phase 2: LP-FW main (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30 and len(best_seq) <= 4500:
        a_out, _ = lp_frank_wolfe(best_seq, t2, solver, verbose=True, top_K=300,
                                   margin_frac=2e-3, use_line_search=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 3: Peak-crushing
    print(f"\n=== Phase 3: Peak-crush (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 15:
        a_out, _ = peak_crush(best_seq, t3, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc1")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 4: LP with tighter margins
    print(f"\n=== Phase 4: LP-FW tight (n={len(best_seq)}) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30 and len(best_seq) <= 4500:
        a_out, _ = lp_frank_wolfe(best_seq, t4, solver, verbose=True, top_K=500,
                                   margin_frac=5e-4, use_line_search=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 5: Multi-length exploration
    n_best = len(best_seq)
    candidate_lengths = sorted(set([
        n_best - 300, n_best - 100, n_best + 100,
        n_best + 300, n_best + 500, n_best + 800,
    ]))
    candidate_lengths = [l for l in candidate_lengths if 2000 <= l <= 5500 and l != n_best]

    print(f"\n=== Phase 5: Multi-length (trying {candidate_lengths[:4]}) ===")
    for new_n in candidate_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n  Trying n={new_n} (remaining={remaining:.0f}s)")
        a_long = interpolate_sequence(best_seq, new_n)
        t5a = min(remaining * 0.15, 120.0)
        if t5a > 20:
            a_out, _ = amd_optimize(a_long, t5a, top_k=50, beta=5.0, c0=3.0,
                                     restart_every=2000, verbose=False, rng_seed=99)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"amd_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60 and new_n <= 4200:
            t5b = min(remaining * 0.12, 100.0)
            if t5b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t5b, solver, verbose=False, top_K=250,
                                           margin_frac=2e-3, use_line_search=True)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 6: AMD fine-tune
    if remaining > 50:
        print(f"\n=== Phase 6: AMD fine-tune (n={len(best_seq)}) ===")
        t6 = min(remaining * 0.15, 150.0)
        if t6 > 20:
            a_out, _ = amd_optimize(best_seq, t6, top_k=80, beta=3.5, c0=4.0,
                                     restart_every=2000, verbose=True, rng_seed=77)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd_final")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 7: Final LP
    if remaining > 50 and len(best_seq) <= 4500:
        print(f"\n=== Phase 7: Final LP (n={len(best_seq)}) ===")
        t7 = min(remaining * 0.20, 200.0)
        if t7 > 30:
            a_out, _ = lp_frank_wolfe(best_seq, t7, solver, verbose=True, top_K=600,
                                       margin_frac=2e-4, use_line_search=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    remaining = time_budget - (time.monotonic() - t_global)

    # Phase 8: Final peak-crush
    if remaining > 20:
        print(f"\n=== Phase 8: Final peak-crush ===")
        t8 = min(remaining * 0.35, 150.0)
        if t8 > 10:
            a_out, _ = peak_crush(best_seq, t8, verbose=True, rng_seed=314)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = make_hann(3200)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining")

    return [float(x) for x in best_seq]
