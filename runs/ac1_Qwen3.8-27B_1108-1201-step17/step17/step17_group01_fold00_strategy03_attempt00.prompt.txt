import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def freq_domain_optimize(a0, time_limit, iters=2000, eta0=0.5, verbose=False):
    """Iterative spectrum flattening via Gerchberg-Saxton-like approach."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(iters):
        if time.monotonic() - t0 > time_limit:
            break

        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Soft flattening: compress dynamic range of spectrum
        eta = eta0 * (1.0 + 0.5 * it / max(1, iters))
        P_target = P_mean * (P / P_mean) ** (1.0 - eta)
        P_target = np.maximum(P_target, 1e-30)

        # Update phase, adjust amplitude
        phase = np.angle(A)
        mag_new = np.sqrt(P_target)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        M_new = fast_autocorr(a_new).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Damped step
            a = 0.5 * a + 0.5 * a_new
            a = norm_sum1(a)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FreqDom it={it} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=200):
    """LP-based Frank-Wolfe with adaptive constraint selection."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    import cvxpy as cp

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()

    T_full = np.zeros((2 * n - 1, n))

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        T_full[:] = 0.0
        for j in range(n):
            T_full[j:j + n, j] = y

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        constraint_vals = T_full @ y
        margins = M_curr - constraint_vals
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        T_sel = T_full[binding_idx, :]
        M_sel = M_curr * np.ones(K_sel)

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def mass_transport_descent(a0, time_limit, verbose=False, rng_seed=42):
    """Coordinate descent based on mirror-position mass transport."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(15, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue

            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Compute cost at each position for all top peaks
            costs = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                for d in range(n):
                    j1 = k2i - d
                    j2 = d - k2i
                    c = 0.0
                    if 0 <= j1 < n:
                        c += a[j1]
                    if 0 <= j2 < n:
                        c += a[j2]
                    costs[d] += c

            # Find best destination: low cost
            candidates = np.argsort(costs)[:max(1, n // 4)]
            best_dst = candidates[np.argmin(a[candidates])]

            if best_dst == src:
                best_dst = int(rng.integers(n))
                if best_dst == src:
                    continue

            found = False
            for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 30:
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Transport ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_power_law(n, power=1.0, center=False):
    x = np.arange(1, n + 1, dtype=np.float64)
    if center:
        x = np.abs(x - (n + 1) / 2.0) + 1.0
        a = x ** (-power)
    else:
        a = x ** power
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_sinc_like(n, bandwidth=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bandwidth
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_multihump(n, n_humps=3, spread=0.15):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_humps):
        center = (i + 1) / (n_humps + 1)
        a += np.exp(-((x - center) ** 2) / (2 * spread ** 2))
    a += 0.01
    return norm_sum1(a)


def make_random_smooth(n, seed=0, n_freq=8):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)
