def fast_autocorr(a):
    """Compute full linear self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def get_subgradient_smooth(a, ac, top_k=20, temperature=1.0):
    """
    Compute smoothed subgradient of max(conv(a,a)) using softmax weighting
    over the top-k peaks. Lower temperature = more peaked (closer to exact).

    n = len(a)
    grad = np.zeros(n)

    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    # Softmax weights
    exp_vals = np.exp(top_vals / (temperature * (top_vals.max() + 1e-30)))
    w = exp_vals / exp_vals.sum()
    for wi, k in zip(w, top_idx):
        if wi < 1e-8:

        k_int = int(k)

        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

            g = get_subgradient_exact(y, ac)
        else:
            g = get_subgradient_smooth(y, ac, top_k=cur_top_k, temperature=cur_temp)

        # Polyak step size
        lower_bound = 1.0 / n
        gap = M - lower_bound
        if gap < 1e-10:
            gap = 1e-10
        eta = cur_c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-10)

        # Mirror descent (exponential) step
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Diverse restart strategy
        if stall > restart_every:
            stall = 0
            restart_count += 1
            restart_type = rng.integers(4)

            if restart_type == 0:
                # Gaussian perturbation
                sigma = 0.005 + 0.01 * min(restart_count, 10)
                pert = rng.normal(0, sigma, n) * best_a
                x_r = best_a + pert
            elif restart_type == 1:
                # Structural: flip a random contiguous block
                x_r = best_a.copy()
                block_len = rng.integers(n // 10, n // 3)
                start = rng.integers(0, n - block_len)
                x_r[start:start + block_len] = x_r[start + block_len - 1:start - 1:-1]
            elif restart_type == 2:
                # Structural: blend with a different init
                x_r = 0.7 * best_a + 0.3 * make_cos_init(n)
            else:
                # Random sparse perturbation
                x_r = best_a.copy()
                n_pert = max(10, n // 50)
                positions = rng.choice(n, size=n_pert, replace=False)
                for p in positions:
                    factor = rng.uniform(0.5, 2.0)
                    x_r[p] *= factor

            x_r = np.maximum(x_r, 1e-12)
            s_r = x_r.sum()
            if s_r < 1e-15:
                x_r = best_a.copy()
            else:
                x_r = x_r / s_r
            x = x_r.copy()
            x_prev = x_r.copy()

            # Adapt parameters
            cur_top_k = min(cur_top_k + 5, 60)
            cur_temp = max(cur_temp * 0.85, 0.02)
            cur_c = min(cur_c + 0.3, 10.0)
            cur_exact_frac = min(cur_exact_frac + 0.03, 0.6)

            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SGD v2 restart {restart_count} ({elapsed:.0f}s): J={best_J:.6f}, tk={cur_top_k}, temp={cur_temp:.3f}")

        if verbose and t % 30000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD v2 t={t} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def lp_frank_wolfe_v3(a0, time_limit, solver, verbose=False, top_K=800):
    """
    LP Frank-Wolfe with Nesterov acceleration and adaptive constraint selection.
    Uses backtracking line search for the FW step.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_y = a.copy()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current convolution
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Build constraint matrix for top-K most binding constraints
        # T_full[k, j] = y[j] if j <= k < j+n else 0
        # conv[k] = sum_j T_full[k,j] * y[j]
        # We want: T_full[k,:] @ b <= M_curr for selected k

        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]

        # Build sparse-ish constraint matrix
        T_sel = np.zeros((K_sel, n))
        for row, k in enumerate(binding_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            T_sel[row, j_start:j_end] = y[j_start:j_end]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr * np.ones(K_sel)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov-accelerated Frank-Wolfe step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        prev_y = y
        y = y_new
        x = x_new

        # Evaluate and update best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW3 iter={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def spectral_flatten(a0, time_limit, verbose=False, n_iter=5000, seed=100):
    """
    Spectral domain optimization: work with log|FFT(a)| to flatten the spectrum.
    The idea: if |FFT(a)| is flat, then the autocorrelation is a delta,
    and the convolution peak should be reduced.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    fft_size = 1 << (2 * n - 1).bit_length()

    for t in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute FFT
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: make |A| as flat as possible (but A[0] is fixed = sum(a) = 1)
        # Use iterative amplitude adjustment
        target_mag = np.ones(len(mag))
        target_mag[0] = mag[0]  # DC is fixed

        # Interpolation between current and target
        blend = 0.01 + 0.02 * (t / n_iter)
        new_mag = (1 - blend) * mag + blend * target_mag
        new_mag[0] = mag[0]

        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Add small random perturbation
            a = a + rng.normal(0, 0.001, n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral t={t} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def peak_reduction_v2(a0, time_limit, verbose=False, seed=77):
    """
    Improved peak reduction: identify peak contributors and use gradient-based
    to find optimal mass transfer destinations.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_lags = np.argsort(ac)[::-1][:10]

        # Compute per-coordinate sensitivity to peak
        sensitivity = np.zeros(n)
        for k in top_lags[:6]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            for ii, jj in zip(idx, j_idx):
                sensitivity[ii] += a[jj]
                sensitivity[jj] += a[ii]

        # Find good source/destination pairs
        high_sens = np.where((sensitivity > sensitivity.mean()) & (a > 0.0005))[0]
        low_sens = np.where((sensitivity < sensitivity.mean() * 0.5) | (sensitivity == 0))[0]

        if len(high_sens) < 5 or len(low_sens) < 5:
            low_sens = np.where(a < np.median(a))[0]
            if len(low_sens) < 5:
                stall += 1
                continue

        improved = False
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            # Try binary search for optimal transfer amount
            lo, hi = 0.0, src_val * 0.9
            best_amt = 0.0
            best_trial_J = best_J

            for _ in range(8):
                mid = (lo + hi) / 2
                a_trial = a.copy()
                a_trial[src] = a[src] - mid
                a_trial[dst] = a[dst] + mid
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    break
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_amt = mid
                    lo = mid
                else:
                    hi = mid

            if best_amt > 0 and best_trial_J < best_J - 1e-14:
                a[src] -= best_amt
                a[dst] += best_amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_J = best_trial_J
                best_a = a.copy()
                improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                # Larger perturbation to escape
                n_pert = max(5, n // 100)
                positions = rng.choice(n, size=n_pert, replace=False)
                for p in positions:
                    if a[p] > 1e-6:
                        factor = rng.uniform(0.3, 3.0)
                        a[p] *= factor
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed v2 ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    PeakRed v2 final ({elapsed:.0f}s): J={best_J:.6f}")
    return best_a, best_J


def make_cos_init(n):
    """U-shaped (high at edges) initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_skew_init(n, skew=0.3):
    """Asymmetric initialization: shifted peak."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - skew)) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)


def make_dual_peak_init(n):
    """Two-peak asymmetric initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.15) / 0.1) ** 2) + 0.5 * np.exp(-((x - 0.85) / 0.12) ** 2) + 0.005
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def coarse_to_fine(a_coarse, n_target, time_per_scale=30, verbose=False):
    """
    Multi-scale optimization: start coarse, refine to target.
    """
    n_cur = len(a_coarse)
    current = a_coarse.copy()
    scales = []
    n_target = int(n_target)
    # Build scale ladder
    n = n_cur
    while n < n_target:
        n = min(n * 2, n_target)
        scales.append(n)
    if not scales:
        return current

    for i, new_n in enumerate(scales):
        t0 = time.monotonic()
        current = interpolate_sequence(current, new_n)
        current, _ = nesterov_subgradient_v2(current, time_per_scale, verbose=False,
                                             top_k=15, temperature=0.1, seed=100 + i)
        if verbose:
            J = compute_J(current)
            print(f"    Coarse-to-fine scale {i}: n={new_n}, J={J:.6f}")
    return current


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator with multi-phase optimization.
    """
    t_global = time.monotonic()

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_cos_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Improved Nesterov SGD (multiple inits) ===
    print(f"\n=== Phase 1: Nesterov SGD v2 ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.22, 380.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
            # Slightly perturbed parent
            rng_init = np.random.default_rng(12345)
            pert = norm_sum1(parent) + rng_init.normal(0, 0.005, n0) * norm_sum1(parent)
            inits.append(norm_sum1(np.maximum(pert, 0)))
            labels.append("parent_pert")

        inits.append(make_cos_init(n0))
        labels.append("cos")

        inits.append(make_skew_init(n0, skew=0.25))
        labels.append("skew0.25")

        inits.append(make_skew_init(n0, skew=0.35))
        labels.append("skew0.35")

        inits.append(make_dual_peak_init(n0))
        labels.append("dual_peak")

        # Remove near-duplicates
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for u in unique_inits:
                if np.sum(np.abs(init - u)) < 0.03:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        t_per_init = max(20.0, t_phase1 / n_inits)

        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per_init, remaining - 30)

                                               top_k=20, temperature=0.08,

        try_update(a_out, f"lpfw3_n{n_best}")

    # === Phase 3: Peak reduction ===

    if remaining > 30:
        t9 = min(remaining * 0.5, 120.0)
        if t9 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = nesterov_subgradient_v2(a_in, t9, verbose=False,
                                               top_k=40, temperature=0.03,
                                               exact_frac=0.5, c=6.0,
                                               restart_every=5000,
                                               seed=4321)
            try_update(a_out, f"sgd_polish_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_cos_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
