def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_lse_gradient(a, ac, beta, K=100):
    """Compute gradient of LSE_beta(autocorr) at point a.
    g[j] = 2 * sum_k w_k * a[k-j] for valid k-j.
    """
    n = len(a)
    ac_shifted = ac - ac.max()
    logits = beta * ac_shifted
    logits = np.clip(logits, -500, 500)
    exp_logits = np.exp(logits)
    w = exp_logits / exp_logits.sum()
    
    top_idx = np.argsort(w)[::-1][:K]
    top_w = w[top_idx]
    
    g = np.zeros(n)
    for wi, k in zip(top_w, top_idx):
        ki = int(k)
        if wi < 1e-14:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        g[js] += 2.0 * wi * a[ki - js]
    return g


def lse_descent(a0, time_limit, beta_start=5.0, beta_end=60.0,
                eta_start=0.03, mu=0.85, K=100, verbose=False, rng_seed=42):
    """LogSumExp smooth gradient descent with momentum."""
    n = len(a0)
    x = norm_sum1(a0)
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    v = np.zeros(n)
    max_iters = 500000
    warmup = max(500, max_iters // 20)
    
    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        frac = min(1.0, t / (0.8 * max_iters))
        beta = beta_start + (beta_end - beta_start) * frac * beta
        
        # Adaptive step size
        if t < warmup:
            eta = eta_start * t / warmup
        else:
            eta = eta_start / (1.0 + 0.005 * (t - warmup))
        
        ac = fast_autocorr(x)
        g = compute_lse_gradient(x, ac, beta, K=K)
        
        # Momentum update
        v = mu * v + (1.0 - mu) * g
        x_new = x - eta * v
        x_new = proj_simplex(x_new)
        
        M_new = fast_autocorr(x_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x_new.copy()
        
        x = x_new
        
        # Periodic restart
        if t % 3000 == 0 and t > warmup:
            pert = rng.normal(0, 0.015, n) * best_a
            x_pert = best_a + pert
            x_pert = np.maximum(x_pert, 0)
            sp = x_pert.sum()
            if sp > 1e-15:
                x_pert = x_pert / sp
                M_pert = fast_autocorr(x_pert).max()
                if M_pert < best_m:
                    x = x_pert
                    v = np.zeros(n)
                elif rng.random() < 0.15 and M_pert < best_m * 1.05:
                    x = x_pert
                    v = np.zeros(n)
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE t={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, max_constraints=1500, K_increase=100):

    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    best_m = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    x_mom = y.copy()
    K_sel = min(max_constraints, 2 * n - 1)
    iter_count = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        # Select top K constraints
        binding_idx = np.argsort(ac)[::-1][:K_sel]

            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_start > j_end:
                A_rows.append(np.zeros(n))
            else:
                row = np.zeros(n)
                js = np.arange(j_start, j_end + 1)
                row[js] = y[ki - js]
                A_rows.append(row)

        constraints = [A_mat @ b_var <= M_curr * np.ones(len(binding_idx))]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step size
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)

        if sy < 1e-15:
            y_new = y.copy()

            y_new = y_new / sy

            sx = x_mom.sum()

            else:
                x_mom = x_mom / sx

        iter_count += 1
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x_mom).max()
        if mx < best_m:
            best_m = mx
            best_a = x_mom.copy()
        
        # Adaptively increase constraint count
        if t % 30 == 0:
            K_sel = min(K_sel + K_increase, 2 * n - 1)
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K_sel}")
    
    return best_a, best_m


def annealed_local_search(a0, time_limit, verbose=True, rng_seed=111):
    """Local search with simulated annealing acceptance."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    temp = 0.02
    cooldown = 0.999
    min_temp = 0.001
    stall = 0
    max_stall = 300
    n_top = 10
    
    total_iters = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            contrib = a[js] * a[ki - js]
            if contrib.max() < 1e-20:
                continue
            
            src = js[np.argmax(contrib)]
            src_partner = ki - src
            if a[src] < 1e-12:
                continue
            
            # Find low-sensitivity destination
            sens = np.zeros(n)
            for k2 in top_k:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                idx2 = np.arange(s2, e2 + 1)
                sens[idx2] += a[k2i - idx2]
            
            candidates = np.where(sens < sens.max() * 0.3)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            candidates = candidates[candidates != src]
            if len(candidates) == 0:
                continue
            dst = candidates[np.argmin(sens[candidates])]
            
            # Try multiple fractions
            fracs = [0.005, 0.01, 0.02, 0.001, 0.003, 0.05, 0.0005]
            rng.shuffle(fracs)
            found = False
            for frac in fracs:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = fast_autocorr(a_trial).max()
                total_iters += 1
                
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break
                elif M_trial < M_curr and rng.random() < temp:
                    a = a_trial
                    stall = 0
                    found = True
                    break
            
            if found:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            n_pert = rng.integers(1, min(6, n))
            perm = rng.permutation(n)[:n_pert]
            a_new = a.copy()
            for i in range(0, len(perm) - 1, 2):
                src_i = perm[i]
                dst_i = perm[i + 1]
                amt = a_new[src_i] * rng.uniform(0.002, 0.02)
                a_new[src_i] -= amt
                a_new[dst_i] += amt
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
                M_new = fast_autocorr(a_new).max()
                total_iters += 1
                if M_new < best_m:
                    a = a_new
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                elif M_new < M_curr * 1.02 and rng.random() < temp:
                    a = a_new
                    stall = 0
        
        temp = max(min_temp, temp * cooldown)
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Anneal ({elapsed:.0f}s): J={J:.6f}, stall={stall}, temp={temp:.4f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Anneal final ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def precise_coord_descent(a0, time_limit, verbose=True, rng_seed=222):
    """High-precision coordinate descent focusing on the single top peak."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_idx = np.argsort(ac)[::-1][:5]
        
        improved = False
        
        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            contrib = a[js] * a[ki - js]
            if contrib.max() < 1e-20:
                continue
            
            # Try reducing top-3 contributing pairs
            top_contrib_idx = np.argsort(contrib)[::-1][:3]
            for tc_i in top_contrib_idx:
                if time.monotonic() - t0 > time_limit:
                    break
                src = js[tc_i]
                partner = ki - src
                ia, ib = int(src), int(partner)
                
                # Build sensitivity for destination selection
                sens = np.zeros(n)
                for k2 in top_idx:
                    k2i = int(k2)
                    s2 = max(0, k2i - n + 1)
                    e2 = min(n - 1, k2i)
                    idx2 = np.arange(s2, e2 + 1)
                    sens[idx2] += a[k2i - idx2]
                
                # Find best destination (low sensitivity, not the partner)
                mask = np.ones(n, dtype=bool)
                mask[ia] = False
                mask[ib] = False
                cand = np.where(mask)[0]
                if len(cand) == 0:
                    continue
                cand_sens = sens[cand]
                # Try top-10 lowest sensitivity destinations
                top_dst = cand[np.argsort(cand_sens)[:10]]
                
                for dst in top_dst:
                    found = False
                    for frac in [0.001, 0.002, 0.003, 0.005, 0.008, 0.0005]:
                        amt = a[ia] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[ia] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-16:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found = True
                            break
                    if found:
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Random kick
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4000
        best_seq = make_u_shape(n_base, 2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: LSE Descent (new method) ===
    print(f"\n=== Phase 1: LSE Descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.28, 480.0)
    
    if t_phase1 > 80:
        inits = []
        labels = []
        
        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")
        
        inits.append(make_u_shape(n0, 2.0))
        labels.append("u2.0")
        
        inits.append(make_power_u(n0, p=3.0))
        labels.append("pow3")
        
        inits.append(spectral_init(n0, n_modes=25, seed=99))
        labels.append("spec99")
        
        inits.append(make_bimodal(n0, width=0.18))
        labels.append("bimodal")
        
        # Blend parent with different shapes
        if parent is not None:
            blend1 = 0.6 * norm_sum1(parent) + 0.4 * make_u_shape(n0, 3.0)
            inits.append(norm_sum1(blend1))
            labels.append("blend_u3")
        
        # Deduplicate
        seen_sigs = set()
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            sig = tuple(np.round(init[::max(1, n0 // 30)], 3))
            if sig not in seen_sigs:
                seen_sigs.add(sig)
                unique_inits.append(init)
                unique_labels.append(lbl)
        inits = unique_inits
        labels = unique_labels
        
        t_per_init = max(25.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per_init, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 20:
                break
            print(f"  LSE init '{lbl}' (t_avail={t_avail:.0f}s)...")
            a_out, _ = lse_descent(init, t_avail,
                                    beta_start=5.0, beta_end=50.0,
                                    eta_start=0.03, mu=0.85, K=80,
                                    verbose=False, rng_seed=42 + idx * 7)
            try_update(a_out, f"lse_{lbl}")
            
            # Also try with different hyperparams on the best so far
            if idx == 0 and t_avail > 60:
                a_out2, _ = lse_descent(norm_sum1(best_seq), t_avail * 0.5,
                                         beta_start=15.0, beta_end=100.0,
                                         eta_start=0.01, mu=0.9, K=150,
                                         verbose=False, rng_seed=500 + idx)
                try_update(a_out2, "lse_sharp")
    
    # === Phase 2: LP-Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 430.0)
    if t2 > 50 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        max_c = min(2000, 2 * n_best - 1) if n_best <= 3000 else min(1000, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True,
                                    max_constraints=max_c, K_increase=80)
        try_update(a_out, "lp_main")
    
    # === Phase 3: Annealed local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Annealed Local Search (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = annealed_local_search(a_in, t3, verbose=True, rng_seed=333)
        try_update(a_out, "anneal")
    
    # === Phase 4: Precise coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Precise Coordinate Descent (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = precise_coord_descent(a_in, t4, verbose=True, rng_seed=444)
        try_update(a_out, "precise_coord")
    
    # === Phase 5: LP-FW round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP-FW Round 2 (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        max_c = min(2500, 2 * n_best - 1)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True,
                                    max_constraints=max_c, K_increase=50)
        try_update(a_out, "lp_r2")
    
    # === Phase 6: Multi-n exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    
    trial_lengths = []
    for delta in [300, -300, 600, -600]:
        new_n = n_best + delta
        if 2000 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 100]
    
    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        
        t6a = min(remaining * 0.4, 180.0)
        if t6a > 30:
            a_out, _ = lse_descent(a_long, t6a,
                                    beta_start=10.0, beta_end=60.0,
                                    eta_start=0.02, mu=0.88, K=100,
                                    verbose=False, rng_seed=new_n % 997)
            try_update(a_out, f"lse_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and new_n <= 4500:
            t6b = min(remaining * 0.5, 120.0)
            if t6b > 25:
                a_in = interpolate_sequence(best_seq, new_n)
                max_c = min(1200, 2 * new_n - 1)
                a_out, _ = lp_frank_wolfe(a_in, t6b, solver, verbose=False,
                                            max_constraints=max_c, K_increase=100)
                try_update(a_out, f"lp_n{new_n}")
    
    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Polish (n={n_best}) ===")
    
    if remaining > 50:
        t7a = min(remaining * 0.3, 120.0)
        if t7a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = lse_descent(a_in, t7a,
                                    beta_start=30.0, beta_end=120.0,
                                    eta_start=0.005, mu=0.92, K=200,
                                    verbose=False, rng_seed=777)
            try_update(a_out, "lse_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30 and n_best <= 5000:
            t7b = min(remaining * 0.4, 100.0)
            if t7b > 20:
                a_in = norm_sum1(best_seq)
                max_c = min(3000, 2 * n_best - 1)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False,
                                            max_constraints=max_c, K_increase=30)
                try_update(a_out, "lp_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t7c = min(remaining * 0.6, 60.0)
            if t7c > 10:
                a_in = norm_sum1(best_seq)
                a_out, _ = precise_coord_descent(a_in, t7c, verbose=False, rng_seed=888)
                try_update(a_out, "coord_final")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_u_shape(n_fb, 2.5)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
