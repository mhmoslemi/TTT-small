def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:

def _norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _proj_simplex(v, s=1.0):
    """Project onto {x: sum(x)=s, x>=0} via Duchi et al. (2012)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (sv - s))[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (sv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_smooth_grad(a, beta, fft_size):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:2 * n - 1]
    max_conv = conv.max()
    shifted = beta * (conv - max_conv)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / w_sum
    b = a[::-1]
    B = np.fft.rfft(b, fft_size)
    W = np.fft.rfft(w, fft_size)
    grad_full = np.fft.irfft(B * W, fft_size)
    offset = n - 1
    grad = 2.0 * grad_full[offset:offset + n]
    return grad, conv


def _smooth_projected_grad(a0, time_limit, beta_start=0.5, beta_end=8.0,
                           verbose=True, rng_seed=42):
    """Nesterov-accelerated projected gradient with log-sum-exp smoothing."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_M = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    fft_size = 1 << (3 * n - 1).bit_length()

    x = a.copy()
    x_prev = a.copy()
    eta = 0.5
    stall = 0
    total_inters = 0
    stagnation_limit = 8000

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, t / 80000.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        beta_n = (t - 1.0) / (t + 1.5)
        y = x + beta_n * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        grad, conv = _compute_smooth_grad(y, beta, fft_size)
        M_curr = _fast_autocorr(x).max()

        if M_curr < best_M - 1e-16:
            best_M = M_curr
            best_a = x.copy()
            stall = 0

        for ls in range(25):
            step = eta / (1.0 + 0.1 * ls)
            a_new = _proj_simplex(x - step * grad)
            M_new = _fast_autocorr(a_new).max()
            if M_new < M_curr - 1e-18:
                x_prev = x
                x = a_new
                eta = min(eta * 1.05, 2.0)
                break
            else:
                eta = max(eta * 0.8, 1e-6)
        else:
            x_prev = x

        total_inters += 1

        if stall > stagnation_limit:
            stall = 0
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            eta = 1.0
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_M
                print(f"    SPG restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        stall += 1
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    SPG t={t} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, beta={beta:.3f}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_M
    if verbose:
        print(f"    SPG done ({elapsed:.0f}s): J={J:.6f}, iters={total_inters}")
    return best_a, best_M


def _lp_frank_wolfe(a0, time_limit, solver, initial_K=300, max_K=3000,
                    add_per_iter=30, verbose=True):
    """LP-based Frank-Wolfe with cutting-plane constraint management."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    g_dir = a.copy()

    ac_y = _fast_autocorr(y)
    M_curr = ac_y.max()

    init_idx = np.argsort(ac_y)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    fft_size = 1 << (2 * n - 1).bit_length()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()

        ac_x = _fast_autocorr(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()

        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        coeffs_matrix = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs_matrix[jj, idx] = y[k_int - idx]

        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs_matrix @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
                constraint_list = sorted(constraint_set)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:2 * n - 1]

        violations = np.where(conv_yg > M_curr + 1e-8)[0]
        if len(violations) > 0:
            viol_amt = conv_yg[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m


def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _coord_descent_polish(a0, time_limit, rng_seed=123):
    """Fast coordinate descent: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = _norm_sum1(a0)
    best_a = a.copy()
    best_M = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = _fast_autocorr(a)
        M_curr = ac.max()
        improved = False

        n_top = min(20, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        for k in top_k[:10]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                k2i = int(k2)
                peak_mask[max(0, k2i - n + 1):min(n, k2i + 1)] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            dst_scores = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    dst_scores[idx2] += a[k2i - idx2]
                    dst_scores[k2i - idx2] += a[idx2]

            dst = candidates[np.argmin(dst_scores[candidates])]
            if dst == src:
                continue

            found = False
            for frac in [0.01, 0.03, 0.06, 0.10]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = _fast_autocorr(a_trial).max()
                if M_trial < best_M - 1e-16:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    found = True
                    break

        if not improved:
            stall += 1
            if rng.random() < 0.3:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.005, 0.02)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = _fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best_a = a.copy()
                        stall = 0

    return best_a, best_M


def _parametric_init(n, best_current, best_score):
    """Try parametric families for a potentially better initialization."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best_a = best_current.copy()
    best_val = best_score

    def try_update(a, label):
        nonlocal best_a, best_val
        a = _norm_sum1(a)
        val = 2.0 * n * _fast_autocorr(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"    Parametric [{label}]: J={val:.6f}")

    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        try_update((i_arr + 1.0) ** p, f"pow_{p}")
        try_update((n - i_arr) ** p, f"ipow_{p}")

    for rate in [0.5, 1, 2, 5, 10, 20]:
        try_update(np.exp(-rate * x), f"decay_{rate}")
        try_update(np.exp(rate * x), f"grow_{rate}")

    for c in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for s in [5, 10, 20, 50]:
            try_update(1.0 / (1.0 + np.exp(-s * (x - c))), f"sig_{c}_{s}")

    for freq in [1, 2, 3]:
        try_update(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01), f"sin_{freq}")

    if best_score < 1.6:
        for w in [0.2, 0.4, 0.6, 0.8]:
            a_g = np.exp(-1.0 * x)
            a_blend = w * best_current / best_current.sum() + (1.0 - w) * a_g / a_g.sum()
            try_update(a_blend, f"blend_{w}")

    elapsed = time.monotonic() - t0
    print(f"  Parametric done ({elapsed:.1f}s): best J={best_val:.6f}")
    return best_a, best_val

            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric search (brief) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.04, 40.0)
    if t1 > 10:
        print(f"\n=== Phase 1: Parametric (n={n0}, {t1:.0f}s) ===")
        a_p, val_p = _parametric_init(n0, best_seq, 2.0 * n0 * _fast_autocorr(best_seq).max())
        check_update(a_p, "parametric")

    # === Phase 2: Smooth projected gradient (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Smooth PG (n={len(best_seq)}, {t2:.0f}s) ===")
        a_in = _norm_sum1(best_seq)
        a_out, _ = _smooth_projected_grad(a_in, t2, beta_start=0.3, beta_end=12.0,
                                          verbose=True, rng_seed=42)
        check_update(a_out, "spg_main")

    # === Phase 3: LP-FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.30, 400.0)
    if t3 > 30 and n_best <= 5500:
        print(f"\n=== Phase 3: LP-FW (n={n_best}, {t3:.0f}s) ===")
        a_in = _norm_sum1(best_seq)
        a_out, _ = _lp_frank_wolfe(a_in, t3, solver, initial_K=400, max_K=3500,
                                   add_per_iter=35, verbose=True)
        check_update(a_out, "lp_main")

    # === Phase 4: Second SPG pass with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        print(f"\n=== Phase 4: SPG refine (n={len(best_seq)}, {t4:.0f}s) ===")
        a_in = _norm_sum1(best_seq)
        a_out, _ = _smooth_projected_grad(a_in, t4, beta_start=3.0, beta_end=25.0,
                                          verbose=True, rng_seed=99)
        check_update(a_out, "spg_refine")

    # === Phase 5: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Coord descent (n={len(best_seq)}, {t5:.0f}s) ===")
        a_in = _norm_sum1(best_seq)
        a_out, _ = _coord_descent_polish(a_in, t5, rng_seed=77)
        check_update(a_out, "coord")

    # === Phase 6: Try adjacent lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    adj_lengths = sorted(set([n_best - 10, n_best - 5, n_best + 5, n_best + 10, n_best + 20, n_best - 20]))
    adj_lengths = [l for l in adj_lengths if 5000 <= l <= 5500]

    for new_n in adj_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        if new_n == n_best:
            continue
        ta = min(remaining * 0.25, 120.0)
        if ta < 20:
            continue
        print(f"\n=== Phase 6: Adjacent n={new_n} ({ta:.0f}s) ===")
        a_long = _interpolate_seq(best_seq, new_n)
        a_out, _ = _smooth_projected_grad(a_long, ta, beta_start=2.0, beta_end=15.0,
                                          verbose=False, rng_seed=new_n % 1000)
        check_update(a_out, f"adj_n{new_n}")

    # === Phase 7: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60 and n_best <= 5500:
        t7 = min(remaining * 0.50, 250.0)
        if t7 > 30:
            print(f"\n=== Phase 7: Final LP-FW (n={n_best}, {t7:.0f}s) ===")
            a_in = _norm_sum1(best_seq)
            a_out, _ = _lp_frank_wolfe(a_in, t7, solver, initial_K=500, max_K=4000,
                                       add_per_iter=40, verbose=True)
            check_update(a_out, "lp_final")

    # === Phase 8: Final coord polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t8 = min(remaining * 0.60, 100.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final coord (n={len(best_seq)}, {t8:.0f}s) ===")
            a_in = _norm_sum1(best_seq)
            a_out, _ = _coord_descent_polish(a_in, t8, rng_seed=314)
            check_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.exp(-1.0 * x)
        best_seq = _norm_sum1(best_seq)
        best_score = _evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
