"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Frank-Wolfe with HiGHS solver and backtracking line search
2. Adaptive active-set with progressive margin tightening
3. Multi-length exploration (parent length + extensions)
4. Spectral diversity for escape from local minima
5. Final coordinate descent polishing

import numpy as np
import time
from scipy.optimize import linprog


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_perturb(a, strength, rng):
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    mag = np.maximum(mag, 1e-30)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    mag_pert = mag * (1.0 + rng.normal(0, strength * 0.5, len(mag)))
    A_new = mag_pert * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def lp_direction(a, M, active_idx, method='highs'):
    n = len(a)
    K = len(active_idx)
    if K == 0:
        return np.ones(n) / n

    c = -np.ones(n)

    # Build A_ub efficiently using vectorized operations
    # For each k_val, valid j range: [max(0, k_val-n+1), min(n-1, k_val)]
    # A_ub[k_idx, j] = a[k_val - j]
    
    # Pre-compute for all active constraints
    A_ub = np.zeros((K, n), dtype=np.float64)
    for k_idx, k_val in enumerate(active_idx):
        j_start = max(0, k_val - n + 1)
        j_end = min(n - 1, k_val)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        A_ub[k_idx, js] = a[k_val - js]

    b_ub = np.full(K, M, dtype=np.float64)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
    except Exception:
        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        except Exception:
            return None

    if result.success:
        b_val = np.maximum(result.x, 0.0)
        s_val = b_val.sum()
        if s_val > 1e-15:
            return b_val / s_val
    return None


def compute_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        js = np.arange(j_start, j_end)
        grad[js] += 2.0 * wi * a[k_int - js]
    return grad


def fw_optimize(a0, time_limit, top_k=200, margin_init=0.005, margin_min=5e-5,
                verbose=True, seed=42, max_its=50000):
    """
    LP Frank-Wolfe with backtracking line search and adaptive active set.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    margin = margin_init
    stall = 0
    stall_limit = 3000
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_its:
        it += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Adaptive active set
        threshold = margin * M_curr
        active_mask = ac >= (M_curr - threshold)
        n_active = active_mask.sum()

        # Cap active set size
        if n_active > top_k * 3:
            # Use the largest n_active*3 values
            sorted_idx = np.argsort(ac)[::-1][:top_k * 3]
            active_idx = sorted_idx
        elif n_active < 3:
            sorted_idx = np.argsort(ac)[::-1][:top_k]
            active_idx = sorted_idx
        else:
            active_idx = np.where(active_mask)[0]

        # Solve LP
        s_dir = lp_direction(a, M_curr, active_idx)
        if s_dir is None:
            s_dir = compute_subgradient_direction(a, ac, top_k=50)
            if s_dir is None:
                s_dir = np.ones(n) / n

        # Backtracking line search
        base_m = M_curr
        best_alpha = 2.0 / (it + 2.0)
        best_trial_m = base_m
        for alpha in [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05]:
            a_trial = (1.0 - alpha) * a + alpha * s_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            m_trial = fast_autocorr(a_trial).max()
            if m_trial < best_trial_m:
                best_trial_m = m_trial
                best_alpha = alpha

        a_new = (1.0 - best_alpha) * a + best_alpha * s_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        m_new = fast_autocorr(a_new).max()
        a = a_new

        if m_new < best_m - 1e-18:
            best_m = m_new
            best_a = a.copy()
            stall = 0
            # Tighten margin on improvement
            margin = max(margin * 0.97, margin_min)
        else:
            stall += 1

        # Restart on long stall
        if stall > stall_limit:
            stall = 0
            pert = rng.normal(0, 0.01, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            a = a_r.copy()
            margin = max(margin * 1.5, 0.002)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    FW restart it={it} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FW it={it} ({elapsed:.0f}s): J={J:.6f}, K={len(active_idx)}, margin={margin:.2e}")

    return best_a, best_m


def compute_subgradient_direction(a, ac, top_k=50, beta=5.0):
    """Fallback: use subgradient to find a simplex direction."""
    n = len(a)
    g = compute_subgradient(a, ac, top_k=top_k, beta=beta)
    # FW direction: min g^T s over simplex = e_{argmin(g)}
    # But we can do better: move mass from high to low gradient
    j_low = int(np.argmin(g))
    j_high = int(np.argmax(g))
    if a[j_high] < 1e-12:
        return np.ones(n) / n
    # Create a direction that moves mass from j_high to j_low
    s_dir = a.copy()
    # This is just a heuristic fallback
    s_dir[j_low] = 1.0
    s_dir[j_high] = 0.0
    s_dir = np.maximum(s_dir, 0.0)
    s_s = s_dir.sum()
    if s_s < 1e-15:
        return np.ones(n) / n
    return s_dir / s_s


def coordinate_polish(a0, time_limit, verbose=True, seed=123):
    """Coordinate descent polishing targeting top peaks."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        # Find top contributors to the peak
        j_start = max(0, peak_k - n + 1)
        j_end = min(n, peak_k + 1)
        js = np.arange(j_start, j_end)
        contrib = a[js] * a[peak_k - js]
        total = contrib.sum()

        if total < 1e-20:
            stall += 10
            continue

        # Top contributors
        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(js[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                # Try several random destinations
                for _ in range(40):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] = src_val - amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    CoordPolish ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def amd_polish(a0, time_limit, top_k=50, beta=5.0, c0=4.0, verbose=False, seed=77):
    """Nesterov mirror descent for polishing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient(y, ac, top_k=top_k, beta=beta)

        # Backtracking
        eta = gamma / (np.sqrt(t) + 1.0)
        base_m = fast_autocorr(y).max()
        accepted = False
        for _ in range(5):
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            m_trial = fast_autocorr(a_new).max()
            if m_trial < base_m - 1e-18:
                x_prev = x
                x = a_new
                accepted = True
                break
            else:
                eta *= 0.5

        if not accepted:
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            stall = 0
            pert = rng.normal(0, 0.01, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3600
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: LP-FW main optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-FW main (n={n0}, budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = fw_optimize(a_in, t1, top_k=250, margin_init=0.005,
                               margin_min=5e-5, verbose=True, seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "fw_main")

    # === Phase 2: AMD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: AMD polish (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_polish(a_in, t2, top_k=50, beta=5.0, c0=4.0, seed=77)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd1")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord polish (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t3, verbose=True, seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "coord1")

    # === Phase 4: Second LP-FW with tighter margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP-FW tight (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 60:
        a_in = norm_sum1(best_seq)
        a_out, _ = fw_optimize(a_in, t4, top_k=350, margin_init=0.002,
                               margin_min=2e-5, verbose=True, seed=88)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "fw_tight")

    # === Phase 5: Try longer sequences ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = sorted(set([n_best + 400, n_best + 800, n_best + 1200]))
    longer_lengths = [l for l in longer_lengths if l <= 5200 and remaining > 200]

    for new_n in longer_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 150:
            break
        print(f"\n=== Phase 5: Extend to n={new_n} (budget={remaining:.0f}s) ===")
        a_long = interpolate_sequence(best_seq, new_n)

        t5a = min(remaining * 0.35, 300.0)
        if t5a > 40:
            a_out, _ = fw_optimize(a_long, t5a, top_k=250, margin_init=0.005,
                                   margin_min=5e-5, verbose=True, seed=111)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"fw_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80:
            t5b = min(remaining * 0.15, 100.0)
            if t5b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = coordinate_polish(a_in, t5b, verbose=False, seed=222)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"coord_n{new_n}")

    # === Phase 6: Spectral diversity ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Spectral diversity (n={n_best}, budget={remaining:.0f}s) ===")
    rng = np.random.default_rng(99)
    t6_per = min(remaining * 0.06, 80.0)
    if t6_per > 25:
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < t6_per + 30:
                break
            a_pert = spectral_perturb(best_seq, strength=0.02, rng=rng)
            a_out, _ = fw_optimize(a_pert, t6_per, top_k=200, margin_init=0.003,
                                   margin_min=5e-5, verbose=False, seed=200 + trial)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"spectral{trial}")

    # === Phase 7: Final LP-FW on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP-FW (n={n_best}, budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 250.0)
    if t7 > 40:
        a_in = norm_sum1(best_seq)
        a_out, _ = fw_optimize(a_in, t7, top_k=400, margin_init=0.001,
                               margin_min=1e-5, verbose=True, seed=555)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "fw_final")

    # === Phase 8: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final coord polish (n={len(best_seq)}, budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.30, 180.0)
    if t8 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t8, verbose=True, seed=777)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "coord_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    # Safety
    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
