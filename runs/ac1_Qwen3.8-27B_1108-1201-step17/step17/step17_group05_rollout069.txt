def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.maximum(v - mid, 0.0).sum() > 1.0:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - (lo + hi) / 2.0, 0.0)


def compute_smooth_grad(a, beta):
    """Compute gradient of smooth max approximation."""
    n = len(a)
    ac = fast_autocorr(a)
    M = ac.max()
    shifted = ac - M
    w = np.exp(beta * shifted)

    if w_sum < 1e-30:

    else:
        w = w / w_sum
    a_rev = a[::-1]

    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * A, fft_size)[:total_len]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, M

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_smooth_init(n, seed=0):
    """Create a smooth non-negative initialization via Fourier series."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, 8):
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)
