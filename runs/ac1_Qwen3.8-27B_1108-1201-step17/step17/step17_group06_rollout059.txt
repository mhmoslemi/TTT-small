```python
import numpy as np

from scipy.optimize import minimize

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = fast_conv(a)
    S = np.sum(a)
    if S < 1e-15:
        return float('inf')
    return float(2.0 * n * conv.max() / (S * S))

def soft_max_and_weights(conv, tau):
    """Compute log-sum-exp approximation of max and the weights."""
    m = conv.max()
    z = (conv - m) / tau
    ez = np.exp(z)
    ez_sum = ez.sum()
    lse = m + tau * np.log(ez_sum)
    w = ez / ez_sum
    return lse, w

def compute_cross_corr(a, w):
    """Compute c_j = sum_l a_l * w_{j+l} for j=0..n-1."""
    n = len(a)
    a_rev = a[::-1]
    # conv(a_rev, w) has length n + (2n-1) - 1 = 3n-2
    conv_len = 3 * n - 2
    fsize = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fsize)
    W = np.fft.rfft(w, fsize)
    c_full = np.fft.irfft(Ar * W, fsize)
    return c_full[n-1 : n-1+n]

def smooth_obj_grad_logspace(x, tau, n):
    """
    Objective and gradient in log-space.

    # J = 2n * lse / S^2
    # dJ/da_j = 2n * (grad_lse_da_j * S^2 - lse * 2*S) / S^4
    #         = 2n * grad_lse_da_j / S^2 - 4n * lse / S^3

def optimize_lbfgs_logspace(a_init, tau_schedule, max_iter, time_limit, verbose=False):
    """
    Optimize using L-BFGS-B in log-space with continuation on tau.
    a_init: initial sequence (will be normalized)
    tau_schedule: list of tau values to use (decreasing)
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    x = np.log(a)
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def obj_grad(x):
            J, g = smooth_obj_grad_logspace(x, tau, n)
            return J, g
        
        result = minimize(obj_grad, x, method='L-BFGS-B',

                                             'ftol': 1e-10, 'gtol': 1e-8})
        
        x_new = result.x
        a_new = np.exp(x_new - x_new.max())
        a_new /= a_new.sum()
        
        J_new = eval_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        
        # Update x for next iteration
        x = x_new.copy()
        
        if verbose:
            print(f"    tau={tau:.4f}: J_smooth={result.fun:.6f}, J_actual={J_new:.6f}, "
                  f"iters={result.nit}, time={time.monotonic()-t0:.1f}s")
    
    return best_a, best_J

def mirror_descent_lse(a0, time_limit, top_k=None, tau=0.05, c0=3.0,
                       restart_every=5000, verbose=False):
    """Nesterov mirror descent with log-sum-exp subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_tau = tau
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy
        
        # Compute subgradient using log-sum-exp
        conv = fast_conv(y)
        lse, w = soft_max_and_weights(conv, cur_tau)
        
        # Subgradient of lse w.r.t. a
        c = compute_cross_corr(y, w)
        g = 2.0 * c - 4.0 * lse  # Gradient of J at S=1: 2n*c - 4n*lse, but we can drop the n factor for the direction
        
        # Actually, for the direction, we just need the gradient of lse (since J = 2n*lse/S^2 and S=1)
        # grad_direction = 2*c (gradient of lse w.r.t. a)
        # Let's use the full gradient for correctness
        g = 2.0 * n * c - 4.0 * n * lse
        # Normalize the gradient for stable stepping
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Step
        eta = c0 / (np.sqrt(t) + 1.0)
        
        # Mirror step (KL)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()
        
        # Track best
        val = eval_J(a_new)
        if val < best_J - 1e-14:
            best_J = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Adapt tau: decrease it over time (sharper)
        if t % 1000 == 0 and cur_tau > 0.005:
            cur_tau = max(0.005, cur_tau * 0.95)
        
        # Restart
        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.01, 0.05)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            cur_tau = max(0.02, cur_tau * 1.5)  # Sharpen on restart
            if verbose:
                print(f"    MD restart t={t}: J={best_J:.6f}, tau={cur_tau:.4f}")
        
        if verbose and t % 5000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, tau={cur_tau:.4f}")
    
    return best_a, best_J

def block_lbfgs(a0, time_limit, block_size=200, n_blocks=10, verbose=False):
    """
    Block coordinate L-BFGS: optimize blocks of variables sequentially.
    This helps escape local minima by exploring different parts of the space.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Create random block ordering
    order = rng.permutation(n)
    blocks = [order[i:i+block_size] for i in range(0, n, block_size)]
    
    tau = 0.03
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 3 * n_blocks:
        for block_idx, block in enumerate(blocks):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Define sub-objective for this block
            def block_obj_dx(dx):
                a_trial = a.copy()
                a_trial[block] = dx
                a_trial = np.maximum(a_trial, 1e-15)
                a_trial /= a_trial.sum()
                conv = fast_conv(a_trial)
                lse, _ = soft_max_and_weights(conv, tau)
                return 2.0 * n * lse  # J at S=1
            
            # Initial values for the block
            x0 = a[block].copy()
            
            result = minimize(block_obj_dx, x0, method='L-BFGS-B',
                            bounds=[(1e-15, None)] * len(block),
                            options={'maxiter': 30, 'ftol': 1e-12})
            
            if result.success or result.fun < block_obj_dx(x0):
                a[block] = np.maximum(result.x, 1e-15)
                a = a / a.sum()
                J_new = eval_J(a)
                if J_new < best_J - 1e-12:
                    best_J = J_new
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        
        # Shuffle blocks
        rng.shuffle(blocks)
        if verbose:
            print(f"    Block L-BFGS ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")
    
    return best_a, best_J

def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()

def make_bimodal(n, width=80.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.1) ** 2) + np.exp(-width * (x - 0.9) ** 2) + 0.005
    return a / a.sum()

def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 10):
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_triangular(n, sharpness=1.0):
    """Triangular-like distribution with tunable sharpness."""
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0) ** sharpness
    a = np.maximum(a, 0.01)
    return a / a.sum()

def make_arcosh(n, center=0.5, width=0.3):
    """Arcoshine-based distribution (related to flat-spectra filters)."""
    x = np.linspace(0, 1, n)
    a = np.arccos(1.0 - 2.0 * np.abs(x - center)) / (np.pi / 2)
    a = a ** 2 + 0.01
    return a / a.sum()

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smooth approximation with continuation method
2. L-BFGS-B optimization in log-space (softmax parameterization)
3. Mirror descent with full log-sum-exp subgradient
4. Block coordinate L-BFGS for fine-tuning
5. Multi-strategy initialization, spectral init, and length sweeping
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = fast_conv(a)
    S = np.sum(a)
    if S < 1e-15:
        return float('inf')
    return float(2.0 * n * conv.max() / (S * S))


def soft_max_and_weights(conv, tau):
    """Compute log-sum-exp approximation of max and the softmax weights."""
    m = conv.max()
    z = (conv - m) / tau
    ez = np.exp(z)
    ez_sum = ez.sum()
    lse = m + tau * np.log(ez_sum)
    w = ez / ez_sum
    return lse, w


def compute_cross_corr(a, w):
    """Compute c_j = sum_l a_l * w_{j+l} for j=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    conv_len = 3 * n - 2
    fsize = 1 << (conv_len - 1).bit_length()
    Ar = np.fft.rfft(a_rev, fsize)
    W = np.fft.rfft(w, fsize)
    c_full = np.fft.irfft(Ar * W, fsize)
    return c_full[n - 1: n - 1 + n]


def smooth_obj_grad_logspace(x, tau):
    """
    Objective and gradient in log-space for the smooth objective.
    x: log-space variables, length n
    a = softmax(x) normalized to sum=1
    Returns (J, grad_x)
    """
    n = len(x)
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()

    conv = fast_conv(a)
    lse, w = soft_max_and_weights(conv, tau)
    J = 2.0 * n * lse

    c = compute_cross_corr(a, w)
    # dJ/da_j at S=1: 2n * dM/da_j - 4n*M = 2n*(2*c_j) - 4n*lse = 4n*c_j - 4n*lse
    grad_J_da = 4.0 * n * c - 4.0 * n * lse

    # dF/dx_i = a_i * dJ/da_i (softmax chain rule + Euler's theorem)
    grad_x = a * grad_J_da

    return J, grad_x


def optimize_lbfgs_logspace(a_init, tau_schedule, max_iter, time_limit, verbose=False):
    """
    Optimize using L-BFGS-B in log-space with continuation on tau.
    a_init: initial sequence (will be normalized)
    tau_schedule: list of tau values to use (decreasing)
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()

    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit * 0.95:
            break

        x = np.log(a)

        def obj_grad(xx):
            return smooth_obj_grad_logspace(xx, tau)

        try:
            result = minimize(obj_grad, x, method='L-BFGS-B',
                              jac=True, options={'maxiter': max_iter,
                                                 'maxfun': max_iter * 3,
                                                 'ftol': 1e-12, 'gtol': 1e-9,
                                                 'maxls': 30})
            x_new = result.x
            a_new = np.exp(x_new - x_new.max())
            a_new /= a_new.sum()
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()

            J_new = eval_J(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
                a = a_new.copy()

            if verbose:
                print(f"      tau={tau:.4f}: J_smooth={result.fun:.8f}, "
                      f"J_actual={J_new:.8f}, iters={result.nit}, "
                      f"status={result.status}, t={time.monotonic()-t0:.1f}s")
        except Exception as e:
            if verbose:
                print(f"      tau={tau:.4f}: L-BFGS error: {e}")

    return best_a, best_J


def mirror_descent_lse(a0, time_limit, tau_init=0.08, c0=3.0,
                       restart_every=5000, verbose=False):
    """Nesterov mirror descent with log-sum-exp subgradient and adaptive tau."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_tau = tau_init

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        y = y / sy

        # Compute subgradient using log-sum-exp
        conv = fast_conv(y)
        lse, w = soft_max_and_weights(conv, cur_tau)
        c = compute_cross_corr(y, w)
        # Gradient of J at S=1: 4n*c - 4n*lse
        g = 4.0 * n * c - 4.0 * n * lse
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-20:
            g = g / g_norm
        else:
            break

        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)

        # Mirror step (KL projection onto simplex)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        # Track best
        val = eval_J(a_new)
        if val < best_J - 1e-14:
            best_J = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Adapt tau: decrease over time
        if t % 2000 == 0 and cur_tau > 0.003:
            cur_tau = max(0.003, cur_tau * 0.92)

        # Restart
        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.005, 0.04)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            cur_tau = max(0.02, cur_tau * 2.0)
            if verbose:
                print(f"      MD restart t={t}: J={best_J:.6f}, tau={cur_tau:.4f}")

        if verbose and t % 5000 == 0:
            print(f"      MD t={t} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, tau={cur_tau:.4f}")

    return best_a, best_J


def block_lbfgs(a0, time_limit, block_size=150, verbose=False):
    """
    Block coordinate L-BFGS: optimize blocks of variables sequentially.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_J = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    order = rng.permutation(n)
    blocks = [order[i:i + block_size] for i in range(0, n, block_size)]

    tau = 0.02
    stall = 0
    max_stall = 4 * len(blocks)
    n_passes = 0

    while time.monotonic() - t0 < time_limit * 0.95 and stall < max_stall:
        n_passes += 1
        for block in blocks:
            if time.monotonic() - t0 > time_limit * 0.95:
                break

            def block_obj(dx):
                a_trial = a.copy()
                a_trial[block] = dx
                a_trial = np.maximum(a_trial, 1e-15)
                a_trial /= a_trial.sum()
                conv = fast_conv(a_trial)
                lse, _ = soft_max_and_weights(conv, tau)
                return 2.0 * n * lse

            x0 = a[block].copy()
            try:
                result = minimize(block_obj, x0, method='L-BFGS-B',
                                  bounds=[(1e-15, None)] * len(block),
                                  options={'maxiter': 20, 'ftol': 1e-12,
                                           'maxls': 10})
                if result.fun < block_obj(x0) - 1e-14:
                    a[block] = np.maximum(result.x, 1e-15)
                    a = a / a.sum()
                    J_new = eval_J(a)
                    if J_new < best_J - 1e-12:
                        best_J = J_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
            except Exception:
                stall += 1

        rng.shuffle(blocks)
        # Adapt tau
        if n_passes % 3 == 0 and tau > 0.005:
            tau = max(0.005, tau * 0.85)

        if verbose and n_passes % 2 == 0:
            print(f"      Block pass {n_passes} ({time.monotonic()-t0:.0f}s): "
                  f"J={best_J:.6f}, tau={tau:.4f}, stall={stall}")

    return best_a, best_J


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=80.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.1) ** 2) + np.exp(-width * (x - 0.9) ** 2) + 0.005
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 10):
        coeff = rng.normal(0, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_triangular(n, sharpness=1.0):
    """Triangular-like distribution."""
    x = np.linspace(0, 1, n)
    a = 1.0 - np.abs(2.0 * x - 1.0) ** sharpness
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_spectral_init(n, seed=0):
    """Flat-spectrum initialization: random phase, IFFT, threshold to positive."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    # Random phase in frequency domain
    phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    mag = np.ones(fft_size // 2 + 1)
    A = mag * np.exp(1j * phase)
    # IFFT
    a_full = np.fft.irfft(A, fft_size)
    a = np.abs(a_full[:n])
    a = np.maximum(a, 0.0)
    a += 0.01
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_asymmetric(n, p=1.0, peak=0.3):
    """Asymmetric power distribution with peaked一侧."""
    x = np.linspace(0, 1, n)
    a = (x / peak) ** p * (1.0 - x) ** 0.3 + 0.01
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    a_i += 1e-10
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def propose_candidate(budget_s=1800.0):
    """Main optimization driver with multi-phase strategy."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        J = eval_J(a)
        if J < best_score:
            best_score = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize best
    if parent is not None:
        best_a = parent.copy()
        best_score = eval_J(parent)
    else:
        n0 = 3000
        best_a = make_power_ramp(n0, 1.0)
        best_score = eval_J(best_a)
        print(f"Initial: n={len(best_a)}, J={best_score:.6f}")

    n0 = len(best_a)
    print(f"Starting: n={n0}, J={best_score:.6f}")

    # === Phase 1: L-BFGS-B sweep on diverse initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: L-BFGS-B Sweep ===")

    inits = []
    labels = []
    if parent is not None:
        inits.append(parent.copy())
        labels.append("parent")
    inits.append(make_power_ramp(n0, 0.3))
    labels.append("ramp0.3")
    inits.append(make_power_ramp(n0, 0.7))
    labels.append("ramp0.7")
    inits.append(make_power_ramp(n0, 1.0))
    labels.append("ramp1.0")
    inits.append(make_power_ramp(n0, 1.5))
    labels.append("ramp1.5")
    inits.append(make_power_ramp(n0, 2.0))
    labels.append("ramp2.0")
    inits.append(make_triangular(n0, 1.0))
    labels.append("tri1.0")
    inits.append(make_triangular(n0, 0.5))
    labels.append("tri0.5")
    inits.append(make_bimodal(n0, 60.0))
    labels.append("bimodal60")
    inits.append(make_bimodal(n0, 120.0))
    labels.append("bimodal120")
    inits.append(make_spectral_init(n0, seed=1))
    labels.append("spectral1")
    inits.append(make_spectral_init(n0, seed=42))
    labels.append("spectral42")
    inits.append(make_smooth_random(n0, seed=5))
    labels.append("smooth5")
    inits.append(make_asymmetric(n0, p=1.0, peak=0.3))
    labels.append("asym0.3")
    inits.append(make_asymmetric(n0, p=1.5, peak=0.7))
    labels.append("asym0.7")

    if parent is not None:
        blend1 = 0.6 * parent + 0.4 * make_power_ramp(n0, 1.5)
        inits.append(blend1 / blend1.sum())
        labels.append("blend1")
        blend2 = 0.8 * parent + 0.2 * make_triangular(n0, 0.7)
        inits.append(blend2 / blend2.sum())
        labels.append("blend2")

    t_per = max(10.0, remaining * 0.30 / len(inits))
    for idx, (init, lbl) in enumerate(zip(inits, labels)):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 180:
            break
        t_avail = min(t_per, time_budget - 180 - elapsed)
        if t_avail < 8:
            break
        print(f"  L-BFGS [{lbl}] ({t_avail:.0f}s)...")
        tau_sched = [0.12, 0.06, 0.03, 0.015, 0.008]
        a_out, _ = optimize_lbfgs_logspace(init, tau_sched, max_iter=150,
                                            time_limit=t_avail, verbose=False)
        try_update(a_out, f"p1_{lbl}")

    # === Phase 2: Deep L-BFGS-B on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Deep L-BFGS-B (n={len(best_a)}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        tau_sched = [0.08, 0.05, 0.03, 0.02, 0.012, 0.007, 0.004, 0.002]
        a_out, _ = optimize_lbfgs_logspace(best_a.copy(), tau_sched,
                                            max_iter=400, time_limit=t2, verbose=True)
        try_update(a_out, "p2_deep")

    # === Phase 3: Mirror descent exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Mirror Descent (n={len(best_a)}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_out, _ = mirror_descent_lse(best_a.copy(), t3, tau_init=0.06, c0=3.0,
                                       restart_every=4000, verbose=True)
        try_update(a_out, "p3_md")

    # === Phase 4: Block L-BFGS fine-tuning ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Block L-BFGS (n={len(best_a)}) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = block_lbfgs(best_a.copy(), t4, block_size=100, verbose=True)
        try_update(a_out, "p4_block")

    # === Phase 5: L-BFGS after MD ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: L-BFGS after MD ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 15:
        tau_sched = [0.04, 0.025, 0.015, 0.008, 0.004]
        a_out, _ = optimize_lbfgs_logspace(best_a.copy(), tau_sched,
                                            max_iter=300, time_limit=t5, verbose=True)
        try_update(a_out, "p5_lbfgs2")

    # === Phase 6: Length sweep ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Length Sweep ===")
    n_best = len(best_a)
    lengths_to_try = []
    for d in [-400, -300, -200, -100, 100, 200, 300, 400, 500]:
        nl = n_best + d
        if 1500 <= nl <= 5500:
            lengths_to_try.append(nl)
    for nl in [2000, 2500, 3000, 3500, 4000, 4500]:
        if nl not in lengths_to_try and 1500 <= nl <= 5500:
            lengths_to_try.append(nl)

    for nl in sorted(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if abs(nl - n_best) <= 30:
            continue
        print(f"  Length n={nl}...")
        a_int = interpolate_seq(best_a, nl)
        t_md = min(remaining * 0.25, 80.0)
        if t_md > 10:
            tau_sched = [0.08, 0.04, 0.02, 0.01]
            a_out, _ = optimize_lbfgs_logspace(a_int, tau_sched, max_iter=200,
                                                time_limit=t_md, verbose=False)
            try_update(a_out, f"p6_len{nl}")

    # === Phase 7: Final deep L-BFGS polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_a)
    print(f"\n=== Phase 7: Final L-BFGS Polish (n={n_cur}) ===")
    t7 = min(remaining * 0.15, 180.0)
    if t7 > 20:
        tau_sched = [0.03, 0.02, 0.012, 0.007, 0.004, 0.002, 0.001]
        a_out, _ = optimize_lbfgs_logspace(best_a.copy(), tau_sched,
                                            max_iter=600, time_limit=t7, verbose=True)
        try_update(a_out, "p7_polish")

    # === Phase 8: Final mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final MD (n={len(best_a)}) ===")
    t8 = min(remaining * 0.10, 120.0)
    if t8 > 15:
        a_out, _ = mirror_descent_lse(best_a.copy(), t8, tau_init=0.04, c0=2.5,
                                       restart_every=6000, verbose=True)
        try_update(a_out, "p8_md")

    # === Phase 9: Fine random perturbations ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 9: Fine Perturbations ===")
        n_fin = len(best_a)
        a_fin = best_a.copy()
        rng = np.random.default_rng(31415)
        n_pert = min(2000, int(remaining * 5))

        for i in range(n_pert):
            if time.monotonic() - t_global > time_budget - 10:
                break
            k = int(rng.integers(3, 50))
            idx = rng.choice(n_fin, size=k, replace=False)
            scale = rng.uniform(0.0005, 0.008)
            a_try = a_fin.copy()
            a_try[idx] *= (1.0 + rng.normal(0, scale, k))
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            J_try = eval_J(a_try)
            if J_try < best_score:
                best_score = J_try
                best_a = a_try.copy()
                a_fin = a_try.copy()
                if i % 50 == 0:
                    print(f"    Pert {i}: J={best_score:.6f}")

        # One more L-BFGS on the perturbed result
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            tau_sched = [0.008, 0.005, 0.003, 0.002]
            a_out, _ = optimize_lbfgs_logspace(best_a.copy(), tau_sched,
                                                max_iter=300, time_limit=remaining - 5,
                                                verbose=False)
            try_update(a_out, "p9_final")

    # Final output
    if best_a is not None:
        final_J = eval_J(best_a)
        best_score = min(best_score, final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={best_score:.6f}, n={len(best_a)}, time={elapsed:.0f}s")
        return [float(x) for x in best_a]
    else:
        n_fb = 3000
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
```
