def fast_conv(a):
    """Compute full convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def lse_obj_and_grad_a(a, tau):
    """Compute LSE-smoothed max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    ac = fast_conv(a)
    
    max_ac = ac.max()
    shifted = ac - max_ac
    exp_shifted = np.exp(tau * shifted)
    log_sum = np.log(exp_shifted.sum())
    f = max_ac + log_sum / tau
    
    # w_k = exp(tau * ac_k) / sum exp(tau * ac_k)
    w = exp_shifted / exp_shifted.sum()
    
    # grad_a[j] = 2 * sum_m a[m] * w[m+j] for m=0..n-1 (w zero-padded outside [0,2n-2])
    # = 2 * cross_corr(a, w)[j] where cross_corr(a,w)[j] = sum_m a[m]*w[m+j]
    # Computed as: conv(flip(a), w)[j+n-1]
    
    fa = a[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    FA = np.fft.rfft(fa, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_full = np.fft.irfft(FA * W, fft_size)[:3 * n - 2]
    g = conv_full[n - 1:n - 1 + n]  # g[j] = sum_m a[m]*w[m+j] for j=0..n-1
    
    grad_a = 2.0 * g
    return f, grad_a, w, ac


def lse_grad_theta(theta, tau):
    """Compute gradient of LSE-smoothed objective w.r.t. theta (softmax parameter)."""
    n = len(theta)
    # Softmax
    t_max = theta.max()
    a = np.exp(theta - t_max)
    a = a / a.sum()
    
    f_val, grad_a, w, ac = lse_obj_and_grad_a(a, tau)
    
    # df/dtheta_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
    mean_grad = np.dot(a, grad_a)
    grad_theta = a * (grad_a - mean_grad)
    
    return f_val, grad_theta


def lse_optimize(a0, time_limit, verbose=True, rng_seed=42):
    """Nesterov-accelerated gradient descent on LSE-smoothed objective with softmax parameterization."""
    n = len(a0)
    a0 = norm_sum1(a0)
    
    # Initialize theta from a0
    t_max = a0.max()
    if t_max < 1e-30:
        theta = np.zeros(n)
    else:
        theta = np.log(a0 / a0.sum())
    
    best_a = a0.copy()
    best_obj = 2.0 * n * fast_conv(a0).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    # Nesterov state
    x = theta.copy()  # current
    x_prev = theta.copy()  # previous
    
    # Adaptive tau
    tau = 10.0
    tau_final = 500.0
    tau_schedule_total = time_limit * 0.8  # ramp up tau over 80% of time
    
    # Step size
    # L (Lipschitz constant) is roughly tau * 2 * max_eig
    # For now, use adaptive step size
    lr = 1.0 / n  # initial
    lr_min = 1e-6
    
    stall = 0
    max_stall = 500
    
    for it in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Update tau
        elapsed_frac = (time.monotonic() - t0) / tau_schedule_total
        tau = tau + (tau_final - tau) * min(1.0, elapsed_frac)
        
        # Nesterov lookahead
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / 2.0
        beta = (it - 1.0) / (it + 1.5)  # Simpler momentum
        y = x + beta * (x - x_prev)
        
        # Gradient at y
        t_max_y = y.max()
        a_y = np.exp(y - t_max_y)
        s_y = a_y.sum()
        if s_y < 1e-30:
            a_y = np.ones(n) / n
        else:
            a_y = a_y / s_y
        
        f_y, grad_y, w, ac = lse_obj_and_grad_a(a_y, tau)
        mean_gy = np.dot(a_y, grad_y)
        grad_theta_y = a_y * (grad_y - mean_gy)
        
        # Step
        x_prev = x.copy()
        x = x - lr * grad_theta_y
        
        # Periodic re-normalization (the softmax should keep this bounded, but just in case)
        if it % 100 == 0:
            t_max_x = x.max()
            a_x = np.exp(x - t_max_x)
            s_x = a_x.sum()
            if s_x < 1e-30 or not np.isfinite(s_x):
                x = theta.copy()
                x_prev = x.copy()
                lr = lr * 0.5
                continue
            a_x = a_x / s_x
            
            obj_x = 2.0 * n * fast_conv(a_x).max()
            if obj_x < best_obj:
                best_obj = obj_x
                best_a = a_x.copy()
                stall = 0
            else:
                stall += 1
            
            # Adapt learning rate
            if stall > 200:
                lr = max(lr * 0.7, lr_min)
                # Perturb
                if stall > max_stall:
                    pert = rng.normal(0, 0.1, n)
                    x = x + pert * a_x
                    stall = 0
            
            if verbose and it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE it={it} ({elapsed:.0f}s): obj={obj_x:.6f}, tau={tau:.1f}, lr={lr:.2e}, stall={stall}")
    
    return best_a, best_obj / (2.0 * n)  # Return M = max(conv)


def lp_frank_wolfe_full(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with ALL constraints (or as many as solver allows)."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    n_constraints = 2 * n - 1
    
    # Try full constraints first, fall back to top-K
    use_full = (n <= 2500)
    K_sel = n_constraints if use_full else min(1500, n_constraints)
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(y)
        M_curr = ac.max()
        
        # Select constraints
        if use_full or K_sel >= n_constraints:
            binding_idx = np.arange(n_constraints)
        else:
            margins = M_curr - ac
            binding_idx = np.argsort(margins)[:K_sel]
        
        # Build constraint matrix A where A[k, j] = y[k-j] if valid (for the LP variable direction)
        # Actually, the FW step solves: max sum(b) s.t. A @ b <= M, b >= 0
        # where A[k, j] represents the (j-th) "derivative" direction for constraint k
        # The constraint is: sum_j b[j] * y[k-j] <= M_curr for all k in binding_idx
        # Wait, that's not right either.
        
        # In the FW step, we want to find b (direction) that maximizes sum(b) 
        # subject to: for all k in binding_idx: conv(y + t*b, y + t*b)[k] <= M_curr + t*L_k
        # At t=0: conv(y,y)[k] + 2t * cross_term <= M_curr
        # cross_term[k] = sum_j b[j] * y[k-j]
        # So constraint: cross_term[k] <= (M_curr - conv(y,y)[k]) / (2t) for small t
        # If we set t=1: cross_term[k] <= M_curr - ac[k]
        
        # So the LP is: max sum(b) s.t. for all k in binding: sum_j b[j]*y[k-j] <= (M_curr - ac[k]), b >= 0
        # The RHS can be negative for binding constraints (where ac[k] is close to M_curr), 
        # which would make the infeasible unless b=0.
        
        # Actually, let me reconsider. The standard FW step for minimizing f(y) = max_k q_k(y)
        # where q_k(y) = sum_i y[i]*y[k-i] (quadratic):
        # Linearize: q_k(y + t*d) ≈ q_k(y) + 2t * sum_j d[j]*y[k-j]
        # We want: q_k(y) + 2t * sum_j d[j]*y[k-j] <= M_curr for all binding k
        # => sum_j d[j]*y[k-j] <= (M_curr - q_k(y)) / (2t)
        # For binding k: q_k(y) ≈ M_curr, so RHS ≈ 0
        # For non-binding k: q_k(y) < M_curr, so RHS > 0
        
        # The FW direction is: max sum(d) s.t. sum_j d[j]*y[k-j] <= (M_curr - ac[k])/(2t) for binding k, d >= 0
        # Normalizing: we can set t=1 and use RHS = (M_curr - ac[k])/2
        
        # But this gives a direction that's very small for binding constraints.
        # A better formulation (as in the previous code): 
        # max sum(b) s.t. sum_j b[j]*y[k-j] <= M_curr for all k, b >= 0
        # This is a relaxation that doesn't account for the current value.
        
        # Let me use the formulation from the previous code:
        # max sum(b) s.t. T @ b <= M, b >= 0
        # where T[k, j] = y[k-j] if 0 <= k-j < n, else 0
        
        # Build T (sparse or dense)
        if n <= 2000 and K_sel <= 2000:
            T_dense = np.zeros((len(binding_idx), n))
            for ii, k in enumerate(binding_idx):
                ki = int(k)
                j_start = max(0, ki - n + 1)
                j_end = min(n, ki + 1)
                for j in range(j_start, j_end):
                    T_dense[ii, j] = y[k - j]
        else:
            # Use sparse
            import scipy.sparse as sp
            rows = []
            cols = []
            vals = []
            for ii, k in enumerate(binding_idx):
                ki = int(k)
                j_start = max(0, ki - n + 1)
                j_end = min(n, ki + 1)
                for j in range(j_start, j_end):
                    rows.append(ii)
                    cols.append(j)
                    vals.append(y[k - j])
            T_dense = sp.csr_matrix((vals, (rows, cols)), shape=(len(binding_idx), n))
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_dense @ b_var <= M_curr * np.ones(len(binding_idx))]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if verbose and t % 10 == 0:
                print(f"    LP solve failed at iter {t}")
            # Fallback: reduce K
            if use_full:
                use_full = False
                K_sel = min(1000, n_constraints)
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov FW update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_conv(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_conv(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}")
        
        # Increase K if not using full
        if not use_full and t % 10 == 0:
            K_sel = min(K_sel + 100, n_constraints)
            if K_sel >= n_constraints:
                use_full = True
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123):
    """Coordinate descent with sensitivity-based moves."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 100
    n_top = 10
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_top]
        
        # Sensitivity
        sens = np.zeros(n)
        in_peak = np.zeros(n, dtype=bool)
        for k in top_k[:5]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            in_peak[idx] = True
            sens[idx] += a[ki - idx]
        
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_sens = sens[candidates]
            dst = candidates[np.argmin(cand_sens)]
            
            if dst == src:
                continue
            
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_conv(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def lse_nesterov_optimize(a0, time_limit, verbose=True, rng_seed=42):

    a = a0.copy()

    rng = np.random.default_rng(rng_seed)

        tau = tau + (tau_end - tau) * elapsed_frac

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)

        y_shift = y - y.max()

        FA = np.fft.rfft(fa, fft_size)
        W = np.fft.rfft(w, fft_size)
        conv_full = np.fft.irfft(FA * W, fft_size)[:3 * n - 2]
        g = conv_full[n - 1:2 * n - 1]  # g[j] for j=0..n-1
        grad_a = 2.0 * g
        
        # Mirror descent update in log-space
        eta = eta0 / (np.sqrt(t) + 1.0)
        x = y - eta * grad_a
        
        # Project back: recover a from x
        x_shift = x - x.max()
        a_new = np.exp(x_shift)
        s_new = a_new.sum()
        if s_new < 1e-30 or not np.isfinite(s_new):
            x = y.copy()
            x_shift = x - x.max()
            a_new = np.exp(x_shift)
            s_new = a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / s_new
        
        x_prev = y.copy()
        
        M_new = fast_conv(a_new).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 100:
            eta0 = max(eta0 * 0.9, 0.5)
        
        if stall > max_stall:
            # Restart with perturbation
            pert = rng.normal(0, 0.05, n) * best_a
            a_r = best_a + pert
            a_r = np.maximum(a_r, 1e-10)
            a_r = a_r / a_r.sum()
            x = np.log(a_r)
            x_prev = x.copy()
            stall = 0
            eta0 = 4.0
            tau = max(tau * 0.5, 20.0)
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE-Nest t={t} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, eta0={eta0:.2f}, stall={stall}")
    
    return best_a, best_m
