def proj_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_anti_pal_init(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.uniform(0.5, 1.5, n)
    # Make anti-palindromic: a[j] * a[n-1-j] should be small
    # Strategy: make first half large, second half small
    half = n // 2
    a[:half] = a[:half] * 2.0
    a[half:] = a[half:] * 0.3
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def make_random_init(n, seed=0):
    rng = np.random.default_rng(seed)
    a = rng.uniform(0, 1, n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
