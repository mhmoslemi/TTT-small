from scipy.fft import fft, ifft, next_fast_len


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = next_fast_len(size)
    A = fft(a, N)
    return np.real(ifft(A * A, N)[:size])


def eval_J_normalized(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * np.max(c) / (s * s)


def soft_max_objective_and_grad(a, p):
    """
    Soft-max approximation of max(conv(a,a))/(sum(a))^2.
    Returns (value, gradient) for f_p(a) = (sum_k exp(p*c_k))^(1/p) / (sum a)^2
    """

    c = fast_conv(a)

def lbgfs_softmax(a0, p, max_iter=200, verbose=False):
    """
    L-BFGS optimization of the soft-max objective with fixed p.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    def objective(x):
        x = np.maximum(x, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            return 1e10
        # Re-normalize for stability during optimization
        x_norm = x / sx
        val, _ = soft_max_objective_and_grad(x_norm, p)
        return val
    
    def gradient(x):
        x = np.maximum(x, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            return np.zeros(n)
        x_norm = x / sx
        _, grad = soft_max_objective_and_grad(x_norm, p)
        return grad
    
    # Use penalty method for non-negativity (L-BFGS-B handles bounds)
    result = minimize(
        objective, a, jac=gradient, method='L-BFGS-B',
        bounds=[(0, 1)] * n,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': 500}
    )
    
    a_out = np.maximum(result.x, 0.0)
    s_out = a_out.sum()
    if s_out > 1e-15:
        a_out = a_out / s_out
    
    return a_out


def coordinate_descent_v2(a0, time_limit, verbose=False, seed=42):
    """
    Improved coordinate descent: try moving mass from high-autocorrelation-contributing
    positions to low-contribution positions.
    """
    n = len(a0)

def make_spectral_flat(n, seed=0):
    """
    Generate a sequence with approximately flat power spectrum.
    This minimizes the autocorrelation peak.
    """
    rng = np.random.default_rng(seed)
    # Random phases in frequency domain
    N = next_fast_len(n)
    A = np.zeros(N, dtype=np.complex128)
    # Set random phases with flat magnitude
    mags = rng.uniform(0.5, 1.0, N)
    phases = rng.uniform(0, 2*np.pi, N)
    A = mags * np.exp(1j * phases)
    # Enforce conjugate symmetry for real signal
    for k in range(1, N//2):
        A[N-k] = np.conj(A[k])
    A[0] = np.abs(A[0])  # real DC
    if N % 2 == 0:
        A[N//2] = np.abs(A[N//2])  # real Nyquist
    
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_waterfill(n):
    """
    Water-filling: start uniform, iteratively reduce mass at positions
    that contribute most to the autocorrelation peak.
    """
    a = np.ones(n) / n
    for _ in range(50):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_star - idx]
            # Reduce mass proportional to contribution
            a[idx] *= (1.0 - 0.01 * contrib / max(contrib.max(), 1e-15))
            a = np.maximum(a, 1e-10)
            a /= a.sum()
    return a


def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=50.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.2) ** 2) + np.exp(-width * (x - 0.8) ** 2) + 0.001
    return a / a.sum()


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.3
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full soft-max gradient (all lags) with adaptive sharpening (annealing on p)
2. L-BFGS quasi-Newton optimization on smooth surrogates
3. Nesterov-accelerated mirror descent with full soft-max subgradient
4. Multi-strategy initialization: spectral-flat, water-filling, power ramps, bimodal
5. Coordinate descent with autocorrelation-aware mass transfer
6. Population-based restarts and length sweep
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft, next_fast_len


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = next_fast_len(size)
    A = fft(a, N)
    return np.real(ifft(A * A, N)[:size])


def eval_J_normalized(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2 assuming a is non-negative."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * np.max(c) / (s * s)


def softmax_grad(a, p):
    """
    Compute the gradient of f_p(a) = (sum_k exp(p*c_k))^(1/p) / (sum a)^2
    where c = conv(a, a).
    Returns (value, gradient).
    """
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return 1e10, np.zeros(n)

    c = fast_conv(a)
    c_max = c.max()
    shifted = p * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    S = exp_shifted.sum()
    if S < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = exp_shifted / S
    log_T = c_max + np.log(S) / p
    T = np.exp(log_T)

    # g_i = 2 * sum_k w_k * a_{k-i} (zero-padded cross-correlation)
    N = next_fast_len(3 * n)
    W = fft(w, N)
    A = fft(a, N)
    g = np.real(ifft(W * np.conj(A), N))[:n] * 2.0

    # f = T / s^2, grad = (s*g - 2*T) / s^3
    f = T / (s * s)
    grad = (s * g - 2.0 * T) / (s ** 3)

    return f, grad


def softmax_mirror_descent(a0, time_limit, p_start=2.0, p_end=500.0,
                           n_anneal_phases=8, c0=3.0, verbose=False):
    """
    Mirror descent with full soft-max gradient and geometric p-annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = eval_J_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    p_values = np.geomspace(p_start, p_end, n_anneal_phases)
    total_iters = 40000
    it_per_phase = max(500, total_iters // n_anneal_phases)

    x = a.copy()
    x_prev = a.copy()
    global_iter = 0
    stall = 0

    for phase_idx, p in enumerate(p_values):
        if time.monotonic() - t0 > time_limit:
            break

        for t in range(it_per_phase):
            if time.monotonic() - t0 > time_limit:
                break
            global_iter += 1

            # Nesterov extrapolation
            if t > 1:
                beta_t = (t - 1.0) / (t + 1.0)
                y = x + beta_t * (x - x_prev)
            else:
                y = x.copy()
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy

            _, g = softmax_grad(y, p)

            eta = c0 / (np.sqrt(global_iter) + 1.0)

            log_r = np.log(y) - eta * g
            log_r -= log_r.max()
            a_new = np.exp(log_r)
            a_new = np.maximum(a_new, 1e-30)
            a_new /= a_new.sum()

            alpha_t = 2.0 / (t + 2.0)
            x_prev = x
            x = (1.0 - alpha_t) * x + alpha_t * a_new
            x = np.maximum(x, 1e-15)
            x /= x.sum()

            if global_iter % 30 == 0:
                val = eval_J_normalized(a_new)
                if val < best_val - 1e-14:
                    best_val = val
                    best_a = a_new.copy()
                    stall = 0
                    if verbose:
                        print(f"    SMMD ph={phase_idx} p={p:.1f} it={global_iter} J={best_val:.6f}")
                else:
                    stall += 1

            # Restart if stalled too long within a phase
            if stall > it_per_phase // 2:
                stall = 0
                scale = rng.uniform(0.005, 0.02)
                pert = rng.normal(0, scale, n) * best_a
                x_r = best_a + pert
                x_r = np.maximum(x_r, 1e-12)
                x_r /= x_r.sum()
                x = x_r.copy()
                x_prev = x_r.copy()

        if verbose:
            print(f"    SMMD phase={phase_idx} p={p:.1f} done, best J={eval_J_normalized(best_a):.6f}")

    return best_a, best_val


def lbgfs_run(a0, p, max_iter=150):
    """Run L-BFGS on the soft-max objective with fixed p."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    def obj(x):
        x = np.maximum(x, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            return 1e10
        x_n = x / sx
        val, _ = softmax_grad(x_n, p)
        return val

    def grad(x):
        x = np.maximum(x, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            return np.zeros(n)
        x_n = x / sx
        _, gr = softmax_grad(x_n, p)
        return gr

    try:
        result = minimize(
            obj, a, jac=grad, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': max_iter * 3}
        )
        a_out = np.maximum(result.x, 0.0)
        s_out = a_out.sum()
        if s_out > 1e-15:
            a_out = a_out / s_out
        return a_out
    except Exception:
        return a0.copy()


def coordinate_descent_v2(a0, time_limit, seed=42):
    """Coordinate descent with multi-peak pressure analysis."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = eval_J_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    max_stall = 400

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        top_peaks = np.argsort(c)[::-1][:8]

        # Contribution vector: how much each position contributes to top peaks
        contrib = np.zeros(n)
        for rank, k_p in enumerate(top_peaks[:4]):
            k_p = int(k_p)
            weight = 1.0 / (rank + 1)
            i_s = max(0, k_p - n + 1)
            i_e = min(n - 1, k_p)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                contrib[idx] += weight * a[idx] * a[k_p - idx]

        src_score = contrib * a
        src = int(np.argmax(src_score))
        if a[src] < 1e-10:
            stall += 1
            continue

        # Find best destination
        dst_pressure = np.zeros(n)
        for rank, k_p in enumerate(top_peaks[:4]):
            k_p = int(k_p)
            weight = 1.0 / (rank + 1)
            i_s = max(0, k_p - n + 1)
            i_e = min(n - 1, k_p)
            if i_s <= i_e:
                idx = np.arange(i_s, i_e + 1)
                dst_pressure[idx] += weight * a[k_p - idx]

        dst_pressure[src] = np.inf
        dst = int(np.argmin(dst_pressure))

        # Try transfers
        improved = False
        best_t_J = best_J
        best_t = a
        for frac in [0.003, 0.007, 0.015, 0.03, 0.06, 0.12, 0.25]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0.0)
            a_t /= a_t.sum()
            J_t = eval_J_normalized(a_t)
            if J_t < best_t_J:
                best_t_J = J_t
                best_t = a_t.copy()

        if best_t_J < best_J - 1e-14:
            a = best_t
            best_J = best_t_J
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 25 == 0:
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * rng.uniform(0.002, 0.04)
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a /= a.sum()
                        J_new = eval_J_normalized(a)
                        if J_new < best_J:
                            best_J = J_new
                            best_a = a.copy()
                            stall = 0
                            break

    return best_a, best_J


def make_power_ramp(n, p=1.0):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=50.0):
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.2) ** 2) + np.exp(-width * (x - 0.8) ** 2) + 0.001
    return a / a.sum()


def make_spectral_flat(n, seed=0):
    """Sequence with approximately flat power spectrum (low autocorrelation peak)."""
    rng = np.random.default_rng(seed)
    N = next_fast_len(2 * n)
    A = np.zeros(N, dtype=np.complex128)
    mags = rng.uniform(0.3, 1.0, N)
    phases = rng.uniform(0, 2 * np.pi, N)
    A = mags * np.exp(1j * phases)
    for k in range(1, N // 2):
        A[N - k] = np.conj(A[k])
    A[0] = np.abs(A[0])
    if N % 2 == 0:
        A[N // 2] = np.abs(A[N // 2])
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_waterfill(n):
    """Iteratively reduce mass where autocorrelation peak contribution is highest."""
    a = np.ones(n) / n
    for _ in range(80):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        i_s = max(0, k_star - n + 1)
        i_e = min(n - 1, k_star)
        if i_s <= i_e:
            idx = np.arange(i_s, i_e + 1)
            contrib = a[idx] * a[k_star - idx]
            max_c = contrib.max()
            if max_c > 1e-15:
                a[idx] *= (1.0 - 0.008 * contrib / max_c)
            a = np.maximum(a, 1e-12)
            a /= a.sum()
    return a


def make_smooth_random(n, seed=0):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.4
    for freq in range(1, 12):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s
