def peak_reduction(a0, time_limit, verbose=False):
    """Iterative peak reduction: transfer mass from peak-contributing positions."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()

        conv = fast_conv(a)
        M = conv.max()
        
        # Find the top few lags
        top_k = min(20, 2*n-1)
        top_lags = np.argsort(conv)[::-1][:top_k]

        # and find the position that contributes most to the peak
        # while having the least "elsewhere" impact
        
        improved = False
        for lag in top_lags[:5]:
            k = int(lag)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            
            # Contributions to this lag
            contrib = np.zeros(n)
            for i in range(i_start, i_end + 1):
                j = k - i
                contrib[i] = a[i] * a[j]  # contribution of position i to lag k
                contrib[j] = max(contrib[j], a[i] * a[j])  # also position j contributes
            
            # Find the position with highest mass that contributes to the peak
            # and transfer a small amount to a position that doesn't contribute
            peak_positions = np.where(contrib > 0.5 * contrib.max())[0]
            if len(peak_positions) == 0:
                continue
            
            # Source: highest mass among peak positions
            src = peak_positions[np.argmax(a[peak_positions])]
            if a[src] < 1e-10:
                continue
            
            # Destination: position with lowest total contribution to all top lags
            total_contrib = np.zeros(n)
            for lag2 in top_lags[:10]:
                k2 = int(lag2)
                for i in range(max(0, k2-n+1), min(n-1, k2)+1):
                    j = k2 - i
                    total_contrib[i] += a[i] * a[j]
                    total_contrib[j] += a[i] * a[j]
            
            total_contrib[src] = np.inf
            dst = np.argmin(total_contrib)
            
            # Try transferring
            for frac in [0.001, 0.002, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                a_trial = a.copy()

        else:
            stall = 0
    
    return best_a, best_J

        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 1e-10)
    return a_i / a_i.sum()
