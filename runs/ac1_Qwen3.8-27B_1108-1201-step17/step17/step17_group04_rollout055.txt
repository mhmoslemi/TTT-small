```python
    """Compute log(sum(exp(x/T))) * T as a smooth max."""
```

```python
    n = len(a)
    size = 2 * n - 1  # length of autocorrelation
    # FFT size for autocorrelation
    fft_size_ac = 1 << (size - 1).bit_length()
    
    # Compute autocorrelation
    A = np.fft.fft(a, fft_size_ac)
    ac = np.fft.ifft(A * np.conj(A)).real[:size]
    # Wait, autocorrelation is ifft(FFT(a)^2) not ifft(FFT(a)*conj(FFT(a)))
    # FFT(a)*conj(FFT(a)) = |FFT(a)|^2 which gives the power spectrum
    # ifft(|FFT(a)|^2) gives the circular autocorrelation
    # For linear autocorrelation, we need ifft(FFT(a, N)^2) with N >= 2n-1
    A = np.fft.fft(a, fft_size_ac)
    ac = np.fft.ifft(A * A).real[:size]
    
    # Compute softmax weights
    ac_max = ac.max()
    shifted = (ac - ac_max) / temperature
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    weights_sum = weights.sum()
    weights = weights / weights_sum
    
    # Compute cross-correlation: C[m] = sum_j weights[m+j] * a[j] for m = 0, ..., n-1
    # N >= len(weights) + len(a) - 1 = (2n-1) + n - 1 = 3n - 2
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.fft(weights, N)
    A2 = np.fft.fft(a, N)
    C = np.fft.ifft(W * np.conj(A2)).real[:n]

    # Wait, weights are normalized to sum to 1, so weights_sum = 1
    grad = 2.0 * (C + a)
    
    return grad, ac
```

Let me also think about the **projection onto the simplex** more carefully.

The projection of a vector v onto the simplex {x : sum(x) = 1, x >= 0} is:
x_i = max(v_i - tau, 0) where tau is chosen so that sum(x) = 1.

This can be found by sorting v and doing a binary search on tau.

Now, for the Adam optimizer with projection:
1. Compute gradient g of the smooth objective
2. Update Adam moments: m = beta1*m + (1-beta1)*g, v = beta2*v + (1-beta2)*g^2
3. Compute update: delta = lr * m_hat / (sqrt(v_hat) + eps)
4. Tentative update: a_new = a - delta
5. Project: a_projected = simplex_project(a_new)
6. Set a = a_projected

The issue with projection + Adam is that the projection changes the effective step size and direction. An alternative is to use **mirror descent** with the entropy regularizer (which has a natural projection onto the simplex via the softmax).

Mirror descent with entropy:
1. Compute gradient g
2. Update: a_new[i] = exp(log(a[i]) - eta * g[i])
3. Normalize: a = a_new / sum(a_new)

This is what the previous code does (called "exponential" step). It automatically maintains non-negativity and can be easily normalized.

The advantage of mirror descent over projected gradient is that it respects the exponential structure of the probability simplex.

**Mirror descent**:
- a_{t+1} = softmax(log(a_t) - eta_t * grad_f(a_t))
- Where softmax here means: a_{t+1}[i] = exp(log(a_t)[i] - eta_t * grad[i]) / sum_j exp(log(a_t)[j] - eta_t * grad[j])

- Polyak: eta_t = (f(a_t) - f*) / ||grad||^2 (but we don't know f*)
- Adaptive: use a running average of gradient norms

For the smooth objective (log-sum-exp), the gradient is Lipschitz continuous, so standard convergence results apply.

```python
def adam_mirror_descent(a0, time_limit, temperature_start=1.0, temperature_end=0.01,
                       lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                       restart_every=5000, seed=42):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Adaptive temperature
        progress = min(t / restart_every, 1.0)
        temperature = temperature_start * (temperature_end / temperature_start) ** progress
        
        # Compute smooth gradient
        grad, ac = compute_smooth_grad_fast(a, temperature)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate
        step_size = lr / (np.sqrt(v_hat) + eps)
        
        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-300)) - step_size
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        # Evaluate
        J = compute_J(a)
        if J < best_J - 1e-14:
            best_J = J
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.01, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
    
    return best_a, best_J
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam + mirror descent with adaptive log-sum-exp smoothing
2. FFT-based efficient gradient computation for the smooth surrogate
3. Coordinate-wise minimax refinement (exact 1D subproblem per coordinate)
4. Multi-start with diverse parametric initializations
5. Exact subgradient Nesterov for final polish
Key innovation: combining smooth Adam optimization (for global structure) with
exact coordinate-wise refinement (for local precision), using FFT to compute
gradients in O(n log n) instead of O(n^2).
"""

import numpy as np
import time

# ============ Core Computation Functions ============

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective assuming sum(a)=1. Returns 2*n*max(autocorr)."""
    n = len(a)
    ac = fast_autocorr(a)
    return 2.0 * n * ac.max()


def simplex_project(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find(u * np.arange(1, n + 1) > (cssv - 1))
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1) / (rho[-1] + 1)
    return np.maximum(v - theta, 0)


def compute_smooth_grad_fft(a, temperature):
    """
    Compute gradient of T*log(sum(exp(ac/T))) using FFT.
    Returns (gradient, autocorrelation, smooth_value).
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Compute autocorrelation
    A = np.fft.rfft(a, fft_size)
    ac = np.fft.irfft(A * A, fft_size)[:size]

    # Compute softmax weights for the smooth max
    ac_max = ac.max()
    if ac_max < 1e-30:
        return np.zeros(n), ac, 0.0
    shifted = (ac - ac_max) / temperature
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    weights_sum = weights.sum()
    if weights_sum < 1e-30:
        weights = np.ones(size) / size
        weights_sum = 1.0
    else:
        weights = weights / weights_sum

    smooth_val = ac_max + temperature * np.log(weights_sum)

    # Compute gradient: grad[m] = 2 * (C[m] + a[m])
    # where C[m] = sum_j weights[j+m] * a[j] (cross-correlation)
    N_cross = 1 << (size + n - 2 - 1).bit_length()
    W = np.fft.rfft(weights, N_cross)
    A2 = np.fft.rfft(a, N_cross)
    # Cross-correlation: C[m] = sum_j w[j+m] * a[j]
    # = ifft(fft(w) * conj(fft(a))) gives sum_j w[(m+j) mod N] * a[j]
    # For N >= len(w) + len(a) - 1, this equals the linear cross-correlation
    cross = np.fft.irfft(W * np.conj(A2), N_cross)[:n]

    grad = 2.0 * (cross + a)  # since sum(weights) = 1
    return grad, ac, smooth_val


def get_exact_subgradient(a, ac):
    """Compute exact subgradient of max(autocorr(a))."""
    n = len(a)
    k = int(np.argmax(ac))
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n, k + 1)
    idx = np.arange(i_start, i_end)
    j_idx = k - idx
    grad[idx] = 2.0 * a[j_idx]
    return grad


def get_soft_subgradient(a, ac, top_k=20, beta=3.0):
    """Compute soft subgradient using top-k peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


# ============ Optimization Methods ============

def adam_mirror_descent(a0, time_limit, temp_start=0.5, temp_end=0.005,
                        lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                        restart_every=3000, verbose=False, seed=42):
    """Adam + mirror descent with adaptive log-sum-exp temperature."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Adaptive temperature: exponential decay with periodic boosts
        progress = min(t / max(restart_every, 1), 1.0)
        temperature = temp_start * (temp_end / temp_start) ** progress
        # Add small temperature floor
        temperature = max(temperature, temp_end)

        # Compute smooth gradient
        grad, ac, _ = compute_smooth_grad_fft(a, temperature)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Adaptive step size
        step_size = lr * 0.5 ** (t // 5000)
        delta = step_size * m_hat / (np.sqrt(v_hat) + eps)

        # Mirror descent update (exponential)
        log_a = np.log(np.maximum(a, 1e-300)) - delta
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new

        # Evaluate
        J = compute_J(a)
        if J < best_J - 1e-14:
            best_J = J
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Restart with adaptation
        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.005 * min(stall // restart_every, 5)
            pert = rng.normal(0, sigma, n) * best_a
            a = best_a + pert
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            lr *= 0.7  # Reduce learning rate on restart

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AMD t={t} ({elapsed:.0f}s): J={best_J:.6f}, temp={temperature:.4f}")

    return best_a, best_J


def exact_nesterov(a0, time_limit, top_k=15, beta=3.0, c=3.0,
                   exact_frac=0.4, restart_every=2000, seed=42):
    """Nesterov-accelerated subgradient with adaptive exact/soft mixing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    cur_c = c
    cur_exact_frac = exact_frac

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = x.copy()
        else:
            y = y / s

        ac = fast_autocorr(y)
        M = ac.max()

        # Adaptive subgradient
        use_exact = rng.random() < cur_exact_frac
        if use_exact:
            g = get_exact_subgradient(y, ac)
        else:
            g = get_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        # Polyak-like step size
        lower_bound = 1.0 / n
        gap = M - lower_bound
        eta = cur_c * gap / (np.dot(g, g) + 1e-20)
        eta = min(eta, 2.0 / (t + 1.0))
        eta = max(eta, 1e-8)

        # Mirror descent step
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        J_new = compute_J(x)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Restart with adaptation
        if stall > restart_every:
            stall = 0
            sigma = 0.005 + 0.005 * min(stall // restart_every, 5)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 5, 60)
            cur_beta = max(cur_beta - 0.2, 1.5)
            cur_c = min(cur_c + 0.3, 8.0)
            cur_exact_frac = min(cur_exact_frac + 0.05, 0.7)

    return best_a, best_J


def coordinate_refinement(a0, time_limit, verbose=False, seed=123):
    """
    Coordinate-wise minimax refinement.
    For each coordinate, find the best value that minimizes the max autocorrelation.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()
        improved = False
        n_improved = 0

        # Process coordinates in random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            a_old = a[i]
            if a_old < 1e-12:
                # Try setting this coordinate to a small positive value
                candidates = [1e-6, a.mean() * 0.1, a.mean() * 0.5]
            else:
                # Generate candidates around current value
                candidates = [
                    0.0,
                    a_old * 0.25,
                    a_old * 0.5,
                    a_old * 0.75,
                    a_old * 1.0,
                    a_old * 1.25,
                    a_old * 1.5,
                    a_old * 2.0,
                ]
                # Add some random candidates
                candidates.append(a_old * rng.uniform(0.1, 3.0))
                # Add value based on sensitivity
                # The "sensitivity" of coordinate i is roughly a[i] * (sum of relevant a values)
                sensitivity = a_old * a.max()
                candidates.append(sensitivity * 0.5)

            best_a_new = a_old
            best_M = M_curr

            for a_new in candidates:
                a_new = max(a_new, 0.0)
                delta = a_new - a_old
                if abs(delta) < 1e-15:
                    continue

                # Compute new max autocorrelation efficiently
                # ac[k] changes by delta * a[k-i] for k where k-i is in [0, n-1]
                # Plus quadratic correction at k = 2*i
                k_start = i
                k_end = min(i + n, 2 * n - 1)

                # Unaffected max
                unaffected_max = 0.0
                if k_start > 0:
                    unaffected_max = max(unaffected_max, ac[:k_start].max())
                if k_end < 2 * n - 1:
                    unaffected_max = max(unaffected_max, ac[k_end:].max())

                if k_start < k_end:
                    # Affected lags: new_ac[k] = ac[k] + delta * a[k-i]
                    affected = ac[k_start:k_end] + delta * a[:k_end - k_start]
                    # Quadratic correction at k = 2*i (if in range)
                    if 2 * i >= k_start and 2 * i < k_end:
                        local_idx = 2 * i - k_start
                        # Correct: new_ac[2i] should be ac[2i] - a_old^2 + a_new^2
                        # Linear gives: ac[2i] + delta * a[i] = ac[2i] + (a_new-a_old)*a_old
                        # Correct: ac[2i] + a_new^2 - a_old^2
                        # Difference: a_new^2 - a_old^2 - (a_new-a_old)*a_old = a_new*(a_new-a_old)
                        affected[local_idx] += a_new * (a_new - a_old)
                    affected_max = affected.max()
                else:
                    affected_max = 0.0

                new_M = max(unaffected_max, affected_max)

                if new_M < best_M - 1e-15:
                    best_M = new_M
                    best_a_new = a_new

            # Apply best move
            if best_a_new != a_old:
                a[i] = best_a_new
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                improved = True
                n_improved += 1

        # Check improvement after full sweep
        J = compute_J(a)
        if J < best_J - 1e-14:
            best_J = J
            best_a = a.copy()

        if not improved:
            # Try random perturbation to escape
            n_pert = max(10, n // 200)
            idx = rng.choice(n, size=n_pert, replace=False)
            for p in idx:
                if a[p] > 1e-8:
                    frac = rng.uniform(0.05, 0.3)
                    amt = a[p] * frac
                    a[p] -= amt
                    a[rng.integers(n)] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
            else:
                a = best_a.copy()
                if sweep > 5:
                    break

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordRef sweep={sweep} ({elapsed:.0f}s): J={best_J:.6f}")

    return best_a, best_J


def peak_relocation(a0, time_limit, verbose=False, seed=77):
    """
    Peak relocation: identify peak contributors and relocate mass to safe zones.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Get top peaks
        n_peaks = min(10, len(ac))
        top_lags = np.argsort(ac)[::-1][:n_peaks]

        # Compute sensitivity
        sensitivity = np.zeros(n)
        for k in top_lags[:6]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            contribution = a[idx] * a[j_idx]
            for ii, jj, cc in zip(idx, j_idx, contribution):
                sensitivity[ii] += cc
                sensitivity[jj] += cc

        high_sens = np.where((sensitivity > sensitivity.mean() + sensitivity.std()) & (a > 0.001))[0]
        low_sens = np.where((sensitivity < sensitivity.mean() - 0.3 * sensitivity.std()) | (sensitivity == 0))[0]

        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            continue

        improved = False
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            src = high_sens[rng.integers(len(high_sens))]
            dst = low_sens[rng.integers(len(low_sens))]
            if src == dst:
                continue

            src_val = a[src]
            if src_val < 1e-8:
                continue

            fracs = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.35]
            found = False
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    break

            if found:
                stall = 0

        if not improved:
            stall += 1
            if stall % 20 == 0:
                n_pert = max(5, n // 100)
                pert_pos = rng.choice(n, size=n_pert, replace=False)
                for p in pert_pos:
                    if a[p] > 1e-8:
                        frac = rng.uniform(0.01, 0.1)
                        amt = a[p] * frac
                        a[p] -= amt
                        a[rng.integers(n)] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                else:
                    a = best_a.copy()

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRel ({elapsed:.0f}s): J={best_J:.6f}, stall={stall}")

    return best_a, best_J


# ============ Initialization Functions ============

def make_exponential_init(n, alpha=0.005, both_sides=True):
    """Exponential decay from edges."""
    x = np.arange(n) / n
    if both_sides:
        a = np.exp(-alpha * n * x) + np.exp(-alpha * n * (1 - x)) + 0.01
    else:
        a = np.exp(-alpha * n * x) + 0.01
    return norm_sum1(a)


def make_glassmann_init(n, power=1.0):
    """Glassman-based: a[i] ~ (i+1)^(-power) type distribution."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** (-power)
    a += a[::-1]  # Symmetric
    return norm_sum1(a)


def make_bimodal_init(n, center_frac=0.3, width=0.15):
    """Bimodal: two bumps away from center."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center_frac) / width) ** 2) + np.exp(-((x - (1 - center_frac)) / width) ** 2) + 0.005
    return norm_sum1(a)


def make_one_sided_init(n, decay=0.5):
    """One-sided: concentrated at the beginning with power-law decay."""
    x = np.arange(n, dtype=np.float64)
    a = (x + 1) ** (-decay)
    return norm_sum1(a)


def make_random_smooth_init(n, seed=42, n_freq=5):
    """Random smooth initialization using Fourier basis."""
    rng = np.random.default_rng(seed)
    freqs = rng.choice(range(1, n // 2 + 1), size=n_freq, replace=False)
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    amps = rng.uniform(0.1, 1.0, n_freq)
    x = np.linspace(0, 2 * np.pi, n)
    a = np.ones(n) * 0.5
    for f, p, amp in zip(freqs, phases, amps):
        a += amp * np.cos(f * x + p)
    a = np.abs(a) + 0.01
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = norm_sum1(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


# ============ Main Search Function ============

def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Multi-start Adam mirror descent with diverse inits (50% time)
    Phase 2: Exact Nesterov subgradient refinement (20% time)
    Phase 3: Coordinate-wise refinement (15% time)
    Phase 4: Peak relocation + final polish (15% time)
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_sequence([float(x) for x in a_norm])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(x) for x in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_exponential_init(n_base, both_sides=True)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: Multi-start Adam Mirror Descent ===
    print(f"\n=== Phase 1: Adam Mirror Descent (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.50, 800.0)

    if t_phase1 > 60:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        inits.append(make_exponential_init(n0, alpha=0.003, both_sides=True))
        labels.append("exp_both")

        inits.append(make_exponential_init(n0, alpha=0.008, both_sides=False))
        labels.append("exp_one")

        inits.append(make_bimodal_init(n0, center_frac=0.25, width=0.12))
        labels.append("bimodal")

        inits.append(make_glassmann_init(n0, power=0.5))
        labels.append("glassmann")

        inits.append(make_one_sided_init(n0, decay=0.7))
        labels.append("one_sided")

        inits.append(make_random_smooth_init(n0, seed=1, n_freq=7))
        labels.append("rand_smooth1")

        inits.append(make_random_smooth_init(n0, seed=2, n_freq=10))
        labels.append("rand_smooth2")

        # Blend parent with various inits
        if parent is not None:
            for frac in [0.3, 0.7]:
                blend = frac * norm_sum1(parent) + (1 - frac) * make_exponential_init(n0, alpha=0.005)
                inits.append(norm_sum1(blend))
                labels.append(f"blend_{int(frac*100)}")

        # Remove near-duplicates
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for u in unique_inits:
                if np.sum(np.abs(init - u)) < 0.03:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        n_inits = len(unique_inits)
        print(f"  {n_inits} unique initializations")
        t_per = max(30.0, t_phase1 / n_inits)

        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_avail = min(t_per, remaining - 60)
            if t_avail < 20:
                break
            print(f"  AMD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = adam_mirror_descent(init, t_avail,
                                           temp_start=0.3, temp_end=0.005,
                                           lr=0.04, restart_every=4000,
                                           verbose=False, seed=42 + idx * 13)
            try_update(a_out, f"amd_{lbl}")

    # === Phase 2: Exact Nesterov Subgradient ===
    print(f"\n=== Phase 2: Exact Nesterov (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = exact_nesterov(a_in, t2, top_k=15, beta=3.0, c=3.0,
                                  exact_frac=0.4, restart_every=2000, seed=555)
        try_update(a_out, f"nesterov_n{len(a_in)}")

    # === Phase 3: Coordinate Refinement ===
    print(f"\n=== Phase 3: Coordinate Refinement (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_refinement(a_in, t3, verbose=True, seed=999)
        try_update(a_out, f"coord_n{len(a_in)}")

    # === Phase 4: Peak Relocation ===
    print(f"\n=== Phase 4: Peak Relocation (n={len(best_seq)}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_relocation(a_in, t4, verbose=True, seed=333)
        try_update(a_out, f"peak_n{len(a_in)}")

    # === Phase 5: Final Adam polish (low temperature) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Final AMD polish (n={len(best_seq)}) ===")
    t5 = min(remaining * 0.20, 200.0)
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = adam_mirror_descent(a_in, t5,
                                       temp_start=0.05, temp_end=0.002,
                                       lr=0.02, restart_every=5000,
                                       verbose=False, seed=7777)
        try_update(a_out, f"amd_polish_n{len(a_in)}")

    # === Phase 6: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length Exploration ===")

    for new_n in [n_best + 200, n_best + 500]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.3, 150.0)
        if t6a > 30:
            a_out, _ = adam_mirror_descent(a_long, t6a,
                                           temp_start=0.2, temp_end=0.005,
                                           lr=0.03, restart_every=3000,
                                           verbose=False, seed=2000)
            try_update(a_out, f"amd_n{new_n}")

    # === Phase 7: Final coordinate refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final Coord Ref (n={len(best_seq)}) ===")
    t7 = min(remaining * 0.3, 150.0)
    if t7 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_refinement(a_in, t7, verbose=True, seed=4444)
        try_update(a_out, f"coord_final_n{len(a_in)}")

    # === Phase 8: One more Nesterov if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final Nesterov (n={n_best}) ===")
        t8 = min(remaining * 0.5, 100.0)
        if t8 > 15:
            a_in = norm_sum1(best_seq)
            a_out, _ = exact_nesterov(a_in, t8, top_k=25, beta=2.5, c=4.0,
                                      exact_frac=0.5, restart_every=5000, seed=8888)
            try_update(a_out, f"nesterov_final_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        final_score = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_exponential_init(n_fb, both_sides=True)
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
