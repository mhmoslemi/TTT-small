def compute_full_subgrad(a, b, threshold_frac=0.001):
    """Compute full subgradient of max(b) where b = conv(a,a).
    Uses all lags within threshold_frac of the maximum."""
    n = len(a)

        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * a[k_int - idx]
    
    return g

def lp_remez_step(a, max_cons=50, r=0.1, solver=None):
    """Single LP Remez step."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # More active constraints for better accuracy
    threshold = max_b * 0.999
    active = np.where(b >= threshold)[0]
    if len(active) > max_cons:
        active = np.argsort(b)[::-1][:max_cons]
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        if solver:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    return a_new / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):

    if kind == 'triangle':

    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':

    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    
    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            coeffs = 2.0 * a[k_int - idx]
            constraints.append(cp.sum(cp.multiply(coeffs, a_var[idx])) - M_var <= b[k_int])
    
    constraints.append(a_var <= a * (1.0 + r))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)

            prob.solve(verbose=False)

    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    return a_new / s

                a_trial = a_trial / a_trial.sum()
                m_trial = fast_conv(a_trial).max()
                if m_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = m_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_init(n, kind='triangle'):
    x = np.linspace(0.0, 1.0, n)
    if kind == 'triangle':
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    elif kind == 'raised_cos':
        a = (1.0 + np.cos(2.0 * np.pi * x)) / 2.0
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.1)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'quartic':
        a = (1.0 - (2.0 * x - 1.0) ** 4)
    elif kind == 'sin':
        a = np.sin(np.pi * x)
    elif kind == 'bimodal':
        a = np.exp(-20.0 * (x - 0.3) ** 2) + np.exp(-20.0 * (x - 0.7) ** 2)
    elif kind == 'water':
        a = 1.0 - 0.5 * np.sin(2.0 * np.pi * x)
    else:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def update_best(a, best_seq, best_score, label, t_global):
    """Try to update best. Returns (new_best_seq, new_best_score)."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)}, t={elapsed:.0f}s)")
    return best_seq, best_score

def propose_candidate(budget_s=1800.0):
    """Main search orchestrator with full-subgradient Nesterov and improved Remez."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    n_base = 2600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = make_init(n_base, 'triangle')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    n0 = len(best_seq)
    
    # === Phase 1: Full-subgradient Nesterov from parent (35% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: NMD full-subgrad n={n0} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_mirror_descent_v2(norm_sum1(best_seq), t1, threshold_frac=0.001, c0=3.0, use_backtrack=True, verbose=True)
        best_seq, best_score = update_best(a_out, best_seq, best_score, f"NMD_v2_n{n0}", t_global)
    
    # === Phase 2: LP Remez (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: LP Remez n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = remez_method_v2(norm_sum1(best_seq), t2, solver=solver, r_init=0.15, verbose=True)
        best_seq, best_score = update_best(a_out, best_seq, best_score, f"Remez_n{len(best_seq)}", t_global)
    
    # === Phase 3: Multi-init exploration (15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3_total = min(remaining * 0.15, 200.0)
    if t3_total > 60:
        print(f"\n=== Phase 3: Multi-init Nesterov (budget={t3_total:.0f}s) ===")
        inits = ['raised_cos', 'gaussian', 'sin', 'bimodal', 'water']
        t_per_init = t3_total / len(inits)
        for kind in inits:
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 15:
                break
            a_init = make_init(n0, kind)
            a_out, _ = nesterov_mirror_descent_v2(a_init, min(t_per_init, remaining - 5),
                                                 threshold_frac=0.002, c0=4.0, use_backtrack=False, verbose=False)
            best_seq, best_score = update_best(a_out, best_seq, best_score, f"NMD_{kind}_n{n0}", t_global)
    
    # === Phase 4: Tight-threshold polish Nesterov (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Polish Nesterov (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_mirror_descent_v2(norm_sum1(best_seq), t4, threshold_frac=0.0001, c0=1.5, use_backtrack=True, verbose=True)
        best_seq, best_score = update_best(a_out, best_seq, best_score, f"Polish_n{len(best_seq)}", t_global)

    t5 = min(remaining * 0.10, 120.0)

    # === Phase 6: Coordinate descent (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.05, 60.0)
    print(f"\n=== Phase 6: CD polish (budget={t6:.0f}s) ===")
    if t6 > 10:

        best_seq, best_score = update_best(a_out, best_seq, best_score, f"CD_{len(best_seq)}", t_global)
    
    # === Phase 7: Alt length exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n0 + 100
        print(f"\n=== Phase 7: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
        a_alt = interpolate_sequence(best_seq, n_alt)
        
        t7a = min(remaining * 0.6, 150.0)
        if t7a > 30:
            a_out, _ = nesterov_mirror_descent_v2(a_alt, t7a, threshold_frac=0.001, c0=3.0, use_backtrack=True, verbose=True)
            best_seq, best_score = update_best(a_out, best_seq, best_score, f"NMD_n{n_alt}", t_global)
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.5, 80.0)
            if t7b > 15:
                a_out, _ = remez_method_v2(norm_sum1(best_seq), t7b, solver=solver, r_init=0.05, verbose=False)
                best_seq, best_score = update_best(a_out, best_seq, best_score, f"Remez_alt_{len(best_seq)}", t_global)
    
    # === Phase 8: Final alternating push (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 8: Final alternating push n={n_best} (budget={remaining:.0f}s) ===")
        while remaining > 15:
            t_chunk_nmd = min(25.0, remaining * 0.5)
            if t_chunk_nmd > 10:
                a_out, _ = nesterov_mirror_descent_v2(norm_sum1(best_seq), t_chunk_nmd,
                                                     threshold_frac=0.00005, c0=0.8, use_backtrack=True, verbose=False)
                best_seq, best_score = update_best(a_out, best_seq, best_score, f"Final_NMD_{n_best}", t_global)
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                t_chunk_remez = min(20.0, remaining * 0.5)
                if t_chunk_remez > 8:
                    a_out, _ = remez_method_v2(norm_sum1(best_seq), t_chunk_remez,
                                              solver=solver, r_init=0.02, verbose=False)
                    best_seq, best_score = update_best(a_out, best_seq, best_score, f"Final_Remez_{n_best}", t_global)
            
            remaining = time_budget - (time.monotonic() - t_global)
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
