def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_M(a):
    """Compute max autocorrelation."""
    return fast_autocorr(a).max()


def norm_sum1(a):
    """Normalize to sum 1, enforce non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_J(a):
    """Compute objective: 2*n*max(conv) when sum=1."""
    n = len(a)
    return 2.0 * n * compute_M(a)


def get_subgradient(a, ac, exact=False, top_k=10, beta=3.0):
    """Compute subgradient of max(conv(a,a))."""
    n = len(a)
    grad = np.zeros(n)
    if exact:
        k = int(np.argmax(ac))
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k - idx
        grad[idx] = 2.0 * a[j_idx]
    else:
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def lp_step(a, M, top_K, solver, warm_b=None):
    """
    Solve LP: max sum(b) s.t. (a*b)(k) <= M for top-K lags, b >= 0.
    Returns the optimal b vector (normalized to sum 1).
    """
    n = len(a)
    ac = fast_autocorr(a)
    margins = M - ac
    K_sel = min(top_K, 2 * n - 1)
    binding_idx = np.argsort(margins)[:K_sel]

    # Build constraint matrix rows for selected lags
    # (a*b)(k) = sum_j a[k-j] * b_j for valid j
    T = np.zeros((K_sel, n))
    for ii, k in enumerate(binding_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        T[ii, idx] = a[j_idx]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

            break

        # Nesterov extrapolation

            y = x.copy()

        # Mirror step

            for _ in range(8):
                mid = (lo + hi) / 2
                a_trial = a.copy()
                a_trial[i] = mid
                # Redistribute the difference
                diff = current_val - mid
                if abs(diff) > 1e-12:
                    # Add to average of other coordinates
                    others = np.ones(n)
                    others[i] = 0
                    a_trial += diff * others / (n - 1)
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                M_trial = compute_M(a_trial)
                if M_trial < best_local_M:
                    best_local_M = M_trial
                    best_local_val = mid
                    hi = mid
                else:
                    lo = mid

            if best_local_val != current_val and best_local_M < best_M - 1e-14:
                a[i] = best_local_val
                diff = current_val - best_local_val
                if abs(diff) > 1e-12:
                    others = np.ones(n)
                    others[i] = 0
                    a += diff * others / (n - 1)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                best_M = best_local_M
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            # Perturb to escape
            if stall % 50 == 0:
                n_pert = max(5, n // 50)
                idx = rng.choice(n, size=n_pert, replace=False)
                for i in idx:
                    a[i] *= rng.uniform(0.7, 1.3)
                a = np.maximum(a, 0)
                a = a / a.sum()
                M_new = compute_M(a)
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): M={best_M:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordDesc final ({elapsed:.0f}s): M={best_M:.6f}")
    return best_a, best_M


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safe wrapper for evaluate_sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining LP-FW, subgradient, peak-chasing,
    and coordinate descent methods with adaptive time allocation.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        x = np.linspace(0, 1, n_base)
        best_seq = np.cos(np.pi * x) ** 2 + 0.01
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LP Frank-Wolfe (main driver) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 1: LP-FW Nesterov (n={n0}, t={t1:.0f}s) ===")
    if t1 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_nesterov(a_in, t1, solver, verbose=True,
                                            initial_K=300, max_K=1200,
                                            K_grow_rate=1.04)
        try_update(a_out, "lpfw1")

        # Second LP-FW run with different parameters
        remaining = time_budget - (time.monotonic() - t_global)
        t1b = min(remaining * 0.15, 200.0)
        if t1b > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_nesterov(a_in, t1b, solver, verbose=True,
                                                initial_K=500, max_K=2000,
                                                K_grow_rate=1.02)
            try_update(a_out, "lpfw1b")

    # === Phase 2: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: Subgradient polish (n={n_best}, t={t2:.0f}s) ===")
    if t2 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = subgradient_polish(a_in, t2, verbose=True, seed=42)
        try_update(a_out, "sg_polish")

    # === Phase 3: Peak chasing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Peak chasing (n={n_best}, t={t3:.0f}s) ===")
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_chase_a(a_in, t3, verbose=True, seed=77)
        try_update(a_out, "peakchase")

    # === Phase 4: Another LP-FW round ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 4: LP-FW round 2 (n={n_best}, t={t4:.0f}s) ===")
    if t4 > 30 and n_best <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe_nesterov(a_in, t4, solver, verbose=True,
                                            initial_K=600, max_K=2500,
                                            K_grow_rate=1.01)
        try_update(a_out, "lpfw2")

    # === Phase 5: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 5: Coordinate descent (n={n_best}, t={t5:.0f}s) ===")
    if t5 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_descent_peaks(a_in, t5, verbose=True, seed=333)
        try_update(a_out, "coorddesc")

    # === Phase 6: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    for delta in [200, -200, 400]:
        new_n = n_best + delta
        if new_n < 1000 or new_n > 5000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)
        # Quick LP-FW
        t6a = min(remaining * 0.25, 150.0)
        if t6a > 30:
            a_out, _ = lp_frank_wolfe_nesterov(a_long, t6a, solver, verbose=False,
                                                initial_K=300, max_K=1000,
                                                K_grow_rate=1.05)
            try_update(a_out, f"lpfw_n{new_n}")

        # Quick subgradient
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_in = interpolate_sequence(best_seq, new_n)
            a_out, _ = subgradient_polish(a_in, min(remaining * 0.2, 80.0),
                                           verbose=False, seed=555)
            try_update(a_out, f"sg_n{new_n}")

    # === Phase 7: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish (n={n_best}) ===")
    if remaining > 60:
        # LP-FW
        t7a = min(remaining * 0.4, 200.0)
        if t7a > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = lp_frank_wolfe_nesterov(a_in, t7a, solver, verbose=True,
                                                initial_K=800, max_K=3000,
                                                K_grow_rate=1.005)
            try_update(a_out, "lpfw_final")

        # Subgradient
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_in = norm_sum1(best_seq)
            a_out, _ = subgradient_polish(a_in, min(remaining * 0.4, 150.0),
                                           verbose=True, seed=999)
            try_update(a_out, "sg_final")

        # Peak chase
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_chase_a(a_in, min(remaining * 0.4, 100.0),
                                     verbose=True, seed=1234)
            try_update(a_out, "peakchase_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
