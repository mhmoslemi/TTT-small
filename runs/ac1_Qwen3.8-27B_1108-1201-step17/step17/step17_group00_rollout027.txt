```python
def compute_weighted_gradient(a, lags, weights):
    """Compute the weighted sum of gradients at given lags.
    grad[i] = sum_l weights[l] * 2*a[l-i] (for valid indices)"""
    n = len(a)
    grad = np.zeros(n)
    for w, k in zip(weights, lags):
        k = int(k)
        # grad[i] += 2*w*a[k-i] for 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2 * w * a[k - idx]
    return grad
```

This is O(K*n) per iteration which could be slow for large K and n. But with FFT, we can do it in O(n log n + K*n) for K lags.

Actually, for a single lag k, the gradient is just 2*a[k-i] for valid i, which is a shifted copy of a. For multiple lags with different weights, it's a sum of shifted copies. If K is small (say 10-50), this is fast enough.

Actually, let me reconsider the gradient formula. 

J(a) = 2n * max_k c[k] / S^2

where c[k] = sum_j a[j]*a[k-j] and S = sum_j a[j].

If we fix S = 1 (by normalization, since J is scale-invariant), then:
J(a) = 2n * max_k c[k]

And ∂J/∂a[i] = 2n * ∂(max_k c[k])/∂a[i]

Using the max subgradient: if the max is attained at lags k_1, ..., k_m with equal values:
∂(max_k c[k])/∂a[i] = (1/m) * sum_l ∂c[k_l]/∂a[i]

For a single lag k:
∂c[k]/∂a[i] = 2*a[k-i] (if 0 <= k-i < n)

So ∂J/∂a[i] = 2n * (1/m) * sum_l 2*a[k_l - i] (for valid indices)
= (4n/m) * sum_l a[k_l - i]

To minimize J, we take a step in the -gradient direction, projected onto the constraint set {a >= 0, sum(a) = 1}.

With multiplicative updates (mirror descent):
a[i] <- a[i] * exp(-eta * grad[i])
a[i] <- max(a[i], 0)
a <- a / sum(a)

This is the "AdaGrad-like" multiplicative update that maintains non-negativity.

Let me also think about what happens at the optimum. At the optimum, the gradient should be "balanced" - meaning the positions that contribute to the peak should be reduced until the peak lags are all equal (the max is attained at multiple lags simultaneously with equal values).

This is analogous to the "water-filling" principle: we pour water (mass) into the sequence, and the peak lags are like containers that we try to level.

One more thought: I should also consider that the "peak" might shift as we optimize. So at each iteration, I should recompute which lags are the peaks.

For efficiency, I'll keep a small set of "active" lags (the top K by value) and update it periodically.

Let me also think about the step size. With multiplicative updates a[i] <- a[i] * exp(-eta * grad[i]), the effective step size depends on the magnitude of grad. I should normalize the gradient or use an adaptive step size.

A simple approach:
- Compute grad
- Normalize: grad = grad / max(grad) (so the largest component has magnitude 1)
- eta = initial_eta (e.g., 0.01-0.1)
- After the update, if J increased, reduce eta; if decreased, keep or slightly increase eta

Let me also add a "temperature" schedule: start with larger steps (exploration) and decrease over time (exploitation).

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-lag multiplicative gradient descent (mirror flow)
2. Block-level local search with random perturbations  
3. Adaptive multi-length scaling
4. Final coordinate polish

import numpy as np
import time
import math

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def compute_gradient_multi_lag(a, lags, weights):
    """Compute sum_l weights[l] * 2*a[l-i] for valid i."""
    n = len(a)
    grad = np.zeros(n)
    for w, k in zip(weights, lags):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * w * a[k - idx]
    return grad

def multiplicative_gradient_descent(a0, time_limit, top_k=20, eta0=0.05, verbose=True):
    """Main optimization: multiplicative gradient descent on the simplex."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    best_j = 2.0 * n * best_m
    t0 = time.monotonic()
    
    eta = eta0
    patience = 0
    max_patience = 500
    
    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute autocorrelation
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            best_j = 2.0 * n * M
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        # Find top k lags
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = ac[top_idx]
        
        # Weight by value (or by excess over second-largest)
        w = top_vals ** 2  # emphasize the largest peaks
        w = w / w.sum()
        
        # Compute gradient
        grad = compute_gradient_multi_lag(a, top_idx, w)
        
        # Project gradient onto tangent space of simplex (subtract mean)
        grad = grad - grad.mean()
        
        # Scale gradient
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check if improvement
        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()
        
        if M_new <= M + 1e-12:
            a = a_new
            # Slowly decrease eta
            eta *= 0.999
        else:
            # Rejection: reduce eta
            eta *= 0.5
            if eta < 0.001:
                # Try a smaller top_k
                top_k = max(5, top_k // 2)
                eta = 0.02 * (top_k / 20.0)
                
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD iter={it} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.4f}, tk={top_k}")
    
    return best_a, best_j

def block_local_search(a0, time_limit, block_size=20, verbose=True):
    """Block-level local search: perturb small blocks and keep improvements."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        
        improved = False
        
        # Try random block perturbations
        for attempt in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick a random block
            start = int(rng.integers(0, n - block_size + 1))
            end = start + block_size
            block = a[start:end].copy()
            
            # Perturb block
            scale = rng.uniform(0.01, 0.1)
            perturbation = rng.normal(0, scale * block.mean() + 1e-6, block_size)
            new_block = np.maximum(block + perturbation, 0.0)
            
            # Maintain sum
            total_change = new_block.sum() - block.sum()
            if abs(total_change) > 1e-15:
                # Distribute change uniformly (approximately)
                other_sum = a.sum() - block.sum()
                if other_sum > 1e-15:
                    adjustment = -total_change / (n - block_size)
                    a_temp = a.copy()
                    a_temp[start:end] = new_block
                    a_temp[:start] += adjustment
                    a_temp[end:] += adjustment
                    a_temp = np.maximum(a_temp, 0.0)
                    s = a_temp.sum()
                    if s > 1e-15:
                        a_temp = a_temp / s
                        M_temp = fast_autocorr(a_temp).max()
                        if M_temp < M - 1e-14:
                            a = a_temp
                            M = M_temp
                            if M < best_m:
                                best_m = M
                                best_a = a.copy()
                            improved = True
                            break
        
        if not improved:
            stall += 1
        else:
            stall = 0
    
    return best_a, 2.0 * len(best_a) * best_m

def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Fast coordinate descent: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        n_top = min(15, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            cmax = contrib.max()
            if cmax < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination (low contribution to peaks)
            peak_contrib = np.zeros(n)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    peak_contrib[idx2] += a[k2i - idx2]
                    peak_contrib[k2i - idx2] += a[idx2]
            
            # Avoid peak-contributing positions
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2 + 1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            cand_scores = peak_contrib[candidates]
            dst = candidates[np.argmin(cand_scores)]
            if dst == src:
                continue
            
            # Try moving mass
            for frac in [0.01, 0.02, 0.04, 0.08]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, 2.0 * n * best_m
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative gradient descent (mirror flow on simplex) with multi-lag subgradient
2. Adaptive step-size control and periodic restarts
3. Block-level local search with structured perturbations
4. Coordinate descent polish targeting peak-contributing positions
5. Multi-length exploration via interpolation and re-optimization
"""

import numpy as np
import time
import math


def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def compute_gradient_multi_lag(a, lags, weights):
    """
    Compute the weighted subgradient: sum_l weights[l] * 2*a[k_l - i] for valid i.
    This gives d(max_l c[k_l])/da[i] approximately.
    """
    n = len(a)
    grad = np.zeros(n)
    for wi, k in zip(weights, lags):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * wi * a[ki - idx]
    return grad


def mgd_optimize(a0, time_limit, top_k, eta0, verbose):
    """
    Multiplicative Gradient Descent on the simplex.
    Uses multi-lag subgradient with adaptive step size.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = eta0
    patience = 0
    max_patience = 300
    cur_top_k = top_k
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = fast_autocorr(a)
        M = ac.max()

        if M < best_m - 1e-15:
            best_m = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            break

        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** 2.0
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = compute_gradient_multi_lag(a, top_idx, w)
        grad = grad - grad.mean()

        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        ac_new = fast_autocorr(a_new)
        M_new = ac_new.max()

        if M_new <= M + 1e-12:
            a = a_new
            eta *= 0.9995
        else:
            eta *= 0.5
            if eta < 0.0005:
                cur_top_k = max(5, cur_top_k // 2)
                eta = 0.02
                patience = max(0, patience - 100)

        iter_count += 1
        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MGD iter={iter_count} ({elapsed:.0f}s): J={J:.6f}, eta={eta:.5f}, tk={cur_top_k}")

    elapsed = time.monotonic() - t0
    if verbose:
        J = 2.0 * n * best_m
        print(f"    MGD done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a


def block_search(a0, time_limit, block_size, verbose):
    """
    Block-level local search: perturb small blocks and keep improvements.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    stall = 0
    max_stall = 300
    n_blocks = max(1, (n - block_size) // block_size)

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        improved = False

        # Focus perturbations near peak-contributing regions
        top_lags = np.argsort(ac)[::-1][:10]

        for attempt in range(30):
            if time.monotonic() - t0 > time_limit:
                break

            if attempt < 10 and len(top_lags) > 0:
                # Targeted: perturb near a peak-contributing position
                k_pick = int(top_lags[attempt % len(top_lags)])
                js = max(0, k_pick - n + 1)
                je = min(n - 1, k_pick)
                center = (js + je) // 2
                start = max(0, min(n - block_size, center - block_size // 2 + int(rng.integers(-5, 5))))
            else:
                start = int(rng.integers(0, max(1, n - block_size + 1)))

            end = min(n, start + block_size)
            bs = end - start
            if bs < 2:
                continue

            block = a[start:end].copy()
            mean_block = block.mean() + 1e-8
            scale = rng.uniform(0.02, 0.08)
            pert = rng.normal(0, scale * mean_block, bs)
            new_block = np.maximum(block + pert, 0.0)

            if abs(new_block.sum() - block.sum()) < 1e-15:
                continue

            a_temp = a.copy()
            a_temp[start:end] = new_block
            a_temp = np.maximum(a_temp, 0.0)
            s_t = a_temp.sum()
            if s_t < 1e-15:
                continue
            a_temp = a_temp / s_t
            M_temp = fast_autocorr(a_temp).max()

            if M_temp < M - 1e-14:
                a = a_temp
                M = M_temp
                if M < best_m:
                    best_m = M
                    best_a = a.copy()
                improved = True
                if attempt >= 20:
                    break

        if not improved:
            stall += 1
            # Random escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.04)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    Block done ({elapsed:.0f}s): J={J:.6f}")
    return best_a

            a_out = mgd_optimize(norm_sum1(best_seq), t7, top_k=50, eta0=0.008, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final MGD: {sc:.6f}")

    # === Phase 8: Final coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.50, 100.0)
        if t8 > 10:
            a_out = coord_polish(norm_sum1(best_seq), t8, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** NEW BEST final coord: {sc:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        n_fb = 3200
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = norm_sum1(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
```
