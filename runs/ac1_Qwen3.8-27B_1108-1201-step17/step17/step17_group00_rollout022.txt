2. Smoothed subgradient optimization with Nesterov acceleration

    old_x = np.linspace(0.0, 1.0, on)

    best = y.copy()
    best_m = fast_ac(y).max()
    t0 = time.monotonic()

    ac_y = fast_ac(y)
    cs = set(int(k) for k in np.argsort(ac_y)[::-1][:init_K])
    no_lp_fail = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > tlimit:
            break

        ac = fast_ac(y)
        M = ac.max()
        if M < best_m - 1e-14:
            best_m = M
            best = y.copy()

        cl = sorted(cs)
        if len(cl) > max_K:
            margins = np.array([M - ac[k] for k in cl])
            keep = np.argsort(margins)[:max_K]
            cs = set(int(cl[i]) for i in keep)
            cl = sorted(cs)

        g = solve_lp_step(y, cl, n, solver)
        if g is None:
            no_lp_fail += 1
            if no_lp_fail > 3:
                cs = set(int(k) for k in cl[:min(init_K, len(cl))])
                cl = sorted(cs)
                g = solve_lp_step(y, cl, n, solver)
                no_lp_fail = 0
            if g is None:
                continue
        else:
            no_lp_fail = 0

        size = 2 * n - 1
        fs = 1 << (size - 1).bit_length()
        yg = np.fft.irfft(np.fft.rfft(y, fs) * np.fft.rfft(g, fs), fs)[:size]
        viol = np.where(yg > M + 1e-8)[0]
        if len(viol) > 0:
            top_v = viol[np.argsort(yg[viol])[::-1][:add_K]]
            for k in top_v:
                cs.add(int(k))

        best_alpha = 0.0
        best_obj_m = M
        alphas = [0.01, 0.02, 0.04, 0.07, 0.1, 0.15, 0.2, 0.3, 0.5]
        for alpha in alphas:
            cand = (1.0 - alpha) * y + alpha * g
            cand = np.maximum(cand, 0.0)
            sc = cand.sum()
            if sc < 1e-15:
                continue
            cand = cand / sc
            mc = fast_ac(cand).max()
            if mc < best_m - 1e-14:
                best_m = mc
                best = cand.copy()
            if mc < best_obj_m:
                best_obj_m = mc
                best_alpha = alpha

        if best_alpha == 0.0:
            best_alpha = 2.0 / (t + 2.0)
            if best_alpha > 0.5:
                best_alpha = 0.5

        y = (1.0 - best_alpha) * y + best_alpha * g
        y = ns(y)

        if verbose and t % 25 == 0:
            el = time.monotonic() - t0
            print(f"  SS-FW t={t} ({el:.0f}s): J={2.0*n*best_m:.6f}, |S|={len(cs)}, a={best_alpha:.3f}")

    el = time.monotonic() - t0
    J = 2.0 * n * best_m
    print(f"  SS-FW done ({el:.0f}s): J={J:.6f}")
    return best, best_m


def smooth_opt(a0, tlimit, beta_start, beta_end, c0, restart_every, verbose):
    n = len(a0)
    a = ns(a0)
    best = a.copy()
    best_m = fast_ac(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    tk = 50

    for t in range(1, 20000000):
        if time.monotonic() - t0 > tlimit:
            break

        bt = (t - 1.0) / (t + 1.5)
        y = x + bt * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_ac(y)
        M = ac.max()

        beta = beta_start + (beta_end - beta_start) * min(1.0, t / 8000.0)
        ti = np.argsort(ac)[::-1][:tk]
        tv = np.maximum(ac[ti], 1e-30)
        w = np.exp(beta * (tv - tv[0]))
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(ti)) / len(ti)
        else:
            w = w / ws

        grad = np.zeros(n)
        for wi, k in zip(w, ti):
            ki = int(k)
            i0 = max(0, ki - n + 1)
            i1 = min(n, ki + 1)
            if i0 < i1:
                idx = np.arange(i0, i1)
                grad[idx] += 2.0 * wi * y[ki - idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        mn = log_a.max()
        a_new = np.exp(log_a - mn)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x
        x = a_new

        mx = fast_ac(x).max()
        if mx < best_m - 1e-14:
            best_m = mx
            best = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.015, n) * best
            x = ns(best + pert)
            x_prev = x.copy()
            tk = min(tk + 10, 100)
            if verbose:
                el = time.monotonic() - t0
                print(f"  Smooth restart t={t} ({el:.0f}s): J={2.0*n*best_m:.6f}, tk={tk}")

    el = time.monotonic() - t0
    print(f"  Smooth done ({el:.0f}s): J={2.0*n*best_m:.6f}")
    return best, best_m


def coord_polish(a0, tlimit, seed, verbose):
    n = len(a0)
    a = ns(a0)
    best = a.copy()
    best_m = fast_ac(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < tlimit and stall < 400:
        ac = fast_ac(a)
        M = ac.max()
        nk = min(25, len(ac))
        top_k = np.argsort(ac)[::-1][:nk]

        improved = False
        for k in top_k[:10]:
            if time.monotonic() - t0 > tlimit:
                break
            ki = int(k)
            i0 = max(0, ki - n + 1)
            i1 = min(n - 1, ki)
            if i0 > i1:
                continue
            idx = np.arange(i0, i1 + 1)
            contrib = a[idx] * a[ki - idx]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            if a[src] < 1e-12:
                continue

            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    in_peak[s2:e2 + 1] = True

            scores = np.zeros(n)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[k2i - idx2] += a[idx2]

            cands = np.where(~in_peak)[0]
            if len(cands) == 0:
                cands = np.arange(n)
            dst = cands[np.argmin(scores[cands])]

            if dst == src:
                continue

            for frac in [0.003, 0.007, 0.015, 0.03, 0.06, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    break
                at = a.copy()
                at[src] -= amt
                at[dst] += amt
                at = ns(at)
                mt = fast_ac(at).max()
                if mt < best_m - 1e-15:
                    a = at
                    best_m = mt
                    best = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 50:
                n_try = min(20, n)
                for _ in range(n_try):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.025)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = ns(a)
                        mn = fast_ac(a).max()
                        if mn < best_m:
                            best_m = mn
                            best = a.copy()
                            stall = 0
                            break

        if verbose and stall > 0 and stall % 100 == 0:
            el = time.monotonic() - t0
            print(f"  Coord ({el:.0f}s): J={2.0*n*best_m:.6f}, stall={stall}")

    el = time.monotonic() - t0
    print(f"  Coord done ({el:.0f}s): J={2.0*n*best_m:.6f}")
    return best, best_m


def parametric_init(n, best_a, best_score):
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    best = best_a.copy()
    best_v = best_score
    improved = []

    families = []
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        families.append((i_arr + 1.0) ** p)
    for p in [0.5, 1.0, 1.5, 2.0]:
        families.append((n - i_arr) ** p)
    for alpha in [0.5, 1.0, 2.0]:
        for beta in [0.5, 1.0, 2.0]:
            families.append(np.maximum(i_arr + 1.0, 0.5) ** alpha * np.maximum(n - i_arr, 0.5) ** beta)
    for c in [0.3, 0.5, 0.7]:
        for sigma in [0.15, 0.3, 0.5]:
            families.append(np.exp(-((x - c) ** 2) / (2.0 * sigma * sigma)))
    for freq in [1, 2, 3]:
        families.append(np.maximum(np.sin(np.pi * freq * x) ** 2, 0.01))
    for rate in [1, 3, 10]:
        families.append(np.exp(-rate * x))
        families.append(np.exp(rate * x))

    for f in families:
        f = ns(f)
        v = 2.0 * n * fast_ac(f).max()
        if v < best_v:
            best_v = v
            best = f.copy()
            improved.append(v)

    if best_score < 2.0:
        for w in [0.3, 0.5, 0.7]:
            a_pow = (i_arr + 1.0) ** 1.0
            a_pow = ns(a_pow)
            a_blend = w * best_a + (1.0 - w) * a_pow
            a_blend = ns(a_blend)
            v = 2.0 * n * fast_ac(a_blend).max()
            if v < best_v:
                best_v = v
                best = a_blend.copy()

    return best, best_v


def propose_candidate(budget_s=1800.0):
    tG = time.monotonic()
    TB = min(float(budget_s) - 40.0, 1760.0)

    solver = get_solver()
    print(f"Total budget: {TB:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            best_seq = ns(p)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_default = 5000
        x = np.linspace(0, 1, n_default)
        best_seq = ns(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 0: Quick parametric check (only if parent score is poor)
    if best_score > 1.55:
        rem = TB - (time.monotonic() - tG)
        print(f"\n=== Phase 0: Parametric (n={n0}) ===")
        tp = min(rem * 0.05, 60.0)
        if tp > 10:
            a_par, val_par = parametric_init(n0, best_seq, 2.0 * n0 * fast_ac(best_seq).max())
            sc_par = evaluate_safe(a_par)
            if sc_par < best_score:
                best_score = sc_par
                best_seq = a_par.copy()
                print(f"  *** Parametric improved: {sc_par:.6f}")

    # Phase 1: SS-FW main optimization
    rem = TB - (time.monotonic() - tG)
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: SS-FW main (n={n_cur}) ===")
    t1 = min(rem * 0.28, 420.0)
    if t1 > 60 and n_cur <= 6500:
        a_out, m_out = ss_fw(best_seq, t1, solver, init_K=400, max_K=3500, add_K=40, verbose=True)
        sc_out = evaluate_safe(a_out)
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            el = time.monotonic() - tG
            print(f"  *** SS-FW improved: {sc_out:.6f} (t={el:.0f}s)")

    # Phase 2: Smooth optimization
    rem = TB - (time.monotonic() - tG)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Smooth opt (n={n_cur}) ===")
    t2 = min(rem * 0.12, 160.0)
    if t2 > 30:
        a_out, m_out = smooth_opt(best_seq, t2, beta_start=80.0, beta_end=500.0, c0=3.0,
                                  restart_every=5000, verbose=True)
        sc_out = evaluate_safe(a_out)
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            el = time.monotonic() - tG
            print(f"  *** Smooth improved: {sc_out:.6f} (t={el:.0f}s)")

    # Phase 3: Coordinate descent
    rem = TB - (time.monotonic() - tG)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Coord descent (n={n_cur}) ===")
    t3 = min(rem * 0.10, 150.0)
    if t3 > 20:
        a_out, m_out = coord_polish(best_seq, t3, seed=12345, verbose=True)
        sc_out = evaluate_safe(a_out)
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            el = time.monotonic() - tG
            print(f"  *** Coord improved: {sc_out:.6f} (t={el:.0f}s)")

    # Phase 4: Try neighboring lengths
    rem = TB - (time.monotonic() - tG)
    n_cur = len(best_seq)
    lengths_to_try = [n_cur - 300, n_cur + 300, n_cur + 600]
    lengths_to_try = [l for l in lengths_to_try if 2000 <= l <= 7000]

    for new_n in lengths_to_try[:2]:
        rem = TB - (time.monotonic() - tG)
        if rem < 100:
            break
        print(f"\n=== Phase 4: Explore n={new_n} ===")
        a_long = interp_seq(best_seq, new_n)
        sc_interp = evaluate_safe(a_long)
        if sc_interp < best_score:
            best_score = sc_interp
            best_seq = a_long.copy()
            print(f"  Interpolated: {sc_interp:.6f}")

        t4a = min(rem * 0.30, 150.0)
        if t4a > 40:
            a_out, m_out = smooth_opt(a_long, t4a, beta_start=80.0, beta_end=400.0, c0=2.5,
                                      restart_every=4000, verbose=False)
            sc_out = evaluate_safe(a_out)
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                el = time.monotonic() - tG
                print(f"  *** Smooth n={new_n} improved: {sc_out:.6f} (t={el:.0f}s}")

        rem = TB - (time.monotonic() - tG)
        if rem > 60 and new_n <= 5500:
            t4b = min(rem * 0.35, 150.0)
            if t4b > 30:
                a_out, m_out = ss_fw(a_long if len(a_long) == new_n else best_seq,
                                     t4b, solver, init_K=300, max_K=3000, add_K=30, verbose=False)
                sc_out = evaluate_safe(a_out)
                if sc_out < best_score:
                    best_score = sc_out
                    best_seq = a_out.copy()
                    el = time.monotonic() - tG
                    print(f"  *** SS-FW n={new_n} improved: {sc_out:.6f} (t={el:.0f}s}")

    # Phase 5: Second SS-FW round on best
    rem = TB - (time.monotonic() - tG)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: SS-FW round 2 (n={n_cur}) ===")
    if rem > 80 and n_cur <= 6500:
        t5 = min(rem * 0.28, 300.0)
        if t5 > 40:
            a_out, m_out = ss_fw(best_seq, t5, solver, init_K=500, max_K=4000, add_K=50, verbose=True)
            sc_out = evaluate_safe(a_out)
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                el = time.monotonic() - tG
                print(f"  *** SS-FW2 improved: {sc_out:.6f} (t={el:.0f}s)")

    # Phase 6: Final coordinate polish
    rem = TB - (time.monotonic() - tG)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final coord (n={n_cur}) ===")
    if rem > 25:
        t6 = min(rem * 0.55, 200.0)
        if t6 > 15:
            a_out, m_out = coord_polish(best_seq, t6, seed=777, verbose=True)
            sc_out = evaluate_safe(a_out)
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                el = time.monotonic() - tG
                print(f"  *** Final coord improved: {sc_out:.6f} (t={el:.0f}s)")

    # Phase 7: One more short smooth pass
    rem = TB - (time.monotonic() - tG)
    if rem > 30:
        n_cur = len(best_seq)
        print(f"\n=== Phase 7: Final smooth (n={n_cur}) ===")
        t7 = min(rem * 0.40, 120.0)
        if t7 > 20:
            a_out, m_out = smooth_opt(best_seq, t7, beta_start=150.0, beta_end=600.0, c0=2.0,
                                      restart_every=3000, verbose=False)
            sc_out = evaluate_safe(a_out)
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                el = time.monotonic() - tG
                print(f"  *** Final smooth improved: {sc_out:.6f} (t={el:.0f}s)")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    else:
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = ns(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    el = time.monotonic() - tG
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={el:.0f}s")

    return [float(x) for x in best_seq]
