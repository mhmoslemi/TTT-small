def compute_subgradient_topk(a, ac, top_k=50, beta=4.0):
    """Compute weighted subgradient using top-k peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start > j_end:
            continue
        idx = np.arange(j_start, j_end + 1)
        g[idx] += wi * 2.0 * a[ki - idx]
    return g


def lp_active_set_step(a, ac, top_K, solver, max_iter_lp=5):
    """Solve LP linearization to find a descent direction.
    LP: min M s.t. grad_k(a_0)^T a_new <= M + f_k(a_0) for active k,
         a_new >= 0, sum(a_new) = 1
    Returns the LP solution (normalized to sum 1) or None."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    k_max = min(top_K, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    M_curr = ac.max()
    f_vals = np.maximum(ac[top_idx], 1e-30)

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(top_idx):
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        for j in range(j_start, j_end + 1):
            v = 2.0 * a[ki - j]
            if v > 1e-18:
                rows.append(i)
                cols.append(j)
                vals.append(v)

    if len(rows) == 0:
        return None

    C_mat = csr_matrix((vals, (rows, cols)), shape=(k_max, n))

    # LP: min M s.t. C @ a_new - M <= f_vals, a_new >= 0, sum(a_new) = 1
    a_new = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones_k = cp.ones(k_max)
    constraints = [
        C_mat @ a_new - M_var * ones_k <= f_vals,
        cp.sum(a_new) == 1.0
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    a_val = np.maximum(np.asarray(a_new.value, dtype=np.float64), 0.0)
    s = a_val.sum()
    if s < 1e-12:
        return None
    return a_val / s


def amd_optimize(a0, time_limit, solver, top_k=60, beta=5.0, c0=4.0,
                 restart_every=5000, verbose=False, rng_seed=42, use_lp=True, lp_top_k=100):
    """Combined mirror descent + LP step optimization."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    lp_interval = 100
    step = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        step += 1
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_subgradient_topk(y, ac, top_k=cur_top_k, beta=cur_beta)

        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-16:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Periodic LP step
        if use_lp and step % lp_interval == 0 and n <= 5000:
            lp_result = lp_active_set_step(best_a, fast_autocorr(best_a), lp_top_k, solver)
            if lp_result is not None:
                # Blend LP solution with current best
                for alpha in [0.3, 0.5, 0.7, 1.0]:
                    blend = (1 - alpha) * best_a + alpha * lp_result
                    blend = np.maximum(blend, 0.0)
                    sb = blend.sum()
                    if sb < 1e-15:
                        continue
                    blend = blend / sb
                    M_blend = fast_autocorr(blend).max()
                    if M_blend < best_m:
                        best_m = M_blend
                        best_a = blend.copy()
                        x = blend.copy()
                        x_prev = blend.copy()
                        stall_count = 0
                        break

            cur_beta = max(cur_beta * 0.92, 2.0)
            cur_top_k = min(cur_top_k + 3, 300)

        if stall_count > restart_every:
            stall_count = 0
            # Symmetry-breaking perturbation
            pert = rng.normal(0, 0.02, n) * best_a
            # Add asymmetric component

            x_r = best_a + pert + asym
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0

def directed_pair_descent(a0, time_limit, verbose=True, rng_seed=55):
    """Directed coordinate descent: for each active peak, find the most responsible
    pair and try to reduce it by moving mass to low-sensitivity positions."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    max_stall = 300
    n_peaks = 20

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:n_peaks]

        # Build global sensitivity map
        sens = np.zeros(n)
        for k in top_k[:10]:
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            sens[idx] += a[ki - idx]

        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]
            if contrib.max() < 1e-20:
                continue

            # Try reducing top-3 contributing pairs
            top_contrib = np.argsort(contrib)[::-1][:3]
            for tc in top_contrib:
                src = idx[tc]
                src_partner = partner[tc]

                # Determine which to reduce
                imp_src = sens[src]
                imp_part = sens[src_partner]
                if imp_src >= imp_part:
                    reduce_idx = src
                else:
                    reduce_idx = src_partner

                if a[reduce_idx] < 1e-12:
                    continue

                # Find best destination: positions with low sensitivity
                exclude = set()
                if reduce_idx < n:
                    exclude.add(int(reduce_idx))
                if src_partner < n and src_partner != src:
                    exclude.add(int(src_partner))

                # Use low-sensitivity positions
                sorted_sens = np.argsort(sens)
                found = False
                for frac in [0.005, 0.01, 0.02, 0.002, 0.05]:
                    if found:
                        break
                    for dst_pos in sorted_sens[:20]:
                        if int(dst_pos) in exclude:
                            continue
                        amt = a[reduce_idx] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[reduce_idx] -= amt
                        a_trial[dst_pos] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-17:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            improved = True
                            found = True
                            break

            if time.monotonic() - t0 > time_limit:
                break

        if not improved:
            stall += 1
            # Random asymmetric kick
            if stall > 30:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    DirPair ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def make_asymmetric_init(n, seed=0):

    # Add a small bump on the other side
    bump_center = rng.uniform(0.3, 0.7)
    bump_width = rng.uniform(0.05, 0.15)
    a += 0.1 * np.exp(-((x - bump_center) / bump_width) ** 2)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_linear_ramp(n, steepness=1.0):
    """Linearly decreasing sequence (asymmetric)."""
    x = np.linspace(0, 1, n)
    a = 1.0 - steepness * x + 0.1
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_two_bump_asym(n, seed=0):
    """Two Gaussian bumps at asymmetric positions."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    c1 = rng.uniform(0.1, 0.3)
    c2 = rng.uniform(0.6, 0.9)
    w1 = rng.uniform(0.05, 0.12)
    w2 = rng.uniform(0.05, 0.12)
    a = np.exp(-((x - c1) / w1) ** 2) + 0.7 * np.exp(-((x - c2) / w2) ** 2) + 0.002
    return norm_sum1(a)


def make_u_shape(n, sharpness=2.0):
    x = np.linspace(-1, 1, n)
    a = 1.0 + sharpness * x ** 2
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def fine_polish(a0, time_limit, verbose=False):
    """Very fine-grained local search with adaptive step sizes."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_k = np.argsort(ac)[::-1][:15]

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            s = max(0, ki - n + 1)
            e = min(n - 1, ki)
            idx = np.arange(s, e + 1)
            partner = ki - idx
            contrib = a[idx] * a[partner]

            # Top contributing pair
            best_c = np.argmax(contrib)
            src = idx[best_c]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Try moving tiny amounts to random positions
            n_tries = 10
            for _ in range(n_tries):
                j = int(rng.integers(n))
                if j == src:
                    continue
                for frac in [0.001, 0.003, 0.005, 0.0005]:
                    amt = src_val * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-17:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Tiny random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.0005, 0.005)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    FinePolish ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search orchestrator."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm_sum1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 4000
        x = np.linspace(0, 1, n_base)
        best_seq = make_u_shape(n_base, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Multi-init exploration (asymmetric focus) ===
    print(f"\n=== Phase 1: Multi-init (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.15, 250.0)

    if t_phase1 > 40:
        inits = []
        labels = []

        if parent is not None:
            inits.append(norm_sum1(parent))
            labels.append("parent")

        # Asymmetric inits
        inits.append(make_asymmetric_init(n0, seed=1))
        labels.append("asym1")
        inits.append(make_asymmetric_init(n0, seed=2))
        labels.append("asym2")
        inits.append(make_asymmetric_init(n0, seed=3))
        labels.append("asym3")
        inits.append(make_linear_ramp(n0, 1.5))
        labels.append("ramp")
        inits.append(make_two_bump_asym(n0, seed=4))
        labels.append("twobump")

        # Symmetric (for comparison)
        inits.append(make_u_shape(n0, 1.5))
        labels.append("u1.5")
        inits.append(make_u_shape(n0, 3.0))
        labels.append("u3.0")

        # Blend parent with asymmetric
        if parent is not None:
            blend = 0.6 * norm_sum1(parent) + 0.4 * make_asymmetric_init(n0, seed=10)
            inits.append(norm_sum1(blend))
            labels.append("blend_asym")

        t_per = max(20.0, t_phase1 / len(inits))
        for idx, (init, lbl) in enumerate(zip(inits, labels)):
            elapsed_so_far = time.monotonic() - t_global
            if elapsed_so_far > time_budget - 60:
                break
            t_avail = min(t_per, time_budget - 60 - (time.monotonic() - t_global))
            if t_avail < 15:
                break
            print(f"  AMD init '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = amd_optimize(init, t_avail, solver, top_k=50, beta=5.0,
                                     c0=4.0, restart_every=4000,
                                     verbose=False, rng_seed=100 + idx,
                                     use_lp=True, lp_top_k=80)
            try_update(a_out, f"p1_{lbl}")

    # === Phase 2: Active-set LP intensive (main phase) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Active-set LP (n={n_best}) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 50:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t2, solver, top_k=80, beta=6.0,
                                 c0=3.5, restart_every=6000,
                                 verbose=True, rng_seed=777,
                                 use_lp=True, lp_top_k=150)
        try_update(a_out, "p2_lp_main")

    # === Phase 3: Directed pair descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Directed pair descent (n={n_best}) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = directed_pair_descent(a_in, t3, verbose=True, rng_seed=33)
        try_update(a_out, "p3_dirpair")

    # === Phase 4: AMD polish with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: AMD polish (n={n_best}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 30:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t4, solver, top_k=120, beta=8.0,
                                 c0=3.0, restart_every=10000,
                                 verbose=False, rng_seed=500,
                                 use_lp=False)
        try_update(a_out, "p4_polish")

    # === Phase 5: LP round 2 with larger active set ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP round 2 (n={n_best}) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t5, solver, top_k=100, beta=5.0,
                                 c0=4.0, restart_every=5000,
                                 verbose=True, rng_seed=888,
                                 use_lp=True, lp_top_k=250)
        try_update(a_out, "p5_lp2")

    # === Phase 6: Fine polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Fine polish (n={n_best}) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = fine_polish(a_in, t6, verbose=True)
        try_update(a_out, "p6_fine")

    # === Phase 7: Length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")

    trial_lengths = []
    for delta in [300, -300, 600, -600]:
        new_n = n_best + delta
        if 800 <= new_n <= 6000:
            trial_lengths.append(new_n)
    trial_lengths = list(dict.fromkeys(trial_lengths))
    trial_lengths = [l for l in trial_lengths if remaining > 100]

    for new_n in trial_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"  Trying n={new_n}...")
        a_long = interpolate_sequence(best_seq, new_n)
        t7 = min(remaining * 0.35, 200.0)
        if t7 > 30:
            a_out, _ = amd_optimize(a_long, t7, solver, top_k=60, beta=5.0,
                                     c0=4.0, restart_every=4000,
                                     verbose=False, rng_seed=new_n % 1000,
                                     use_lp=True, lp_top_k=100)
            try_update(a_out, f"p7_n{new_n}")

    # === Phase 8: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish (n={n_best}) ===")
    if remaining > 30:
        t8a = min(remaining * 0.4, 100.0)
        if t8a > 20:
            a_in = norm_sum1(best_seq)
            a_out, _ = directed_pair_descent(a_in, t8a, verbose=False, rng_seed=99)
            try_update(a_out, "p8a_dirpair")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.4, 80.0)
            if t8b > 15:
                a_in = norm_sum1(best_seq)
                a_out, _ = fine_polish(a_in, t8b, verbose=False)
                try_update(a_out, "p8b_fine")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 4000
        best_seq = make_u_shape(n_fb, 2.0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
