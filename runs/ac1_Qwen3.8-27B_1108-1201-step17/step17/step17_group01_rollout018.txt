For the simplex projection, I'll implement the standard algorithm:

```python
def project_simplex(v, s=1.0):
    """Project v onto {x: x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

- Or simply: δ * (a[k-i] + a[i-k]) where a[j] = 0 for j outside [0,n-1]

Wait, a[k-i] and a[i-k] are different! a[k-i] is a at position (k-i), and a[i-k] is a at position (i-k). These are different unless k-i = i-k, i.e., k = 2i.

So the increase is:
`Δc[k] = δ * (a[k-i] if 0<=k-i<n else 0) + δ * (a[i-k] if 0<=i-k<n else 0)`

Note: k-i and i-k have the same absolute value but opposite signs. Since a is defined on [0, n-1], a[k-i] is non-zero when 0 <= k-i < n, and a[i-k] is non-zero when 0 <= i-k < n, i.e., k-i > -n (always true for k >= 0) and i-k < n (i.e., k > i-n).

So for a given i:
- `a[k-i]` is non-zero for k in [i, i+n-1]
- `a[i-k]` is non-zero for k in [max(0, i-n+1), i] (since i-k >= 0 means k <= i, and i-k < n means k > i-n)

The "dangerous" k values are those where c[k] is already large. So the cost of adding mass at position i is:
`cost(i) = max over k of (a[k-i] * [c[k] is large] + a[i-k] * [c[k] is large])`

For simplicity, let's just use `cost(i) = max(a)` as a rough upper bound, or better, compute it as the max of the "shadow" weighted by the current peak values.

A practical approach: for each position i, compute:
`shadow(i) = max_k (p[k] * (a[k-i] + a[i-k]))`
where p[k] = exp(c[k]/τ)/sum(exp(c/τ)) is the LSE weight.

This is exactly the LSE gradient! So the positions with the smallest LSE gradient are the "safest" to add mass.

This gives a natural coordinate descent:
1. Compute LSE gradient g
2. Find i_min = argmin g[i] (safest to increase)
3. Find i_max = argmax g[i] (most expensive, should decrease)
4. Move mass from i_max to i_min
5. Adjust the amount by line search

This is essentially a "mass transport" guided by the LSE gradient.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp (LSE) smooth gradient descent with Nesterov momentum
2. Adaptive temperature schedule focusing on near-maximal peaks
3. Simplex projection for the sum-to-one constraint
4. LSE-guided mass transport for local polish
5. Multi-length and multi-initialization search

import numpy as np
import time


def fast_autocorr(a):
    """Compute full linear self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the objective."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def project_simplex(v, s=1.0):
    """Project vector v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1)
    cond = u * rho_arr > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, tau):
    """
    Compute gradient of LSE(c/tau) where c = conv(a,a).
    Returns (gradient, lse_value, c, p) where p = exp(c/tau)/sum(exp(c/tau)).
    """
    n = len(a)
    c = fast_autocorr(a)
    
    # Compute p (softmax of c/tau)
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / tau)
    p = exp_c / exp_c.sum()
    
    lse_val = tau * (c.max() + np.log(exp_c.sum()))
    
    # Gradient: grad[i] = tau * (conv1[n-1+i] + conv2[i])
    a_flipped = a[::-1]
    
    # Need FFT size for convolutions of length (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n).bit_length()
    P = np.fft.rfft(p, fft_size)
    Af = np.fft.rfft(a_flipped, fft_size)
    A = np.fft.rfft(a, fft_size)
    
    conv1 = np.fft.irfft(P * Af, fft_size)
    conv2 = np.fft.irfft(P * A, fft_size)
    
    grad = tau * (conv1[n - 1:2 * n - 1] + conv2[0:n])
    
    return grad, lse_val, c, p


def lse_optimize(a0, time_limit, tau_init=None, tau_decay=0.995, 
                 eta_init=0.5, verbose=False):
    """
    LSE smooth gradient descent with Nesterov momentum and simplex projection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    if tau_init is None:
        c_init = fast_autocorr(a)
        tau_init = c_init.max() / 20.0
    
    tau = tau_init
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    eta = eta_init
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute LSE gradient
        grad, lse_val, c, p = lse_gradient(y, tau)
        
        # Gradient step
        y_new = y - eta * grad
        y_new = project_simplex(y_new)
        
        # Evaluate
        M_new = fast_autocorr(y_new).max()
        
        # Update best
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = y_new.copy()
        
        # Update
        x_prev = x
        x = y_new
        
        # Adaptive tau: track the gap between top peaks
        # (decrease tau to focus more on the peak)
        tau *= tau_decay
        c_curr = c.max()
        if tau < c_curr * 0.005:
            tau = c_curr / 20.0
        
        # Adaptive eta
        eta = eta_init / (1.0 + 0.001 * t)
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LSE t={t} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.2e}")
    
    return best_a, best_m


def lse_mass_transport(a0, time_limit, verbose=False, rng_seed=42):
    """
    LSE-guided mass transport: move mass from high-gradient to low-gradient positions.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_autocorr(a)
        M_curr = c.max()
        
        # Compute LSE weights (with small tau to focus on peaks)
        tau = M_curr / 50.0
        c_shifted = c - c.max()
        exp_c = np.exp(c_shifted / tau)
        p = exp_c / exp_c.sum()
        
        # Compute "cost" of adding mass at each position (the LSE gradient)
        a_flipped = a[::-1]
        fft_size = 1 << (3 * n).bit_length()
        P = np.fft.rfft(p, fft_size)
        Af = np.fft.rfft(a_flipped, fft_size)
        A = np.fft.rfft(a, fft_size)
        conv1 = np.fft.irfft(P * Af, fft_size)
        conv2 = np.fft.irfft(P * A, fft_size)
        cost = conv1[n - 1:2 * n - 1] + conv2[0:n]
        
        # Source: high cost (expensive positions)
        # Destination: low cost (cheap positions)
        src_candidates = np.argsort(cost)[::-1][:20]
        dst_candidates = np.argsort(cost)[:20]
        
        improved = False
        
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit:
                break
            if a[src] < 1e-12:
                continue
            
            for dst in dst_candidates:
                if dst == src:
                    continue
                if time.monotonic() - t0 > time_limit:
                    break
                
                # Try moving a fraction of mass
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                        # Actually we should continue trying other fracs
                        # but let's break for now
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Transport ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def make_ramp(n, power=1.0, reverse=False):
    """Create a one-sided ramp (asymmetric)."""
    x = np.arange(1, n + 1, dtype=np.float64)
    if reverse:
        x = x[::-1]
    a = x ** power
    return norm_sum1(a)


def make_bell(n, center=0.5, width=0.15):
    """Create a bell-shaped sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    return norm_sum1(a)


def make_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_sinc(n, bandwidth=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bandwidth
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_random_smooth(n, seed=0, n_freq=6):
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_freq + 1):
        coeff = rng.normal(0, 0.15 / np.sqrt(f))
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.001)
    return norm_sum1(a)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
