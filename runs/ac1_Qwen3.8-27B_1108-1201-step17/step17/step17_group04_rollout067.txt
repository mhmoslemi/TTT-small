def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def compute_J(a):

def smoothed_obj_grad(a, T):
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    S = exp_Tc.sum()
    w = exp_Tc / S
    obj = (2.0 * n / T) * (Tc_max + np.log(S))
    b = np.correlate(w, a, mode='valid')
    grad = 4.0 * n * b
    return obj, grad

def lbfqsb_refine(a0, time_limit, T=20000.0, verbose=False):
    n = len(a0)
    a_init = norm_sum1(a0)
    best_a = a_init.copy()
    best_J = compute_J(a_init)
    t0 = time.monotonic()
    z_init = np.log(a_init + 1e-30)
    max_z = z_init.max()
    z_init = z_init - max_z

    def obj_grad(z):
        zc = z - z.max()
        ez = np.exp(zc)
        a = ez / ez.sum()
        c = fast_autocorr(a)
        Tc = T * c
        Tc_max = Tc.max()
        exp_Tc = np.exp(Tc - Tc_max)
        S = exp_Tc.sum()
        w = exp_Tc / S
        fval = (2.0 * n / T) * (Tc_max + np.log(S))
        b = np.correlate(w, a, mode='valid')
        grad_a = 4.0 * n * b
        grad_z = a * (grad_a - np.dot(a, grad_a))
        return fval, grad_z

    result = scipy_minimize(obj_grad, z_init, method='L-BFGS-B',
                            jac=True, options={'maxiter': 200, 'maxfun': 500,
                                               'ftol': 1e-14, 'gtol': 1e-10,
                                               'maxls': 30, 'maxcor': 20})
    if result.success or hasattr(result, 'fun'):
        z_opt = result.x
        zc = z_opt - z_opt.max()
        a_opt = np.exp(zc)
        a_opt = a_opt / a_opt.sum()
        J_opt = compute_J(a_opt)
        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()
    if verbose:
        print(f"    L-BFGS-B done ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")
    return best_a, best_J

        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        c = np.fft.irfft(P, N)[:2 * n - 1]
        top_k = np.argsort(c)[::-1][:4]
        for k_max in top_k:
            bin_idx = int(k_max) % N
            if bin_idx < n_bins:
                A[bin_idx] *= (1.0 - gamma)
        a_new = np.fft.irfft(A, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
        if verbose and it % 50 == 0:
            print(f"    FreqDepeak it={it} ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")
    return best_a, best_J

def peak_reduction_v2(a0, time_limit, verbose=False, seed=77):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 1000:
        c = fast_autocorr(a)
        top_lags = np.argsort(c)[::-1][:8]
        sensitivity = np.zeros(n)
        for k in top_lags[:5]:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            for ii, jj in zip(idx, j_idx):
                sensitivity[ii] += a[jj]
                sensitivity[jj] += a[ii]
        mean_s = sensitivity.mean()
        high_sens = np.where(sensitivity > mean_s + 0.1 * sensitivity.std())[0]
        low_sens = np.where(sensitivity < mean_s)[0]
        if len(high_sens) == 0 or len(low_sens) == 0:
            stall += 1
            continue
        improved = False
        for _ in range(40):
            if time.monotonic() - t0 > time_limit:
                break
            src = int(high_sens[rng.integers(len(high_sens))])
            dst = int(low_sens[rng.integers(len(low_sens))])
            if src == dst or a[src] < 1e-10:
                continue
            lo, hi = 0.0, a[src] * 0.8
            best_amt = 0.0
            best_trial_J = best_J
            for _ in range(10):
                mid = (lo + hi) / 2
                if mid < 1e-13:
                    break
                a_trial = a.copy()
                a_trial[src] -= mid
                a_trial[dst] += mid
                a_trial = norm_sum1(a_trial)
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_amt = mid
                    lo = mid
                else:
                    hi = mid
            if best_amt > 1e-13 and best_trial_J < best_J - 1e-14:
                a[src] -= best_amt
                a[dst] += best_amt
                a = norm_sum1(a)
                best_J = best_trial_J
                best_a = a.copy()
                improved = True
                stall = 0
        if not improved:
            stall += 1
            if stall > 50:
                for _ in range(max(3, n // 300)):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * rng.uniform(0.005, 0.05)
                        a[i] -= amt
                        a[j] += amt
                a = norm_sum1(a)
                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()
                stall = 0
        if verbose and stall % 200 == 0:
            print(f"    PeakRed2 ({time.monotonic()-t0:.0f}s): J={best_J:.6f}, stall={stall}")
    return best_a, best_J

def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=600):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    for t in range(1, 200):
        if time.monotonic() - t0 > time_limit:
            break
        T_full = np.zeros((2 * n - 1, n))
        for j in range(n):
            T_full[j:j + n, j] = y
        ac = T_full @ y
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        margins = M_curr - ac
        K_sel = min(top_K, 2 * n - 1)
        binding_idx = np.argsort(margins)[:K_sel]
        T_sel = T_full[binding_idx, :]
        M_sel = (M_curr - margins[binding_idx] * 0.0) * np.ones(K_sel)
        M_sel = np.full(K_sel, M_curr)
        b_var = cp.Variable(n, nonneg=True)
        constraints_list = [T_sel @ b_var <= M_sel]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints_list)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        if t >= 2:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
            mx = fast_autocorr(x_new).max()
            if mx < best_m:
                best_m = mx
                best_a = x_new.copy()
        my = fast_autocorr(y_new).max()
        if my < best_m:
            best_m = my
            best_a = y_new.copy()
        y = y_new
        if verbose and t % 5 == 0:
            print(f"    LP-FW it={t} ({time.monotonic()-t0:.0f}s): J={2*n*best_m:.6f}")
    return best_a, best_m

def make_symmetric_init(n):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.01
    a = np.maximum(a, 0)
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coordinate_polish(a0, time_limit, verbose=False, seed=555):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_J = compute_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        k_max = int(np.argmax(c))
        i_start = max(0, k_max - n + 1)
        i_end = min(n, k_max + 1)
        idx = np.arange(i_start, i_end)
        improved = False
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            p = int(rng.integers(len(idx)))
            q = int(rng.integers(len(idx)))
            if p == q:
                continue
            i1, i2 = int(idx[p]), int(idx[q])
            if i1 == i2 or a[i1] < 1e-12:
                continue
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = a[i1] * frac
                if amt < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[i1] -= amt
                a_trial[i2] += amt
                a_trial = norm_sum1(a_trial)
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            low_idx = list(set(range(n)) - set(idx))
            if len(low_idx) > 0:
                dst = int(low_idx[rng.integers(len(low_idx))])
                src = int(idx[rng.integers(len(idx))])
                if a[src] > 1e-10:
                    for frac in [0.003, 0.008, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = norm_sum1(a_trial)
                        J_trial = compute_J(a_trial)
                        if J_trial < best_J - 1e-14:
                            a = a_trial
                            best_J = J_trial
                            best_a = a.copy()
                            improved = True
                            break
        if verbose:
            print(f"    CoordPolish ({time.monotonic()-t0:.0f}s): J={best_J:.6f}")
    return best_a, best_J

def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1780.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = norm_sum1(p)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = make_symmetric_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # Phase 1: Frequency-domain de-peak
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Frequency de-peak (n={n0}) ===")
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = freq_depeak(a_in, t1, gamma=0.02, verbose=True)
        try_update(a_out, "freq_depeak")

    # Phase 2: Smoothed gradient - low T (exploration)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: NS low-T (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        n_cur = len(best_seq)
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smoothed(a_in, t2, T_start=n_cur / 10.0, T_end=n_cur,
                                      verbose=True, seed=111)
        try_update(a_out, "ns_lowT")

    # Phase 3: Smoothed gradient - medium T (refinement)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: NS med-T (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        n_cur = len(best_seq)
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smoothed(a_in, t3, T_start=n_cur / 3.0, T_end=10.0 * n_cur,
                                      verbose=True, seed=222)
        try_update(a_out, "ns_medT")

    # Phase 4: L-BFGS-B refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: L-BFGS-B (n={n_cur}) ===")
    t4 = min(remaining * 0.12, 200.0)
    if t4 > 20 and n_cur <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lbfqsb_refine(a_in, t4, T=15.0 * n_cur, verbose=True)
        try_update(a_out, "lbfqsb")

    # Phase 5: Peak reduction
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Peak reduction (n={n_cur}) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_reduction_v2(a_in, t5, verbose=True, seed=333)
        try_update(a_out, "peakred")

    # Phase 6: NS high-T (polish)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: NS high-T (n={len(best_seq)}) ===")
    t6 = min(remaining * 0.20, 300.0)
    if t6 > 20:
        n_cur = len(best_seq)
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smoothed(a_in, t6, T_start=20.0 * n_cur, T_end=200.0 * n_cur,
                                      verbose=True, seed=444)
        try_update(a_out, "ns_highT")

    # Phase 7: LP Frank-Wolfe
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: LP Frank-Wolfe (n={n_cur}) ===")
    t7 = min(remaining * 0.15, 250.0)
    if t7 > 20 and n_cur <= 4000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t7, solver, verbose=True, top_K=500)
        try_update(a_out, "lpfw")

    # Phase 8: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Coordinate polish (n={len(best_seq)}) ===")
    t8 = min(remaining * 0.08, 120.0)
    if t8 > 10:
        a_in = norm_sum1(best_seq)
        a_out, _ = coordinate_polish(a_in, t8, verbose=True, seed=555)
        try_update(a_out, "coordpolish")

    # Phase 9: Final NS from best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final NS (n={len(best_seq)}) ===")
    t9 = min(remaining * 0.3, 300.0)
    if t9 > 20:
        n_cur = len(best_seq)
        a_in = norm_sum1(best_seq)
        a_out, _ = nesterov_smoothed(a_in, t9, T_start=5.0 * n_cur, T_end=50.0 * n_cur,
                                      verbose=True, seed=666)
        try_update(a_out, "ns_final")

    # Phase 10: Try slightly different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    new_n = n_cur + 200
    if remaining > 60 and new_n <= 5000:
        print(f"\n=== Phase 10: Try n={new_n} ===")
        a_long = interpolate_sequence(best_seq, new_n)
        t10 = min(remaining * 0.4, 200.0)
        if t10 > 20:
            a_out, _ = nesterov_smoothed(a_long, t10, T_start=new_n / 5.0, T_end=20.0 * new_n,
                                          verbose=True, seed=777)
            try_update(a_out, f"ns_n{new_n}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = make_symmetric_init(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
