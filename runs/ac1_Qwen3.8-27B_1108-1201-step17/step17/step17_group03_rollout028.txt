def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_soft_subgradient(a, ac, top_k=50, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad

    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        if use_backtrack:
            eta = gamma / (np.sqrt(t) + 1.0)
            base_m = fast_autocorr(y).max()
            accepted = False
            for _ in range(8):
                log_a = np.log(y) - eta * g
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)

                if s_new < 1e-15:

                a_new = a_new / s_new

                if m_trial < base_m - 1e-18:
                    x_prev = x
                    x = a_new
                    accepted = True
                    break
                else:
                    eta *= 0.5
            if not accepted:
                log_a = np.log(y) - eta * g
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()

                else:
                    a_new = a_new / s_new
                x_prev = x

        else:
            eta = gamma / (np.sqrt(t) + 1.0)
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:

def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   margin_frac=1e-3, use_line_search=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    prev_b = None

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K * 4:
            active_idx = np.argsort(margins)[:top_K * 4]
        elif n_active < 8:
            active_idx = np.argsort(margins)[:top_K]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            for i in range(i_start, i_end):
                T_sel[row, i] = y[k_int - i]

        b_var = cp.Variable(n, nonneg=True)
        if prev_b is not None:
            b_var.value = prev_b
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        prev_b = b_val.copy()

        if use_line_search:
            best_t = 2.0 / (t + 2.0)
            best_val = best_m
            candidates_t = [2.0/(t+2.0), 1.0/(t+2.0), 0.5*2.0/(t+2.0),
                            3.0/(t+2.0), 4.0/(t+2.0), 0.1, 0.05]
            for trial_t in candidates_t:
                trial_t = min(trial_t, 1.0)
                y_trial = (1.0 - trial_t) * y + trial_t * g_dir
                y_trial = np.maximum(y_trial, 0.0)
                sy = y_trial.sum()
                if sy < 1e-15:
                    continue
                y_trial = y_trial / sy
                m_trial = fast_autocorr(y_trial).max()
                if m_trial < best_val:
                    best_val = m_trial
                    best_t = trial_t
            alpha = best_t
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            if margin_frac > 1e-5:
                margin_frac = max(margin_frac * 0.85, 1e-5)

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 3 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, margin={margin_frac:.2e}")

    return best_a, best_m


def peak_crush_coordinate(a0, time_limit, verbose=True, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 1
            continue

        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                n_trials = 40
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakCrush final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def pair_swap_descent(a0, time_limit, verbose=True, rng_seed=77):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        sort_idx = np.argsort(ac)[::-1][:10]
        improved = False

        for peak_k in sort_idx:
            if time.monotonic() - t0 > time_limit or improved:
                break
            peak_k = int(peak_k)
            i_start = max(0, peak_k - n + 1)
            i_end = min(n, peak_k + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[peak_k - idx]

            if contrib.sum() < 1e-20:
                continue

            top_src = np.argsort(contrib)[::-1][:5]
            low_dst_candidates = np.argsort(contrib)[:n // 4]

            for si in top_src:
                if time.monotonic() - t0 > time_limit or improved:
                    break
                src = int(idx[si])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                for di in low_dst_candidates[:20]:
                    if time.monotonic() - t0 > time_limit:
                        break
                    dst = int(idx[di])
                    if dst == src:
                        continue

                    for frac in [0.01, 0.03, 0.05, 0.1]:
                        amt = src_val * frac
                        if amt > a[src]:
                            amt = a[src] * 0.5
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = a_trial / a_trial.sum()
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_m - 1e-15:
                            a = a_trial
                            best_m = M_trial
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = min(a[i] * frac, a[i])
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PairSwap ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def freq_perturb(a, strength=0.01, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def spectral_equalize(a0, time_limit, verbose=True):
    """Try to equalize the autocorrelation peaks via iterative cutting."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(314)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M_curr = ac.max()

        top_peaks = np.argsort(ac)[::-1][:20]
        mean_offpeak = np.median(ac)

        improved = False
        for k_int in top_peaks[:5]:
            if time.monotonic() - t0 > time_limit or improved:
                break
            k_int = int(k_int)
            if ac[k_int] < M_curr * 0.9:
                continue

            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k_int - idx]

            if contrib.sum() < 1e-20:
                continue

            top_contrib = np.argsort(contrib)[::-1][:8]
            for tc in top_contrib:
                if time.monotonic() - t0 > time_limit or improved:
                    break
                src = int(idx[tc])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                for frac in [0.005, 0.01, 0.02, 0.04]:
                    amt = src_val * frac
                    n_dst = 50
                    best_trial_m = M_curr
                    best_dst = None
                    for _ in range(n_dst):
                        dst = int(rng.integers(n))
                        if dst == src:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = a_trial / a_trial.sum()
                        m_trial = fast_autocorr(a_trial).max()
                        if m_trial < best_trial_m:
                            best_trial_m = m_trial
                            best_dst = dst

                    if best_dst is not None and best_trial_m < best_m - 1e-15:
                        a[src] -= amt
                        a[best_dst] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                        improved = True
                        break

        if improved:
            continue

        # No improvement, try random perturbation
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j and a[i] > 1e-10:
            frac = rng.uniform(0.003, 0.02)
            amt = a[i] * frac
            a[i] -= amt
            a[j] += amt
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            M_new = fast_autocorr(a).max()
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()

        if verbose and it % 5000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    SpectralEq ({elapsed:.0f}s): J={J:.6f}, M={best_m:.8f}")

    return best_a, best_m


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    elif family == 'uniform':
        return np.ones(n) / n
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):

    best_seq = None

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

                                restart_every=3000, verbose=True, use_backtrack=True)

    # === Phase 2: LP-Frank-Wolfe main (generous) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP-FW main (n={len(best_seq)}) ===")
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 420.0)
    if t2 > 40 and n_best <= 5000:

        a_out, _ = lp_frank_wolfe(a_in, t2, solver, verbose=True, top_K=250,
                                   margin_frac=2e-3, use_line_search=True)

    # === Phase 3: Spectral equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral equalization (n={len(best_seq)}) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = spectral_equalize(a_in, t3, verbose=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "spectral_eq")

    # === Phase 4: LP-FW tight margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW tight (n={n_best}) ===")
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t4, solver, verbose=True, top_K=350,
                                   margin_frac=5e-4, use_line_search=True)

    # === Phase 5: Pair-swap coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)

    t5 = min(remaining * 0.10, 150.0)

        a_in = norm_sum1(best_seq)
        a_out, _ = pair_swap_descent(a_in, t5, verbose=True, rng_seed=77)

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    longer_lengths = sorted(set([n_best + 200, n_best + 500, n_best + 1000,

    longer_lengths = [l for l in longer_lengths if 1500 <= l <= 5500]

    explored = 0
    for new_n in longer_lengths[:3]:

        if remaining < 100 or explored >= 2:

                                           margin_frac=2e-3, use_line_search=True)

    # === Phase 7: Frequency diversification + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global)

    if t7_per > 20:

            a_pert = freq_perturb(best_seq, strength=0.01, rng=rng)

                                     restart_every=2000, verbose=False, use_backtrack=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"freq{trial}")

    # === Phase 8: Final LP on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 30 and n_best <= 5000:

    # === Phase 9: Final peak-crush ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final peak-crush (n={n_best}) ===")
    if remaining > 25:
        t9 = min(remaining * 0.30, 160.0)
        if t9 > 10:

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):

        n_fb = 3200
        best_seq = make_parametric(n_fb, 'sine')

        print(f"Fallback: score={best_score:.6f}")

    # Safety: ensure we return before budget
    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
