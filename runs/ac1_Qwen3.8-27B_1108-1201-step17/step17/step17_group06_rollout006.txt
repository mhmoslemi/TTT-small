def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_conv(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_subgradient(a, conv, top_k=100, beta_w=8.0):
    """Compute soft subgradient of max(conv) at a using top-K lags."""
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(conv)[::-1][:k_max]
    top_vals = np.maximum(conv[top_idx], 1e-30)
    w = top_vals ** beta_w
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    g = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            j_idx = k_int - idx
            g[idx] += 2.0 * wi * a[j_idx]
    return g

def peak_reduction(a0, time_limit, verbose=False, seed=99):

        # d(conv[k])/da[i] = 2*a[k-i] (symmetric)

                j_idx = k_int - idx
                grad[idx] += 2.0 * w * a[j_idx]
        
        # Project gradient to sum=0 (tangent to simplex)
        grad -= grad.mean()

def blend_solutions(solutions, weights=None):
    """Blend multiple solutions by averaging (on same-length sequences)."""
    if len(solutions) < 2:
        return solutions[0] if solutions else None
    
    n = len(solutions[0])
    if weights is None:
        weights = np.ones(len(solutions)) / len(solutions)
    
    # All must have same length
    same_len = all(len(s) == n for s in solutions)
    if not same_len:
        # Interpolate to common length
        max_n = max(len(s) for s in solutions)
        resampled = []
        for s in solutions:
            if len(s) == max_n:
                resampled.append(s)
            else:
                old_x = np.linspace(0, 1, len(s))
                new_x = np.linspace(0, 1, max_n)
                resampled.append(np.interp(new_x, old_x, s))
        n = max_n
        solutions = [np.array(s, dtype=np.float64) for s in resampled]
    
    result = np.zeros(n)
    for s, w in zip(solutions, weights):
        s = np.asarray(s, dtype=np.float64)
        result += w * s
    
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-15:
        return None
    return result / s


def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=80.0):

def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s

def fast_conv(a):

def make_power_ramp(n, p=1.0):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_bimodal(n, width=80.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.15) ** 2) + np.exp(-width * (x - 0.85) ** 2) + 0.003
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 12):
        coeff = rng.normal(0, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_asymmetric_ramp(n, p=1.0, peak_pos=0.6):
    """Asymmetric ramp peaked at a specific position."""
    x = np.linspace(0, 1, n)
    a = np.exp(-3.0 * (x - peak_pos) ** 2 / (0.3 * (1 - 0.3)))
    a = a ** (1.0 / (p + 0.1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

    return a / s


def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s


def blend_two(a1, a2, w1):
    """Blend two same-length sequences."""
    n1 = len(a1)
    n2 = len(a2)
    if n1 != n2:
        a2 = interpolate_seq(a2, n1)
    result = w1 * a1 + (1.0 - w1) * a2
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-15:
        return None
    return result / s

    if J < state['best_score']:
        state['best_score'] = J
        state['best_a'] = a.copy()
        elapsed = time.monotonic() - state['t0']
        print(f"  *** NEW BEST [{label}]: J={J:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        state['solutions'].append(a.copy())
        # Keep only top 5 sorted by J
        sol_vals = []
        for sol in state['solutions']:
            sol_vals.append(eval_J(sol))
        order = np.argsort(sol_vals)
        state['solutions'] = [state['solutions'][i] for i in order[:5]]


def propose_candidate(budget_s=1800.0):
    """
    Main optimization driver with multi-phase strategy:
    1. Multi-init exploration with improved MD
    2. Focused deep MD on best
    3. Peak reduction
    4. Coordinate descent
    5. Blend & re-optimize
    6. Length sweep
    7. Ultra-fine final polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    state = {
        'best_a': None,
        'best_score': float('inf'),
        'solutions': [],
        't0': t_global
    }

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p.copy()
            parent = parent / parent.sum()
            pJ = eval_J(parent)
            print(f"Parent: n={len(parent)}, J={pJ:.6f}")
    except Exception:
        pass

    # Initialize best
    if parent is not None:
        state['best_a'] = parent.copy()
        state['best_score'] = eval_J(parent)
        state['solutions'].append(parent.copy())
    else:
        n0 = 3000
        best_a = make_power_ramp(n0, 1.0)
        state['best_a'] = best_a
        state['best_score'] = eval_J(best_a)
        print(f"Initial: n={len(best_a)}, J={state['best_score']:.6f}")

    n0 = len(state['best_a'])

    # === Phase 1: Multi-init MD exploration (30% of time) ===
    print(f"\n=== Phase 1: Multi-Init MD (n={n0}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = remaining * 0.30

    inits = []
    labels = []
    if parent is not None:
        inits.append(parent.copy())
        labels.append("parent")
    inits.append(make_power_ramp(n0, 0.5))
    labels.append("ramp0.5")
    inits.append(make_power_ramp(n0, 0.75))
    labels.append("ramp0.75")
    inits.append(make_power_ramp(n0, 1.0))
    labels.append("ramp1.0")
    inits.append(make_power_ramp(n0, 1.5))
    labels.append("ramp1.5")
    inits.append(make_power_ramp(n0, 2.0))
    labels.append("ramp2.0")
    inits.append(make_bimodal(n0))
    labels.append("bimodal")
    inits.append(make_smooth_random(n0, seed=7))
    labels.append("smooth7")
    inits.append(make_smooth_random(n0, seed=123))
    labels.append("smooth123")
    inits.append(make_asymmetric_ramp(n0, p=1.0, peak_pos=0.5))
    labels.append("asym_center")
    inits.append(make_asymmetric_ramp(n0, p=1.5, peak_pos=0.7))
    labels.append("asym_right")

    if parent is not None:
        for alpha in [0.3, 0.5, 0.7]:
            for p_exponent in [0.5, 1.0, 1.5, 2.0]:
                b = alpha * parent + (1.0 - alpha) * make_power_ramp(n0, p_exponent)
                b = np.maximum(b, 0)
                b = b / b.sum()
                inits.append(b)
                labels.append(f"bl_{alpha:.1f}_{p_exponent}")

    # Deduplicate
    seen = set()
    u_inits = []
    u_labels = []
    for init, lbl in zip(inits, labels):
        sig = tuple(np.round(init[::max(1, n0 // 200)], 4))
        if sig not in seen:
            seen.add(sig)
            u_inits.append(init)
            u_labels.append(lbl)
    inits = u_inits
    labels = u_labels

    # Limit to fit in budget
    max_inits = min(len(inits), max(3, int(t_phase1 / 35)))
    t_per = t_phase1 / max_inits

    for idx in range(max_inits):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget - 60:
            break
        t_avail = min(t_per * 1.3, time_budget - 60 - elapsed)
        if t_avail < 12:
            break
        init, lbl = inits[idx], labels[idx]
        init_J = eval_J(init)
        print(f"  MD [{lbl}] ({t_avail:.0f}s) J_init={init_J:.6f}...")
        a_out, _ = mirror_descent_improved(init, t_avail, top_k=80, beta_w=10.0,
                                            c0=4.0, seed=idx * 137 + 1)
        update_best(state, a_out, f"md_{lbl}")

    # === Phase 2: Focused deep MD (25% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Focused MD (n={len(state['best_a'])}, J={state['best_score']:.6f}) ===")
    t2 = remaining * 0.25

    if t2 > 50:
        a_out, _ = mirror_descent_improved(state['best_a'].copy(), t2,
                                            top_k=120, beta_w=6.0, c0=3.0,
                                            restart_every=8000, verbose=True, seed=555)
        update_best(state, a_out, "focused_md")

    # === Phase 3: Peak reduction (12% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak Reduction ===")
    t3 = remaining * 0.12

    if t3 > 30:
        a_out, _ = peak_reduction(state['best_a'].copy(), t3, verbose=True, seed=777)
        update_best(state, a_out, "peak_red")

    # === Phase 4: Coordinate descent (10% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord Descent ===")
    t4 = remaining * 0.10

    if t4 > 20:
        a_out, _ = coordinate_descent_v2(state['best_a'].copy(), t4, verbose=True, seed=314)
        update_best(state, a_out, "coord_v2")

    # === Phase 5: MD after CD (8% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: MD after CD ===")
    t5 = remaining * 0.08

    if t5 > 20:
        a_out, _ = mirror_descent_improved(state['best_a'].copy(), t5,
                                            top_k=150, beta_w=5.0, c0=2.5,
                                            restart_every=10000, seed=444)
        update_best(state, a_out, "md_after_cd")

    # === Phase 6: Blend solutions and re-optimize (8% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Blend & Re-opt ({len(state['solutions'])} solutions) ===")
    t6 = remaining * 0.08

    if t6 > 20 and len(state['solutions']) >= 2:
        rng_blend = np.random.default_rng(42)
        for i in range(min(3, len(state['solutions']))):
            for j in range(i + 1, min(4, len(state['solutions']))):
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining < 30:
                    break
                w1 = rng_blend.uniform(0.2, 0.8)
                blended = blend_two(state['solutions'][i], state['solutions'][j], w1)
                if blended is not None:
                    bl_J = eval_J(blended)
                    if bl_J < state['best_score']:
                        print(f"  Blend {i}/{j} w={w1:.2f}: J={bl_J:.6f} (improvement!)")
                    t_bl = min(remaining * 0.15, 50.0)
                    if t_bl > 10:
                        a_out, _ = mirror_descent_improved(blended, t_bl,
                                                            top_k=100, beta_w=7.0,
                                                            c0=3.0, seed=i * 10 + j)
                        update_best(state, a_out, f"blend_{i}_{j}")

    # === Phase 7: Length sweep (7% of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Length Sweep ===")
    t7 = remaining * 0.07
    n_best = len(state['best_a'])
    lengths_to_try = []
    for d in [-400, -200, 200, 400, 600, 1000]:
        nl = n_best + d
        if 1500 <= nl <= 6000:
            if nl not in lengths_to_try:
                lengths_to_try.append(nl)

    for nl in sorted(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        if abs(nl - n_best) <= 50:
            continue
        print(f"  Length: n={nl}...")
        a_int = interpolate_seq(state['best_a'], nl)
        t_md = min(remaining * 0.25, 70.0)
        if t_md > 15:
            a_out, _ = mirror_descent_improved(a_int, t_md, top_k=80, beta_w=10.0,
                                                c0=3.5, seed=nl)
            update_best(state, a_out, f"len{nl}")

    # === Phase 8: Final ultra-fine polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Ultra-Fine Polish (n={len(state['best_a'])}) ===")
    if remaining > 30:
        a_out, _ = mirror_descent_improved(state['best_a'].copy(), remaining - 20,
                                            top_k=250, beta_w=3.5, c0=1.0,
                                            restart_every=15000, verbose=True, seed=9999)
        update_best(state, a_out, "ultra_fine")

    # Final validation
    if state['best_a'] is not None:
        final_J = eval_J(state['best_a'])
        state['best_score'] = min(state['best_score'], final_J)
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL RESULT: J={state['best_score']:.6f}, n={len(state['best_a'])}, time={elapsed:.0f}s")
        try:
            ev = evaluate_sequence([float(x) for x in state['best_a']])
            print(f"evaluate_sequence: {ev:.6f}")
        except Exception as e:
            print(f"Warning: evaluate_sequence failed: {e}")
        return [float(x) for x in state['best_a']]
    else:
        n_fb = 3000
        fb = make_power_ramp(n_fb, 1.0)
        print(f"\nFALLBACK: n={n_fb}, J={eval_J(fb):.6f}")
        return [float(x) for x in fb]
