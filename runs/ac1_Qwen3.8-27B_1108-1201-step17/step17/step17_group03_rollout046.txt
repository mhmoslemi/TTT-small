                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  Solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def compute_soft_subgradient(a, ac, top_k=60, beta=5.0):
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def amd_optimize(a0, time_limit, top_k=60, beta=5.0, c0=4.0,
                 restart_every=4000, verbose=False, use_backtrack=True):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        g = compute_soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)

        if use_backtrack:
            eta = gamma / (np.sqrt(t) + 1.0)
            base_m = fast_autocorr(y).max()
            accepted = False
            for _ in range(5):
                log_a = np.log(y) - eta * g
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    break
                a_new = a_new / s_new
                m_trial = fast_autocorr(a_new).max()
                if m_trial < base_m - 1e-18:
                    x_prev = x
                    x = a_new
                    accepted = True
                    break
                else:
                    eta *= 0.5
            if not accepted:
                log_a = np.log(y) - eta * g
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    a_new = y.copy()
                else:
                    a_new = a_new / s_new
                x_prev = x
                x = a_new
        else:
            eta = gamma / (np.sqrt(t) + 1.0)
            log_a = np.log(y) - eta * g
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 100)
            cur_beta = max(cur_beta - 0.5, 2.5)

        if verbose and t % 10000 == 0:
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}")

    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K_max=400,
                   margin_init=5e-3, margin_min=1e-5, tighten_factor=0.85):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    x = a.copy()
    x_prev = a.copy()
    margin_frac = margin_init
    stall = 0
    max_stall = 200

    t = 0
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        t += 1
        ac = fast_autocorr(a)
        M_curr = ac.max()

        if M_curr < 1e-16:
            break

        margins = M_curr - ac
        threshold = margin_frac * M_curr
        active_mask = margins <= threshold
        n_active = active_mask.sum()

        if n_active > top_K_max:
            active_idx = np.argsort(margins)[:top_K_max]
        elif n_active < 10:
            active_idx = np.argsort(margins)[:min(50, top_K_max)]
        else:
            active_idx = np.where(active_mask)[0]

        K = len(active_idx)
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k = int(k_int)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            T_sel[row, idx] = a[k - idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * a + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new / sy

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - a)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        a = y_new
        x_prev = x_new

        my = fast_autocorr(a).max()
        mx = fast_autocorr(x_new).max()

        if my < best_m or mx < best_m:
            if my < best_m:
                best_m = my
                best_a = a.copy()
            else:
                best_m = mx
                best_a = x_new.copy()
            stall = 0
            margin_frac = max(margin_frac * tighten_factor, margin_min)
        else:
            stall += 1

        if verbose and t % 5 == 0:
            J = 2.0 * n * best_m
            print(f"    LP-FW t={t} ({time.monotonic()-t0:.0f}s): J={J:.6f}, K={K}, m={margin_frac:.1e}")

    return best_a, best_m


def peak_crush(a0, time_limit, verbose=False, rng_seed=123):
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_k = int(np.argmax(ac))

        i_start = max(0, peak_k - n + 1)
        i_end = min(n, peak_k + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-20:
            stall += 1
            continue

        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]

        improved = False
        for li in top_local:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[li])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            best_new_m = M_curr
            best_move = None

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                n_trials = 25
                for _ in range(n_trials):
                    dst = int(rng.integers(n))
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_new_m - 1e-18:
                        best_new_m = M_trial
                        best_move = (src, dst, amt)

            if best_move is not None:
                src_i, dst_i, amt = best_move
                a[src_i] -= amt
                a[dst_i] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 20 == 0:
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({time.monotonic()-t0:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        J = 2.0 * n * best_m
        print(f"    PeakCrush final ({time.monotonic()-t0:.0f}s): J={J:.6f}")

    return best_a, best_m


def freq_perturb(a, strength=0.01, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    phase_pert = phase + rng.normal(0, strength, len(phase))
    A_new = mag * np.exp(1j * phase_pert)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def symmetric_peak_rebalance(a, ac, top_peaks=5, time_limit=30.0, rng_seed=77):
    """Targeted reduction of top peaks by moving mass to low-contribution positions."""
    n = len(a)
    a = norm_sum1(a)
    best_a = a.copy()
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    for _ in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        M_curr = ac.max()
        top_idx = np.argsort(ac)[::-1][:top_peaks]

        for pk in top_idx:
            k = int(pk)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[k - idx]
            top_c = np.argsort(contrib)[::-1][:5]

            for tc in top_c:
                src = int(idx[tc])
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                # Find destination with low contribution to this peak
                dst_candidates = np.where(a > 1e-6)[0]
                if len(dst_candidates) < 10:
                    dst_candidates = np.arange(n)
                # Score by how much moving to dst reduces peak
                best_dst = None
                best_reduction = 0
                for _ in range(40):
                    dst = int(rng.choice(dst_candidates))
                    if dst == src:
                        continue
                    # Check if dst contributes to this peak
                    if i_start <= dst <= i_end:
                        continue  # Would still contribute
                    amt = src_val * 0.01
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    new_peak_at_k = np.sum(a_trial[max(0,k-n+1):min(n,k+1)] *
                                           a_trial[k - np.arange(max(0,k-n+1),min(n,k+1))])
                    reduction = ac[pk] - new_peak_at_k
                    if reduction > best_reduction:
                        best_reduction = reduction
                        best_dst = dst

                if best_dst is not None and best_reduction > 1e-15:
                    amt = src_val * 0.008
                    a[src] -= amt
                    a[best_dst] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                    break

    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3600
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
            best_seq = a_norm.copy()
            best_score = sc

    # === Phase 1: Quick AMD to get into basin ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AMD warmup (n={n0}) ===")
    t1 = min(remaining * 0.08, 130.0)
    if t1 > 20:
        a_out, _ = amd_optimize(best_seq, t1, top_k=60, beta=4.5, c0=3.5,
                                 restart_every=3000, verbose=True)
        try_update(a_out, "amd_warm")

    # === Phase 2: Main LP-FW (bulk of time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LP-FW main (n={n_best}) ===")
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 40 and n_best <= 5000:
        a_out, _ = lp_frank_wolfe(best_seq, t2, solver, verbose=True,
                                   top_K_max=350, margin_init=3e-3,
                                   margin_min=5e-5, tighten_factor=0.80)
        try_update(a_out, "lp_main")

    # === Phase 3: Peak crush ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Peak crush (n={n_best}) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        a_out, _ = peak_crush(best_seq, t3, verbose=True, rng_seed=42)
        try_update(a_out, "peakhug")

    # === Phase 4: LP-FW tighter ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW tight (n={n_best}) ===")
    t4 = min(remaining * 0.20, 330.0)
    if t4 > 30 and n_best <= 5000:
        a_out, _ = lp_frank_wolfe(best_seq, t4, solver, verbose=True,
                                   top_K_max=500, margin_init=1e-3,
                                   margin_min=1e-5, tighten_factor=0.75)
        try_update(a_out, "lp_tight")

    # === Phase 5: Try adjacent lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Length exploration ===")
    lengths_to_try = []

        remaining = time_budget - (time.monotonic() - t_global)

            break

                                     restart_every=2000, verbose=False)
            try_update(a_out, f"amd_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 100 and new_n <= 4500:
            t5b = min(remaining * 0.12, 120.0)

                                     restart_every=1500, verbose=False)

    # === Phase 7: Final LP on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP (n={n_best}) ===")
    t7 = min(remaining * 0.20, 300.0)
    if t7 > 30 and n_best <= 5000:
        a_out, _ = lp_frank_wolfe(best_seq, t7, solver, verbose=True,
                                   top_K_max=500, margin_init=5e-4,
                                   margin_min=1e-5, tighten_factor=0.70)
        try_update(a_out, "lp_final")

    t8 = min(remaining * 0.30, 200.0)
    if t8 > 15:
        a_out, _ = peak_crush(best_seq, t8, verbose=True, rng_seed=555)
        try_update(a_out, "polish")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        x = np.linspace(0, 1, n_fb)
        best_seq = norm_sum1(np.sin(np.pi * x) ** 2)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 60:
        print(f"WARNING: Only {remaining_time:.0f}s remaining")

    return [float(x) for x in best_seq]
