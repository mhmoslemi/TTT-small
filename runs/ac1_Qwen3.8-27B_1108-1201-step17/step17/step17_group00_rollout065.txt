def _fast_autocorr(a):

def _interpolate_sequence(f, new_n):

        return np.ones(new_n) / new_n

    ss = f_interp.sum()
    if ss < 1e-15:

def _lp_solve_direction(coeffs, solver):
    n = coeffs.shape[1]
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
    s = g_val.sum()
    if s < 1e-15:
        return None, None
    m_val = float(M_var.value) if M_var.value is not None else float(coeffs @ (g_val / s)).max()
    return g_val / s, m_val

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = _evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = _norm_sum1(parent)
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    if best_seq is None:
        n_base = 5000
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = _norm_sum1(best_seq)
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def _check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Parametric init ===
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.05, 40.0)
    if t1 > 10:
        a_p, v_p = _parametric_init(n0, best_seq, 2.0 * n0 * _fast_autocorr(best_seq).max())
        _check_update(a_p, f"param_n{n0}")

    # === Phase 2: Mirror descent (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Mirror Descent (n={n_best}) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _soft_max_mirror_descent(a_in, t2, verbose=True)
        _check_update(a_out, f"md_n{n_best}")

    # === Phase 3: Iterative balancing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Iterative Balancing (n={n_best}) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _iterative_balancing(a_in, t3, verbose=True)
        _check_update(a_out, f"bal_n{n_best}")

    # === Phase 4: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-FW (n={n_best}) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30 and n_best <= 6000:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _lp_frank_wolfe(a_in, t4, solver, verbose=True)
        _check_update(a_out, f"lp_n{n_best}")

    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord polish (n={n_best}) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _coordinate_polish(a_in, t5, rng_seed=77, verbose=True)
        _check_update(a_out, f"coord_n{n_best}")

    # === Phase 6: Try a different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    alt_lengths = []
    if n_best > 4500:
        alt_lengths = [n_best - 500, n_best + 500]
    else:
        alt_lengths = [n_best + 500, n_best + 1000]
    alt_lengths = [l for l in alt_lengths if 3000 <= l <= 6500]
    for new_n in alt_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"\n=== Phase 6: Alt length n={new_n} ===")
        a_long = _interpolate_sequence(best_seq, new_n)
        t6a = min(remaining * 0.35, 180.0)
        if t6a > 30:
            a_out, _ = _soft_max_mirror_descent(a_long, t6a, verbose=False)
            _check_update(a_out, f"md_alt_n{new_n}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t6b = min(remaining * 0.40, 150.0)
            if t6b > 20:
                a_in = _norm_sum1(best_seq)
                a_out, _ = _iterative_balancing(a_in, t6b, verbose=False)
                _check_update(a_out, f"bal_alt_n{new_n}")

    # === Phase 7: Final MD on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final MD (n={n_best}) ===")
    t7 = min(remaining * 0.40, 250.0)
    if t7 > 20:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _soft_max_mirror_descent(a_in, t7, verbose=True)
        _check_update(a_out, f"md_final_n{n_best}")

    # === Phase 8: Final balance + coord ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final balance (n={n_best}) ===")
    t8 = min(remaining * 0.35, 120.0)
    if t8 > 10:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _iterative_balancing(a_in, t8, verbose=True)
        _check_update(a_out, f"bal_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final coord (n={n_best}) ===")
    t9 = min(remaining * 0.50, 80.0)
    if t9 > 10:
        a_in = _norm_sum1(best_seq)
        a_out, _ = _coordinate_polish(a_in, t9, rng_seed=314, verbose=True)
        _check_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 5000
        x = np.linspace(0, 1, n_fb)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = _norm_sum1(best_seq)
        best_score = _evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
