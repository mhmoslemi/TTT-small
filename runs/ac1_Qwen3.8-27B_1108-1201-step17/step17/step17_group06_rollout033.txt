```python
def compute_gradient(a, tau):
    n = len(a)
    conv = fast_conv(a)
    s = a.sum()
    
    # Softmax weights
    shifted = tau * (conv - conv.max())
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    # LSE value
    L = (conv.max() + np.log(exp_shifted.sum())) / tau
    
    # Cross-correlation: R[i] = sum_j p[i+j] * a[j]
    # Using FFT: R = correlate(p, a) at lags 0 to n-1
    a_flip = a[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AF = np.fft.rfft(a_flip, fft_size)
    conv_full = np.fft.irfft(P * AF, fft_size)[:3*n-2]
    R = conv_full[n-1:2*n-1]
    
    # Gradient
    grad = 4.0 * n * (s * R - L) / (s ** 3)
    
    # Function value
    fval = 2.0 * n * L / (s * s)
    
    return fval, grad
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B with log-sum-exp smoothed objective and analytic gradients
2. Mirror descent (multiplicative weight updates) with adaptive subgradient
3. Coordinate descent with pressure-based mass transfer
4. Multi-scale length exploration
5. Power-ramp and other diverse initializations
Key insight: gradient of LSE-smoothed max-conv objective computed via FFT cross-correlation
"""

import numpy as np
import time
from scipy.optimize import minimize_bfgs


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_J(a):
    """Compute J = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = fast_conv(a)
    s = np.sum(a)
    if s < 1e-15:
        return float('inf')
    return float(2.0 * n * conv.max() / (s * s))


def lse_obj_grad(a, tau):
    """Compute log-sum-exp smoothed objective and its gradient."""
    n = len(a)
    conv = fast_conv(a)
    s = np.sum(a)
    
    # Numerically stable log-sum-exp
    c_max = conv.max()
    shifted = tau * (conv - c_max)
    exp_shifted = np.exp(shifted)
    log_sum = c_max + np.log(exp_shifted.sum()) / tau
    L = log_sum  # This is (1/tau)*log(sum(exp(tau*conv)))
    
    # Softmax weights
    p = exp_shifted / exp_shifted.sum()
    
    # Cross-correlation: R[i] = sum_j p[i+j] * a[j] for i=0..n-1
    a_flip = a[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    AF = np.fft.rfft(a_flip, fft_size)
    conv_full = np.fft.irfft(P * AF, fft_size)[:3 * n - 2]
    R = conv_full[n - 1:2 * n - 1]
    
    # Gradient: df/da[i] = 4n * (s * R[i] - L) / s^3
    fval = 2.0 * n * L / (s * s)
    grad = 4.0 * n * (s * R - L) / (s ** 3)
    
    return fval, grad


def lse_obj_grad_wrapper(a_flat, tau):
    """Wrapper for scipy minimize."""
    fval, grad = lse_obj_grad(a_flat, tau)
    return fval, grad


def optimize_lbfgsb(a0, time_limit, tau_seq=None):
    """Optimize using L-BFGS-B with annealed log-sum-exp temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_J = eval_J(a)
    t_start = time.monotonic()
    
    if tau_seq is None:
        tau_seq = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    
    for tau in tau_seq:
        if time.monotonic() - t_start > time_limit - 5:
            break
        t_phase_start = time.monotonic()
        phase_time = min(60.0, time_limit - 5 - (time.monotonic() - t_start))
        
        def obj_and_grad(x):
            x_pos = np.maximum(x, 1e-15)
            return lse_obj_grad(x_pos, tau)
        
        try:
            res = minimize_bfgs(
                obj_and_grad,
                a,
                jac=True,
                options={'maxiter': 200, 'gtol': 1e-8, 'maxfun': 500}
            )
            a_new = np.maximum(res.x, 1e-15)
            a_new = a_new / a_new.sum()
            J_new = eval_J(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                # Keep a as is for next phase
                pass
        except Exception:
            pass
        
        # Check time
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit - 10:
            break
    
    return best_a, best_J


def mirror_descent(a0, time_limit, top_k=100, beta_w=8.0, c0=3.0,
                   restart_every=5000, verbose=False):
    """Nesterov-accelerated mirror descent with adaptive parameters."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val = eval_J(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = top_k
    cur_beta = beta_w

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 2.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Subgradient from top-K lags
        conv = fast_conv(y)
        k_max = min(cur_top_k, 2 * n - 1)
        top_idx = np.argsort(conv)[::-1][:k_max]
        top_vals = np.maximum(conv[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                j_idx = k_int - idx
                g[idx] += 2.0 * wi * y[j_idx]

        # Step size
        eta = c0 / (np.sqrt(t) + 1.0)

        # Mirror step (multiplicative weight)
        log_r = np.log(y) - eta * g
        log_r -= log_r.max()
        a_new = np.exp(log_r)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x /= x.sum()

        # Track best
        val = eval_J(a_new)
        if val < best_val - 1e-14:
            best_val = val
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Restart logic
        if stall > restart_every:
            stall = 0
            scale = rng.uniform(0.003, 0.02)
            pert = rng.normal(0, scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-12)
            x_r /= x_r.sum()
            x = x_r.copy()
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 300)
            cur_beta = max(cur_beta - 0.5, 2.0)
            c0 = max(c0 * 0.92, 0.5)

        if verbose and t % 5000 == 0:
            print(f"    MD t={t} ({time.monotonic()-t0:.0f}s): J={best_val:.6f}, k={cur_top_k}")

    return best_a, best_val

def make_power_ramp(n, p=1.0):
    """Power-law ramp: a_i proportional to (i+1)^p."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_gaussian(n, width_frac=0.3, center=0.5):
    """Gaussian bump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width_frac) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_bimodal(n, width=60.0):
    """Bimodal distribution."""
    x = np.linspace(0, 1, n)
    a = np.exp(-width * (x - 0.15) ** 2) + np.exp(-width * (x - 0.85) ** 2) + 0.01
    return a / a.sum()


def make_smooth_random(n, seed=0):
    """Smooth random via low-frequency Fourier."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for freq in range(1, 15):
        coeff = rng.normal(0, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_weighted_ramp(n, p=1.0, q=0.5):
    """Weighted ramp: a_i ∝ (i+1)^p * (n-i)^q."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = i ** p * j ** q
    return a / a.sum()


def make_sinusoidal(n, freq=1.0, phase=0.0):
    """Sinusoidal modulation on a ramp."""
    x = np.linspace(0, 1, n)
    a = (x + 0.1) * (1.0 + 0.3 * np.sin(2 * np.pi * freq * x + phase))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
