_global_solver = None

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.2])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")
    return _global_solver


def fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return result


def eval_local(a):
    """Compute objective value for normalized non-negative sequence."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    ac = fast_autocorr(a)
    return 2.0 * len(a) * ac.max()


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interp_to_length(a, new_n):
    """Interpolate sequence to new length, preserving sum=1 normalization."""
    a = norm_sum1(a)
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0.0)
    return norm_sum1(a_interp)


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, K_init=100, K_max=None,
                    nesterov=True, max_iters=10000):
    """
    LP-based Frank-Wolfe with Nesterov acceleration and adaptive constraint selection.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = eval_local(a)
    t0 = time.monotonic()
    
    if K_max is None:
        K_max = min(500, 2 * n - 1)
    
    K = min(K_init, 2 * n - 1)
    y = a.copy()
    x_mom = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current convolution
        ac = fast_autocorr(y)
        M = ac.max()
        
        # Select binding constraints (top K peaks)
        if K >= 2 * n - 1:
            sel_idx = np.arange(2 * n - 1)
        else:
            sel_idx = np.argsort(ac)[::-1][:K]
        
        # Build constraint matrix for selected lags
        # T[i, j] = y[k_i - j] where 0 <= k_i - j < n
        T = np.zeros((len(sel_idx), n))
        for i, k in enumerate(sel_idx):
            ki = int(k)
            js = np.arange(max(0, ki - n + 1), min(n, ki + 1))
            T[i, js] = y[ki - js]
        
        # Solve LP: max sum(b) s.t. T @ b <= M, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M * np.ones(len(sel_idx))]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = b_var.value
        if b_val is None:
            continue
        b_val = np.maximum(np.asarray(b_val, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            b_val = a.copy()
            b_sum = 1.0
        
        g = b_val / b_sum  # Frank-Wolfe direction
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if nesterov:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        else:
            x_new = y_new.copy()
        
        y = y_new
        x_mom = x_new
        
        # Update best
        obj_y = eval_local(y)
        if obj_y < best_obj:
            best_obj = obj_y
            best_a = y.copy()
        
        obj_x = eval_local(x_mom)
        if obj_x < best_obj:
            best_obj = obj_x
            best_a = x_mom.copy()
        
        # Adaptively increase K
        if t % 10 == 0 and K < K_max:
            K = min(K + 20, K_max)
        
        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} K={K} ({elapsed:.0f}s): obj={best_obj:.6f}")
    
    return best_a, best_obj


def soft_subgradient(a, ac, top_k=30, beta=4.0):
    """Compute weighted subgradient focusing on top peaks."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        js = np.arange(max(0, ki - n + 1), min(n, ki + 1))
        grad[js] += 2.0 * wi * a[ki - js]
    return grad


def smooth_descent(a0, time_limit, top_k=50, beta=5.0, c0=4.0, 
                   restart_every=5000, verbose=False, rng_seed=42):
    """Nesterov-accelerated mirror descent with soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    cur_top_k = top_k
    cur_beta = beta
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        g = soft_subgradient(y, ac, top_k=cur_top_k, beta=cur_beta)
        
        eta = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * g
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > restart_every:
            stall = 0
            pert = rng.normal(0, 0.02, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 10, 100)
            cur_beta = max(cur_beta - 0.5, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    SD restart ({elapsed:.0f}s): obj={2*best_m:.6f}")
    
    return best_a, 2.0 * n * best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=123, n_top=10):
    """Aggressive coordinate descent for final polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_obj = eval_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        for k in top_k[:5]:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = np.arange(max(0, ki - n + 1), min(n, ki + 1))
            contrib = a[js] * a[ki - js]
            if contrib.max() < 1e-20:
                continue
            
            src = js[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find best destination (low contribution to peaks)
            sens = np.zeros(n)
            for kk in top_k[:5]:
                kki = int(kk)
                jss = np.arange(max(0, kki - n + 1), min(n, kki + 1))
                sens[jss] += a[kki - jss]
            
            candidates = np.where(sens < sens.mean())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            dst = candidates[np.argmin(sens[candidates])]
            
            if dst == src:
                continue
            
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                if a_trial.sum() < 1e-15:
                    continue
                a_trial = a_trial / a_trial.sum()
                obj_trial = eval_local(a_trial)
                if obj_trial < best_obj - 1e-10:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            if stall % 50 == 0 and verbose:
                elapsed = time.monotonic() - t0
                print(f"    Coord ({elapsed:.0f}s): obj={best_obj:.6f}, stall={stall}")
            
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj_new = eval_local(a)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_obj


def multi_resolution_optimize(a_coarse, target_n, time_limit, solver, verbose=True):
    """Multi-resolution optimization: interpolate to intermediate lengths and optimize."""
    t0 = time.monotonic()
    current = a_coarse.copy()
    n_current = len(current)
    
    if n_current >= target_n:
        return current, eval_local(current)
    
    # Define resolution ladder
    resolutions = []
    while n_current < target_n:
        # Double or go to target
        n_next = min(2 * n_current, target_n)
        if n_next == n_current:
            n_next = target_n
        resolutions.append(n_next)
        n_current = n_next
    
    if verbose:
        print(f"  Resolution ladder: {len(a_coarse)} -> {resolutions}")
    
    for i, res_n in enumerate(resolutions):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 30:
            break
        
        current = interp_to_length(current, res_n)
        
        # Allocate time: more time for larger resolutions
        time_frac = (i + 1) / len(resolutions)
        t_alloc = remaining * (time_frac - (i / len(resolutions))) * 0.8
        t_alloc = max(20.0, min(t_alloc, remaining * 0.4))
        
        if verbose:
            print(f"  Resolving at n={res_n} (t_alloc={t_alloc:.0f}s)...")
        
        # LP-FW at this resolution
        K_init = min(50, 2 * res_n - 1)
        K_max = min(200, 2 * res_n - 1)
        a_out, obj_out = lp_frank_wolfe(current, t_alloc, solver, 
                                          verbose=False, K_init=K_init, K_max=K_max)
        current = a_out
        
        if verbose:
            obj_final = eval_local(current)
            print(f"  n={res_n}: obj={obj_final:.6f}")
    
    return current, eval_local(current)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            sc = float(evaluate_sequence([float(x) for x in a_norm]))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
