def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    return normalize(f_interp)


def spectral_flatten(a, n_iter=500, eta_init=0.3, eta_end=0.1, verbose=False):

            break

        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Frank-Wolfe step with Nesterov
        if nesterov:
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = normalize(y_new)
            
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - prev_y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        else:
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g_dir
            y_new = normalize(y_new)
            x_new = y_new
        
        prev_y = y
        y = y_new
        
        # Check both y and x (Nesterov)
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        if x_new is not None:
            mx = fast_autocorr(x_new).max()
            if mx < best_m:
                best_m = mx
                best_a = x_new.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={2*n*best_m:.6f}, M={M_curr:.2e}")
    
    return best_a, best_m


def mirror_descent(a0, time_limit, verbose=False, top_k=50, beta=5.0, c0=4.0, 
                   restart_every=5000, adaptive_k=True):
    """
    Nesterov-accelerated mirror descent with multi-peak subgradient.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta
    restart_count = 0
    
    while time.monotonic() - t0 < time_limit:
        beta_t_factor = (restart_count * restart_every + 1 - 1.0) / (restart_count * restart_every + 1 + 1.5)
        
        # Do one step
        y = x + beta_t_factor * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-30:
            y = x.copy()
        else:
            y = y / sy
        
        ac = fast_autocorr(y)
        
        # Compute subgradient: weighted sum of contributions from top peaks
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        
        eta = gamma / (np.sqrt(restart_count * restart_every + 2) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > restart_every:
            stall_count = 0
            restart_count += 1
            # Perturb and restart
            strength = min(0.15, 0.03 + 0.02 * min(restart_count, 5))
            pert = rng.normal(0, strength, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            
            if adaptive_k:
                cur_top_k = min(cur_top_k + 10, 150)
                cur_beta = max(cur_beta - 0.5, 2.0)
            gamma = c0 * (1.0 + 0.2 * min(restart_count, 5))
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    MD restart #{restart_count} ({elapsed:.0f}s): J={2*n*best_m:.6f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, verbose=False, rng_seed=42):
    """
    Coordinate descent: try moving mass from high-contributing positions to low-cost positions.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        n_top = min(10, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]
        
        improved = False
        
        for k in top_k:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Compute "cost" at each position (contribution to all top peaks)
            costs = np.zeros(n)
            for k2 in top_k[:6]:
                k2i = int(k2)
                for d in range(max(0, k2i - n + 1), min(n, k2i + 1)):
                    j1 = k2i - d
                    j2 = d - k2i
                    c = 0.0
                    if 0 <= j1 < n:
                        c += a[j1]
                    if 0 <= j2 < n and j2 != j1:
                        c += a[j2]
                    costs[d] += c
            
            # Find good destination (low cost, not already high)
            candidates = np.argsort(costs + 10*a)[:max(1, n // 5)]
            best_dst = candidates[0]
            
            if best_dst == src:
                if len(candidates) > 1:
                    best_dst = candidates[1]
                else:
                    best_dst = int(rng.integers(n))
                    if best_dst == src:
                        continue
            
            # Try different transfer fractions
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if not improved and stall > 50:
                # Random perturbation
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
        
        if not improved:
            stall += 1
            # Random single-pair swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord ({elapsed:.0f}s): J={2*n*best_m:.6f}, stall={stall}")
    
    return best_a, best_m


def multi_peak_mirror_descent(a0, time_limit, verbose=False, top_k=80, beta=4.0, c0=5.0):
    """
    Enhanced mirror descent with:
    - Multiple subgradient estimators (different top_k values)
    - Adaptive switching between them
    - Better restart strategy
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall = 0
    t_iter = 0
    
    # Multiple configurations to try
    configs = [
        {'top_k': 30, 'beta': 6.0, 'c0': 6.0},
        {'top_k': 50, 'beta': 5.0, 'c0': 5.0},
        {'top_k': 80, 'beta': 4.0, 'c0': 4.0},
        {'top_k': 120, 'beta': 3.0, 'c0': 3.5},
        {'top_k': 200, 'beta': 2.5, 'c0': 3.0},
    ]
    cfg_idx = 1  # start with middle config
    
    while time.monotonic() - t0 < time_limit:
        t_iter += 1
        cfg = configs[cfg_idx]
        
        y = x + (t_iter / (t_iter + 1.5)) * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        ac = fast_autocorr(y)
        k_max = min(cfg['top_k'], len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cfg['beta']
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * a[j_idx]
        
        eta = cfg['c0'] / (np.sqrt(t_iter) + 1.0)
        log_a = np.log(y) - eta * grad
        a_new = np.exp(log_a - log_a.max())
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive restart
        if stall > 3000:
            stall = 0
            cfg_idx = (cfg_idx + 1) % len(configs)
            
            # Different restart strategies
            strategy = rng.integers(3)
            if strategy == 0:
                # Gaussian perturbation
                strength = rng.uniform(0.02, 0.1)
                x = best_a + strength * rng.normal(0, 1, n)
            elif strategy == 1:
                # Mix with uniform
                x = (1 - 0.1) * best_a + 0.1 * np.ones(n) / n
            else:
                # Mirror descent restart with different config
                x = best_a + 0.05 * rng.normal(0, 1, n)
            
            x = np.maximum(x, 1e-10)
            x = x / x.sum()
            x_prev = x.copy()
            t_iter = 0
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    MMD restart ({elapsed:.0f}s): J={2*n*best_m:.6f}, cfg={cfg_idx}")
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """
    Main search orchestrator combining multiple optimization methods:
    1. Spectral flattening (fast initial improvement)
    2. LP-based Frank-Wolfe (principled optimization)
    3. Multi-peak mirror descent (robust global search)
    4. Coordinate descent (final polishing)
    5. Multi-length exploration
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = find_lp_solver()
    print(f"Total time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = normalize(parent)
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 4200
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = normalize(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Spectral flattening (n={n0}) ===")
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 30:
        a_in = normalize(best_seq)
        a_out, _ = spectral_flatten(a_in, n_iter=1000, eta_init=0.5, eta_end=0.15, verbose=False)
        try_update(a_out, f"spectral_n{n0}")
    
    # === Phase 2: Multi-peak mirror descent (primary optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror descent (n={len(best_seq)}) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30:
        a_in = normalize(best_seq)
        a_out, _ = multi_peak_mirror_descent(a_in, t2, verbose=True, top_k=80, beta=4.0, c0=5.0)
        try_update(a_out, f"mmd_n{len(best_seq)}")
    
    # === Phase 3: LP Frank-Wolfe ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW (n={n_best}) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30 and n_best <= 5000:
        a_in = normalize(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True, top_K=300, nesterov=True)
        try_update(a_out, f"lpfw_n{n_best}")
    
    # === Phase 4: Additional mirror descent with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: MD variant (n={n_best}) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_in = normalize(best_seq)
        # Slightly perturb before running
        a_pert = a_in + 0.02 * rng.normal(0, 1, n_best)
        a_pert = normalize(a_pert)
        a_out, _ = mirror_descent(a_pert, t4, verbose=True, top_k=60, beta=5.0, c0=4.5)
        try_update(a_out, f"md_var_n{n_best}")
    
    # === Phase 5: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coordinate polish (n={n_best}) ===")
    t5 = min(remaining *
0.10, 120.0)
    if t5 > 15:
        a_in = normalize(best_seq)
        a_out, _ = coordinate_polish(a_in, t5, verbose=True, rng_seed=77)
        try_update(a_out, f"coord_n{n_best}")
    
    # === Phase 6: Spectral + MD combo on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Spectral+MD (n={n_best}) ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 20:
        a_in = normalize(best_seq)
        a_spectral, _ = spectral_flatten(a_in, n_iter=500, eta_init=0.4, eta_end=0.1)
        try_update(a_spectral, f"spectral2_n{n_best}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t6b = min(remaining * 0.5, 80.0)
            if t6b > 10:
                a_out, _ = multi_peak_mirror_descent(a_spectral, t6b, verbose=False)
                try_update(a_out, f"mmd_after_spec_n{n_best}")
    
    # === Phase 7: Try slightly different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration ===")
    for delta in [200, -200, 500, -500]:
        new_n = n_best + delta
        if new_n < 1000 or new_n > 8000:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}...")
        a_interp = interpolate_to_length(best_seq, new_n)
        try_update(a_interp, f"interp_n{new_n}")
        
        t7 = min(remaining * 0.3, 100.0)
        if t7 > 20:
            a_out, _ = multi_peak_mirror_descent(a_interp, t7, verbose=False)
            try_update(a_out, f"mmd_n{new_n}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t7b = min(remaining * 0.3, 60.0)
            if t7b > 15:
                a_out, _ = coordinate_polish(normalize(best_seq if len(best_seq)==new_n else a_interp), t7b, verbose=False, rng_seed=99)
                try_update(a_out, f"coord_n{new_n}")
    
    # === Phase 8: Final LP if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    if remaining > 40 and n_best <= 5000:
        t8 = min(remaining * 0.4, 150.0)
        if t8 > 20:
            a_in = normalize(best_seq)
            a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=350, nesterov=True)
            try_update(a_out, f"lp_final_n{n_best}")
    
    # === Phase 9: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final polish (n={n_best}) ===")
    if remaining > 15:
        t9 = min(remaining * 0.8, 100.0)
        if t9 > 10:
            a_in = normalize(best_seq)
            a_out, _ = coordinate_polish(a_in, t9, verbose=True, rng_seed=
