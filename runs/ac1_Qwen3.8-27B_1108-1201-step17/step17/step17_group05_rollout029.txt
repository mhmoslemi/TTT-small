Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Peak-directed mirror descent with exact subgradient
2. Sensitivity-based coordinate descent with lookahead

4. Multi-length exploration with interpolation

import numpy as np
import time

def fast_autocorr(a):
    """Compute linear autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def exact_subgradient(a, top_k=5):
    """
    Compute subgradient of max_k conv(a,a)[k] using top-k maximizing lags.
    grad[j] = 2 * sum_{k in top} w_k * a[k-j] (if 0 <= k-j < n)
    """
    n = len(a)
    ac = fast_autocorr(a)
    # Get top-k lags
    k_top = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_top]
    top_vals = ac[top_idx]
    # Weights: uniform (exact for single peak) or proportional
    if k_top == 1:
        weights = np.array([1.0])
    else:
        # Use the relative values as weights (favors the dominant peak)
        weights = top_vals / top_vals.sum()
    
    grad = np.zeros(n, dtype=np.float64)
    for k, w in zip(top_idx, weights):
        ki = int(k)
        # grad[j] += 2*w*a[ki-j] for 0<=ki-j<n
        # j ranges from max(0, ki-n+1) to min(n-1, ki)
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        vals = a[ki - j_lo:ki - j_hi + 1][::-1]  # a[ki-j] for j from j_lo to j_hi
        grad[j_lo:j_hi+1] += 2.0 * w * vals
    
    return grad, ac, float(ac.max())

def mirror_descent_exact(a0, time_limit, verbose=True, top_k=3, c0=3.0, seed=42):
    """
    Mirror descent on simplex with exact subgradient (top-k lags).
    Uses log-domain updates for numerical stability.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    stall = 0
    iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        # Nesterov extrapolation
        beta_t = (iterations - 1.0) / (iterations + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute exact subgradient
        g, _, M_curr = exact_subgradient(y, top_k=top_k)
        
        # Mirror descent step in log domain
        eta = c0 / (np.sqrt(iterations) + 1.0)
        log_y = np.log(y)
        log_new = log_y - eta * g
        log_new -= log_new.max()  # numerical stability
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        x_prev = x.copy()
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall > 3000:
            stall = 0
            # Perturb around best
            noise = rng.normal(0, 0.05, n) * best_a
            x_r = best_a + noise
            x_r = norm_sum1(x_r)
            x = x_r
            x_prev = x_r.copy()
            # Increase diversity: try different top_k
            top_k = int(rng.integers(1, 8))
        
        if verbose and iterations % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD-exact iter={iterations} ({elapsed:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

            if j_lo >= j_hi:
                continue

            s[j_lo:j_hi+1] = a[ki - j_lo:ki - j_hi + 1][::-1]

            src_candidates = np.where(s > s.mean())[0]
            dst_candidates = np.where(s < s.mean())[0]

                dst = dst_candidates[rng.integers(len(dst_candidates))]
                if src == dst or a[src] < 1e-12:
                    continue

                    M_trial = fast_autocorr(a_trial).max()

                        a = a_trial
                        best_m = M_trial

                        accepted += 1

                    break

                break

                    j = int(rng.integers(n))

                        M_new = fast_autocorr(a).max()

def spectral_smooth_init(n, seed=0, n_modes=5):
    """Create a smooth non-negative initialization."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for f in range(1, n_modes + 1):
        coeff = rng.normal(0, 0.1)
        phase = rng.uniform(0, 2 * np.pi)
        a += coeff * np.sin(2 * np.pi * f * x + phase)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = f / f.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def antisymmetric_envelope(n, seed=0):
    """
    Create an asymmetric envelope that avoids symmetric autocorrelation peaks.
    Uses a shifted Gaussian or sigmoid shape.
    """
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    # Asymmetric: peak at a random position (not center)
    center = rng.uniform(0.25, 0.75)
    width = rng.uniform(0.15, 0.35)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    # Add some asymmetry
    a *= (1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 3) * x + rng.uniform(0, 2*np.pi)))
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]

def soft_subgradient(a, beta=6.0, top_k=50):
    """Compute soft subgradient using top-k lags with exp weighting."""
    n = len(a)
    ac = fast_autocorr(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx]
    shifted = top_vals - top_vals.max()
    w_top = np.exp(beta * shifted)
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        w_top = np.ones(k_max) / k_max
    else:
        w_top = w_top / w_sum

    # Build effective weight vector (sparse)
    total_len = n + k_max  # conservative
    fft_size = 1 << (2 * n - 1).bit_length()
    a_rev = np.zeros(fft_size)
    a_rev[:n] = a[::-1]
    w_full = np.zeros(fft_size)
    for wi, idx in zip(w_top, top_idx):
        w_full[int(idx)] = wi
    conv = np.fft.irfft(np.fft.rfft(a_rev) * np.fft.rfft(w_full), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, ac.max()


def mirror_descent_optimize(a0, time_limit, verbose=True, beta_init=7.0,
                            beta_min=3.0, c0=4.0, restart_every=4000,
                            top_k_init=60, top_k_max=100, seed=42):
    """Mirror descent on simplex with adaptive soft subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        iterations += 1
        beta_t = (iterations - 1.0) / (iterations + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        # Adaptive top-k and beta
        cur_beta = max(beta_min, beta_init - 0.002 * iterations)
        cur_top_k = min(top_k_max, top_k_init + iterations // 100)

        g, M_curr = soft_subgradient(y, beta=cur_beta, top_k=cur_top_k)

        eta = c0 / (np.sqrt(iterations) + 1.0)
        log_y = np.log(y)
        log_new = log_y - eta * g
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        x_prev = x.copy()
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-13:
            best_m = M_new
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > restart_every:
            stall = 0
            noise_scale = max(0.02, 0.08 * np.exp(-stall / 2000))
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = norm_sum1(x_r)
            x = x_r
            x_prev = x_r.copy()
            if rng.random() < 0.25:
                # More aggressive restart

                x = x_r
                x_prev = x_r.copy()

        if verbose and iterations % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter={iterations} ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m

            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Adaptive number of constraints
        K = min(K_max, K_init + t // 5)

        # Select constraints: mix of top by value and spread
        top_idx = np.argsort(ac)[::-1][:K]
        T = np.zeros((K, n))
        for col_idx, m in enumerate(top_idx):
            mi = int(m)
            j_start = max(0, mi - n + 1)
            j_end = min(n - 1, mi)
            js = np.arange(j_start, j_end + 1)
            if len(js) > 0:
                T[col_idx, js] = y[mi - js]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K),
                       cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            no_improve += 1
            if no_improve > 20:
                # Restart
                noise = np.random.default_rng(t).normal(0, 0.03, n) * best_a
                y = norm_sum1(best_a + noise)
                prev_y = y.copy()
                no_improve = 0
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Nesterov step with line search
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        # Line search: try a few step sizes
        best_alpha_eff = alpha
        best_M_trial = fast_autocorr(y_new).max()
        for trial_alpha in [0.5 * alpha, 1.5 * alpha, 2.0 * alpha]:
            yt = (1.0 - trial_alpha) * y + trial_alpha * g_dir
            yt = np.maximum(yt, 0.0)
            st = yt.sum()
            if st < 1e-15:
                continue
            yt = yt / st
            Mt = fast_autocorr(yt).max()
            if Mt < best_M_trial:
                best_M_trial = Mt
                best_alpha_eff = trial_alpha
                y_new = yt

        y_new = (1.0 - best_alpha_eff) * y + best_alpha_eff * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy

        prev_y = y
        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and t % 2 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}")

    return best_a, best_m


def coordinate_descent_targeted(a0, time_limit, verbose=True, seed=123):
    """
    Targeted coordinate descent focusing on peak lag reduction.
    Uses sensitivity analysis and lookahead.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 300
    accepted = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()
        peak_lags = np.argsort(ac)[::-1][:8]

        improvement = False

        for k in peak_lags[:4]:
            if time.monotonic() - t0 > time_limit:
                break

            ki = int(k)
            j_lo = max(0, ki - n + 1)
            j_hi = min(n - 1, ki)
            if j_lo >= j_hi:
                continue

            # Sensitivity: s[j] = a[ki-j]
            s = np.zeros(n)
            s[j_lo:j_hi + 1] = a[ki - j_lo:ki - j_hi + 1][::-1]
            s_mean = s.mean()

            src_candidates = np.where(s > s_mean * 1.5)[0]
            dst_candidates = np.where(s < s_mean * 0.5)[0]

            if len(src_candidates) == 0:
                src_candidates = np.where(s > s_max if (s_max := s.max()) > 0)[0]
            if len(dst_candidates) == 0:
                dst_candidates = np.where(s < s_min if (s_min := s.min()) >= 0)[0]

            if len(src_candidates) == 0 or len(dst_candidates) == 0:
                continue

            n_tries = min(30, len(src_candidates) * max(1, len(dst_candidates)))
            for _ in range(n_tries):
                src = src_candidates[rng.integers(len(src_candidates))]
                dst = dst_candidates[rng.integers(len(dst_candidates))]
                if src == dst or a[src] < 1e-12:
                    continue

                for frac in [0.008, 0.004, 0.015, 0.003, 0.01]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] = max(0.0, a_trial[src] - amt)
                    a_trial[dst] += amt
                    a_trial = a_trial / a_trial.sum()
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improvement = True
                        accepted += 1
                        stall = 0
                        break
                if improvement:
                    break

            if improvement:
                break

        if not improvement:
            stall += 1
            if stall % 15 == 0:
                for _ in range(15):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.003, 0.025)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break

        if verbose and (accepted % 50 == 0 or (stall > 0 and stall % 80 == 0)):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord stall={stall} ({elapsed:.0f}s): J={J:.6f}, acc={accepted}")

    return best_a, best_m


def make_ramp_init(n, direction=1, seed=0, noise_level=0.0):
    """Create a ramp-shaped initialization (asymmetric)."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    if direction == 1:
        a = x + 0.1
    else:
        a = (1.0 - x) + 0.1
    if noise_level > 0:
        a += noise_level * rng.normal(0, 1, n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)


def make_bump_init(n, center=0.3, width=0.25, seed=0):
    """Create a single off-center bump."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a += 0.01 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)
