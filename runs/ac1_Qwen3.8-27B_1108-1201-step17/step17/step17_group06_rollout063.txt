def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    c = np.fft.irfft(A*A, fsize)[:size]
    # remove tiny negatives
    np.maximum(c, 0.0, out=c)
    return c

"""
Search for non-negative step-function heights minimizing
    2*n*max(conv(a,a))/(sum(a))**2

The implementation combines:
- FFT-based exact convolution and exact pairwise transfer evaluation,
- exponentiated-gradient / mirror descent with smoothed max coefficients,
- L-BFGS polishing in a softmax parameterization,
- local coordinate descent with exact one-pair mass-transfer moves,
- multi-start initializers and length sweeping.

import numpy as np
import time

try:
    from scipy.optimize import minimize
    SCIPY_AVAILABLE = True
except Exception:
    minimize = None
    SCIPY_AVAILABLE = False


class TimeLimitException(Exception):
    pass


EG_MAIN = ((5, "exp", 0.25),
           (20, "exp", 0.30),
           (60, "exp", 0.25),
           (0, "active", 0.20))

EG_FINAL = ((0, "active", 0.55),
            (20, "exp", 0.25),
            (60, "exp", 0.20))


def normalize_pos(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300 or not np.isfinite(s):
        n = len(a)
        if n <= 0:
            return np.array([1.0], dtype=np.float64)
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    c = np.fft.irfft(A * A, fsize)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def local_J(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    c = fast_conv(p)
    s = float(p.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return float("inf")
    return float(2.0 * len(p) * float(c.max()) / (s * s))


def make_weights(c, concentration=10.0, mode="exp", thresh=0.985):
    c = np.asarray(c, dtype=np.float64).ravel()
    L = len(c)
    m = float(np.max(c))
    if m <= 0.0 or not np.isfinite(m):
        return np.ones(L, dtype=np.float64) / float(L)

    if mode == "active":
        w = (c >= thresh * m).astype(np.float64)
        s = float(w.sum())
        if s <= 0.0:
            w = np.zeros(L, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= s
        return w

    if concentration <= 0.0:
        return np.ones(L, dtype=np.float64) / float(L)

    lam = float(concentration) / m
    z = lam * (c - m)
    z = np.maximum(z, -745.0)
    e = np.exp(z)
    e[e < 1e-300] = 0.0
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def gradient_from_weights(p, w):
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = len(p)
    nz = int(np.count_nonzero(w > 0.0))

    if nz <= 128:
        idxs = np.nonzero(w)[0]
        g = np.zeros(n, dtype=np.float64)
        for k in idxs:
            wk = float(w[k])
            if wk <= 0.0:
                continue
            kk = int(k)
            i_s = kk - n + 1
            if i_s < 0:
                i_s = 0
            i_e = kk
            if i_e >= n:
                i_e = n - 1
            if i_s <= i_e:
                seg = p[kk - i_e:kk - i_s + 1][::-1]
                g[i_s:i_e + 1] += (2.0 * wk) * seg
        return g

    L = 2 * n - 1
    size = L + n - 1
    fsize = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fsize)
    Q = np.fft.rfft(p[::-1], fsize)
    conv = np.fft.irfft(W * Q, fsize)
    return 2.0 * conv[n - 1:n - 1 + n]


def perturb_sequence(a, scale, rng):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    logp = np.log(np.maximum(a, 1e-18))
    if scale > 0.0:
        logp = logp + rng.normal(0.0, float(scale), n)
    logp -= logp.mean()
    p = np.exp(logp - logp.max())
    p = np.maximum(p, 1e-18)
    s = float(p.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return p / s

    rng = np.random.default_rng(777001)
    noise_scale = 0.02
    base = float(initial_base)
    cum = 0.0

    for conc, mode, frac in schedule:
        cum += float(frac)
        seg_end = min(deadline, t0 + float(time_limit) * cum)
        conc_cur = float(conc)
        no_acc = 0

        while time.monotonic() < seg_end:
            if time.monotonic() >= deadline:
                break
            it += 1

            w = make_weights(c, conc_cur, mode, 0.985)
            g = gradient_from_weights(p, w)
            gmax = float(np.max(g))
            if not np.isfinite(gmax) or gmax <= 1e-30:
                break
            g = g / gmax

            accepted = False
            for _ in range(4):
                logp = np.log(np.maximum(p, 1e-18))
                logp = logp - base * g
                logp -= logp.max()
                p_new = np.exp(logp)
                p_new = np.maximum(p_new, 1e-18)
                s = float(p_new.sum())
                if s <= 1e-30 or not np.isfinite(s):
                    base = max(base * 0.5, 1e-6)
                    continue
                p_new /= s
                c_new = fast_conv(p_new)
                J_new = float(2.0 * n * c_new.max())

                if np.isfinite(J_new) and (J_new < current_J - 1e-15 or J_new < best_J - 1e-15):
                    current_J = J_new
                    p = p_new
                    c = c_new
                    if J_new < best_J:
                        best_J = J_new
                        best_p = p.copy()
                        no_acc = 0
                    base = min(base * 1.1, 1.2)
                    accepted = True
                    if verbose and it % 5000 == 0:
                        print(f"    EG it={it} elapsed={time.monotonic()-t0:.0f}s J={best_J:.6f}")
                    break
                else:
                    base = max(base * 0.55, 1e-6)

            if not accepted:
                no_acc += 1
                if no_acc > no_acc_limit or current_J > best_J * 1.02:
                    no_acc = 0
                    logp = np.log(np.maximum(best_p, 1e-18))
                    logp = logp + rng.normal(0.0, noise_scale, n)
                    logp -= logp.mean()
                    p = np.exp(logp - logp.max())
                    p = np.maximum(p, 1e-18)
                    p /= p.sum()
                    c = fast_conv(p)
                    current_J = float(2.0 * n * c.max())
                    base = float(initial_base)
                    noise_scale = min(noise_scale * 1.3, 0.25)
                    if mode != "active" and conc_cur > 2.0:
                        conc_cur = max(2.0, conc_cur * 0.6)

        if time.monotonic() >= deadline:
            break

    return best_p, best_J


def lbfgs_fun(z, state):
    if time.monotonic() - state["t0"] > state["time_limit"]:
        raise TimeLimitException

    n = state["n"]
    concentration = state["concentration"]
    z = np.asarray(z, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return 1e100, np.zeros(n, dtype=np.float64)

    p = z - z.mean()
    p = p - p.max()
    ep = np.exp(p)
    s = float(ep.sum())
    if s <= 1e-300 or not np.isfinite(s):
        p = np.ones(n, dtype=np.float64) / float(n)
    else:
        p = ep / s
    p = np.maximum(p, 1e-18)
    p = p / float(p.sum())

    c = fast_conv(p)
    m = float(c.max())
    if m <= 0.0 or not np.isfinite(m):
        return 1e100, np.zeros(n, dtype=np.float64)

    true_J = float(2.0 * n * m)
    if true_J < state["best_J"]:
        state["best_J"] = true_J
        state["best_p"] = p.copy()

    if concentration <= 0.0:
        smooth_val = m
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        lam = float(concentration) / m
        zc = lam * (c - m)
        zc = np.maximum(zc, -745.0)
        e = np.exp(zc)
        e[e < 1e-300] = 0.0
        denom = float(e.sum())
        if denom <= 0.0 or not np.isfinite(denom):
            smooth_val = m
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / denom
            smooth_val = m + np.log(denom) / lam

    g = gradient_from_weights(p, w)
    gradz = p * (g - float(np.dot(p, g)))
    if not np.all(np.isfinite(gradz)):
        gradz = np.zeros(n, dtype=np.float64)

    return float(2.0 * n * smooth_val), gradz


def lbfgs_optimize(p, time_limit, concentration=15.0, verbose=False):
    p = normalize_pos(p)
    n = len(p)
    c = fast_conv(p)
    state = {
        "n": n,
        "concentration": float(concentration),
        "t0": time.monotonic(),
        "time_limit": float(time_limit),
        "best_p": p.copy(),
        "best_J": float(2.0 * n * c.max()),
    }

    z0 = np.log(np.maximum(p, 1e-18))
    z0 -= z0.mean()

    try:
        t1 = time.monotonic()
        lbfgs_fun(z0, state)
        t_call = max(1e-3, time.monotonic() - t1)
    except Exception:
        return state["best_p"], state["best_J"]

    maxfun = int(max(60, min(10000, (float(time_limit) * 0.85) / t_call / 4.0)))
    maxiter = int(max(20, min(2000, maxfun // 8)))

    try:
        minimize(lbfgs_fun,
                 z0,
                 args=(state,),
                 jac=True,
                 method="L-BFGS-B",
                 options={"maxfun": maxfun,
                          "maxiter": maxiter,
                          "ftol": 1e-12,
                          "gtol": 1e-8,
                          "maxcor": 50})
    except Exception:
        pass

    if verbose:
        print(f"    LBFGS elapsed={time.monotonic()-state['t0']:.0f}s J={state['best_J']:.6f}")

    return state["best_p"], state["best_J"]


def transfer_max(c, p, i, j, delta, n):
    new_c = c.copy()
    if delta <= 0.0:
        return float(np.max(new_c))
    two_d = 2.0 * delta
    new_c[i:i + n] -= two_d * p
    new_c[j:j + n] += two_d * p
    q = delta * delta
    new_c[2 * i] += q
    new_c[2 * j] += q
    new_c[i + j] -= 2.0 * q
    return float(np.max(new_c))


def coordinate_descent_fast(p, time_limit, src_count=14, dst_count=14,
                            fracs=(0.005, 0.02, 0.06, 0.15, 0.35, 0.65),
                            weight_mode="active", thresh=0.99,
                            concentration=30.0, max_no_improve=60, verbose=False):
    p = normalize_pos(p)
    n = len(p)
    c = fast_conv(p)
    best_p = p.copy()
    best_J = float(2.0 * n * c.max())
    best_c = c.copy()

    t0 = time.monotonic()
    deadline = t0 + float(time_limit)
    no_improve = 0
    noise_scale = 0.01
    it = 0
    rng = np.random.default_rng(543211)

    while time.monotonic() < deadline:
        if time.monotonic() >= deadline:
            break
        it += 1

        w = make_weights(c, concentration, weight_mode, thresh)
        g = gradient_from_weights(p, w)

        valid = np.nonzero(p > 1e-12)[0]
        if len(valid) < 2:
            break

        score = p * g
        src_top = np.argsort(score)[::-1][:src_count]
        dst_top = np.argsort(g)[:dst_count]

        rs = min(src_count // 2, len(valid))
        rd = min(dst_count // 2, len(valid))
        if rs > 0:
            rand_src = rng.choice(valid, size=rs, replace=False)
            src_list = np.unique(np.concatenate((src_top, rand_src)))
        else:
            src_list = np.unique(src_top)
        if rd > 0:
            rand_dst = rng.choice(valid, size=rd, replace=False)
            dst_list = np.unique(np.concatenate((dst_top, rand_dst)))
        else:
            dst_list = np.unique(dst_top)

        best_move = None
        best_move_J = best_J

        for i in src_list:
            pi = float(p[i])
            if pi <= 1e-12:
                continue
            for j in dst_list:
                if i == j:
                    continue
                if g[i] <= g[j]:
                    continue
                for frac in fracs:
                    if time.monotonic() >= deadline:
                        break
                    delta = float(frac) * pi
                    if delta <= 1e-15:
                        continue
                    M = transfer_max(c, p, i, j, delta, n)
                    J = 2.0 * n * M
                    if J < best_move_J - 1e-15:
                        best_move_J = J
                        best_move = (int(i), int(j), delta)
                if time.monotonic() >= deadline:
                    break
            if time.monotonic() >= deadline:
                break

        if best_move is not None:
            i, j, d = best_move
            p_new = p.copy()
            p_new[i] = max(0.0, p_new[i] - d)
            p_new[j] += d
            p_new = normalize_pos(p_new)
            c_new = fast_conv(p_new)
            J_new = float(2.0 * n * c_new.max())

            if J_new < best_J - 1e-16:
                best_J = J_new
                best_p = p_new
                best_c = c_new
                p = p_new
                c = c_new
                no_improve = 0
                if verbose:
                    print(f"    CD it={it} elapsed={time.monotonic()-t0:.0f}s J={best_J:.6f}")
            else:
                no_improve += 1
                p = best_p.copy()
                c = best_c.copy()
        else:
            no_improve += 1
            if no_improve % 3 == 0:
                p = perturb_sequence(best_p, noise_scale, rng)
                c = fast_conv(p)
                noise_scale = min(noise_scale * 1.3, 0.15)
            if no_improve > max_no_improve:
                p = best_p.copy()
                c = best_c.copy()
                break

    return best_p, best_J


def optimize_sequence(p, budget, label="", verbose=False, use_lbfgs=True):
    t_start = time.monotonic()
    p = normalize_pos(p)
    c = fast_conv(p)
    best_p = p.copy()
    best_J = float(2.0 * len(p) * c.max())
    n = len(p)

    eg_budget = float(budget) * 0.55
    if eg_budget > 20.0:
        p_out, J_out = eg_optimize(best_p.copy(), eg_budget, schedule=EG_MAIN,
                                   verbose=verbose)
        if J_out < best_J:
            best_p, best_J = p_out, J_out

    remaining = float(budget) - (time.monotonic() - t_start)
    if use_lbfgs and SCIPY_AVAILABLE and remaining > 60.0 and n <= 6000:
        lb_budget = min(remaining * 0.30, 240.0)
        if lb_budget > 30.0:
            p_out, J_out = lbfgs_optimize(best_p.copy(), lb_budget,
                                          concentration=15.0, verbose=verbose)
            if J_out < best_J:
                best_p, best_J = p_out, J_out

    remaining = float(budget) - (time.monotonic() - t_start)
    cd_budget = min(remaining * 0.50, 350.0)
    if cd_budget > 15.0:
        p_out, J_out = coordinate_descent_fast(best_p.copy(), cd_budget,
                                               verbose=verbose)
        if J_out < best_J:
            best_p, best_J = p_out, J_out

    remaining = float(budget) - (time.monotonic() - t_start)
    if remaining > 30.0:
        p_out, J_out = eg_optimize(best_p.copy(), remaining * 0.55,
                                   schedule=EG_FINAL, verbose=verbose)
        if J_out < best_J:
            best_p, best_J = p_out, J_out

    remaining = float(budget) - (time.monotonic() - t_start)
    if remaining > 20.0:
        p_out, J_out = coordinate_descent_fast(best_p.copy(), remaining - 5.0,
                                               verbose=False)
        if J_out < best_J:
            best_p, best_J = p_out, J_out

    if verbose:
        print(f"  optimize_sequence {label}: J={best_J:.6f}, n={len(best_p)}, elapsed={time.monotonic()-t_start:.0f}s")

    return best_p, best_J


def interpolate_seq(a, new_n):
    a = np.asarray(a, dtype=np.float64).ravel()
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n <= 0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    ai = np.interp(new_x, old_x, a)
    ai = np.maximum(ai, 1e-18)
    return normalize_pos(ai)


def make_uniform(n):
    return np.ones(n, dtype=np.float64) / float(n)


def make_power_ramp(n, alpha):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** float(alpha)
    return normalize_pos(a)


def make_end_bump(n, alpha=1.0):
    x = np.linspace(0.0, 1.0, n)
    a = np.abs(2.0 * x - 1.0) ** float(alpha) + 1e-5
    return normalize_pos(a)


def make_center_bump(n, alpha=1.0):
    x = np.linspace(0.0, 1.0, n)
    a = (1.0 - np.abs(2.0 * x - 1.0)) ** float(alpha) + 1e-5
    return normalize_pos(a)


def make_gaussian_mixture(n, centers, width):
    x = np.linspace(0.0, 1.0, n)
    a = np.full(n, 1e-5, dtype=np.float64)
    for c in centers:
        a += np.exp(-float(width) * (x - float(c)) ** 2)
    return normalize_pos(a)


def make_smooth_random(n, seed=0, modes=10, amp=0.08):
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 1.0, n)
    a = np.full(n, 0.5, dtype=np.float64)
    for m in range(1, modes + 1):
        coeff = float(rng.normal(0.0, amp))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        a += coeff * np.sin(2.0 * np.pi * float(m) * x + phase)
    return normalize_pos(a)


def blend_sequence(a, b, t):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = len(a)
    if len(b) != n:
        b = np.interp(np.linspace(0.0, 1.0, n),
                      np.linspace(0.0, 1.0, len(b)),
                      b)
    out = (1.0 - float(t)) * a + float(t) * b
    return normalize_pos(out)


def load_parent():
    g = globals().get("height_sequence_1")
    if g is None:
        return None
    try:
        p = np.asarray(g, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size <= 0 or not np.all(np.isfinite(p)) or p.sum() <= 1e-8:
            return None
        p = normalize_pos(p)
        if len(p) > 8000:
            p = interpolate_seq(p, 5120)
        elif len(p) < 1500:
            p = interpolate_seq(p, 3200)
        return p
    except Exception:
        return None


def build_candidates(n, parent, rng):
    cand = []

    if parent is not None:
        p0 = parent if len(parent) == n else interpolate_seq(parent, n)
        cand.append((p0, "parent"))
        cand.append((perturb_sequence(p0, 0.015, rng), "parent_p1"))
        cand.append((perturb_sequence(p0, 0.04, rng), "parent_p4"))

    cand.append((make_power_ramp(n, 0.5), "ramp05"))
    cand.append((make_power_ramp(n, 1.0), "ramp10"))
    cand.append((make_power_ramp(n, 1.8), "ramp18"))
    cand.append((make_end_bump(n, 0.8), "end08"))
    cand.append((make_end_bump(n, 1.4), "end14"))
    cand.append((make_center_bump(n, 0.8), "center08"))
    cand.append((make_gaussian_mixture(n, [0.25, 0.75], 40.0), "gauss2"))
    cand.append((make_gaussian_mixture(n, [0.15, 0.5, 0.85], 32.0), "gauss3"))
    cand.append((make_smooth_random(n, seed=17), "rand17"))
    cand.append((make_smooth_random(n, seed=88), "rand88"))

    if parent is not None:
        p0 = parent if len(parent) == n else interpolate_seq(parent, n)
        cand.append((blend_sequence(p0, make_power_ramp(n, 1.0), 0.35), "blend_ramp"))
        cand.append((blend_sequence(p0, make_smooth_random(n, seed=33), 0.40), "blend_rand"))

    return cand


def try_symmetric(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    s = normalize_pos(0.5 * (p + p[::-1]))
    return s, local_J(s)


def try_roll(p, shift):
    p = np.asarray(p, dtype=np.float64).ravel()
    r = np.roll(p, int(shift))
    return normalize_pos(r), local_J(r)


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    abs_limit = min(float(budget_s), 1800.0)
    time_limit = abs_limit - 90.0
    if time_limit < 120.0:
        time_limit = max(120.0, abs_limit - 30.0)

    parent = load_parent()
    rng = np.random.default_rng(20240527)

    if parent is not None:
        n_main = len(parent)
    else:
        n_main = 3200

    if n_main < 1500:
        n_main = 3200
    if n_main > 8000:
        n_main = 5120

    print(f"=== propose_candidate start: budget={abs_limit:.0f}s, main_n={n_main} ===")

    candidates = build_candidates(n_main, parent, rng)
    scored = []
    for p, lab in candidates:
        if len(p) != n_main:
            p = interpolate_seq(p, n_main)
        J = local_J(p)
        if not np.isfinite(J):
            J = 1e100
        scored.append((J, lab, p))
    scored.sort()

    if sorted(scored):
        print("Initial candidates:")
        for J, lab, _ in scored[:6]:
            print(f"   {lab}: J={J:.6f}")

    if parent is not None:
        primary = parent if len(parent) == n_main else interpolate_seq(parent, n_main)
        primary_label = "parent"
    else:
        primary = scored[0][2]
        primary_label = scored[0][1]

    best_p = primary.copy()
    best_J = local_J(best_p)

    budget1 = time_limit * 0.50
    if budget1 < 90.0:
        budget1 = time_limit * 0.70

    print(f"\nPhase 1: primary {primary_label}, budget={budget1:.0f}s")
    p_out, J_out = optimize_sequence(best_p.copy(), budget1, label=primary_label,
                                     verbose=True, use_lbfgs=True)
    if J_out < best_J:
        best_p, best_J = p_out, J_out

    # If an untouched initializer was already clearly better, keep it.
    for J_init, lab, p in scored:
        if J_init < best_J:
            best_p = p.copy()
            best_J = J_init
            print(f"  Better initializer found: {lab}, J={best_J:.6f}")

    used = time.monotonic() - t_global
    remaining = time_limit - used

    # Alternate initializers, especially if there is enough time.
    alt_count = 0
    print(f"\nPhase 2: alternate initializers, remaining={remaining:.0f}s")
    for J_init, lab, p in scored:
        if remaining < 130.0:
            break
        if lab == primary_label:
            continue
        if parent is not None and lab.startswith("parent"):
            continue
        budget_alt = min(remaining * 0.28, 250.0)
        if budget_alt < 70.0:
            break
        print(f"    alternate {lab} at n={n_main}, budget={budget_alt:.0f}s")
        p_out, J_out = optimize_sequence(p.copy(), budget_alt, label=lab,
                                         verbose=False,
                                         use_lbfgs=(n_main <= 4500 and budget_alt > 80.0))
        if J_out < best_J:
            best_p, best_J = p_out, J_out
            print(f"    NEW BEST from {lab}: J={best_J:.6f}")
        alt_count += 1
        if alt_count >= 2:
            break
        used = time.monotonic() - t_global
        remaining = time_limit - used

    # Length sweep around the current best length.
    print(f"\nPhase 3: length sweep, best_n={len(best_p)}, J={best_J:.6f}, remaining={remaining:.0f}s")
    tried_lengths = {len(best_p)}

    if len(best_p) <= 4500:
        trials = [4096, 5120, 6144, 7168]
    elif len(best_p) <= 6500:
        trials = [5120, 7168, 4096, 6144]
    else:
        trials = [6144, 5120, 4096]

    for nl in trials:
        if remaining < 150.0:
            break
        if nl in tried_lengths or not (1500 <= nl <= 7168):
            continue
        tried_lengths.add(nl)
        p_trial = interpolate_seq(best_p, nl)
        J_trial = local_J(p_trial)
        if J_trial < best_J:
            best_p = p_trial.copy()
            best_J = J_trial
            print(f"  Interpolation to n={nl} improved immediately: J={best_J:.6f}")
        budget_trial = min(remaining * 0.35, 320.0)
        if budget_trial < 80.0:
            budget_trial = remaining * 0.55
        if budget_trial < 60.0:
            break
        print(f"    length trial n={nl}, initial J={J_trial:.6f}, budget={budget_trial:.0f}s")
        p_out, J_out = optimize_sequence(p_trial, budget_trial, label=f"len{nl}",
                                         verbose=False,
                                         use_lbfgs=(nl <= 5000 and budget_trial > 100.0))
        if J_out < best_J:
            best_p, best_J = p_out, J_out
            print(f"    NEW BEST from length n={nl}: J={best_J:.6f}")
        used = time.monotonic() - t_global
        remaining = time_limit - used

    # Cheap global transformations.
    print(f"\nPhase 4: transformations, remaining={remaining:.0f}s")
    for p_try, J_try in [try_symmetric(best_p)]:
        if J_try < best_J:
            best_p, best_J = p_try, J_try
            print(f"  Symmetric transform improved: J={best_J:.6f}")

    for s in [-8, -4, -2, -1, 1, 2, 4, 8]:
        if abs(s) >= max(1, len(best_p) // 2):
            continue
        p_try, J_try = try_roll(best_p, s)
        if J_try < best_J:
            best_p, best_J = p_try, J_try
            print(f"  Roll {s} improved: J={best_J:.6f}")

    # Final polish.
    print(f"\nPhase 5: final polish, remaining={remaining:.0f}s")
    used = time.monotonic() - t_global
    remaining = time_limit - used
    if remaining > 60.0:
        cd_budget = min(remaining * 0.55, 350.0)
        p_out, J_out = coordinate_descent_fast(best_p.copy(), cd_budget,
                                               verbose=True)
        if J_out < best_J:
            best_p, best_J = p_out, J_out

    used = time.monotonic() - t_global
    remaining = time_limit - used
    if remaining > 40.0:
        p_out, J_out = eg_optimize(best_p.copy(), remaining * 0.70,
                                   schedule=EG_FINAL, verbose=False)
        if J_out < best_J:
            best_p, best_J = p_out, J_out

    used = time.monotonic() - t_global
    remaining = time_limit - used
    if remaining > 25.0:
        p_out, J_out = coordinate_descent_fast(best_p.copy(), remaining - 5.0,
                                               verbose=False)
        if J_out < best_J:
            best_p, best_J = p_out, J_out

    # Final sanity and output.
    best_p = normalize_pos(best_p)
    final_J = local_J(best_p)
    if final_J < best_J:
        best_J = final_J

    print(f"\nFINAL: J_local={best_J:.6f}, n={len(best_p)}, total_time={time.monotonic()-t_global:.0f}s")

    if not np.all(np.isfinite(best_p)) or len(best_p) == 0:
        fallback = make_power_ramp(3200, 1.0)
        return [float(x) for x in fallback]

    return [float(max(0.0, x)) for x in best_p]
