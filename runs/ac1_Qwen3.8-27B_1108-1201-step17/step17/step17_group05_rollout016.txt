def fast_autocorr(a):
    """Compute conv(a, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def spectral_flatten(a, n_iters=200, alpha=0.3, verbose=False):
    """
    DFT-based spectral flattening: iteratively compress the power spectrum
    to reduce the peak of the autocorrelation.
    """
    n = len(a)
    a = norm_sum1(a).copy()
    N = 1 << (2 * n - 1).bit_length()
    a_padded = np.zeros(N)
    a_padded[:n] = a
    best_score = evaluate_safe(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        A = np.fft.rfft(a_padded)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        
        # Compress: reduce values above mean
        excess = np.maximum(P - P_mean, 0)
        P_new = P - alpha * excess
        P_new = np.maximum(P_new, 0)
        
        # Also try: P_new = P ** (1 - alpha_iters)
        # P_new = P ** (1 - alpha * (it+1) / n_iters)
        
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Update
        score = evaluate_safe(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a_padded[:n] = a_new
        a = a_new
        
        if verbose and it % 50 == 0:
            print(f"    SpecFlatten iter={it}: score={best_score:.6f}")
    
    return best_a, best_score

    # grad_a[i] = 2 * sum_{j=max(0,-i)}^{min(n-1, n-1-i)} p[i+j]*a[j]
    # Hmm, this is getting complicated. Let me just use the direct formula with FFT.
    
    # Alternative: grad_a = 2 * ifft(fft(p) * fft(a_reversed))... 
    # Actually, the simplest is:
    # grad_a[i] = 2 * sum_k p[k] * a[(k-i) mod ???]
    
    # Let me think differently. Define the "cross-convolution" c[i] = sum_k p[k]*a[k-i] where
    # we interpret a[j] = 0 for j outside [0, n-1].
    # This is (p * a)[i] in the convolution sense, but with specific index ranges.
    # c[i] = sum_{k} p[k] * a[k-i] where we need 0 <= k-i < n, i.e., i <= k < i+n
    # So c[i] = sum_{k=i}^{i+n-1} p[k] * a[k-i] (with p[k]=0 outside [0, 2n-2])
    # This is the "sliding dot product" of p with a, which is the cross-correlation.
    # c = np.correlate(p, a, mode='full')[n-1: 2*n-1]... 
    # Actually, let me just compute it directly using FFT.
    
    N_len = 1 << (2 * n - 1).bit_length()
    p_padded = np.zeros(N_len)
    p_padded[:len(p)] = p
    a_padded = np.zeros(N_len)
    a_padded[:n] = a
    # What we want: c[i] = sum_k p[k] * a[k-i] for i = 0..n-1
    # This is the cross-correlation of p and a, evaluated at specific lags.
    # cross_corr[p, a][i] = sum_k p[k+i] * a[k] (for the "same" alignment)
    # Hmm, let me just use np.correlate for clarity (might be slow for large n, but let's try)
    
    # Actually for large n, let me use FFT:
    # c[i] = sum_k p[k] * a[k-i]
    # = sum_j p[j+i] * a[j]  (substituting k = j+i)
    # This is the cross-correlation of p and a at lag i.
    # Using FFT: cross_corr = ifft(fft(p) * conj(fft(a)))
    # But we need to be careful about the definition.
    
    # Let's use: c = np.fft.irfft(np.fft.rfft(p_padded) * np.fft.rfft(a_padded)[::-1], N_len)
    # No, that's not right either.
    
    # Simplest correct approach:
    # c[i] = sum_{k} p[k] * a[k-i] where a[j]=0 for j<0 or j>=n
    # = sum_{j=0}^{n-1} p[j+i] * a[j] where p[m]=0 for m<0 or m>=2n-1
    # This is the correlation of p and a at lag i.
    
    # Using the FFT-based correlation formula:
    # r[i] = sum_j p[j+i] * a[j] = ifft(fft(p) * conj(fft(a)))[i] (circular)
    # For linear correlation, we need N_len >= len(p) + n - 1 = (2n-1) + n - 1 = 3n-2
    
    N_corr = 1 << (3 * n - 1).bit_length()
    p_corr = np.zeros(N_corr)
    p_corr[:len(p)] = p
    a_corr = np.zeros(N_corr)
    a_corr[:n] = a
    # Linear cross-correlation: r[i] = sum_j p[j+i]*a[j] for i=0..2n-2
    # In terms of convolution: r = ifft(fft(p) * fft(a_flipped))
    # where a_flipped is a reversed and shifted...
    
    # Actually, the standard formula:
    # corr(p, a)[i] = sum_j p[j+i] * conj(a[j]) = ifft(fft(p) * conj(fft(a)))[i] (circular)
    # For linear: pad to length >= len(p) + len(a) - 1
    
    # Let me just do it the simple way for correctness:
    # grad_a[i] = 2 * sum_k p[k] * a[k-i] * 1[0 <= k-i < n]
    
    # Use FFT: this is 2 * (p cross-correlated with a) at lags 0..n-1
    # p has length 2n-1, a has length n
    # We want: for each i in 0..n-1: sum_{k=0}^{2n-2} p[k] * a[k-i] * 1[0<=k-i<n]
    # = sum_{j=0}^{n-1} p[j+i] * a[j]
    
    N_fft = 1 << (3 * n).bit_length()
    P_fft = np.fft.rfft(p, N_fft)
    A_fft = np.fft.rfft(a, N_fft)
    # Cross-correlation: ifft(P_fft * conj(A_fft))
    # But numpy's rfft doesn't work well with this. Let me use full FFT.
    
    # Actually, let me just use a different approach. The gradient of the objective
    # with respect to a is: grad_a = 2 * T^T @ p where T is the "Toeplitz" matrix
    # T[k, j] = 1[j + (k-j) = k]... this is getting circular.
    
    # Let me just compute it directly using the fact that:
    # grad_a[i] = 2 * (p * a_rev)[something]
    
    # You know what, let me just do the O(n^2) computation for now and optimize later if needed.
    # For n=3200, O(n^2) = 10M operations per gradient, which is feasible.
    
    # Actually no, let me think about this more carefully.
    # 
    # ac[k] = sum_j a[j] * a[k-j] (valid j)
    # 
    # The Jacobian J[k,i] = dac[k]/da[i] = 2*a[k-i] if 0<=k-i<n else 0
    # 
    # So J is a (2n-1) x n matrix where J[k,i] = 2*a[k-i] if 0<=k-i<n
    # 
    # grad_a = J^T @ p (where p is the softmax weight vector of length 2n-1)
    # 
    # grad_a[i] = sum_k J[k,i] * p[k] = 2 * sum_k a[k-i]*1[0<=k-i<n]*p[k]
    #           = 2 * sum_{j=0}^{n-1} a[j] * p[j+i] (substituting j=k-i)
    #
    # So grad_a[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]
    #
    # This is exactly 2 * the cross-correlation of a and p, evaluated at lags 0 to n-1.
    # 
    # Using FFT: cross_corr[a, p][i] = sum_j a[j] * p[i+j]
    # = sum_j a[j] * conjugate(conj(p))[i+j]... 
    # 
    # The standard formula: (a * p)[i] = sum_j a[j] * p[i-j] (convolution)
    # We want: sum_j a[j] * p[i+j] = sum_j a[j] * p'[-j+i] where p'[k] = p[-k] = p_reversed
    # = (a * p_reversed_shifted)[i]
    #
    # Let me define q[m] = p[2n-2-m] (reverse of p, length 2n-1)
    # Then p[i+j] = q[2n-2-i-j]
    # sum_j a[j]*p[i+j] = sum_j a[j]*q[2n-2-i-j]
    # Let k = 2n-2-i: = sum_j a[j]*q[k-j] = (a * q)[k] = (a * q)[2n-2-i]
    #
    # So grad_a[i] = 2 * (conv(a, q))[2n-2-i] where q = p[::-1]
    # = 2 * conv(a, p[::-1])[2n-2-i]
    
    q = p[::-1].copy()
    # Compute conv(a, q) using FFT
    N_cv = len(a) + len(q) - 1
    N_fft2 = 1 << (N_cv - 1).bit_length()
    A_fft2 = np.fft.rfft(a, N_fft2)
    Q_fft2 = np.fft.rfft(q, N_fft2)
    conv_aq = np.fft.irfft(A_fft2 * Q_fft2, N_fft2)[:N_cv]
    # grad_a[i] = 2 * conv_aq[2n-2-i]
    grad_a = 2.0 * conv_aq[2*n-2 : n-1 : -1]
    # Let me verify: conv_aq has length n + (2n-1) - 1 = 3n-2
    # We want indices 2n-2, 2n-3, ..., n-1 (i.e., from 2n-2 down to n-1, that's n values)
    # conv_aq[2n-2-i] for i=0..n-1 gives indices 2n-2, 2n-3, ..., n-1
    # So grad_a[i] = 2 * conv_aq[2*n-2-i]
    
    # Now chain rule: grad_x = J_a^T @ grad_a where J_a[i,j] = da[i]/dx[j] = a[i]*(delta[i,j] - a[j])
    # grad_x[j] = sum_i grad_a[i] * a[i]*(delta[i,j] - a[j])
    #           = grad_a[j]*a[j] - a[j]*sum_i grad_a[i]*a[i]
    
    grad_x = grad_a * a - a * np.dot(grad_a, a)
    
    return logsumexp_val, grad_x


def lbfgs_optimize_smoothed(a0, time_limit, beta=10.0, verbose=True):
    """
    Optimize using L-BFGS-B on the smoothed objective with softmax parameterization.
    """
    n = len(a0)
    a0 = norm_sum1(a0)
    
    # Initialize x such that softmax(x) ≈ a0
    # x = log(a0) works if a0 > 0
    x0 = np.log(np.maximum(a0, 1e-30))
    
    t0 = time.monotonic()
    best_x = x0.copy()
    best_score = evaluate_safe(a0)
    
    def func_and_grad(x):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            # Return cached best to signal stopping
            return best_score, np.zeros_like(x)
        val, grad = smoothed_objective_and_grad(x, beta=beta)
        return val, grad
    
    try:
        result = minimize(
            func_and_grad, x0, method='L-BFGS-B', jac=True,
            options={'maxiter': 500, 'maxfun': 2000, 'ftol': 1e-10, 'gtol': 1e-8}
        )
    except Exception as e:
        print(f"    L-BFGS failed: {e}")
        return a0, best_score
    
    # Evaluate result
    x_final = result.x
    x_shifted = x_final - x_final.max()
    a_final = np.exp(x_shifted)
    a_final = a_final / a_final.sum()
    a_final = np.maximum(a_final, 0.0)
    
    score = evaluate_safe(a_final)
    if score < best_score:
        best_score = score
    
    # Also check the best during optimization
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    L-BFGS done ({elapsed:.1f}s): score={best_score:.6f}")
    
    return a_final, best_score


def mirror_descent_opt(a0, time_limit, beta_init=8.0, beta_min=3.0, c0=5.0, 
                       restart_every=5000, seed=42, verbose=False):
    """Mirror descent with full subgradient."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac = fast_autocorr(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()
    x_prev = a.copy()
    cur_beta = beta_init
    stall_count = 0
    iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        beta_t = (iterations - 1.0) / (iterations + 2.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute subgradient (full)
        ac_y = fast_autocorr(y)
        shifted = ac_y - ac_y.max()
        w = np.exp(cur_beta * shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(ac_y) / len(ac_y)
        else:
            w = w / w_sum
        
        # Gradient: 2 * conv(a, w_reversed)[2n-2-i]
        q = w[::-1]
        N_cv = n + len(w) - 1
        N_fft = 1 << (N_cv - 1).bit_length()
        Y_fft = np.fft.rfft(y, N_fft)
        Q_fft = np.fft.rfft(q, N_fft)
        conv_yq = np.fft.irfft(Y_fft * Q_fft, N_fft)[:N_cv]
        g = 2.0 * conv_yq[2*n-2 : n-1 : -1]
        
        # Mirror descent step
        eta = c0 / (np.sqrt(iterations) + 1.0)
        log_y = np.log(np.maximum(y, 1e-30))
        log_a_new = log_y - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        x_prev = x.copy()
        x = a_new
        
        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-12:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        cur_beta = max(beta_min, beta_init - 0.002 * iterations)
        
        if stall_count > restart_every:
            stall_count = 0
            pert = rng.normal(0, 0.03, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            if rng.random() < 0.3:
                x_r = rng.dirichlet(np.ones(n) * 0.5)
                x = x_r
                x_prev = x_r.copy()
    
    return best_a, best_m


def lp_frank_wolfe(a0, time_limit, solver, verbose=False, top_K=300):
    """LP-based Frank-Wolfe with top-K constraints."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break
        
        # Select top-K most binding constraints
        K_sel = min(top_K, 2 * n - 1)
        top_idx = np.argsort(ac)[::-1][:K_sel]
        T = np.zeros((K_sel, n))
        for col_idx, m in enumerate(top_idx):
            m_int = int(m)
            j_start = max(0, m_int - n + 1)
            j_end = min(n - 1, m_int)
            js = np.arange(j_start, j_end + 1)
            T[col_idx, js] = y[m_int - js]
        
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M_curr * np.ones(K_sel),
                        cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        y_new = y_new / max(sy, 1e-15)
        
        if t == 1:
            x_new = g_dir.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            x_new = x_new / max(sx, 1e-15)
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m
