"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed mirror descent (Nesterov-accelerated)
2. L-BFGS-B on softmax parameterization
3. LP-based Frank-Wolfe with adaptive cutting planes
4. Randomized coordinate descent
5. Multi-length exploration (5200-7000)
"""

import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate as sig_correlate

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def compute_ls_gradient(a, p, n):
    """Compute gradient of (1/beta)*log(sum exp(beta*conv_k)) w.r.t. a."""
    # s[i] = sum_j p[i+j] * a[j]
    # Using: s = correlate(p, a, 'full')[:2n-1]
    s = sig_correlate(p, a, mode='full')[:2*n-1]
    return 2.0 * s

def mirror_descent_ls(a0, time_limit, beta_start=20.0, beta_end=500.0, 
                      c0=5.0, verbose=True):
    """Nesterov-accelerated mirror descent with log-sum-exp smoothing."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    x = a.copy()
    x_prev = a.copy()
    
    total_iters = 50000
    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive beta
        beta = beta_start + (beta_end - beta_start) * t / total_iters
        
        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.0)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-30)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy
        
        # Compute objective gradient
        conv = fast_autocorr(y)
        # Softmax weights (numerically stable)
        beta_conv = beta * conv
        beta_conv -= beta_conv.max()
        p = np.exp(beta_conv)
        p_sum = p.sum()
        p = p / p_sum
        
        # Gradient
        grad = compute_ls_gradient(y, p, n)
        
        # Mirror descent step
        eta = c0 / (np.sqrt(t) + 1.0)
        log_update = np.log(y) - eta * grad
        log_update -= log_update.max()
        a_new = np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        x_prev = x
        x = a_new

        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD-LS iter={t} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.0f}")
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    MD-LS done ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def lbfgs_optimize(a0, time_limit, beta=200.0, max_iter=500, verbose=True):
    """L-BFGS-B optimization on softmax parameterization."""
    from scipy.optimize import minimize as sp_minimize
    n = len(a0)
    a0 = norm_sum1(a0)
    z0 = np.log(np.maximum(a0, 1e-20))
    
    best_z = z0.copy()
    best_m = fast_autocorr(a0).max()
    t0 = time.monotonic()
    
    def objective(z):
        a = softmax(z)
        conv = fast_autocorr(a)
        bc = beta * conv
        bc -= bc.max()
        return np.log(np.exp(bc).sum()) / beta
    
    def gradient(z):
        a = softmax(z)
        conv = fast_autocorr(a)
        bc = beta * conv
        bc -= bc.max()
        p = np.exp(bc)
        p = p / p.sum()
        grad_a = compute_ls_gradient(a, p, n)
        # Chain rule through softmax
        return a * (grad_a - np.dot(a, grad_a))
    
    n_calls = [0]
    def time_check():
        n_calls[0] += 1
        return time.monotonic() - t0 > time_limit * 0.85
    
    try:
        result = sp_minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'maxfun': 2000,
                                    'ftol': 1e-12, 'gtol': 1e-10})
    except Exception as e:
        print(f"    L-BFGS-B error: {e}")
        return best_a, best_m
    
    a_result = softmax(result.x)
    M_result = fast_autocorr(a_result).max()
    if M_result < best_m:
        best_m = M_result
        best_z = result.x.copy()
    
    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    L-BFGS-B done ({elapsed:.0f}s): J={J:.6f}, iters={result.nit}")
    return norm_sum1(softmax(best_z)), best_m
