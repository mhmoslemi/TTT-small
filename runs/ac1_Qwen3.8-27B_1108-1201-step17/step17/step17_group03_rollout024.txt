def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n + 1)  # include endpoint
    new_x = np.linspace(0.0, 1.0, new_n + 1)
    f_pad = np.concatenate([f_norm, [0.0]])
    f_interp = np.interp(new_x, old_x, f_pad)[:-1]  # remove endpoint
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def spectral_flatten(a, n_iter=200, alpha=0.1, gamma=0.5):
    """
    Iteratively flatten the power spectrum |FFT(a)|^2 to reduce autocorrelation peaks.
    Uses iterative projection: shape spectrum, inverse FFT, project to non-negative.
    """
    n = len(a)
    a = norm_sum1(a.copy())
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        P = mag ** 2  # power spectrum
        
        # Compute current autocorrelation from P
        c = np.fft.irfft(P, fft_size)[:2*n-1]
        M = c.max()
        
        if M < 1e-15:
            break
        
        # Flatten: reduce peaks in autocorrelation by adjusting spectrum
        # For lags where c_k is high, reduce the corresponding spectral contributions
        # Simple approach: scale P by a factor that's < 1 where it contributes to peaks
        
        # More direct: compute the "excess" at each lag
        # c_k = sum_i a_i a_{i+k}, so |A|^2 in freq domain gives the autocorrelation
        # We want to make the inverse FFT of |A|^2 as flat as possible
        
        # Use a soft thresholding approach on the autocorrelation
        target = M * (1.0 - alpha)
        excess = np.maximum(0.0, c - target)
        
        # Convert excess back to frequency domain to get correction
        correction = np.fft.rfft(excess, fft_size)
        
        # Apply correction to power spectrum
        P_new = P - alpha * np.abs(correction) ** 2 / (fft_size * M + 1e-30)
        P_new = np.maximum(P_new, 1e-30 * P.max())
        
        # New magnitude
        mag_new = np.sqrt(P_new)
        
        # Reconstruct: new_a = ifft(mag_new * exp(i*phase))
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Blend with previous (for stability)
        a = gamma * a + (1 - gamma) * a_new
        a = norm_sum1(a)
    
    return a

def lp_frank_wolfe_full_ls(a0, time_limit, solver, verbose=True, top_K=200,

    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()

    t0 = time.monotonic()

    for t in range(1, 500000):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral flattening pre-conditioning (frequency-domain projection)
2. LP Frank-Wolfe with full line search and adaptive active set (main workhorse)
3. Nesterov mirror descent with backtracking (fast refinement)
4. Structured peak-crushing coordinate descent
5. Multi-length exploration with spectral interpolation
Key innovations over previous: full bisection line search in FW, spectral
pre-conditioning, dual-weighted active set selection, and better restart strategy.
"""

import numpy as np
import time
import cvxpy as cp

_global_solver = None


def get_solver():

        return _global_solver

            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue

    print("  LP solver (fallback): ECOS")

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_flatten(a, n_iter=150, alpha=0.08, blend=0.6):
    """
    Iteratively flatten the power spectrum |FFT(a)|^2 to reduce autocorrelation peaks.
    Uses: shape spectrum in freq domain -> inverse FFT -> project non-negative -> blend.
    """
    n = len(a)
    a = norm_sum1(a.copy())
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        P = mag ** 2

        # Compute current autocorrelation
        c = np.fft.irfft(P, fft_size)[:2 * n - 1]
        M = c.max()
        if M < 1e-15:
            break

        # Compute excess peaks
        target = M * (1.0 - alpha)
        excess = np.maximum(0.0, c - target)

        # Convert excess to frequency domain correction
        corr_freq = np.fft.rfft(excess, fft_size)
        corr_mag = np.abs(corr_freq)

        # Apply correction: reduce spectral power where it contributes to peaks
        P_new = P - 0.5 * corr_mag * (corr_mag / (P + 1e-30)) * 0.1
        P_new = np.maximum(P_new, 1e-10 * P.max())

        mag_new = np.sqrt(P_new)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        a = blend * a + (1.0 - blend) * a_new
        a = norm_sum1(a)

    return a


def spectral_flatten_v2(a, n_iter=80, flatness=0.15):
    """
    Direct spectral flattening: interpolate |FFT(a)|^2 towards uniform,
    then reconstruct with original phase and project to non-negative.
    """
    n = len(a)
    a = norm_sum1(a.copy())
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        P = mag ** 2
        P_mean = P.mean()

        # Blend towards flat spectrum
        P_new = (1.0 - flatness) * P + flatness * P_mean
        mag_new = np.sqrt(P_new)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

    return a


def lp_frank_wolfe(a0, time_limit, solver, verbose=True, top_K=200,
                   margin_frac=2e-3, ls_steps=20, ls_refine=10):
    """
    LP Frank-Wolfe with full line search and adaptive active set.
    Key improvements:
    - Full bisection line search for optimal step size
    - Adaptive active set that grows and tightens over time
    - Periodic parameter resets to escape stagnation
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    cur_margin = margin_frac
    cur_top_K = top_K

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-16:
            break

        # Adaptive active set: top peaks + margin-based
        top_idx = np.argsort(ac)[::-1][:cur_top_K]
        margins = M_curr - ac
        threshold = cur_margin * M_curr
        active_mask = margins <= threshold

        active_idx = np.unique(np.concatenate([top_idx, np.where(active_mask)[0]]))

        # Cap active set size
        max_constraints = cur_top_K * 6
        if len(active_idx) > max_constraints:
            active_idx = np.argsort(ac)[::-1][:max_constraints]

        K = len(active_idx)

        # Build constraint matrix
        T_sel = np.zeros((K, n))
        for row, k_int in enumerate(active_idx):
            k_int = int(k_int)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            T_sel[row, idx] = y[j_idx]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T_sel @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b_val / b_sum

        # Full line search
        best_alpha = 0.0
        best_obj = M_curr

        # Coarse grid search
        alphas = np.linspace(0.0, 1.0, ls_steps)
        for alpha in alphas:
            y_trial = (1.0 - alpha) * y + alpha * g_dir
            y_trial = np.maximum(y_trial, 0.0)
            sy = y_trial.sum()
            if sy < 1e-15:
                continue
            y_trial = y_trial / sy
            m_trial = fast_autocorr(y_trial).max()
            if m_trial < best_obj:
                best_obj = m_trial
                best_alpha = alpha

        # Bisection refinement
        if 0.0 < best_alpha < 1.0:
            step_wide = 1.0 / ls_steps
            lo = max(0.0, best_alpha - step_wide)
            hi = min(1.0, best_alpha + step_wide)
            for _ in range(ls_refine):
                mid = (lo + hi) / 2.0
                y_trial = (1.0 - mid) * y + mid * g_dir
                y_trial = np.maximum(y_trial, 0.0)
                sy = y_trial.sum()
                if sy < 1e-15:
                    break
                y_trial = y_trial / sy
                m_trial = fast_autocorr(y_trial).max()
                if m_trial < best_obj:
                    best_obj = m_trial
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        # Update
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()
            # Tighten active set when improving
            cur_margin = max(cur_margin * 0.93, 1e-5)

        # Periodically expand the search
        if t % 50 == 0:
            cur_top_K = min(cur_top_K + 3, 400)

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.0f}s): J={J:.6f}, K={K}, alpha={best_alpha:.3f}, margin={cur_margin:.2e}")

    return best_a, best_m


def amd_optimize(a0, time_limit, top_k=50, beta=4.0, c0=3.5,
                 restart_every=3000, verbose=False, use_backtrack=True):
    """
    Nesterov mirror descent (Fenchel) with backtracking line search.
    Uses exponential parameterization for non-negativity.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    x = a.copy()
    x_prev = a.copy()
    gamma = c0
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)

        # Soft subgradient: weight top peaks by their value^beta
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        # Mirror descent step with backtracking
        if use_backtrack:
            eta = gamma / (np.sqrt(t) + 1.0)
            base_m = fast_autocorr(y).max()
            accepted = False
            for _ in range(8):
                log_a = np.log(y) - eta * grad
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    break
                a_new = a_new / s_new
                m_trial = fast_autocorr(a_new).max()
                if m_trial < base_m - 1e-18:
                    x_prev = x
                    x = a_new
                    accepted = True
                    break
                else:
                    eta *= 0.5
            if not accepted:
                log_a = np.log(y) - eta * grad
                max_log = log_a.max()
                a_new = np.exp(log_a - max_log)
                a_new = np.maximum(a_new, 1e-30)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    a_new = y.copy()
                else:
                    a_new = a_new / s_new
                x_prev = x
                x = a_new
        else:
            eta = gamma / (np.sqrt(t) + 1.0)
            log_a = np.log(y) - eta * grad
            max_log = log_a.max()
            a_new = np.exp(log_a - max_log)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new
            x_prev = x
            x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-15:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            # Spectral perturbation restart
            fft_size = 1 << (2 * n - 1).bit_length()
            A = np.fft.rfft(best_a, fft_size)
            phase = np.angle(A)
            mag = np.abs(A)
            phase_pert = phase + rng.normal(0, 0.02, len(phase))
            A_new = mag * np.exp(1j * phase_pert)
            a_new = np.fft.irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()
            x = a_new
            x_prev = a_new.copy()
            cur_top_k = min(cur_top_k + 5, 100)
            cur_beta = max(cur_beta - 0.3, 2.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    AMD restart t={t} ({elapsed:.0f}s): J={J:.6f}")

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    AMD t={t} ({elapsed:.0f}s): J={J:.6f}, stall={stall_count}")

    return best_a, best_m


def peak_crush(a0, time_limit, verbose=True, rng_seed=123):
    """
    Targeted peak-crushing: identify the largest autocorrelation peak,
    find the mass contributor, and relocate it to minimize the new max peak.
    Uses structured candidate selection (low-mass positions + random).
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0
    max_stall = 250

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M_curr = ac.max()

        # Process top peaks
        n_peaks = min(3, len(ac))
        top_peaks = np.argsort(ac)[::-1][:n_peaks]

        improved = False
        for peak_k in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break

            peak_k = int(peak_k)
            i_start = max(0, peak_k - n + 1)
            i_end = min(n, peak_k + 1)
            idx = np.arange(i_start, i_end)
            contrib = a[idx] * a[peak_k - idx]
            total_contrib = contrib.sum()

            if total_contrib < 1e-20:
                continue

            n_top = min(15, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top]

            for li in top_local:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[li])
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                best_new_m = M_curr
                best_move = None

                for frac in [0.003, 0.008, 0.015, 0.03, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue

                    # Build candidate destinations
                    candidates = np.unique(np.concatenate([
                        rng.integers(0, n, size=15),
                        np.where(a < np.percentile(a, 30))[0][:10]
                    ]))
                    candidates = candidates[candidates != src][:25]

                    for dst in candidates:
                        dst = int(dst)
                        a_trial = a.copy()
                        a_trial[src] = max(0.0, a_trial[src] - amt)
                        a_trial[dst] += amt
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_new_m:
                            best_new_m = M_trial
                            best_move = (src, dst, amt)

                if best_move is not None:
                    src_i, dst_i, amt = best_move
                    a[src_i] = max(0.0, a[src_i] - amt)
                    a[dst_i] += amt
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a[i] = max(0.0, a[i] - amt)
                a[j] += amt
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    PeakCrush ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m
        print(f"    PeakCrush final ({elapsed:.0f}s): J={J:.6f}")

    return best_a, best_m


def make_parametric(n, family='sine'):
    if family == 'sine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        return norm_sum1(np.cos(np.pi * x) ** 2 + 0.01)
    elif family == 'beta_2_2':
        x = np.linspace(0.01, 0.99, n)
        return norm_sum1(x ** 2 * (1 - x) ** 2)
    else:
        x = np.linspace(0, 1, n)
        return norm_sum1(np.sin(np.pi * x) ** 2)


def try_update(a, best_seq, best_score, t_global, label=""):
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return a_norm.copy(), sc
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = norm_sum1(parent)
        best_score = evaluate_safe(best_seq)
    else:
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Spectral pre-conditioning ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Spectral pre-conditioning (n={n0}) ===")
    t1 = min(remaining * 0.04, 60.0)
    if t1 > 15:
        a_spec = spectral_flatten(best_seq, n_iter=80, alpha=0.06, blend=0.65)
        best_seq, best_score = try_update(a_spec, best_seq, best_score, t_global, "spec1")

        a_spec2 = spectral_flatten_v2(best_seq, n_iter=40, flatness=0.12)
        best_seq, best_score = try_update(a_spec2, best_seq, best_score, t_global, "spec2")

    # === Phase 2: AMD warmup ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: AMD warmup (n={n_best}) ===")
    t2 = min(remaining * 0.08, 120.0)
    if t2 > 25:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t2, top_k=50, beta=4.0, c0=3.0,
                                restart_every=2500, verbose=False, use_backtrack=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd_warm")

    # === Phase 3: LP Frank-Wolfe main optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-FW main (n={n_best}) ===")
    t3 = min(remaining * 0.28, 450.0)
    if t3 > 40 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t3, solver, verbose=True, top_K=200,
                                   margin_frac=2e-3, ls_steps=20, ls_refine=10)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_main")

    # === Phase 4: Peak crushing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak-crush (n={n_best}) ===")
    t4 = min(remaining * 0.08, 120.0)
    if t4 > 15:
        a_in = norm_sum1(best_seq)
        a_out, _ = peak_crush(a_in, t4, verbose=True, rng_seed=42)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "peakcrush")

    # === Phase 5: LP with tighter margins ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-FW tight (n={n_best}) ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 30 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t5, solver, verbose=True, top_K=300,
                                   margin_frac=5e-4, ls_steps=25, ls_refine=12)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_tight")

    # === Phase 6: AMD refinement with adjusted params ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: AMD refine (n={n_best}) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 20:
        a_in = norm_sum1(best_seq)
        a_out, _ = amd_optimize(a_in, t6, top_k=80, beta=3.0, c0=2.5,
                                restart_every=2000, verbose=False, use_backtrack=True)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "amd_refine")

    # === Phase 7: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Length exploration (base n={n_best}) ===")

    candidate_lengths = []
    if n_best > 500:
        for delta in [300, -300, 600, -600]:
            cl = n_best + delta
            if 500 <= cl <= 5000:
                candidate_lengths.append(cl)

    for new_n in candidate_lengths[:2]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 120:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_sequence(best_seq, new_n)

        t7a = min(remaining * 0.12, 100.0)
        if t7a > 25:
            a_out, _ = amd_optimize(a_long, t7a, top_k=50, beta=4.0, c0=3.0,
                                     restart_every=2000, verbose=False, use_backtrack=True)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"amd_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 80 and new_n <= 4000:
            t7b = min(remaining * 0.08, 80.0)
            if t7b > 20:
                a_in = norm_sum1(best_seq)
                a_out, _ = lp_frank_wolfe(a_in, t7b, solver, verbose=False, top_K=150,
                                           margin_frac=2e-3, ls_steps=15, ls_refine=8)
                best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, f"lp_n{new_n}")

    # === Phase 8: Final LP on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP (n={n_best}) ===")
    t8 = min(remaining * 0.15, 180.0)
    if t8 > 25 and n_best <= 5000:
        a_in = norm_sum1(best_seq)
        a_out, _ = lp_frank_wolfe(a_in, t8, solver, verbose=True, top_K=350,
                                   margin_frac=2e-4, ls_steps=25, ls_refine=12)
        best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "lp_final")

    # === Phase 9: Final peak-crush ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final peak-crush (n={n_best}) ===")
    if remaining > 25:
        t9 = min(remaining * 0.25, 150.0)
        if t9 > 10:
            a_in = norm_sum1(best_seq)
            a_out, _ = peak_crush(a_in, t9, verbose=True, rng_seed=999)
            best_seq, best_score = try_update(a_out, best_seq, best_score, t_global, "pc_final")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3600
        best_seq = make_parametric(n_fb, 'sine')
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    remaining_time = budget_s - (time.monotonic() - t_global)
    if remaining_time < 30:
        print(f"WARNING: Only {remaining_time:.0f}s remaining, returning immediately")

    return [float(x) for x in best_seq]
