"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth subgradient mirror descent with exact FFT-based gradient
2. Nesterov acceleration with adaptive step size
3. Gerchberg-Saxton spectral flattening for global exploration

import numpy as np
import time

def fast_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    """Normalize a to have sum 1, element-wise non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def make_init_cosine(n, mode=1):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * mode * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def make_init_sinc(n, bw=5.0):
    x = np.linspace(0, 1, n)
    t = (x - 0.5) * bw
    a = np.sinc(t) ** 2
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def make_init_flat_spectrum(n, seed=0):
    """Initialize with flat magnitude spectrum, random phases."""
    rng = np.random.default_rng(seed)
    fft_size = n
    # Flat magnitude with random phase (conjugate symmetric for real signal)
    nfft = fft_size // 2 + 1
    mag = np.ones(nfft)
    phase = np.zeros(nfft)
    phase[1:-1] = rng.uniform(-np.pi, np.pi, nfft - 2)
    # Handle conjugate symmetry: A[0] and A[nfft-1] (if even) should be real
    A = mag * np.exp(1j * phase)
    A[0] = mag[0]  # Real
    if fft_size % 2 == 0:
        A[-1] = mag[-1]  # Real (Nyquist)
    a = np.fft.irfft(A, fft_size)
    a = np.maximum(a, 0.0)
    return norm_sum1(a)

def smooth_subgrad_md(a0, time_limit, tau0=20.0, tau1=200.0, lr=1.0, 
                       restart_every=2000, momentum=True, seed=42, verbose=False):
    """
    Smooth subgradient mirror descent with Nesterov acceleration.
    
    Minimizes f(a) = max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    Uses smooth approximation: f_tau(a) = (1/tau) * log(sum exp(tau * conv[k]))
    Gradient computed exactly via np.correlate.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    x = a.copy()  # Momentum variable
    x_prev = a.copy()
    
    iter_count = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        progress = min(1.0, iter_count / 10000.0)
        
        # Adaptive temperature
        tau = tau0 + (tau1 - tau0) * progress
        # Learning rate decay
        eta = lr / (1.0 + 0.05 * np.sqrt(iter_count))
        
        # Nesterov step
        if momentum:
            beta_t = (iter_count - 1.0) / (iter_count + 1.5)
            y = x + beta_t * (x - x_prev)
            y = np.maximum(y, 1e-15)
            sy = y.sum()
            if sy < 1e-15:
                y = x.copy()
            else:
                y = y / sy
        else:
            y = a
        
        # Compute convolution
        conv = fast_conv(y)
        
        # Softmax weights (shifted for numerical stability)
        conv_shifted = conv - conv.max()
        exp_conv = np.exp(tau * conv_shifted)
        w = exp_conv / exp_conv.sum()
        
        # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
        # = 2 * np.correlate(w, y, mode='valid')
        grad = 2.0 * np.correlate(w, y, mode='valid')
        
        # Ensure gradient is non-negative (it should be since w, y >= 0)
        grad = np.maximum(grad, 0.0)
        
        # Mirror descent update (in log space)
        log_y = np.log(y + 1e-30)
        a_new = np.exp(log_y - eta * grad)
        a_new = a_new / a_new.sum()
        
        # Update momentum
        x_prev = x
        x = a_new
        
        # Evaluate
        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Restart if stalled
        if stall_count >= restart_every:
            stall_count = 0
            # Perturb from best
            noise_scale = 0.05 + 0.05 * min(iter_count / 10000.0, 1.0)
            pert = rng.normal(0, noise_scale, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            a = x_r.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart at iter {iter_count} ({elapsed:.0f}s): J={J:.6f}")

            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD iter {iter_count} ({elapsed:.0f}s): J={J:.6f}, tau={tau:.1f}, stall={stall_count}")
    
    return best_a, best_m

def gerchberg_saxton(a0, time_limit, blend=0.3, iters=None, seed=42, verbose=False):
    """
    Gerchberg-Saxton / Fienup error reduction for flat spectrum.
    Alternates between:
    - Time domain: enforce non-negativity and normalization
    - Frequency domain: flatten magnitude
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Add small random phase perturbation to escape symmetric solutions
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    phase[1:] += rng.normal(0, 0.1, len(phase) - 1)
    mag = np.abs(A)
    A = mag * np.exp(1j * phase)
    a = np.fft.irfft(A, fft_size)[:n]
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Frequency domain: flatten magnitude
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target magnitude: geometric mean of current and flat
        target_mag = np.full_like(mag, mag.mean())
        new_mag = (1 - blend) * mag + blend * target_mag
        new_mag = np.maximum(new_mag, 1e-10)
        
        A_new = new_mag * np.exp(1j * phase)
        
        # Time domain: enforce non-negativity
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Evaluate
        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GS iter {iter_count} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def gerchberg_saxton_phase(a0, time_limit, seed=42, verbose=False):
    """
    GS with phase refinement: after getting a good magnitude, 
    optimize phases to maximize non-negativity.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    fft_size = 1 << (2 * n - 1).bit_length()
    nfft = fft_size // 2 + 1
    
    # Initialize with flat magnitude, optimized phases
    A = np.fft.rfft(a, fft_size)
    target_mag = np.abs(A).mean()
    
    # Start with current phases
    phase = np.angle(A).copy()
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Update: a = IFFT(target_mag * exp(i*phase))
        A = target_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A, fft_size)[:n]
        
        # Soft non-negativity: use softplus-like clipping
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            # Reset phases
            phase = rng.uniform(-np.pi, np.pi, nfft)
            phase[0] = 0
            continue
        a_new = a_new / s
        
        # Update phases from new a
        A_new = np.fft.rfft(a_new, fft_size)
        phase_new = np.angle(A_new)
        
        # Blend phases (to avoid oscillation)
        blend = 0.5
        phase = (1 - blend) * phase + blend * phase_new
        # Unwrap to avoid discontinuities
        # (simpler: just use the new phases)
        phase = phase_new
        
        # Evaluate
        m_new = fast_conv(a_new).max()
        if m_new < best_m - 1e-15:
            best_m = m_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    GS-phase iter {iter_count} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_refine(a0, time_limit, n_perturb=2, rng_seed=42, verbose=False):
    """
    Simple coordinate perturbation: randomly pick 2 indices, 
    transfer mass between them, accept if improvement.
    """
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        # Pick source (high value) and destination (low value)
        src = int(rng.integers(n))
        dst = int(rng.integers(n))
        if src == dst:
            continue
        
        # Try several transfer fractions
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            m_trial = fast_conv(a_trial).max()
            if m_trial < best_m - 1e-16:
                a = a_trial
                best_m = m_trial
                best_a = a_trial.copy()
                break
        
        if verbose and iter_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord iter {iter_count} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
