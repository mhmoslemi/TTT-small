    ac = fast_autocorr(f)
    M = ac.max()
    
    # Find binding constraints
    margins = M - ac
    binding = np.argsort(margins)[:K]
    
    # Build constraint matrix for binding constraints only
    T_bind = np.zeros((K, n))
    for i, k in enumerate(binding):
        k = int(k)
        for j in range(max(0, k-n+1), min(n, k+1)):
            T_bind[i, j] = f[k-j]
    
    # Solve LP: max sum(b) s.t. T_bind @ b <= M, b >= 0
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_bind @ b_var <= M * np.ones(K)])
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return f, M
    
    b = np.maximum(np.asarray(b_var.value), 0)

    if b_sum < 1e-15:

    x_prev = a.copy()

        if time.monotonic() - t_start > time_limit:

            beta_t = (t - 1.0) / (t + 1.0)

        x = a_new

            j_end = min(n, k + 1)
            if j_start < j_end:
                T_mat[i, j_start:j_end] = y[k - j_start:k - j_start + (j_end - j_start)]

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(
            cp.Maximize(cp.sum(b_var)),
            [T_mat @ b_var <= M_curr * np.ones(K_sel)]
        )
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b / b_sum

        # Line search: try several alpha values
        alphas = [2.0 / (t + 2.0), 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        best_alpha = alphas[0]
        best_trial = y.copy()
        best_trial_M = M_curr

        for alpha in alphas:
            trial = (1.0 - alpha) * y + alpha * g_dir
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            m = peak_val(trial)
            if m < best_trial_M - 1e-15:
                best_trial_M = m
                best_trial = trial.copy()
                best_alpha = alpha

        # Nesterov momentum
        if t == 1:
            x_new = best_trial.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = best_trial + beta_t * (best_trial - y)
            x_new = np.maximum(x_new, 0.0)
            xs = x_new.sum()
            if xs < 1e-15:
                x_new = best_trial.copy()
            else:
                x_new = x_new / xs

        y = best_trial
        x = x_new

        # Track best
        m_y = best_trial_M
        m_x = peak_val(x)
        if m_y < best_M:
            best_M = m_y
            best_f = y.copy()
        if m_x < best_M:
            best_M = m_x
            best_f = x.copy()

        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t_start
            score = 2.0 * n * best_M
            print(f"    FW t={t} ({elapsed:.0f}s): score={score:.6f}")

    return best_f, best_M


def perturbation_refine(a0, time_limit, solver, n_perturbs=5, verbose=False, seed=777):
    """Try multiple perturbations of a0 and refine each with short FW runs."""
    n = len(a0)
    a = norm1(a0)
    best_a = a.copy()
    best_M = peak_val(a)
    t_start = time.monotonic()
    rng = np.random.default_rng(seed)

    for p in range(n_perturbs):
        if time.monotonic() - t_start > time_limit - 10:
            break
        remaining = time_limit - (time.monotonic() - t_start)
        if remaining < 20:
            break
        t_sub = remaining / (n_perturbs - p) * 0.8

        # Perturb
        sigma = 0.02 + 0.01 * p
        pert = rng.normal(0, sigma, n) * a
        a_pert = a + pert
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert / s

        # Short FW refinement
        a_out, m_out = fw_with_linsearch(a_pert, t_sub, solver, verbose=False, K=300, max_iters=5000)
        if m_out < best_M:
            best_M = m_out
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t_start
                print(f"    Pert {p}: score={2.0 * n * best_M:.6f} ({elapsed:.0f}s)")

    return best_a, best_M


def init_cos2(n):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2 + 0.005
    return norm1(a)


def init_power_law(n, power=0.7):
    x = np.linspace(-1, 1, n)
    a = (1 - x ** 2) ** power + 0.002
    return norm1(a)


def init_gaussian(n, sigma=0.12):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.05) / sigma) ** 2) + np.exp(-((x - 0.95) / sigma) ** 2) + 0.003
    return norm1(a)


def init_two_bump(n, pos1=0.3, pos2=0.7, width=0.08):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - pos1) / width) ** 2) + np.exp(-((x - pos2) / width) ** 2) + 0.002
    return norm1(a)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_i = np.interp(new_x, old_x, f_n)
    f_i = np.maximum(f_i, 0.0)
    ss = f_i.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_i / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1760.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_score = float('inf')
    best_seq = None

    def try_update(a, label):
        nonlocal best_score, best_seq
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_n = a / s
        sc = evaluate_safe(a_n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_n.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_n)}, t={elapsed:.0f}s)")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = norm1(parent)
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 3200
        best_seq = init_cos2(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")

    # === Phase 1: LSE Mirror Descent from parent and other inits ===
    print(f"\n=== Phase 1: LSE Mirror Descent ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 420.0)

    if t1 > 30:
        inits = []
        labels = []
        if parent is not None:
            inits.append(norm1(parent))
            labels.append("parent")
        inits.append(init_cos2(n0))
        labels.append("cos2")
        inits.append(init_power_law(n0, 0.7))
        labels.append("pw07")
        inits.append(init_gaussian(n0, 0.12))
        labels.append("gauss")

        # Remove near-duplicates
        unique_inits = []
        unique_labels = []
        for init, lbl in zip(inits, labels):
            is_dup = False
            for u in unique_inits:
                if np.sum(np.abs(init - u)) < 0.03:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(init)
                unique_labels.append(lbl)

        t_per = max(15.0, t1 / len(unique_inits))
        for idx, (init, lbl) in enumerate(zip(unique_inits, unique_labels)):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            t_avail = min(t_per, remaining - 20)
            if t_avail < 10:
                break
            print(f"  LSE-MD '{lbl}' (t={t_avail:.0f}s)...")
            a_out, _ = lse_mirror_descent(init, t_avail, seed=100 + idx)
            try_update(a_out, f"lse_{lbl}")

    # === Phase 2: LP Frank-Wolfe with line search (main workhorse) ===
    print(f"\n=== Phase 2: LP Frank-Wolfe + Line Search ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.30, 500.0)

    if t2 > 30:
        a_in = norm1(best_seq)
        a_out, _ = fw_with_linsearch(a_in, t2, solver, verbose=True, K=500, max_iters=100000)
        try_update(a_out, f"fw_ls_n{n_best}")

    # === Phase 3: Perturbation refinement ===
    print(f"\n=== Phase 3: Perturbation Refinement ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)

    if t3 > 40:
        a_in = norm1(best_seq)
        a_out, _ = perturbation_refine(a_in, t3, solver, n_perturbs=6, verbose=True, seed=555)
        try_update(a_out, f"pert_n{n_best}")

    # === Phase 4: Another FW round on best ===
    print(f"\n=== Phase 4: FW Round 2 ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.15, 250.0)

    if t4 > 20:
        a_in = norm1(best_seq)
        a_out, _ = fw_with_linsearch(a_in, t4, solver, verbose=True, K=600, max_iters=50000)
        try_update(a_out, f"fw2_n{n_best}")

    # === Phase 5: LSE polish ===
    print(f"\n=== Phase 5: LSE Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 150.0)

    if t5 > 20:
        a_in = norm1(best_seq)
        a_out, _ = lse_mirror_descent(a_in, t5, seed=999)
        try_update(a_out, f"lse_polish_n{n_best}")

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Length Exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    length_candidates = []
    for delta in [-200, -100, 100, 200, 400, 600]:
        cand = n_best + delta
        if 1500 <= cand <= 6000:
            length_candidates.append(cand)

    for new_n in length_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"  Trying n={new_n}")
        a_long = interpolate_seq(best_seq, new_n)
        t6a = min(remaining * 0.3, 200.0)
        if t6a > 20:
            a_out, _ = fw_with_linsearch(a_long, t6a, solver, verbose=False, K=400, max_iters=30000)
            try_update(a_out, f"fw_n{new_n}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t6b = min(remaining * 0.2, 100.0)
            if t6b > 15:
                a_out, _ = lse_mirror_descent(norm1(best_seq), t6b, seed=333)
                try_update(a_out, f"lse_n{new_n}")

    # === Phase 7: Final perturbation + FW ===
    print(f"\n=== Phase 7: Final Refinement ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    if remaining > 60:
        t7 = min(remaining * 0.5, 300.0)
        a_in = norm1(best_seq)
        # Small perturbation
        rng = np.random.default_rng(42)
        for p in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            sigma = 0.01 + 0.005 * p
            pert = rng.normal(0, sigma, n_best) * a_in
            a_p = np.maximum(a_in + pert, 0.0)
            a_p = a_p / a_p.sum()
            t_sub = min(remaining * 0.3, 150.0)
            if t_sub > 10:
                a_out, _ = fw_with_linsearch(a_p, t_sub, solver, verbose=False, K=500)
                try_update(a_out, f"final_pert{p}_n{n_best}")

    # === Phase 8: Final FW on absolute best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        n_best = len(best_seq)
        t8 = min(remaining * 0.6, 200.0)
        if t8 > 15:
            a_in = norm1(best_seq)
            a_out, _ = fw_with_linsearch(a_in, t8, solver, verbose=False, K=500)
            try_update(a_out, f"final_fw_n{n_best}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        best_seq = init_cos2(n_fb)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback: score={best_score:.6f}")

    return [float(x) for x in best_seq]
