    """Optimize using L-BFGS-B with smooth approximation."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    t0 = time.monotonic()
    
    def obj_and_grad(x):
        a_x = np.maximum(x, 0.0)
        s = a_x.sum()
        if s < 1e-15:
            return 1e10, np.zeros(n)
        a_norm = a_x / s
        conv = fast_conv(a_norm)
        # Soft max
        m = conv.max()
        shifted = beta * (conv - m)
        exp_s = np.exp(shifted)
        sum_exp = exp_s.sum()
        weights = exp_s / sum_exp
        soft_val = m + np.log(sum_exp) / beta
        
        # Gradient of soft_max w.r.t. a_norm
        # grad_f[i] = 2 * sum_j a_norm[j] * weights[i+j]
        grad_f = np.zeros(n)
        for i in range(n):
            grad_f[i] = 2.0 * np.dot(a_norm, weights[i:i+n])
        
        # Chain rule for normalization
        # dJ/da_i = (1/s) * (grad_f[i] - a_norm[i] * sum(grad_f))
        # But we want gradient w.r.t. x (the unnormalized variable)
        # Since a_x = max(x, 0), da_x/da_i = 1 if x_i > 0, else 0
        sum_grad_f = grad_f.sum()
        grad_x = (grad_f - a_norm * sum_grad_f) / s
        grad_x[x < 0] = 0  # subgradient at boundary
        
        val = 2.0 * n * soft_val
        return val, grad_x
    
    # Add small regularization to break homogeneity
    def obj_and_grad_reg(x):
        val, grad = obj_and_grad(x)
        s = np.maximum(x, 0).sum()
        # Regularization: eps * (s - 1)^2
        eps = 1e-8
        val += eps * (s - 1.0)**2
        if s > 0:
            grad += 2.0 * eps * (s - 1.0) / s * np.where(x > 0, 1.0, 0.0)
        return val, grad
    
    result = minimize(
        obj_and_grad_reg, a,
        method='L-BFGS-B',
        bounds=[(0, None)] * n,
        options={'maxiter': maxiter, 'maxfun': 10000, 
                 'ftol': 1e-12, 'gtol': 1e-8,
                 'max time': time_limit}  # scipy doesn't support max time directly
    )
    
    a_opt = np.maximum(result.x, 0.0)
    s = a_opt.sum()
    if s > 1e-15:
        a_opt = a_opt / s
    else:
        a_opt = a.copy()
    
    return a_opt
