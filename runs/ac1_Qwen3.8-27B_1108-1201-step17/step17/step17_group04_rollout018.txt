    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    ac = fast_conv(a)
    return 2.0 * n * ac.max()


def evaluate_safe(a):
    """Safe evaluation."""
    try:
        a_list = [float(max(0.0, x)) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def make_taper_ramp(n, p=0.8, q=0.3, alpha=2.0, gamma=0.45):
    """
    Parametrized family: a[i] = (i+1)^p * (n-1-i+1)^q * exp(-alpha*(i/n - gamma)^2)
    This creates an asymmetric shape with tapering at both ends.
    """
    i = np.arange(n, dtype=np.float64)
    x = i / max(n - 1, 1)
    a = (i + 1.0) ** p * (n - i) ** q * np.exp(-alpha * (x - gamma) ** 2)
    a = np.maximum(a, 1e-10)
    return normalize(a)


def make_symmetric_bump(n, sigma=0.2, center=0.5):
    """Gaussian bump with tapering."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(n - 1, 1)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    # Taper at edges
    taper = np.minimum(x * 10.0, 1.0) * np.minimum((1.0 - x) * 10.0, 1.0)
    a = a * (0.1 + 0.9 * taper)
    return normalize(a)


def get_subgradient_weighted(a, ac, top_k=12, beta=4.0):
    """Weighted subgradient with exponential peak weighting."""
    n = len(a)
    conv_len = 2 * n - 1
    grad = np.zeros(n)
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        grad[idx] += 2.0 * wi * a[j_idx]
    return grad


def get_subgradient_balanced(a, ac, top_k=15, n_iter=50):
    """
    Balanced subgradient: find weights on active peaks that make the
    resulting gradient as uniform as possible (KKT-inspired).
    """
    n = len(a)
    conv_len = 2 * n - 1
    k_max = min(top_k, conv_len)
    top_idx = np.argsort(ac)[::-1][:k_max]

    # Build gradient matrix G: G[i, j] = (grad of c[k_j] w.r.t. a[i])
    G = np.zeros((n, k_max))
    for j, k in enumerate(top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        G[idx, j] = 2.0 * a[j_idx]

    # Start with weights proportional to peak values
    peak_vals = np.maximum(ac[top_idx], 1e-30)
    w = peak_vals ** 3.0
    w = w / w.sum()

    # Iterate: adjust weights to make G@w more uniform
    for _ in range(n_iter):
        g = G @ w  # shape (n,)
        g_mean = g.mean()
        g_centered = g - g_mean
        # Gradient of ||g_centered||^2 w.r.t. w: 2 * G^T @ g_centered
        grad_w = 2.0 * (G.T @ g_centered)
        # Project gradient descent step onto simplex over w
        lr = 0.01
        w_new = w - lr * grad_w
        w_new = np.maximum(w_new, 0)
        sw = w_new.sum()
        if sw < 1e-15:
            break
        w_new = w_new / sw
        if np.max(np.abs(w_new - w)) < 1e-10:
            w = w_new
            break
        w = w_new

    return G @ w


def mirror_descent_step(a, grad, eta):
    """Mirror descent step on the simplex (exponential family)."""
    log_a = np.log(np.maximum(a, 1e-30)) - eta * grad
    max_log = log_a.max()
    a_new = np.exp(log_a - max_log)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def refined_subgradient(a0, time_limit, verbose=False, seed=42):
    """
    Nesterov-accelerated mirror descent with backtracking line search
    and adaptive peak weighting.
    """
    global _global_best_seq, _global_best_score
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_obj = compute_obj(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall = 0
    cur_top_k = 12
    cur_beta = 4.0
    use_balanced = False
    balanced_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 1:
            beta_t = (t - 1.0) / (t + 1.0)
            y = x + beta_t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_conv(y)
        M = ac.max()

        # Choose subgradient type (alternate)
        balanced_count += 1
        if balanced_count % 5 == 0:
            g = get_subgradient_balanced(y, ac, top_k=cur_top_k)
        else:
            g = get_subgradient_weighted(y, ac, top_k=cur_top_k, beta=cur_beta)

        # Backtracking line search
        eta = 2.0 / (t + 1.0)
        best_step_obj = compute_obj(y)
        best_step_a = y.copy()
        for bt in range(8):
            a_candidate = mirror_descent_step(y, g, eta)
            obj_candidate = compute_obj(a_candidate)
            if obj_candidate < best_step_obj - 1e-14:
                best_step_obj = obj_candidate
                best_step_a = a_candidate
                # Try one more step with larger eta
                eta *= 1.5
                continue
            else:
                eta *= 0.5
                if eta < 1e-10:
                    break

        x_prev = x
        x = best_step_a

        if best_step_obj < best_obj - 1e-14:
            best_obj = best_step_obj
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive restarts
        if stall > 3000:
            stall = 0
            sigma = 0.005 + 0.005 * min(stall // 3000, 5)
            pert = rng.normal(0, sigma, n) * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 3, 40)
            cur_beta = max(cur_beta - 0.3, 1.5)

    # Update global
    sc = evaluate_safe(best_a)
    if sc < _global_best_score:
        _global_best_score = sc
        _global_best_seq = best_a.copy()
        if verbose:
            print(f"  Subgradient: obj={best_obj:.6f}, score={sc:.6f}")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False, seed=123):
    """
    Coordinate descent: for each coordinate, find optimal value via
    incremental convolution update.
    """
    global _global_best_seq, _global_best_score
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_ac = fast_conv(a)
    best_M = best_ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    stall = 0
    max_stall = 2000
    order = np.random.permutation(n)
    pos = 0

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        i = order[pos % n]
        pos += 1

        if best_M < 1e-15:
            break

        cur_val = a[i]
        if cur_val < 1e-12:
            continue

        # Try several delta values
        deltas = [
            -cur_val * 0.5, -cur_val * 0.3, -cur_val * 0.15,
            -cur_val * 0.05, cur_val * 0.05, cur_val * 0.15,
            cur_val * 0.3, cur_val * 0.5
        ]
        # Add some random deltas
        for _ in range(3):
            deltas.append(cur_val * (rng.uniform(-0.5, 0.5)))

        best_delta = 0.0
        best_trial_M = best_M

        for d in deltas:
            new_val = cur_val + d
            if new_val < -1e-12:
                continue
            new_val = max(new_val, 0.0)
            if abs(new_val - cur_val) < 1e-14:
                continue

            # Update convolution incrementally
            # c[k] changes by: d * (a[k-i] + a[k-i] for the cross terms)
            # More precisely: c[k] = sum_j a[j]*a[k-j]
            # Changing a[i] by d: c[k] += d * a[k-i] for k where k-i is valid
            # And c[2i] += a[i]*d (the self-term)
            # Actually: c[k] = sum_j a[j]*a[k-j], so:
            # delta c[k] = d * a[k-i] + a[i] * d * (if k-i == i, i.e., k=2i)
            # Wait, let's be more careful:
            # c[k] = sum_j a[j] * a[k-j]
            # d(c[k])/d(a[i]) = a[k-i] (for the term j=i) + a[k-i] (for the term k-j=i, j=k-i)
            # = a[k-i] + a[k-i] = 2*a[k-i] if k-i != i
            # = 2*a[i] if k-i == i (i.e., k == 2i), but actually it's a[i] (the term j=i gives a[i]*a[k-i]=a[i]*a[i])
            # Hmm, let me just compute directly.
            #
            # c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
            # The terms involving a[i] are:
            #   j = i: a[i] * a[k-i] (if 0 <= k-i < n)
            #   k-j = i, i.e., j = k-i: a[k-i] * a[i] (if 0 <= k-i < n, and k-i != i)
            # So: d(c[k])/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i] if k-i != i and 0<=k-i<n
            #                               = 2*a[i] if k-i == i (k=2i)
            # Wait, if k-i == i, then j=i and k-j=i are the same term, so we only count it once:
            # d(c[2i])/d(a[i]) = 2*a[i] (from the two contributions of a[i]^2)
            # Actually no: c[2i] includes the term a[i]*a[i] = a[i]^2, so d/d(a[i]) = 2*a[i].
            # For k != 2i: c[k] includes a[i]*a[k-i] and a[k-i]*a[i] (two separate terms),
            # so d(c[k])/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i].
            #
            # But wait, in the linear convolution, c[k] = sum_j a[j]*a[k-j].
            # The term with j=i is a[i]*a[k-i].
            # The term with j=k-i is a[k-i]*a[i] (if k-i != i and 0 <= k-i < n).
            # So if k-i != i: d(c[k])/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i]
            # If k-i == i (k=2i): d(c[k])/d(a[i]) = 2*a[i] (derivative of a[i]^2)
            #
            # For the second-order effect (changing a[i] by d):
            # delta c[k] ≈ 2*a[k-i]*d + (if k==2i) d^2
            #
            # For simplicity, just compute the new max by updating only affected terms.

            # Affected k values: k = i + j for j in [0, n-1], so k in [i, i+n-1]
            # Also k = (k-i) + i, same range.
            # Actually the affected range is [i, i+n-1] capped to [0, 2n-2].
            k_min = max(0, i)
            k_max_val = min(2 * n - 2, i + n - 1)

            # For efficiency, only check if the new max could be lower
            # Compute the change in each affected c[k]
            # delta_c[k] = 2 * a[k-i] * d + (d^2 if k == 2*i else 0)
            # where a[k-i] is the ORIGINAL value (before modification) for k-i != i
            # For k = 2*i: delta = 2*a[i]*d + d^2

            new_M = best_M
            for k in range(k_min, k_max_val + 1):
                j = k - i
                if 0 <= j < n:
                    if k == 2 * i:
                        delta_c = 2.0 * cur_val * d + d * d
                    else:
                        delta_c = 2.0 * a[j] * d
                    new_val_k = best_ac[k] + delta_c
                    if new_val_k > new_M:
                        new_M = new_val_k
                    if new_M > best_M + 1e-12:
                        break  # Already worse, skip this delta

            if new_M < best_trial_M - 1e-14:
                best_trial_M = new_M
                best_delta = d

        if best_delta != 0.0:
            a[i] = max(a[i] + best_delta, 0.0)
            # Renormalize
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                best_ac = fast_conv(a)
                best_M = best_ac.max()
                continue
            a = a / s
            # Recompute full convolution (periodically for accuracy)
            if pos % n == 0:
                best_ac = fast_conv(a)
                best_M = best_ac.max()
            else:
                # Incremental update (approximate)
                best_ac = fast_conv(a)
                best_M = best_ac.max()

            if best_M < 1e-15:
                stall = 0
            else:
                stall = 0
        else:
            stall += 1

        # Periodic full recompute
        if pos % (3 * n) == 0:
            best_ac = fast_conv(a)
            best_M = best_ac.max()

    sc = evaluate_safe(a)
    if sc < _global_best_score:
        _global_best_score = sc
        _global_best_seq = a.copy()
        if verbose:
            print(f"  CoordDesc: obj={compute_obj(a):.6f}, score={sc:.6f}")

    return a, compute_obj(a)


def optimize_params(a0, n_params, time_limit, verbose=False):
    """
    Optimize parameters of the parametric family around the current best.
    Uses the current sequence to infer parameters and then refines.
    """
    global _global_best_seq, _global_best_score
    n = len(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    best_a = normalize(a0)
    best_obj = compute_obj(best_a)

    # Try different parametric shapes and blend with current best
    param_grid = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.3]:
        for q in [0.0, 0.2, 0.5, 0.8]:
            for alpha in [1.0, 2.0, 3.0, 5.0]:
                for gamma in [0.35, 0.42, 0.5, 0.58, 0.65]:
                    param_grid.append((p, q, alpha, gamma))

    # Shuffle and try
    rng.shuffle(param_grid)
    tried = 0
    for p, q, alpha, gamma in param_grid:
        if time.monotonic() - t0 > time_limit:
            break
        tried += 1
        param_a = make_taper_ramp(n, p, q, alpha, gamma)
        # Blend with current best
        for mix in [0.3, 0.5, 0.7]:
            blend = mix * best_a + (1 - mix) * param_a
            blend = normalize(blend)
            obj = compute_obj(blend)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = blend.copy()

    sc = evaluate_safe(best_a)
    if sc < _global_best_score:
        _global_best_score = sc
        _global_best_seq = best_a.copy()
        if verbose:
            print(f"  ParamOpt: obj={best_obj:.6f}, score={sc:.6f} (tried {tried} shapes)")

    return best_a, best_obj


def multi_peak_reduction(a0, time_limit, verbose=False, seed=55):
    """
    Targeted multi-peak reduction: identify top peaks and find
    mass relocation moves that reduce the worst peak.
    """
    global _global_best_seq, _global_best_score
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    iterations = 0
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        ac = fast_conv(a)
        M_curr = ac.max()

        if M_curr < best_M - 1e-14:
            best_M = M_curr
            best_a = a.copy()

        # Find the peak
        k_peak = int(np.argmax(ac))

        # Find contributors to the peak
        i_start = max(0, k_peak - n + 1)
        i_end = min(n, k_peak + 1)
        idx = np.arange(i_start, i_end)
        j_idx = k_peak - idx
        contributions = a[idx] * a[j_idx]

        # Find the top contributor
        top_contrib = np.argmax(contributions)
        src_i = idx[top_contrib]
        src_j = j_idx[top_contrib]

        # Find a safe destination (low contribution to peaks)
        # Compute "danger" for each position
        danger = np.zeros(n)
        top_lags = np.argsort(ac)[::-1][:8]
        for k in top_lags:
            k_int = int(k)
            s2 = max(0, k_int - n + 1)
            e2 = min(n, k_int + 1)
            ix = np.arange(s2, e2)
            jx = k_int - ix
            danger[ix] += a[jx]
            danger[jx] += a[ix]

        # Safe positions: low danger AND currently low value
        safe_mask = (danger < np.mean(danger)) & (a < 0.5 * np.mean(a))
        if not np.any(safe_mask):
            safe_mask = danger < np.percentile(danger, 30)
        if not np.any(safe_mask):
            break

        safe_positions = np.where(safe_mask)[0]
        dst = safe_positions[rng.integers(len(safe_positions))]
        if dst == src_i:
            continue

        # Try several transfer fractions
        src_val = a[src_i]
        if src_val < 1e-10:
            continue

        best_frac = None
        best_trial_M = M_curr

        for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
            amt = src_val * frac
            a_trial = a.copy()
            a_trial[src_i] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_frac = frac

        if best_frac is not None and best_trial_M < M_curr - 1e-14:
            a[src_i] -= a[src_i] * best_frac
            a[dst] += a[src_i] * best_frac / (1 - best_frac) if best_frac < 1 else 0
            # Redo properly
            a = a.copy()
            amt = (1 - best_frac) * a[src_i]
            a[src_i] = amt
            a[dst] += (a.sum() - 1.0)  # This is getting messy, let me redo

        # Simpler: just redo the trial
        if best_frac is not None and best_trial_M < M_curr - 1e-14:
            a_trial = a.copy()
            amt = a_trial[src_i] * best_frac
            a_trial[src_i] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            a = a_trial
            best_M = best_trial_M
            best_a = a.copy()
        else:
            # Random perturbation to escape
            if iterations % 100 == 0:
                n_pert = max(3, n // 50)
                pert_pos = rng.choice(n, size=n_pert, replace=False)
                for pp in pert_pos:
                    if a[pp] > 1e-8:
                        dst2 = int(rng.integers(n))
                        amt2 = a[pp] * rng.uniform(0.01, 0.05)
                        a[pp] -= amt2
                        a[dst2] += amt2
                a = np.maximum(a, 0)
                a = a / a.sum()

    sc = evaluate_safe(best_a)
    if sc < _global_best_score:
        _global_best_score = sc
        _global_best_seq = best_a.copy()
        if verbose:
            print(f"  MultiPeak: score={sc:.6f} ({iterations} iters)")

    return best_a, best_obj if 'best_obj' in dir() else compute_obj(best_a)


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    return normalize(f_interp)

    # === Phase 2: Refined subgradient - main optimization (30% time) ===
    print(f"\n=== Phase 2: Subgradient n={n0} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_best, _ = refined_subgradient(a_best, t2, verbose=True, seed=42)

    # === Phase 3: Multi-peak reduction (15% time) ===
    print(f"\n=== Phase 3: Multi-peak reduction n={len(_global_best_seq)} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_best, _ = multi_peak_reduction(_global_best_seq, t3, verbose=True, seed=55)

    # === Phase 4: Coordinate descent polish (20% time) ===
    print(f"\n=== Phase 4: Coordinate descent n={len(_global_best_seq)} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 20:
        a_best, _ = coordinate_descent(_global_best_seq, t4, verbose=True, seed=123)

    # === Phase 5: Second subgradient pass with different params (15% time) ===
    print(f"\n=== Phase 5: Subgradient pass 2 n={len(_global_best_seq)} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        a_best, _ = refined_subgradient(_global_best_seq, t5, verbose=True, seed=999)

    # === Phase 6: Coordinate descent again (10% time) ===
    print(f"\n=== Phase 6: Coord descent pass 2 n={len(_global_best_seq)} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 15:
        a_best, _ = coordinate_descent(_global_best_seq, t6, verbose=True, seed=456)

    # === Phase 7: Try different length (10% time) ===
    print(f"\n=== Phase 7: Length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(_global_best_seq)

    # Try a longer length
    n_long = min(n_best + 400, 4500)
    if remaining > 60 and n_long != n_best:
        print(f"  Trying n={n_long}")
        a_long = interpolate_seq(_global_best_seq, n_long)
        t7a = min(remaining * 0.4, 200.0)
        if t7a > 20:
            a_long, _ = refined_subgradient(a_long, t7a, verbose=True, seed=555)
            sc = evaluate_safe(a_long)
            if sc < _global_best_score:
                _global_best_score = sc
                _global_best_seq = a_long.copy()
                print(f"    Improved: score={sc:.6f} at n={n_long}")

    # Try a shorter length
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(n_best - 400, 2000)
    if remaining > 40 and n_short != len(_global_best_seq):
        print(f"  Trying n={n_short}")
        a_short = interpolate_seq(_global_best_seq, n_short)
        t7b = min(remaining * 0.5, 150.0)
        if t7b > 15:
            a_short, _ = refined_subgradient(a_short, t7b, verbose=True, seed=666)
            sc = evaluate_safe(a_short)
            if sc < _global_best_score:
                _global_best_score = sc
                _global_best_seq = a_short.copy()
                print(f"    Improved: score={sc:.6f} at n={n_short}")

    # === Phase 8: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish n={len(_global_best_seq)} ===")
    if remaining > 30:
        t8 = min(remaining * 0.6, 200.0)
        if t8 > 15:
            a_best, _ = coordinate_descent(_global_best_seq, t8, verbose=True, seed=789)

    # === Phase 9: Final subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t9 = min(remaining * 0.7, 100.0)
        if t9 > 10:
            a_best, _ = refined_subgradient(_global_best_seq, t9, verbose=True, seed=1000)

    # === Final validation ===
    if _global_best_seq is not None:
        fs = evaluate_safe(_global_best_seq)
        if np.isfinite(fs):
            _global_best_score = fs
        print(f"\nFINAL: score={_global_best_score:.6f}, n={len(_global_best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        n_fb = 3200
        _global_best_seq = make_taper_ramp(n_fb)
        _global_best_score = evaluate_safe(_global_best_seq)
        print(f"Fallback: score={_global_best_score:.6f}")

    return [float(x) for x in _global_best_seq]
