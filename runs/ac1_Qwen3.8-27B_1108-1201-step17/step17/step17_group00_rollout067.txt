"""

import numpy as np
import time
import cvxpy as cp

def get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS

def lin_conv(a):
    """Compute linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def lse_smooth_conv_obj(a, temperature):
    """Compute log-sum-exp smoothed version of max(conv(a,a)) and its gradient."""
    n = len(a)
    c = lin_conv(a)
    t = temperature
    c_shifted = c - c.max()
    exp_t_c = np.exp(t * c_shifted)
    w = exp_t_c / exp_t_c.sum()
    lse_val = c.max() + np.log(exp_t_c.sum()) / t
    # Gradient of LSE wrt a: grad[j] = sum_k w[k] * dc[k]/da[j] = 2 * sum_m w[j+m]*a[m]
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return lse_val, grad, w

def adam_logspace(a0, time_limit, t0_temp=20.0, t_max=200.0, t_anneal=1.001,
                  lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8,
                  n_steps_max=2000000, verbose=True, verbose_every=2000):
    """Adam optimization in log-space with annealed temperature."""
    n = len(a0)
    b = np.log(np.maximum(a0, 1e-30))
    b = b - b.mean()  # center to avoid overflow
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t_count = 0
    temp = t0_temp
    
    best_a = a0.copy()
    best_obj = lin_conv(a0).max()
    t0 = time.monotonic()
    
    for step in range(1, n_steps_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        a = np.exp(b - b.max())
        a = a / a.sum()
        
        lse_val, grad_a, w = lse_smooth_conv_obj(a, temp)
        
        # Chain rule: b = log(a), so da[j]/db[j] = a[j]
        grad_b = grad_a * a
        
        t_count += 1
        m_hat = beta1 * m_hat + (1 - beta1) * grad_b
        v_hat = beta2 * v_hat + (1 - beta2) * grad_b ** 2
        m_hat_hat = m_hat / (1 - beta1 ** t_count)
        v_hat_hat = v_hat / (1 - beta2 ** t_count)
        
        b = b - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)
        
        # Periodically check and update best
        if step % 500 == 0:
            a_check = np.exp(b - b.max())
            a_check = a_check / a_check.sum()
            M = lin_conv(a_check).max()
            if M < best_obj - 1e-15:
                best_obj = M
                best_a = a_check.copy()
            
            # Anneal temperature
            temp = min(temp * t_anneal, t_max)
        
        if verbose and step % verbose_every == 0:
            a_check = np.exp(b - b.max())
            a_check = a_check / a_check.sum()
            M = lin_conv(a_check).max()
            J = 2.0 * n * M
            elapsed = time.monotonic() - t0
            print(f"    Adam step={step} ({elapsed:.0f}s): J={J:.6f}, temp={temp:.1f}")
    
    # Final
    a_final = np.exp(b - b.max())
    a_final = a_final / a_final.sum()
    M = lin_conv(a_final).max()
    if M < best_obj:
        best_obj = M
        best_a = a_final.copy()
    
    return best_a, best_obj

def lp_frank_wolfe(a0, time_limit, solver, initial_K=400, max_K=3500,
                   add_per_iter=40, verbose=True, verbose_every=25):
    """LP-based Frank-Wolfe with adaptive cutting-plane."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = lin_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    ac = lin_conv(y)
    M_curr = ac.max()
    init_idx = np.argsort(ac)[::-1][:initial_K]
    constraint_set = set(int(k) for k in init_idx)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = lin_conv(y)
        M_curr = ac.max()
        
        if M_curr < best_m - 1e-15:
            best_m = M_curr
            best_a = y.copy()
        
        ac_x = lin_conv(x)
        M_x = ac_x.max()
        if M_x < best_m - 1e-15:
            best_m = M_x
            best_a = x.copy()
        
        # Manage constraints
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)
        
        K = len(constraint_list)
        if K == 0:
            break
        
        # Build constraint matrix (only for near-active constraints to keep LP small)
        # Sort by activity (most active first)
        sort_order = np.argsort([ac[k] for k in constraint_list])[::-1]
        active_list = [constraint_list[i] for i in sort_order]
        
        # Use at most max_K constraints, prioritize most active
        n_use = min(len(active_list), max_K)
        use_list = active_list[:n_use]
        
        coeffs = np.zeros((n_use, n))
        for jj, k in enumerate(use_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]
        
        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1]
        
        # Add constraints in batches for solver efficiency
        constraints.append(coeffs @ g_var <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            # Reduce constraints and retry
            if n_use > initial_K:
                constraint_set = set(int(k) for k in active_list[:initial_K])
            continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum
        
        # Find violated constraints
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_yg = np.fft.irfft(Yf * Gf, fft_size)[:size]
        
        violations = conv_yg > M_curr + 1e-10
        n_viol = violations.sum()
        if n_viol > 0:
            viol_idx = np.where(violations)[0]
            viol_amt = conv_yg[viol_idx] - M_curr
            top_n = min(add_per_iter, len(viol_idx))
            top_v = viol_idx[np.argsort(viol_amt)[::-1][:top_n]]
            for k in top_v:
                constraint_set.add(int(k))
        
        # Frank-Wolfe step with Nesterov
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t <= 2:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        if verbose and t % verbose_every == 0:
            J = 2.0 * n * best_m
            elapsed = time.monotonic() - t0
            gap = ac.max() - M_x
            print(f"    LP-FW t={t} ({elapsed:.0f}s): J={J:.6f}, |S|={len(constraint_set)}, gap={gap:.2e}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, rng_seed=42, verbose=True, verbose_every=50):
    """Coordinate descent polish."""
    n = len(a0)
    a = norm_sum1(a0).copy()
    best_a = a.copy()
    best_m = lin_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)
    stall = 0
    count = 0
    
    while time.monotonic() - t0 < time_limit and stall < 600:
        count += 1
        ac = lin_conv(a)
        M_curr = ac.max()
        improved = False
        
        top_k = np.argsort(ac)[::-1][:25]
        
        for k in top_k[:12]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Score destinations: positions that contribute least to peak constraints
            scores = np.zeros(n)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    other = k2i - idx2
                    if (other >= 0).any() and (other < n).any():
                        mask = (other >= 0) & (other < n)
                        scores[other[mask]] += a[idx2[mask]]
            
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:10]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                in_peak[s2:e2+1] = True
            
            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            dst = candidates[np.argmin(scores[candidates])]
            if dst == src:
                continue
            
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                M_trial = lin_conv(a_trial).max()
                if M_trial < best_m - 1e-16:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            if stall > 100:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = lin_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
        else:
            stall = 0
        
        if verbose and count % verbose_every == 0:
            J = 2.0 * n * best_m
            print(f"    Coord ({time.monotonic()-t0:.0f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def parametric_init(n, time_limit, best_init=None, best_val=2.0):
    """Try various parametric shapes."""
    t0 = time.monotonic()
    x = np.linspace(0.0, 1.0, n)
    i_arr = np.arange(n, dtype=np.float64)
    
    best_a = norm_sum1(best_init) if best_init is not None else np.ones(n, dtype=np.float64) / n
    best_score = best_val
    updated = False
    
    def try_shape(a_arr, label):
        nonlocal best_a, best_score, updated
        a_arr = norm_sum1(a_arr)
        c = lin_conv(a_arr)
        val = 2.0 * n * c.max()
        if val < best_score:
            best_score = val
            best_a = a_arr.copy()
            updated = True
            print(f"    Param [{label}]: J={val:.6f}")
    
    # Power ramps (increasing)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        try_shape((x + 0.01) ** p, f"ramp_{p}")
    
    # Exponential ramps
    for rate in [1, 2, 3, 5, 8, 12]:
        try_shape(np.exp(rate * x), f"exp_{rate}")
        try_shape(np.exp(-rate * x), f"expm_{rate}")
    
    # Beta-like shapes
    for a_par in [0.5, 1.0, 2.0, 3.0]:
        for b_par in [0.5, 1.0, 2.0, 3.0]:
            try_shape((x + 0.01) ** (a_par - 1) * (1 - x + 0.01) ** (b_par - 1), f"bet_{a_par}_{b_par}")
    
    # Sine powers
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        try_shape(np.sin(np.pi * x) ** p, f"sinp_{p}")
    
    return best_a, best_score
