def fast_conv(a):

    size = 2 * n - 1

def eval_J(a):

def compute_subgradient(a, conv, n_top=100, beta=8.0):
    """Compute subgradient of max(conv(a,a)) at a using top-K lags with softmax weighting."""
    n = len(a)
    conv_max = conv.max()
    n_active = min(n_top, len(conv))
    
    # Get top-K lags
    top_idx = np.argpartition(conv, -n_active)[-n_active:]
    top_idx = top_idx[np.argsort(conv[top_idx])[::-1]]
    top_vals = conv[top_idx]
    
    # Softmax weighting (sharp for peaks, but includes near-peaks)
    margins = conv_max - top_vals
    weights = np.exp(-beta * margins / (conv_max + 1e-15))
    weights = weights / weights.sum()
    
    g = np.zeros(n)
    for wk, k in zip(weights, top_idx):
        ki = int(k)
        i_s = max(0, ki - n + 1)
        i_e = min(n - 1, ki)
        if i_s <= i_e:
            g[i_s:i_e+1] += 2.0 * wk * a[ki - i_s : ki - i_e - 1 : -1]
    
    return g

def fft_flatten(a, alpha=0.5, iterations=50):
    """Apply Gonzales-Lagarias style spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a_current = a.copy()
    
    for _ in range(iterations):
        # Pad and FFT
        a_padded = np.zeros(L)
        a_padded[:n] = a_current
        A = np.fft.rfft(a_padded)
        
        # Target: flat spectrum
        power = np.abs(A) ** 2
        target_power = power.sum() / len(power)
        target_amp = np.sqrt(target_power)
        
        # Blend current and target amplitude
        current_amp = np.abs(A)
        current_amp = np.maximum(current_amp, 1e-15)
        phase = A / current_amp
        
        # New amplitude: geometric mean of current and target
        new_amp = current_amp ** (1 - alpha) * target_amp ** alpha
        A_new = phase * new_amp
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, L)[:n]
        
        # Project: non-negative, normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check if close to current
        if np.allclose(a_new, a_current, atol=1e-10):
            a_current = a_new
            break
        a_current = a_new
    
    return a_current

        # Nesterov combination
        alpha_t = 2.0 / (t + 2.0)
        x_prev = x
        x = (1.0 - alpha_t) * x + alpha_t * a_new
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        
        # Track best

        if val < best_val - 1e-14:

def make_smooth_random(n, seed=0):

def interpolate_seq(a, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_i = np.interp(new_x, old_x, a)
    a_i = np.maximum(a_i, 0.0)
    s = a_i.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_i / s
