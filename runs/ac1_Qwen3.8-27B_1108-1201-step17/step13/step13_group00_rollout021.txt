        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_window(a, time_limit, rng, window=30, verbose=True):
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1):
                j_candidates = [(i + offset) % n, (i - offset) % n]
                for j in j_candidates:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    delta_vec = np.zeros(size)
                    lo1 = i
                    hi1 = min(size, i + n)
                    if hi1 > lo1:
                        src = np.arange(lo1 - i, hi1 - i)
                        delta_vec[lo1:hi1] += a[src]
                    lo2 = max(0, n - 1 - i)
                    hi2 = min(size, n - 1 + i + 1)
                    if hi2 > lo2:
                        src2 = n - 1 - np.arange(lo2, hi2)
                        valid = (src2 >= 0) & (src2 < n)
                        delta_vec[lo2:hi2][valid] += a[src2[valid]]

                    lo3 = j
                    hi3 = min(size, j + n)
                    if hi3 > lo3:
                        src3 = np.arange(lo3 - j, hi3 - j)
                        delta_vec[lo3:hi3] -= a[src3]
                    lo4 = max(0, n - 1 - j)
                    hi4 = min(size, n - 1 + j + 1)
                    if hi4 > lo4:
                        src4 = n - 1 - np.arange(lo4, hi4)
                        valid4 = (src4 >= 0) & (src4 < n)
                        delta_vec[lo4:hi4][valid4] -= a[src4[valid4]]

                    delta_vec *= 2.0

                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 15
                    deltas = np.linspace(0.0, a[i] * 0.99, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-13:
                            best_val = new_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_polish(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 6
    temps = np.geomspace(10.0, 0.3, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(20, min(200, int(remaining * 4)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_autocorr(a)
            max_conv = conv.max()
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()
            w = w / w.sum()

            A_fft = rfft(a, fft_size)
            W_fft = rfft(w, fft_size)
            cross1 = irfft(W_fft * np.conj(A_fft), fft_size)
            g1 = np.zeros(n)
            for i in range(n):
                k_lo = max(0, i)
                k_hi = min(n, i + n)
                if k_hi > k_lo:
                    g1[i] = np.dot(w[k_lo:k_hi], a[k_lo - i:k_hi - i])
            cross2 = irfft(A_fft * np.conj(W_fft), fft_size)
            g2 = np.zeros(n)
            for i in range(n):
                k_lo2 = max(0, i - n + 1)
                k_hi2 = min(n, i + 1)
                if k_hi2 > k_lo2:
                    g2[i] = np.dot(w[k_lo2:k_hi2], a[i - np.arange(k_lo2, k_hi2)])

            grad = 2.0 * (g1 + g2)

            step = norm / (T * n) / (1.0 + 0.1 * it)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov - main workhorse (65% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={remaining * 0.65:.0f}s) ===")
    t1 = remaining * 0.65
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # Phase 2: Pair CD with window (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Pair CD window n={n_best} (budget={remaining * 0.20:.0f}s) ===")
    t2 = remaining * 0.20
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t2, rng, window=20, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 3: LSE polish (10% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE polish n={n_best} (budget={remaining * 0.10:.0f}s) ===")
    t3 = remaining * 0.10
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_polish(a_norm, t3, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 4: Second LP Nesterov round (refine after polish)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={n_best} (budget={remaining * 0.30:.0f}s) ===")
    t4 = remaining * 0.30
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 5: Final pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final Pair CD n={n_best} (budget={remaining * 0.40:.0f}s) ===")
    t5 = remaining * 0.40
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_window(a_norm, t5, rng, window=40, verbose=True)
        do_try_update(a_out, f"pcd2_n{n_best}")

    # Phase 6: Try alternate length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n_base + 100
        print(f"\n=== Phase 6: Alternate length n={n_alt} (budget={remaining * 0.5:.0f}s) ===")
        t6 = remaining * 0.5
        if t6 > 30:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_norm_alt = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            t6a = min(t6 * 0.6, remaining * 0.3)
            a_out_alt, _ = nesterov_lp_optimize(a_norm_alt, t6a, solver, verbose=False)
            do_try_update(a_out_alt, f"lp_n{n_alt}")
            remaining2 = time_budget - (time.monotonic() - t_global)
            if remaining2 > 30 and best_seq is not None and len(best_seq) == n_alt:
                a_norm_alt2 = best_seq * np.sqrt(2.0 * n_alt) / best_seq.sum()
                a_out_alt2, _ = pair_cd_window(a_norm_alt2, min(remaining2 * 0.5, 40.0), rng, window=20, verbose=False)
                do_try_update(a_out_alt2, f"pcd_n{n_alt}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
