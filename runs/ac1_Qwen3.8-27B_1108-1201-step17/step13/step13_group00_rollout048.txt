def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def normalize_seq(a, target_sum):
    """Normalize a to have the given sum, keeping non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s


def build_conv_matrix(a):
    """Build the convolution matrix T where (T @ b)[k] = sum_j b_j * a[k-j]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response_screened(a, M, solver, active_set=None, tol_frac=0.02, min_active=20):
    """LP best response with constraint screening."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    conv_current = T @ a

    if active_set is None or len(active_set) < min_active:
        threshold = M * (1.0 - tol_frac)
        active_set = np.where(conv_current >= threshold)[0]
        if len(active_set) < min_active:
            active_set = np.argsort(conv_current)[-min_active:]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T[active_set, :] @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set

    # Check dropped constraints
    if len(active_set) < size:
        full_conv = T @ b_val
        violated = np.where(full_conv > M * (1 + 1e-6))[0]
        if len(violated) > 0:
            new_active = np.union1d(active_set, violated)
            if len(new_active) > 0.85 * size:
                # Full LP
                b_var2 = cp.Variable(n, nonneg=True)
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T @ b_var2 <= M])
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, active_set
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                    s = b_val.sum()
                    if s < 1e-15:
                        return None, active_set
                return b_val, active_set
            else:
                b_var2 = cp.Variable(n, nonneg=True)
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T[new_active, :] @ b_var2 <= M])
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, active_set
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                    s = b_val.sum()
                    if s < 1e-15:
                        return b_val, active_set

    return b_val, active_set


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    active_set = None
    tol_frac = 0.02

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Adaptively tighten the screening tolerance
        if n_iters % 100 == 50:
            tol_frac = max(0.005, tol_frac * 0.95)

        b, active_set = lp_best_response_screened(y, M_curr, solver, active_set, tol_frac)
        if b is None:
            active_set = None
            # Fallback: use subgradient step
            conv_y = fast_autocorr(y)
            k_star = int(np.argmax(conv_y))
            grad = np.zeros(n)
            for i in range(n):
                j1 = k_star - i
                if 0 <= j1 < n:
                    grad[i] += y[j1]
                j2 = i - (k_star - (n - 1))
                if 0 <= j2 < n:
                    grad[i] += y[j2]
            grad *= 2.0
            i_min = int(np.argmin(grad))
            b = np.zeros(n)
            b[i_min] = norm

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def compute_pair_gain(a, i, j):
    """Compute the gain vector g_k = 2*(a_{k-j} - a_{k-i}) for pair (i,j)."""
    n = len(a)
    size = 2 * n - 1
    g = np.zeros(size)
    # a_{k-j} for k in [j, j+n-1]
    lo1 = j
    hi1 = min(size, j + n)
    g[lo1:hi1] += a[:hi1 - lo1]
    # a_{k-i} for k in [i, i+n-1]
    lo2 = i
    hi2 = min(size, i + n)
    g[lo2:hi2] -= a[:hi2 - lo2]
    # a_{j-k} = a_{n-1-(k-j)} for k in [j, j+n-1]... using symmetry
    # a_{i-k} = a_{n-1-(k-i)} for k in [i, i+n-1]
    # a_{n-1-m} for m in [0, n-1] -> positions n-1-m
    # a_{j-k}: set m = k-j, then a_{j-k} = a_{n-1-m} for m in [0,n-1], k = j+m
    # So at position k=j+m, value is a[n-1-m]
    lo3 = j
    hi3 = min(size, j + n)
    m_idx = np.arange(hi3 - lo3)
    src = n - 1 - m_idx
    valid = (src >= 0) & (src < n)
    g[lo3:lo3 + valid.sum()][valid] += a[src[valid]]

    lo4 = i
    hi4 = min(size, i + n)
    m_idx2 = np.arange(hi4 - lo4)
    src2 = n - 1 - m_idx2
    valid2 = (src2 >= 0) & (src2 < n)
    g[lo4:lo4 + valid2.sum()][valid2] -= a[src2[valid2]]

    g *= 2.0
    return g


def compute_pair_quadratic(a, i, j):
    """Compute h_k = delta_{k,2j} - 2*delta_{k,i+j} + delta_{k,2i} for pair (i,j)."""
    size = 2 * len(a) - 1
    h = np.zeros(size)
    if 2 * j < size:
        h[2 * j] += 1.0
    if 2 * i < size:
        h[2 * i] += 1.0
    if i + j < size:
        h[i + j] -= 2.0
    return h


def min_pair_1d(conv, g, h, max_delta):
    """
    Find delta in [0, max_delta] that minimizes max_k (conv_k + delta*g_k + delta^2*h_k).
    Uses coarse grid + golden section refinement.
    """
    if max_delta <= 0:
        return 0.0, conv.max()

    size = len(conv)
    # Coarse grid
    n_coarse = 30
    deltas = np.linspace(0.0, max_delta, n_coarse)
    # Compute values: conv + delta*g + delta^2*h
    # Vectorized: shape (n_coarse, size)
    D = deltas[:, None]
    vals = conv[None, :] + D * g[None, :] + D**2 * h[None, :]
    max_vals = vals.max(axis=1)
    best_idx = int(np.argmin(max_vals))
    best_val = max_vals[best_idx]
    best_d = deltas[best_idx]

    # Golden section refinement around best
    if 0 < best_idx < n_coarse - 1:
        lo = deltas[best_idx - 1]
        hi = deltas[best_idx + 1]
    else:
        lo = max(0.0, best_d - max_delta / n_coarse)
        hi = min(max_delta, best_d + max_delta / n_coarse)

    # Check boundaries
    if best_idx == 0:
        lo, hi = 0.0, deltas[1]
    elif best_idx == n_coarse - 1:
        lo, hi = deltas[-2], max_delta

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(30):
        if hi - lo < 1e-12 * max_delta + 1e-15:
            break
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = (conv + c * g + c * c * h).max()
        fd = (conv + d * g + d * d * h).max()
        if fc < fd:
            hi = d
        else:
            lo = c
    mid = 0.5 * (lo + hi)
    vmid = (conv + mid * g + mid * mid * h).max()
    if vmid < best_val:
        best_d = mid
        best_val = vmid

    return best_d, best_val


def systematic_pair_cd(a0, time_limit, rng, verbose=False):
    """
    Systematic pair coordinate descent with exact 1D minimization.
    Sweeps through all i's, choosing j's ranked by a heuristic.

    n = len(a0)
    norm = np.sqrt(2.0 * n)

def systematic_pair_cd_v2(a0, time_limit, rng, verbose=False):
    """
    Improved systematic pair coordinate descent.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve_count = 0
    n_target_j = 20

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        pass_improvement = 0.0

        # Order by descending a_i
        active_indices = np.where(a > norm * 5e-6)[0]
        if len(active_indices) == 0:
            break
        order = active_indices[np.argsort(-a[active_indices])]

        for idx in range(len(order)):
            if time.monotonic() - t0 > time_limit - 3:
                break
            i = int(order[idx])
            ai = a[i]
            if ai < norm * 5e-6:
                continue

            # Find the peak position
            k_star = int(np.argmax(conv))

            # Score j's by how much they'd reduce the peak
            # Moving delta from i to j: change at k_star = 2*delta*(a_{k_star-j} - a_{k_star-i})
            # We want a_{k_star-j} to be small
            ki = k_star - i
            if 0 <= ki < n:
                peak_contrib = a[ki]
            else:
                peak_contrib = 0.0

            # a_{k_star - j} is valid for j in [max(0, k_star-n+1), min(n-1, k_star)]
            j_lo = max(0, k_star - (n - 1))
            j_hi = min(n - 1, k_star)
            j_scores = -np.full(n, 1e10)
            if j_hi >= j_lo:
                j_scores[j_lo:j_hi + 1] = -(peak_contrib - a[k_star - np.arange(j_lo, j_hi + 1)])
            # Add diversity: also consider randomness
            j_scores += 0.05 * rng.random(n) * norm
            j_scores[i] = np.inf

            # Pick top candidates
            n_try = min(n_target_j, n - 1)
            top_j = np.argsort(j_scores)[:n_try]

            best_d = 0.0
            best_j = -1
            best_val = conv.max()

            for jj in top_j:
                j = int(jj)
                if j == i:
                    continue

                # Compute gain: 2*(a_{k-j} - a_{k-i}) for all k
                # Vectorized computation
                g = np.zeros(size)
                # +2 * a shifted by j (for a_{k-j})
                lo1 = j
                hi1 = min(size, j + n)
                g[lo1:hi1] += 2.0 * a[:hi1 - lo1]
                # -2 * a shifted by i (for a_{k-i})
                lo2 = i
                hi2 = min(size, i + n)
                g[lo2:hi2] -= 2.0 * a[:hi2 - lo2]
                # +2 * a reversed shifted by j (for a_{j-k})
                # a_{j-k} = a[n-1-(k-j)] for k in [j, j+n-1]
                lo3 = j
                hi3 = min(size, j + n)
                m3 = np.arange(hi3 - lo3)
                src3 = n - 1 - m3
                valid3 = (src3 >= 0) & (src3 < n)
                if valid3.any():
                    g[lo3:lo3 + int(valid3.sum())][valid3] += 2.0 * a[src3[valid3]]
                # -2 * a reversed shifted by i (for a_{i-k})
                lo4 = i
                hi4 = min(size, i + n)
                m4 = np.arange(hi4 - lo4)
                src4 = n - 1 - m4
                valid4 = (src4 >= 0) & (src4 < n)
                if valid4.any():
                    g[lo4:lo4 + int(valid4.sum())][valid4] -= 2.0 * a[src4[valid4]]

                # Quadratic term
                h = np.zeros(size)
                if 2 * j < size:
                    h[2 * j] += 1.0
                if 2 * i < size:
                    h[2 * i] += 1.0
                if i + j < size:
                    h[i + j] -= 2.0

                d_opt, val_opt = min_pair_1d(conv, g, h, ai)

                if val_opt < best_val - 1e-14:
                    best_val = val_opt
                    best_d = d_opt
                    best_j = j

            if best_d > 1e-14 and best_j >= 0:
                a[i] -= best_d
                a[best_j] += best_d
                # Renormalize
                s_new = a.sum()
                a = a * norm / s_new
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-14:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    pass_improvement += best_score - cur_max
                    no_improve_count = 0
                else:
                    # Revert
                    a[best_j] -= best_d
                    a[i] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)

        if not pass_improved:
            no_improve_count += 1
            if no_improve_count >= 4:
                break
        else:
            no_improve_count = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCDv2 pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def kkt_polish(a0, time_limit, verbose=False):
    """
    KKT fixed-point polishing.
    At the optimum, for active constraints k: sum_k w_k * (a_{k-i} + a_{i-k}) = lambda
    We iterate: update weights from current conv, then solve for a.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Anneal temperature
    n_iters = 0
    for stage, T in enumerate([1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]):
        if time.monotonic() - t0 > time_limit:
            break
        n_inner = max(10, int((time_limit - (time.monotonic() - t0)) / max(1, len([T2 for T2 in [0.1, 0.05, 0.02, 0.01] if T2 <= T]))))
        n_inner = min(n_inner, 200)

        for it in range(n_inner):
            if time.monotonic() - t0 > time_limit:
                break
            n_iters += 1

            conv = fast_autocorr(a)
            max_conv = conv.max()

            # Softmax weights
            shifted = (conv - max_conv) / T
            exp_shifted = np.exp(shifted)
            w = exp_shifted / exp_shifted.sum()

            # Compute the "effective gradient": sum_k w_k * (a_{k-i} + a_{i-k})
            # This is the gradient of the smoothed objective
            # Using FFT for efficiency:
            # sum_k w_k * a_{k-i} = cross-correlation of w and a at lag i
            # sum_k w_k * a_{i-k} = cross-correlation of w and a at negative lag i

            # Method: conv(w, a) gives sum_j w_j * a[k-j]
            # sum_k w_k * a[i-k] = (conv(w, a))[i] (the i-th element)
            # sum_k w_k * a[k-i] = (conv(a, w))[i] = (conv(w, a))[i] (same by commutativity)
            # Wait: conv(w,a)[k] = sum_j w_j * a[k-j]
            # sum_k w_k * a[i-k]: let j = k, = sum_j w_j * a[i-j] = conv(w,a)[i]
            # sum_k w_k * a[k-i]: let m = k-i, k = m+i, = sum_m w_{m+i} * a_m
            # = sum_m a_m * w_{m+i} = conv(a, w_rev)[n-1-i]...
            # Actually for the full convolution of w and a (both length n):
            # (w*a_full)[k] = sum_j w_j * a[k-j] for valid j
            # We want: S1_i = sum_k w_k * a[i-k] = (w*a_full)[i]
            # And: S2_i = sum_k w_k * a[k-i]
            #   S2_i = sum_k w_k * a[k-i]. Let l = k-i: sum_l w_{l+i} * a_l
            #   = sum_l a_l * w_{l+i}. This is the cross-correlation of a and w at lag i.
            #   = (a * w_rev_full)[n-1-i] where w_rev = w[::-1]
            #   But since we're dealing with the full convolution:
            #   (a * w)[k] = sum_j a_j * w[k-j]
            #   sum_l a_l * w_{l+i} = (a * w)[n-1 - (n-1-i)]... hmm

            # Simpler: just use the fact that for the FULL convolution of two length-n sequences:
            # full_conv[k] = sum_{j=0}^{n-1} w[j] * a[k-j] (valid when 0<=k-j<n)
            # S1_i = sum_k w[k] * a[i-k] = sum_j w[j] * a[i-j] = full_conv_wa[i]
            # S2_i = sum_k w[k] * a[k-i]: substitute m=k-i, k=m+i:
            #   = sum_m w[m+i] * a[m] (valid when 0<=m<n and 0<=m+i<n)
            #   = sum_m a[m] * w[m+i]
            #   This is the full_conv of a and w_rev at position (n-1-i):
            #   Let w_rev[m] = w[n-1-m]
            #   full_conv_aw_rev[k] = sum_j a[j] * w_rev[k-j] = sum_j a[j] * w[n-1-k+j]
            #   Set k = n-1-i: sum_j a[j] * w[n-1-(n-1-i)+j] = sum_j a[j] * w[i+j]
            #   = S2_i (with j=m). Yes!

            fft_size = 1 << (size - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            full_wa = irfft(W * A, fft_size)[:size]  # sum_j w_j * a[k-j]
            w_rev = w[::-1]
            Wr = rfft(w_rev, fft_size)
            full_awr = irfft(A * Wr, fft_size)[:size]  # sum_j a_j * w_rev[k-j]

            # S1_i = full_wa[i] for i in [0, n-1]
            # S2_i = full_awr[n-1-i] for i in [0, n-1]
            S1 = full_wa[:n]
            S2 = full_awr[(n - 1): (2 * n - 1)][::-1]  # positions n-1, n-2, ..., 0
            # Wait: full_awr has length size = 2n-1
            # full_awr[n-1-i] for i=0,...,n-1 gives indices n-1, n-2, ..., 0
            S2 = full_awr[:n][::-1]

            grad = S1 + S2  # This is sum_k w_k * (a_{k-i} + a_{i-k})

            # At the KKT optimum, grad should be constant (= lambda) on the support of a
            # So we want to find a' such that the "effective gradient" is flat
            # Update: a' = a - eta * (grad - mean(grad_on_support))
            # But this is just gradient descent on the smoothed objective

            # Better: directly solve the linear system
            # We want: W_mat @ a' = lambda (on support)
            # where W_mat[i, j] = w contribution...
            # This is essentially what we already computed: grad depends linearly on a
            # grad_i = sum_j M[i,j] * a_j where M[i,j] = sum_k w_k * (delta_{j,k-i} + delta_{j,i-k})
            # At KKT: M @ a = lambda * 1 for a > 0

            # For the fixed-point: a_new proportional to the solution of M @ a = lambda
            # But M depends on w which depends on a...

            # Practical update: use the current grad as a "dual" variable
            # a_new_i = a_i * exp(-eta * (grad_i - grad_mean)) (multiplicative update)
            # This is the "mirror descent" step

            eta = 1.0 / (T * 5.0)
            grad_clipped = np.clip(grad, 1e-10, None)
            # Symmetric log update
            log_ratio = -eta * (np.log(grad_clipped) - np.log(grad_clipped).mean())
            log_ratio = np.clip(log_ratio, -5.0, 5.0)
            a_new = a * np.exp(log_ratio)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new =
