import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    b = fast_autocorr(a)
    return 2.0 * n * b.max()


def evaluate_safe(arr):
    """Safely evaluate a sequence via the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_grad_z(z, T):
    """
    Compute the LogSumExp objective and its gradient w.r.t. z.
    a = softmax(z), obj = (1/T) log sum_k exp(T*(a*a)_k)
    Returns (obj, grad_z).
    """
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()

    w = exp_Tb / exp_Tb.sum()
    obj = (max_Tb + np.log(exp_Tb.sum())) / T

    # Actually for autocorrelation: d(a*a)_k/da_j = a_{k-j} if k-j in [0,n-1], plus a_{j-k} if j-k in [0,n-1]
    # But since we're using the full convolution a*a (not symmetric), let's be careful.
    # np.convolve(a,a)[k] = sum_j a[j]*a[k-j]
    # d/d a[m] of sum_j a[j]*a[k-j] = a[k-m] + a[m-k] (if both indices valid)
    # For the full (2n-1) convolution, d/d a[m] = sum over valid pairs

    # which is 2 * (a_rev * w)[m] in the right range

    A_rev = rfft(a[::-1].copy(), fft_size)

    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1: n - 1 + n]

    # Chain rule: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def mirror_descent_lse(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=30, lr=0.5, verbose=True):
    """
    LogSumExp mirror descent for the z = log(a) parameterization.
    Uses adaptive learning rate and periodic restarts for exploration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()

    best_a = a.copy()
    best_true_score = internal_score(a)

    temps = np.geomspace(T_start, T_end, n_stages)
    rng = np.random.default_rng(12345)

    total_iters = 0
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Adaptive learning rate: start larger, decay within stage
        base_lr = lr * (1.0 + 0.5 * np.cos(np.pi * si / n_stages))
        stage_time = remaining / (n_stages - si + 1)

        # Estimate how many iterations we can do (rough: ~0.5ms per iter for n=1600)
        estimated_iters_per_sec = 1500 if n <= 2000 else 800
        max_iters = max(20, int(stage_time * estimated_iters_per_sec * 0.8))

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break

            obj, grad_z = lse_grad_z(z, stage_T)

            # Adaptive step: scale by gradient norm
            gnorm = np.linalg.norm(grad_z)
            if gnorm < 1e-15:
                lr_t = base_lr
            else:
                lr_t = base_lr * min(1.0, 0.1 / (gnorm + 1e-10))

            z = z - lr_t * grad_z
            total_iters += 1

            # Periodic true score check
            if (it + 1) % 200 == 0 or it == max_iters - 1:
                a = softmax_fn(z)
                score = internal_score(a)
                if score < best_true_score:
                    best_true_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_true_score:
            best_true_score = score
            best_a = a.copy()

        # Occasional perturbation to escape local minima
        if si % 5 == 4 and remaining > 60:
            z_pert = z + rng.normal(0, 0.1, n)
            a_pert = softmax_fn(z_pert)
            score_pert = internal_score(a_pert)
            if score_pert < best_true_score:
                best_true_score = score_pert
                best_a = a_pert.copy()
                z = z_pert

        if verbose and si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD stage {si}/{n_stages} (T={stage_T:.3f}, {elapsed:.0f}s): best={best_true_score:.6f}")

    return best_a, best_true_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T[k,j] = a_fixed[k-j] if 0<=k-j<n else 0."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, warm_b=None):
    """Solve: min M s.t. (a_fixed*b)_k <= M for all k, sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=100)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=200)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_zero_sum_game(a0, time_limit, solver, verbose=True):
    """
    Zero-sum game: min_a max_b max_k (a*b)_k = max_b min_a max_k (a*b)_k
    Using alternating best responses with averaging.
    """
    import cvxpy as cp
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Best response for player 2 (given a)
        b_star, M_val = solve_lp_best_response(a, solver)
        if b_star is None:
            continue

        # Update a: solve min max_k (b_star * a)_k s.t. sum(a)=1, a>=0
        # This is also an LP (swap roles)
        a_star, _ = solve_lp_best_response(b_star, solver)
        if a_star is None:
            continue

        # Time-average
        avg_a += a_star
        n_avg += 1

        # Check scores
        score_a = internal_score(a_star)
        if score_a < best_score:
            best_score = score_a
            best_a = a_star.copy()
            stall = 0
        else:
            stall += 1

        score_avg = internal_score(avg_a / n_avg)
        if score_avg < best_score:
            best_score = score_avg
            best_a = (avg_a / n_avg).copy()
            stall = 0

        a = a_star

        if stall > 200:
            break

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-game t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def targeted_peak_reduction(a0, time_limit, seed=42):
    """
    Identify top peaks in autocorrelation and perform targeted mass redistribution
    to reduce them. Uses LP dual information for directionality.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        num_peaks = min(15, 2 * n - 1)
        top_k = np.argsort(b)[::-1][:num_peaks]

        improved = False
        for k in top_k:
            # Find contributing pairs for lag k
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find the pair contributing most to this peak
            src_local = np.argmax(contrib)
            src = idx[src_local]
            partner = k - src

            # Find destination: position that contributes least to top peaks
            peak_indices = set(top_k)
            dest_scores = np.zeros(n)
            for kk in peak_indices:
                js2 = max(0, kk - n + 1)
                je2 = min(n - 1, kk)
                idx2 = np.arange(js2, je2 + 1)
                contrib2 = a[idx2] * a[kk - idx2]
                for ii, jj in enumerate(idx2):
                    dest_scores[jj] += contrib2[ii]

            # Prefer destinations with low peak-impact and low current value
            combined = dest_scores + 0.1 * a
            # Exclude source and its partner
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            if 0 <= partner < n:
                mask[partner] = False
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                continue

            # Pick from the lowest-impact candidates
            cand_scores = combined[candidates]
            n_pick = max(1, len(candidates) // 4)
            low_cands = candidates[np.argsort(cand_scores)[:n_pick]]
            dst = low_cands[rng.integers(len(low_cands))]

            # Try different amounts
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                if amt < 1e-15 or amt > a[src]:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Random exploration
            stall += 1
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if stall > 300:
                break

        if time.monotonic() - t0 < time_limit and t0 > 0:
            pass

    return best_a, best_score


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def adam_lse_polish(a0, time_limit, T_start=10.0, T_end=0.02, n_stages=15, lr=0.01):
    """Adam optimizer on LogSumExp for final polish."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        iters = max(20, min(100, int(remaining * 80 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            obj, grad = lse_grad_z(z, stage_T)
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            lr_t = lr / np.sqrt(1 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 10 == 0:
                a = softmax_fn(z)
                score = internal_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score

    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 6: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 7: Another round of targeted peak reduction
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        print(f"\n=== Phase 7: Final Peak Reduction n={len(best_seq)} (remaining={remaining:.0f}s) ===")
        t7 = min(remaining * 0.40, 200.0)
        if t7 > 10:
            a_out, sc_out = targeted_peak_reduction(best_seq, t7, seed=777)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [FinalPeak]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 7: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 8: Very final mirror descent with low temperature
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t8 = min(remaining * 0.50, 150.0)
        if t8 > 10:
            a_out, sc_out = mirror_descent_lse(best_seq, t8, T_start=5.0, T_end=0.01, n_stages=15, lr=0.3, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [FinalMD]: {best_score:.6f}")

    # Final verification and cleanup
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
