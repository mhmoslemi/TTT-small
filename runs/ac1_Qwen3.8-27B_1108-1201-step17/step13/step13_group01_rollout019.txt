def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def lse_objective_and_grad(a, n, T):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    c = fast_autocorr(a)
    max_c = c.max()
    if max_c < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T * (c - max_c)
    exp_vals = np.exp(shifted)
    log_sum = max_c + np.log(exp_vals.sum()) / T
    w = exp_vals / exp_vals.sum()
    obj = 2.0 * n * log_sum / (s * s)
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj * a / s
    return obj, grad


def adam_lse_optimize(a0, time_limit, seed=42, T_start=80.0, T_end=0.3, lr_init=0.005):
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    stall_no_improve = 0
    patience = 2000
    total_iters = 0
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        lr = lr_init * (0.5 ** (t_count // 50000))
        _, grad = lse_objective_and_grad(a, n, T)
        t_count += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)

            a_new = a.copy()

        total_iters += 1

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stall_no_improve = 0

                t_count = 0

        if b_max < 1e-15:
            break
        threshold = M / (n * b_max)
        b_new = np.maximum(1.0 - b / b_max, 0.0)
        b_new = b_new / b_new.sum()
        alpha = 2.0 / (t_count + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        if t_count == 1:
            a_new = b_new.copy()
        else:
            momentum = (t_count - 1.0) / (t_count + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        y = y_new
        a = a_new
        if t_count % 100 == 0:
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                last_improve = time.monotonic()
        if time.monotonic() - last_improve > 60.0:
            stall_count += 1
            if stall_count > 5:
                break
        else:
            stall_count = 0
        if verbose and t_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov t={t_count} ({elapsed:.0f}s): best={best_score:.10f}")
    score_final = fast_autocorr(a).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a.copy()
    return best_a, best_score


def smooth_init(n, shape_id, rng):
    x = np.linspace(0.0, 1.0, n)
    if shape_id == 0:
        a = np.sin(np.pi * x) ** 2
    elif shape_id == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif shape_id == 2:
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    elif shape_id == 3:
        a = np.exp(-5.0 * (x - 0.4) ** 2) + 0.5 * np.exp(-5.0 * (x - 0.6) ** 2)
    elif shape_id == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif shape_id == 5:
        a = np.sin(np.pi * x) ** 1.5
    elif shape_id == 6:
        a = np.exp(-3.0 * x) + 0.3
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    rng = np.random.default_rng(seed)
    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    n_base = len(best_seq) if best_seq is not None else 1600
    if best_seq is None:
        best_seq = smooth_init(n_base, 0, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # Phase 1: Adam-LSE at base length (long run, slow cooling)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam-LSE n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.35
    a_in = best_seq.copy()
    a_out, sc_out = adam_lse_optimize(a_in, t1, seed=seed, T_start=100.0, T_end=0.3, lr_init=0.003)
    try_update(a_out, f"adam_lse_n{n_base}")

    # Phase 2: Try a longer length
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adam-LSE n={n_long} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 300.0)
    if t2 > 30:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, sc_out = adam_lse_optimize(init_long, t2, seed=seed + 1, T_start=80.0, T_end=0.3, lr_init=0.003)
        try_update(a_out, f"adam_lse_n{n_long}")

    # Phase 3: Nesterov with approximate best response at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 20:
        a_out, sc_out = nesterov_approx_br(best_seq, t3, verbose=True)
        try_update(a_out, f"nesterov_n{n_best}")

    # Phase 4: Adam-LSE low temperature polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 20:
        a_out, sc_out = adam_lse_optimize(best_seq, t4, seed=seed + 2, T_start=5.0, T_end=0.1, lr_init=0.001)
        try_update(a_out, f"lse_polish_n{n_best}")

    # Phase 5: Coordinate descent final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.30, 400.0)
    if t5 > 20:
        a_out, sc_out = peak_coord_descent(best_seq, t5, seed=seed + 3)
        try_update(a_out, "coord_polish_1")

    # Phase 6: More coordinate descent with different seed
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Coord descent 2 n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.50, 500.0)
    if t6 > 20:
        a_out, sc_out = peak_coord_descent(best_seq, t6, seed=seed + 100)
        try_update(a_out, "coord_polish_2")

    # Phase 7: Final LSE very low temp
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LSE ultra-low T n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.60, 300.0)
    if t7 > 15:
        a_out, sc_out = adam_lse_optimize(best_seq, t7, seed=seed + 5, T_start=2.0, T_end=0.05, lr_init=0.0005)
        try_update(a_out, f"lse_ultra_n{n_best}")

    # Phase 8: Final coord descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 15:
        a_out, sc_out = peak_coord_descent(best_seq, remaining * 0.8, seed=seed + 200)
        try_update(a_out, "coord_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
