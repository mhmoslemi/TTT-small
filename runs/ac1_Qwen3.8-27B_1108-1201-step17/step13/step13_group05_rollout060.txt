def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def lp_best_response(a, M, solver):

            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-30:
                continue
            src = idx[np.argmax(contrib)]
            contrib_frac = contrib.max() / total

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            cand = np.where(~peak_mask)[0]
            if len(cand) == 0:
                cand = np.where(np.arange(n) != src)[0]
            cands_sorted = cand[np.argsort(a[cand])]
            n_pick = max(1, len(cands_sorted) // 4)
            dst = cands_sorted[rng.integers(n_pick)]
            if dst == src:
                continue

            for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-18 or a[src] - amt < 0:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial * c / st
                M_trial = fast_convolution(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < c * 1e-8:
                    continue
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                if a[i] >= 0:
                    a = a * c / a.sum()
                    M_new = fast_convolution(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a * c / a.sum()

    if verbose:
        el = time.monotonic() - t0
        J = 2.0 * n * best_m / (c * c)
        print(f"    Coord ({el:.1f}s): J={J:.6f}")
    return best_a, best_m
