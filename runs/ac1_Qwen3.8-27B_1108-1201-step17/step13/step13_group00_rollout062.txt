    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:2 * n - 1]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} via Euclidean projection."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, conv, T, fft_size):
    """Compute gradient of T*log(sum exp(conv_k/T)) w.r.t. a."""
    n = len(a)
    max_c = conv.max()
    w = np.exp((conv - max_c) / T)
    w /= w.sum()
    
    # grad_i = (1/T) * sum_k w_k * (a_{k-i} + a[i-k])
    # First term: sum_k w_k * a[k-i] = circular cross-correlation
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    
    # corr1[i] = sum_k w[k] * a[k-i] (circular) = ifft(W * conj(A))[i]... 
    # Actually: ifft(fft(w)*conj(fft(a))[i] = sum_k w[k]*conj(a[k-i]) = sum_k w[k]*a[k-i] for real
    corr1 = irfft(W * np.conj(A), fft_size)
    # corr1[i] = sum_k w[k] * a[(k-i) mod fft_size]
    # For i in [0, n-1] and fft_size >= 2n-1: corr1[i] = sum_k w[k]*a[k-i] (linear, valid)
    
    # Second term: sum_k w[k] * a[i-k] = sum_k a[i-k]*w[k]
    # = ifft(A * conj(W))[i]... let's verify:
    # ifft(fft(a)*conj(fft(w)))[i] = sum_k a[k]*w[k-i] (circular)
    # That's not quite what we want. We want sum_k w[k]*a[i-k].
    # Let j = i-k: sum_j w[i-j]*a[j] = sum_j a[j]*w[i-j]
    # = ifft(fft(w)*conj(fft(a)))[i] with the roles swapped... 
    # Actually: ifft(fft(a)*conj(fft(w)))[i] = sum_k a[k]*w[(k-i) mod N]
    # For real signals and k in range: = sum_k a[k]*w[k-i]
    # We want sum_k w[k]*a[i-k]. Let's substitute: = sum_k a[i-k]*w[k]
    # Let j = i-k: = sum_j a[j]*w[i-j] = sum_j a[j]*w[-(j-i)]
    # = ifft(fft(a)*conj(fft(w)))[-i mod N] = ifft(fft(a)*conj(fft(w)))[N-i] for i>0
    # = corr2[fft_size - i] where corr2 = ifft(A * conj(W))
    
    corr2 = irfft(A * np.conj(W), fft_size)
    # corr2[i] = sum_k a[k]*w[k-i]
    # We want sum_k w[k]*a[i-k] = corr2[fft_size - i] for i > 0, corr2[0] for i = 0
    
    grad = np.zeros(n)
    # First term: corr1[i] = sum_k w[k]*a[k-i] for i=0..n-1
    grad[:n] += corr1[:n]
    # Second term: for i=0: corr2[0]; for i>0: corr2[fft_size - i]
    grad[0] += corr2[0]
    if n > 1:
        grad[1:] += corr2[fft_size - np.arange(1, n)]
    
    grad /= T
    return grad * 2.0  # factor of 2 from d(conv_k)/d(a_i) having two terms


def lse_projected_gradient(a0, time_limit, T_start, T_end, verbose=True):
    """LSE-smoothed projected gradient with Nesterov acceleration and annealing."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    fft_size = get_fft_size(n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_score = fast_conv(a, fft_size).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    n_total = max(100, int(time_limit * 200))  # estimate total iters
    
    t_k = 1.0
    t_prev = 1.0
    a_prev = a.copy()
    
    for t in range(1, n_total + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        frac = min(1.0, t / n_total)
        T = T_start * (T_end / T_start) ** frac
        
        conv_y = fast_conv(y, fft_size)
        grad = compute_lse_gradient(y, conv_y, T, fft_size)
        
        # Lipschitz estimate: L ~ 2*norm(a)_inf * n / T (rough)
        # Use adaptive step: lr = T / (2 * max(a) * n) approximately
        # Better: use a fixed fraction that works in practice
        max_a = max(y.max(), 1e-8)
        L_est = 2.0 * max_a * n / T
        lr = 1.0 / (L_est * 1.1)
        
        a_new = y - lr * grad
        a_new = project_simplex(a_new, S)
        
        # Nesterov momentum
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + ((t_k - 1.0) / t_curr) * (a_new - a_prev)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new * S / s_y
        
        a_prev = a_new
        a = a_new
        y = y_new
        t_prev = t_k
        t_k = t_curr
        n_iters += 1
        
        # Track best
        score_y = fast_conv(y, fft_size).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_conv(a, fft_size).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-PGD t={t} ({elapsed:.0f}s): T={T:.3f}, best={best_score:.8f}")
    
    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    """Subgradient method with averaging for the exact max."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    fft_size = get_fft_size(n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_score = fast_conv(a, fft_size).max()
    
    y = a.copy()
    a_prev = a.copy()
    t0 = time.monotonic()
    t_k = 1.0
    a_avg = np.zeros(n)
    n_avg = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv(y, fft_size)
        k_star = int(np.argmax(conv_y))
        
        # Subgradient at y: grad_i = 2*(y[k_star-i] + y[i-k_star]) (valid indices)
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += y[j2]
        grad *= 2.0
        
        # Step size: use 1/sqrt(t) or similar
        lr = S / (n * np.sqrt(t + 1.0))
        a_new = y - lr * grad
        a_new = project_simplex(a_new, S)
        
        # Nesterov
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + ((t_k - 1.0) / t_curr) * (a_new - a_prev)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new * S / s_y
        
        a_prev = a_new
        a = a_new
        y = y_new
        t_k = t_curr
        
        # Averaging
        a_avg += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = a_avg / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * S / s_avg
                sc = fast_conv(avg_check, fft_size).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
        
        score_a = fast_conv(a, fft_size).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Subgrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def pair_cd_fast(a0, time_limit, rng, verbose=True):
    """Fast pair coordinate descent with vectorized evaluation."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    fft_size = get_fft_size(n)
    size = 2 * n - 1
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * S / s0
    
    conv = fast_conv(a, fft_size)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve >= 5:
            break
        
        improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if no_improve >= 5:
                break
            
            i = perm[idx]
            if a[i] < S * 1e-10:
                continue
            
            # Try transferring from i to random j
            for attempt in range(5):
                j = int(rng.integers(n))
                if j == i:
                    continue
                
                ai = a[i]
                aj = a[j]
                
                # Fast evaluation: compute change in conv for transfer of delta from i to j
                # conv_new[k] = conv[k] + delta*(a[k-j]+a[j-k] - a[k-i]-a[i-k]) + O(delta^2)
                gain = np.zeros(size)
                
                # +delta at position j: adds delta * (a[k-j] + a[j-k]) to conv[k]
                # a[k-j] valid for k in [j, j+n-1]
                lo, hi = j, min(size, j + n)
                if hi > lo:
                    gain[lo:hi] += a[:hi - lo]
                lo2, hi2 = max(0, n - 1 - j), min(size, n - 1 + j + 1)
                if hi2 > lo2:
                    gain[lo2:hi2] += a[hi2 - lo2 - 1 :: -1][:hi2 - lo2] if False else a[max(0, lo2 - (n-1-j)) : max(0, hi2 - (n-1-j))][::-1]
                
                # -delta at position i: subtracts delta * (a[k-i] + a[i-k]) from conv[k]
                lo3, hi3 = i, min(size, i + n)
                if hi3 > lo3:
                    gain[lo3:hi3] -= a[:hi3 - lo3]
                lo4, hi4 = max(0, n - 1 - i), min(size, n - 1 + i + 1)
                if hi4 > lo4:
                    src = np.arange(lo4, hi4)
                    idx_src = (n - 1) - src
                    valid = (idx_src >= 0) & (idx_src < n)
                    if valid.any():
                        gain[lo4:hi4][valid] -= a[idx_src[valid]]
                
                # Find best delta in [0, ai]
                # Use a few samples
                n_samples = 16
                deltas = np.linspace(0, ai, n_samples + 1)[1:]
                best_d = 0.0
                best_val = conv.max()
                
                for d in deltas:
                    new_conv_est = conv + d * gain
                    # Second order correction at k=2i: (ai-d)^2 vs ai^2, change = -2*ai*d + d^2
                    # At k=2j: (aj+d)^2 vs aj^2, change = 2*aj*d + d^2
                    if 2 * i < size:
                        new_conv_est[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv_est[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    # Cross terms at k=i+j: add d^2 (from a_i*a_j becoming (ai-d)(aj+d) = ai*aj + d*ai - d*aj - d^2)
                    # Actually this is O(d^2) and small, skip for speed
                    
                    new_max = new_conv_est.max()
                    if new_max < best_val - 1e-14:
                        best_val = new_max
                        best_d = d
                
                if best_d > 1e-14:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * S / s
                    conv = fast_conv(a, fft_size)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-14:
                        best_score = cur_max
                        best_a = a.copy()
                        improved = True
                    else:
                        # Revert
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * S / a.sum()
                        conv = fast_conv(a, fft_size)
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and no_improve % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PCD ({elapsed:.0f}s): best={best_score:.8f}, no_improve={no_improve}")
    
    return best_a, best_score


def multiplicative_updates(a0, time_limit, T, verbose=True):
    """Multiplicative (mirror) descent with LSE smoothing."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    fft_size = get_fft_size(n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0) + 1e-10
    a = a * S / a.sum()
    
    best_a = a.copy()
    best_score = fast_conv(a, fft_size).max()
    t0 = time.monotonic()
    
    eta = 1.0 / n  # base step size
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_conv(a, fft_size)
        max_c = conv.max()
        w = np.exp((conv - max_c) / T)
        w /= w.sum()
        
        # Gradient as before
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        corr1 = irfft(W * np.conj(A), fft_size)
        corr2 = irfft(A * np.conj(W), fft_size)
        
        grad = np.zeros(n)
        grad[:n] += corr1[:n]
        grad[0] += corr2[0]
        if n > 1:
            grad[1:] += corr2[fft_size - np.arange(1, n)]
        grad *= 2.0 / T
        
        # Multiplicative update: a_i *= exp(-eta * grad_i)
        eta_t = eta / (1.0 + 0.01 * t)
        update = np.exp(-np.clip(eta_t * grad, -5, 5))
        a = a * update
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a * S / s
        
        score = fast_conv(a, fft_size).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MU t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def structured_init(n, rng, style=None):
    """Generate diverse structured initializations."""
    if style is None:
        style = int(rng.integers(0, 10))
    
    if style == 0:
        # Smooth bump
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif style == 1:
        # Double bump
        x = np.linspace(0, 1, n)
        a = np.exp(-12 * (x - 0.3) ** 2) + np.exp(-12 * (x - 0.7) ** 2)
    elif style == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 4
    elif style == 3:
        # Truncated cosine (always positive)
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.4 * np.cos(x) + 0.2 * np.cos(2 * x) + 0.1 * np.cos(3 * x)
    elif style == 4:
        # Random smooth
        a = np.abs(rng.standard_normal(n))
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif style == 5:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif style == 6:
        # Exponential decay
        a = np.exp(-np.linspace(0, 3, n))
        a = a + np.exp(-np.linspace(3, 0, n))  # symmetric
    elif style == 7:
        # Random binary smoothed
        a = (rng.random(n) > 0.4).astype(float)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
    elif style == 8:
        # Power law
        x = np.linspace(1e-3, 1, n)
        a = x ** 0.5 + (1 - x) ** 0.5
    else:
        # Pure random
        a = np.abs(rng.standard_normal(n))
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def detailed_pair_cd(a0, time_limit, rng, verbose=True):
    """More thorough pair CD: tries all pairs in a smart order."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    fft_size = get_fft_size(n)
    size = 2 * n - 1
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * S / s0
    
    conv = fast_conv(a, fft_size)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    # Precompute the "gain vectors" for each position change
    # gain_add[j] = vector of length size: change in conv when adding 1 to position j
    # gain_add[j][k] = a[k-j] + a[j-k] (valid indices)
    # We can precompute these for all j, but that's O(n^2) memory
    # Instead, compute on the fly but cache for the top positions
    
    # Strategy: focus on pairs where a[i] is large (source) and a[j] is small (target)
    # or where the transfer would reduce the peak
    
    peak_idx = np.argsort(conv)[::-1][:50]  # top 50 peak positions
    
    no_improve = 0
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve >= 8:
            break
        pass_count += 1
        
        improved = False
        # Sort indices by value descending (likely sources first)
        order = np.argsort(a)[::-1]
        
        for ii in range(min(n, 800)):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[ii]
            ai = a[i]
            if ai < S * 1e-8:
                break
            
            # Try a few targets
            n_targets = 8
            targets = rng.choice(n, size=n_targets, replace=False)
            
            best_move = None
            best_move_val = conv.max()
            
            for j in targets:
                if j == i:
                    continue
                aj = a[j]
                
                # Compute gain vector
                gain = np.zeros(size)
                # Adding to j
                lo, hi = j, min(size, j + n)
                gain[lo:hi] += a[:hi - lo]
                # a[j-k] part: for k in [0, j], a[j-k]
                for k in range(min(j + 1, n)):
                    gain[k] += a[j - k]
                # Actually let me just do it more carefully
                # conv[k] = sum_p a[p]*a[k-p]
                # Adding delta to a[j]: new conv[k] = conv[k] + delta*(a[k-j] + a[j-k])
                # a[k-j] valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
                # a[j-k] valid when 0 <= j-k < n, i.e., j-n+1 <= k <= j
                
                # Recompute cleanly
                gain = np.zeros(size)
                # a[k-j] contribution
                k_range = np.arange(j, min(size, j + n))
                gain[k_range] += a[:len(k_range)]
                # a[j-k] contribution  
                k_range2 = np.arange(max(0, j - n + 1), j + 1)
                gain[k_range2] += a[j - k_range2]
                
                # Removing from i
                k_range3 = np.arange(i, min(size, i + n))
                gain[k_range3] -= a[:len(k_range3)]
                k_range4 = np.arange(max(0, i - n + 1), i + 1)
                gain[k_range4] -= a[i - k_range4]
                
                # Find best delta
                n_samples = 12
                deltas = np.linspace(0, ai * 0.9, n_samples + 1)[1:]
                for d in deltas:
                    new_conv = conv + d * gain
                    # Second order: at k=2j: +2*aj*d + d^2; at k=2i: -2*ai*d + d^2
                    if 2 * j < size:
                        new_conv[2 * j] += 2 * aj * d + d * d
                    if 2 * i < size:
                        new_conv[2 * i] -= 2 * ai * d + d * d
                    nm = new_conv.max()
                    if nm < best_move_val - 1e-13:
                        best_move_val = nm
                        best_move = (i, j, d, gain.copy())
            
            if best_move is not None:
                i2, j2, d2, _ = best_move
                a[i2] -= d2
                a[j2] += d2
                s = a.sum()
                a = a * S / s
                conv = fast_conv(a, fft_size)
                cur = conv.max()
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_a = a.copy()
                    improved = True
        
        if improved:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LSE-smoothed projected gradient with Nesterov acceleration
2. Multiplicative (mirror) descent with LSE smoothing
3. Subgradient method with averaging
4. Vectorized pair coordinate descent for local polish
5. Multi-start with diverse structured initializations
Key innovation: Replace expensive LP with O(n log n) FFT-based gradient steps.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def get_fft_size(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def fast_conv(a, fft_size=None):
    n = len(a)
    if fft_size is None:
        fft_size = get_fft_size(n)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:2 * n - 1]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    if not cond.any():
        return np.full(n, S / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_gradient(a, w, fft_size):
    """Compute gradient of sum_k w_k * conv(a,a)[k] w.r.t. a."""
    n = len(a)
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # corr1[i] = sum_k w[k] * a[k-i] (circular, valid for our range)
    corr1 = irfft(W * np.conj(A), fft_size)
    # corr2[i] = sum_k a[k] * w[k-i]
    corr2 = irfft(A * np.conj(W), fft_size)
    
    grad = np.zeros(n)
    # First term: sum_k w[k]*a[k-i] for i=0..n-1 is corr1[i]
    grad[:n] += corr1[:n]
    # Second term: sum_k w[k]*a[i-k] = corr2[(fft_size - i) % fft_size]
    grad[0] += corr2[0]
    if n > 1:
        grad[1:] += corr2[fft_size - np.arange(1, n)]
    
    return grad * 2.0


def lse_pgd(a0, time_limit, T_start, T_end, verbose=True, label=""):
    n = len(a0)
    S = np.sqrt(2.0 * n)
    fft_size = get_fft_size(n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_score = fast_conv(a, fft_size).max()
    y = a.copy()
    a_prev = a.copy()
    t_k = 1.0
    t0 = time.time()
    
    for t in range(1, 10000000):
        if time.time() - t0 > time_limit:
            break
        
        frac = min(1.0, t / 10000)
        T = T_start * (T_end / T_start) ** frac
        
        conv_y = fast_conv(y, fft_size)
        max_c = conv_y.max()
        w = np.exp((conv_y - max_c) / T)
        w /= w.sum()
        
        grad = compute_conv_gradient(y, w, fft_size) / T
        
        # Adaptive step size based on Lipschitz estimate
        L_est = 4.0 * max(y.max(), 1e-6) * n / T
        lr = 1.0 / max(L_est, 1e-10)
        
        a_new = y - lr * grad
        a_new = project_simplex(a_new, S)
        
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + ((t_k - 1.0) / t_curr) * (a_new - a_prev)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new * S / s_y
        
        a_prev = a_new
        a = a_new
        y = y_new
        t_k = t_curr
        
        score_y = fast_conv(y, fft_size).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if verbose and t % 10000 == 0:
