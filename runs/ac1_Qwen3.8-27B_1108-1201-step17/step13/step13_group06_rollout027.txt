Let me also think about the "away Frank-Wolfe" more carefully, as it might be the most effective local optimization method.

Standard Frank-Wolfe for our problem:
- Objective: minimize `f(a) = max_k conv(a,a)[k]` subject to `a >= 0`, `sum(a) = 1`
- At each step:
  1. Compute subgradient `g` at current point `a`
  2. Find `v = argmin_{x in simplex} <g, x> = e_{argmin(g)}`
  3. Step size: `t = 2/(t+2)` (or line search)
  4. Update: `a = (1-t)*a + t*v`

The issue with standard FW is the O(1/t) convergence rate. Away-FW can improve this to O(1/t^2) by also allowing "away" moves.

Away-FW:
- Maintain a "support" set S of vertices that are "active" (have positive weight)
- At each step:
  1. Compute subgradient `g`
  2. Find `v = argmin(g)` (toward direction)
  3. Find `w = argmax_{j in S} a[j]` (away direction)
  4. Try both:
     - Toward: `a_t = (1-t)*a + t*e_v`
     - Away: `a_a = a / (1 - a[w])` with `a_a[w] = 0` (project away from w)
       Then: `a_a = (1-t)*a_a_projected + t*e_v`
  5. Choose the better of the two
  6. Update support: add v if it has enough weight, remove w if its weight is small

Actually, the standard away-FW is:
- `a = sum_{j in S} w_j * e_j` (representation in terms of support)
- Toward: `a_t = (1-t)*a + t*e_v`
- Away: First compute `a' = (a - a[w]*e_w) / (1 - a[w])` (remove w from the support)
  Then: `a_a = (1-t)*a' + t*e_v`

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain GS initialization for diverse starting points
2. Multi-stage Adam + top-k LSE gradient descent
3. Gradient-guided peak shaving with multi-direction search
4. Away-step conditional gradient (Frank-Wolfe)
5. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_obj_grad(a, topk, T):
    """Compute top-k LSE objective and gradient."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    
    # LSE of top values
    max_tv = top_vals.max()
    exp_tv = np.exp((top_vals - max_tv) * T)
    lse_val = (max_tv + np.log(exp_tv.sum()) / T)
    w = exp_tv / exp_tv.sum()
    
    # Gradient
    g = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx] * T  # weight (already includes T from derivative of LSE)
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            i_range = np.arange(js, je + 1)
            g[i_range] += 2.0 * wK * a[K - i_range]
    
    return lse_val, g

def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=20, topk=15, lr_scale=1.0):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        base_lr = 2e-3 * min(1.0, 200.0 / n) * lr_scale
        lr = base_lr * (0.9 ** si)
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        
        max_it = max(30, int(min(remaining, 40.0) * 8))
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            a = softmax_fn(z)
            obj, grad_a = lse_obj_grad(a, topk, stage_T)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 4 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 20 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))
                        m_z = np.zeros(n)
                        v_z = np.zeros(n)
                        t_adam = 0
        
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
    
    return best_a, best_score

Now let me think about the peak shaving more carefully. The key idea is:
- Find the peak at lag k
- The peak value is `conv(a,a)[k] = sum_i a[i]*a[k-i]`
- To reduce this peak, we want to decrease some `a[i]*a[k-i]` products
- We can do this by:
  - Decreasing `a[i]` (and moving the mass elsewhere)
  - Decreasing `a[k-i]` (and moving the mass elsewhere)
  - The "elsewhere" should be a position that doesn't contribute much to any peak

The gradient tells us which positions are most "important" for the peak. We should move mass from important positions to unimportant ones.

def peak_shaving_advanced(a0, time_limit, seed=42):

        bmax = b.max()
        # Top peaks
        top_k = np.argsort(b)[::-1][:8]
        
        # Compute importance
        importance = np.zeros(n)
        for k in top_k[:5]:
            js = max(0, k - (n-1))
            je = min(n-1, k)
            if js <= je:
                i_range = np.arange(js, je+1)
                importance[i_range] += np.abs(a[k - i_range])
        
        imax = importance.max()
        if imax > 1e-15:
            importance = importance / imax
        
        # Find source (high importance) and destination (low importance)
        src_candidates = np.argsort(importance)[::-1][:20]
        dst_candidates = np.argsort(importance)[:20]
        
        improved = False
        for src in src_candidates[:10]:
            if a[src] < 1e-10:
                continue
            for dst in dst_candidates[:10]:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-12:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    new_score = 2.0 * n * new_max
                    if new_score < best_score - 1e-12:
                        a = a_trial
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                    # Simulated annealing: accept worse moves with small probability
                    elif new_score < best_score + current_temp and rng.random() < 0.1:
                        a = a_trial
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            current_temp = max(1e-6, current_temp * 0.95)
            if stall > 100:
                # Random restart
                noise = rng.normal(0, 0.005, n)
                a = a * (1 + noise)
                a = np.maximum(a, 0)
                st = a.sum()
                if st > 1e-10:
                    a = a / st
                stall = 0
            if stall > 500:
                break
    
    return best_a, best_score

def gs_initialize(n, n_iter=200, seed=42):
    """Gerchberg-Saxton style initialization for flat-spectrum sequences."""
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n))
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iter):
        A = np.fft.rfft(a, fft_size)
        # Target: flat magnitude (but preserve DC)
        mag_target = np.full(fft_size // 2 + 1, np.sqrt(np.mean(np.abs(A[1:])**2) + 1e-10))
        mag_target[0] = abs(A[0])  # preserve DC
        # Phase from current
        phase = np.angle(A)
        A_new = mag_target * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n].real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new / s
        else:
            a = rng.dirichlet(np.ones(n))
    
    return a
