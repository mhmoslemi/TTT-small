def frank_wolfe(a0, time_limit):

    n = len(a0)

                g[m] += a[idx2]

        s_vec = np.zeros(n)
        s_vec[s_idx] = 1.0

        a_new = (1 - t) * a + t * s_vec
        a = a_new
        
        avg_a += a
        count += 1
        it += 1
        
        if count % 100 == 0:
            avg = avg_a / count
            score = fast_autocorr(avg).max()
            if score < best_score:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE gradient with Nesterov acceleration (improved cooling schedule)
2. Frank-Wolfe conditional gradient on max subgradient (new)
3. Annealed peak-targeting local search with diverse moves (improved)

5. Final coordinate-descent polish
Key new ideas: FW subgradient method for fast global progress,

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation (convolution with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    """Compute gradient of LSE surrogate for the autocorrelation max."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]
    return r1 + r2


def nesterov_lse(a0, time_limit, T_start=15.0, T_end=0.08, n_stages=16):
    """Nesterov-accelerated LSE gradient descent with aggressive cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_iters_stage = max(80, int(remaining * 3.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break
            if it == 0:
                y_new = y - (1.0 / (it + 3.0)) * lse_gradient(y, T)
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_new = y_extrap - (1.0 / (it + 3.0)) * lse_gradient(y_extrap, T)
            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            y = y_new
            it_count += 1

            if it_count % 5 == 0:
                score = fast_autocorr(y).max()
                if score < best_score:
                    best_score = score
                    best_a = y.copy()

            if it_count % 300 == 0:
                print(f"    LSE stage={si}/{n_stages} T={T:.4f} it={it_count} best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def frank_wolfe(a0, time_limit, verbose=True):
    """Frank-Wolfe (conditional gradient) on the max autocorrelation subgradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    count = 0
    it = 0
    # Track active set for away steps
    active_set = []
    top_k = 3  # Use top-k lags for randomized subgradient

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Randomized subgradient: pick from top few peaks
        sorted_idx = np.argsort(b)[::-1][:top_k]
        k_star = sorted_idx[it % top_k]

        # Compute subgradient g[m] = a[k*-m] + a[m-k*]
        g = np.zeros(n)
        # a[k*-m] valid for m in [max(0, k*-n+1), min(n-1, k*)]
        m_start = max(0, k_star - n + 1)
        m_end = min(n - 1, k_star)
        for m in range(m_start, m_end + 1):
            g[m] += a[k_star - m]
        # a[m-k*] valid for m in [max(0, k*), min(n-1, k*+n-1)]
        m_start2 = max(0, k_star)
        m_end2 = min(n - 1, k_star + n - 1)
        for m in range(m_start2, m_end2 + 1):
            g[m] += a[m - k_star]

        # Standard FW step: move toward argmin g
        s_idx = g.argmin()
        t = 2.0 / (it + 2.0)
        a_new = (1.0 - t) * a + t * 0  # start from zero contribution
        a_new = (1.0 - t) * a
        a_new[s_idx] += t

        # Away step: if s_idx is in active set, also move away from a random active element
        if s_idx in active_set:
            # Move away from a random element in active set (not s_idx)
            others = [x for x in active_set if x != s_idx]
            if others:
                away_idx = others[it % len(others)]
                t_away = 0.5 * t
                a_new[s_idx] += t_away
                a_new[away_idx] -= t_away
                active_set.remove(away_idx)

        if s_idx not in active_set:
            active_set.append(s_idx)
            if len(active_set) > 20:
                active_set.pop(0)

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        avg_a += a
        count += 1
        it += 1

        # Periodic evaluation
        if count % 200 == 0:
            avg = avg_a / count
            score_avg = fast_autocorr(avg).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg.copy()

            score_curr = fast_autocorr(a).max()
            if score_curr < best_score:
                best_score = score_curr
                best_a = a.copy()

            if verbose and count % 2000 == 0:
                print(f"    FW it={it} best={best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    # Final check
    if count > 0:
        avg = avg_a / count
        score_avg = fast_autocorr(avg).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg.copy()

    return best_a, best_score


def annealed_peak_search(a0, time_limit, seed=42, verbose=True):
    """Local search with simulated annealing and diverse targeted moves."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

        a = a * norm / s0

    t0 = time.monotonic()
    it = 0
    T_start = 0.05
    T_end = 0.001
    n_peaks = 5

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        T = T_start * (T_end / T_start) ** (elapsed / time_limit)

        b = fast_autocorr(a)
        peak = b.max()

        # Choose move type
        move_type = rng.integers(0, 5)
        a_new = a.copy()

        if move_type == 0:
            # Move mass from high-contribution to low-contribution position
            top_ks = np.argsort(b)[::-1][:n_peaks]
            k_star = top_ks[it % n_peaks]
            contrib = np.zeros(n)
            for m in range(n):
                idx1 = k_star - m
                idx2 = m - k_star
                if 0 <= idx1 < n:
                    contrib[m] += a[idx1]
                if 0 <= idx2 < n:
                    contrib[m] += a[idx2]
            i = contrib.argmax()
            # Find low-contribution target
            low_mask = contrib < contrib.mean()
            if not low_mask.any():
                low_mask = np.arange(n) != i
            candidates = np.where(low_mask)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            j = candidates[rng.integers(len(candidates))]
            if i != j:
                frac = rng.uniform(0.02, 0.15)
                amt = a[i] * frac
                a_new[i] -= amt
                a_new[j] += amt

        elif move_type == 1:
            # Random 2-swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                amt = min(a[i], a[j]) * rng.uniform(0.05, 0.4)
                a_new[i] -= amt
                a_new[j] += amt

        elif move_type == 2:
            # Multiplicative perturbation (smooth)
            delta = rng.normal(0, 0.05, n)
            a_new = a * np.exp(delta)

        elif move_type == 3:
            # Targeted: reduce peak by adjusting top contributors
            k_star = b.argmax()
            m_start = max(0, k_star - n + 1)
            m_end = min(n - 1, k_star)
            idx = np.arange(m_start, m_end + 1)
            contrib_vals = np.zeros(len(idx))
            for ii in range(len(idx)):
                m = idx[ii]
                c = 0.0
                idx1 = k_star - m
                idx2 = m - k_star
                if 0 <= idx1 < n:
                    c += a[idx1] * a[m]
                if 0 <= idx2 < n:
                    c += a[m] * a[idx2]
                contrib_vals[ii] = c
            # Reduce the top contributor
            top_pos = idx[contrib_vals.argmax()]
            i = top_pos
            j = rng.integers(n)
            if i != j:
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a_new[i] -= amt
                a_new[j] += amt
        else:
            # Block swap: swap values between two random blocks
            block_size = max(1, n // 8)
            bi = rng.integers(0, n // block_size)
            bj = rng.integers(0, n // block_size)
            if bi != bj:
                si, ei = bi * block_size, (bi + 1) * block_size
                sj, ej = bj * block_size, (bj + 1) * block_size
                a_new[si:ei], a_new[sj:ej] = a_new[sj:ej].copy(), a_new[si:ei].copy()

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()
        delta_score = score_new - peak

        # Accept if improvement or with annealing probability
        if delta_score < 0 or (T > 1e-10 and rng.random() < np.exp(-delta_score / T)):
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()

        it += 1
        if verbose and it % 2000 == 0:
            print(f"    Anneal it={it} T={T:.6f} best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, seed)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ========== Phase 1: LSE-Nesterov at base length ==========
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t1, T_start=12.0, T_end=0.05, n_stages=16, verbose=True)
        do_try_update(a_out, f"lse_n{n_base}")

    # ========== Phase 2: Frank-Wolfe at base length ==========
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Frank-Wolfe n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        n_best = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = frank_wolfe(a_work, t2, verbose=True)
        do_try_update(a_out, f"fw_n{n_best}")

    # ========== Phase 3: Annealed peak search ==========
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: AnnealedSearch n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        n_best = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = annealed_peak_search(a_work, t3, seed=777, verbose=True)
        do_try_update(a_out, f"anneal_n{n_best}")

    # ========== Phase 4: Multi-length exploration ==========
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    # Try a few nearby lengths
    lengths_to_try = [n_best + 50, n_best + 100, n_best - 50]
    lengths_to_try = [l for l in lengths_to_try if l > 100]

    for li, nl in enumerate(lengths_to_try[:2]):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t4 = min(remaining * 0.15, 180.0)
        if t4 < 30:
            break
        print(f"\n=== Phase 4.{li}: LSE n={nl} (budget={t4:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, nl)
        a_work = init_long * np.sqrt(2.0 * nl) / init_long.sum()
        a_out, _ = nesterov_lse(a_work, t4, T_start=10.0, T_end=0.1, n_stages=10, verbose=False)
        do_try_update(a_out, f"lse_n{nl}")

        # Follow with annealed search
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t4b = min(remaining * 0.08, 100.0)
            if t4b > 20:
                a_work2 = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
                a_out2, _ = annealed_peak_search(a_work2, t4b, seed=999 + li, verbose=False)
                do_try_update(a_out2, f"anneal_n{len(best_seq)}_r{li}")

    # ========== Phase 5: Coordinate polish ==========
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: CoordPolish n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 250.0)
    if t5 > 20:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_polish(a_work, t5, seed=12345, verbose=True)
        do_try_update(a_out, f"coord_n{n_best}")

    # ========== Phase 6: Final LSE polish with very low T ==========
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE polish (low T) n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.30, 200.0)
    if t6 > 20:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t6, T_start=3.0, T_end=0.03, n_stages=10, verbose=True)
        do_try_update(a_out, "lse_polish_lowT")

    # ========== Phase 7: Final annealed search ==========
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t7 = min(remaining * 0.5, 150.0)
        if t7 > 20:
            print(f"\n=== Phase 7: Final Anneal n={n_best} (budget={t7:.0f}s) ===")
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = annealed_peak_search(a_work, t7, seed=55555, verbose=True)
            do_try_update(a_out, "anneal_final")

    # ========== Final ==========
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
