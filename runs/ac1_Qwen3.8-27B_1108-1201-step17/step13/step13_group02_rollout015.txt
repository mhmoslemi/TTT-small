def interpolate_seq(f, new_n):

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based fast autocorrelation evaluation
2. LogSumExp-smoothed gradient descent with adaptive step size
3. Simulated annealing with multi-destination peak shaving
4. Multi-length exploration with interpolation
5. Iterative refinement with restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Fast local evaluation matching evaluate_sequence logic."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = np.minimum(a, 1000.0)
    n = len(a)
    if n == 0:
        return float('inf')
    b = fast_autocorr(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def lse_weights(b, temperature):
    """Compute softmax weights for LSE approximation of max."""
    b_max = b.max()
    if b_max <= 0:
        return np.ones_like(b) / len(b)
    w = np.exp((b - b_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.ones_like(b) / len(b)
    return w / w_sum


def lse_gradient(a, w):
    """Gradient of sum_k w[k] * (a*a)[k] w.r.t. a using FFT."""
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def sgd_optimize(a0, time_budget, verbose=True):
    """LogSumExp smoothed gradient descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = eval_local(a)
    t0 = time.monotonic()

    b_init = fast_autocorr(a)
    T_start = max(0.005, best_score / n * 0.5)
    T_end = max(0.0005, best_score / n * 0.02)

    step_count = 0
    stall = 0

    while time.monotonic() - t0 < time_budget:
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = lse_weights(b, T)
        grad = lse_gradient(a, w)

        # Adaptive backtracking step size
        lr = 4.0 / (n * max(T, 1e-8))
        improved = False
        for _ in range(15):
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break

            score_new = eval_local(a_new)
            if score_new < best_score:
                a = a_new
                best_score = score_new
                best_a = a.copy()
                stall = 0
                improved = True
                break
            elif score_new < a_score + lr * np.abs(grad).max() * 0.1:
                a = a_new
                improved = True
                break
            lr *= 0.5

        if not improved and stall > 0:
            stall += 1
            if stall > 200:
                # Perturb to escape local minimum
                rng_p = np.random.default_rng(step_count * 31 + 7)
                a = np.maximum(a + rng_p.normal(0, 0.01 * a.mean(), n), 0)
                a = a / a.sum()
                best_score = max(best_score, eval_local(a))
                stall = 0

        step_count += 1
        if verbose and step_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD step={step_count} ({elapsed:.0f}s): T={T:.6f}, best={best_score:.6f}")

    return best_a, best_score


def peak_shave_anneal(a0, time_budget, seed=42, verbose=True):
    """Simulated annealing with peak-shaving moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)

    n_stall = 0

    while time.monotonic() - t0 < time_budget:
        progress = min(1.0, (time.monotonic() - t0) / time_budget)

        b = fast_autocorr(a)
        peak_idx = int(np.argmax(b))

        # Find contributing pairs to the peak

        if je < js:
            js, je = 0, n - 1
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]

        if contrib.max() < 1e-20:
            # Random move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                n_stall += 1
                continue
            amt = a[i] * rng.uniform(0.05, 0.2)

                n_stall += 1
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:

                a = a_trial
                best_score = score_t

                n_improve += 1
                n_stall = 0

                a = a_trial
                n_stall = 0
            else:
                n_stall += 1

        amounts = amt_total * wts

        a_trial[src] -= amt_total

        a_trial = np.maximum(a_trial, 0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            n_stall += 1
            continue
        a_trial = a_trial / s_t

        score_t = eval_local(a_trial)
        if score_t < best_score:
            a = a_trial
            best_score = score_t
            best_a = a.copy()
            n_improve += 1
            n_stall = 0
        elif rng.random() < np.exp(-(score_t - best_score) / max(T, 1e-10)):

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        a_plus = np.maximum(a + sigma * noise, 0)

        if s_plus < 1e-15:

        score_plus = eval_local(a_plus)
        score_minus = best_score  # approximate

            sigma = min(0.05, sigma * 1.1)
