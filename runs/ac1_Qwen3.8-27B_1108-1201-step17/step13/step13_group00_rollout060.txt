1. LP best-response with Nesterov acceleration and adaptive constraint screening
2. Improved pair coordinate descent with golden-section search and active-set guided pair selection

import numpy as np

from scipy.fft import rfft, irfft


def fast_selfconv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the score 2*n*max(conv)/(sum)^2 directly."""
    a = np.asarray(a, dtype=np.float64)
    if a.sum() < 0.01:
        return float('inf')
    conv = fast_selfconv(a)
    return float(2.0 * len(a) * conv.max() / (a.sum() ** 2))

def get_lp_solver():
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    print("  LP solver (fallback): ECOS")
    return cp.ECOS


def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp(a, M, solver, active_set=None):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k (or active k), b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    if active_set is not None and len(active_set) < size:
        constraints = [T[active_set, :] @ b_var <= M]
    else:
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    return b_val, active_set

def fast_selfconv_actual(a, b):
    """Compute conv(a, b) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_gain_vec(a, i, j):
    """Compute first-order gain vector g where conv change ≈ delta * g for transfer i->j."""
    n = len(a)
    size = 2 * n - 1
    g = np.zeros(size)
    # +a[k-j] (contribution of position j as first factor)
    g[j:j + n] += a
    # +a[j-k] (contribution of position j as second factor)
    k_lo = max(0, j - n + 1)
    k_hi = min(j, size - 1)
    if k_hi >= k_lo:
        g[k_lo:k_hi + 1] += a[j - k_hi:j - k_lo + 1][::-1]
    # -a[k-i] (removing position i as first factor)
    g[i:i + n] -= a
    # -a[i-k] (removing position i as second factor)
    k_lo2 = max(0, i - n + 1)
    k_hi2 = min(i, size - 1)
    if k_hi2 >= k_lo2:
        g[k_lo2:k_hi2 + 1] -= a[i - k_hi2:i - k_lo2 + 1][::-1]
    return g


def eval_conv_max(conv_base, g, i, j, delta, size, n):
    """Evaluate max conv after transferring delta from i to j (with 2nd order correction)."""
    val = conv_base + delta * g
    if 2 * j < size:
        val[2 * j] += delta * delta
    if 2 * i < size:
        val[2 * i] += delta * delta
    if i + j < size:
        val[i + j] -= 2.0 * delta * delta
    return val.max()

            if opt_d < 1e-12:
                continue

            opt_val = eval_conv_max(conv, g, i, j, opt_d, size, n)

            if opt_val < cur_peak - 1e-12:
                # Apply the move
                a[i] -= opt_d
                a[j] += opt_d
                s = a.sum()
                if s < 1e-15:
                    a[i] += opt_d
                    a[j] -= opt_d
                    continue
                a = a * norm / s
                conv = fast_selfconv(a)
                new_peak = conv.max()
                if new_peak < best_score - 1e-12:
                    best_score = new_peak
                    best_a = a.copy()
                pass_improved = True
                no_improve = 0
                cur_peak = new_peak
            else:
                # Try reverse direction (j -> i)
                g_rev = -g
                test_rev = ai * 0.3
                if a[j] > norm * 1e-10:
                    rv_lo = 0.0
                    rv_hi = a[j] * 0.5  # limit to half
                    for _ in range(15):
                        d1 = rv_hi - golden_r * (rv_hi - rv_lo)
                        d2 = rv_lo + golden_r * (rv_hi - rv_lo)
                        if d1 <= rv_lo or d2 >= rv_hi:
                            break
                        v1 = eval_conv_max(conv, g_rev, i, j, d1, size, n)
                        v2 = eval_conv_max(conv, g_rev, i, j, d2, size, n)
                        if v1 < v2:
                            rv_hi = d2
                        else:
                            rv_lo = d1
                    rev_d = (rv_lo + rv_hi) * 0.5
                    if rev_d > 1e-12:
                        rev_val = eval_conv_max(conv, g_rev, i, j, rev_d, size, n)
                        if rev_val < cur_peak - 1e-12:
                            a[j] -= rev_d
                            a[i] += rev_d
                            a = a * norm / a.sum()
                            conv = fast_selfconv(a)
                            new_peak = conv.max()
                            if new_peak < best_score - 1e-12:
                                best_score = new_peak
                                best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                            cur_peak = new_peak

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}, tries/pass={n_tries}")

    return best_a, best_score


def systematic_pcd(a0, time_limit, rng, verbose=True):
    """Systematic sweep PCD: go through pairs in structured order."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_selfconv(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    golden_r = (np.sqrt(5.0) - 1.0) / 2.0

    # Sort indices by value (high to low)
    # Try transferring from high-value positions to low-value positions
    sorted_idx = np.argsort(a)[::-1]
    window = min(n, 300)  # only consider top/bottom 300 for efficiency

    n_sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_sweeps += 1
        sweep_improved = False
        cur_peak = conv.max()

        # Shuffle within groups for diversity
        hi_group = sorted_idx[:window].copy()
        rng.shuffle(hi_group)
        lo_group = np.argsort(a)[:window].copy()
        rng.shuffle(lo_group)

        for i_idx in hi_group:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(i_idx)
            if a[i] < norm * 1e-8:
                continue

            # Try a few random targets from the low group
            n_targets = min(5, window)
            for t_idx in range(n_targets):
                j = int(lo_group[t_idx % window])
                if i == j:
                    continue

                g = compute_gain_vec(a, i, j)
                ai = a[i]

                # Quick test
                test_d = ai * 0.2
                test_val = eval_conv_max(conv, g, i, j, test_d, size, n)
                if test_val >= cur_peak:
                    continue

                # Golden section
                lo = 0.0
                hi = ai
                for _ in range(18):
                    d1 = hi - golden_r * (hi - lo)
                    d2 = lo + golden_r * (hi - lo)
                    if d1 <= lo or d2 >= hi:
                        break
                    v1 = eval_conv_max(conv, g, i, j, d1, size, n)
                    v2 = eval_conv_max(conv, g, i, j, d2, size, n)
                    if v1 < v2:
                        hi = d2
                    else:
                        lo = d1
                opt_d = (lo + hi) * 0.5
                if opt_d < 1e-12:
                    continue

                opt_val = eval_conv_max(conv, g, i, j, opt_d, size, n)
                if opt_val < cur_peak - 1e-12:
                    a[i] -= opt_d
                    a[j] += opt_d
                    a = a * norm / a.sum()
                    conv = fast_selfconv(a)
                    cur_peak = conv.max()
                    if cur_peak < best_score - 1e-12:
                        best_score = cur_peak
                        best_a = a.copy()
                    sweep_improved = True
                    no_improve = 0

        if not sweep_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        # Re-sort periodically
        if n_sweeps % 10 == 0:
            sorted_idx = np.argsort(a)[::-1]

        if verbose and n_sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SysPCD sweep={n_sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_perturbation_search(a0, time_limit, rng, verbose=True):
    """Very fine random perturbation search (local search with tiny steps)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_selfconv(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    step_scale = norm / n * 0.1  # very small steps
    n_accepted = 0
    n_rejected = 0
    temperature = step_scale * 5.0
    n_total = 0

    while time.monotonic() - t0 < time_limit:
        # Decay temperature
        progress = (time.monotonic() - t0) / time_limit
        current_temp = temperature * (1.0 - 0.8 * progress)

        # Pick two random positions
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue

        # Small random transfer
        delta = rng.uniform(-1, 1) * step_scale * (1.0 + 0.5 * rng.random())
        delta = max(-a[i], min(a[j] * 0.5, delta))  # ensure non-negativity
        if abs(delta) < 1e-15:
            continue

        # Evaluate
        g = compute_gain_vec(a, i, j)
        # For negative delta, swap i and j roles
        if delta > 0:
            test_val = eval_conv_max(conv, g, i, j, delta, size, n)
        else:
            test_val = eval_conv_max(conv, -g, j, i, -delta, size, n)

        cur_peak = conv.max()
        if test_val < cur_peak - 1e-13:
            # Accept (improvement)
            if delta > 0:
                a[i] -= delta
                a[j] += delta
            else:
                a[j] -= (-delta)
                a[i] += (-delta)
            a = a * norm / a.sum()
            conv = fast_selfconv(a)
            new_peak = conv.max()
            if new_peak < best_score - 1e-13:
                best_score = new_peak
                best_a = a.copy()
            n_accepted += 1
        elif rng.random() < np.exp(-(test_val - cur_peak) / (current_temp + 1e-15)):
            # Accept (worse, simulated annealing)
            if delta > 0:
                a[i] -= delta
                a[j] += delta
            else:
                a[j] -= (-delta)
                a[i] += (-delta)
            a = a * norm / a.sum()
            conv = fast_selfconv(a)
            n_accepted += 1
        else:
            n_rejected += 1

        n_total += 1
        if verbose and n_total % 50000 == 0:
            elapsed = time.monotonic() - t0
            acc_rate = n_accepted / max(1, n_accepted + n_rejected)
            print(f"    Fine t={elapsed:.0f}s: best={best_score:.8f}, acc={acc_rate:.3f}, steps={step_scale:.6f}")
