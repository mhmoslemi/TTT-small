    t0 = time.monotonic()
    
    for stage_i, T in enumerate(T_schedule):
        # Lipschitz constant estimate
        L = max(n / (T * 0.5), 2.0 * n)
        t_fista = 1
        staging = a.copy()
        
        while time.monotonic() - t0 < time_limit:
            grad = _lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)
            
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()
            
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            
            if t_fista % 1000 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  LSE-FISTA stage={stage_i} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def _build_lp_matrix(a):
    """Build constraint matrix C for LP: C[k,j] = a[k-j] if valid."""
    n = len(a)
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    C = _build_lp_matrix(a)
    constraints = [C @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    return b_val
