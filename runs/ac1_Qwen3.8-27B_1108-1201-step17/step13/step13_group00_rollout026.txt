def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size_ac = 1 << (size - 1).bit_length()  # For autocorrelation
    fft_size_grad = 1 << (3 * n - 3).bit_length()  # For gradient
    
    # Temperature schedule
    n_stages = 15
    temps = np.geomspace(100.0, 0.05, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Learning rate: start larger for larger T
        base_lr = 0.5 / (T + 1.0) * norm
        
        # Number of steps: more for later stages (smaller T)
        n_steps = max(20, int(remaining * 2))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute autocorrelation
            A_ac = rfft(a, fft_size_ac)
            conv = irfft(A_ac * A_ac, fft_size_ac)[:size]
            max_conv = conv.max()
            
            # LSE weights
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()
            
            # Compute gradient
            w_padded = np.zeros(fft_size_grad)
            w_padded[:size] = w
            a_padded = np.zeros(fft_size_grad)
            a_padded[:n] = a
            
            A_g = rfft(a_padded, fft_size_grad)
            W_g = rfft(w_padded, fft_size_grad)
            
            # h1[i] = cross_corr: ifft(conj(A) * W)
            h1 = irfft(np.conj(A_g) * W_g, fft_size_grad)[:n]
            # h2[i] = conv: ifft(A * W)
            h2 = irfft(A_g * W_g, fft_size_grad)[:n]
            
            grad = 2.0 * (h1 + h2)
            
            # Learning rate with decay
            lr = base_lr / (1.0 + 0.005 * it)
            
            # Gradient step with projection
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new
            
            # Check improvement
            if it % 10 == 0 or it == n_steps - 1:
                score = conv.max()  # Use current conv (slightly stale but OK)
                # Actually compute fresh
                A_ac2 = rfft(a, fft_size_ac)
                conv2 = irfft(A_ac2 * A_ac2, fft_size_ac)[:size]
                score = conv2.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

**Modified spectral method:**
1. Compute a, get spectrum A(ω) = FFT(a)
2. Compute autocorrelation c = IFFT(|A|^2)
3. Identify the lag k* with maximum c
4. In the spectrum, the "excess" at k* corresponds to a specific phase relationship in A(ω)
5. To reduce c[k*], we need to modify the phase of A(ω) such that the contribution to c[k*] is reduced
6. c[k*] = (1/n) * sum_ω |A(ω)|^2 * exp(2πiωk*/n) (for cyclic correlation)
7. To reduce this, we can adjust the phase of A(ω) at frequencies where the exponential has positive real part

This is essentially the "iterative amplitude and phase adjustment" (IAPA) algorithm.

Actually, for our purpose, a simpler approach works:

**Direct peak reduction:**
At each iteration:
1. Find k* = argmax c[k]
2. Compute the "blame" of each a[i] for the peak at k*:

3. Reduce a[i] proportional to blame[i] and distribute the mass to positions with low blame
4. This is essentially a subgradient step

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based LSE gradient descent with temperature annealing (fast global optimizer)
2. Spectral peak-flattening via iterative amplitude adjustment
3. LP Frank-Wolfe with Nesterov acceleration (principled global method)
4. Vectorized pair coordinate descent (fast local refinement)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def normalize(a, target_sum=None):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * (target_sum if target_sum is not None else 1.0) / n
    if target_sum is None:
        target_sum = s
    return a * target_sum / s


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size_ac = 1 << (size - 1).bit_length()
    fft_size_grad = 1 << (3 * n - 3).bit_length()

    n_stages = 20
    temps = np.geomspace(80.0, 0.03, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        base_lr = 0.8 / (T + 0.5) * norm
        n_steps = max(30, int(remaining * 3))

        w_padded = np.zeros(fft_size_grad)
        w_padded[:size] = 0.0
        a_padded = np.zeros(fft_size_grad)
        a_padded[:n] = a

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            A_ac = rfft(a, fft_size_ac)
            conv = irfft(A_ac * A_ac, fft_size_ac)[:size]
            max_conv = conv.max()

            exp_conv = np.exp((conv - max_conv) / T)
            w_sum = exp_conv.sum()
            if w_sum < 1e-300:
                break
            w = exp_conv / w_sum

            w_padded[:size] = w
            a_padded[:n] = a

            A_g = rfft(a_padded, fft_size_grad)
            W_g = rfft(w_padded, fft_size_grad)

            h1 = irfft(np.conj(A_g) * W_g, fft_size_grad)[:n]
            h2 = irfft(A_g * W_g, fft_size_grad)[:n]

            grad = 2.0 * (h1 + h2)

            lr = base_lr / (1.0 + 0.003 * it)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            if it % 20 == 0 or it == n_steps - 1:
                A_ac2 = rfft(a, fft_size_ac)
                conv2 = irfft(A_ac2 * A_ac2, fft_size_ac)[:size]
                score = conv2.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_flatten(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size_ac = 1 << (size - 1).bit_length()
    fft_size_spec = 1 << (2 * n - 2).bit_length()

    n_iters = 0
    alpha = 0.3

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 100 == 0:
            alpha = max(0.05, 0.3 * (1.0 - it / 5000.0))

        A = rfft(a, fft_size_spec)
        power = np.abs(A) ** 2
        avg_power = power.mean()

        power_new = power + alpha * (avg_power - power)
        power_new = np.maximum(power_new, 1e-20)
        amp_new = np.sqrt(power_new)
        phase = A / np.abs(A + 1e-30)
        A_new = amp_new * phase

        a_new = irfft(A_new, fft_size_spec)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * norm / s
        a = a_new
        n_iters += 1

        if it % 50 == 0:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if it % 2000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    Spec iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    conv_current = T @ a
    active_set = np.where(conv_current > M * 0.8)[0]
    if len(active_set) < 5:
        active_set = np.where(conv_current > M * 0.5)[0]
    if len(active_set) < 3:
        active_set = np.argsort(conv_current)[-10:]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T[active_set, :] @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    full_conv = T @ b_val
    violated = np.where(full_conv > M * 1.001)[0]
    if len(violated) > 0 and len(active_set) < size * 0.9:
        new_active = np.union1d(active_set, violated)
        b_var2 = cp.Variable(n, nonneg=True)
        constraints2 = [T[new_active, :] @ b_var2 <= M]
        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
        try:
            prob2.solve(solver=solver, verbose=False)
        except Exception:
            return b_val
        if prob2.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0

    y = a.copy()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_coordinate_descent(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize(a0, norm)

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            for attempt in range(2):
                j = int(rng.integers(n))
                if j == i or a[j] >= a[i]:
                    continue

                d = np.zeros(size)
                lo1 = max(0, i)
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    d[lo1:hi1] += a[max(0, lo1 - i):max(0, hi1 - i)]
                lo2 = max(0, n - 1 - i)
                hi2 = min(size, n + i)
                if hi2 > lo2:
                    idx2 = np.arange(lo2, hi2)
                    src2 = n - 1 - idx2
                    valid = (src2 >= 0) & (src2 < n)
                    d[lo2:hi2][valid] += a[src2[valid]]

                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    d[lo3:hi3] -= a[max(0, lo3 - j):max(0, hi3 - j)]
                lo4 = max(0, n - 1 - j)
                hi4 = min(size, n + j)
                if hi4 > lo4:
                    idx4 = np.arange(lo4, hi4)
                    src4 = n - 1 - idx4
                    valid4 = (src4 >= 0) & (src4 < n)
                    d[lo4:hi4][valid4] -= a[src4[valid4]]

                ai = a[i]
                n_samples = 15
                deltas = np.linspace(0.0, ai * 0.9, n_samples)
                best_d = 0.0
                best_val = conv.max()

                new_convs = conv[None, :] + deltas[:, None] * d[None, :]
                if 2 * i < size:
                    mask_i = (np.arange(size) == 2 * i)
                    new_convs[:, mask_i] = conv[2 * i] - 2 * ai * deltas + deltas ** 2
                if 2 * j < size:
                    mask_j = (np.arange(size) == 2 * j)
                    new_convs[:, mask_j] = conv[2 * j] + 2 * a[j] * deltas + deltas ** 2
                max_vals = new_convs.max(axis=1)

                valid_mask = max_vals < best_val - 1e-11
                if np.any(valid_mask):
                    best_idx = np.argmax(np.where(valid_mask, -max_vals, -np.inf))
                    best_d = deltas[best_idx]
                    best_val = max_vals[best_idx]

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = a * norm / a.sum()
                    A = rfft(a, fft_size)
                    conv = irfft(A * A, fft_size)[:size]
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        A = rfft(a, fft_size)
                        conv = irfft(A * A, fft_size)[:size]
                    break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.5) ** 2) * (1.0 + 0.3 * np.cos(20.0 * np.pi * x))
        a = np.maximum(a, 0.0)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.ones(n) / n
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lse_gradient_descent(a_norm, t5 * 0.5, verbose=True)
        do_try_update(a_out, f"lse_long_n{n2}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out2, _ = pair_coordinate_descent(a_out, min(remaining * 0.3, 60.0), rng, verbose=False)
            do_try_update(a_out2, f"pcd_long_n{n2}")

    # Phase 6: Try shorter length (sometimes shorter is better)
    n_short = max(800, n_base - 400)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Shorter n={n_short} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 10:
        init_short = interpolate_sequence(best_seq, n_short)
        a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
        a_out, _ = lse_gradient_descent(a_norm, t6 * 0.6, verbose=False)
        do_try_update(a_out, f"lse_short_n{n_short}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            a_out2, _ = pair_coordinate_descent(a_out, min(remaining * 0.3, 50.0), rng, verbose=False)
            do_try_update(a_out2, f"pcd_short_n{n_short}")

    # Phase 7: Final LSE + pair CD polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_curr} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        t7a = min(remaining * 0.4, 100.0)
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t7a, verbose=True)
        do_try_update(a_out, f"lse_final_n{n_curr}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    if remaining > 10:
        t7b = min(remaining * 0.6, 150.0)
        a_norm = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t7b, rng, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n_curr}")

    # Phase 8: Try a fresh structured init at the best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_curr = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Fresh init n={n_curr} (budget={remaining:.0f}s) ===")
        for trial in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 10:
                break
            a_fresh = structured_init(n_curr, rng)
            a_fresh = a_fresh * np.sqrt(2.0 * n_curr) / a_fresh.sum()
            t_fresh = min(remaining * 0.3, 60.0)
            a_out, _ = lse_gradient_descent(a_fresh, t_fresh, verbose=False)
            do_try_update(a_out, f"fresh{trial}_n{n_curr}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
