def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)

def project_simplex(v, target_sum=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def compute_grad(a, w, fft_size):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = 2 * sum_k w[k] * a[k-i] (valid indices only)
    """
    n = len(a)
    a_rev = a[::-1].copy()
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def spectral_shaping(a, n_iter=400, alpha=0.35):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (non_dc + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def make_initializations(n, seed=42):
    """Generate diverse non-symmetric initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    # Power-law shapes (non-symmetric, peaked at one end)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = (1.0 - x + 0.01) ** p
        a = a / a.sum()
        inits.append(a)

    # Exponential decay
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.exp(-alpha * x)
        a = a / a.sum()
        inits.append(a)

    # Power-law increasing
    for p in [0.5, 1.0, 2.0]:
        a = (x + 0.01) ** p
        a = a / a.sum()
        inits.append(a)

    # Skewed Gaussian
    for center in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            a = np.exp(-((x - center) / width) ** 2)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0:
                a = a / s
                inits.append(a)

    # Linear ramp
    a = 1.0 - 0.5 * x
    a = a / a.sum()
    inits.append(a)

    # Quadratic (peaked at edge)
    a = (1 - x) ** 2 + 0.01
    a = a / a.sum()
    inits.append(a)

    return inits


def mirror_descent_optimize(a0, time_limit, n_stages=40, lr_base=0.05, verbose=True):
    """
    Mirror descent (log-domain Adam) optimization.
    Minimizes max(conv(a,a)) using the zero-sum game formulation.
    Uses softmax peak weighting with adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    a = a / s

    z = np.log(a)
    best_a = a.copy()
    best_score = internal_score(a)

    fft_size = 1 << (3 * n - 2).bit_length()
    t0 = time.monotonic()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_adam = 0

    # Temperature stages (from hot to cold)
    T_start = max(10.0, n / 50.0)
    T_end = 0.3
    temps = np.geomspace(T_start, T_end, n_stages)

    stall_count = 0
    stall_threshold = 2000

    for si, T_peak in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Adaptive iterations per stage
        frac_remaining = (n_stages - si) / n_stages
        iters = max(10, int(remaining / frac_remaining * 0.9))
        iters = min(iters, 200)

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            b = fast_autocorr(a)

            # Softmax weights on peaks (temperature controls focus)
            Tb = T_peak * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient in a-space
            grad_a = compute_grad(a, w, fft_size)

            # Chain rule for z-space: d/dz = a * d/da
            grad_z = a * grad_a

            # Adam update
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            # Learning rate with decay
            progress = (si + it / max(1, iters)) / n_stages
            lr_t = lr_base * (1.0 - 0.5 * progress) / (1.0 + 0.02 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            # Periodic score check
            if it % 15 == 0:
                a_check = softmax_fn(z)
                score = internal_score(a_check)
                if score < best_score - 1e-10:
                    best_score = score
                    best_a = a_check.copy()
                    stall_count = 0
                else:
                    stall_count += 15

        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall_count = 0

        if verbose and si % 8 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD stage {si}/{n_stages} (T={T_peak:.2f}, t={elapsed:.0f}s): {best_score:.6f}")

        if stall_count > stall_threshold:
            # Perturb to escape local minimum
            z = z + 0.02 * np.random.default_rng(si).standard_normal(n)
            stall_count = 0

    return best_a, best_score


def subgradient_optimize(a0, time_limit, lr_base=0.08, verbose=True):
    """
    Direct subgradient descent on the simplex.
    Uses top-K peak weighting with softmax.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    t0 = time.monotonic()

    v = np.zeros(n)
    beta = 0.7
    t = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)
        max_b = b.max()

        # Softmax weights: focus on top peaks
        T_smooth = max(2.0, 30.0 * np.exp(-t / 5000.0))
        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        grad = compute_grad(a, w, fft_size)

        # Learning rate
        lr_t = lr_base / (1.0 + 0.005 * t)

        # Momentum
        v = beta * v + (1 - beta) * grad
        a_new = a - lr_t * v
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1

        if stall > 3000:
            break

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): {best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=123, verbose=True):
    """
    Peak-targeting coordinate descent.
    Moves mass from coordinates contributing to peaks to less critical ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top peaks
        top_k = np.argsort(b)[::-1][:20]
        max_peak_idx = top_k[0]

        # Identify coordinates most responsible for the top peak
        js = max(0, max_peak_idx - n + 1)
        je = min(n - 1, max_peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[max_peak_idx - idx]

        # Source: coordinate with highest contribution to peak
        src_local = np.argmax(contrib)
        src = idx[src_local]

        # Build set of "critical" coordinates (those in any top peak's support)
        critical = set()
        for k in top_k:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            critical.update(range(js2, je2 + 1))

        # Destination: non-critical coordinate with low value
        non_critical = [j for j in range(n) if j not in critical and j != src]
        if not non_critical:
            non_critical = [j for j in range(n) if j != src]
        non_critical.sort(key=lambda j: a[j])
        dst_pool = non_critical[:max(1, len(non_critical) // 2)]
        dst = dst_pool[rng.integers(len(dst_pool))]

        # Try several move fractions
        improved = False
        for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
            amt = a[src] * frac
            if amt < 1e-16:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < 0:
                continue
            a_trial = a_trial / a_trial.sum()
            score = internal_score(a_trial)
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            # Random perturbation
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 1000:
                break

        if verbose and stall % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordDesc (stall={stall}, t={elapsed:.0f}s): {best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_weights_game(a0, time_limit, n_iter_max=50000, verbose=True):
    """
    Multiplicative weights (fictitious play) for the zero-sum game.
    Player 1 (a) vs Player 2 (peak index k).
    Update: a_new[i] = a[i] * exp(-eta * g[i]) / sum(...)
    where g[i] = 2 * a_avg[k*-i] (subgradient from averaged opponent).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()

    # Empirical distribution over peaks
    y = np.ones(2 * n - 1) / (2 * n - 1)

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    fft_size = 1 << (3 * n - 2).bit_length()
    eta = 0.05
    avg_a = np.zeros(n)
    count = 0

    for t in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break

        # Player 1's best response to y: minimize y^T (a*a) = sum_k y[k](a*a)[k]
        # Gradient: g[i] = 2 * sum_k y[k] * a[k-i]
        grad = compute_grad(a, y, fft_size)

        # Multiplicative update
        log_a = np.log(a) - eta * grad
        a_new = np.exp(log_a - log_a.max())
        a_new = a_new / a_new.sum()

        # Player 2's best response to a: find max peak
        b = fast_autocorr(a_new)
        k_star = np.argmax(b)

        # Update empirical distribution (exponentially weighted)
        gamma = 0.99
        y = gamma * y + (1 - gamma) * np.zeros(2 * n - 1)
        y[k_star] += (1 - gamma)

        a = a_new
        avg_a += a
        count += 1

        # Periodic checks
        if (t + 1) % 50 == 0:
            score_a = internal_score(a)
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
            avg_check = avg_a / count
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if verbose and (t + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MW t={t+1} ({elapsed:.0f}s): best={best_score:.6f}, b_max={b.max():.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization targeting score < 1.5027.
    Uses mirror descent + multiplicative weights + coordinate descent.
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1700.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Load starting point from parent
    parent_a = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent_a = p / p.sum()
            best_seq = parent_a.copy()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if parent_a is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.exp(-1.5 * x)
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Try diverse initializations and pick the best ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Initialization (remaining={remaining:.0f}s) ===")
    n_try = n_base

    inits = make_initializations(n_try, seed=seed)
    if parent_a is not None and len(parent_a) == n_try:
        inits.append(parent_a.copy())

    best_init = best_seq.copy()
    best_init_score = best_score

    # Quick spectral shaping on top candidates
    for i, init_a in enumerate(inits):
        if time.monotonic() - t_global_start > 60:
            break
        # Quick spectral on first few
        if i < 6:
            shaped = spectral_shaping(init_a, n_iter=100, alpha=0.3)
            sc = internal_score(shaped)
            if sc < best_init_score:
                best_init_score = sc
                best_init = shaped.copy()
                print(f"  Init {i} shaped: {sc:.6f}")
        else:
            sc = internal_score(init_a)
            if sc < best_init_score:
                best_init_score = sc
                best_init = init_a.copy()

    if best_init_score < best_score:
        best_score = best_init_score
        best_seq = best_init.copy()
    print(f"  Best init: {best_score:.6f} (n={n_base})")

    # === Phase 2: Mirror descent (primary optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Mirror Descent n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.45, 750.0)
    if t2 > 30:
        a_out, sc_out = mirror_descent_optimize(best_seq, t2, n_stages=50, lr_base=0.04)
        sc = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in a_out])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [MD]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 2: best={best_score:.6f} (t={elapsed:.0f}s)")

    # === Phase 3: Multiplicative weights game ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Multiplicative Weights n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 20:
        a_out, sc_out = multiplicative_weights_game(best_seq, t3)
        sc = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in a_out])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [MW]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 3: best={best_score:.6f} (t={elapsed:.0f}s)")

    # === Phase 4: Subgradient optimization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Subgradient n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 15:
        a_out, _ = subgradient_optimize(best_seq, t4, lr_base=0.06)
        sc = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in a_out])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [SubGrad]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 4: best={best_score:.6f} (t={elapsed:.0f}s)")

    # === Phase 5: Mirror descent round 2 (from current best) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: MD Round 2 n={n_base} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        a_out, _ = mirror_descent_optimize(best_seq, t5, n_stages=30, lr_base=0.02)
        sc = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in a_out])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [MD2]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 5: best={best_score:.6f} (t={elapsed:.0f}s)")

    # === Phase 6: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Coord Descent n={n_base} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 10:
        a_out, _ = coordinate_descent(best_seq, t6, seed=777)
        sc = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in a_out])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [CD]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 6: best={best_score:.6f} (t={elapsed:.0f}s)")

    # === Phase 7: Try larger n ===
    n_long = min(int(n_base * 1.3), 2400)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Larger n={n_long} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 300.0)
    if t7 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral
        a_spec = spectral_shaping(init_long, n_iter=150, alpha=0.25)
        sc = internal_score(a_spec)
        sc_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in a_spec])
        if np.isfinite(sc_eval) and sc_eval < best_score:
            best_score = sc_eval
            best_seq = np.maximum(a_spec, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [spec_long]: {best_score:.6f}")

        # Mirror descent at longer length
        t7b = t7 * 0.7
        a_out, _ = mirror_descent_optimize(best_seq if len(best_seq) == n_long else init_long, t7b, n_stages=35, lr_base=0.03)
        sc = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in a_out])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [MD_long]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 7: best={best_score:.6f} (t={elapsed:.0f}s)")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final CD n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t8 = min(remaining * 0.4, 150.0)
    if t8 > 5:
        a_out, _ = coordinate_descent(best_seq, t8, seed=999)
        sc = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in a_out])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [CD_final]: {best_score:.6f}")

    # === Phase 9: One more MD round if time remains ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"\n=== Phase 9: Final MD (remaining={remaining:.0f}s) ===")
        t9 = min(remaining * 0.5, 100.0)
        a_out, _ = mirror_descent_optimize(best_seq, t9, n_stages=20, lr_base=0.015)
        sc = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in a_out])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [MD_final]: {best_score:.6f}")

    # Final verification and cleanup
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        fs = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
