_global_solver = None


def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient_correct(a, T):
    """
    Compute gradient of LSE surrogate: T*log(sum_k exp(b_k/T)) where b = autocorr(a).
    dL/da[m] = sum_k w[k] * (a[k-m] + a[m-k]) = r1[m] + r2[m]
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2


def lse_value(a, T):
    """Compute LSE surrogate value."""
    b = fast_autocorr(a)
    max_b = b.max()
    log_sum = max_b / T + np.log(np.sum(np.exp((b - max_b) / T)))
    return T * log_sum


def nesterov_lse(a0, time_limit, T_start=20.0, T_end=0.5, n_stages=12, verbose=True):
    """Nesterov-accelerated LSE gradient descent with geometric cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(30, int(remaining * 3.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            if it == 0:
                grad = lse_gradient_correct(y, T)
                y_new = y - (1.0 / (it + 3.0)) * grad
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s_e = y_extrap.sum()
                if s_e > 1e-15:
                    y_extrap = y_extrap * norm / s_e
                grad = lse_gradient_correct(y_extrap, T)
                y_new = y_extrap - (1.0 / (it + 3.0)) * grad

            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y = y_new
            it_count += 1

            if it_count % 50 == 0:
                score = fast_autocorr(y).max()
                if score < best_score:
                    best_score = score
                    best_a = y.copy()

    score = fast_autocorr(y).max()
    if score < best_score:
        best_score = score
        best_a = y.copy()

    return best_a, best_score


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_j <= M for all j, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    from scipy.sparse import lil_matrix
    T = lil_matrix((2 * n - 1, n))
    for j in range(2 * n - 1):
        for k in range(n):
            i = j - k
            if 0 <= i < n:
                T[j, k] = a[i]
    T = T.tocsr()

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):

    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Polyak-Ruppert average
        if t >= 10:
            avg_a += a
            n_avg += 1

        if n_iters % 20 == 0:
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

            if n_avg > 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    score_avg = fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final averaging check
    if n_avg > 50:
        avg_check = avg_a / n_avg
        s_avg = avg_check.sum()
        if s_avg > 1e-15:
            avg_check = avg_check * norm / s_avg
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

    return best_a, best_score

        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        sg = np.zeros(n)
        sg[idx] += a[k_star - idx]
        sg_rev = np.zeros(n)
        sg_rev[idx] += a[idx]
        # Full subgradient: sg[m] = a[m-k_star] + a[k_star-m] (where valid)
        sg = np.zeros(n)
        for m in range(n):
            val = 0.0
            i1 = m - k_star
            if 0 <= i1 < n:
                val += a[i1]
            i2 = k_star - m
            if 0 <= i2 < n:
                val += a[i2]
            sg[m] = val

        # Step size: Polyak-like
        alpha = 1.0 / (np.sqrt(it + 1) * (M_curr / norm + 1e-10))
        a_new = a - alpha * M_curr * sg / norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new

        if (it + 1) % 20 == 0:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            if verbose and (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Subgrad it={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_targeting_search(a0, time_limit, seed=42, verbose=True):
    """Aggressive peak-targeting local search with diverse move strategies."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_ks = np.argsort(b)[::-1][:8]

        improved = False
        for k_star in top_ks:
            # Find contributing pairs
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

            for src in top_contrib_idx:
                if a[src] < norm * 1e-8:
                    continue

                # Find positions far from all top peaks
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_ks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True

                candidates = np.where(~peak_mask)[0]
                if len(candidates) < 5:
                    candidates = np.where(np.arange(n) != src)[0]

                # Sort by value (prefer low-value destinations)
                candidates = candidates[np.argsort(a[candidates])[:max(5, len(candidates) // 4)]]

                for dst in candidates[:5]:
                    for frac in [0.001, 0.003, 0.007, 0.015, 0.025, 0.04, 0.06, 0.08]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-16:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation for escaping local optima
            if stall % 10 == 0:
                seed_try = rng.integers(0, 5)
                if seed_try < 2:
                    # Single random move
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > norm * 1e-8:
                        frac = rng.uniform(0.003, 0.05)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a_trial.copy()
                                stall = 0
                elif seed_try == 2:
                    # Random 2-move
                    i1 = rng.integers(n)
                    j1 = rng.integers(n)
                    i2 = rng.integers(n)
                    j2 = rng.integers(n)
                    valid = (i1 != j1) and (i2 != j2) and (i1 != i2)
                    if valid and a[i1] > norm * 1e-8 and a[i2] > norm * 1e-8:
                        a_trial = a.copy()
                        amt1 = a[i1] * rng.uniform(0.003, 0.03)
                        amt2 = a[i2] * rng.uniform(0.003, 0.03)
                        a_trial[i1] -= amt1
                        a_trial[j1] += amt1
                        a_trial[i2] -= amt2
                        a_trial[j2] += amt2
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a_trial.copy()
                                stall = 0
                else:
                    # Smoothing perturbation
                    i = rng.integers(1, n - 1)
                    a_trial = a.copy()
                    amt = a[i] * 0.01
                    a_trial[i] -= amt
                    a_trial[i-1] += amt / 2
                    a_trial[i+1] += amt / 2
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            stall = 0

            if stall > 300:
                if verbose:
                    print(f"    PeakSearch: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                break

        if verbose and it_count % 500 == 0:
            print(f"    PeakSearch it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, seed):
    """Generate a diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    else:
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def adaptive_lse(a0, time_limit, verbose=True):
    """LSE with adaptive temperature: starts high, cools based on progress."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    T = 50.0
    it_count = 0
    lse_prev = lse_value(y, T)
    patience = 0

    while time.monotonic() - t0 < time_limit and T > 0.05:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Adaptive iterations per temperature
        if T > 10:
            n_iters = max(20, int(remaining * 2.0))
        elif T > 2:
            n_iters = max(30, int(remaining * 2.5))
        else:
            n_iters = max(50, int(remaining * 3.0))

        lse_stage_start = lse_value(y, T)

        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            if it == 0:
                grad = lse_gradient_correct(y, T)
                y_new = y - (1.0 / (it + 3.0)) * grad
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s_e = y_extrap.sum()
                if s_e > 1e-15:
                    y_extrap = y_extrap * norm / s_e
                grad = lse_gradient_correct(y_extrap, T)
                y_new = y_extrap - (1.0 / (it + 3.0)) * grad

            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            y = y_new
            it_count += 1

            if it_count % 30 == 0:
                score = fast_autocorr(y).max()
                if score < best_score:
                    best_score = score
                    best_a = y.copy()

        # Check progress and adapt temperature
        lse_stage_end = lse_value(y, T)
        lse_improvement = lse_stage_start - lse_stage_end

        if lse_improvement / max(lse_stage_start, 1e-10) < 0.001:
            patience += 1
        else:
            patience = 0

        # Cool down
        if patience >= 2:
            T = max(T * 0.4, 0.05)
            patience = 0
        else:
            T = max(T * 0.7, 0.05)

        if verbose and it_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdaptiveLSE T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_start_optimize(best_init, n_target, time_budget, solver, verbose=True):
    """Run optimization from the best init at target length with time split."""
    t0 = time.monotonic()
    a_work = best_init.copy()
    n = len(a_work)
    norm = np.sqrt(2.0 * n)
    s_work = a_work.sum()
    if s_work < 1e-15:
        a_work = np.ones(n) * norm / n
    else:
        a_work = a_work * norm / s_work

    best_a = a_work.copy()
    best_score = fast_autocorr(a_work).max()

    # Phase A: Adaptive LSE (30% of budget)
    t_a = time_budget * 0.30
    if t_a > 10:
        if verbose:
            print(f"    [A] Adaptive LSE, budget={t_a:.0f}s")
        a_out, sc = adaptive_lse(a_work, t_a, verbose=True)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
        a_work = best_a.copy()
        np.multiply(a_work, norm / a_work.sum(), out=a_work)

    # Phase B: LP Nesterov (35% of budget)
    remaining = time_budget - (time.monotonic() - t0)
    t_b = min(remaining * 0.55, time_budget * 0.35)
    if t_b > 15:
        if verbose:
            print(f"    [B] LP Nesterov, budget={t_b:.0f}s")
        a_out, sc = nesterov_lp_optimize(a_work, t_b, solver, verbose=True)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
        a_work = best_a.copy()
        np.multiply(a_work, norm / a_work.sum(), out=a_work)

    # Phase C: Subgradient (10% of budget)
    remaining = time_budget - (time.monotonic() - t0)
    t_c = min(remaining * 0.30, time_budget * 0.10)
    if t_c > 5:
        if verbose:
            print(f"    [C] Subgradient, budget={t_c:.0f}s")
        a_out, sc = subgradient_optimize(a_work, t_c, verbose=True)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
        a_work = best_a.copy()
        np.multiply(a_work, norm / a_work.sum(), out=a_work)

    # Phase D: Peak targeting (25% of budget)
    remaining = time_budget - (time.monotonic() - t0)
    t_d = max(remaining * 0.8, time_budget * 0.1)
    if t_d > 5:
        if verbose:
            print(f"    [D] Peak targeting, budget={t_d:.0f}s")
        a_out, sc = peak_targeting_search(a_work, t_d, seed=777, verbose=True)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses multi-phase optimization with adaptive LSE,
    LP best-response with Nesterov acceleration, subgradient polish, and
    aggressive local search at multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()

    try:

            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Strategy: Try base length first, then explore neighbors ============

    # Determine which lengths to try
    n_primary = n_base
    lengths_to_try = [n_primary]
    # Add a couple of alternative lengths
    if n_primary > 100:
        lengths_to_try.append(n_primary + 200)
    lengths_to_try.append(n_primary + 400)

    # Allocate time: 60% to primary, 25% to first alt, 15% to second alt
    time_allocs = [0.55, 0.25, 0.20]

    current_best = best_seq.copy()

    for li, n_try in enumerate(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break

        t_alloc = remaining * time_allocs[li] if li < len(time_allocs) else remaining * 0.15
        t_alloc = max(t_alloc, 20.0)
        if li > 0:
            t_alloc = min(t_alloc, remaining * 0.4)

        if t_alloc < 20:
            break

        print(f"\n=== Length n={n_try}, allocated={t_alloc:.0f}s, remaining={remaining:.0f}s ===")

        # Interpolate from current best to target length
        if n_try != len(current_best):
            init_long = interpolate_sequence(current_best, n_try)
        else:
            init_long = current_best.copy()

        # Run multi-start optimization
        a_out, sc = multi_start_optimize(init_long, n_try, t_alloc, solver, verbose=True)

        do_try_update(a_out, f"multi_n{n_try}")

        # Update current best for interpolation
        if best_seq is not None and len(best_seq) != n_try:
            current_best = interpolate_sequence(best_seq, n_try)

            current_best = best_seq.copy()

    # ============ Final: Thorough peak targeting on the overall best ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10 and best_seq is not None:

            a_work = best_seq.copy()
            norm = np.sqrt(2.0 * n_best)
            a_work = a_work * norm / a_work.sum()
            a_out, sc = peak_targeting_search(a_work, t_final, seed=31415, verbose=True)

    # ============ Final: Quick LSE polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 5 and best_seq is not None:
        n_best = len(best_seq)
        t_polish = min(remaining * 0.5, 40.0)
        if t_polish > 3:

            a_work = best_seq.copy()
            norm = np.sqrt(2.0 * n_best)

            a_out, sc = nesterov_lse(a_work, t_polish, T_start=3.0, T_end=0.1, n_stages=5, verbose=False)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
