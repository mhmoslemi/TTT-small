def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def smooth_subgradient(a, ac, sharpness=8.0):
    """
    Compute a smooth approximation of the subgradient of max(ac).
    Uses softmax-like weighting over all lags.
    """
    n = len(a)
    ac_max = ac.max()
    # Softmax weights with temperature
    shifted = ac - ac_max
    weights = np.exp(sharpness * shifted)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros(len(ac))
        weights[np.argmax(ac)] = 1.0
    else:
        weights = weights / w_sum
    
    grad = np.zeros(n)
    # For each lag k, the subgradient contribution is 2*a[k-j] for j in valid range
    active_lags = np.where(weights > 1e-10 / len(ac))[0]
    for k in active_lags:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        j = np.arange(j_start, j_end)
        grad[j] += weights[k_int] * 2.0 * a[k_int - j]
    return grad


def hard_subgradient(a, ac, top_k=8):
    """Compute subgradient using top-k lags uniformly weighted."""
    n = len(a)
    top_idx = np.argsort(ac)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        if j_start >= j_end:
            continue
        j = np.arange(j_start, j_end)
        grad[j] += (1.0 / top_k) * 2.0 * a[k_int - j]
    return grad

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build convolution matrix T where (T*b)[k] = sum_j a[|k-j|]*b[j]
    # Use efficient construction: T[k,j] = a[k-j] if 0<=k-j<n else 0
    T_rows = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        row = np.zeros(n)
        j = np.arange(j_start, j_end)
        row[j] = a[k - j]
        T_rows.append(row)
    T = np.array(T_rows)
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

    rng = np.random.default_rng(12345)
    
    ac = fast_autocorr(a)
    M = ac.max()
    stall = 0
    max_stall = 50
    epoch = 0
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        epoch += 1
        ac = fast_autocorr(a)
        # Get top lags
        top_k = min(16, len(ac))
        top_idx = np.argsort(ac)[::-1][:top_k]
        
        improved = False
        # Try random pairs of coordinates
        n_pairs = min(2000, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if a[i] < 1e-15:
                continue
            
            # Try shifting some mass from i to j
            for frac in [0.02, 0.05, 0.1, 0.2, 0.5]:
                amt = a[i] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * c / s_t
                ac_trial = fast_autocorr(a_trial)
                M_trial = ac_trial.max()
                if M_trial < best_m * (1 - 1e-8):
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    ac = ac_trial
                    M = M_trial
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Perturb to escape local minimum
            n_perturb = min(50, n // 10)
            for _ in range(n_perturb):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c * 1e-8:
                    frac = rng.uniform(0.05, 0.2)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a * c / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                            break
        
        if verbose and epoch % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    CD epoch={epoch} ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m


def multi_start_gradient_polish(a0, time_limit, n_restarts=3, verbose=True):
    """
    Try multiple random perturbations of the input and optimize each.
    Return the best result.
    """
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n, dtype=np.float64) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    per_restart = time_limit / n_restarts
    
    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Create perturbation
        if r == 0:
            a_start = a.copy()
        else:
            a_start = a.copy()
            # Random perturbation
            sigma = 0.1 + 0.1 * r
            noise = rng.normal(0, sigma, n)
            a_start = a_start + noise * a.max() * 0.01
            a_start = np.maximum(a_start, 1e-30)
            s = a_start.sum()
            if s > 1e-15:
                a_start = a_start * c / s
        
        # Quick Nesterov mirror descent
        a_out, m_out = nesterov_mirror_descent(a_start, per_restart, 
                                                  sharpness=10 + 2*r, 
                                                  top_k=6 + 2*r,
                                                  verbose=False)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MultiStart restart={r} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length using smooth interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss


def spectral_init(n, mode='raised_cosine'):
    """Create spectrally-inspired initialization."""
    if mode == 'raised_cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif mode == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / n)
    elif mode == 'optimal_bandlimited':
        # Approximation of optimal non-negative bandlimited function
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 * (1 + 0.3 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a
