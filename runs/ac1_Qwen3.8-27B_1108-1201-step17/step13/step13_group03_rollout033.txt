"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-Nesterov with adaptive step sizes and averaged iterates
- Spectral (Gerchberg-Saxton-like) initialization in frequency domain
- Multi-restart with perturbation to escape local minima
- Active-set gradient descent for final polish
- Multi-length strategy (1600, 2000, 2400)
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print(f"  LP solver (fallback): CLARABEL")
    return _global_solver


def _fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n <= 1:
        return np.array([target_sum])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_arr) == 0:
        return np.full(n, target_sum / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    return out


def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.array(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, solver, verbose=True, avg_window=50):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % avg_window == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final check on average
    if n_avg > 0:
        avg_check = avg_a / n_avg
        s_avg = avg_check.sum()
        if s_avg > 1e-15:
            avg_check = avg_check * norm / s_avg
            score_avg = _fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

    return best_a, best_score


def _spectral_init(n, num_iters=200, target_flatness=0.5):
    """Initialize using frequency-domain iterations to flatten the convolution."""
    a = np.ones(n) / n
    fft_size = 1 << (3 * n - 2).bit_length()

    for it in range(num_iters):
        R = _fast_autocorr(a)
        R_max = R.max()
        R_min = R[R > 1e-10].min() if np.any(R > 1e-10) else R.min()

        A = rfft(a, fft_size)
        A_mag = np.abs(A)
        A_phase = np.angle(A)

        # We want |A|^2 to be flatter. The convolution is ifft(A^2), not ifft(|A|^2).
        # A^2 = |A|^2 * exp(2i*angle(A))
        # To flatten the convolution, we want A^2 to have flat magnitude... 
        # Actually, let's just try to equalize the autocorrelation directly.
        # Use: a_new = a * (1 + gamma * (target - R) / R_max) projected
        target = R_max * target_flatness
        ratio = np.ones(n)
        for i in range(n):
            # Influence of a[i] on R
            # R[k] depends on a[i] through terms a[i]*a[k-i] and a[k-i]*a[i]
            # Simplified: reduce a at positions that contribute to peaks
            pass

        # Simpler approach: multiplicative update
        # a_new[i] proportional to a[i] * (sum of R kde at positions involving i)^{-1/2}
        # For now, use a power iteration in frequency domain
        A_sq = A * A
        A_sq_mag = np.abs(A_sq)
        A_sq_mag = np.maximum(A_sq_mag, 1e-10)
        # Compress dynamic range
        A_sq_mag_new = A_sq_mag ** 0.5
        A_new = A_sq_mag_new * np.exp(1j * np.angle(A_sq))
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        if np.allclose(a_new, a, atol=1e-12):
            break
        a = a_new

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _active_set_descent(a0, time_limit, verbose=True):
    """Gradient descent on active set with FFT-accelerated gradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    lr = 0.5
    fft_size = 1 << (3 * n - 2).bit_length()

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        R = _fast_autocorr(a)
        R_max = R.max()
        tol = max(0.005 * R_max, 1e-10)
        S = np.where(R >= R_max - tol)[0]

        w = np.zeros(2 * n - 1)
        w[S] = 1.0

        # grad[j] = 2 * sum_{k in S} a[k-j] (with a[m]=0 for m<0 or m>=n)
        W_f = rfft(w, fft_size)
        A_f = rfft(a, fft_size)
        cross = irfft(W_f * np.conj(A_f), fft_size)
        grad = 2.0 * cross[:n]

        # Projected gradient step
        a_new = _proj_simplex(a - lr * grad, norm)

        score_new = _fast_autocorr(a_new).max()
        if score_new < best_score * (1 - 1e-10):
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 50.0)
        else:
            lr *= 0.6
            if lr < 1e-8:
                break

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AS-Grad it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.4f} |S|={len(S)}")

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _perturbed_restart(a_best, rng, n, norm_factor):
    """Create a perturbed version of the best solution."""
    a = np.asarray(a_best, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm_factor / n
    a = a * norm_factor / s

    # Random perturbation: multiply by random positive factors
    perturbation_scale = 0.1
    noise = 1.0 + perturbation_scale * (rng.random(n) - 0.5)
    a_pert = a * noise
    a_pert = np.maximum(a_pert, 0.0)
    s_p = a_pert.sum()
    if s_p < 1e-15:
        return a.copy()
    a_pert = a_pert * norm_factor / s_p
    return a_pert
