```python
"""
Optimization of step-function sequences minimizing
    2*n*max(np.convolve(a,a))/(sum(a))**2
using fast FFT-based multiplicative / mirror-descent updates,
inverse-gradient fixed-point updates, adaptive active-lag weighting,
multiple seeds, multi-length interpolation, and averaging.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft


def next_pow2(m):
    s = 1
    while s < m:
        s <<= 1
    return s


def normalize_p(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-300:
        p = np.ones_like(p, dtype=np.float64)
        return p / len(p)
    return p / s


def convolution_size(a, size):
    n = len(a)
    A = rfft(a, size)
    out = irfft(A * A, size)
    out = out[:2 * n - 1]
    return np.maximum(out, 0.0)


def score_of_norm(p):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = len(p)
    s = float(p.sum())
    if s <= 1e-300:
        return float("inf")
    size = next_pow2(2 * n - 1)
    q = convolution_size(p, size)
    M = float(q.max()) if q.size else 0.0
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (s * s))


def half_grad_size(p, w, size):
    n = len(p)
    pr = p[::-1]
    A = rfft(pr, size)
    B = rfft(w, size)
    out = irfft(A * B, size)
    h = out[n - 1:2 * n - 1]
    return np.maximum(h, 0.0)


def make_weights(q, mode="top_frac", alpha=20.0, fraction=0.02, beta=3.0):
    L = len(q)
    M = float(q.max())
    if M <= 0.0 or not np.isfinite(M):
        return np.ones(L, dtype=np.float64) / L

    if mode == "softmax":
        val = alpha * (q - M)
        val -= float(val.max())
        w = np.exp(val)
        s = float(w.sum())
        if s <= 1e-300 or not np.isfinite(s):
            w = np.ones(L, dtype=np.float64)
            s = float(L)
        return w / s

    k = int(round(fraction * L))
    if k <= 0:
        k = 1
    if k >= L:
        w = q.copy()
    else:
        if k > 0 and k < L:
            thr = float(np.partition(q, L - k)[L - k])
        else:
            thr = 0.0
        if thr > 0.0:
            w = np.where(q >= thr, q, 0.0)
            if float(w.sum()) <= 1e-300:
                idx = np.argsort(q)[-k:]
                w = np.zeros(L, dtype=np.float64)
                w[idx] = q[idx]
        else:
            idx = np.argsort(q)[-k:]
            w = np.zeros(L, dtype=np.float64)
            w[idx] = q[idx]

    if beta != 1.0:
        w = np.power(w, beta)
    s = float(w.sum())
    if s <= 1e-300 or not np.isfinite(s):
        idx = np.argsort(q)[-max(1, k):]
        w = np.zeros(L, dtype=np.float64)
        w[idx] = q[idx]
        s = float(w.sum())
    if s <= 1e-300:
        return np.ones(L, dtype=np.float64) / L
    return w / s


def md_fixed(p0, time_limit, label="md", mode="exp", step_r=1.0,
             weight_mode="top_frac", alpha=20.0, fraction=0.02, beta=3.0,
             damping=1.0, inv_floor=1e-4, dual_memory=0.0,
             avg_interval=50, verbose=False, use_decay=False):
    p = normalize_p(p0)
    n = len(p)
    c_size = next_pow2(2 * n - 1)
    g_size = next_pow2(3 * n - 2)

    t0 = time.monotonic()
    limit = t0 + max(0.0, float(time_limit))

    q = convolution_size(p, c_size)
    M = float(q.max())
    best = p.copy()
    best_score = float(2.0 * n * M)
    best_M = M

    avg = p.copy()
    avg_count = 1
    stall = 0
    t = 0
    w_prev = None
    log_floor = 1e-300

    while time.monotonic() < limit:
        t += 1
        if t > 5000000:
            break

        cur_score = float(2.0 * n * M)
        if cur_score < best_score - 1e-15:
            best = p.copy()
            best_score = cur_score
            best_M = M
            stall = 0
            avg = p.copy()
            avg_count = 1
        else:
            stall += 1

        if stall > 3000 and (t % 20 == 0):
            p_new = 0.7 * p + 0.3 * best
            p_new = np.maximum(p_new, 0.0)
            ss = float(p_new.sum())
            if ss > 1e-300:
                p = p_new / ss
                q = convolution_size(p, c_size)
                M = float(q.max())
                stall = 0
            continue
        if stall > 12000:
            break

        w_new = make_weights(q, mode=weight_mode, alpha=alpha,
                             fraction=fraction, beta=beta)

        if dual_memory > 0.0 and w_prev is not None:
            w = dual_memory * w_prev + (1.0 - dual_memory) * w_new
            sw = float(w.sum())
            if sw > 1e-300 and np.isfinite(sw):
                w = w / sw
            else:
                w = w_new
        else:
            w = w_new
        w_prev = w

        h = half_grad_size(p, w, g_size)
        hmax = float(h.max())
        if hmax <= 1e-300:
            break
        hs = h / hmax

        if mode == "exp":
            step = step_r
            if use_decay:
                step = step_r / np.sqrt(1.0 + float(t) / 2000.0)
            z = np.log(np.maximum(p, log_floor)) - step * hs
            zmax = float(z.max())
            if not np.isfinite(zmax):
                zmax = 0.0
            pn = np.exp(z - zmax)
            sp = float(pn.sum())
            if sp <= 1e-300 or not np.isfinite(sp):
                break
            pn = pn / sp
            if damping != 1.0:
                pn = damping * pn + (1.0 - damping) * p
                sp = float(pn.sum())
                if sp > 1e-300 and np.isfinite(sp):
                    pn = pn / sp
            p = pn
        else:
            pflat = p / (hs + inv_floor)
            sp = float(pflat.sum())
            if sp <= 1e-300 or not np.isfinite(sp):
                break
            pflat = pflat / sp
            tau = float(np.clip(step_r, 0.0, 1.0))
            p = tau * pflat + (1.0 - tau) * p
            p = np.maximum(p, 0.0)
            sp = float(p.sum())
            if sp <= 1e-300 or not np.isfinite(sp):
                break
            p = p / sp

        q = convolution_size(p, c_size)
        M = float(q.max())

        avg += p
        avg_count += 1

        if t % avg_interval == 0:
            pa = avg / float(avg_count)
            pa = np.maximum(pa, 0.0)
            sa = float(pa.sum())
            if sa > 1e-300:
                pa = pa / sa
                qa = convolution_size(pa, c_size)
                Ma = float(qa.max())
                sa_score = float(2.0 * n * Ma)
                if sa_score < best_score:
                    best = pa.copy()
                    best_score = sa_score
                    best_M = Ma
                    stall = 0
                    avg = pa.copy()
                    avg_count = 1

        if verbose and (t % 500 == 0):
            print(f"    {label}: t={t} best={best_score:.6f} cur={2.0*n*M:.6f}")

    pa = avg / float(avg_count)
    pa = np.maximum(pa, 0.0)
    sa = float(pa.sum())
    if sa > 1e-300:
        pa = pa / sa
        M = float(convolution_size(pa, c_size).max())
        sc = float(2.0 * n * M)
        if sc < best_score:
            best = pa.copy()
            best_score = sc

    Mp = float(convolution_size(p, c_size).max())
    scp = float(2.0 * n * Mp)
    if scp < best_score:
        best = p.copy()
        best_score = scp

    return best, best_score


def interpolate_sequence(p, new_n):
    p = normalize_p(p)
    old_n = len(p)
    if old_n == new_n:
        return p.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    q = np.interp(x_new, x_old, p)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-300:
        q = np.ones(new_n, dtype=np.float64)
        s = float(new_n)
    return q / s


def make_uniform(n):
    return np.ones(n, dtype=np.float64) / float(n)


def make_fourier_seed(n, freq=1.0, phase=0.0, amp=0.8):
    i = np.arange(n, dtype=np.float64)
    v = np.ones(n, dtype=np.float64)
    v += amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 1e-300:
        return make_uniform(n)
    return v / s


def make_multi_seed(n, modes):
    i = np.arange(n, dtype=np.float64)
    v = np.ones(n, dtype=np.float64)
    for freq, phase, amp in modes:
        v += amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 1e-300:
        return make_uniform(n)
    return v / s


def try_update(state, p, label):
    if p is None:
        return
    p = normalize_p(p)
    sc = score_of_norm(p)
    if np.isfinite(sc) and sc < state["score"] - 1e-14:
        state["score"] = sc
        state["best_seq"] = p.copy()
        print(f"  * new best [{label}]: score={sc:.6f}, n={len(p)}")


def remaining_time(limit):
    return max(0.0, float(limit - time.monotonic()))


def default_params(mode="exp", step_r=1.0, weight_mode="top_frac",
                   fraction=0.02, beta=3.0, alpha=20.0, damping=1.0,
                   inv_floor=1e-4, dual_memory=0.0, avg_interval=50,
                   verbose=False, use_decay=False):
    return {
        "mode": mode,
        "step_r": step_r,
        "weight_mode": weight_mode,
        "fraction": fraction,
        "beta": beta,
        "alpha": alpha,
        "damping": damping,
        "inv_floor": inv_floor,
        "dual_memory": dual_memory,
        "avg_interval": avg_interval,
        "verbose": verbose,
        "use_decay": use_decay,
    }


def run_task(state, limit, alloc, p0, label, params):
    rem = remaining_time(limit)
    if rem < 25.0:
        return
    t = min(float(alloc), rem * 0.95)
    if t < 20.0:
        return
    out, _ = md_fixed(p0, t, label=label, **params)
    try_update(state, out, label)


def propose_candidate(budget_s=1800.0):
    start = time.monotonic()
    search_limit = start + min(max(60.0, float(budget_s) - 60.0), 1740.0)

    state = {
        "best_seq": None,
        "score": float("inf"),
    }

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and float(p.sum()) > 0:
            parent = normalize_p(p)
            state["best_seq"] = parent.copy()
            state["score"] = score_of_norm(parent)
            print(f"Parent load: n={len(parent)}, score={state['score']:.6f}")
    except Exception:
        pass

    if state["best_seq"] is None:
        n0 = 1600
        state["best_seq"] = make_uniform(n0)
        state["score"] = score_of_norm(state["best_seq"])
        print(f"No parent; using uniform n={n0}, score={state['score']:.6f}")

    n_base = len(state["best_seq"])

    seed_items = []
    seed_items.append(("parent", state["best_seq"].copy()))
    seed_items.append(("uniform", make_uniform(n_base)))

    sym_parent = normalize_p(0.5 * (state["best_seq"] + state["best_seq"][::-1]))
    seed_items.append(("sym_parent", sym_parent))

    single_modes = [
        (1.0, np.pi / 2.0, 0.5),
        (1.0, np.pi / 2.0, 0.8),
        (1.0, np.pi / 2.0, 1.0),
        (1.0, 0.0, 0.5),
        (1.0, np.pi, 0.5),
        (0.5, 0.0, 0.3),
        (0.5, 0.0, 0.7),
        (0.5, np.pi, 0.5),
        (2.0, np.pi / 2.0, 0.5),
        (3.0, np.pi / 2.0, 0.4),
    ]
    for f, ph, a in single_modes:
        seed_items.append((f"four_f{f:.1f}_ph{ph:.2f}_a{a:.1f}", make_fourier_seed(n_base, f, ph, a)))

    multi_modes = [
        [(1.0, np.pi / 2.0, 0.8), (0.5, 0.0, 0.5)],
        [(1.0, np.pi / 2.0, 0.8), (0.5, np.pi, 0.5)],
        [(1.0, 0.0, 0.6), (0.5, 0.0, 0.4)],
        [(1.0, np.pi / 2.0, 0.7), (2.0, 0.0, 0.3)],
        [(1.0, np.pi / 2.0, 0.7), (2.0, np.pi, 0.3)],
    ]
    for idx, modes in enumerate(multi_modes):
        seed_items.append((f"multi_{idx}", make_multi_seed(n_base, modes)))

    for name, pseed in seed_items:
        pseed = normalize_p(pseed)
        try_update(state, pseed, f"seed_{name}")

    scored = []
    for name, pseed in seed_items:
        sc = score_of_norm(pseed)
        scored.append((sc, name, pseed))
    scored.sort(key=lambda x: x[0])
    top_seeds = scored[:4]
    print("Top initial seeds:")
    for sc, nm, _ in top_seeds:
        print(f"    {nm}: {sc:.6f}")

    # Quick parameter/seed sweep around the current best and promising seeds.
    for r in (0.3, 1.0, 2.0, 4.0):
        params = default_params(mode="exp", step_r=r, fraction=0.02, beta=3.0,
                                avg_interval=40)
        run_task(state, search_limit, 55.0, state["best_seq"], f"quick_exp_r{r}", params)

    for seed_score, seed_name, seed_arr in top_seeds[:2]:
        params = default_params(mode="exp", step_r=1.0, fraction=0.01, beta=2.5,
                                dual_memory=0.3, avg_interval=40)
        run_task(state, search_limit, 50.0, seed_arr, f"quick_seed_{seed_name}", params)

    params = default_params(mode="inv", step_r=0.2, fraction=0.02, beta=3.0,
                            inv_floor=1e-4, avg_interval=40)
    run_task(state, search_limit, 45.0, state["best_seq"], "quick_inv0.2", params)

    params = default_params(mode="inv", step_r=0.5, fraction=0.02, beta=3.0,
                            inv_floor=1e-4, avg_interval=40)
    run_task(state, search_limit, 45.0, state["best_seq"], "quick_inv0.5", params)

    # Medium polishing on the best object found so far.
    params = default_params(mode="exp", step_r=1.0, fraction=0.01, beta=3.0,
                            dual_memory=0.4, use_decay=True, avg_interval=50)
    run_task(state, search_limit, 140.0, state["best_seq"], "med_decay_r1", params)

    params = default_params(mode="exp", step_r=2.0, weight_mode="softmax", alpha=40.0,
                            fraction=1.0, beta=3.0, avg_interval=50)
    run_task(state, search_limit, 140.0, state["best_seq"], "med_softmax_a40", params)

    params = default_params(mode="inv", step_r=0.3, fraction=0.02, beta=3.0,
                            dual_memory=0.3, inv_floor=1e-4, avg_interval=50)
    run_task(state, search_limit, 140.0, state["best_seq"], "med_inv0.3", params)

    params = default_params(mode="exp", step_r=0.3, fraction=0.05, beta=2.0,
                            damping=0.6, avg_interval=50)
    run_task(state, search_limit, 120.0, state["best_seq"], "med_damped_r0.3", params)

    # Try larger lengths.
    for n_large in (3200, 4096):
        rem = remaining_time(search_limit)
        if rem < 160.0:
            break

        p_interp = interpolate_sequence(state["best_seq"], n_large)
        try_update(state, p_interp, f"interp_{n_large}")

        for a in (0.5, 0.8, 1.0):
            p_four = make_fourier_seed(n_large, 1.0, np.pi / 2.0, a)
            try_update(state, p_four, f"fourier_{n_large}_a{a}")

        p_multi = make_multi_seed(
            n_large,
            [(1.0, np.pi / 2.0, 0.8), (0.5, 0.0, 0.5)]
        )
        try_update(state, p_multi, f"multi_{n_large}")

        p_init = p_interp if len(state["best_seq"]) != n_large else state["best_seq"].copy()
        p_four_init = make_fourier_seed(n_large, 1.0, np.pi / 2.0, 0.8)

        params = default_params(mode="exp", step_r=1.0, fraction=0.015, beta=3.0,
                                dual_memory=0.3, avg_interval=60)
        run_task(state, search_limit, 120.0, p_init, f"large_{n_large}_md", params)

        params = default_params(mode="exp", step_r=1.5, fraction=0.01, beta=3.0,
                                dual_memory=0.2, avg_interval=60)
        run_task(state, search_limit, 90.0, p_four_init, f"large_{n_large}_four_md", params)

    # Final fine-polish stage.
    rem = remaining_time(search_limit)
    if rem > 80.0:
        params = default_params(mode="exp", step_r=0.2, fraction=0.05, beta=2.0,
                                damping=0.8, avg_interval=100)
        run_task(state, search_limit, max(40.0, min(180.0, rem * 0.5)),
                 state["best_seq"], "final_fine_exp", params)

    rem = remaining_time(search_limit)
    if rem > 40.0:
        params = default_params(mode="inv", step_r=0.1, fraction=0.05, beta=2.0,
                                inv_floor=1e-3, avg_interval=100)
        run_task(state, search_limit, max(25.0, min(120.0, rem * 0.6)),
                 state["best_seq"], "final_fine_inv", params)

    best = np.asarray(state["best_seq"], dtype=np.float64)
    best = np.nan_to_num(best, nan=0.0, posinf=0.0, neginf=0.0)
    best = np.clip(best, 0.0, 1000.0)
    s = float(best.sum())
    if s < 0.01 or not np.isfinite(s):
        best = make_uniform(1600)
        s = float(best.sum())
    best = best / s

    out = [float(x) for x in best]
    try:
        ev = float(evaluate_sequence(out))
        print(f"Final internal score={state['score']:.6f}, evaluate_sequence={ev:.6f}, n={len(out)}")
        if not np.isfinite(ev):
            # Should not happen, but keep a safe fallback.
            best = make_uniform(1600)
            out = [float(x) for x in best]
    except Exception as ex:
        print("Final evaluate_sequence failed:", ex)
        best = make_uniform(1600)
        out = [float(x) for x in best]

    return out
```
