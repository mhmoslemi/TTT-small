def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Bisection on the threshold
    if v.max() <= 0:
        return np.full(n, target_sum/n)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum/n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference to the largest elements
        idx = np.argsort(result)[::-1]
        result[idx[0]] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result

def compute_lse_grad(a, T):
    """Compute gradient of LSE-smoothed objective w.r.t. a.
    Returns (grad, lse_value)."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    lse_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    # Compute grad = 2 * conv(a_reversed, w)[n-1:n-1+n]
    fft_size = 1 << (n + (2*n-1) - 1 - 1).bit_length()

    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n-1:n-1+n]
    
    return grad, lse_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping for fast initial improvement
2. LSE-smoothed gradient descent with Adam (primary fast optimizer, O(n log n) per iter)
3. Nesterov-accelerated subgradient-LSE blend (complementary dynamics)
4. Aggressive peak-shaving coordinate descent (final refinement)
Key difference from previous: avoids slow LP solves, uses thousands of fast gradient steps
with adaptive temperature annealing and momentum for rapid convergence.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return float('inf')
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} using sorting method."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    if v.max() <= 0:
        return np.full(n, target_sum / n)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-10:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def compute_lse_grad(a, T):
    """Compute LSE-smoothed gradient. Returns (grad, lse_value)."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    lse_val = (max_Tb + np.log(exp_Tb.sum())) / T

    # grad_j = 2 * sum_k w_k * a_{k-j} (valid indices)
    # = 2 * conv(a_reversed, w)[n-1 : n-1+n]
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    a_rev = a[::-1].copy()
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]

    return grad, lse_val


def spectral_shaping(a, n_iter=150, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def lse_adam_optimize(a0, time_limit, T_start=60.0, T_end=0.2, lr_base=0.04, verbose=True):
    """
    LSE-smoothed gradient descent with Adam optimizer.
    Fast per-iteration (O(n log n)), allowing many iterations.
    Temperature anneals from T_start (smooth) to T_end (sharp).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t_adam = 0

    # Nesterov momentum in a-space
    a_prev = a.copy()
    nest_beta = 0.7

    stall = 0
    stall_threshold = 3000

    # Precompute FFT size for gradient
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1.0), 1.0)

        # Temperature annealing: exponential
        T_curr = T_start * (T_end / T_start) ** progress

        # Nesterov lookahead
        if t_adam > 0:
            y = a + nest_beta * (a - a_prev)
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            else:
                y = a.copy()
        else:
            y = a.copy()

        # Compute gradient at y
        b = fast_autocorr(y)
        Tb = T_curr * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        a_rev = y[::-1].copy()
        A_rev = rfft(a_rev, fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        # Learning rate: decay with iteration
        lr_t = lr_base / (1 + 0.0005 * t_adam)

        a_new = y - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        a_prev = a.copy()
        a = a_new

        # Track best
        score = internal_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if verbose and t_adam % 5000 == 0:
            print(f"    LSE-Adam t={t_adam} (T={T_curr:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

        if stall > stall_threshold:
            # Perturb to escape
            rng = np.random.default_rng(hash((t_adam, 0)))
            idx = rng.integers(0, n, size=max(1, n // 100))
            a[idx] *= rng.uniform(0.5, 1.5, size=len(idx))
            a = project_simplex(a)
            stall = 0

    return best_a, best_score


def subgrad_blend_optimize(a0, time_limit, verbose=True):
    """
    Nesterov-accelerated method blending subgradient (at argmax) with LSE-smoothed gradient.
    Uses time-averaged iterates for convergence guarantee.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    t = 0
    stall = 0

    avg_a = np.zeros(n)
    n_avg = 0

    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Nesterov lookahead
        if t > 1:
            momentum = (t - 1) / (t + 2)
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            else:
                y = a.copy()
        else:
            y = a.copy()

        a_prev = a.copy()

        # Compute blended gradient
        b = fast_autocorr(y)

        # Adaptive temperature: starts high (smooth), decreases
        T_adapt = max(3.0, 50.0 * np.exp(-t / 2000.0))
        Tb = T_adapt * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        a_rev = y[::-1].copy()
        A_rev = rfft(a_rev, fft_size)
        W = rfft(w, fft_size)
        conv_w = irfft(A_rev * W, fft_size)
        grad_lse = 2.0 * conv_w[n - 1:n - 1 + n]

        # Also compute subgradient at argmax for sharpness
        k_star = b.argmax()
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        grad_sub = np.zeros(n)
        indices = np.arange(js, je + 1)
        grad_sub[indices] = 2.0 * y[k_star - indices]

        # Blend: more weight on subgradient as T decreases
        blend = min(1.0, T_adapt / 50.0)
        grad = blend * grad_lse + (1 - blend) * grad_sub

        # Step size: O(1/sqrt(t))
        lr = 1.5 / (np.sqrt(t) + 1)

        a_new = y - lr * grad
        a_new = project_simplex(a_new)

        # Time average
        avg_a += a_new
        n_avg += 1

        # Track best
        score = internal_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if n_avg >= 200 and n_avg % 1000 == 0:
            score_avg = internal_score(avg_a / n_avg)
            if score_avg < best_score:
                best_score = score_avg
                best_a = (avg_a / n_avg).copy()
                stall = 0

        a = a_new

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradBlend t={t} ({elapsed:.0f}s): best={best_score:.6f}")

        if stall > 15000:
            break

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Aggressive coordinate-descent peak shaving with adaptive step sizes."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_k = b.argmax()

        # Top contributors to the main peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]

        # Find top 5 peaks
        top_peaks = np.argsort(b)[::-1][:5]

        # Build dangerous set (indices that contribute significantly to top peaks)
        dangerous = set()
        for k in top_peaks:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            for j in range(js2, je2 + 1):
                if a[j] * a[k - j] > 0.1 * b[k]:
                    dangerous.add(j)

        # Source: highest contributor to main peak
        src = idx[np.argmax(contrib)]

        # Destination: safest position (minimizes impact on all peaks)
        safe_candidates = []
        for j in range(n):
            if j in dangerous:
                continue
            # Score: total contribution to all top peaks
            danger = 0.0
            for k in top_peaks:
                if 0 <= k - j < n:
                    danger += a[j] * a[k - j]
            safe_candidates.append((danger, j))

        if not safe_candidates:
            safe_candidates = [(0.0, j) for j in range(n) if j != src]

        safe_candidates.sort()
        dst_pool = [j for _, j in safe_candidates[:max(1, len(safe_candidates) // 5)]]
        if not dst_pool:
            dst = (src + 1) % n
        else:
            dst = dst_pool[rng.integers(len(dst_pool))]

        # Try various transfer sizes
        moved = False
        for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < -1e-15:
                continue
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score = internal_score(a_trial)
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                moved = True
                stall = 0
                break

        if not moved:
            # Try multiple random moves
            improved = False
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-13:
                    continue
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if not improved:
                stall += 1
                if stall > 500:
                    break

    return best_a, best_score


def short_lse_burst(a0, time_limit, verbose=False):
    """Quick LSE optimization with very low temperature for final polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    T = 5.0
    lr = 0.01
    v = np.zeros(n)
    v2 = np.zeros(n)
    bm, bv = 0.9, 0.999
    t_step = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        a_rev = a[::-1].copy()
        A_rev = rfft(a_rev, fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        t_step += 1
        v = bm * v + (1 - bm) * grad
        v2 = bv * v2 + (1 - bv) * grad ** 2
        v_hat = v / (1 - bm ** t_step)
        v2_hat = v2 / (1 - bv ** t_step)

        lr_t = lr / (1 + 0.001 * t_step)
        a_new = a - lr_t * v_hat / (np.sqrt(v2_hat) + 1e-8)
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral a={alpha_val}: {best_score:.6f}")
    print(f"  Phase 1 done: {best_score:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # Phase 2: LSE-Adam (primary workhorse - fast, many iterations)
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 2: LSE-Adam (rem={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 20:
        a_out, sc_out = lse_adam_optimize(best_seq, t2, T_start=80.0, T_end=0.1, lr_base=0.05)
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [LSE-Adam]: {best_score:.6f}")
    print(f"  Phase 2 done: {best_score:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # Phase 3: Subgradient-LSE blend (different dynamics)
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 3: SubGrad-Blend (rem={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 15:
        a_out, sc_out = subgrad_blend_optimize(best_seq, t3)
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [SubGrad]: {best_score:.6f}")
    print(f"  Phase 3 done: {best_score:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # Phase 4: Peak shaving (coordinate descent)
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 4: Peak shaving (rem={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10:
        a_out, sc_out = peak_shaving(best_seq, t4, seed=42)
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [PeakShave]: {best_score:.6f}")
    print(f"  Phase 4 done: {best_score:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # Phase 5: LSE-Adam round 2 with different params
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 5: LSE-Adam R2 (rem={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 10:
        a_out, sc_out = lse_adam_optimize(best_seq, t5, T_start=30.0, T_end=0.05, lr_base=0.02)
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [LSE-Adam2]: {best_score:.6f}")
    print(f"  Phase 5 done: {best_score:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # Phase 6: Short LSE burst for final polish
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 6: LSE burst (rem={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 5:
        a_out, sc_out = short_lse_burst(best_seq, t6)
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [LSE-burst]: {best_score:.6f}")
    print(f"  Phase 6 done: {best_score:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # Phase 7: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 10:
        t7 = min(remaining * 0.15, 80.0)
        print(f"\n=== Phase 7: Final peak shave (rem={remaining:.0f}s) ===")
        a_out, sc_out = peak_shaving(best_seq, t7, seed=777)
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [PeakShave2]: {best_score:.6f}")
        print(f"  Phase 7 done: {best_score:.6f} (t={time.monotonic()-t_start:.0f}s)")

    # Phase 8: Use remaining time for one more optimization round
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 30:
        t8 = min(remaining * 0.5, 100.0)
        print(f"\n=== Phase 8: Final opt (rem={remaining:.0f}s) ===")
        a_out, sc_out = lse_adam_optimize(best_seq, t8, T_start=10.0, T_end=0.01, lr_base=0.01, verbose=False)
        if sc_out < best_score:
            best_score = sc_out
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [Final]: {best_score:.6f}")

        remaining2 = time_budget - (time.monotonic() - t_start)
        if remaining2 > 10:
            t9 = min(remaining2 * 0.5, 40.0)
            a_out, sc_out = peak_shaving(best_seq, t9, seed=31337)
            if sc_out < best_score:
                best_score = sc_out
                best_seq = a_out.copy()
                print(f"  *** NEW BEST [FinalShave]: {best_score:.6f}")

    # Final evaluation
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        try:
            final_score = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(final_score):
                best_score = float(final_score)
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
