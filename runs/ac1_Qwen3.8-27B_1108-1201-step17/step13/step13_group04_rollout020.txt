def lse_gradient(a, T):
    """
    Compute subgradient-based LSE gradient.
    dL/da[m] = sum_k w[k] * (a[k-m] + a[m-k]) where w = softmax(b/T)
    """

    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    # dL/da[m] = sum_k w[k] * (a[k-m] + a[m-k])
    # = (rev(a) * w)[m + n - 1] + (a * w)[m]
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2

            T[j, k] = a[j - k]
    return T.tocsr()


def solve_lp_full(a, solver):
    """
    Solve LP: minimize t s.t. T @ b <= t, sum(b)=1, b>=0.
    Returns (b_normalized, t_value) or (None, None).
    """
    n = len(a)
    T = build_conv_matrix(a)
    return solve_lp_with_matrix(T, n, solver)


def solve_lp_with_matrix(T, n, solver):
    """Solve LP given prebuilt matrix T."""
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    t_val = max(0.0, float(t_var.value))
    return b_val, t_val


def solve_lp_reduced(a, active_rows, solver):
    """
    Solve LP with only active constraints (constraint generation approach).
    Returns (b, t, violated_rows) or (None, None, None).
    """
    n = len(a)
    if len(active_rows) < 10:
        active_rows = list(range(2 * n - 1))
    
    # Build partial matrix
    T_part = lil_matrix((len(active_rows), n))
    for idx, row in enumerate(active_rows):
        k_min = max(0, row - n + 1)
        k_max = min(n - 1, row)
        for k in range(k_min, k_max + 1):
            T_part[idx, k] = a[row - k]
    T_part = T_part.tocsr()
    
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_part @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    t_val = max(0.0, float(t_var.value))
    
    # Check all constraints for violations
    conv_ab = fast_conv(a, b_val)
    violated = np.where(conv_ab > t_val + 1e-6)[0]
    return b_val, t_val, set(violated.tolist())


def nesterov_lp(a0, solver, time_limit, verbose=True, use_reduced=False):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    y = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = y.sum()
    if s0 < 1e-15:
        y = np.ones(n) * norm / n
    else:
        y = y * norm / s0

    best_a = y.copy()
    best_score = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    avg_x = np.zeros(n)
    n_avg = 0
    it_count = 0
    active_rows = list(range(2 * n - 1))
    
    for t_iter in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_start = time.monotonic()
        if use_reduced:
            b_star, t_obj, violated = solve_lp_reduced(y, active_rows, solver)
            if b_star is not None and len(violated) > 0:
                # Add violated constraints and re-solve
                active_rows = list(set(active_rows) | violated)
                if len(active_rows) > 2 * n - 1:
                    active_rows = list(range(2 * n - 1))
                b_star, t_obj, violated = solve_lp_reduced(y, active_rows, solver)
                # Keep active set for next iteration
                if b_star is not None:
                    conv_ab = fast_conv(y, b_star)
                    top_k = 200
                    active_rows = list(np.argsort(conv_ab)[::-1][:top_k])
        else:
            b_star, t_obj = solve_lp_full(y, solver)
        
        lp_time = time.monotonic() - t_start
        
        if b_star is None:
            b_star = np.ones(n) / n
            if lp_time > 30:
                if verbose:
                    print(f"    LP: solver failed repeatedly, stopping")
                break
        
        b_star = b_star * norm / b_star.sum()
        it_count += 1
        
        # Nesterov update
        alpha = 2.0 / (t_iter + 2.0)
        if t_iter == 1:
            y_new = b_star.copy()
            x = y_new.copy()
        else:
            y_prev = y.copy()
            y_new = (1.0 - alpha) * y + alpha * b_star
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            
            momentum = (t_iter - 1.0) / (t_iter + 2.0)
            x = y_new + momentum * (y_new - y_prev)
            x = np.maximum(x, 0.0)
            s_x = x.sum()
            if s_x < 1e-15:
                x = y_new.copy()
            else:
                x = x * norm / s_x
        
        y = y_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_score:
            best_score = score_x
            best_a = x.copy()
        
        # Running average (Polak averaging)
        avg_x += x
        n_avg += 1
        if n_avg % 10 == 0:
            avg_a = avg_x / n_avg
            s_avg = avg_a.sum()
            if s_avg > 1e-15:
                avg_a = avg_a * norm / s_avg
                score_avg = fast_autocorr(avg_a).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_a.copy()
        
        # Update active set for next iteration
        if use_reduced:
            conv_y = fast_autocorr(y)
            top_k = min(300, 2 * n - 1)
            active_rows = list(np.argsort(conv_y)[::-1][:top_k])
        
        if verbose and it_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it_count} ({elapsed:.0f}s, lp={lp_time:.1f}s): best={best_score:.8f}")
    
    return best_a, best_score, it_count


def subgradient_nesterov(a0, time_limit, verbose=True):
    """
    Subgradient method with Nesterov acceleration.
    Uses the subgradient of max_k conv(a,a)_k (hard max).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    y = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = y.sum()
    if s0 < 1e-15:
        y = np.ones(n) * norm / n
    else:
        y = y * norm / s0

    best_a = y.copy()
    best_score = fast_autocorr(y).max()
    t0 = time.monotonic()
    
    y_prev = y.copy()
    it_count = 0
    
    for t_iter in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(y)
        max_b = b.max()
        
        # Find argmax indices (within tolerance)
        tol = max_b * 1e-6
        argmax_set = np.where(b >= max_b - tol)[0]
        
        # Subgradient: sg[m] = (1/|S|) * sum_{k in S} (a[k-m] + a[m-k])
        sg = np.zeros(n)
        for k in argmax_set:
            # (a*a)_k derivative w.r.t. a[m] = a[k-m] + a[m-k] (with valid index handling)
            # a[k-m] is valid if 0 <= k-m < n
            # a[m-k] is valid if 0 <= m-k < n
            for m in range(n):
                dm1 = k - m  # index for a[k-m]
                dm2 = m - k  # index for a[m-k]
                val = 0.0
                if 0 <= dm1 < n:
                    val += y[dm1]
                if 0 <= dm2 < n:
                    val += y[dm2]
                sg[m] += val
        sg /= len(argmax_set)
        
        # Nesterov
        if t_iter == 1:
            step = 1.0 / (2.0 * t_iter)
            y_new = y - step * (sg / max_b)
            x = y_new.copy()
        else:
            momentum = (t_iter - 1.0) / (t_iter + 2.0)
            y_hat = y + momentum * (y - y_prev)
            # Compute gradient at y_hat
            b_hat = fast_autocorr(y_hat)
            max_hat = b_hat.max()
            tol_hat = max_hat * 1e-6
            argmax_hat = np.where(b_hat >= max_hat - tol_hat)[0]
            sg_hat = np.zeros(n)
            for k in argmax_hat:
                for m in range(n):
                    dm1 = k - m
                    dm2 = m - k
                    val = 0.0
                    if 0 <= dm1 < n:
                        val += y_hat[dm1]
                    if 0 <= dm2 < n:
                        val += y_hat[dm2]
                    sg_hat[m] += val
            sg_hat /= len(argmax_hat)
            
            step = 1.0 / (2.0 * t_iter)
            y_new = y_hat - step * (sg_hat / max_hat)
            x = y_new.copy()
        
        y_prev = y.copy()
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
            x = y.copy()
        else:
            y_new = y_new * norm / s_y
            s_x = x.sum()
            if s_x < 1e-15:
                x = y_new.copy()
            else:
                x = x * norm / s_x
        
        y = y_new
        it_count += 1
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_score:
            best_score = score_x
            best_a = x.copy()
        
        if verbose and it_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it_count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def lse_gradient_fft(a, T):
    """Efficient LSE gradient using FFT."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2


def nesterov_lse(a0, time_limit, T_start=15.0, T_end=0.3, n_stages=12, verbose=True):
    """Nesterov-accelerated LSE gradient descent with geometric cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            if it == 0:
                grad = lse_gradient_fft(y, T)
                y_new = y - (1.0 / (it + 3.0)) * (grad / max(1e-15, T))
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_hat = y + momentum * (y - y_prev)
                grad = lse_gradient_fft(y_hat, T)
                y_new = y_hat - (1.0 / (it + 3.0)) * (grad / max(1e-15, T))

            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y = y_new
            it_count += 1

            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if it_count % 200 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"    LSE stage={si}/{n_stages} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_targeting_search(a0, time_limit, seed=42, verbose=True):
    """Peak-targeting local search with 1-swap moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_ks = np.argsort(b)[::-1][:8]

        improved = False
        for k_star in top_ks:
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:4]]

            for src in top_contrib_idx:
                if a[src] < norm * 1e-9:
                    continue

                # Penalty mask: positions that contribute to top peaks
                peak_penalty = np.zeros(n)
                for k2 in top_ks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    for m in range(js2, je2 + 1):
                        peak_penalty[m] += b[k2] * a[m] * a[k2 - m] / max(1e-30, b[k2])

                # Candidates: low penalty, not src
                candidates = np.array([m for m in range(n) if m != src])
                candidates = candidates[np.argsort(peak_penalty[candidates])[:max(5, len(candidates) // 5)]]

                for dst in candidates[:5]:
                    for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if rng.integers(10) == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.003, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 150:
                if verbose:
                    print(f"    PeakSearch: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                break

        if verbose and it_count % 200 == 0:
            print(f"    PeakSearch it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x)) ** 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.3
    else:
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """
    Main search function. Uses LP best-response with Nesterov as the primary
    optimizer, complemented by LSE gradient descent at longer lengths and
    peak-targeting local search for final polish.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = 1710.0  # Leave 90s safety margin

    solver = get_solver()
    print(f"Solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: LP Nesterov at base length (main workhorse) ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.40, 680.0)
    if t1 > 30 and solver is not None:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _, _ = nesterov_lp(a_work, solver, t1, verbose=True, use_reduced=False)
        do_try_update(a_out, f"lp_n{n_best}")

    # ============ Phase 2: Subgradient Nesterov (complementary) ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: SubGrad-Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 150.0)
    if t2 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_nesterov(a_work, t2, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_best}")

    # ============ Phase 3: LSE Nesterov at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE-Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 15:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t3, T_start=10.0, T_end=0.2, n_stages=12, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # ============ Phase 4: Interpolate to longer length, LP ============
    n2 = n_best + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30 and solver is not None:
        init_long = interpolate_sequence(best_seq, n2)
        a_work = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _, _ = nesterov_lp(a_work, solver, t4, verbose=True, use_reduced=True)
        do_try_update(a_out, f"lp_n{n2}")

    # ============ Phase 5: LSE at even longer length ============
    n3 = n_best + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE-Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 20:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_work = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lse(a_work, t5, T_start=20.0, T_end=0.5, n_stages=10, verbose=True)
        do_try_update(a_out, f"lse_n{n3}")

    # ============ Phase 6: Peak targeting on best ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: PeakTargeting n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_targeting_search(a_work, t6, seed=12345, verbose=True)
        do_try_update(a_out, f"peak_n{n_best}")

    # ============ Phase 7: Final LP polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP polish n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.30, 300.0)
    if t7 > 20 and solver is not None:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _, _ = nesterov_lp(a_work, solver, t7, verbose=True, use_reduced=False)
        do_try_update(a_out, f"lp_polish_n{n_best}")

    # ============ Phase 8: Final peak targeting ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        t8 = min(remaining * 0.8, 120.0)
        if t8 > 5:
            print(f"\n=== Phase 8: Final peak targeting n={n_best} (budget={t8:.0f}s) ===")
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = peak_targeting_search(a_work, t8, seed=99999, verbose=True)
            do_try_update(a_out, "peak_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence! Using fallback.")
        return [1.0]

    return [float(x) for x in best_seq]
