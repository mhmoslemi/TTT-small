def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response_fast(a, M):
    """Solve max sum(b) s.t. conv(a,b) <= M, b >= 0 using scipy.linprog.
    Returns b or None."""
    n = len(a)
    T = build_conv_matrix(a)
    # Maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    # Constraints: T @ b <= M (2n-1 constraints)
    A_ub = T
    b_ub = np.full(2 * n - 1, M)
    # b >= 0 handled by bounds
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, 
                  method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-6})
    if res.status != 0:
        return None
    b = np.maximum(res.x, 0)
    if b.sum() < 1e-15:
        return None
    return b

    n = len(a)
    T = build_conv_matrix(a)

                  bounds=[(0, None)]*n, method='highs')
    if res.status != 0:
        return None
    b = np.maximum(res.x, 0)
    
    # Identify active constraints and re-solve
    for _ in range(5):

        tol = 1e-7 * M
        active = circ >= M - tol

        b_new = np.maximum(res.x, 0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-5):

        else:
            break

            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb, n), 0)
            a = a * norm / a.sum()
            stall = 0
            if verbose:
                print(f"    [Restart at iter {n_iters}]")
    
    return best_a, best_score

def compute_lse_gradient(a, w):
    """Gradient of weighted sum of autocorrelation values."""
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]

def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smooth gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.001, best_score * 0.003)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        grad = compute_lse_gradient(a, w)
        
        # Backtracking line search
        step = 1.0 / max(T * n * 0.3, 1e-6)
        for _ in range(10):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            score_new = fast_autocorr(a_new).max()
            if score_new < max_b:
                break
            step *= 0.5
        
        a = a_new
        steps += 1
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 100 == 0:
            print(f"    LSE-GD step={steps} ({time.monotonic()-t0:.0f}s): T={T:.4f}, best={best_score:.8f}")
    
    return best_a, best_score

def subgradient_descent(a0, time_limit, verbose=True):
    """Projecting subgradient descent with Nesterov averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    cumulative = np.zeros(n)
    steps = 0
    best_averaged_score = best_score
    
    # Step size schedule: eta_t = c / sqrt(t)
    c = 1.0
    
    for t in range(1, 500000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        b = fast_autocorr(y)
        k = np.argmax(b)
        M = b[k]
        
        # Subgradient of max at point y: 2 * H_k * y
        # H_k * y = the vector where element i is sum_j H_k[i,j] * y[j]
        # H_k[i,j] = 1 if i+j=k (within bounds)
        # (H_k * y)[i] = y[k-i] if 0 <= k-i < n, else 0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad = np.zeros(n)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k - idx]
        
        # Step
        eta = c / np.sqrt(t)
        y_new = y - eta * grad
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s
        
        # Nesterov momentum
        alpha = 2.0 / (t + 2.0)
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        cumulative += a_new
        y = y_new
        steps += 1
        
        # Check averaged solution every 100 steps
        if steps % 100 == 0:
            avg = cumulative / steps
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg_norm = avg * norm / s_avg
                score_avg = fast_autocorr(avg_norm).max()
                if score_avg < best_averaged_score:
                    best_averaged_score = score_avg
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_norm.copy()
        
        # Check current
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        if verbose and steps % 500 == 0:
            print(f"    SGD step={steps} ({elapsed:.0f}s): best={best_score:.8f}, avg_best={best_averaged_score:.8f}")
        
        # Adaptive c
        if steps % 1000 == 0 and best_score > best_averaged_score * 1.01:
            c *= 0.9
    
    return best_a, best_score

def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Peak shaving with simulated annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    
    T_start = max(0.3, best_score * 0.03)
    T_end = best_score * 0.001
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        b = fast_autocorr(a)
        # Find top peaks
        top_k = np.argsort(b)[::-1][:5]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-18:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Try moving mass from src to a random other position
            # Choose destination that minimizes impact
            candidates = np.where(np.arange(n) != src)[0]
            # Prefer positions with low value
            n_cand = min(20, len(candidates))
            sorted_cand = candidates[np.argsort(a[candidates])[:n_cand * 3]]
            dst = sorted_cand[rng.integers(len(sorted_cand))]
            
            frac = rng.choice([0.02, 0.05, 0.1, 0.15, 0.2, 0.3])
            amt = a[src] * frac
            if amt < 1e-18:
                continue
            
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            
            score = fast_autocorr(a_trial).max()
            
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                n_improve += 1
                break
            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break
        
        if not improved:
            n_stall += 1
            if n_stall > 100:
                # Larger perturbation
                n_pert = max(3, n // 50)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.1, 0.5)
                    j = rng.integers(n)
                    if j != idx:
                        a_trial[idx] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.02:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0
        
        if verbose and n_improve % 1000 == 0 and n_improve > 0:
            print(f"    PeakShave improve={n_improve} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def generate_initializations(n, seed=42):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sine family
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        x = np.linspace(0.0, 1.0, n)
        f = np.sin(np.pi * x) ** p
        s = f.sum()
        if s > 1e-15:
            inits.append(f / s)
    
    # Cosine window
    for p in [1.0, 2.0, 4.0]:
        x = np.linspace(0.0, 1.0, n)
        f = (0.5 + 0.5 * np.cos(2 * np.pi * x / (n - 1))) ** p
        f = f + 0.01  # avoid zeros
        s = f.sum()
        if s > 1e-15:
            inits.append(f / s)
    
    # Gaussian
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        x = np.linspace(-1.0, 1.0, n)
        f = np.exp(-alpha * x**2)
        s = f.sum()
        if s > 1e-15:
            inits.append(f / s)
    
    # U-shaped (inverse cosine)
    x = np.linspace(0.0, 1.0, n)
    f = 0.5 - 0.5 * np.cos(2 * np.pi * x)
    f = f + 0.01
    s = f.sum()
    if s > 1e-15:
        inits.append(f / s)
    
    # Random
    f = rng.dirichlet(np.ones(n))
    inits.append(f)
    
    # Bi-modal
    x = np.linspace(-1.0, 1.0, n)
    f = np.exp(-2.0 * (x - 0.5)**2) + np.exp(-2.0 * (x + 0.5)**2)
    s = f.sum()
    if s > 1e-15:
        inits.append(f / s)
    
    return inits
