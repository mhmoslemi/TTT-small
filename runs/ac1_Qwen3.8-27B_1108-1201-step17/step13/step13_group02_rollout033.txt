def fast_autocorr(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix_rows(a, rows):
    """Build only specified rows of the convolution matrix T.
    T[k, :] @ b = conv(a, b)[k]
    For row k: (T[k])_j = a[j] if 0 <= k-j < n, else 0
    Equivalently: T[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    """
    n = len(a)
    T = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            T[idx, js:je + 1] = a[k - je:k - js + 1][::-1]
    return T


def lp_best_response_fast(a, M, solver, top_k=50, warm_start=None):
    """Solve LP: max sum(b) s.t. T_top @ b <= M_top, b >= 0
    Using only top_k constraints for speed."""
    n = len(a)
    b = fast_autocorr(a)
    # Get indices of top_k largest autocorrelation values
    top_idx = np.argsort(b)[::-1][:top_k]
    top_idx = sorted(top_idx.tolist())
    M_top = np.array([b[k] for k in top_idx])
    
    T = build_conv_matrix_rows(a, top_idx)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M_top]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        y_new = np.maximum(y_new, 0.0)

        if score_y < best_score:

            best_a = y.copy()
            stall = 0

        else:

        score_a = fast_autocorr(a).max()
        if score_a < best_score:

            stall = 0

            stall = 0

        # grad[i] = sum_k w[k] * d(conv_k)/da[i]
        # = sum_k w[k] * (a[k-i] + (i-k) != 0 * a[i-k]) for valid indices
        # Simplification: grad[i] = 2 * sum_k w[k]*a[k-i] for valid k-i in [0,n-1]
        #                    - sum_k w[k]*a[k-i]*delta_{k,2i} (to correct double count)
        # Actually for the "full" conv convention where conv[k] = sum_j a[j]*a[k-j]:
        # d(conv[k])/da[i] = a[k-i] (j=i term) + a[i] (j=k-i term, if k-i != i)
        #                    = a[k-i] + a[i] if k-i != i and 0<=k-i<n
        #                    = 2*a[i] if k=2i
        #                    = a[k-i] if only one is valid
        pass
    
    # Use FFT-based approach (most efficient):
    # The key identity: if f(a) = sum_k w[k]*conv(a,a)[k], then
    # grad[i] = 2 * (w_cross_corr_a)[i] - w_diag[i]
    # where w_cross_corr_a[i] = sum_k w[k]*a[k-i] (validated range)
    # and w_diag[i] accounts for the self-term.
    
    # Actually, the most reliable FFT method:
    # Define the "reversed w": wr[i] = w[n-1-i] for i=0..n-1 (take the "center" part)
    # No, this is getting too complicated. Let me just use the original approach
    # from the parent program which worked:
    
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smooth gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.005, best_score * 0.003)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = max(b)
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = compute_lse_gradient(a, w)

        # Adaptive step with line search
        step = 0.5 / (n * 0.1)
        for ls in range(5):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new * norm / s_new
            score = fast_autocorr(a_new).max()
            if score < max_b:
                a = a_new
                break
            step *= 0.5
        else:
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new * norm / s_new
            else:
                a = a.copy()

        steps += 1
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.5f}, best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42, verbose=True):
    """Local search: try perturbing individual elements or pairs."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    T = max(0.1, best_score * 0.01)

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        T = T * (0.999 ** int(elapsed_frac * 1000))
        T = max(T, best_score * 0.001)

        b = fast_autocorr(a)
        top_k_idx = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_k_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_c = contrib.max()
            if max_c < 1e-18:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find best destination
            # Try a few random destinations
            for _ in range(10):
                dst = rng.integers(n)
                if dst == src:
                    continue
                frac = rng.uniform(0.05, 0.3)
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    n_improve += 1
                    break
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial
                    improved = True
                    break

        if not improved:
            n_stall += 1
            if n_stall > 100:
                # Larger perturbation
                npert = max(1, n // 100)
                idx_p = rng.choice(n, size=npert, replace=False)
                a_trial = a.copy()
                for idx in idx_p:
                    amt = a[idx] * rng.uniform(0.1, 0.3)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.01:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                n_stall = 0

        if verbose and n_improve > 0 and n_improve % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def spectral_init(n, alpha=2.0, beta=1.0):
    """Create initial sequence based on spectral shaping.
    We want flat autocorrelation, which means flat power spectrum.
    For a non-negative sequence, we shape the spectrum to be as flat as possible
    while maintaining non-negativity of the time-domain signal.
    """
    # Start with a window function that gives good spectral properties
    x = np.linspace(0.0, 1.0, n)
    # Use a raised cosine (Hann-like) window
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    # Add some spectral shaping via frequency domain
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    # Try to flatten the spectrum
    magnitude = np.abs(A)
    magnitude[magnitude < 1e-10] = 1e-10
    # Shape toward flat
    target = np.ones_like(magnitude)
    shaped = magnitude ** 0.5 * target ** 0.5  # geometric mean
    # Reconstruct
    phase = np.angle(A)
    A_new = shaped * np.exp(1j * phase)
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = a.copy()
        s = a_new.sum()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function to find best sequence."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    print(f"Total time budget: {time_budget:.0f}s")

    solver = get_solver()

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        best_seq = spectral_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting n={n_base}, score={best_score:.6f}")

    # Phase 1: Fast LP Nesterov at base length (most time here)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.50
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, top_k_start=20, top_k_end=60, verbose=True)
        try_update(a_out, f"lp_n{n_base}")

    # Phase 2: LSE gradient descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LSE-GD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 15:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t2, verbose=True)
        try_update(a_out, f"lse_n{n_cur}")

    # Phase 3: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: CoordDesc n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 15:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t3, seed=99, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # Phase 4: Second LP round with different parameters
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov round 2 n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, top_k_start=40, top_k_end=100, verbose=True)
        try_update(a_out, f"lp2_n{n_cur}")

    # Phase 5: LSE polish again
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t5, verbose=False)
        try_update(a_out, f"lse2_n{n_cur}")

    # Phase 6: Try longer length
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP Nesterov n={n_long} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, top_k_start=20, top_k_end=50, verbose=True)
        try_update(a_out, f"lp_long_n{n_long}")

    # Phase 7: Final coordinate descent on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CoordDesc n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t7, seed=314, verbose=True)
        try_update(a_out, f"cd_final_n{n_best}")

    # Phase 8: Final LSE on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LSE n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 120.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t8, verbose=True)
        try_update(a_out, f"lse_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0, 1.0]

    return [float(x) for x in best_seq]
