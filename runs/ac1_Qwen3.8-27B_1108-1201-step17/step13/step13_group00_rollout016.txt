    temps = np.geomspace(100.0, 0.2, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        # Adjust number of steps based on remaining time
        n_stages_left = n_stages - si
        n_steps = max(5, int(remaining / n_stages_left * 1.5))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute autocorrelation
            A = rfft(a, fft_size)
            c = irfft(A * A, fft_size)[:size]
            c_max = c.max()
            
            # LSE weights
            # Clip to avoid overflow
            c_shifted = (c - c_max) / T
            c_shifted = np.maximum(c_shifted, -50.0)
            w = np.exp(c_shifted)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(size)
                w[int(np.argmax(c))] = 1.0
            else:
                w = w / w_sum
            
            # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
            # = 2 * (a_flipped * w)[n-1+i]
            a_flip = a[::-1]
            Af = rfft(a_flip, fft_size2)
            Wf = rfft(w, fft_size2)
            conv_fw = irfft(Af * Wf, fft_size2)
            grad = 2.0 * conv_fw[n - 1:2 * n - 1]
            
            # Mirror descent step
            eta = 1.0 / max(T, 0.1)
            # Scale gradient for stability
            grad_scale = np.abs(grad).max()
            if grad_scale > 1e-10:
                grad_norm = grad / grad_scale
            else:
                grad_norm = grad
            
            a_new = a * np.exp(-eta * grad_norm)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            a = a_new
            n_iters += 1
            
            # Check score periodically
            if n_iters % 10 == 0:
                A2 = rfft(a, fft_size)
                c2 = irfft(A2 * A2, fft_size)[:size]
                score = c2.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            A2 = rfft(a, fft_size)
            c2 = irfft(A2 * A2, fft_size)[:size]
            print(f"    LSE-MD T={T:.2f} ({elapsed:.0f}s): score={c2.max():.8f}, best={best_score:.8f}")
    
    # Final check
    A2 = rfft(a, fft_size)
    c2 = irfft(A2 * A2, fft_size)[:size]
    if c2.max() < best_score:
        best_score = c2.max()
        best_a = a.copy()
    
    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with FFT-based gain computation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    size = 2 * n - 1
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < 1e-10:
                continue
            
            for attempt in range(2):
                j = int(rng.integers(n))
                if j == i:
                    continue
                
                ai = a[i]
                # Gain vector: change in conv when transferring delta from i to j
                # First order: g[k] = a[k-j] + a[j-k] - a[k-i] - a[i-k]
                # (each with appropriate zero-padding)
                g = np.zeros(size)
                
                # +a[k-j] for k in [j, j+n-1]
                lo = j
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[:hi - lo]
                # +a[j-k] for k in [0, j] (a[j-k] valid when 0 <= j-k < n, i.e., j-n+1 <= k <= j)
                lo2 = max(0, j - n + 1)
                hi2 = min(j + 1, size)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                
                # -a[k-i] for k in [i, i+n-1]
                lo3 = i
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[:hi3 - lo3]
                # -a[i-k] for k in [0, i]
                lo4 = max(0, i - n + 1)
                hi4 = min(i + 1, size)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                # Find best delta
                best_d = 0.0
                best_val = conv.max()
                deltas = np.linspace(0.01, ai, 15)
                for d in deltas:
                    new_c = conv + d * g
                    # Second order at k=2i: (ai-d)^2 - ai^2 = -2*ai*d + d^2
                    if 2 * i < size:
                        new_c[2 * i] -= 2 * ai * d - d * d  # wait this is wrong
                        # Actually: new value at 2i = old - 2*ai*d + d^2
                        # conv[2i] already includes ai^2, so new = conv[2i] - 2*ai*d + d^2
                        # The first-order approximation already accounts for -ai from the -a[k-i] term at k=2i
                        # and +ai from... hmm this is getting complicated. Let me just use first order.
                    
                    new_max = new_c.max()
                    if new_max < best_val - 1e-12:
                        best_val = new_max
                        best_d = d
                
                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = np.maximum(a, 0.0)
                    s2 = a.sum()
                    if s2 > 1e-15:
                        a = a / s2
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a / a.sum()
                        conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        
        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(10 * np.pi * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    elif choice == 5:
        a = np.sin(np.pi * x) ** 0.8
    elif choice == 6:
        # Sinc-like (flat spectrum approx)
        t = np.linspace(-4 * np.pi, 4 * np.pi, n)
        a = np.sinc(t / (2 * np.pi)) ** 2  # sinc^2 = triangular spectrum
    else:
        # Random smooth
        k = rng.integers(3, 8)
        coeffs = rng.random(k)
        freqs = np.arange(1, k + 1)
        a = 1.0 + 0.3 * np.sum(c * np.cos(2 * np.pi * f * x) for c, f in zip(coeffs, freqs))
        a = np.maximum(a, 0.0)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
