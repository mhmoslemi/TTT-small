"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast FFT-based gradient descent on the true max autocorrelation peak
2. Softmax-smoothed multi-peak gradient with adaptive step size and momentum
3. Random restarts and mass-redistribution moves to escape local minima
4. Multi-length exploration (1600, 1920, 2048) with interpolation transfer
5. Final fine-grained polish phase with very small steps

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_a(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def peak_gradient(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a using vectorized ops."""
    n = len(a)
    grad = np.zeros(n)
    i_lo = max(0, k - (n - 1))
    i_hi = min(k, n - 1)
    idx = np.arange(i_lo, i_hi + 1)
    j = k - idx
    mask_neq = (j != idx)
    if mask_neq.any():
        grad[idx[mask_neq]] = 2.0 * a[j[mask_neq]]
    mask_eq = (j == idx)
    if mask_eq.any():
        grad[idx[mask_eq]] = 2.0 * a[idx[mask_eq]]
    return grad


def multi_peak_gradient(a, peaks, weights):
    """Weighted sum of gradients at specified peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):

        grad += w * peak_gradient(a, k)

    mom = momentum_start

        progress = min(1.0, elapsed_frac / max(time_limit, 1e-10))

                    a_try[j] += amt
                    a_try = normalize_a(a_try)
                    score_try = fast_autocorr(a_try).max()
                    if score_try < best_score:
                        best_score = score_try
                        best_a = a_try.copy()
                        a = a_try

        steps += 1
        if verbose and steps % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Polish step={steps} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def swap_optimization(a0, time_limit, seed=42, verbose=True):
    """Local search via pairwise mass swaps."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_a(a)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    stall = 0

    T = best_score * 0.01

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / max(time_limit, 1e-10))
        T = best_score * 0.01 * (0.1 ** progress)

        b = fast_autocorr(a)
        max_b = b.max()

        # Find top peaks to target
        top_k = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_k:
            # Find which pair contributes most to this peak
            i_lo = max(0, k - (n - 1))
            i_hi = min(k, n - 1)
            idx = np.arange(i_lo, i_hi + 1)
            contrib = a[idx] * np.maximum(a[k - idx], 0)

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            # Build peak mask
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                lo2 = max(0, k2 - (n - 1))
                hi2 = min(k2, n - 1)
                peak_mask[lo2:hi2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                continue

            # Choose destination: prefer low-value non-peak positions
            n_try = min(5, len(non_peak))
            candidates = non_peak[np.argsort(a[non_peak])][:n_try * 2]
            dst = candidates[rng.integers(len(candidates))]

            # Try different fractions
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = normalize_a(a_trial)
                score = fast_autocorr(a_trial).max()

                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    n_improve += 1
                    improved = True
                    break
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-12)):
                    a = a_trial
                    improved = True
                    break

        if not improved:
            stall += 1
            # Random swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.02, 0.1)
                a[i] -= amt
                a[j] += amt
                a = normalize_a(a)
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    n_improve += 1
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-12)):
                    pass
                stall = 0

            if stall > 500:
                # Larger perturbation
                n_pert = max(1, n // 200)
                sources = rng.choice(n, size=n_pert, replace=False)
                a_pert = a.copy()
                for s in sources:
                    amt = a_pert[s] * rng.uniform(0.2, 0.5)
                    t = rng.integers(n)
                    a_pert[s] -= amt
                    a_pert[t] += amt
                a_pert = normalize_a(a_pert)
                if fast_autocorr(a_pert).max() < best_score * 1.02:
                    a = a_pert
                stall = 0

        if verbose and n_improve % 2000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    Swap improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def adversarial_gradient_descent(a0, time_limit, seed=0, verbose=True):
    """GD that specifically targets the hardest-to-reduce peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_a(a)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    stall = 0
    v = np.zeros(n)

    # Track which peaks are "persistent" (hard to reduce)
    peak_history = {}

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / max(time_limit, 1e-10))
        T = best_score * (0.1 * (0.05 ** elapsed_frac))

        b = fast_autocorr(a)
        max_b = b.max()

        # Update peak history (which peaks remain high)
        top_20 = np.argsort(b)[::-1][:20]
        for k in top_20:
            if b[k] > max_b * 0.95:
                peak_history[k] = peak_history.get(k, 0) + 1
            else:
                if k in peak_history:
                    peak_history[k] = max(0, peak_history[k] - 1)

        # Weight peaks by persistence (harder peaks get more weight)
        top_count = min(12, len(top_20))
        top_indices = top_20[:top_count]
        persist = np.array([peak_history.get(k, 0) + 1 for k in top_indices], dtype=np.float64)
        # Also weight by how close to max
        closeness = np.array([(b[k] / max_b) ** 5 for k in top_indices])
        weights = persist * closeness
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.ones(top_count) / top_count
        else:
            weights = weights / w_sum

        # Compute gradient
        grad = multi_peak_gradient(a, top_indices, weights)

        # Reduce gradient component in the direction that increases sum
        # (to help with projection)
        grad_bias = grad.mean()
        grad = grad - grad_bias  # Remove mean to reduce sum-changing component

        grad_norm = np.sqrt(np.sum(grad ** 2))
        if grad_norm < 1e-15:
            stall += 1
            if stall > 200:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-6:
                    amt = a[i] * 0.05
                    a[i] -= amt
                    a[j] += amt
                    a = normalize_a(a)
                stall = 0
            continue

        # Adaptive step
        step = 0.3 / (n ** 0.5) * (max_b / (grad_norm + max_b))
        step = min(step, 0.2 / n)

        # Momentum
        mom = 0.9 + 0.08 * elapsed_frac
        v = mom * v + grad
        a_new = a - step * v
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_a(a_new)

        score = fast_autocorr(a_new).max()
        steps += 1

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50 and stall % 50 == 0:
                # Dampen momentum and try again
                v *= 0.5

        if stall > 400:
            # Escape: random redistribution
            n_move = max(2, n // 80)
            sources = rng.choice(n, size=n_move, replace=False)
            a_pert = a.copy()
            for s in sources:
                amt = a_pert[s] * rng.uniform(0.1, 0.4)
                t = rng.integers(n)
                a_pert[s] -= amt
                a_pert[t] += amt
            a_pert = normalize_a(a_pert)
            score_pert = fast_autocorr(a_pert).max()
            if score_pert < best_score * 1.03:
                a = a_pert
                v = np.zeros(n)
            stall = 0

        if verbose and steps % 3000 == 0 and steps > 0:
            elapsed = time.monotonic() - t0
            print(f"    AdvGD step={steps} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase gradient-based optimization with multi-length exploration."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize from parent or default
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting n={n_base}, score={best_score:.6f}")

    # Phase 1: Adversarial gradient descent at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: AdvGD n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 20:
        a_out, _ = adversarial_gradient_descent(best_seq, t1, seed=1, verbose=True)
        do_try_update(a_out, f"adv_n{n_base}")

    # Phase 2: Standard smoothed GD
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: SmoothGD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        n_cur = len(best_seq)
        a_out, _ = gradient_descent_phase(best_seq, t2, seed=2, verbose=True,
                                          T_start=best_score * 0.1, T_end=best_score * 0.005)
        do_try_update(a_out, f"sgd_n{n_cur}")

    # Phase 3: Swap optimization
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Swap n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 15:
        n_cur = len(best_seq)
        a_out, _ = swap_optimization(best_seq, t3, seed=3, verbose=True)
        do_try_update(a_out, f"swap_n{n_cur}")

    # Phase 4: Try longer length
    n2 = n_base + 320  # e.g., 1920
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Multi-length n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 20:
        init_long = interpolate_sequence(best_seq, n2)
        a_out, _ = adversarial_gradient_descent(init_long, t4, seed=4, verbose=True)
        do_try_update(a_out, f"adv_n{n2}")

        remaining = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining * 0.08, 100.0)
        if t4b > 15:
            n_cur = len(best_seq)
            a_out, _ = gradient_descent_phase(best_seq, t4b, seed=5, verbose=False,
                                              T_start=best_score * 0.05, T_end=best_score * 0.003)
            do_try_update(a_out, f"sgd_n{n_cur}")

    # Phase 5: Try power-of-2 length for FFT efficiency
    n3 = 2048
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: n={n3} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        init_2048 = interpolate_sequence(best_seq, n3)
        a_out, _ = adversarial_gradient_descent(init_2048, t5, seed=6, verbose=True)
        do_try_update(a_out, f"adv_n{n3}")

    # Phase 6: Fine polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Fine polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 10:
        a_out, _ = fine_polish(best_seq, t6, verbose=True)
        do_try_update(a_out, f"polish_n{n_best}")

    # Phase 7: Another adversarial GD round
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: AdvGD round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 180.0)
    if t7 > 15:
        a_out, _ = adversarial_gradient_descent(best_seq, t7, seed=7, verbose=True)
        do_try_update(a_out, f"adv2_n{n_best}")

    # Phase 8: Final fine polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 150.0)
    if t8 > 10:
        a_out, _ = fine_polish(best_seq, t8, verbose=True)
        do_try_update(a_out, f"finalpolish_n{n_best}")

    # Phase 9: One more swap round on the very best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final swap n={n_best} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.20, 150.0)
    if t9 > 10:
        a_out, _ = swap_optimization(best_seq, t9, seed=9, verbose=True)
        do_try_update(a_out, f"finalswap_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
