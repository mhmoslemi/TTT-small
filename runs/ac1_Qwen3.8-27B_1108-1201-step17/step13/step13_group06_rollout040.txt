import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as sp_minimize


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad(z, topk, T):
    """Compute LSE objective and gradient in z-space."""
    a = softmax_fn(z)
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    
    x_max = top_vals.max()
    shifted = T * (top_vals - x_max)
    exp_shifted = np.exp(shifted)
    lse_val = x_max + np.log(exp_shifted.sum()) / T
    
    w = exp_shifted / exp_shifted.sum()
    
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        idx_range = np.arange(js, je + 1)
        grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    obj = 2.0 * n * lse_val
    grad = 2.0 * n * grad_z
    
    return obj, grad

2. Power-sum (L4, L8) smooth objectives for diverse starting points
3. LP Nesterov zero-sum game for global exploration
4. Aggressive peak shaving for final polish
5. Multi-length exploration and re-optimization
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as sp_minimize


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def weighted_autocorr_grad(a, w):
    """
    Compute grad_a[i] = 2 * sum_j w[i+j] * a[j] for i=0,...,n-1.
    w has length 2n-1, a has length n.
    This is the gradient of sum_k w[k]*c[k] w.r.t. a, where c = a*a.
    """
    n = len(a)
    idx = np.arange(n)
    W_mat = w[idx[:, None] + idx[None, :]]
    return 2.0 * (W_mat @ a)


def lse_z_obj_grad(z, fixed_top, T, n):
    """LSE objective and gradient in z-space with fixed top indices."""
    a = softmax_fn(z)
    c = fast_autocorr(a)
    top_vals = c[fixed_top]

    x_max = top_vals.max()
    shifted = T * (top_vals - x_max)
    exp_shifted = np.exp(shifted)
    lse_val = x_max + np.log(exp_shifted.sum()) / T

    w_full = np.zeros(2 * n - 1)
    w_full[fixed_top] = exp_shifted / exp_shifted.sum()

    grad_a = weighted_autocorr_grad(a, w_full)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    obj = 2.0 * n * lse_val
    grad = 2.0 * n * grad_z
    return obj, grad


def power_sum_z_obj_grad(z, p, n):
    """Power-sum objective and gradient in z-space."""
    a = softmax_fn(z)
    c = np.maximum(fast_autocorr(a), 0.0)
    c_p = c ** p
    S = c_p.sum()
    if S < 1e-300:
        return 2.0 * n / (2 * n - 1), np.zeros(n)
    ps_val = S ** (1.0 / p)
    w = c ** (p - 1) / S ** (1.0 - 1.0 / p)

    grad_a = weighted_autocorr_grad(a, w)
    Sc = np.dot(grad_a, a)
    grad_z = a * (grad_a - Sc)

    obj = 2.0 * n * ps_val
    grad = 2.0 * n * grad_z
    return obj, grad


def get_top_indices(a, topk):
    """Get the top-k autocorrelation indices."""
    c = fast_autocorr(a)
    return np.argsort(c)[::-1][:topk]


def lbfgs_lse_anneal(a0, time_limit, T_start=100.0, T_end=0.05, n_temps=30, topk=120):
    """
    L-BFGS-B with multi-temperature LSE annealing and periodic index re-selection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    temps = np.geomspace(T_start, T_end, n_temps)

    for ti, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 3:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        a_curr = softmax_fn(z)
        fixed_top = get_top_indices(a_curr, topk)

        max_iter = max(40, int(remaining * 0.8))
        try:
            result = sp_minimize(
                lse_z_obj_grad,
                z.copy(),
                method='L-BFGS-B',
                args=(fixed_top, T, n),
                jac=True,
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-15,
                    'gtol': 1e-12,
                    'maxls': 60,
                },
            )
            if result.x is not None:
                z = result.x
        except Exception:
            pass

        a_curr = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a_curr).max()
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()

        if ti % 5 == 0 or ti == n_temps - 1:
            elapsed = time.monotonic() - t0
            print(f"      LSE T={T:.2f} ({elapsed:.0f}s): score={score:.6f}, best={best_score:.6f}")

    return best_a, best_score


def lbfgs_power_sum(a0, time_limit, p=4):
    """L-BFGS-B with power-sum objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    max_iter = max(100, int(time_limit * 2))
    try:
        result = sp_minimize(
            power_sum_z_obj_grad,
            z.copy(),
            method='L-BFGS-B',
            args=(p, n),
            jac=True,
            options={
                'maxiter': max_iter,
                'ftol': 1e-14,
                'gtol': 1e-10,
                'maxls': 50,
            },
        )
        if result.x is not None:
            z = result.x
    except Exception:
        pass

    a_curr = softmax_fn(z)
    score = 2.0 * n * fast_autocorr(a_curr).max()
    if score < best_score:
        best_score = score
        best_a = a_curr.copy()

    return best_a, best_score


def build_toeplitz(a_fixed):
    """Build (2n-1) x n Toeplitz matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS, cp.MOSEK]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(y, T_mat, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.CLARABEL, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_buf = []
    stall_count = 0
    warm_b = None
    last_print = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 2:
            break

        T_mat = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T_mat, solver, warm_b)
        if b_star is None:
            warm_b = None
            continue
        warm_b = b_star.copy()

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new
        avg_buf.append(a.copy())
        if len(avg_buf) > 50:
            avg_buf.pop(0)

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if len(avg_buf) >= 30 and t % 10 == 0:
            avg_check = np.mean(avg_buf[-30:], axis=0)
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if stall_count > 800 and t > 200:
            a = best_a.copy()
            y = best_a.copy()
            avg_buf = []
            stall_count = 0
            warm_b = None

        if t - last_print > 2000:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
            last_print = t

    return best_a, best_score


def peak_shaving_aggressive(a0, time_limit, seed=42):
    """
    Aggressive peak shaving with multiple strategies.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_peaks[:12]:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue

            # Find the most contributing source
            src = idx[np.argmax(contrib)]

            # Compute danger zone
            danger = np.zeros(n, dtype=bool)
            for kp in top_peaks[:15]:
                js2 = max(0, kp - (n - 1))
                je2 = min(n - 1, kp)
                danger[js2:je2 + 1] = True

            # Find safe destinations (prefer low-value positions outside danger)
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(10, len(safe_sorted) // 3)]

            if len(dst_pool) == 0:
                continue

            found = False
            for dst in dst_pool[:15]:
                for frac in [0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 15 == 0:
                # Random perturbation
                n_pert = rng.integers(2, 6)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.04)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                else:
                    a = best_a.copy()

            if stall > 300:
                break

    return best_a, best_score


def mirror_descent_refine(a0, time_limit, eta=0.008, seed=999):
    """Multiplicative weights refinement."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    current_eta = eta

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.99
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        importance = np.zeros(n)
        for k in top_peaks[:15]:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            importance[idx] += a[k - idx]

        imax = importance.max()
        if imax > 1e-15:
            importance = importance / imax

        updates = 1.0 - current_eta * importance
        updates = np.maximum(updates, 0.01)
        updates = updates / updates.mean()
        a_new = a * updates
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score - 1e-12:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                current_eta *= 0.7
                if current_eta < 1e-7:
                    current_eta = eta
                    a = best_a.copy()
                    noise = rng.normal(0, 0.02, n)
                    a = a * (1.0 + noise)
                    a = np.maximum(a, 0)
                    sn = a.sum()
                    if sn > 1e-10:
                        a = a / sn
            if stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_init(a0, rng, scale=0.08):
    """Create a perturbed version of a sequence."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining L-BFGS-B/LSE, power-sum, LP Nesterov,
    peak shaving, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 80.0, 1720.0)

    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None
        n_default = 1600
        best_seq = np.ones(n_default) / n_default
        best_score = evaluate_safe(best_seq)
        print(f"No parent, using uniform n={n_default}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)

    # === Phase 1: Power-sum p=4 (fast, smooth, different landscape) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Power-sum p=4, n={n_base} ===")
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 30:
        a_out, _ = lbfgs_power_sum(best_seq, t1, p=4)
        try_update(a_out, f"ps4_n{n_base}")

    # Power-sum p=8
    remaining = time_budget - (time.monotonic() - t_global)
    t1b = min(remaining * 0.08, 120.0)
    if t1b > 20:
        print(f"=== Phase 1b: Power-sum p=8, n={n_base} ===")
        a_out, _ = lbfgs_power_sum(best_seq, t1b, p=8)
        try_update(a_out, f"ps8_n{n_base}")

    # === Phase 2: Main L-BFGS-B LSE annealing (the heavy workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: L-BFGS-B LSE anneal, n={n_base} ===")
    t2 = min(remaining * 0.35, 600.0)
    if t2 > 60:
        a_out, _ = lbfgs_lse_anneal(best_seq, t2, T_start=80.0, T_end=0.03,
                                     n_temps=35, topk=150)
        try_update(a_out, f"lse_main_n{n_base}")

    # === Phase 3: L-BFGS-B LSE from power-sum result (different basin) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: L-BFGS-B LSE from PS start, n={n_base} ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 40:
        # Use a perturbed version of current best as starting point
        start_ps = perturbed_init(best_seq, rng, scale=0.03)
        a_out, _ = lbfgs_lse_anneal(start_ps, t3, T_start=40.0, T_end=0.1,
                                     n_temps=20, topk=120)
        try_update(a_out, f"lse_ps_start_n{n_base}")

    # === Phase 4: LP Nesterov for global exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: LP Nesterov, n={n_base} ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 50:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver)
        try_update(a_out, f"nlp_n{n_base}")

    # === Phase 5: Peak shaving (aggressive) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 5: Peak shaving, n={n_base} ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        a_out, _ = peak_shaving_aggressive(best_seq, t5, seed=123)
        try_update(a_out, f"peak_n{n_base}")

    # === Phase 6: Mirror descent refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 6: Mirror descent, n={n_base} ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 15:
        a_out, _ = mirror_descent_refine(best_seq, t6, eta=0.005, seed=777)
        try_update(a_out, f"mirror_n{n_base}")

    # === Phase 7: Final LSE polish (very low temperature, many iterations) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 7: Final LSE polish, n={n_base} ===")
    t7 = min(remaining * 0.12, 200.0)
    if t7 > 20:
        a_out, _ = lbfgs_lse_anneal(best_seq, t7, T_start=10.0, T_end=0.01,
                                     n_temps=20, topk=200)
        try_update(a_out, f"lse_polish_n{n_base}")

    # === Phase 8: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        n_long = n_base + 300
        t8 = min(remaining * 0.15, 150.0)
        if t8 > 40:
            print(f"=== Phase 8: LSE at n={n_long} ===")
            init_long = interpolate_sequence(best_seq, n_long)
            a_out, _ = lbfgs_lse_anneal(init_long, t8, T_start=50.0, T_end=0.1,
                                         n_temps=15, topk=120)
            try_update(a_out, f"lse_long_n{n_long}")

    # === Phase 9: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t9 = min(remaining * 0.5, 100.0)
        if t9 > 10:
            print(f"=== Phase 9: Final peak shaving ===")
            a_out, _ = peak_shaving_aggressive(best_seq, t9, seed=2024)
            try_update(a_out, "peak_final")

    # === Phase 10: Ultra-fine LSE on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t10 = min(remaining * 0.8, 80.0)
        if t10 > 10:
            print(f"=== Phase 10: Ultra-fine LSE ===")
            a_out, _ = lbfgs_lse_anneal(best_seq, t10, T_start=5.0, T_end=0.005,
                                         n_temps=15, topk=250)
            try_update(a_out, "ultra_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
