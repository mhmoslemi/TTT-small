def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_obj_grad(a, T):
    """Compute LSE objective value and gradient."""
    n = len(a)
    c = fast_autocorr(a)  # length 2n-1
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    log_sum = np.log(exp_Tc.sum())
    obj = (Tc_max + log_sum) / T  # = (1/T)*log(sum exp(T*c))
    
    w = exp_Tc / exp_Tc.sum()  # softmax weights, length 2n-1
    
    # Gradient: grad[i] = 2 * sum_j a[j] * w[i+j] for i=0..n-1
    # = 2 * IFFT(conj(FFT(a_pad)) * FFT(w_pad))[i]
    fft_size = 1 << (2*n - 2).bit_length()
    A_pad = np.zeros(fft_size)
    A_pad[:n] = a
    W_pad = np.zeros(fft_size)
    W_pad[:2*n-1] = w
    
    AF = np.fft.rfft(A_pad)
    WF = np.fft.rfft(W_pad)
    grad = 2.0 * np.fft.irfft(np.conj(AF) * WF, fft_size)[:n]
    
    return obj, grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort-based projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def fista_lse(a0, time_limit, T_init=10.0, T_end=0.1, n_iters_max=50000):
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    n_stages = 20
    T_schedule = np.geomspace(T_init, T_end, n_stages)
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        T = T_schedule[stage]
        lr = 1.0  # Will be adjusted by backtracking
        
        for it in range(n_iters_max):
            if time.monotonic() - t0 > time_limit:
                break
            
            obj, grad = compute_lse_obj_grad(y, T)
            
            # Project gradient onto tangent space of simplex
            grad_proj = grad - np.mean(grad)
            
            # Backtracking line search
            step = lr
            y_new = y - step * grad_proj
            x_new = project_simplex(y_new)
            
            # Check decrease
            obj_new, _ = compute_lse_obj_grad(x_new, T)
            while obj_new > obj + step * np.dot(grad_proj, x_new - y) + 0.5 * step**2 * np.dot(grad_proj, grad_proj) and step > 1e-10:
                step /= 2
                y_new = y - step * grad_proj
                x_new = project_simplex(y_new)
                obj_new, _ = compute_lse_obj_grad(x_new, T)
            
            # Nesterov update
            t_new = 0.5 * (1 + np.sqrt(1 + 4 * t * t))
            momentum = (t - 1) / t_new
            y = x_new + momentum * (x_new - x)
            x = x_new
            t = t_new
            lr = step  # Keep the step for next iteration
            
            # Track best
            score = 2.0 * n * fast_autocorr(x).max()
            if score < best_score:
                best_score = score
                best_a = x.copy()
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with LogSumExp-smoothed max-autocorrelation (FFT-accelerated, O(n log n)/iter)
2. Nesterov LP zero-sum game with fast solver (CLARABEL/PDLCP) and constraint generation
3. Peak shaving with danger-zone analysis and global verification
4. Multi-length exploration with interpolation
5. Coordinate descent polish
6. Multi-start with structured initializations

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_obj_grad(a, T):
    """Compute LSE objective value and gradient of (1/T)*log(sum_k exp(T*c_k))."""
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    log_sum = np.log(exp_Tc.sum())
    obj = (Tc_max + log_sum) / T
    w = exp_Tc / exp_Tc.sum()
    # Gradient: grad[i] = 2 * sum_j a[j] * w[i+j]
    fft_size = 1 << (2 * n - 2).bit_length()
    A_pad = np.zeros(fft_size)
    A_pad[:n] = a
    W_pad = np.zeros(fft_size)
    W_pad[:2 * n - 1] = w
    AF = np.fft.rfft(A_pad)
    WF = np.fft.rfft(W_pad)
    grad = 2.0 * np.fft.irfft(np.conj(AF) * WF, fft_size)[:n]
    return obj, grad


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_lse(a0, time_limit, T_init=15.0, T_end=0.05, n_stages=30):
    """FISTA with adaptive-temperature LSE smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_init, T_end, n_stages)
    x = a.copy()
    y = a.copy()
    t_mom = 1.0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit - 2:
            break
        T = temps[stage]
        lr = 0.8 / T
        x = x.copy()
        y = x.copy()
        t_mom = 1.0
        max_it = 50000
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 2:
                break
            obj_y, grad_y = compute_lse_obj_grad(y, T)
            grad_proj = grad_y - np.mean(grad_y)
            x_new = project_simplex(y - lr * grad_proj)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
            y = np.maximum(y, 0)
            yn = y.sum()
            if yn > 1e-15:
                y = y / yn
            else:
                y = x_new.copy()
            x = x_new
            t_mom = t_new
            if (it + 1) % 40 == 0:
                score = 2.0 * n * fast_autocorr(x).max()
                if score < best_score:
                    best_score = score
                    best_a = x.copy()
                score_y = 2.0 * n * fast_autocorr(y).max()
                if score_y < best_score:
                    best_score = score_y
                    best_a = y.copy()
        score = 2.0 * n * fast_autocorr(x).max()
        if score < best_score:
            best_score = score
            best_a = x.copy()
    return best_a, best_score


def build_toeplitz(a_fixed):
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(y, T, solver, warm_b=None):
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    sb = b_val.sum()
    if sb < 1e-15:
        return None, None
    b_val = b_val / sb
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    avg_buf = np.zeros(n)
    n_avg = 0
    stall = 0
    warm_b = None
    last_print = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        T = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T, solver, warm_b)
        if b_star is None:
            warm_b = None
            continue
        warm_b = b_star
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        y = y_new
        a = a_new
        avg_buf += a
        n_avg += 1
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 800 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            avg_buf = np.zeros(n)
            n_avg = 0
            stall = 0
            warm_b = None
        if n_avg % 50 == 0 and n_avg >= 100:
            avg_check = avg_buf / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall = 0
        if t - last_print > 2000:
            print(f"      LP t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
            last_print = t

def peak_shaving(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.985
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]
        danger = np.zeros(n, dtype=bool)
        for k in top_peaks[:20]:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            danger[js:je + 1] = True
        improved = False
        for k in top_peaks[:15]:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue
            found = False
            for dst in dst_pool[:10]:
                for frac in [0.003, 0.007, 0.015, 0.03, 0.05, 0.08, 0.12]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if stall % 15 == 0:
                n_pert = rng.integers(1, 6)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.04)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break
    return best_a, best_score


def coord_descent_polish(a0, time_limit, seed=123):
    """Coordinate descent: optimize pairs of coordinates to reduce peak."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        k_peak = np.argmax(b)
        js = max(0, k_peak - (n - 1))
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        # Try moving mass within the peak-contributing indices
        improved = False
        for _ in range(200):
            if time.monotonic() - t0 > time_limit - 2:
                break
            i = idx[rng.integers(len(idx))]
            j = rng.integers(n)
            if i == j:
                continue
            # Try multiple amounts
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[i] * frac
                if amt < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-12:
                    continue
                a_trial = a_trial / st
                nm = fast_autocorr(a_trial).max()
                if nm < bmax * (1 - 1e-10):
                    a = a_trial
                    sc = 2.0 * n * nm
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    improved = True
                    bmax = nm
                    stall = 0
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if stall > 100:
                break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.1
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.6 * np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2 + 0.05
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

            pert = best_seq + rng.normal(0, 0.03, len(best_seq)) * (best_seq + 1e-6)
            pert = np.maximum(pert, 0)
            ps = pert.sum()
            if ps > 1e-10:
                pert = pert / ps
            a_out, _ = fista_lse(pert, t_each, T_init=10.0, T_end=0.1, n_stages=15)
            try_update(a_out, f"multi_{si}")

    # Phase 10: Final LP polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        t10 = min(remaining * 0.35, 200.0)
        print(f"=== Phase 10: Final LP polish n={len(best_seq)} ===")
        a_out, _ = nesterov_lp_optimize(best_seq, t10, solver)
        try_update(a_out, "lp_final")

    # Phase 11: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t11 = min(remaining * 0.5, 50.0)
        if t11 > 5:
            print(f"=== Phase 11: Final peak shave ===")
            a_out, _ = peak_shaving(best_seq, t11, seed=2024)
            try_update(a_out, "peak_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0] * 1600

    return [float(x) for x in best_seq]
