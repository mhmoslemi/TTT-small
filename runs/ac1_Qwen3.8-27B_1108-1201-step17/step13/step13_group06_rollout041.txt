    """Full linear autocorrelation a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_obj_grad_a(a, T):
    """
    Compute the LSE objective value and gradient w.r.t. a.
    LSE = (1/T) * log(sum_k exp(T * c[k]))
    grad[i] = sum_k w[k] * d(c[k])/d(a[i]) where w = softmax(T*c)
    """
    n = len(a)
    c = fast_autocorr(a)
    max_c = c.max()
    Tc = T * (c - max_c)
    exp_Tc = np.exp(Tc)
    sum_exp = exp_Tc.sum()
    lse_val = max_c + np.log(sum_exp) / T
    w = exp_Tc / sum_exp
    
    # Gradient computation via FFT
    N = 1 << (3 * n).bit_length()
    W_hat = rfft(w, N)
    A_hat = rfft(a, N)
    A_rev_hat = rfft(a[::-1], N)
    
    # f2[i] = (w * a)[i] = sum_k w[k]*a[i-k]
    f2 = irfft(W_hat * A_hat, N)[:n]
    # f1[i] = (w * a_rev)[n-1+i] = sum_k w[k]*a[k-i]
    conv_w_arev = irfft(W_hat * A_rev_hat, N)
    f1 = conv_w_arev[n-1:2*n-1]
    
    grad = f1 + f2
    return lse_val, grad

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def score_of(a):
    """Compute the actual objective: 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (s * s)

def adam_lse_optimize(a0, time_limit, T_start=200.0, T_end=0.3, n_stages=40, 
                      lr_start=0.05, lr_end=0.001, restart_interval=500, seed=42):
    """
    Adam-optimized full LSE with multi-stage temperature annealing.
    Works in z-space where a = softmax(z).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = score_of(a_init)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    lrs = np.linspace(lr_start, lr_end, n_stages)
    
    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_T = temps[si]
        stage_lr = lrs[si]
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        
        # How many iterations for this stage
        max_it = max(50, int(remaining * 3))  # ~3 iterations per second initially
        stall = 0
        stage_best_score = best_score
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break
            
            a = softmax_fn(z)
            lse_val, grad_a = lse_obj_grad_a(a, stage_T)
            
            # Convert gradient to z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Adam update
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - stage_lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                sc = score_of(a_check)
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
                    stage_best_score = sc
                    stall = 0
                else:
                    stall += 1
                
                if stall > 80:
                    # Restart from best
                    z = np.log(np.maximum(best_a, 1e-30)) + rng.normal(0, 0.01, n)
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0
                    stall = 0
        
        # End of stage: check and update
        a_stage = softmax_fn(z)
        sc = score_of(a_stage)
        if sc < best_score:
            best_score = sc
            best_a = a_stage.copy()
        
        elapsed = time.monotonic() - t0
        if si % 8 == 0:
            print(f"    LSE stage {si+1}/{n_stages} (T={stage_T:.1f}, lr={stage_lr:.4f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def peak_shave_fft(a0, time_limit, seed=42):
    """Peak shaving using FFT-based gradient with aggressive step sizes."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        bmax = c.max()
        k_peak = c.argmax()
        
        # Compute gradient of c[k_peak] w.r.t. a
        # d(c[k])/d(a[i]) = a[k-i] + a[i-k] (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            ki = k_peak - i
            if 0 <= ki < n:
                val += a[ki]
            ik = i - k_peak
            if 0 <= ik < n:
                val += a[ik]
            grad[i] = val
        
        # Find the best single-element move
        # Try moving mass from high-grad elements to low-grad elements
        grad_sorted = np.argsort(grad)
        # High grad indices (contribute most to peak)
        top_src = np.argsort(grad)[::-1][:20]
        # Low grad indices (don't contribute to peak)
        bottom_dst = grad_sorted[:20]
        
        improved = False
        for src in top_src:
            if a[src] < 1e-10:
                continue
            for dst in bottom_dst:
                if src == dst:
                    continue
                for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-10):
                        a = a_trial
                        sc = 2.0 * n * new_max
                        if sc < best_score:
                            best_score = sc
                            best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try random perturbation
            a_trial = a.copy()
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                sc = score_of(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                elif time.monotonic() - t0 > time_limit * 0.8:
                    break
            elif time.monotonic() - t0 > time_limit * 0.8:
                break
    
    return best_a, best_score

4. Targeted peak shaving using FFT-based directional moves
5. Multi-length exploration with interpolation and re-optimization
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full linear autocorrelation a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_obj_grad_a(a, T):
    """
    Compute LSE objective and gradient w.r.t. a.
    LSE = (1/T)*log(sum_k exp(T*c[k])) where c = autocorrelation(a).
    Returns (lse_value, gradient_vector).
    """
    n = len(a)
    c = fast_autocorr(a)
    max_c = c.max()
    Tc = T * (c - max_c)
    exp_Tc = np.exp(Tc)
    sum_exp = exp_Tc.sum()
    lse_val = max_c + np.log(sum_exp) / T
    w = exp_Tc / sum_exp

    # FFT-based gradient: grad[i] = sum_k w[k]*(a[k-i] + a[i-k])
    N = 1 << (3 * n).bit_length()
    W_hat = rfft(w, N)
    A_hat = rfft(a, N)
    A_rev_hat = rfft(a[::-1], N)

    f2 = irfft(W_hat * A_hat, N)[:n]
    conv_w_arev = irfft(W_hat * A_rev_hat, N)
    f1 = conv_w_arev[n - 1:2 * n - 1]

    grad = f1 + f2
    return lse_val, grad

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def score_of(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (s * s)

def normalize_a(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.8 + 0.15 * np.cos(2 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * (x - 0.3) ** 2) + 0.5 * np.exp(-6 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (0.5 + 0.5 * np.cos(4 * np.pi * x)) ** 0.5
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - np.abs(x) ** 0.8)
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = np.exp(-30 * x ** 2) + np.exp(-30 * (x - 1) ** 2) + 0.1
    elif choice == 9:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.5) ** 2)
    elif choice == 10:
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.5 + 5.0 * np.sin(np.pi * x) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-1.5 * (x - 0.5) ** 2) * (1 + 0.3 * np.cos(8 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def perturbed_init(a0, rng, scale=0.05):
    """Create a perturbed version of a sequence."""
    a = normalize_a(a0)
    noise = rng.normal(0, scale, len(a))
    a_new = a * (1.0 + noise)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn

def adam_lse_optimize(a0, time_limit, T_start, T_end, n_stages,
                      lr_start, lr_end, seed, label=""):
    """
    Adam-optimized full LSE with multi-stage temperature annealing.
    Works in z-space where a = softmax(z).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)

    a_init = normalize_a(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = score_of(a_init)

    temps = np.geomspace(T_start, T_end, n_stages)
    lrs = np.linspace(lr_start, lr_end, n_stages)

    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_T = temps[si]
        stage_lr = lrs[si]

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        stall = 0

        max_it = max(50, int(remaining * 4))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            a = softmax_fn(z)
            lse_val, grad_a = lse_obj_grad_a(a, stage_T)

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - stage_lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                sc = score_of(a_check)
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
                    stall = 0
                else:
                    stall += 1

                if stall > 100:
                    z = np.log(np.maximum(best_a, 1e-30)) + rng.normal(0, 0.005, n)
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0
                    stall = 0

        a_stage = softmax_fn(z)
        sc = score_of(a_stage)
        if sc < best_score:
            best_score = sc
            best_a = a_stage.copy()

        if si % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-{label} stage {si+1}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def peak_shave_fft(a0, time_limit, seed=42):
    """Peak shaving using FFT-based autocorrelation checks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_a(a0)
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        bmax = c.max()

        # Find top peaks
        threshold = bmax * 0.995
        top_peaks = np.where(c >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(c)[::-1][:5]

        improved = False
        for k in top_peaks[:8]:
            # Compute gradient of c[k]
            grad = np.zeros(n)
            for i in range(n):
                ki = k - i
                ik = i - k
                val = 0.0
                if 0 <= ki < n:
                    val += a[ki]
                if 0 <= ik < n:
                    val += a[ik]
                grad[i] = val

            # High-grad sources (contribute to peak)
            top_src = np.argsort(grad)[::-1][:15]
            # Low-grad destinations
            bottom_dst = np.argsort(grad)[:15]

            for src in top_src:
                if a[src] < 1e-10:
                    continue
                for dst in bottom_dst:
                    if src == dst:
                        continue
                    for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                        amt = a[src] * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-12:
                            continue
                        a_trial = a_trial / st
                        new_max = fast_autocorr(a_trial).max()
                        if new_max < bmax * (1.0 - 1e-10):
                            a = a_trial
                            sc = 2.0 * n * new_max
                            if sc < best_score:
                                best_score = sc
                                best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve % 20 == 0:
                a_trial = a.copy()
                for _ in range(5):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-8:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    sc = score_of(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        no_improve = 0
            if no_improve > 200:
                break

    return best_a, best_score

def find_solver():
    """Find a working LP solver."""
    try:
        import cvxpy as cp
        for solver_name in [
            getattr(cp, 'MOSEK', None),
            getattr(cp, 'GUROBI', None),
            getattr(cp, 'CLARABEL', None),
            getattr(cp, 'SCS', None),
            getattr(cp, 'ECOS', None),
            getattr(cp, 'PDLP', None),
            getattr(cp, 'GLPK', None),
        ]:
            if solver_name is None:
                continue
            try:
                x = cp.Variable(5, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    return solver_name
            except Exception:
                continue
    except ImportError:
        pass
    return None

def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP best-response zero-sum game."""
    import cvxpy as cp

    n = len(a0)
    a = normalize_a(a0)
    y = a.copy()

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    warm_b = None
    avg_buf = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Build Toeplitz matrix T where T@b = convolve(y, b)
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y

        # Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0
        b_var = cp.Variable(n, nonneg=True)
        if warm_b is not None:
            b_var = cp.Variable(n, nonneg=True, value=warm_b)
        M_var = cp.Variable(value=1.0 / n)
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        resolved = False
        for sv in [solver, cp.SCS, cp.ECOS, cp.PDLP]:
            try:
                prob.solve(solver=sv)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    resolved = True
                    break
            except Exception:
                continue
        if not resolved:
            warm_b = None
            continue

        b_star = np.asarray(b_var.value, dtype=np.float64)
        b_star = np.maximum(b_star, 0)
        s_b = b_star.sum()
        if s_b < 1e-15:
            warm_b = None
            continue
        b_star = b_star / s_b
        warm_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        avg_buf += a
        n_avg += 1

        score_a = score_of(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if n_avg % 20 == 0 and n_avg >= 40:
            avg_check = avg_buf / n_avg
            score_avg = score_of(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            M_val = float(M_var.value)
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.6f}")

    return best_a, best_score

def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
