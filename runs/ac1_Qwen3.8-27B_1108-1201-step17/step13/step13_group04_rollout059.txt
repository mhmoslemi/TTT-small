```python
import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_grad(a, T):
    """Gradient of LSE surrogate T*log(sum exp(b_k/T)) where b=autocorr(a)."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2*n-1)
        w[n-1] = 1.0
    else:
        w /= w_sum
    
    # gradient[m] = sum_k w_k * (a_{k-m} + a_{m-k})
    # = (a_rev * w)[m+n-1] + (a * w)[m]

    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    
    r1 = conv1[n-1:n-1+n]
    r2 = conv2[:n]
    
    return r1 + r2
```

```python
def interpolate_to(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss
```

```python
def autocorr_delta(a, k, p, q, delta):
    """Compute change in b[k] when moving delta from position p to position q."""
    n = len(a)
    db = 0.0
    # Remove from p
    if 0 <= k - p < n:
        db -= delta * a[k - p]
    if 0 <= p - k < n:
        db -= delta * a[p - k]
    # Add to q
    if 0 <= k - q < n:
        db += delta * a[k - q]
    if 0 <= q - k < n:
        db += delta * a[q - k]
    # Cross term (only if p and q pair up for this k)
    # b[k] includes a[p]*a[k-p], after move it's (a[p]-δ)*a[k-p] if k-p != q
    # and a[q]*a[k-q], after move it's a[q+δ]*a[k-q] if k-q != p
    # The δ^2 term: if k-p == q, then the term a[p]*a[k-p] becomes (a[p]-δ)*(a[q]+δ)
    # This introduces a -δ^2 term
    if k - p == q and 0 <= k - p < n:
        db -= delta * delta
    if p - k == q and 0 <= p - k < n:
        db -= delta * delta
    return db
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LSE gradient descent with adaptive cooling and backtracking line search
2. Gradient-informed peak targeting with O(1) delta evaluation

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation via FFT. Returns array of length 2n-1."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:2 * n - 1]


def lse_value(a, T):
    """Compute LSE surrogate value: T * log(sum_k exp(b_k/T))."""
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < -1e300:
        return 0.0
    shifted = (b - max_b) / T
    shifted = np.clip(shifted, -700, 700)
    return T * (max_b / T + np.log(np.sum(np.exp(shifted))))


def lse_gradient(a, T):
    """
    Gradient of LSE(a; T) = T * log(sum_k exp(b_k / T)) where b = full autocorrelation of a.
    grad[m] = sum_k w_k * (a_{k-m} + a_{m-k}) where w_k = exp(b_k/T) / sum_j exp(b_j/T).
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    exp_arg = np.clip((b - max_b) / T, -700, 700)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 3).bit_length()
    A_rev = rfft(a[::-1], fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]
    return r1 + r2

    y_prev = a.copy()
    avg = a.copy()

    global_it = 0

        stage_it = 0

                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial / s_t
                        maxb_trial = fast_autocorr(a_trial).max()
                        if maxb_trial < best_maxb:
                            a = a_trial
                            best_maxb = maxb_trial
                            best_a = a.copy()
                            stall = 0
            if stall % 200 == 0 and verbose:
                print(f"  PeakSearch stalled at {best_maxb:.10f} ({time.monotonic()-t0:.0f}s)")

        if verbose and it_count > 0 and it_count % 300 == 0:
            print(f"  PeakSearch it={it_count} ({time.monotonic()-t0:.0f}s): best={best_maxb:.10f}")

    return best_a, best_maxb


def interpolate_to(a, new_n):
    """Interpolate a sequence to a new length, normalized to sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / ss


def subgradient_polish(a0, time_limit, verbose=True):
    """
    Direct optimization of max(b) using a smooth-max approximation with
    very low temperature and Nesterov acceleration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_maxb = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    it = 0
    T = 0.05

    while time.monotonic() - t0 < time_limit:
        if it > 50000:
            break
        it += 1

        if it <= 1:
            y_ext = y.copy()
        else:
            t_k = float(it)
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            beta = (t_k - 1.0) / t_next
            y_ext = y + beta * (y - y_prev)
            y_ext = np.maximum(y_ext, 0.0)
            s_ext = y_ext.sum()
            if s_ext > 1e-15:
                y_ext = y_ext / s_ext

        grad = lse_gradient(y_ext, T)
        g_norm = float(np.abs(grad).max())
        if g_norm < 1e-30:
            break

        step = 1.0 / (it * g_norm + 1.0)
        y_prev = y.copy()
        y_new = y_ext - step * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        y = y_new

        if it % 50 == 0:
            maxb = fast_autocorr(y).max()
            if maxb < best_maxb:
                best_maxb = maxb
                best_a = y.copy()
        if verbose and it % 500 == 0:
            print(f"  Subgrad it={it} ({time.monotonic()-t0:.0f}s): best={best_maxb:.10f}")

    maxb = fast_autocorr(y).max()
    if maxb < best_maxb:
        best_maxb = maxb
        best_a = y.copy()

    return best_a, best_maxb


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Combines LSE Nesterov, peak targeting, and multi-length
    exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = safe_evaluate(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n0 = 1600
        best_seq = np.ones(n0) / n0
        best_score = safe_evaluate(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")

    # Inline update helper (avoids closure)
    state = {'seq': best_seq, 'score': best_score}

    # ============ Phase 1: LSE Nesterov on parent length ============
    n_curr = len(state['seq'])
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.32, 500.0)
    print(f"\n=== Phase 1: LSE Nesterov n={n_curr} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lse(state['seq'].copy(), t1, T_start=25.0, T_end=0.03, n_stages=30, verbose=True)
        sc = safe_evaluate(a_out)
        if np.isfinite(sc) and sc < state['score']:
            state['score'] = sc
            state['seq'] = a_out.copy()
            print(f"  *** NEW BEST [lse_n{n_curr}]: {sc:.6f} ({time.monotonic()-t_global:.0f}s)")

    # ============ Phase 2: Peak targeting ============
    n_curr = len(state['seq'])
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.18, 280.0)
    print(f"\n=== Phase 2: Peak Targeting n={n_curr} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_out, _ = peak_targeting(state['seq'].copy(), t2, verbose=True)
        sc = safe_evaluate(a_out)
        if np.isfinite(sc) and sc < state['score']:
            state['score'] = sc
            state['seq'] = a_out.copy()
            print(f"  *** NEW BEST [peak_n{n_curr}]: {sc:.6f} ({time.monotonic()-t_global:.0f}s)")

    # ============ Phase 3: LSE at longer length ============
    n_long = n_curr + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.22, 380.0)
    print(f"\n=== Phase 3: LSE n={n_long} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_init = interpolate_to(state['seq'], n_long)
        a_out, _ = nesterov_lse(a_init, t3, T_start=20.0, T_end=0.03, n_stages=25, verbose=True)
        sc = safe_evaluate(a_out)
        if np.isfinite(sc) and sc < state['score']:
            state['score'] = sc
            state['seq'] = a_out.copy()
            print(f"  *** NEW BEST [lse_n{n_long}]: {sc:.6f} ({time.monotonic()-t_global:.0f}s)")

    # ============ Phase 4: Peak targeting on best ============
    n_curr = len(state['seq'])
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 220.0)
    print(f"\n=== Phase 4: Peak Targeting n={n_curr} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = peak_targeting(state['seq'].copy(), t4, verbose=True)
        sc = safe_evaluate(a_out)
        if np.isfinite(sc) and sc < state['score']:
            state['score'] = sc
            state['seq'] = a_out.copy()
            print(f"  *** NEW BEST [peak_n{n_curr}]: {sc:.6f} ({time.monotonic()-t_global:.0f}s)")

    # ============ Phase 5: Subgradient polish ============
    n_curr = len(state['seq'])
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: Subgradient polish n={n_curr} (budget={t5:.0f}s) ===")
    if t5 > 10:

            print(f"  *** NEW BEST [subgrad_n{n_curr}]: {sc:.6f} ({time.monotonic()-t_global:.0f}s)")

    # ============ Phase 6: Final LSE polish (low T) ============
    n_curr = len(state['seq'])
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 6: Final LSE polish n={n_curr} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = nesterov_lse(state['seq'].copy(), t6, T_start=3.0, T_end=0.01, n_stages=15, verbose=True)
        sc = safe_evaluate(a_out)
        if np.isfinite(sc) and sc < state['score']:
            state['score'] = sc
            state['seq'] = a_out.copy()
            print(f"  *** NEW BEST [lse_polish]: {sc:.6f} ({time.monotonic()-t_global:.0f}s)")

    # ============ Phase 7: Final peak targeting ============
    n_curr = len(state['seq'])
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.6, 150.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Final Peak n={n_curr} (budget={t7:.0f}s) ===")
        a_out, _ = peak_targeting(state['seq'].copy(), t7, verbose=True)
        sc = safe_evaluate(a_out)
        if np.isfinite(sc) and sc < state['score']:
            state['score'] = sc
            state['seq'] = a_out.copy()
            print(f"  *** NEW BEST [peak_final]: {sc:.6f} ({time.monotonic()-t_global:.0f}s)")

    # Final evaluation
    if state['seq'] is not None:
        final_score = safe_evaluate(state['seq'])
        if np.isfinite(final_score):
            state['score'] = final_score
        print(f"\nFINAL: score={state['score']:.6f}, n={len(state['seq'])}, time={time.monotonic()-t_global:.0f}s")
        return [float(x) for x in state['seq']]
    else:
        print("ERROR: No valid sequence!")
        return [1.0]
```
