- After seeing a_t, B updates: q_{t+1,k} ∝ q_{t,k} * exp(eta * (a_t*a_t)_k)
- A plays best response to q_t

This gives an O(1/eta + 1/(eta*T)) approximation after T iterations.

But actually, for our purpose, we don't need the game value - we just need a good a. So I can:
1. Run the game for T iterations
2. Return the best a seen (or the average a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping + diverse initializations
2. Zero-sum game with QP oracle (multiplicative weight updates)
3. Smoothed subgradient with Nesterov acceleration
4. Peak shaving coordinate descent
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:n]

def internal_score(a):
    """Compute 2*n*max(b) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result

def qp_best_response(q, max_iters=30, lr=None):
    """Solve min sum_k q_k (a*a)_k s.t. sum(a)=1, a>=0 using projected gradient."""
    n = len(q) // 2 + 1  # q has length 2n-1, so n = (len(q)+1)//2
    # Actually, q has 2n-1 entries, so n = (len(q)+1)//2
    n = (len(q) + 1) // 2
    
    a = np.ones(n) / n  # initialize with uniform
    
    # Compute initial gradient
    def compute_grad(a):
        q_rev = q[::-1]
        c = fast_convolve(a, q_rev)
        return 2.0 * c[n-1:2*n-1][::-1]
    
    grad = compute_grad(a)
    
    # Estimate Lipschitz constant (rough)
    # The Hessian is 2*Q where Q_{j,l} = q_{j+l}
    # lambda_max(Q) <= sum_j |row_j| <= n * max(q) * 2 (rough)
    L = 4.0 * n * q.max() + 0.1
    
    if lr is None:
        lr = 1.0 / (L * 2.0)
    
    for it in range(max_iters):
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        if np.allclose(a_new, a, atol=1e-12):
            a = a_new
            break
        a = a_new
        grad = compute_grad(a)
    
    return a

def game_optimize(a0, time_limit, eta=0.1, qp_iters=20, verbose=True):
    """Zero-sum game with QP oracle and multiplicative weight updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # B's strategy (distribution over lags)
    q = np.ones(2 * n - 1) / (2 * n - 1)
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    # Accumulators for time-averaged iterate
    avg_a = np.zeros(n)
    T_count = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # A's best response to q (approximate QP)
        a_br = qp_best_response(q, max_iters=qp_iters)
        
        # Update a with momentum
        alpha_t = 2.0 / (t + 2.0)
        a_momentum = (1 - alpha_t) * a + alpha_t * a_br
        a = project_simplex(a_momentum)
        
        # B's multiplicative weight update
        b = fast_autocorr(a)
        # Smoothed weights (temperature annealing)
        T_smooth = max(0.01, 1.0 / (1.0 + 0.001 * t))
        log_q = np.log(q + 1e-300)
        log_q += eta * b / T_smooth
        log_q -= log_q.max()
        q = np.exp(log_q)
        q = q / q.sum()
        
        # Track best
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Time average
        avg_a += a
        T_count += 1
        if T_count % 50 == 0:
            avg = avg_a / T_count
            score_avg = internal_score(avg)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Game t={t} ({elapsed:.0f}s): score={score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

def nesterov_subgrad(a0, time_limit, T_start=50.0, T_end=0.1, beta=0.9, base_lr=0.05, verbose=True):
    """Smoothed subgradient with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealed temperature
        frac = min(1.0, t / max(1, 10000))
        T = T_start * (T_end / T_start) ** frac
        
        b = fast_autocorr(y)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        # Gradient
        q_rev = w[::-1]
        c = fast_convolve(y, q_rev)
        grad = 2.0 * c[n-1:2*n-1][::-1]
        
        # Nesterov step
        v = beta * v + grad
        lr_t = base_lr / (1.0 + 0.01 * t)
        y_new = y - lr_t * v
        y_new = project_simplex(y_new)
        y = y_new
        
        score = internal_score(y)
        if score < best_score:
            best_score = score
            best_a = y.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): score={score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = internal_score(a)
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            src = idx[np.argmax(contrib)]
            
            # Find a good destination (low contribution to all top peaks)
            dst_candidates = []
            for j in range(n):
                if j == src:
                    continue
                # Estimate how much adding to j would increase peaks
                peak_increase = 0
                for k2 in top_k:
                    if 0 <= k2 - j < n and 0 <= j < n:
                        peak_increase += a[k2 - j]
                dst_candidates.append((peak_increase, j))
            
            dst_candidates.sort()
            dst = dst_candidates[rng.integers(0, min(10, len(dst_candidates)))][1]
            
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 300:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping for initialization
2. Zero-sum game with QP oracle (multiplicative weight updates, no cvxpy)
3. Smoothed subgradient with Nesterov acceleration
4. Peak shaving coordinate descent for final refinement
Key: QP oracle uses FFT-based projected gradient (O(n log n) per step),
massively faster than LP, enabling many more iterations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:n]


def internal_score(a):
    """Compute 2*n*max(b) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} using O(n log n) sort."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def qp_gradient(a, q):
    """Compute gradient of sum_k q_k*(a*a)_k w.r.t. a using FFT.
    grad_m = 2 * sum_j a_j * q[m+j]
    """
    n = len(a)
    q_rev = q[::-1]
    c = fast_convolve(a, q_rev)
    # grad_m = 2 * c[2n-2-m] for m=0,...,n-1
    return 2.0 * c[n - 1:2 * n - 1][::-1]


def qp_best_response(q, max_iters=30, lr_scale=0.5):
    """Solve min_a sum_k q_k*(a*a)_k s.t. sum(a)=1, a>=0 via projected gradient."""
    n = (len(q) + 1) // 2
    a = np.ones(n) / n

    # Estimate Lipschitz constant
    L = 4.0 * n * q.max() + 0.01
    lr = lr_scale / L

    for it in range(max_iters):
        grad = qp_gradient(a, q)
        a_new = project_simplex(a - lr * grad)
        if np.max(np.abs(a_new - a)) < 1e-13:
            a = a_new
            break
        a = a_new

    return a


def game_optimize(a0, time_limit, eta=0.1, qp_iters=25, verbose=True):
    """Zero-sum game: A minimizes max autocorr, B picks lag distribution.
    A uses approximate QP best response; B uses multiplicative weight updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    q = np.ones(2 * n - 1) / (2 * n - 1)
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    T_count = 0
    no_improve = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # A's approximate best response to q
        a_br = qp_best_response(q, max_iters=qp_iters)

        # Nesterov-type update for A
        alpha_t = 2.0 / (t + 2.0)
        if t == 1:
            a_new = a_br.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            y_new = (1 - alpha_t) * a + alpha_t * a_br
            y_new = np.maximum(y_new, 0)
            ss = y_new.sum()
            if ss > 1e-15:
                y_new = y_new / ss
            a_new = y_new + momentum * (y_new - a)
            a_new = project_simplex(a_new)
        a = a_new

        # B's multiplicative weight update
        b = fast_autocorr(a)
        # Annealed temperature for B's beliefs
        T_anneal = max(0.05, 1.0 / (1.0 + 0.0005 * t))
        log_q = np.log(q + 1e-300) + (eta / T_anneal) * b
        log_q -= log_q.max()
        q = np.exp(log_q)
        q = q / q.sum()

        # Track best
        score = internal_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Time average
        avg_a += a
        T_count += 1
        if T_count % 100 == 0:
            avg = avg_a / T_count
            sc_avg = internal_score(avg)
            if sc_avg < best_score - 1e-12:
                best_score = sc_avg
                best_a = avg.copy()
                no_improve = 0

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Game t={t} ({elapsed:.0f}s): curr={score:.6f}, best={best_score:.6f}")

        if no_improve > 20000:
            break

    return best_a, best_score


def nesterov_smoothed(a0, time_limit, T_start=50.0, T_end=0.01, beta=0.9, base_lr=0.05, verbose=True):
    """Smoothed subgradient with Nesterov acceleration using LogSumExp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, t / 20000.0)
        T = T_start * (T_end / T_start) ** frac

        b = fast_autocorr(y)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Gradient via FFT
        grad = qp_gradient(y, w)

        # Nesterov momentum
        v = beta * v + grad
        lr_t = base_lr / (1.0 + 0.005 * t)
        y = project_simplex(y - lr_t * v)

        score = internal_score(y)
        if score < best_score - 1e-12:
            best_score = score
            best_a = y.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): curr={score:.6f}, best={best_score:.6f}")

    return best_a, best_score


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent: move mass from peak contributors to low-impact positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Score destinations by their impact on top peaks
            dst_scores = []
            for j in range(0, n, max(1, n // 200)):
                if j == src:
                    continue
                impact = 0.0
                for k2 in top_k:
                    lj = k2 - j
                    if 0 <= lj < n:
                        impact += a[lj]
                dst_scores.append((impact, j))

            dst_scores.sort()
            pool_size = min(5, len(dst_scores))
            dst = dst_scores[rng.integers(pool_size)][1]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 500:
                break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                best_seq = best_seq / s
            print(f"  *** NEW BEST [Game2]: {best_score:.6f}")
        elif sc_out < best_score:
            best_score = sc_out
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  Game2 internal: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 5: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 6: Final peak shaving + subgradient polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t6a = min(remaining * 0.5, 150.0)
        t6b = min(remaining * 0.5, 150.0)
        print(f"\n=== Phase 6: Final polish n={n_base} (remaining={remaining:.0f}s) ===")
        if t6a > 10:
            a_out, _ = peak_shaving(best_seq, t6a, seed=456)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [PeakShave2]: {best_score:.6f}")
        if t6b > 15:
            a_out, sc_out = nesterov_smoothed(best_seq, t6b, T_start=5.0, T_end=0.005, base_lr=0.02)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [SubGrad2]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
