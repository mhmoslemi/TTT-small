```python
def banach_nesterov(a0, time_limit, solver, verbose=True):
    """Banach LP best-response with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0  # normalize to sum=1

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    n_iters = 0
    last_build = -1

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        T = build_T(y)
        M_curr = T @ y  # This is conv(y, y) = autocorr(y)
        # Actually let's use FFT for speed
        M_curr = fast_autocorr(y)
        M = M_curr.max()
        if M < 1e-15:

        b = lp_best_response(y, M, solver, T)
        if b is None:
            # Try without the matrix (in case it was stale)
            b = lp_best_response(y, M, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum  # normalize to sum=1
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        # Nesterov extrapolation point
        if t == 1:
            a_its = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_its = y_new + momentum * (y_new - y)
            a_its = np.maximum(a_its, 0.0)
            s_a = a_its.sum()
            if s_a < 1e-15:
                a_its = y_new.copy()
            else:
                a_its = a_its / s_a

        y_prev = y
        y = y_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_its = fast_autocorr(a_its).max()
        if score_its < best_score:
            best_score = score_its
            best_a = a_its.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Banach iter={n_iters} ({elapsed:.0f}s): max_b={best_score:.8f}, obj={2*n*best_score:.6f}")

    return best_a, best_score


def subgradient_method(a0, time_limit, verbose=True):
    """Direct subgradient descent on max autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        max_b = b.max()
        # Find peaks (within 1% of max)
        peak_mask = b >= max_b * 0.99
        peak_indices = np.where(peak_mask)[0]

        # Compute subgradient: average over peaks
        grad = np.zeros(n)
        for k_star in peak_indices:
            for m in range(n):
                # db_kstar/da_m = a[j] where j = k_star - m (if valid) + a[j] where j = m - k_star (if valid)
                j1 = k_star - m
                j2 = m - k_star
                if 0 <= j1 < n:
                    grad[m] += y[j1]
                if 0 <= j2 < n:
                    grad[m] += y[j2]
        grad /= len(peak_indices)

        # Nesterov step
        eta = 1.0 / (np.sqrt(t) + 1.0)
        if t == 1:
            y_new = y - eta * grad
        else:
            momentum = (t - 1.0 + np.sqrt(1.0 + 4.0 * (t - 1.0) ** 2)) / (2.0 * (t + 1.0))
            y_extrap = y + momentum * (y - y_prev)
            # Recompute gradient at extrapolated point
            b_extrap = fast_autocorr(y_extrap)
            max_extrap = b_extrap.max()
            peak_mask_extrap = b_extrap >= max_extrap * 0.99
            peak_indices_extrap = np.where(peak_mask_extrap)[0]
            grad_extrap = np.zeros(n)
            for k_star in peak_indices_extrap:
                for m in range(n):
                    j1 = k_star - m
                    j2 = m - k_star
                    if 0 <= j1 < n:
                        grad_extrap[m] += y_extrap[j1]
                    if 0 <= j2 < n:
                        grad_extrap[m] += y_extrap[j2]
            grad_extrap /= len(peak_indices_extrap)
            y_new = y_extrap - eta * grad_extrap

        y_prev = y
        y = np.maximum(y_new, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = y_prev.copy()
        else:
            y = y / s_y

        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()

        avg_a += y
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): max_b={best_score:.8f}")

    return best_a, best_score
```

Wait, the subgradient computation above is O(n * |peaks|) which is O(n^2) in the worst case. For n=1800, that's 3.2M operations per iteration. With FFT we can do it in O(n log n). Let me use the FFT-based approach:

The gradient of b_k with respect to a_m is:
- db_k/da_m = a_{k-m} (if 0 <= k-m < n) + a_{m-k} (if 0 <= m-k < n)

This is the same as: (a_rev * delta_k)[m] + (a * delta_k)[m] where delta_k is a unit vector at position k.

For multiple peaks, we sum over peaks. So:
grad[m] = sum_{k in peaks} (a_{k-m} + a_{m-k}) (with valid index checks)

= sum_{k in peaks} a_{k-m} (valid) + sum_{k in peaks} a_{m-k} (valid)

The first term is the cross-correlation of a with the peak indicator, evaluated at specific lags. We can compute this with FFT:
- Let w = peak_indicator (length 2n-1)
- First term: (a_rev * w)[m + n - 1] for m=0..n-1
- Second term: (a * w)[m] for m=0..n-1

This is exactly what the LSE gradient code does! Let me reuse that:

```python
def subgrad_max(a, b):
    """Compute subgradient of max(b) with respect to a, where b=autocorr(a)."""

    # Peaks within some tolerance
    w = (b >= max_b * 0.95).astype(np.float64)
    if w.sum() == 0:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    w /= w.sum()

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1: n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Banach LP best-response with Nesterov acceleration (core method)
2. FFT-based subgradient descent on peak autocorrelation (complementary)
3. Simulated annealing polish for final fine-tuning
4. Multi-length exploration with interpolation
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_T(a):
    n = len(a)
    k_arr = np.repeat(np.arange(n), n)
    i_arr = np.tile(np.arange(n), n)
    j_arr = k_arr + i_arr
    val_arr = np.tile(a, n)
    return csr_matrix((val_arr, (j_arr, k_arr)), shape=(2 * n - 1, n))


def lp_best_response(a, M, solver, T=None):
    n = len(a)
    if T is None:
        T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def banach_nesterov(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    n_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y)
        M = M_curr.max()
        if M < 1e-15:
            break

        T = build_T(y)
        b = lp_best_response(y, M, solver, T)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_its = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_its = y_new + momentum * (y_new - y)
            a_its = np.maximum(a_its, 0.0)
            s_a = a_its.sum()
            if s_a < 1e-15:
                a_its = y_new.copy()
            else:
                a_its = a_its / s_a

        y = y_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_its = fast_autocorr(a_its).max()
        if score_its < best_score:
            best_score = score_its
            best_a = a_its.copy()

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Banach iter={n_iters} ({elapsed:.0f}s): max_b={best_score:.8f}, obj={2*n*best_score:.6f}")

    return best_a, best_score


def subgrad_fft(a, b):
    """Compute subgradient of max(b) w.r.t. a using FFT, where b=autocorr(a)."""
    n = len(a)
    max_b = b.max()
    w = (b >= max_b * 0.95).astype(np.float64)
    if w.sum() == 0:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    w /= w.sum()

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1: n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2


def subgradient_method(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        grad = subgrad_fft(y, b)

        eta = 1.0 / (np.sqrt(t) + 1.0)

        else:
            momentum = (t - 1.0 + np.sqrt(1.0 + 4.0 * (t - 1.0) ** 2)) / (2.0 * (t + 2.0))
            y_extrap = y + momentum * (y - y_prev)
            y_extrap = np.maximum(y_extrap, 0.0)
            s_e = y_extrap.sum()
            if s_e > 1e-15:
                y_extrap = y_extrap / s_e
            b_e = fast_autocorr(y_extrap)
            grad_e = subgrad_fft(y_extrap, b_e)
            y_new = y_extrap - eta * grad_e

        y_prev = y
        y = np.maximum(y_new, 0.0)
        s_y = y.sum()

                    best_a = avg_check.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): max_b={best_score:.8f}, obj={2*n*best_score:.6f}")

    return best_a, best_score


def simulated_annealing(a0, time_limit, seed=42, verbose=True):
    """Polish with simulated annealing: random 2-coordinate moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    T_start = 0.01
    T_end = 1e-6
    n_total = 50000

    for it in range(n_total):
        if time.monotonic() - t0 > time_limit:
            break

        T = T_start * (T_end / T_start) ** (it / max(1, n_total - 1))

        # Pick two random indices
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if a[i] < 1e-10:
            continue

        # Move a fraction from i to j
        frac = rng.uniform(0.001, 0.02)
        amt = a[i] * frac

        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        score_trial = fast_autocorr(a_trial).max()
        delta = score_trial - best_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a_trial.copy()

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA it={it} ({elapsed:.0f}s): best={best_score:.8f}, obj={2*n*best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
```

OK let me also think about a potentially more effective approach: **adaptive peak targeting**.

Instead of the generic Banach iteration, we can focus specifically on reducing the peak. The idea:
1. Find the peak lag k*
2. The peak value is b_{k*} = sum_i a_i * a_{k*-i}
3. To reduce this, we want to reduce the contributions from the largest terms
4. For each pair (i, k*-i) that contributes significantly, try to move mass away from one of them

This is more targeted than the generic Banach approach and might converge faster to a lower peak.

```python
def peak_reduction(a0, time_limit, verbose=True):
    """Targeted peak reduction by moving mass away from contributing positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        k_star = np.argmax(b)

        # Find contributing pairs
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            break

        # Sort by contribution
        sorted_idx = idx[np.argsort(contrib)[::-1]]
        top_k = min(10, len(sorted_idx))

        improved = False
        for src_pos in sorted_idx[:top_k]:
            src_val = a[src_pos]
            if src_val < best_score * 1e-6:
                continue

            # Try moving to positions that don't contribute to the peak
            # Peak-contributing positions are those i where i or k_star-i is in [js, je]
            # Actually, moving from src_pos to dst_pos affects:
            # - b_{k_star}: decreases by 2*a[src_pos]*dst_contribution - amt^2 (roughly)
            # - Other b_k: might increase
            
            # Simple approach: try moving to random positions and check
            for _ in range(20):
                dst_pos = np.random.randint(n)
                if dst_pos == src_pos:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src_pos] -= amt
                    a_trial[dst_pos] += amt
                    if a_trial[src_pos] < -1e-16:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 500:
                break

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed ({elapsed:.0f}s): max_b={best_score:.8f}, obj={2*n*best_score:.6f}")

    return best_a, best_score
```

```python
def build_T(a):
    n = len(a)
    k_arr = np.repeat(np.arange(n), n)  # [0,0,...,0, 1,1,...,1, ..., n-1,...]
    i_arr = np.tile(np.arange(n), n)   # [0,1,...,n-1, 0,1,...,n-1, ...]
    j_arr = k_arr + i_arr              # row indices
    val_arr = np.tile(a, n)            # values
    return csr_matrix((val_arr, (j_arr, k_arr)), shape=(2 * n - 1, n))
```
