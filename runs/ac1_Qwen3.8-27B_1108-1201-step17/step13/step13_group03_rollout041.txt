import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return _global_solver
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print(f"  LP solver (fallback): CLARABEL")
    return _global_solver

def _fast_autocorr(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _lse_smoothed_max(a, T):
    """LSE-smoothed max of autocorrelation."""
    ac = _fast_autocorr(a)
    ac_max = ac.max()
    return ac_max + T * np.log(np.sum(np.exp((ac - ac_max) / T)))

    """Gradient of LSE-smoothed max autocorrelation."""
    n = len(a)
    ac = _fast_autocorr(a)
    ac_max = ac.max()
    shifted = np.clip((ac - ac_max) / T, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()

    # = 2 * (w * a_rev)[j] where appropriate
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)

def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n

    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:

    return out

    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size2 = 1 << (3 * n - 1).bit_length()
    W_full = rfft(w, fft_size2)
    A_full = rfft(y, fft_size2)
    c = irfft(W_full * np.conj(A_full), fft_size2)
    grad = 2.0 * c[:n]
    return grad


def _lse_mirror_descent(a0, T_schedule, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        eta = 2.0 / np.sqrt(n)
        max_iters = 300000
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            grad = _lse_grad(a, T)
            gnorm = np.sqrt(grad @ grad)
            if gnorm < 1e-15:
                break
            step = eta / gnorm
            step = min(step, 1.0 / max(T, 0.01))
            log_a = np.log(a + 1e-30) - step * grad
            log_a_max = log_a.max()
            exp_log_a = np.exp(log_a - log_a_max)
            a_new = exp_log_a / exp_log_a.sum()
            a = a_new

            if it % 5000 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    MD stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    sc = _fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _lp_best_response_full(y, M, solver):
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        coeff = y[k - np.arange(js, je + 1)]
        constraints.append(coeff @ b_var[js:je+1] <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _lp_best_response_active(y, M, solver, top_k=400):
    n = len(y)
    ac = _fast_autocorr(y)
    active = np.argsort(ac)[::-1][:top_k]
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        coeff = y[k - np.arange(js, je + 1)]
        constraints.append(coeff @ b_var[js:je+1] <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, solver, verbose=True, use_active=True, top_k=400):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        if use_active:
            b = _lp_best_response_active(y, M_curr, solver, top_k=top_k)
        else:
            b = _lp_best_response_full(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _subgradient_polish(a0, time_limit, verbose=False):
    """Subgradient method with decreasing step size for direct optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = 1.0 / np.sqrt(n)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = _fast_autocorr(a)
        k_star = int(np.argmax(ac))
        M = ac[k_star]

        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * a[k_star - idx]

        step = eta / (it + 1.0)
        a_new = a - step * grad
        a_new = _proj_simplex(a_new)
        a = a_new

        if it % 100 == 0:
            sc = _fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _local_pair_swap(a0, time_limit, seed=42, verbose=False):
    """Local search: try moving mass between pairs of positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = _fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:5]
        improved = False

        for k in top_peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = int(idx[np.argmax(contrib)])

            # Find destination: position that doesn't contribute to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            candidates = candidates[np.argsort(a[candidates])]
            dst_candidates = candidates[:max(1, len(candidates) // 3)]

            if len(dst_candidates) == 0:
                continue

            dst = int(dst_candidates[rng.integers(len(dst_candidates))])

            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 300:
                break

    return best_a, best_score


def _spectral_peak_shave(a0, time_limit, seed=12345):
    """Identify peaks in frequency domain and reduce them."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        ac = _fast_autocorr(a)
        k_peak = int(np.argmax(ac))
        peak_val = ac[k_peak]

        # Find which pairs contribute most to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        top_contrib = np.argsort(contrib)[::-1][:10]

        improved = False
        for tc in top_contrib:
            i_src = int(idx[tc])
            j_src = int(k_peak - i_src)
            # Try to reduce by moving mass from i_src to a safe position
            peak_mask = np.zeros(n, dtype=bool)
            top_peaks = np.argsort(ac)[::-1][:5]
            for k2 in top_peaks:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                peak_mask[js2:je2+1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != i_src)[0]
            candidates = candidates[np.argsort(a[candidates])]
            dst_list = candidates[:max(1, len(candidates) // 4)]

            if len(dst_list) == 0:
                continue
            dst = int(dst_list[rng.integers(len(dst_list))])

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[i_src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[i_src] -= amt
                a_trial[dst] += amt
                if a_trial[i_src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try random pair
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                amt = a[i] * rng.uniform(0.002, 0.015)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()

            # Check for stall
            ac_now = _fast_autocorr(a)
            if ac_now.max() >= best_score * (1 - 1e-10):
                stall_check = time.monotonic() - t0
                if stall_check > time_limit * 0.8:
                    break

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _lse_fista(a0, T_schedule, time_limit, verbose=False):
    """FISTA on LSE-smoothed max autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    t_fista = 1

    for stage_i, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        L = max(n / (T * 2.0), 1.0)
        staging = a.copy()
        max_iters = 50000
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            grad = _lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 3000 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    sc = _fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-method, multi-length optimization combining
    LSE-FISTA, Mirror Descent, LP-Nesterov (active set), and local search."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.sin(np.pi * np.linspace(0, 1, n_base)) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LSE-FISTA (fast coarse optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 15:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [10.0, 3.0, 1.0, 0.3, 0.1]
        a_out, _ = _lse_fista(best_seq, T_sched, t1, verbose=True)
        _try_update(a_out, "fista1")

    # === Phase 2: Mirror Descent (exponential weights) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 120.0)
    if t2 > 10:
        print(f"\n=== Phase 2: Mirror Descent n={len(best_seq)} (budget={t2:.0f}s) ===")
        T_sched = [5.0, 1.5, 0.4, 0.1]
        a_out, _ = _lse_mirror_descent(best_seq, T_sched, t2, verbose=True)
        _try_update(a_out, "mirror1")

    # === Phase 3: LP-Nesterov with active set (main optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP-Nesterov active n={n_cur} (budget={t3:.0f}s) ===")
        top_k = min(400, 2 * n_cur - 1)
        a_out, _ = _lp_nesterov(best_seq, t3, solver, verbose=True, use_active=True, top_k=top_k)
        _try_update(a_out, "lp_active1")

    # === Phase 4: Local pair-swap polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.08, 80.0)
    if t4 > 10:
        print(f"\n=== Phase 4: Local pair-swap n={len(best_seq)} (budget={t4:.0f}s) ===")
        a_out, _ = _local_pair_swap(best_seq, t4, seed=42)
        _try_update(a_out, "local1")

    # === Phase 5: Expand to larger length ===
    n2 = len(best_seq) + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 30:
        print(f"\n=== Phase 5: Expand to n={n2} (budget={t5:.0f}s) ===")
        init_long = _interpolate_sequence(best_seq, n2)
        # Quick FISTA at new length
        t5a = min(t5 * 0.2, 60.0)
        T_sched = [5.0, 1.0, 0.3]
        a_out, _ = _lse_fista(init_long, T_sched, t5a, verbose=False)
        _try_update(a_out, f"fista_n{n2}")
        # LP at new length
        remaining5 = t5 - (time.monotonic() - t_global) + t5  # approximate
        remaining5 = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining5 * 0.5, 250.0)
        if t5b > 20:
            top_k2 = min(500, 2 * n2 - 1)
            a_out, _ = _lp_nesterov(best_seq if len(best_seq) == n2 else init_long,
                                     t5b, solver, verbose=True, use_active=True, top_k=top_k2)
            _try_update(a_out, f"lp_n{n2}")

    # === Phase 6: LP-Nesterov full (with all constraints) at best length ===
