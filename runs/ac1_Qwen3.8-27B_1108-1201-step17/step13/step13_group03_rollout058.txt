def _fast_autocorr(a):
    """Compute full autocorrelation (convolution with self) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _score(a):
    """Compute the objective value."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = _fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def _project_simplex(v):
    """Project onto probability simplex {x: sum=1, x>=0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _spectral_flatten(a, iters=200, time_limit=None):
    """Spectral flattening: iteratively equalize the power spectrum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()
    
    for it in range(iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        
        # Target: flatten the magnitude
        target_mag = np.full_like(mag, np.mean(mag))
        
        # Blend: don't fully flatten, keep some structure
        alpha = 0.5  # blending parameter
        new_mag = alpha * mag + (1 - alpha) * target_mag
        
        # Reconstruct with original phase
        phase = np.exp(1j * np.angle(A))
        A_new = new_mag * phase
        a_new = irfft(A_new, 2 * n - 1)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
    
    return best_a, best_score


def _lse_grad(y, T):
    """Gradient of LSE-smoothed max autocorrelation at temperature T."""
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = np.clip((ac - ac_max) / T, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    # Gradient of sum w_k * ac_k with respect to y
    # d(ac_k)/dy_i = y_{k-i} + y_{i-k} (for valid indices)
    # grad = sum_k w_k * 2 * (circular-like cross corr)
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    # The gradient of the autocorrelation weighted sum
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def _lse_fista(a0, T_schedule, iters_per_stage, time_limit, verbose=False):
    """FISTA on LSE-smoothed max autocorrelation with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        # Lipschitz constant estimate
        L = max(n / (T * 2.0), 1.0)
        t_fista = 1
        staging = a.copy()
        
        max_iters = max(int(iters_per_stage), 2000)
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad = _lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _project_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 1000 == 0:
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

def _build_conv_matrix(a, n):
    """Build the convolution constraint matrix T where (T@b)_k = conv(a,b)[k]."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T


def _lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def _mirror_descent(a0, time_limit, learning_rate=None, verbose=False):
    """Mirror descent (multiplicative weights) for non-negative optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a.sum()
    a = a * norm / s

    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()
    
    if learning_rate is None:
        learning_rate = 1.0 / np.sqrt(n)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient of max autocorrelation
        ac = _fast_autocorr(a)
        k_star = int(np.argmax(ac))
        
        # Gradient of ac[k_star] with respect to a
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]
        grad *= 2.0  # because ac = a*a
        
        # Mirror descent step (multiplicative weights)
        eta = learning_rate / (1.0 + 0.1 * t)  # decaying step size
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        a_new = a_new * norm / s_new
        
        a = a_new
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def _coordinate_descent(a, time_limit, seed=42):

        a = a * norm / s0

    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()
    
    # Precompute which lags each pair of indices affects
    # Moving mass from i to j affects lags in range [j-i, j-i+1] approximately
    
    iterations = 0
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        # Sample a random pair
        i = rng.integers(n)
        j = rng.integers(n)

def _interpolate_sequence(f, new_n):
    """Linearly interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _fast_autocorr(a):
    """Compute full autocorrelation (convolution with self) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _score(a):
    """Compute the objective value 2*n*max_ac/(sum_a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    ac = _fast_autocorr(a)
    return 2.0 * n * ac.max() / (s * s)


def _project_simplex(v):
    """Project onto probability simplex {x: sum=1, x>=0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _spectral_flatten(a, iters=300, time_limit=None):
    """Spectral flattening: iteratively equalize the power spectrum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()

    for it in range(iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break

        A = rfft(a)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15
        target_mag = np.full_like(mag, np.mean(mag))
        alpha = 0.6
        new_mag = alpha * mag + (1.0 - alpha) * target_mag
        phase = np.exp(1j * np.angle(A))

        a_new = irfft(A_new, 2 * n - 1)[:n]

            best_a = a_new.copy()

def _lse_grad(y, T):
    """Gradient of LSE-smoothed max autocorrelation at temperature T."""
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = np.clip((ac - ac_max) / T, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _build_conv_matrix(a, n):
    """Build matrix T where (T @ b)[k] = conv(a, b)[k]."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T


def _lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

            best_a = a.copy()
        if verbose and lp_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov lp={lp_count} t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _mirror_descent(a0, time_limit, verbose=False):
    """Mirror descent (multiplicative weights) for non-negative optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s = a.sum()
    a = a * norm / s
    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()
    learning_rate = 1.0 / np.sqrt(n * 10.0)

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = _fast_autocorr(a)
        k_star = int(np.argmax(ac))
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]
        grad *= 2.0
        eta = learning_rate / (1.0 + 0.05 * t)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        a_new = a_new * norm / s_new
        a = a_new
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _peak_shave(a, time_limit, seed=42):
    """Targeted peak reduction via mass redistribution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = _fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:10]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            local_best = idx[np.argmax(contrib)]
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2: je2 + 1] = True
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != local_best)[0]
            candidates_sorted = candidates[np.argsort(a[candidates])]
            for dst in candidates_sorted[:min(5, len(candidates_sorted))]:
                for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[local_best] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[local_best] -= amt
                    a_trial[dst] += amt
                    if a_trial[local_best] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = _score(a_trial)
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-15:
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = _score(a_trial)
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 1000:
                break

    return best_a, best_score


def _coordinate_descent(a, time_limit, seed=42):
    """Coordinate descent with random pair moves."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        iterations += 1
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        for direction in [1, -1]:
            for frac in [0.005, 0.01, 0.02, 0.05]:
                src = i if direction == 1 else j
                dst = j if direction == 1 else i
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15 or a_trial[dst] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    break
            else:
                continue
            break
        if iterations % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc iter={iterations} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    """Linearly interpolate a sequence to a new length (preserving sum=1)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def prop_update_best(best_seq, best_score, a, label, t_global):
    """Try to update best with candidate a. Returns (new_best, new_score)."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = _evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_seq = a.copy()
        best_score = sc
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-method ensemble with spectral init, LSE-FISTA,
    LP-Nesterov, mirror descent, peak shaving, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1700.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Initialize from parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_seq = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    n_init = 1600
    if parent is not None:
        n_init = len(parent)

    # Generate alternative initializations
    x = np.linspace(0, 1, n_init)

    # Raised cosine
    init_cos = 0.5 * (1 + np.cos(2 * np.pi * x))
    init_cos = np.maximum(init_cos, 0.0)
    init_cos = init_cos / init_cos.sum()
    sc_cos = _score(init_cos)

    # Gaussian-like
    init_gauss = np.exp(-4.0 * (x - 0.5) ** 2)
    init_gauss = init_gauss / init_gauss.sum()
    sc_gauss = _score(init_gauss)

    # Random
    init_rand = np.abs(rng.standard_normal(n_init))
    init_rand = init_rand / init_rand.sum()
    sc_rand = _score(init_rand)

    print(f"Init scores: cos={sc_cos:.6f}, gauss={sc_gauss:.6f}, rand={sc_rand:.6f}")

    # Pick best initialization
    init_candidates = [
        ('cos', init_cos, sc_cos),
        ('gauss', init_gauss, sc_gauss),
        ('rand', init_rand, sc_rand),
    ]
    if parent is not None:
        init_candidates.append(('parent', parent, best_score))

    best_init_name, best_init_arr, best_init_sc = min(init_candidates, key=lambda x: x[2])
    print(f"Best init: {best_init_name} with score={best_init_sc:.6f}")
    best_seq = best_init_arr.copy()
    best_score = min(best_init_sc, best_score)
    if best_seq is None or best_score != _evaluate_safe(best_seq):
        best_score = _evaluate_safe(best_seq)

    # === Phase 1: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 60.0)
    if t1 > 10:
        print(f"\n=== Phase 1: Spectral flatten n={len(best_seq)} ({t1:.0f}s) ===")
        a_out, _ = _spectral_flatten(best_seq, iters=500, time_limit=t1)
        best_seq, best_score = prop_update_best(best_seq, best_score, a_out, "spectral", t_global)

    # === Phase 2: LSE-FISTA coarse ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 15:
        n_cur = len(best_seq)
        print(f"\n=== Phase 2: LSE-FISTA coarse n={n_cur} ({t2:.0f}s) ===")
        T_sched = [30.0, 8.0, 2.0, 0.5, 0.12, 0.03]
        a_out, _ = _lse_fista(best_seq, T_sched, 20000, t2, verbose=True)
        best_seq, best_score = prop_update_best(best_seq, best_score, a_out, "lse_coarse", t_global)

    # === Phase 3: LP-Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.30, 400.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP-Nesterov n={n_cur} ({t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t3, solver, verbose=True)
        best_seq, best_score = prop_update_best(best_seq, best_score, a_out, "lp_main", t_global)

    # === Phase 4: Longer sequence LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    t4 = min(remaining * 0.18, 220.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP-Nesterov long n={n_long} ({t4:.0f}s) ===")
        init_long = _interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = _lp_nesterov(a_norm, t4, solver, verbose=True)
        best_seq, best_score = prop_update_best(best_seq, best_score, a_out, f"lp_long_{n_long}", t_global)

    # === Phase 5: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.08, 80.0)
    if t5 > 10:
        print(f"\n=== Phase 5: Mirror descent n={n_cur} ({t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _mirror_descent(a_norm, t5, verbose=False)
        best_seq, best_score = prop_update_best(best_seq, best_score, a_out, "mirror", t_global)

    # === Phase 6: LSE-FISTA fine ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 10:
        print(f"\n=== Phase 6: LSE-FISTA fine n={n_cur} ({t6:.0f}s) ===")
        T_sched = [1.0, 0.2, 0.05, 0.01, 0.003]
        a_out, _ = _lse_fista(best_seq, T_sched, 25000, t6, verbose=True)
        best_seq, best_score = prop_update_best(best_seq, best_score, a_out, "lse_fine", t_global)

    # === Phase 7: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.12, 120.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Peak shave n={n_cur} ({t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, t7, seed=99999)
        best_seq, best_score = prop_update_best(best_seq, best_score, a_out, "peak_shave", t_global)

    # === Phase 8: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.08, 80.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Coord descent n={n_cur} ({t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _coordinate_descent(a_norm, t8, seed=55555)
        best_seq, best_score = prop_update_best(best_seq, best_score, a_out, "coord_desc", t_global)

    # === Phase 9: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t9 = min(remaining * 0.15, 120.0)
    if t9 > 20:
        print(f"\n=== Phase 9: Final LP n={n_cur} ({t9:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t9, solver, verbose=True)
        best_seq, best_score = prop_update_best(best_seq, best_score, a_out, "lp_final", t_global)

    # === Phase 10: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t10 = min(remaining * 0.6, 80.0)
        if t10 > 5:
            n_cur = len(best_seq)
            print(f"\n=== Phase 10: Final shave n={n_cur} ({t10:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = _peak_shave(a_norm, t10, seed=123456)
            best_seq, best_score = prop_update_best(best_seq, best_score, a_out, "final_shave", t_global)

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
