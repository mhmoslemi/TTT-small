def lse_gradient_step(a, w):
    """Compute gradient of sum_k w_k * (a*a)_k.
    grad_i = 2 * sum_j w[k]*a[j] where k = i+j... 
    Actually: (a*a)[k] = sum_j a[j]*a[k-j]
    d/d a[i] of (a*a)[k] = a[k-i] + a[i-k] (when valid)
    So grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # conv1[i] = sum_k w[k] * a[k-i] (for valid k-i)
    # This is the cross-correlation of w and a
    # conv1[i] = sum_j w[i+j] * a[j] ... let me use FFT
    # Actually: grad[i] = 2 * (w * a_reversed)[i + n - 1]
    # where * denotes convolution
    a_rev = a[::-1]
    fft_size2 = 1 << (size + n - 1 - 1).bit_length()
    conv = irfft(rfft(w, fft_size2) * rfft(a_rev, fft_size2), fft_size2)
    return 2.0 * conv[n - 1:2 * n - 1]


def lse_gradient_descent(a0, time_limit, verbose=True):

    else:
        a = a * norm / s0
    
    best_a = a.copy()

    t0 = time.monotonic()
    steps = 0
    prev_score = best_score
    
    T_start = max(0.5, best_score * 0.05)
    T_end = max(0.01, best_score * 0.003)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        # Softmax weights with outlier damping
        shifted = b - max_b
        w = np.exp(shifted / T)
        # Cap extremely large weights

            w = np.minimum(w, 1e6)
        w_sum = w.sum()
        if w_sum < 1e-30:

def local_peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted local search: reduce the peak by moving mass from contributing positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    n_stall = 0

        # Anneal temperature
        elapsed_frac = (time.monotonic() - t0) / time_limit
        T = max(best_score * 0.003, best_score * 0.02 * (1.0 - elapsed_frac))
        
        b = fast_autocorr(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()
        
        if total_contrib < 1e-18:
            break
        
        # Find the most contributing source
        src_local = np.argmax(contrib)
        src = idx[src_local]
        
        # Find good destinations: positions that don't contribute to top peaks
        top_peaks = np.argsort(b)[::-1][:10]
        peak_mask = np.zeros(n, dtype=bool)
        for k in top_peaks:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            peak_mask[js2:je2 + 1] = True
        
        # Candidate destinations: not in peak mask, or low value
        candidates = np.where(~peak_mask)[0]
        if len(candidates) < 5:
            candidates = np.where(np.arange(n) != src)[0]
        
        # Try moving a fraction from src to a random destination
        n_trials = 3
        improved = False
        for _ in range(n_trials):
            frac = rng.uniform(0.01, 0.08)
            amt = a[src] * frac

                continue
            
            dst = candidates[rng.integers(len(candidates))]
            if dst == src:
                continue
            
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:

                a = a_trial
                best_score = score
                best_a = a.copy()
                n_improve += 1

                break
            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break
        
        if not improved:
            n_stall += 1
            # Try pair swaps
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial[i] = max(a_trial[i], 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial
        
        if n_stall > 100:
            # Larger perturbation
            n_pert = max(5, n // 20)
            idx_p = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_p:
                amt = a[idx] * rng.uniform(0.1, 0.3)
                j = rng.integers(n)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score * 1.005:
                    a = a_trial
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
            n_stall = 0
        
        if verbose and n_improve % 500 == 0 and n_improve > 0:
            print(f"    PeakShave[{n}]: improve={n_improve} ({time.monotonic()-t0:.0f}s) best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):

            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_parent = len(best_seq)
            print(f"Parent loaded: n={n_parent}, score={best_score:.6f}")
    except Exception:
        pass
    
    # Initialize if no parent
    if best_seq is None:
        x = np.linspace(0.0, 1.0, n_parent)
        best_seq = np.sin(np.pi * x) ** 1.2

        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")
    
    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    def get_normed(a, n_target=None):
        """Normalize a for Nesterov (sum = sqrt(2n))."""
        a = np.asarray(a, dtype=np.float64)
        n = a.shape[0]
        norm = np.sqrt(2.0 * n)
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) * norm / n
        return a * norm / s
    
    # ============ PHASE 1: LP Nesterov at parent length ============
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)

    if t1 > 30:

    # ============ PHASE 2: LSE gradient polish at parent length ============
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.08, 100.0)

    if t2 > 10:
        a_in = get_normed(best_seq)
