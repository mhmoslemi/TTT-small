    """Find optimal a[i] given all other coordinates fixed.
    Returns optimal x and the new autocorrelation."""
    n = len(a)
    x_old = a[i]
    
    if x_max is None:
        x_max = max(2 * x_old, np.max(a) * 1.5, 1e-10)
    
    # Compute base autocorrelation (without a[i]'s contribution)
    # Relevant k range: [i, min(2n-2, i+n-1)]
    k_start = i
    k_end = min(2*n - 2, i + n - 1)
    
    # base[k] = ac[k] - contribution of a[i]
    # For k in [k_start, k_end]:
    #   if k == 2*i: contrib = x_old^2
    #   else: contrib = 2 * x_old * a[k-i]
    
    # We'll compute h(x) = max of relevant terms
    # Terms:
    #   For k in [k_start, k_end], k != 2*i: base[k] + 2*x*a[k-i]
    #   For k = 2*i: base[2i] + x^2
    #   For k outside [k_start, k_end]: ac[k] (constant, contributes to max)
    
    # Precompute the constant part (max outside relevant range)
    if k_start > 0 or k_end < 2*n-2:

        if k_start > 0:
            outside_max_left = np.max(ac[:k_start]) if k_start > 0 else -np.inf
        else:
            outside_max_left = -np.inf
        if k_end < 2*n-2:
            outside_max_right = np.max(ac[k_end+1:]) if k_end + 1 <= 2*n-2 else -np.inf
        else:
            outside_max_right = -np.inf
        outside_max = max(outside_max_left, outside_max_right)
    else:
        outside_max = -np.inf
    
    # Precompute for the relevant range
    ks = np.arange(k_start, k_end + 1)
    # a[k-i] for k in [k_start, k_end]: indices are 0 to n-1
    a_shifted = a[0:n]  # a[k-i] where k-i ranges from 0 to n-1

    base = ac[k_start:k_end+1].copy()

    for idx, k in enumerate(ks):

            base[idx] = ac[k] - 2 * x_old * a[k - i]

        # For k in relevant range:
        vals = base.copy()

                vals[idx] = base[idx] + x * x

        result = np.max(vals)

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def coord_descend(a_init, time_limit, t_start=None, verbose=False, verbose_every=50):

    n = len(a_init)

    if s < 1e-15:

        a = a / s

    best_M = ac.max()
    t0 = t_start if t_start is not None else time.monotonic()
    
    gr = (np.sqrt(5) - 1) / 2.0
    a_max = a.max()
    
    for sweep in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        any_improved = False
        for i in range(n):
            x_old = a[i]
            
            k_start = i
            k_end = min(2*n - 2, i + n - 1)
            # n_rel = k_end - k_start + 1 = n (always, for i in [0, n-1])

            base = ac[k_start:k_end+1] - 2.0 * x_old * a

            base[idx_2i] = ac[k_start + idx_2i] - x_old * x_old
            
            # Outside max
            outside_max = -np.inf
            if k_start > 0:
                outside_max = max(outside_max, float(ac[:k_start].max()))
            if k_end < 2*n - 2:
                outside_max = max(outside_max, float(ac[k_end+1:].max()))

            lo = 0.0
            hi = max(2.0 * x_old, a_max * 1.5, 1e-10)

            # h(x) = max(base + 2*x*a[0:n], outside_max) with quadratic at idx_2i
            
            def h_eval(x):
                vals = base + 2.0 * x * a
                vals[idx_2i] = base[idx_2i] + x * x
                mx = vals.max()
                if outside_max > mx:
                    return outside_max
                return mx
            
            h_lo = h_eval(lo)
            h_hi = h_eval(hi)
            # Extend hi if needed
            iters = 0
            while h_hi > h_lo and hi < 2.0 and iters < 10:
                hi *= 2.0
                h_hi = h_eval(hi)
                iters += 1
            
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            f_c = h_eval(c)
            f_d = h_eval(d)

                    hi = d
                    d = c
                    f_d = f_c
                    c = hi - gr * (hi - lo)
                    f_c = h_eval(c)
                else:
                    lo = c
                    c = d
                    f_c = f_d
                    d = lo + gr * (hi - lo)
                    f_d = h_eval(d)

            ac[k_start:k_end+1] += 2.0 * delta * a
            ac[2*i] += delta * delta  # correction for k=2i

            if x_opt > a_max:
                a_max = x_opt

def make_sin_init(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:

    return a

def make_cos_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:

    return a

def make_flat_init(n, edge_frac=0.1):
    a = np.zeros(n)
    start = int(edge_frac * n)
    end = n - start
    a[start:end] = 1.0
    # Smooth edges
    for i in range(n):
        if i < start:
            a[i] = i / float(start) if start > 0 else 0
        elif i >= end:
            a[i] = (n - 1 - i) / float(start) if start > 0 else 0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_random_init(n, seed=42):
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n))
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
