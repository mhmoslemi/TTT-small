import numpy as np

def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_mask = v_sorted * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(rho_mask):
        return np.full(n, target_sum / n)
    rho = np.nonzero(rho_mask)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    # Fix numerical issues
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def compute_weighted_gradient(a, w):
    """
    Compute g[j] = 2 * sum_k w[k] * a[k-j] for j = 0,...,n-1.
    This is the gradient of sum_k w[k] * conv(a,a)[k] with respect to a.
    """
    n = len(a)
    w_len = len(w)  # should be 2n-1
    N = 1 << (max(n, w_len) - 1).bit_length()
    if N < n + w_len - 1:
        N <<= 1
    
    A = np.fft.fft(a, N)
    W = np.fft.fft(w, N)
    # cross-correlation: ifft(conj(A) * W)[j] = sum_k a[k] * w[k+j]
    # We want sum_k w[k] * a[k-j] = sum_m a[m] * w[m+j] (sub m=k-j)
    # This is exactly ifft(conj(A) * W)[j] for j=0,...,n-1
    corr = np.fft.ifft(np.conj(A) * W)[:n]
    return 2.0 * corr.real


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start initialization with spectral shaping
2. Subgradient descent with backtracking line search and adaptive temperature
3. Systematic targeted coordinate descent with bisection optimization
4. Fine-grained random polishing

systematic peak-targeting in CD, and better time allocation to refinement.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_mask = v_sorted * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(rho_mask):
        return np.full(n, target_sum / n)
    rho = np.nonzero(rho_mask)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def compute_weighted_gradient(a, w):
    """
    Compute g[j] = 2 * sum_k w[k] * a[k-j] for j = 0,...,n-1.
    Gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a.
    """
    n = len(a)
    w_len = len(w)
    N = 1 << (max(n, w_len) - 1).bit_length()
    if N < n + w_len - 1:
        N <<= 1
    A = np.fft.fft(a, N)
    W = np.fft.fft(w, N)
    corr = np.fft.ifft(np.conj(A) * W)[:n]
    return 2.0 * corr.real


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = internal_score(a)
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a

                            st = internal_score(a_trial)
                            if st < best_s:
                                best_s = st
                                best_amt = mid
                                lo2 = mid
                            else:
                                hi2 = mid

                if best_amt > 0 and best_s < best_score:
                    a[src] -= best_amt
                    dst_idx = top_safes[0]  # We need to track which dst was best
                    # Actually we need to redo this properly
                    break

            if improved_this_round:
                break

        # If no improvement from systematic approach, try random
        if not improved_this_round:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < 0:
                        break
                    a_trial = a_trial / a_trial.sum()
                    st = internal_score(a_trial)
                    if st < best_score:
                        a = a_trial
                        best_score = st
                        best_a = a.copy()
                        no_improve = 0
                        improved_this_round = True
                        break
                if not improved_this_round:
                    no_improve += 1
            else:
                no_improve += 1

        round_count += 1
        if no_improve > max_no_improve:
            break

        if verbose and round_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD round={round_count} ({elapsed:.0f}s): best={best_score:.8f}, no_imp={no_improve}")

    return best_a, best_score


def fine_polish(a0, time_limit, seed=123, verbose=True):
    """Very fine random perturbations for final polish."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        batch_size = min(50, max(1, int((time_limit - (time.monotonic() - t0)) * 100)))
        for _ in range(batch_size):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or a[i] < 1e-12:
                continue
            for frac in [0.0003, 0.001, 0.003, 0.01]:
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = a_trial / a_trial.sum()
                st = internal_score(a_trial)
                if st < best_score:
                    a = a_trial
                    best_score = st
                    best_a = a.copy()
                    break
        count += 1
        if count % 20 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"      Polish ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
