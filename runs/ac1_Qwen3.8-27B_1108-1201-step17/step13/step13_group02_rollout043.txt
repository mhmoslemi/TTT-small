    Let me implement this carefully:

    ```python
    def gauss_seidel_best_response(y, M, n_sweeps=1):
        """Find g >= 0 maximizing sum(g) subject to conv(g, y) <= M.
        Uses Gauss-Seidel coordinate updates."""
        n = len(y)
        g = np.zeros(n)
        # C[k] = conv(g, y)[k]
        C = np.zeros(2*n - 1)
        
        for sweep in range(n_sweeps):
            for i in range(n):
                # Find max g[i] such that for all k: C[k] + (g[i]_new - g[i]_old) * y[k-i] <= M
                # i.e., g[i]_new <= g[i]_old + (M - C[k]) / y[k-i] for all k where y[k-i] > 0
                # Valid k: i <= k <= i + n - 1 (and k <= 2n-2)
                k_start = i
                k_end = min(i + n, 2*n)  # exclusive
                yk = y[:k_end - i]  # y[k-i] for k in [k_start, k_end)
                Ck = C[k_start:k_end]
                
                mask = yk > 1e-15
                if np.any(mask):
                    feasible = g[i] + (M - Ck[mask]) / yk[mask]
                    g_new = max(0.0, np.min(feasible))
                else:
                    g_new = g[i]  # no constraint binds
                
                delta = g_new - g[i]
                if abs(delta) > 1e-15:
                    C[k_start:k_end] += delta * yk
                    g[i] = g_new
        
        return g
    ```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Gauss-Seidel Nesterov (O(n^2) coordinate updates, no LP solver)
2. Subgradient Nesterov with analytic subgradients via FFT
3. Direct coordinate-wise convex piecewise-linear minimization
4. Multi-length exploration with interpolation transfer
5. Adaptive restart for escaping local minima

import numpy as np

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def gauss_seidel_best_response(y, M, n_sweeps=3):
    """Approximate best response: find g>=0 with conv(g,y)<=M maximizing sum(g)."""
    n = len(y)
    g = np.zeros(n)
    C = np.zeros(2 * n - 1)
    k_lens = np.minimum(n, 2 * n - 1 - np.arange(n))
    y_pos = y > 1e-15

    for _ in range(n_sweeps):
        for i in range(n):
            L = k_lens[i]
            yk = y[:L]
            Ck = C[i:i + L]
            mask = y_pos[:L]
            if mask.any():
                feasible = (M - Ck[mask]) / yk[mask]
                g_new = max(0.0, np.min(feasible))
            else:
                g_new = 0.0
            delta = g_new - g[i]
            if abs(delta) > 1e-14:
                C[i:i + L] += delta * yk
                g[i] = g_new
    return g


def nesterov_gs(a0, time_limit, n_sweeps=3, verbose=True):
    """Nesterov acceleration with Gauss-Seidel best response."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        g = gauss_seidel_best_response(y, M_curr, n_sweeps)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g * norm / g_sum
        n_iters += 1

        t = n_iters
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GS-Newton iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            pert_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, pert_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, pert_scale, n), 0)
            a = a * norm / a.sum()
            y_prev = a.copy()
            stall = 0
            if verbose:
                print(f"    [Restart at iter {n_iters}]")

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, verbose=True):
    """Nesterov-accelerated subgradient method for min max(conv(a,a))."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_steps = 0

    y = a.copy()
    y_prev = a.copy()

    # Adaptive learning rate
    lr = norm * 0.1

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(y)
        max_b = b.max()
        if max_b < 1e-12:
            break

        # Find peak lags
        tol = 1e-8 * max_b
        peak_mask = b >= max_b - tol
        peak_lags = np.where(peak_mask)[0]

        # Compute subgradient
        subgrad = np.zeros(n)
        for k in peak_lags:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx = k - np.arange(i_start, i_end + 1)
                subgrad[i_start:i_end + 1] += 2.0 * a[idx]

        # Nesterov step
        t = n_steps + 1
        alpha = 2.0 / (t + 2.0)
        y_new = y - lr * subgrad / max_b
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y
        y = y_new
        a = a_new
        n_steps += 1

        # Adaptive lr
        if n_steps % 500 == 0:
            lr = norm * 0.1 / np.sqrt(1.0 + n_steps / 1000.0)

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and n_steps % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={n_steps} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.4f}")

    return best_a, best_score


def coordinate_optimize(a0, n_passes=5, verbose=True):
    """Direct coordinate-wise minimization of max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    for p in range(n_passes):
        for i in range(n):
            # For coordinate i, f(x) = max_k (b_k + (x - a[i]) * 2 * a[k-i])
            # = max_k (C_k + x * m_k) where C_k = b_k - 2*a[i]*a[k-i], m_k = 2*a[k-i]
            b = fast_autocorr(a)
            # Valid lag range for coordinate i: k from i to i+n-1
            k_start = i
            k_end = min(n, 2 * n - 1 - i)
            yk = a[:k_end]  # a[k-i] for k in [i, i+k_end)
            bsl = b[k_start:k_start + k_end]

            # f(x) = max_k (bsl[k] - 2*a[i]*yk[k] + x * 2*yk[k])
            # = max_k (bsl[k] + (x - a[i]) * 2*yk[k])
            # At x = a[i]: f = max_k bsl[k] = b.max() (approximately, for the relevant range)
            # We want to find x >= 0 minimizing max_k (bsl[k] + (x-a[i])*2*yk[k])
            # This is a convex PL function in x.
            # The slope at x is 2*max{yk[k] : k such that bsl[k]+(x-a[i])*2*yk[k] = max}
            # Minimum: find x where slope changes from negative to positive.
            # Equivalent: minimize max_k (bsl[k] + 2*x*yk[k]) over x >= 0 (shifted)
            # = minimize max_k (2*yk[k]*x + (bsl[k] - 2*a[i]*yk[k]))

            # Use bisection on the subgradient (slope)
            # At x=0: slope = 2*max{yk[k] : active at x=0}
            # Active at x=0: k where bsl[k] - 2*a[i]*yk[k] = max_j (bsl[j] - 2*a[i]*yk[j])
            # = k where bsl[k] = max_j bsl[j] (since -2*a[i]*yk[k] is the shift)
            # Actually the active set at x is: {k : (bsl[k] + 2*x*yk[k]) = max_j (bsl[j] + 2*x*yk[j])}

            # Simpler: bisection to find x where f has minimum
            # f(0) = max_k (bsl[k] - 2*a[i]*yk[k]) -- this is the max of bsl at x=0
            # f(inf) = +inf (if any yk[k] > 0)
            # f is convex, so minimum is where left slope <= 0 <= right slope

            # Find approximate minimum by golden section or bisection on slope
            # Left bound: x_lo = 0
            # Right bound: x_hi = a[i] * 2 (reasonable upper bound)
            x_lo = 0.0
            x_hi = a[i] * 3.0 + 1e-10

            # Check if minimum is at boundary x=0
            vals_0 = bsl + (0.0 - a[i]) * 2.0 * yk
            f_0 = vals_0.max()
            # Slope at x=0+: 2 * max{yk[k] : vals_0[k] = f_0}
            active_0 = vals_0 >= f_0 - 1e-12
            if active_0.any():
                slope_0 = 2.0 * yk[active_0].max()
                if slope_0 > 1e-10:
                    # Minimum is at x=0
                    continue

            # Bisection for 30 iterations
            for _ in range(30):
                x_mid = (x_lo + x_hi) / 2.0
                vals_mid = bsl + (x_mid - a[i]) * 2.0 * yk
                f_mid = vals_mid.max()
                active_mid = vals_mid >= f_mid - 1e-10
                if not active_mid.any():
                    break
                slope_mid = 2.0 * yk[active_mid].max()
                # Also check minimum slope (for PL function, subgradient is [min_active_slope, max_active_slope])
                slope_min = 2.0 * yk[active_mid].min()
                if slope_min > 1e-10:
                    # All active slopes positive, minimum is to the left
                    x_hi = x_mid
                elif slope_max_check := np.sort(vals_mid)[-2] >= f_mid - 1e-10:
                    # Check if 0 is in subgradient
                    if slope_mid < -1e-10:
                        x_lo = x_mid
                    else:
                        x_hi = x_mid
                else:
                    if slope_mid > 1e-10:
                        x_hi = x_mid
                    else:
                        x_lo = x_mid

            x_opt = (x_lo + x_hi) / 2.0
            a[i] = max(0.0, x_opt)

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a * norm / s
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CoordOpt pass={p+1}/{n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def lse_smooth_descent(a0, time_limit, verbose=True):
    """LogSumExp-smoothed gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.5, best_score * 0.05)
    T_end = max(0.001, best_score * 0.003)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
        # = 2 * (w * a_rev)[i+n-1] where a_rev = a[::-1]
        # Actually: sum_k w[k] * a[k-i] = cross-correlation
        # Let's compute via FFT:
        # sum_k w[k] * a[k-i] = (w_conj * a)[i] in some sense...
        # Simpler: for each i, sum over k of w[k]*a[k-i] where 0<=k-i<n
        # = sum_j w[i+j] * a[j] for valid i+j < 2n-1
        # This is the cross-correlation of w and a: (w ⊛ a)[i] = sum_j w[i+j]*a[j]
        # = (w_rev * a)[i + len(w) - 1]... let me just use the convolution formula
        # sum_j w[i+j]*a[j] = (full_convolution(w[::-1], a))[i + len(w) - 1]
        # where w[::-1] has length 2n-1 and a has length n
        # Full conv length: (2n-1) + n - 1 = 3n-2
        a_rev = a[::-1]
        w_rev = w[::-1]
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w_rev, fft_size)
        A = rfft(a_rev, fft_size)
        conv = irfft(W * A, fft_size)
        # (w_rev * a_rev)[i + 2n - 2] = sum_j w_rev[i+2n-2-j] * a_rev[j]
        # = sum_j w[2n-1-(i+2n-2-j)] * a[n-1-j]
        # = sum_j w[j-1-i] * a[n-1-j]  ... this is getting confusing
        # Let me just do it directly with the formula:
        # grad[i] = 2 * sum_{j=0}^{min(n-1, 2n-2-i)} w[i+j] * a[j]
        # This is a "sliding dot product" which can be done via convolution:
        # Let v = w (length 2n-1), u = a (length n)
        # sum_j v[i+j] * u[j] = (v * u_flip)[i + n - 1] where u_flip = u[::-1] (length n)
        # Full conv of v (len 2n-1) and u_flip (len n) has length 3n-2
        # (v * u_flip)[k] = sum_j v[j] * u_flip[k-j] = sum_j v[j] * u[n-1-k+j]
        # At k = i + n - 1: sum_j v[j] * u[n-1-i-n+1+j] = sum_j v[j] * u[j-i]
        # Hmm, that's not quite right either.

        # Let me just use the direct formula with numpy:
        # grad[i] = 2 * sum_j w[i+j] * a[j] for j=0..min(n-1, 2n-2-i)
        # For speed, use FFT-based approach:
        # Define v of length 2n-1: v[k] = w[k]
        # Define u of length n: u[j] = a[j]
        # We want c[i] = sum_j v[i+j] * u[j] = sum_j w[i+j] * a[j]
        # This equals: (u * v_flip)[2n-2 - i] where v_flip = v[::-1]
        # Because (u * v_flip)[m] = sum_j u[j] * v_flip[m-j] = sum_j u[j] * v[2n-2-m+j]
        # At m = 2n-2-i: sum_j u[j] * v[i+j] = sum_j a[j] * w[i+j] = c[i] ✓

        v_flip = w[::-1]  # length 2n-1
        u = a  # length n
        # Full convolution length: (2n-1) + n - 1 = 3n-2
        conv_full = irfft(rfft(v_flip, fft_size) * rfft(u, fft_size), fft_size)
        # c[i] = conv_full[2n-2-i] for i=0..n-1
        grad = 2.0 * conv_full[2 * n - 2 : n - 1 : -1]

        # Step
        step_size = norm / (np.max(np.abs(grad)) * n + 1e-10) * 0.5
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score

            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lse_smooth_descent(a_norm, t4b, verbose=False)
            do_try_update(a_out, f"lse_n{n_cur}")

    # Phase 5: Subgradient Nesterov on current best
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 5: SubGrad n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_nesterov(a_norm, t5, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_cur}")

    # Phase 6: Coordinate optimization on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 6: CoordOpt n={n_cur} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_optimize(a_norm, n_passes=5, verbose=True)
        do_try_update(a_out, f"coord2_n{n_cur}")

    # Phase 7: Final GS Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 7: Final GS n={n_cur} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_gs(a_norm, t7, n_sweeps=5, verbose=True)
        do_try_update(a_out, f"gs_final_n{n_cur}")

    # Phase 8: Try n+400
    n3 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 8: GS-Nesterov n={n3} (budget={t8:.0f}s) ===")
    if t8 > 30:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_gs(a_norm, t8, n_sweeps=3, verbose=True)
        do_try_update(a_out, f"gs_n{n3}")

    # Phase 9: Final polish - coordinate optimize + LSE
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t9 = min(remaining * 0.5, 300.0)
    print(f"\n=== Phase 9: Final polish n={n_cur} (budget={t9:.0f}s) ===")
    if t9 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_optimize(a_norm, n_passes=4, verbose=True)
        do_try_update(a_out, f"coord_final_n{n_cur}")

        remaining = time_budget - (time.monotonic() - t_global)
        t9b = min(remaining * 0.5, 100.0)
        if t9b > 10:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lse_smooth_descent(a_norm, t9b, verbose=True)
            do_try_update(a_out, f"lse_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
