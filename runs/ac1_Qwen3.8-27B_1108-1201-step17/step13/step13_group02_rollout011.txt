def lp_solve(a, M):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    import cvxpy as cp
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix: T @ b <= M
    # T[k, j] = a[j] if 0 <= k-j < n, else 0
    # conv(a,b)[k] = sum_j a[j] * b[k-j] (valid indices)
    # = sum_j T[k,j] * b[j]
    
    # For efficiency, use the fact that T is banded
    # T[k, j] = a[j] if max(0, k-n+1) <= j <= min(k, n-1)
    # So for row k, the non-zero entries are at columns max(0,k-n+1) to min(k, n-1)
    
    # Build T as a sparse matrix
    from scipy.sparse import lil_matrix
    
    T = lil_matrix((2*n-1, n))
    for j in range(n):
        # a[j] contributes to conv positions j to j+n-1
        T[j:j+n, j] = a
    T = T.tocsc()
    
    solver = get_solver()
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val
