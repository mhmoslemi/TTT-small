    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    return out

def _build_lp_matrix(a):
    """Build the (2n-1) x n LP constraint matrix."""
    n = len(a)
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # This is a "convolution matrix"
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = a  # a[k-i] for k = i, i+1, ..., i+n-1
    return A

def _lp_best_response_scipy(a, M):

    n = len(a)
    A = _build_lp_matrix(a)
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='HiGHS')
    if res.status != 0:

    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(a)
        k_star = int(np.argmax(ac))
        peak_val = ac[k_star]

        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]

        grad_max = grad.max()
        if grad_max < 1e-15:
            break

        eta = 1.0 / np.sqrt(t)
        log_w = np.zeros(n)
        mask = a > 1e-15
        log_w[mask] = np.log(a[mask])
        log_a_new = log_w - eta * grad / grad_max
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        score = _fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and t % 1000 == 0:
            print(f"    ExpGrad t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _subgradient_polish(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    step = norm * 0.1

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(a)
        k_star = int(np.argmax(ac))

        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += a[j2]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        for _ in range(10):
            a_trial = a - step * grad / grad_norm
            a_trial = _proj_simplex(a_trial)
            a_trial = a_trial * norm
            score = _fast_autocorr(a_trial).max()
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
                break
            step *= 0.5
            if step < norm * 1e-8:
                break

        if t % 5000 == 0:
            step = min(step * 2.0, norm * 0.1)

        if verbose and t % 2000 == 0:
            print(f"    SubGrad t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _peak_shave(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                break

            candidates_sorted = candidates[np.argsort(a[candidates])][
                :max(1, len(candidates) // 5)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-15:
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 300:
                break

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _spectral_init(n, alpha=1.0):
    """Initialize using a Lorentzian spectral shape for near-flat autocorrelation."""
    freqs = np.fft.rfftfreq(n) * n
    spectrum = 1.0 / (1.0 + alpha * (freqs / (n * 0.1)) ** 2)
    a = np.abs(np.fft.irfft(spectrum, n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
