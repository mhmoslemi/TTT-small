    """
    Subgradient method with momentum and adaptive step size.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = S.
    """
    global _total_time_start
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    ac = fast_conv(a)
    best_m = ac.max()
    t0 = time.monotonic()
    
    # Momentum state
    v = a.copy()
    x_prev = a.copy()
    mu = momentum
    
    # AdaGrad state
    g_sq = np.zeros(n)
    ada_lr = 0.1
    ada_eps = 1e-8
    ada_decay = 0.999
    
    # Track previous gradient for BB step
    g_prev = None
    x_diff_prev = None
    
    for t in range(1, 50000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        ac = fast_conv(v)
        M = ac.max()
        
        g = soft_subgrad(v, ac, top_k=top_k, beta=beta)
        
        # AdaGrad step size
        g_sq = ada_decay * g_sq + (1 - ada_decay) * g**2
        step = ada_lr / (np.sqrt(g_sq) + ada_eps)
        step_max = 1.0 / (np.linalg.norm(g, ord=np.inf) + 1e-10)
        step = np.minimum(step, 10.0 * step_max)
        
        # Gradient step
        v_new = v - step * g
        v_new = proj_simplex(v_new, S)
        
        if v_new.sum() < 1e-15:
            break
        
        M_new = fast_conv(v_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = v_new.copy()
        
        # Nesterov momentum
        beta_mom = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        x_new = v_new + (mu * (1.0 - beta_mom) / beta_mom) * (v_new - x_prev)
        x_new = np.maximum(x_new, 0.0)
        sx = x_new.sum()
        if sx > 1e-15:
            x_new = x_new * S / sx
        x_prev = v_new
        v = x_new
        
        M_v = fast_conv(v).max()
        if M_v < best_m:
            best_m = M_v
            best_a = v.copy()
        
        if verbose and t % 10000 == 0:
            J = 2.0 * n * best_m / (S * S)
            print(f"    [{label}] t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.6f}")
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m / (S * S)
        print(f"    [{label}] Done ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coord_descent_polish(a0, time_limit, verbose=True, label="coord"):
    """Coordinate descent: move mass to reduce the max peak."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    ac = fast_conv(a)
    best_m = ac.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_conv(a)
        M = ac.max()
        k_star = int(np.argmax(ac))
        
        # Find the top contributors to b_{k_star}
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]
        
        if contrib.max() < 1e-15:
            break
        
        # Sort by contribution
        order = np.argsort(contrib)[::-1]
        
        improved = False
        for oi in order[:20]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(idx[oi])
            rev_j = int(k_star - src)
            
            # Find destination: position with low contribution to all top peaks
            top_peaks = np.argsort(ac)[::-1][:10]
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                peak_mask[js2:je2+1] = True
            
            # Candidates: positions not in any top peak's support, or with low value
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(a < a.mean())[0]
            if len(candidates) == 0:
                continue
            
            # Pick a few random candidates
            n_try = min(5, len(candidates))
            try_idx = rng.choice(candidates, size=n_try, replace=False)
            
            for dst in try_idx:
                dst = int(dst)
                if dst == src:
                    continue
                
                best_trial = None
                best_trial_m = best_m
                for frac in [0.001, 0.003, 0.01, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial * S / a_trial.sum()
                    ac_trial = fast_conv(a_trial)
                    M_trial = ac_trial.max()
                    if M_trial < best_trial_m - 1e-12:
                        best_trial_m = M_trial
                        best_trial = a_trial
                
                if best_trial is not None:
                    a = best_trial
                    best_m = best_trial_m
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape local minimum
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > S * 1e-6:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    if a[i] >= 0:
                        a = a * S / a.sum()
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m / (S * S)
        print(f"    [{label}] ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_to_new_length(f, new_n):
    """Interpolate sequence to new length using cubic spline."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) * np.sqrt(2.0 * new_n) / new_n
    
    # Use PCHIP-like interpolation (monotone cubic)
    # Pad with zeros for natural boundary conditions
    f_padded = np.concatenate([[0], f, [0]])
    old_x = np.linspace(0, 1, old_n + 2)
    new_x = np.linspace(0, 1, new_n)
    # Only interpolate in the [1/(old_n+1), old_n/(old_n+1)] range
    # Actually, let's just do linear for simplicity
    from scipy.interpolate import PchipInterpolator
    # Filter out zeros for PCHIP
    mask = f_padded > 1e-15
    if mask.sum() < 4:
        # Fall back to linear
        f_interp = np.interp(new_x, old_x, f_padded)
    else:
        pchip = PchipInterpolator(old_x[mask], f_padded[mask])
        f_interp = pchip(new_x)
    
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        f_interp = np.ones(new_n) / new_n
    return f_interp

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
