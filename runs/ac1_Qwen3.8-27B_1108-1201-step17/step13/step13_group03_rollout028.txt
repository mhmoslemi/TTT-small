from scipy.optimize import linprog

def _fast_autocorr(a):
    """Compute linear autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _build_conv_matrix_sparse(a, n):
    """Build sparse convolution matrix T such that T @ b = conv(a, b)."""
    # T[j, i] = a[j-i] if 0 <= j-i < n
    rows = []
    cols = []
    data = []
    for i in range(n):
        for k in range(n):
            j = i + k
            rows.append(j)
            cols.append(i)
            data.append(a[k])
    return csc_matrix((data, (rows, cols)), shape=(2*n-1, n))

def _simplex_project(v):
    """Project onto {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _lp_solve(T, M, n):
    """Solve LP: max sum(b) s.t. T @ b <= M, b >= 0."""
    c = -np.ones(n)
    res = linprog(c, A_ub=T, b_ub=M * np.ones(T.shape[0]), 
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def _simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out
