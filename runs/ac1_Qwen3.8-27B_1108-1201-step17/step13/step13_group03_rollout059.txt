def _fast_conv(a):
    """Compute np.convolve(a, a) efficiently via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, n=fft_size)
    return irfft(A * A, n=fft_size)[:size]


def _proj_simplex(v):
    """Project v onto the simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _build_conv_matrix(a, n):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_best_response(a, M, solver, top_k=None):
    """
    Solve: max sum(b) s.t. T@b <= M, b >= 0.
    Optionally use only top_k most binding constraints.
    """
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if top_k is not None and top_k < 2 * n - 1:
        # Compute a @ zeros to get constraint structure, select top-k
        # The binding constraints are those where (T @ a)_k is closest to M
        T = _build_conv_matrix(a, n)
        constraints_all = T @ a
        # Select top_k constraints that are highest (most binding)
        idx = np.argsort(constraints_all)[::-1][:top_k]
        T_sub = T[idx, :]
        constraints = [T_sub @ b_var <= M]
    else:
        T = _build_conv_matrix(a, n)
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def _lse_fista(a0, T_schedule, time_limit, verbose=False):
    """FISTA on LSE-smoothed max convolution with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = _fast_conv(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        L = max(n / (T * 2.0), n / 100.0)
        t_fista = 1
        staging = a.copy()

        while time.monotonic() - t0 < time_limit:
            # LSE gradient
            ac = _fast_conv(y)
            ac_max = ac.max()
            shifted = (ac - ac_max) / T
            shifted = np.clip(shifted, -60.0, 60.0)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()

            # Gradient via cross-correlation
            fft_size2 = 1 << (3 * n - 2).bit_length()
            A = rfft(y, n=fft_size2)
            W = rfft(w, n=fft_size2)
            cross = irfft(np.conj(A) * W, n=fft_size2)
            grad = 2.0 * cross[n - 1: n - 1 + n]

            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if t_fista % 500 == 0:
                sc = _fast_conv(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_conv(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _coordinate_descent(a0, time_limit, verbose=False):
    """
    Adaptive coordinate descent: find coordinates contributing to peak,
    try small reductions, accept if improves.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a.copy(), _fast_conv(a).max()
    a = a / s0

    best_a = a.copy()
    best_score = _fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_conv(a)
        k_star = int(np.argmax(b))
        current_max = b[k_star]
        
        # Find coordinates contributing to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        
        if len(idx) == 0:
            break
        
        contrib = a[idx] * a[k_star - idx]
        
        # Sort by contribution (highest first)
        order = np.argsort(contrib)[::-1]
        top_coords = idx[order[:min(50, len(order))]]
        
        improved = False
        for src in top_coords:
            if a[src] < 1e-12:
                continue
            
            # Try reducing src and adding to random low-coordinate
            for frac in [0.002, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                
                # Find destination: coordinate with low value, far from peak
                mask = np.ones(n, dtype=bool)
                mask[max(0, k_star - n + 1):min(n, k_star + 1)] = False
                candidates = np.where(mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                
                # Pick the one with lowest value among candidates
                dst = candidates[np.argmin(a[candidates])]
                
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                score = _fast_conv(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            # Try random perturbations
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial / s_t
                        score = _fast_conv(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                        else:
                            stall += 1
            else:
                stall += 1
            
            if stall > 300:
                break
    
    return best_a, best_score


def _interpolate_sequence(f, new_n):
    """Linearly interpolate sequence to new length, preserve shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    # Apply slight smoothing to avoid artifacts
    kernel = np.array([0.25, 0.5, 0.25])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0.0)
    ss = f_smooth.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_smooth / ss


def _evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _subgradient_method(a0, time_limit, verbose=False):
    """
    Subgradient method with Polyak step size and diagonal averaging.
    Used as a complementary approach to LP.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = _fast_conv(a).max()
    target = best_score * 0.999  # Slightly below current for Polyak step
    diag_avg = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    t_iter = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_conv(a)
        k_star = int(np.argmax(b))
        
        # Subgradient: 2 * a[k_star - i] for valid i
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
        
        # Polyak step size
        current_val = b[k_star]
        if current_val > target:
            step = (current_val - target) / (np.sum(grad**2) + 1e-15)
            step = min(step, 0.1)  # Cap step size
        else:
            step = 0.01 / (t_iter + 1)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        t_iter += 1
        diag_avg += a
        n_avg += 1
        
        if t_iter % 20 == 0:
            score = _fast_conv(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                target = min(target, best_score * 0.999)
            
            # Check diagonal average
            avg = diag_avg / n_avg
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg / s_avg
                score_avg = _fast_conv(avg).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg.copy()
        
        if verbose and t_iter % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={t_iter} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function: multi-method optimization combining LP best-response,
    LSE-FISTA, subgradient methods, and coordinate descent across multiple
    sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        # Create a reasonable initial sequence
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP + Nesterov + DiagAvg at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LP-NA n={n_base} (budget={t1:.0f}s) ===")
        a_out, _ = _lp_nesterov_diagavg(best_seq, t1, solver, verbose=True)
        _try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: LSE-FISTA warmup at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 15:
        print(f"\n=== Phase 2: LSE-FISTA n={len(best_seq)} (budget={t2:.0f}s) ===")
        T_sched = [10.0, 3.0, 1.0, 0.3, 0.1]
        a_out, _ = _lse_fista(best_seq, T_sched, t2, verbose=True)
        _try_update(a_out, f"lse1_n{len(best_seq)}")

    # === Phase 3: LP at base length (second pass, possibly with top-k) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP-NA round 2 n={n_cur} (budget={t3:.0f}s) ===")
        a_out, _ = _lp_nesterov_diagavg(best_seq, t3, solver, verbose=True)
        _try_update(a_out, f"lp2_n{n_cur}")

    # === Phase 4: Interpolate to n+800, LP ===
    n2 = len(best_seq) + 800
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP-NA n={n2} (budget={t4:.0f}s) ===")
        init_long = _interpolate_sequence(best_seq, n2)
        a_out, _ = _lp_nesterov_diagavg(init_long, t4, solver, verbose=True)
        _try_update(a_out, f"lp3_n{n2}")

    # === Phase 5: Subgradient at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 10:
        print(f"\n=== Phase 5: SubGrad n={n_best} (budget={t5:.0f}s) ===")
        a_out, _ = _subgradient_method(best_seq, t5, verbose=False)
        _try_update(a_out, f"subgrad_n{n_best}")

    # === Phase 6: Interpolate to n_best+800, LP ===
    n3 = len(best_seq) + 800
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 20:
        print(f"\n=== Phase 6: LP-NA n={n3} (budget={t6:.0f}s) ===")
        init_long2 = _interpolate_sequence(best_seq, n3)
        a_out, _ = _lp_nesterov_diagavg(init_long2, t6, solver, verbose=True)
        _try_update(a_out, f"lp4_n{n3}")

    # === Phase 7: Coordinate descent on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.12, 120.0)
    if t7 > 10:
        print(f"\n=== Phase 7: CoordDesc n={n_best} (budget={t7:.0f}s) ===")
        a_out, _ = _coordinate_descent(best_seq, t7, verbose=False)
        _try_update(a_out, f"coord_n{n_best}")

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.35, 300.0)
    if t8 > 20:
        print(f"\n=== Phase 8: Final LP n={n_best} (budget={t8:.0f}s) ===")
        a_out, _ = _lp_nesterov_diagavg(best_seq, t8, solver, verbose=True)
        _try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 9: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t9 = min(remaining * 0.25, 100.0)
    if t9 > 5:
        print(f"\n=== Phase 9: Final CoordDesc n={n_best} (budget={t9:.0f}s) ===")
        a_out, _ = _coordinate_descent(best_seq, t9, verbose=False)
        _try_update(a_out, f"coord_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            if fs < _evaluate_safe(best_seq) - 1e-10:
                best_seq = best_seq.copy()
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
