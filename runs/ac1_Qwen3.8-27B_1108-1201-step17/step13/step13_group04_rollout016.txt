def lse_gradient(a, T):
    """
    Compute gradient of LSE surrogate: T*log(sum_k exp(c_k/T)) where c = conv(a,a).
    Correct gradient: dL/da[m] = 2 * sum_j a[j] * w[m+j]
    """
    n = len(a)
    c = fast_autocorr(a)
    max_c = c.max()
    w = np.exp((c - max_c) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    # Compute r[m] = sum_j a[j] * w[m+j] = (rev(a) * w)[m + n - 1]
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1], fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    r = conv[n-1 : n-1+n]
    
    return 2.0 * r

def lp_best_response(a, M, n, solver=None):
    """
    Solve: max sum(b) s.t. conv(a,b) <= M (element-wise), b >= 0.
    Uses direct computation for small n, or a simpler heuristic for large n.
    """
    from scipy.sparse import csr_matrix, lil_matrix
    import cvxpy as cp
    
    # Build constraint matrix: (a*b)_k = sum_i a[|k-i| ... ] hmm
    # conv(a,b)_k = sum_i a[i]*b[k-i] for valid i, k-i
    # Constraint: for each k in 0..2n-2, sum_{i: 0<=i<n, 0<=k-i<n} a[i]*b[k-i] <= M
    
    # For each k, the constraint is: sum_{j: max(0,k-n+1)<=j<=min(n-1,k)} a[k-j]*b[j] <= M
    # Let's build the matrix T[k,j] = a[k-j] if 0<=k-j<n, else 0
    
    T = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            T[k, j] = a[k - j]
    T = T.tocsr()
    
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    
    try:
        if solver is not None:
            prob.solve(solver=solver, verbose=False)
        else:
            prob.solve(verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val
