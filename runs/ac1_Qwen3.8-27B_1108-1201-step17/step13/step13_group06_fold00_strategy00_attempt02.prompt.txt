def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.

    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:

        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    sequence = [min(1000.0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def compute_topk_lse(a, T, topk):
    """Compute top-k LSE objective and gradient."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_lse(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_topk_lse(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=30, topk=20):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.85 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 60.0) * 5))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 6 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 30 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))
                        m_z = np.zeros(n)
                        v_z = np.zeros(n)
                        t_adam = 0

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """
    Systematic peak shaving: for each top peak, find the most contributing
    pair and try to reduce it while checking global peak doesn't increase.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.99
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        # Compute danger zones
        danger = np.zeros(n, dtype=bool)
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            danger[js:je + 1] = True

        improved = False
        for k in top_peaks[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destinations
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue

            # Try multiple destinations and step sizes
            found = False
            for dst in dst_pool[:8]:
                for frac in [0.005, 0.01, 0.02, 0.04, 0.07, 0.10]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-8):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 10 == 0:
                n_pert = rng.integers(1, 5)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 200:
                break

    return best_a, best_score


def mirror_descent_refine(a0, time_limit, eta=0.01, seed=999):
    """
    Multiplicative weights (mirror descent) refinement.
    At each step, compute gradient at top peaks and apply multiplicative update.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    current_eta = eta
    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        # Find peaks within 1% of max
        threshold = bmax * 0.99
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        # Compute importance: how much each position contributes to top peaks
        importance = np.zeros(n)
        for k in top_peaks[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            importance[idx] += a[k - idx]

        # Multiplicative update: decrease important positions, increase unimportant
        # Scale importance to [0, 1]
        imax = importance.max()
        if imax > 1e-15:
            importance = importance / imax

        # Update: a[j] *= (1 - eta * importance[j]) / (1 - eta * avg_importance)
        updates = 1.0 - current_eta * importance
        updates = np.maximum(updates, 0.01)
        updates = updates / updates.mean()
        a_new = a * updates
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Try smaller eta or random perturbation
            if stall % 50 == 0:
                current_eta /= 2
                if current_eta < 1e-6:
                    current_eta = eta
                    # Random restart from best
                    a = best_a.copy()
                    noise = rng.normal(0, 0.01, n)
                    a = a * (1 + noise)
                    a = np.maximum(a, 0)
                    sn = a.sum()
                    if sn > 1e-10:
                        a = a / sn

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):

        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)

        x = np.linspace(0, 1, n)

    s = a.sum()

def perturbed_init(a0, rng, scale=0.08):
    """Create a perturbed version of a sequence."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
