"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length mirror descent (n=1600 -> 2400 -> 3200 -> 4096)
2. Adaptive step-size mirror descent with Exponential Weighted Gradient

5. Random restarts with perturbation for escaping local minima

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute linear autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list or float(np.sum(a_list)) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def normalize_to_sum(a, target_sum):
    """Scale a non-negative array to have the given sum."""
    s = a.sum()
    if s < 1e-30:
        return np.full(len(a), target_sum / len(a))
    return a * (target_sum / s)

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    return result  # sum will be approximately 1 (trapezoidal)

def compute_subgrad_topk(a, ac, top_k=8, beta=4.0):
    """Compute subgradient weighted by top-k peak heights."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0.0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        j = k - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def mirror_descent(a0, time_limit, top_k=8, beta=4.0, lr_scale=1.0, verbose=True):
    """Mirror descent with exponential weights on the simplex."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) * c / n
    else:
        a = a * c / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average for convergence
    avg_a = np.zeros(n)
    n_avg = 0
    
    gamma = lr_scale
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        step = gamma / (np.sqrt(t) + 1.0)
        
        g = compute_subgrad_topk(a, ac, top_k=top_k, beta=beta)
        
        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        
        # Check improvement
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-30:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def surgical_coordinate_descent(a0, time_limit, verbose=True):

    """

    else:
        a = a * c / s
    
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    ac = fast_autocorr(a)
    peak_lags = np.argsort(ac)[::-1][:16]
    
    stall_count = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall_count < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        peak_lags = np.argsort(ac)[::-1][:16]
        
        improved = False
        
        for k in peak_lags[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            j = k - idx
            contrib = a[idx] * a[j]
            
            if contrib.max() < 1e-20:
                continue
            
            # Find the source: position with max contribution
            src_pos = idx[np.argmax(contrib)]
            
            # Compute "danger" for each position: how much it contributes to top peaks
            danger = np.zeros(n)
            for k2 in peak_lags[:8]:
                k2 = int(k2)
                js = max(0, k2 - n + 1)
                je = min(n - 1, k2)
                if js > je:
                    continue
                idx2 = np.arange(js, je + 1)
                j2 = k2 - idx2
                danger[idx2] += a[j2]  # contribution to peak if this position gets more mass
            
            # Find safe destinations: low danger, not at source
            danger[src_pos] = float('inf')
            # Also exclude positions that are sources for other peaks
            for k2 in peak_lags[:8]:
                k2 = int(k2)
                js = max(0, k2 - n + 1)
                je = min(n - 1, k2)
                idx2 = np.arange(js, je + 1)
                # The "other" member of each pair
                j2 = k2 - idx2
                danger[j2] += a[idx2] * 0.5
            
            # Pick from least dangerous positions
            safe_positions = np.where(danger < danger.max() * 0.1)[0]
            if len(safe_positions) == 0:
                safe_positions = np.argsort(danger)[:max(1, n // 4)]
            
            # Try moving small amounts
            best_trial_M = M
            best_trial_a = None
            best_dst = None
            
            # Sample a few destinations
            n_try = min(20, len(safe_positions))
            dst_candidates = safe_positions[rng.choice(len(safe_positions), size=n_try, replace=False)]
            
            for dst in dst_candidates:
                if dst == src_pos:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = a[src_pos] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src_pos] -= amt
                    a_trial[dst] += amt
                    if a_trial[src_pos] < 0:
                        continue
                    a_trial = a_trial * c / a_trial.sum()
                    M_trial = fast_autocorr(a_trial).max()
                    if M_trial < best_trial_M:
                        best_trial_M = M_trial
                        best_trial_a = a_trial.copy()
                        best_dst = dst
            
            if best_trial_a is not None and best_trial_M < best_m:
                a = best_trial_a
                best_m = best_trial_M
                best_a = a.copy()
                improved = True
                stall_count = 0
                break
        
        if not improved:
            stall_count += 1
            # Random perturbation to escape
            if stall_count % 10 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c * 1e-6:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    if a[i] >= 0:
                        a = a * c / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall_count = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m / (c * c)
        print(f"    SCD ({elapsed:.1f}s): J={J:.6f}")

def gradient_descent_averaged(a0, time_limit, verbose=True):
    """
    Gradient method with Polyak averaging, using subgradient of max convolution.
    """
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) * c / n
    else:
        a = a * c / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    elif shape_type == 'triangle':
        f = 1 - np.abs(2 * x - 1)
    elif shape_type == 'gaussian':

import numpy as np

_global_rng = None

def get_rng(seed=42):
    global _global_rng
    if _global_rng is None:
        _global_rng = np.random.default_rng(seed)
    return _global_rng

def fast_autocorr(a):
    """Compute linear autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list:
            return float('inf')
        if float(sum(a_list)) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0.0)
    ss = result.sum()
    if ss < 1e-30:
        return np.ones(new_n) / new_n
    return result / ss

def water_filling_init(a, n_target, clip_percentile=95.0, iterations=20):
    """Create a flattened initialization by clipping peaks and redistributing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n_target) / n_target
    a = a / s
    # Interpolate to target length first
    a_interp = interpolate_to_length(a, n_target)
    # Iteratively clip peaks and redistribute
    for _ in range(iterations):
        threshold = np.percentile(a_interp, clip_percentile)
        if threshold < 1e-15:
            break
        excess = np.maximum(a_interp - threshold, 0.0)
        a_interp = np.minimum(a_interp, threshold)
        deficit_mask = a_interp < threshold
        n_deficit = deficit_mask.sum()
        if n_deficit == 0:
            break
        total_excess = excess.sum()
        a_interp[deficit_mask] = np.minimum(
            a_interp[deficit_mask] + total_excess / n_deficit, threshold
        )
        # Fix any rounding
        ss = a_interp.sum()
        if ss > 1e-30:
            a_interp = a_interp / ss
    return a_interp

def compute_subgrad_topk(a, ac, top_k=10, beta=4.0):
    """Compute subgradient weighted by top-k peak heights with exponential weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0.0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j = k - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def compute_subgrad_smooth(a, ac, power=8.0):
    """Compute subgradient using all lags weighted by (b_k/max_b)^power via FFT."""
    n = len(a)
    max_b = ac.max()
    if max_b < 1e-30:
        return np.zeros(n)
    w = (np.maximum(ac, 0.0) / max_b) ** power
    # grad_i = 2 * sum_k w[k] * a[k-i] for valid k-i
    # This is 2 * correlate(w, a)[i] where correlate uses 'valid' mode
    # Using FFT: correlate(w, a, 'valid')[i] = sum_j w[j+i] * a[j]
    # We can compute this as: ifft(fft(w) * conj(fft(a))) gives circular correlation
    # For linear: pad to size >= len(w) + n - 1
    N = 1 << (len(w) + n - 2).bit_length()
    W = rfft(w, N)
    A = rfft(np.zeros(N))
    A[:n] = a
    # Cross-correlation: ifft(conj(FFT(w)) * FFT(a)) gives corr[i] = sum_k w[k] * a[k-i+N]
    # We want sum_k w[k] * a[k-i] which is the correlation at lag i
    # Actually: ifft(FFT(w) * conj(FFT(a))[::-1]) ... this is getting confusing
    # Let me use a different approach: 
    # corr[i] = sum_j w[j] * a[j-i] ... no
    # I want: g[i] = sum_k w[k] * a[k-i] where 0 <= k-i < n
    # Let m = k-i: g[i] = sum_m w[m+i] * a[m]
    # This is the "sliding dot product" of w shifted by i with a
    # = correlate(w, a, 'full')[i + n - 1] ... let me verify
    # np.correlate(w, a, 'full')[k] = sum_j w[j] * conj(a[j - k + len(a) - 1])
    # For real: = sum_j w[j] * a[j - k + n - 1]
    # We want sum_j w[j+i] * a[j] = sum_j w[j'] * a[j' - i] where j' = j+i
    # = sum_j' w[j'] * a[j' - i]
    # This matches np.correlate(w, a, 'full')[k] with j - k + n - 1 = j' - i
    # So k = n - 1 + i
    # Thus g[i] = np.correlate(w, a, 'full')[n - 1 + i]
    
    # For FFT-based computation:
    # correlate(w, a, 'full') = ifft(fft(w, N) * conj(fft(a, N)))[:len(w)+len(a)-1]
    # Let me verify the indexing with a simple case
    # Actually, in scipy: correlate(W, V) = ifft(fft(W) * conj(fft(V)))
    # The result has length len(W) + len(V) - 1
    # result[k] = sum_n W[n] * V[n - k + len(V) - 1]
    
    # So: corr_full[k] = sum_n w[n] * a[n - k + n - 1]  (where n here is len(a))
    # We want g[i] = sum_m w[m+i] * a[m]
    # Let's set: in corr_full[k], we have sum_n w[n] * a[n-k+n_a-1]
    # We want: sum_m w[m+i] * a[m]
    # Let n = m+i: sum_m w[m+i] * a[m] = sum_n w[n] * a[n-i]
    # In corr_full: sum_n w[n] * a[n-k+n_a-1]
    # Match: n-k+n_a-1 = n-i => k = n_a - 1 + i
    # So g[i] = corr_full[n_a - 1 + i]
    
    N_fft = 1 << (len(w) + n - 2).bit_length()
    W_fft = rfft(w, N_fft)
    A_fft = rfft(a, N_fft)
    corr_full = irfft(W_fft * np.conj(A_fft), N_fft)[:len(w) + n - 1]
    
    # g[i] = corr_full[n - 1 + i] for i = 0, ..., n-1
    grad = 2.0 * corr_full[n-1:n-1+n]
    return grad

def mirror_descent(a0, time_limit, top_k=10, beta=4.0, lr_scale=1.0, 
                   use_smooth_subgrad=False, smooth_power=8.0, verbose=True):
    """Mirror descent with exponential weights update on the probability simplex."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) * c / n
    else:
        a = a * c / s

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    gamma = lr_scale

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(a)
        M = ac.max()

        step = gamma / (np.sqrt(t) + 1.0)

        if use_smooth_subgrad:
            g = compute_subgrad_smooth(a, ac, power=smooth_power)
        else:
            g = compute_subgrad_topk(a, ac, top_k=top_k, beta=beta)

        norm_g = np.linalg.norm(g, ord=np.inf)
        if norm_g < 1e-30:
            break
        # Scale step to keep update reasonable
        step = min(step, 0.5 / norm_g)

        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new

        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-30:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.6f}")

    return best_a, best_m

            danger[src_pos] = float('inf')

            # Find safe destinations
            threshold = np.percentile(danger[danger < float('inf')], 30)
            safe_positions = np.where(danger <= threshold)[0]
            if len(safe_positions) == 0:
                finite_danger = danger[danger < float('inf')]
                threshold = np.percentile(finite_danger, 20)
                safe_positions = np.where(danger <= threshold)[0]
            if len(safe_positions) == 0:
                safe_positions = np.argsort(danger)[:max(1, n // 5)]

            # Try moving mass
            best_trial_M = M
            best_trial_a = None

            n_try = min(30, len(safe_positions))
            if n_try > 0:
                dst_indices = rng.choice(len(safe_positions), size=n_try, replace=False)
                for di in dst_indices:
                    dst = safe_positions[di]
                    if dst == src_pos:
                        continue
                    for frac in [0.003, 0.008, 0.015, 0.03]:
                        amt = a[src_pos] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src_pos] -= amt
                        a_trial[dst] += amt
                        if a_trial[src_pos] < 0:
                            continue
                        a_trial = a_trial * c / a_trial.sum()
                        M_trial = fast_autocorr(a_trial).max()
                        if M_trial < best_trial_M:
                            best_trial_M = M_trial
                            best_trial_a = a_trial.copy()

            if best_trial_a is not None and best_trial_M < best_m * (1 - 1e-10):
                a = best_trial_a
                best_m = best_trial_M
                best_a = a.copy()
                improved = True
                stall_count = 0
                break

        if not improved:
            stall_count += 1
            if stall_count % 20 == 0:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c * 1e-7:
                    amt = a[i] * rng.uniform(0.005, 0.04)
                    a[i] -= amt
                    a[j] += amt
                    if a[i] >= 0:
                        a = a * c / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall_count = 0

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m / (c * c)
        print(f"    SCD ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def multi_start_mirror_descent(a0, time_limit, n_starts=3, verbose=True):
    """Run mirror descent from multiple perturbed starting points."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a_base = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a_base.sum()
    if s < 1e-30:
        a_base = np.ones(n) * c / n
    else:
        a_base = a_base * c / s

    rng = np.random.default_rng(777)
    best_a = a_base.copy()
    best_m = fast_autocorr(a_base).max()
    t0 = time.monotonic()

    time_per_start = time_limit / n_starts

    for start_idx in range(n_starts):
        if time.monotonic() - t0 > time_limit:
            break

        # Create perturbation
        if start_idx == 0:
            a_start = a_base.copy()
        else:
            noise = rng.normal(0, 0.1, n)
            a_start = a_base + noise
            a_start = np.maximum(a_start, 0.0)
            ss = a_start.sum()
            if ss < 1e-30:
                a_start = a_base.copy()
            else:
                a_start = a_start * c / ss

        remaining = time_limit - (time.monotonic() - t0)
        t_this = min(time_per_start, remaining)
        if t_this < 5:
            break

        a_out, m_out = mirror_descent(a_start, t_this, top_k=8, beta=4.0,
                                      lr_scale=1.0 + 0.3 * start_idx, verbose=False)
        if m_out < best_m:
            best_m = m_out
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m / (c * c)
                print(f"    MS start={start_idx} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def make_shape(n, shape_type):
    """Create various initial shapes normalized to sum=1."""
    x = np.linspace(0, 1, n)
    if shape_type == 'sin2':
        f = np.sin(np.pi * x) ** 2
    elif shape_type == 'sin4':
        f = np.sin(np.pi * x) ** 4
    elif shape_type == 'raised_cos':
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif shape_type == 'triangle':
        f = 1.0 - np.abs(2.0 * x - 1.0)
    elif shape_type == 'gaussian':
        f = np.exp(-6.0 * (x - 0.5) ** 2)
    elif shape_type == 'hat':
        # Sharper peak than triangle
        f = np.maximum(0, 1 - 4.0 * (x - 0.5) ** 2)
    elif shape_type == 'bumps':
        # Multi-peak
        f = 0.3 + 0.7 * np.sin(np.pi * x) ** 2 * np.cos(np.pi * x) ** 2
    else:
        f = np.ones(n)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        f = np.ones(n)
        s = n
    return f / s
