    return _global_solver


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v, target=1.0):
    n = len(v)
    if n <= 1:
        return np.array([target])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out = out * target / s
    else:
        out = np.full(n, target / n)
    return out


def _build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lse_gradient(y, T_temp):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = np.clip((ac - ac_max) / T_temp, -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(ac)] = 1.0
    else:
        w = w / w_sum
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1:min(2 * n - 1, fft_size)]
    if len(grad) < n:
        grad = np.resize(grad, n)
    return grad


def _lse_fista(a0, T_schedule, iters_per_stage, time_limit):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    prev_a = a.copy()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        L = max(n / (T * 2.0), 10.0)
        max_iters = max(int(iters_per_stage * (T_schedule[0] / T) ** 0.4), 3000)

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            grad = _lse_gradient(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z, 1.0)

            momentum = prev_a + ((np.sqrt(1.0 + 4.0 * (it + 1) ** 2) - 1.0) / (2.0 * (it + 1))) * (a_new - a)
            momentum = np.maximum(momentum, 0.0)
            ms = momentum.sum()
            if ms > 1e-15:
                momentum = momentum / ms
            else:
                momentum = a_new.copy()

            prev_a = a.copy()
            a = a_new
            y = momentum

            if it % 300 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        elapsed = time.monotonic() - t0
        print(f"    LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    sc = _fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    dual_vals = None
    try:
        dual_vals = np.asarray(prob.dual_values, dtype=np.float64)
    except Exception:
        passive_vals = None
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    return b_val, dual_vals


def _lp_nesterov_polyak(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    tail_avg = np.zeros(n)
    tail_count = 0
    tail_window = 200
    tail_history = []

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        b, duals = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        gap = best_score - M_curr
        if gap < 0:
            gap = 0.0

        y_to_g = y - g
        dist_sq = np.dot(y_to_g, y_to_g)
        if dist_sq < 1e-30:
            alpha = 2.0 / (t + 2.0)
        elif gap > 1e-10:
            alpha_polyak = gap / dist_sq
            alpha_nesterov = 2.0 / (t + 2.0)
            alpha = min(max(alpha_polyak, alpha_nesterov * 0.5), alpha_nesterov * 2.0)
            alpha = min(alpha, 1.0)
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t <= 1:
            a_new = g.copy()
        else:
            momentum_coeff = (np.sqrt(1.0 + 4.0 * t ** 2) - 1.0) / (2.0 * t)
            a_new = y_new + momentum_coeff * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        tail_history.append(a_new.copy())
        if len(tail_history) > tail_window:
            tail_history.pop(0)

        if n_iters % 10 == 0:
            tail_avg = np.mean(tail_history, axis=0)
            s_tail = tail_avg.sum()
            if s_tail > 1e-15:
                tail_avg = tail_avg * norm / s_tail
                score_tail = _fast_autocorr(tail_avg).max()
                if score_tail < best_score:
                    best_score = score_tail
                    best_a = tail_avg.copy()
                    stall_count = 0

        if stall_count > 500 and stall_count % 100 == 0:
            y = best_a.copy()
            prev_y = best_a.copy()
            tail_history = []
            stall_count = 0
            if verbose:
                print(f"    LP-Nest restart at iter={n_iters} ({time.monotonic()-t0:.0f}s)")

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.8f}, alpha={alpha:.4f}")

    if len(tail_history) > 0:
        tail_avg = np.mean(tail_history, axis=0)
        s_tail = tail_avg.sum()
        if s_tail > 1e-15:
            tail_avg = tail_avg * norm / s_tail
            score_tail = _fast_autocorr(tail_avg).max()
            if score_tail < best_score:
                best_score = score_tail
                best_a = tail_avg.copy()

    return best_a, best_score


def _peak_shave(a, time_limit, seed=42):
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if js2 <= je2:
                    peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue

            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 5)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.0005, 0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall <= 2000:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    for frac in [0.002, 0.008, 0.02]:
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        if a_trial[i] >= 0:
                            s_t = a_trial.sum()
                            if s_t > 1e-15:
                                a_trial = a_trial * norm / s_t
                                score = _fast_autocorr(a_trial).max()
                                if score < best_score:
                                    a = a_trial
                                    best_score = score
                                    best_a = a.copy()
                                    improved = True
                                    stall = 0
                                    break
            else:
                break

    return best_a, best_score


def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _grad_exact(a, k_star):
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    for i in range(js, je + 1):
        grad[i] += 2.0 * a[k_star - i]
    return grad


def _subgrad_descent_peak(a0, time_limit, n_peaks=3):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    step_size = 0.05

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = _fast_autocorr(a)
        top_idx = np.argsort(b)[::-1][:n_peaks]
        grad_sum = np.zeros(n)
        for k in top_idx:
            grad_k = _grad_exact(a, k)
            grad_sum += grad_k

        grad_norm = np.linalg.norm(grad_sum)
        if grad_norm < 1e-15:
            break

        direction = -grad_sum / grad_norm
        alpha = step_size / (1.0 + 0.01 * t)
        a_new = a + alpha * direction
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new

        score = _fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.95
            if step_size < 1e-8:
                step_size = 0.01

    return best_a, best_score
