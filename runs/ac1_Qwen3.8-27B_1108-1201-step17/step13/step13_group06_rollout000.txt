Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using gradient-based optimization,
multiplicative weights, and coordinate descent on non-negative sequences.
Key insight: for non-negative a, max(autocorr) = sum(a_i^2), so the objective
is 2*n*sum(a_i^2)/(sum(a_i))^2, minimized by the constant sequence.
However, we also explore local optimization in case the effective objective
differs from this simplification.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective score."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fast_autocorr(a)
    return 2.0 * len(a) * b.max() / (s * s)


def safe_normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gradient_descent(a0, time_limit, lr_init=0.01, decay=0.999, seed=42):
    """
    Gradient descent on the score function.
    For non-negative a, score = 2n * sum(a^2) / (sum(a))^2
    Gradient: 4n*(a_i*S - Q)/S^3 for each i
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = safe_normalize(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = lr_init
    patience = 0
    
    while time.monotonic() - t0 < time_limit - 1:
        S = a.sum()
        Q = np.dot(a, a)
        # Gradient of 2n*Q/S^2
        grad = 4.0 * n * (a * S - Q) / (S ** 3)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        a_new = safe_normalize(a_new)
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            lr *= 0.95
            if patience > 200:
                # Random restart
                a = best_a * (1 + rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                a = safe_normalize(a)
                lr = lr_init
                patience = 0
        
        if (int((time.monotonic() - t0) * 10) % 50 == 0):
            lr *= decay
    
    return best_a, best_score
