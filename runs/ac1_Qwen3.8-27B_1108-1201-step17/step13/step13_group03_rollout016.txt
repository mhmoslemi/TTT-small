def _proj_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    T = 1.0
    eta = 1.0 / np.sqrt(n)
    t_iter = 0

    for stage in range(8):
        T = max(0.05, 10.0 * (0.4 ** stage))
        stage_start = time.monotonic()
        stage_time = time_budget / 8.0

        while time.monotonic() - t0 < time_budget:
            if time.monotonic() - stage_start > stage_time:
                break

            t_iter += 1
            grad = _lse_full_grad(y, T)
            grad = np.maximum(grad, 0.0)
            eta_t = eta / np.sqrt(t_iter + 1)

            log_update = -eta_t * grad
            log_update = np.clip(log_update, -50.0, 50.0)
            a_new = y * np.exp(log_update)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / s_new

            alpha = 2.0 / (t_iter + 2.0)
            y_new = (1.0 - alpha) * y + alpha * a_new
            y_new = np.maximum(y_new, 1e-15)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new / s_y

            y = y_new
            a = a_new

            if t_iter % 100 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                sc_y = _fast_autocorr(y).max()
                if sc_y < best_score:
                    best_score = sc_y
                    best_a = y.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent stage={stage} T={T:.3f} ({elapsed:.0f}s): peak={best_score:.10f}")

    return best_a, best_score


def _multi_peak_opt(a0, time_budget, verbose=True):
    """Optimize using multi-peak composite objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    L = n / 5.0
    t_fista = 1.0
    staging = a.copy()

    for it in range(10000000):
        if time.monotonic() - t0 > time_budget:
            break

        grad = _multi_peak_grad(y, top_K=10, T=0.1)
        step = 1.0 / L
        z = y - step * grad
        a_new = _proj_simplex(z)

        mom = (t_fista - 1.0) / (t_fista + 2.0)
        y_new = a_new + mom * (a_new - staging)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new / s_y
        else:
            y_new = a_new.copy()

        staging = a_new.copy()
        a = a_new
        y = y_new
        t_fista += 1.0

        if it % 100 == 0:
            sc = _fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _peak_shave_multi(a0, time_budget, seed=42):
    """Aggressive multi-peak shaving via mass redistribution."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_budget:
        ac = _fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:12]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue

            cand_sorted = candidates[np.argsort(a[candidates])]
            n_try = max(1, len(cand_sorted) // 8)
            dsts = cand_sorted[rng.integers(0, len(cand_sorted), min(n_try, len(cand_sorted)))]

            for dst in dsts:
                for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    score = _fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial / s_t
                            score = _fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
            if stall > 1000:
                break

    return best_a, best_score


def _structured_restart(a, seed, n_freqs=5):
    """Add structured sinusoidal perturbation to escape local minima."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a_new = a.copy()
    total_amp = 0.02
    for _ in range(n_freqs):
        freq = rng.integers(2, max(3, n // 10))
        phase = rng.uniform(0, 2 * np.pi)
        amp = total_amp / n_freqs * rng.uniform(0.5, 1.5)
        x = np.arange(n)
        a_new += amp * np.sin(2 * np.pi * freq * x / n + phase)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _interpolate_to_length(f, new_n):
    """Interpolate sequence to new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function: multi-method optimization with diversified geometry.
    Uses LSE-FISTA, mirror descent, multi-peak optimization, and peak shaving
    in a carefully scheduled sequence with structured restarts.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LSE-FISTA at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        a_out, _ = _lse_fista(best_seq, t1, verbose=True)
        _try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: Mirror descent (different geometry) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 280.0)
    if t2 > 20:
        print(f"\n=== Phase 2: Mirror Descent n={len(best_seq)} (budget={t2:.0f}s) ===")
        a_out, _ = _mirror_descent(best_seq, t2, verbose=True)
        _try_update(a_out, f"md_n{len(best_seq)}")

    # === Phase 3: LSE-FISTA again (potally improved by mirror descent) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: LSE-FISTA 2 n={len(best_seq)} (budget={t3:.0f}s) ===")
        a_out, _ = _lse_fista(best_seq, t3, verbose=True)
        _try_update(a_out, f"lse2_n{len(best_seq)}")

    # === Phase 4: Multi-peak optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Multi-Peak Opt n={len(best_seq)} (budget={t4:.0f}s) ===")
        a_out, _ = _multi_peak_opt(best_seq, t4, verbose=True)
        _try_update(a_out, f"mp{n}") if False else _try_update(a_out, f"mp_n{len(best_seq)}")

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Peak Shave n={n_cur} (budget={t5:.0f}s) ===")
        a_out, _ = _peak_shave_multi(best_seq, t5, seed=12345)
        _try_update(a_out, f"ps1_n{n_cur}")

    # === Phase 6: Structured restart + LSE-FISTA ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 6: Restart + LSE-FISTA n={n_cur} ===")
        a_restart = _structured_restart(best_seq, seed=777, n_freqs=3)
        t6 = min(remaining * 0.15, 150.0)
        a_out, _ = _lse_fista(a_restart, t6, verbose=False)
        _try_update(a_out, f"restart1_n{n_cur}")

    # === Phase 7: Mirror descent from restart ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 80:
        print(f"\n=== Phase 7: Mirror Descent restart n={n_cur} ===")
        a_restart2 = _structured_restart(best_seq, seed=999, n_freqs=5)
        t7 = min(remaining * 0.12, 120.0)
        a_out, _ = _mirror_descent(a_restart2, t7, verbose=False)
        _try_update(a_out, f"md_restart_n{n_cur}")

    # === Phase 8: Peak shaving with different seed ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 8: Peak Shave 2 n={n_cur} ===")
        t8 = min(remaining * 0.15, 120.0)
        a_out, _ = _peak_shave_multi(best_seq, t8, seed=77777)
        _try_update(a_out, f"ps2_n{n_cur}")

    # === Phase 9: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    if remaining > 200:
        print(f"\n=== Phase 9: LSE-FISTA longer n={n_long} ===")
        init_long = _interpolate_to_length(best_seq, n_long)
        t9 = min(remaining * 0.25, 300.0)
        a_out, _ = _lse_fista(init_long, t9, verbose=True)
        _try_update(a_out, f"lse_long_n{n_long}")

    # === Phase 10: Mirror descent on longer ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 150:
        print(f"\n=== Phase 10: Mirror Descent n={n_cur} ===")
        t10 = min(remaining * 0.25, 200.0)
        a_out, _ = _mirror_descent(best_seq, t10, verbose=True)
        _try_update(a_out, f"md_final_n{n_cur}")

    # === Phase 11: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 11: Final Peak Shave n={n_cur} ===")
        t11 = min(remaining * 0.5, 80.0)
        a_out, _ = _peak_shave_multi(best_seq, t11, seed=4242)
        _try_update(a_out, f"ps_final_n{n_cur}")

    # === Phase 12: Final micro LSE ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 12: Final LSE-FISTA n={n_cur} ===")
        t12 = min(remaining * 0.6, 50.0)
        a_out, _ = _lse_fista(best_seq, t12, verbose=False)
        _try_update(a_out, f"lse_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
