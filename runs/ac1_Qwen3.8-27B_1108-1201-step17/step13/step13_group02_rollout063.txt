import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                print(f"  LP solver: {s}")

        except Exception:

    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")

def lin_autocorr(a):
    """Compute linear autoconvolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient_linear(a, T):
    """Compute gradient of LSE(T * lin_autocorr(a)) with respect to a.
    Returns grad of shape (n,).
    """
    n = len(a)
    b = lin_autocorr(a)  # length 2n-1
    max_b = b.max()
    # Numerically stable softmax
    shifted = (b - max_b) / T
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # grad[i] = 2 * (w * a_rev)[i + n - 1]
    a_rev = a[::-1]
    conv_len = len(w) + n - 1  # (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    grad = 2.0 * conv[n-1:2*n-1]
    
    return grad, w


def lse_descent(a0, time_limit, T_start=None, T_end=None, verbose=True, label=""):
    """LSE smooth gradient descent for minimizing max(linear autoconvolution)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    b0 = lin_autocorr(a)
    max_b0 = b0.max()
    
    if T_start is None:
        T_start = max_b0 * 0.5
    if T_end is None:
        T_end = max_b0 * 0.005
    
    best_a = a.copy()
    best_score = max_b0
    t0 = time.monotonic()
    steps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        # Anneal temperature (geometric schedule)
        T = T_start * (T_end / T_start) ** min(elapsed_frac, 1.0)
        
        b = lin_autocorr(a)
        max_b = b.max()
        
        # Compute gradient
        grad, w = lse_gradient_linear(a, T)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        eta = 2.0 / (T * grad_norm + 1e-10)
        # Limit step to not go too negative
        max_step = 0.5 * a.min() / max(np.abs(grad).max(), 1e-15) if a.min() > 0 else eta
        eta = min(eta, max_step * 2) if a.min() > 0 else eta
        
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        score = lin_autocorr(a_new).max()
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        steps += 1
        
        if no_improve > 200:
            # Perturb to escape local minimum
            rng = np.random.default_rng(steps * 13 + 7)
            perturb = rng.normal(0, 0.01 * a.mean(), n)
            a = np.maximum(a + perturb, 0)
            a = a * norm / a.sum()
            no_improve = 0
        
        if verbose and steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE[{label}] step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        a = a_new
        score_y = lin_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        score_a = lin_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s) best={best_score:.8f}")
        if stall > 400:
            rng = np.random.default_rng(n_iters * 7 + 3)
            ps = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, ps, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, ps, n), 0)
            a = a * norm / a.sum()
            stall = 0
            if verbose:
                print(f"    [Restart iter {n_iters}]")
    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = lin_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = best_score * (0.04 * (1.0 - elapsed_frac) + 0.001)
        b = lin_autocorr(a)
        k_peak = int(np.argmax(b))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        if contrib.max() < 1e-18:
            n_stall += 1
            continue
        src = idx[int(np.argmax(contrib))]
        top_k = np.argsort(b)[::-1][:12]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_k:
            js2 = max(0, int(k2) - n + 1)
            je2 = min(n - 1, int(k2))
            peak_mask[js2:je2 + 1] = True
        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.where(np.arange(n) != src)[0]
        dst = non_peak[int(np.argmin(a[non_peak]))]
        improved = False
        for frac in [0.01, 0.02, 0.05, 0.08, 0.12]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < -1e-12:
                continue
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score = lin_autocorr(a_trial).max()
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                n_improve += 1
                improved = True
                break
            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break
        if not improved:
            n_stall += 1
            if n_stall > 100:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = float(rng.uniform(0.05, 0.15))
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = lin_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                n_improve += 1
                            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                                a = a_trial
                n_stall = 0
        if verbose and n_improve > 0 and n_improve % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s) best={best_score:.8f}")
    return best_a, best_score


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def random_flatspectrum_init(n, seed=0):
    rng = np.random.default_rng(seed)
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.ones(n) * np.exp(1j * phases)
    a = np.real(np.fft.ifft(A)) * n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.sin(np.pi * np.linspace(0, 1, n)) ** 1.5
        a = np.maximum(a, 0)
        s = a.sum()
    a = a / s
    return a


def sine_init(n, p=1.5):
    x = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * x) ** p
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = sine_init(n_base, 1.5)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: Multiple initializations with quick LSE
    print(f"\n=== Phase 1: Multi-init LSE (n={n_base}) ===")
    t1_total = min(time_budget * 0.20, 300.0)
    remaining = time_budget - (time.monotonic() - t_global)
    t1_total = min(t1_total, remaining)
    t_per_init = max(10.0, t1_total / 4)

    inits = [
        (best_seq, "parent"),
        (sine_init(n_base, 2.0), "sine2"),
        (sine_init(n_base, 1.0), "sine1"),
        (random_flatspectrum_init(n_base, seed=99), "randfs"),
    ]

    for init_a, init_label in inits:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < t_per_init:
            break
        a_norm = init_a * np.sqrt(2.0 * n_base) / init_a.sum()
        a_out, _ = lse_descent(a_norm, t_per_init, verbose=False, label=init_label)
        try_update(a_out, f"lse1_{init_label}")

    # Phase 2: Extended LSE on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Extended LSE n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_descent(a_norm, t2, T_start=0.5, T_end=0.01, verbose=True, label="ext")
        try_update(a_out, "lse2_ext")

    # Phase 3: Top-K LSE
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: TopK-LSE n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.08, 120.0)
    if t3 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = topk_lse_descent(a_norm, t3, k=20, verbose=True, label="topk")
        try_update(a_out, "topk")

    # Phase 4: LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.30, 450.0)
    if t4 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t4, solver, verbose=True)
        try_update(a_out, f"lp_n{n_cur}")

    # Phase 5: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: PeakShave n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t5, seed=123, verbose=True)
        try_update(a_out, f"ps_n{n_cur}")

    # Phase 6: Try longer length
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Longer n={n2} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        init_long = interpolate_to_length(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lse_descent(a_norm, min(t6 * 0.4, 80.0), verbose=False, label=f"long{n2}")
        try_update(a_out, f"lse_long{n2}")
        a_out2, _ = lp_nesterov(a_out * np.sqrt(2.0 * n2) / a_out.sum(), min(t6 * 0.6, 120.0), solver, verbose=True)
        try_update(a_out2, f"lp_long{n2}")

    # Phase 7: LSE polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: LSE polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_descent(a_norm, t7, T_start=0.1, T_end=0.001, verbose=True, label="polish")
        try_update(a_out, f"lse_polish_n{n_cur}")

    # Phase 8: Final LP on best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final LP n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.25, 250.0)
    if t8 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov(a_norm, t8, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_cur}")

    # Phase 9: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Final PeakShave n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.25, 200.0)
    if t9 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t9, seed=777, verbose=True)
        try_update(a_out, f"ps_final_n{n_cur}")

    # Final LSE squeeze
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_descent(a_norm, remaining, T_start=0.05, T_end=0.0005, verbose=False, label="final")
        try_update(a_out, "lse_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
