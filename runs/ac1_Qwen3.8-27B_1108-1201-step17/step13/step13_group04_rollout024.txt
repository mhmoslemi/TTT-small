def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_gradient(a, T):
    """
    Compute gradient of LSE surrogate: T*log(sum_k exp(b_k/T)).
    Returns gradient vector of length n.
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    # grad[m] = sum_k w[k] * (a[k-m] + a[m-k]) for valid indices
    # = (a_rev * w)[m + n - 1] + (a * w)[m]
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2

def lse_value(a, T):
    """Compute LSE surrogate value."""
    b = fast_autocorr(a)
    max_b = b.max()
    log_sum = max_b / T + np.log(np.sum(np.exp((b - max_b) / T)))
    return T * log_sum

def nesterov_lse(a0, time_limit, T_start=15.0, T_end=0.3, n_stages=12, verbose=True, seed=42):
    """Nesterov-accelerated LSE gradient descent with geometric cooling and Polyak averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    # Polyak averaging
    avg_a = np.zeros(n)
    n_avg = 0
    avg_window = 50

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(50, int(remaining * 3.0 / max(1, n_stages - si)))
        # Reset averaging for each stage
        avg_a = np.zeros(n)
        n_avg = 0

        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov extrapolation
            if it == 0:
                y_eval = y
                step = 1.0 / 3.0
            else:
                momentum = (np.sqrt(1.0 + 4.0 * it * it) - 1.0) / (2.0 * it)
                y_eval = y + momentum * (y - y_prev)
                y_eval = np.maximum(y_eval, 0.0)
                s_ey = y_eval.sum()
                if s_ey > 1e-15:
                    y_eval = y_eval * norm / s_ey
                step = 1.0 / (it + 3.0)

            grad = lse_gradient(y_eval, T)

            y_new = y - step * grad
            y_prev = y.copy()

            # Project: clip to non-negative, rescale to norm
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y = y_new
            it_count += 1

            # Track best
            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            # Polyak averaging
            avg_a += y
            n_avg += 1
            if n_avg >= avg_window:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    score_avg = fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()

            if verbose and it_count % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={si+1}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

        # Get top K peaks
        top_ks = np.argsort(b)[::-1][:top_k]
        peak_val = b.max()

        improved = False
        found_any = False

        for k_star in top_ks:
            # Find contributing pairs
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]

            for src in top_contrib_idx:
                if a[src] < norm * 1e-10:
                    continue

                        if s_t < 1e-15:

                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            found_any = True
                            stall = 0
                            it_count += 1
                            break
                        elif score < peak_val:
                            # Accept if it reduces the current peak (simulated annealing-like)
                            pass  # Don't accept for now, keep it strict

                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:

            # Random perturbation to escape local minimum

                        # But don't update best if we made it worse

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a, b)_j <= M for all j, b >= 0."""
    if solver is None:
        return None
    n = len(a)

    from scipy.sparse import lil_matrix
    T = lil_matrix((2 * n - 1, n))
    for j in range(2 * n - 1):
        for k in range(n):
            i = j - k
            if 0 <= i < n:
                T[j, k] = a[i]
    T = T.tocsr()

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        # Nearly symmetric with small asymmetry
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 2 + 0.05 * np.sin(2 * np.pi * x)
    else:
        # Random smooth
        a = np.ones(n) * 0.5
        for _ in range(3 + seed % 5):
            center = (seed * 12345 + _) % n
            width = n * (0.05 + 0.1 * ((seed + _) % 3))
            amp = 0.1 + 0.15 * ((seed * 7 + _) % 3)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses LSE-Nesterov, enhanced local search,
    multi-length exploration, and LP verification.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"Warning: could not load parent: {e}")

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, seed)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: LSE-Nesterov at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 420.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t1, T_start=12.0, T_end=0.2, n_stages=14, verbose=True)
        do_try_update(a_out, f"lse_n{n_base}")

    # ============ Phase 2: Enhanced local search ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    t2 = min(remaining * 0.20, 300.0)
    if t2 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = enhanced_local_search(a_work, t2, top_k=8, verbose=True, seed=12345)
        do_try_update(a_out, f"local_n{n_best}")

    # ============ Phase 3: LSE at longer length ============
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LSE-Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_work = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lse(a_work, t3, T_start=15.0, T_end=0.3, n_stages=12, verbose=True)
        do_try_update(a_out, f"lse_n{n2}")

    # ============ Phase 4: Local search at longer length ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LocalSearch n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 220.0)
    if t4 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = enhanced_local_search(a_work, t4, top_k=10, verbose=True, seed=99999)
        do_try_update(a_out, f"local_n{n_best}_2")

    # ============ Phase 5: LP Nesterov ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30 and solver is not None:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_work, t5, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_best}")

    # ============ Phase 6: Final LSE polish with very low T ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 180.0)
    if t6 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t6, T_start=3.0, T_end=0.05, n_stages=8, verbose=True)
        do_try_update(a_out, "lse_polish")

    # ============ Phase 7: Final enhanced local search ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LocalSearch n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.5, 200.0)
    if t7 > 5:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = enhanced_local_search(a_work, t7, top_k=10, verbose=True, seed=77777)
        do_try_update(a_out, "local_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
