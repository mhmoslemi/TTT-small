The solution to this LP gives a direction `b` such that moving from `a` towards `b` (while staying in the feasible region) will reduce the peak. The key property is that `b` is the "best response" - it maximizes the total mass while keeping the convolution with `a` bounded by M.

1. Start with just the top K constraints
2. Solve the LP
3. Check if the solution violates any other constraint
4. If yes, add that constraint and re-solve
5. Repeat until no violation

This is the "cutting plane" method, and it's much faster when only a few constraints are active.

def softmax_subgrad(ac, a, temperature=0.05):
    """Compute subgradient using softmax weighting of all peaks."""
    n = len(a)
    # Normalize peaks for softmax
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    # Use only peaks above a threshold to reduce computation
    threshold = 0.5 * M
    active = np.where(ac >= threshold)[0]
    
    # Softmax weighting
    exp_vals = np.exp((ac[active] - M) / (temperature * M))
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j = k - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def topk_subgrad(ac, a, top_k=10, beta=2.0):
    """Compute subgradient using top-k peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].astype(float)
    top_vals = np.maximum(top_vals, 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j = k - idx

    return grad

        j = k - idx

        constraints.append(row @ b_var <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        M = ac.max()
        if M < 1e-15:
            break

        improved = False

                break
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            j = k - idx
            contrib = a[idx] * a[j]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find best destination
            # Minimum contribution to top peaks
            peak_contrib = np.zeros(n)
            for k2 in top_k[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                j2 = k2 - idx2
                peak_contrib[idx2] += a[j2]
            
            # Exclude positions that are already sources
            mask = np.ones(n, dtype=bool)
            for k2 in top_k[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                mask[idx2] = False
            mask[src] = False
            
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Pick from low-peak-contribution, low-value positions
            scores = peak_contrib[candidates] + 0.1 * a[candidates]
            sorted_idx = np.argsort(scores)
            # Try a few candidates
            n_try = min(5, len(sorted_idx))
            
            for ci in range(n_try):
                dst = candidates[sorted_idx[ci]]
                if dst == src:
                    continue

                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue

                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:

                        best_m = M_trial
                        best_a = a.copy()
                        ac = ac_trial
                        improved = True
                        stall_count = 0

                    break

                break

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def fast_autocorr(a):

def proj_simplex(v, s):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def smooth_subgrad(ac, a, bandwidth=0.08, floor_frac=0.3):
    """Subgradient using smooth Gaussian weighting of peaks relative to max."""
    n = len(a)
    M = ac.max()
    if M < 1e-30:
        return np.zeros(n)
    # Gaussian weighting: w_k = exp(-(ac_k - M)^2 / (2*(bandwidth*M)^2))
    # Only include lags above floor_frac * M for efficiency
    threshold = floor_frac * M
    active = np.where(ac >= threshold)[0]
    if len(active) == 0:
        active = np.argsort(ac)[::-1][:10]
    sigma = bandwidth * M
    if sigma < 1e-15:
        sigma = 1e-15
    diffs = ac[active] - M  # negative or zero
    w = np.exp(-(diffs * diffs) / (2.0 * sigma * sigma))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(active)) / len(active)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, active):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j = k - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def topk_subgrad(ac, a, top_k=8, beta=2.5):
    """Subgradient using top-k peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].astype(float)
    top_vals = np.maximum(top_vals, 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j = k - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def lp_best_response(a, M, active_lags, solver):
    """LP: max sum(b) s.t. (a*b)_k <= M for k in active_lags, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_lags:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(i_start, i_end)
        j = k - idx
        row = np.zeros(n)
        row[j] = a[idx]
        if row.sum() < 1e-20:
            continue
        constraints.append(row @ b_var <= M)
    if not constraints:
        return None
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

        alpha_fw = 2.0 / (t + 2.0)
        best_alpha = alpha_fw

def coordinate_polish(a0, time_limit, verbose=False):
    """Multi-move coordinate descent with random restarts."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

    else:
        a = a * c / s0
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall_count = 0
    max_stall = 500
    while time.monotonic() - t0 < time_limit and stall_count < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        if M < 1e-15:
            break
        top_lags = np.argsort(ac)[::-1][:min(12, len(ac))]
        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            j = k - idx
            contrib = a[idx] * a[j]
            if contrib.max() < 1e-20:
                continue
            src = idx[np.argmax(contrib)]
            # Compute peak exposure for each position
            peak_exposure = np.zeros(n)
            for k2 in top_lags[:6]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                j2 = k2 - idx2
                peak_exposure[idx2] += a[j2]
            # Find good destinations
            mask = np.ones(n, dtype=bool)
            mask[src] = False
            candidates = np.where(mask)[0]
            if len(candidates) == 0:
                continue

            sorted_cand = candidates[np.argsort(scores)]

                    a_trial = a_trial * c / a_trial.sum()
                    ac_trial = fast_autocorr(a_trial)
                    M_trial = ac_trial.max()
                    if M_trial < best_m - 1e-16:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall_count = 0
                        break
                if improved:
                    break
            if improved:
                break

            stall_count += 1
            # Periodic random perturbation
            if stall_count % 20 == 0:
                n_perturb = max(1, n // 200)
                for _ in range(n_perturb):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > c * 1e-6:
                        frac = rng.uniform(0.02, 0.08)
                        amt = a[i] * frac
                        a[i] -= amt
                        a[j] += amt
                        if a[i] >= 0:
                            a = a * c / a.sum()
                            M_new = fast_autocorr(a).max()
                            if M_new < best_m:

                                stall_count = 0
                                break
                            elif stall_count >= 100:
                                pass  # accept anyway to escape
                            else:
                                a[i] += amt
                                a[j] -= amt
                                a = a * c / a.sum()
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = a * c / a.sum()
        if verbose and stall_count % 100 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)

    if verbose:

    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def make_initialization(n, kind="parent", parent=None, seed=42):
    """Create a diverse initialization."""
    rng = np.random.default_rng(seed)
    if kind == "parent" and parent is not None:
        a = parent.copy()
    elif kind == "raised_cosine":
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif kind == "super_gaussian":
        x = np.linspace(-1, 1, n)
        p = rng.uniform(3.0, 8.0)
        a = np.exp(-np.abs(x) ** p)
    elif kind == "hamming":
        a = np.hamming(n)
        a = np.maximum(a, 0.01)
    elif kind == "uniform":
        a = np.ones(n)
    elif kind == "random_smooth":
        a = rng.dirichlet(np.ones(n) * 0.5)
        # Smooth it
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=max(1.0, n / 100.0))
        a = np.maximum(a, 0.0)
    elif kind == "bimodal":
        center = rng.uniform(0.3, 0.7)
        x = np.linspace(0, 1, n)
        width = rng.uniform(0.1, 0.3)
        a = np.exp(-(x - center) ** 2 / (2 * width ** 2))
        # Add a secondary bump
        center2 = 1.0 - center
        a += 0.5 * np.exp(-(x - center2) ** 2 / (2 * width ** 2))
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            state["best_seq"] = parent.copy()
            state["best_score"] = p_score
    except Exception:
        parent = None
    if state["best_seq"] is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        state["best_seq"] = np.sin(np.pi * x) ** 2
        state["best_seq"] = state["best_seq"] / state["best_seq"].sum()
        state["best_score"] = evaluate_safe(state["best_seq"])
        print(f"Default init: score={state['best_score']:.6f}")
    def _try_update(a, label, state):
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < state["best_score"]:
            state["best_score"] = sc
            state["best_seq"] = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    # Phase 1: Hybrid optimization at parent length
    n0 = len(state["best_seq"])
    c0 = np.sqrt(2.0 * n0)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Hybrid n={n0} (budget {remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 300.0)
    if t1 > 30:
        a_norm = state["best_seq"] * c0
        a_out, _ = hybrid_optimize(a_norm, t1, solver, verbose=True)
        _try_update(a_out, f"hybrid1_n{n0}", state)
    # Phase 2: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state["best_seq"])
    c_best = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 2: Coord n={n_best} (budget {remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 200.0)
    if t2 > 15:
        a_init = state["best_seq"] * c_best
        a_out, _ = coordinate_polish(a_init, t2, verbose=True)
        _try_update(a_out, f"coord1_n{n_best}", state)
    # Phase 3: Subgradient with smooth weighting
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state["best_seq"])
    c_best = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 3: Subgrad n={n_best} (budget {remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30:
        a_init = state["best_seq"] * c_best
        a_out, _ = subgradient_optimize(a_init, t3, mode="smooth", verbose=True)
        _try_update(a_out, f"sg1_n{n_best}", state)
    # Phase 4: Try alternative initialization (super-Gaussian)
    remaining = time_budget - (time.monotonic() - t_global)
    n_try = n_best
    print(f"\n=== Phase 4: Alt init n={n_try} (budget {remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 150.0)
    if t4 > 30:
        a_alt = make_initialization(n_try, kind="super_gaussian", seed=7)
        c_try = np.sqrt(2.0 * n_try)
        a_alt_scaled = a_alt * c_try
        a_out, _ = subgradient_optimize(a_alt_scaled, t4, mode="smooth", verbose=True)
        _try_update(a_out, f"alt_init_n{n_try}", state)
    # Phase 5: LP Frank-Wolfe at best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state["best_seq"])
    c_best = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 5: FW-LP n={n_best} (budget {remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_init = state["best_seq"] * c_best
        a_out, _ = frank_wolfe_lp_optimize(a_init, t5, solver, verbose=True)
        _try_update(a_out, f"fwlp_n{n_best}", state)
    # Phase 6: Coordinate polish at best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state["best_seq"])
    c_best = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 6: Coord2 n={n_best} (budget {remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 15:
        a_init = state["best_seq"] * c_best
        a_out, _ = coordinate_polish(a_init, t6, verbose=True)
        _try_update(a_out, f"coord2_n{n_best}", state)
    # Phase 7: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_best + 400
    print(f"\n=== Phase 7: Longer n={n_long} (budget {remaining:.0f}s) ===")
    t7 = min(remaining * 0.12, 150.0)
    if t7 > 30:
        a_long = interpolate_sequence(state["best_seq"], n_long)
        c_long = np.sqrt(2.0 * n_long)
        a_long_scaled = a_long * c_long
        a_out, _ = hybrid_optimize(a_long_scaled, t7, solver, verbose=True)
        _try_update(a_out, f"long_n{n_long}", state)
    # Phase 8: Final hybrid at best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state["best_seq"])
    c_best = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 8: Final hybrid n={n_best} (budget {remaining:.0f}s) ===")
    t8 = min(remaining * 0.30, 300.0)
    if t8 > 15:
        a_init = state["best_seq"] * c_best
        a_out, _ = hybrid_optimize(a_init, t8, solver, verbose=True)
        _try_update(a_out, f"final_hybrid_n{n_best}", state)
    # Phase 9: Final coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state["best_seq"])
    c_best = np.sqrt(2.0 * n_best)
    print(f"\n=== Phase 9: Final coord n={n_best} (budget {remaining:.0f}s) ===")
    t9 = min(remaining * 0.50, 200.0)
    if t9 > 10:
        a_init = state["best_seq"] * c_best
        a_out, _ = coordinate_polish(a_init, t9, verbose=True)
        _try_update(a_out, f"final_coord_n{n_best}", state)
    # Final validation
    if state["best_seq"] is not None:
        fs = evaluate_safe(state["best_seq"])
        if np.isfinite(fs):
            state["best_score"] = fs
        print(f"\nFINAL: score={state['best_score']:.6f}, n={len(state['best_seq'])}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    return [float(x) for x in state["best_seq"]]
