def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient_correct(a, T):

                    y_eval = y_eval * norm / s_e

            grad = lse_gradient_correct(y_eval, T)
            gnorm = np.linalg.norm(grad)
            grad_norms.append(gnorm)

            # Adaptive step size: scale by T and normalize gradient
            step = T / max(gnorm, 1e-15)
            step = min(step, 1.0 / (si + 1) * 2.0)

            y_new = y - step * lse_gradient_correct(y_eval, T)

            # Project
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y_prev = y.copy()
            y = y_new
            it_count += 1

            # Track best
            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()
                local_restart = 0

            # Adaptive restart: if gradient norm keeps increasing, reset momentum
            if len(grad_norms) > 5:
                recent = np.mean(grad_norms[-5:])
                prev5 = np.mean(grad_norms[-10:-5]) if len(grad_norms) >= 10 else recent
                if recent > 1.5 * prev5 and local_restart < 5:
                    y_prev = y.copy()  # Reset momentum
                    local_restart += 1

            if it_count % 100 == 0:
                if it_count % 500 == 0 and verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    LSE-imp stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_shaping(a0, time_limit, verbose=True):
    """
    Frequency-domain multiplicative update (spectral shaping).
    Iteratively flatten the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    it_count = 0

    # Use a geometric schedule for the spectral shaping strength
    n_iters = int(time_limit * 2.0)  # ~2 iterations per second target
    n_iters = max(n_iters, 200)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()

        # Compute shaping weights: emphasize peaks more at early iterations
        decay = 5.0 * (1.0 - float(it) / max(1, n_iters)) + 1.0
        w = np.exp((b - max_b) * decay / max_b)
        w_sum = w.sum()
        if w_sum < 1e-300:
            continue
        w = w / w_sum

        # FFT of current sequence
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P = P / max(P.sum(), 1e-300)

        # Shape the spectrum: reduce power where autocorrelation peaks are
        # The autocorrelation at lag k gets contribution from all frequencies
        # weighted by exp(2*pi*i*k*f). We approximate by reducing overall power
        # in high-autocorrelation directions.
        # Simpler: multiply spectrum by (1 - alpha * w_shifted)
        # where w_shifted is the spectrum of the weighting

        W_spec = rfft(w[:n], fft_size)  # Use first n of the weight
        P_shaped = P * (1.0 - 0.5 * np.abs(W_spec / max(np.abs(W_spec).sum(), 1e-300)))
        P_shaped = np.maximum(P_shaped, 1e-30)

        # Reconstruct: keep phase, use new magnitude
        mag = np.sqrt(P_shaped)
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = mag * phase

        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        # Small interpolation to smooth
        alpha_mix = 0.3
        a = (1.0 - alpha_mix) * a + alpha_mix * a_new
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a > 1e-15:
            a = a * norm / s_a

        it_count += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if it_count % 200 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    SpectralShaping it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_targeting_improved(a0, time_limit, seed=42, verbose=True):
    """
    Improved peak-targeting local search with pair-splitting and
    multi-peak awareness.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_ks = np.argsort(b)[::-1][:8]

        improved = False
        for k_star in top_ks:
            # Find contributing pairs
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_star - idx]
            top_n = min(5, len(idx))
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:top_n]]

            for src in top_contrib_idx:
                partner = k_star - src
                if partner < 0 or partner >= n:
                    continue
                if a[src] < norm * 1e-10 and a[partner] < norm * 1e-10:
                    continue

                # Build peak mask: positions that contribute to top peaks
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_ks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    if js2 <= je2:
                        peak_mask[js2:je2 + 1] = True

                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]

                # Pick candidates with low current values
                cand_sorted = candidates[np.argsort(a[candidates])[:max(5, len(candidates) // 4)]]

                for dst in cand_sorted[:5]:
                    for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.04]:
                        amt_src = a[src] * frac
                        amt_part = a[partner] * frac if partner != src else 0.0
                        if amt_src < 1e-18:
                            continue

                        a_trial = a.copy()
                        a_trial[src] -= amt_src
                        if partner != src:
                            a_trial[partner] -= amt_part
                        a_trial[dst] += amt_src + amt_part

                        if a_trial[src] < -1e-14 or (partner != src and a_trial[partner] < -1e-14):
                            continue

                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t

                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 3 == 0:
                n_perturb = rng.integers(1, 4)
                for _ in range(n_perturb):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > norm * 1e-8:
                        frac = rng.uniform(0.003, 0.03)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0

            if stall > 300:
                if verbose:
                    print(f"    PeakSearch-imp: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                break

    return best_a, best_score


def subgradient_method(a0, time_limit, verbose=True):
    """
    Subgradient method with Polyak step size for the non-smooth max problem.
    Uses top-k peaks for a richer subgradient.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    it_count = 0

    # Lower bound estimate: 2 (theoretical minimum for non-negative sequences)
    # Actually the true lower bound is higher. Use current best as upper bound.
    upper_bound = best_score

    for it in range(1, 20000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        curr_val = b.max()

        # Find top-k peaks for subgradient
        top_k = min(10, 2 * n - 1)
        top_ks = np.argsort(b)[::-1][:top_k]
        top_vals = b[top_ks]

        # Weight by exp((val - max) / scale)
        scale = max((top_vals[0] - top_vals[-1]) / top_k, 1e-15)
        weights = np.exp((top_vals - top_vals[0]) / scale)
        weights /= weights.sum()

        # Compute subgradient: weighted sum of gradients at top peaks
        # d(a*a)_k / da_m = a[k-m] + a[m-k] (with boundary handling)
        grad = np.zeros(n)
        for ik, k in enumerate(top_ks):
            w_k = weights[ik]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            # Contribution: a[idx] * a[k - idx], gradient w.r.t. a_m
            # d/da_m sum_j a_j * a_{k-j} = a_{k-m} + a_{m-k} (for valid indices)
            # For each m in idx: contribution is a[k-m]
            # For each m in k-idx: contribution is a[m-k]... 
            # Actually: (a*a)_k = sum_j a_j * a_{k-j}
            # d/dm = a_{k-m} + a_{m-k} where valid
            for m in idx:
                jm = k - m
                if 0 <= jm < n:
                    grad[m] += w_k * a[jm]
                # The other term: m contributes as a_{k-j} when j = m, so partner is k-m
                # But also m could be the "k-j" index, meaning j = k-m
                # Already handled above
            # Also need: for each j in idx, d/dm at m = k-j gives a_j
            # This is the same set of indices, so we double count? No:
            # (a*a)_k = sum_j a_j * a_{k-j}
            # = sum_j a_j * a_{k-j}
            # d/d a_m: m appears as j (giving a_{k-m}) and as k-j (giving a_j)
            # So d/da_m = a_{k-m} + a_{m-k} for valid indices
            # The first term: a_{k-m} valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
            # The second term: a_{m-k} valid when 0 <= m-k < n, i.e., k <= m <= k+n-1
            # Combined: m in [max(0,k-n+1), min(n-1,k)] for first, [max(0,k), min(n-1,k+n-1)] for second
            # For k < n: first gives m in [0,k], second gives m in [k,n-1]
            # For k >= n: first gives m in [k-n+1,n-1], second gives m in [k,n-1] (empty if k>=n)
            
        # Actually let me redo this more carefully using the FFT approach
        # But for subgradient, we only need the top few, so direct computation is fine for small top_k

        # Recompute gradient properly
        grad = np.zeros(n)
        for ik, k in enumerate(top_ks):
            w_k = weights[ik]
            # d(a*a)_k / a_m = a[(k-m) mod ... no, it's:
            # For the full convolution: (a*a)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_j * a_{k-j}
            # d/da_m = a_{k-m} [if 0<=k-m<n and m in the sum range] + a_{m-k} [if 0<=m-k<n and ...]
            # More simply: d/da_m (sum_j a_j a_{k-j}) = a_{k-m} + a_{m-k} for valid indices
            m1_lo = max(0, k - n + 1)
            m1_hi = min(n - 1, k)
            if m1_lo <= m1_hi:
                ms = np.arange(m1_lo, m1_hi + 1)
                partners = k - ms
                grad[ms] += w_k * a[partners]
            m2_lo = max(0, k)
            m2_hi = min(n - 1, k + n - 1)
            if m2_lo <= m2_hi:
                ms = np.arange(m2_lo, m2_hi + 1)
                partners = ms - k
                # Only add if partner is valid (0 <= partner < n)
                valid = (partners >= 0) & (partners < n)
                if np.any(valid):
                    grad[ms[valid]] += w_k * a[partners[valid]]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-30:
            break

        # Polyak step size
        gap = curr_val - 1.0 / n  # rough lower bound
        if gap < 1e-15:
            gap = 1e-15
        eta = gap / (gnorm * gnorm)
        eta = min(eta, 0.1 / (it + 1))  # cap to prevent too large steps

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        it_count += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            upper_bound = score

        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if it_count % 500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(5.0 * x) + 0.05 * np.cos(11.0 * x)
        a = np.maximum(a, 0.0)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        # Asymmetric: concentrated on one side
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-10.0 * x) + 0.3 * np.exp(-5.0 * (x - 0.8) ** 2)
    elif choice == 6:
        # Quadratic residue inspired
        p = 17
        a = np.ones(n) * 0.5
        for i in range(n):
            pos = i % p
            if pos * pos % p in [1, 4, 9, 16, 8, 2, 15, 13, 7, 3, 12, 10, 6, 5, 14, 11, 0]:
                a[i] = 1.0
            else:
                a[i] = 0.1
    else:
        # Multi-scale
        a = np.ones(n) * 0.3
        for _ in range(5):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.03, n * 0.2)
            amp = rng.uniform(-0.2, 0.4)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
        a = np.maximum(a, 0.0)

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses a multi-phase approach combining LSE gradient
    descent, spectral shaping, subgradient methods, and peak-targeting search
    across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: LSE Nesterov improved at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Improved n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.28, 460.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lse_improved(a_work, t1, T_start=12.0, T_end=0.1,
                                          n_stages=18, verbose=True)
        do_try_update(a_out, f"lse_imp_n{n_base}")

    # ============ Phase 2: Spectral shaping ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: SpectralShaping n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 180.0)
    if t2 > 15:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = spectral_shaping(a_work, t2, verbose=True)
        do_try_update(a_out, f"spectral_n{n_cur}")

    # ============ Phase 3: Subgradient method ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Subgradient n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 15:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_method(a_work, t3, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_cur}")

    # ============ Phase 4: LSE at larger length ============
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LSE-Improved n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 320.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_work = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lse_improved(a_work, t4, T_start=15.0, T_end=0.2,
                                          n_stages=14, verbose=True)
        do_try_update(a_out, f"lse_imp_n{n2}")

    # ============ Phase 5: Spectral shaping at larger length ============
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: SpectralShaping n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 15:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = spectral_shaping(a_work, t5, verbose=True)
        do_try_update(a_out, f"spectral2_n{n_cur}")

    # ============ Phase 6: Peak targeting improved ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: PeakTargeting-imp n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 240.0)
    if t6 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_targeting_improved(a_work, t6, seed=7777, verbose=True)
        do_try_update(a_out, f"peak_imp_n{n_cur}")

    # ============ Phase 7: Low-temperature LSE polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: LSE Polish n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.12, 160.0)
    if t7 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lse_improved(a_work, t7, T_start=2.0, T_end=0.05,
                                          n_stages=10, verbose=True)
        do_try_update(a_out, "lse_polish")

    # ============ Phase 8: Final peak targeting ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.5, 120.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final Peak Targeting n={n_cur} (budget={t8:.0f}s) ===")
            a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = peak_targeting_improved(a_work, t8, seed=31415, verbose=True)
            do_try_update(a_out, "peak_final")

    # ============ Phase 9: Try even larger length if time allows ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n3 = n_base + 800
        t9 = min(remaining * 0.6, 200.0)
        if t9 > 30:
            print(f"\n=== Phase 9: LSE n={n3} (budget={t9:.0f}s) ===")
            init_long3 = interpolate_sequence(best_seq, n3)
            a_work = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
            a_out, _ = nesterov_lse_improved(a_work, t9, T_start=20.0, T_end=0.3,
                                              n_stages=12, verbose=True)
            do_try_update(a_out, f"lse_n{n3}")

            # Quick peak targeting on the result
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                t10 = min(remaining * 0.4, 60.0)
                a_work = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
                a_out, _ = peak_targeting_improved(a_work, t10, seed=27182, verbose=False)
                do_try_update(a_out, "peak_after_long")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
