"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with soft-max approximation and beta annealing (primary method)
2. Adam optimizer on soft-max objective for faster convergence
3. Differential evolution for global exploration
4. Mirror descent with continuous soft subgradient
5. Multi-scale refinement and coordinate descent polish
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_global_best = None

def fast_autocorr(a):
    """Compute linear autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def soft_max_val(values, beta):
    """Compute soft max: (1/beta)*log(sum(exp(beta*x)))."""
    shifted = beta * values
    m = np.max(shifted)
    if m > 500:
        shifted = shifted - m
    return np.max(values) + np.log(np.sum(np.exp(shifted - np.max(shifted)))) / beta

def soft_max_weights(values, beta):
    """Compute softmax(beta * values) as weights."""
    shifted = beta * values - beta * np.max(values)
    e = np.exp(shifted)
    return e / np.sum(e)

def compute_grad_a(a, g_weights):
    """Compute gradient of soft_max(beta, autocorr(a)) w.r.t. a.
    
    grad_a[j] = 2 * sum_m g_weights[j+m] * a[m]
    This equals 2 * conv(g_weights, a[::-1])[j + n - 1]
    """
    n = len(a)
    a_rev = a[::-1]
    size = (2 * n - 1) + n - 1
    fft_sz = 1 << (size - 1).bit_length()
    G = rfft(g_weights, fft_sz)
    Ar = rfft(a_rev, fft_sz)
    cross = irfft(G * Ar, fft_sz)[:size]
    return 2.0 * cross[n - 1:2 * n - 1]

def softmax_normalize(w):
    """Convert log-parameters to probability vector."""
    shifted = w - np.max(w)
    e = np.exp(shifted)
    return e / np.sum(e)

        elapsed = time.monotonic() - t0

            stall_count = 0
        
        if t % 2000 == 0:
            print(f"    Adam t={t}: best={best_score:.6f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score

def mirror_descent_soft(a0, time_limit):
    """Mirror descent with continuous soft subgradient (no top-k truncation)."""
    global _global_best
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    gamma = 0.5
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        # Use exponential weights on all lags (softer than top-k)
        beta_md = 5.0
        g_w = soft_max_weights(ac, beta_md)
        grad_a = compute_grad_a(a, g_w)
        
        # Mirror descent step
        step = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad_a / max(np.max(grad_a), 1e-10)
        a_new = np.exp(log_a_new - np.max(log_a_new))
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if _global_best is None or score < _global_best[1]:
                _global_best = (a_new.copy(), score)
        
        a = a_new
        
        if t % 5000 == 0:
            print(f"    MD-soft t={t}: best={best_score:.6f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score

def differential_evolution(a0, time_limit, pop_size=25, F=0.8, CR=0.9):
    """Differential evolution in the simplex."""
    global _global_best
    n = len(a0)
    rng = np.random.default_rng(12345)
    
    a_start = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a_start = a_start / a_start.sum()
    
    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = a_start
    for i in range(1, pop_size):
        # Mix of random, perturbed parent, and structured init
        if i % 3 == 1:
            pop[i] = rng.dirichlet(np.ones(n) * 0.5)
        elif i % 3 == 2:
            pop[i] = a_start + rng.normal(0, 0.02, n)
            pop[i] = np.maximum(pop[i], 0)
            pop[i] = pop[i] / pop[i].sum()
        else:
            # Structured: sine, triangle, etc.
            x = np.linspace(0, 1, n)
            if i % 6 == 3:
                base = np.sin(np.pi * x) ** 2
            else:
                base = np.exp(-((x - 0.5) ** 2) / 0.05)
            pop[i] = base * (1 + 0.3 * rng.normal(0, 1, n))
            pop[i] = np.maximum(pop[i], 0)
            s = pop[i].sum()
            if s < 1e-10:
                pop[i] = rng.dirichlet(np.ones(n))
            else:
                pop[i] = pop[i] / s
    
    scores = np.array([2.0 * n * fast_autocorr(pop[i]).max() for i in range(pop_size)])
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]
    best_a = pop[best_idx].copy()
    
    if _global_best is None or best_score < _global_best[1]:
        _global_best = (best_a.copy(), best_score)
    
    t0 = time.monotonic()
    gen = 0
    
    while time.monotonic() - t0 < time_limit:
        gen += 1
        current_best_idx = np.argmin(scores)
        current_best = scores[current_best_idx]
        
        for i in range(pop_size):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Select three distinct random individuals
            r1, r2 = rng.choice(pop_size, size=2, replace=False)
            while r1 == i or r2 == i:
                r1, r2 = rng.choice(pop_size, size=2, replace=False)
            
            # Mutation
            trial = pop[r1] + F * (pop[r2] - pop[i])
            # Crossover (binomial)
            mask = rng.random(n) < CR
            if not mask.any():
                mask[rng.integers(n)] = True
            trial = np.where(mask, trial, pop[i])
            
            # Project to simplex
            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s < 1e-15:
                trial = pop[i].copy()
            else:
                trial = trial / s
            
            trial_score = 2.0 * n * fast_autocorr(trial).max()
            
            if trial_score <= scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_a = trial.copy()
                    if _global_best is None or trial_score < _global_best[1]:
                        _global_best = (trial.copy(), trial_score)
        
        if gen % 50 == 0:
            print(f"    DE gen={gen}: best={best_score:.6f} ({time.monotonic()-t0:.0f}s)")
        
        # Adapt F and CR
        if gen % 100 == 0:
            F = max(0.4, F * 0.95)
            CR = min(0.95, 0.7 + 0.2 * gen / 1000)
    
    return best_a, best_score

def coord_descent_polish(a0, time_limit):
    """Stochastic coordinate descent with line search."""
    global _global_best
    n = len(a0)
    rng = np.random.default_rng(99)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_autocorr(a)
        M = ac.max()
        top_lags = np.argsort(ac)[::-1][:5]
        
        improved = False
        for k in top_lags:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if np.max(contrib) < 1e-15:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Try moving to several candidate destinations
            candidates = rng.choice(n, size=min(20, n), replace=False)
            candidates = candidates[candidates != src]
            
            for dst in candidates:
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = a_trial / a_trial.sum()
                    s_trial = 2.0 * n * fast_autocorr(a_trial).max()
                    if s_trial < best_score - 1e-12:
                        a = a_trial
                        best_score = s_trial
                        best_a = a.copy()
                        improved = True
                        if _global_best is None or s_trial < _global_best[1]:
                            _global_best = (a.copy(), s_trial)
                        break
                if improved:
                    break
            if improved:
                stall = 0
                break
        else:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.1)
                a[i] -= amt
                a[j] += amt
                a = a / a.sum()
                s_new = 2.0 * n * fast_autocorr(a).max()
                if s_new < best_score:
                    best_score = s_new
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_score

def nesterov_soft(a0, time_limit, beta=30.0):
    """Nesterov accelerated gradient on soft-max objective."""
    global _global_best
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    y = a.copy()
    x = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    gamma = 1.0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(y)
        g_w = soft_max_weights(ac, beta)
        grad_y = compute_grad_a(y, g_w)
        
        step = gamma / (np.sqrt(t) + 1.0)
        y_new = y - step * grad_y / max(np.linalg.norm(grad_y, np.inf), 1e-10)
        # Project to simplex
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        # Nesterov extrapolation
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / (t + 1.0) if t > 1 else 0.0
        x_new = y_new + beta_mom * (y_new - x)
        x_new = np.maximum(x_new, 0)
        s_x = x_new.sum()
        if s_x < 1e-15:
            x_new = y_new.copy()
        else:
            x_new = x_new / s_new if s_new > 1e-15 else x_new / s_x
        
        x = y
        y = y_new
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            if _global_best is None or score_y < _global_best[1]:
                _global_best = (y.copy(), score_y)
        
        score_x = 2.0 * n * fast_autocorr(x_new).max()
        if score_x < best_score:
            best_score = score_x
            best_a = x_new.copy()
            if _global_best is None or score_x < _global_best[1]:
                _global_best = (x_new.copy(), score_x)
        
        if t % 2000 == 0:
            print(f"    Nesterov t={t}: best={best_score:.6f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score

def interp_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def propose_candidate(budget_s=1800.0):
    """Main search function combining multiple optimization strategies."""
    global _global_best
    _global_best = None
    
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    # Load parent or create default
    best_seq = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
            _global_best = (parent.copy(), p_score)
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            _global_best = (a_norm.copy(), sc)
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        _global_best = (best_seq.copy(), best_score)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    elapsed = lambda: time.monotonic() - t_global
    
    # Phase 1: L-BFGS with soft-max (strongest method)
    remaining = time_budget - elapsed()

    # Phase 2: L-BFGS with different params (restart from best)
    remaining = time_budget - elapsed()
    print(f"\n=== Phase 2: L-BFGS second run n={len(best_seq)} ===")
    n_cur = len(best_seq)

        try_update(a_out, f"lbfgs2_n{n_cur}")
    
    # Phase 3: Adam optimizer
    remaining = time_budget - elapsed()
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Adam n={n_cur} ===")
    t3 = min(remaining * 0.12, 200.0)

    # Phase 4: Nesterov accelerated

    else:

    return [float(x) for x in best_seq]
