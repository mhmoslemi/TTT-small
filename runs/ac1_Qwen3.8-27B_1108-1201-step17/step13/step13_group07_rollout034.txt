- **Subgradient method**: The subgradient of max_k (T@b)_k at b is T^T * e_{k*} where k* = argmax_k (T@b)_k.
- **Frank-Wolfe**: At each step, solve a linear subproblem over the simplex, then take a step.

- At current b, compute g = T^T @ e_{k*} where k* = argmax_k (T@b)_k

  - So g is just a shifted copy of a!

- Update: b <- (1 - gamma) * b + gamma * e_{j*}

This is extremely fast! Each Frank-Wolfe iteration is:
1. Compute T@b (autocorrelation of a and b): O(n log n) with FFT
2. Find k* = argmax: O(n)
3. Find j* = argmin_j a[k*-j]: O(n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping initialization
2. Nesterov-accelerated zero-sum game with Frank-Wolfe LP approximation
3. MWU (Multiplicative Weight Update) as alternative LP solver

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def internal_score(a):
    n = len(a)
    b = fast_autocorr(a)
    return 2.0 * n * b.max()

def project_simplex(v, target_sum=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.ones(n) * target_sum / n
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    result = result / result.sum() * target_sum
    return result

def fw_lp_solve(a, n_fw=5000, warm_start=None):
    """Solve min max_k (a*b)_k s.t. b in simplex using Frank-Wolfe."""
    n = len(a)
    b = warm_start.copy() if warm_start is not None else np.ones(n) / n
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / s
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    
    for t in range(1, n_fw + 1):
        B = rfft(b, fft_size)
        c = irfft(np.conj(A) * B, fft_size)[:2*n-1]
        k_star = int(np.argmax(c))
        
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        m_lo = k_star - j_hi
        m_hi = k_star - j_lo
        
        # Find m that maximizes a[m] in [m_lo, m_hi]
        seg = a[m_lo:m_hi+1]
        m_star = m_lo + int(np.argmax(seg))
        j_star = k_star - m_star
        
        gamma = 2.0 / (t + 2.0)
        b *= (1.0 - gamma)
        b[j_star] += gamma
    
    return b

def nesterov_fw_optimize(a0, time_limit, n_fw=3000, verbose=True):
    """Nesterov-accelerated LP best-response with FW approximation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    y_old = a.copy()
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    b_warm = np.ones(n) / n  # Warm start for FW
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        # Solve LP best response using FW
        b_star = fw_lp_solve(y, n_fw=n_fw, warm_start=b_warm)
        b_warm = b_star.copy()
        
        t = iteration
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)
        
        y_old = y
        y = y_new
        a = a_new
        
        # Check scores
        score_a = internal_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = internal_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Time-averaged
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NFW iter={iteration} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping initialization
2. Nesterov-accelerated zero-sum game with fast Frank-Wolfe LP approximation
3. Adam LSE polynomial-time polish
4. Aggressive multi-coordinate peak shaving
Key innovation: FW-based LP approximation (O(n log n) per inner iter) enables
thousands of Nesterov outer iterations within the time budget.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max(autocorr) assuming sum(a)=1."""
    n = len(a)
    b = fast_autocorr(a)
    return 2.0 * n * float(b.max())


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_idx = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_idx) == 0:
        return np.full(n, target_sum / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s < 1e-30:
        return np.full(n, target_sum / n)
    result = result / s * target_sum
    return result


def fw_lp_solve(a_fixed, n_fw=5000, warm_start=None, line_search=True):
    """
    Solve min max_k (a_fixed * b)_k s.t. b in simplex using Frank-Wolfe.
    Returns approximate optimal b.
    """
    n = len(a_fixed)
    if warm_start is not None:
        b = np.maximum(warm_start, 0).copy()
        s = b.sum()
        if s < 1e-15:
            b = np.full(n, 1.0 / n)
        else:
            b = b / s
    else:
        b = np.full(n, 1.0 / n)

    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a_fixed, fft_size)

    for t in range(1, n_fw + 1):
        B = rfft(b, fft_size)
        c = irfft(np.conj(A) * B, fft_size)
        c = c[:2 * n - 1]
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        # Find j_star = argmin_j a_fixed[k_star - j] over valid j
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        m_lo = k_star - j_hi
        m_hi = k_star - j_lo
        seg = a_fixed[m_lo:m_hi + 1]
        m_star = m_lo + int(np.argmax(seg))
        j_star = k_star - m_star

        if line_search:
            # Strong Wolfe-type step size
            g_j_star = a_fixed[k_star - j_star]
            # <g, s> where g = col j_star of T^T, s = e_jstar - b
            # <g, s> = g_jstar - g^T b = a_fixed[k_star - j_star] - c[k_star]
            inner = g_j_star - peak_val  # This is negative (g_jstar <= peak)
            # ||s||^2 = ||e_jstar - b||^2 = 1 - 2*b[j_star] + ||b||^2
            b_norm_sq = float(np.dot(b, b))
            s_norm_sq = 1.0 - 2.0 * b[j_star] + b_norm_sq
            if s_norm_sq > 1e-12 and inner < -1e-12:
                gamma_ls = (-inner) / s_norm_sq
            else:
                gamma_ls = 2.0 / (t + 2.0)
            gamma = min(2.0 / (t + 2.0), max(gamma_ls, 1e-8))
        else:
            gamma = 2.0 / (t + 2.0)

        b *= (1.0 - gamma)
        b[j_star] += gamma

    return b


def nesterov_fw_optimize(a0, time_limit, n_fw=5000, verbose=True):
    """Nesterov-accelerated best-response with FW LP approximation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    y = a.copy()
    y_old = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    b_warm = np.full(n, 1.0 / n)
    iteration = 0
    stall_count = 0
    stall_threshold = 2000

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b_star = fw_lp_solve(y, n_fw=n_fw, warm_start=b_warm)
        b_warm = b_star.copy()

        t = iteration
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y_old = y

        a = a_new

        else:
            stall_count += 1

        score_y = internal_score(y)
        if score_y < best_score:
            stall_count = 0
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                stall_count = 0
                best_score = score_avg
                best_a = avg_check.copy()

        if stall_count > stall_threshold:
            break

        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NFW iter={iteration} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def spectral_shaping(a, n_iter=400, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, 1.0 / n)
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def compute_lse_grad(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*autocorr_k)) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    conv_len = 2 * n - 1 + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n]
    return grad


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(z), 1.0 / len(z))
    return e / s


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=20, lr=0.005):
    """LogSumExp-smoothed optimization with Adam."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.full(n, 1.0 / n)
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters = max(20, min(150, int(remaining * 80 / max(1, (n_stages - si)))))
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            a = softmax_fn(z)
            grad_a = compute_lse_grad(a, stage_T)
            S = float(np.dot(grad_a, a))
            grad_z = a * (grad_a - S)
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            lr_t = lr / np.sqrt(1 + 0.05 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            if it % 25 == 0:
                score = internal_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      AdamLSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Multi-coordinate peak shaving via simulated annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    temp = 0.01

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[int(np.argmax(contrib))]

            # Find destination: element not in any peak region
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = list(range(n))
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[int(rng.integers(len(dst_pool)))]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = float(rng.uniform(0.002, 0.015))
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            temp *= 0.999
            if stall > 800:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization targeting score < 1.5027.
    Phase 1: Spectral shaping
    Phase 2: Nesterov FW (primary, most time)
    Phase 3: Adam LSE polish
    Phase 4: Peak shaving
    Phase 5: Second Nesterov round
    Phase 6: Final peak shaving
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 1 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: Spectral shaping
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        for alpha_val in [0.4, 0.2, 0.1]:
            a_out = spectral_shaping(best_seq, n_iter=300, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.6f}")
    print(f"  After Phase 1: best={best_score:.6f} ({time.monotonic()-t_global_start:.0f}s)")

    # Phase 2: Nesterov FW (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov FW n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.45, 800.0)
    if t2 > 30:
        # Use fewer FW iters initially (faster), then increase
        a_out, sc_out = nesterov_fw_optimize(best_seq, t2 * 0.6, n_fw=3000)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NFW round 1: {best_score:.6f}")
        # Second round with more FW iters for accuracy
        remaining2 = time_budget - (time.monotonic() - t_global_start)
        t2b = min(remaining2 * 0.15, t2 * 0.4)
        if t2b > 20:
            a_out, sc_out = nesterov_fw_optimize(best_seq, t2b, n_fw=8000)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NFW round 2 (high-precision): {best_score:.6f}")
    print(f"  After Phase 2: best={best_score:.6f} ({time.monotonic()-t_global_start:.0f}s)")

    # Phase 3: Adam LSE
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Adam LSE n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 10:
        a_out, _ = adam_lse_optimize(best_seq, t3, T_start=30.0, T_end=0.05, n_stages=18, lr=0.004)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** AdamLSE: {best_score:.6f}")
    print(f"  After Phase 3: best={best_score:.6f} ({time.monotonic()-t_global_start:.0f}s)")

    # Phase 4: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Peak shaving n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10:
        a_out, _ = peak_shaving(best_seq, t4, seed=456)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** PeakShave: {best_score:.6f}")
    print(f"  After Phase 4: best={best_score:.6f} ({time.monotonic()-t_global_start:.0f}s)")

    # Phase 5: More Nesterov if time remains
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Nesterov FW round 3 (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 250.0)
    if t5 > 20:
        a_out, _ = nesterov_fw_optimize(best_seq, t5, n_fw=6000, verbose=False)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NFW round 3: {best_score:.6f}")
    print(f"  After Phase 5: best={best_score:.6f} ({time.monotonic()-t_global_start:.0f}s)")

    # Phase 6: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t6 = min(remaining * 0.4, 80.0)
        if t6 > 5:
            a_out, _ = peak_shaving(best_seq, t6, seed=789)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** Final PeakShave: {best_score:.6f}")

    # Final: verify with evaluate_sequence
    if best_seq is not None:
        a_final = np.maximum(best_seq, 0)
        s = a_final.sum()
        if s > 0:
            a_final = a_final / s
        # Scale to avoid overflow in evaluation
        a_final = a_final * np.sqrt(2 * len(a_final))
        a_list = [float(x) for x in a_final]
        try:
            ev = evaluate_sequence(a_list)
            if np.isfinite(ev):
                print(f"\nFINAL verify score: {ev:.6f} (n={len(a_final)})")
        except Exception as e:
            print(f"Eval error: {e}")
        print(f"FINAL internal score: {best_score:.6f} (n={len(best_seq)})")
        return [float(max(0.0, min(1000.0, x))) for x in a_final]
    else:
        print("ERROR: No valid sequence!")
        return [1.0]
