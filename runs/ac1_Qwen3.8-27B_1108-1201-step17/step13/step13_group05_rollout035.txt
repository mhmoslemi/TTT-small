"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smooth relaxation with Adam optimizer (softmax parameterization)
2. Power-sum relaxation with annealing
3. Multiple initializations (parent, sine, cosine, random, edge-weighted)
4. Multi-length scaling (1600 -> 2400 -> 3200 -> 4800)
5. Coordinate descent polish on exact objective
6. Adaptive beta annealing for LSE

import numpy as np
import time


def _fft_size(n):
    return 1 << (n - 1).bit_length()


def fft_conv_self(a):
    """Linear convolution a*a via real FFT."""
    n = len(a)
    size = _fft_size(2 * n - 1)
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def fft_xcorr(a, w):
    """Cross-correlation: result[i] = sum_m a[m]*w[m+i], for i=0..len(a)-1."""
    n = len(a)
    m = len(w)
    size = _fft_size(n + m - 1)
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    return np.fft.irfft(A * np.conj(W), size)[:n]


def lse_eval(conv, beta):
    """Compute (1/beta)*log(sum(exp(beta*conv))) stably."""
    bc = beta * conv
    mx = bc.max()
    if mx > 500:
        return mx / beta
    return (mx / beta) + np.log(np.exp(bc - mx).sum()) / beta


def lse_gradient(a, conv, beta):
    """Gradient of LSE w.r.t. a. Returns grad_a of same shape as a."""
    n = len(a)
    bc = beta * conv
    mx = bc.max()
    if mx > 500:
        w = np.zeros(len(conv))
        w[int(np.argmax(bc))] = 1.0
    else:
        w = np.exp(bc - mx)
        w /= w.sum()
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (if 0<=k-i<n)
    # = 2 * (a cross-correlate w)[i]
    return 2.0 * fft_xcorr(a, w)


def power_eval(conv, p):
    """(sum conv^p)^(1/p)."""
    vals = np.maximum(conv, 0) ** p
    s = vals.sum()
    if s < 1e-300:
        return 0.0
    return s ** (1.0 / p)


def power_gradient(a, conv, p):
    """Gradient of (sum conv^p)^(1/p) w.r.t. a."""
    n = len(a)
    cp = np.maximum(conv, 1e-300) ** (p - 1)
    norm = cp.sum()
    if norm < 1e-300:
        return np.zeros(n)
    # w[k] = (1/p) * (sum conv^p)^(1/p - 1) * p * conv[k]^(p-1) / 1
    # = conv[k]^(p-1) / (sum conv^p)^(1/p)
    # Simplified: w[k] = conv[k]^(p-1) / sum(conv^p) * p / p ... 
    # Actually: d/d(conv_k) [ (sum conv^p)^(1/p) ] = conv_k^(p-1) / (sum conv^p)^(1/p)
    # So w[k] = conv[k]^(p-1) / (sum conv^p)^(1/p)
    denom = np.sum(np.maximum(conv, 1e-300) ** p)
    if denom < 1e-300:
        return np.zeros(n)
    w = np.maximum(conv, 1e-300) ** (p - 1)
    w = w * (len(conv) ** (1.0 / p)) / (denom ** (1.0 / p))  # normalize
    return 2.0 * fft_xcorr(a, w)


def softmax_from_x(x):
    """Convert unconstrained x to probability simplex (sum=1, positive)."""
    x = x - x.max()
    a = np.exp(x)
    a = a / a.sum()
    return a


def grad_through_softmax(grad_a, a):
    """Convert gradient w.r.t. a to gradient w.r.t. x (log parameters)."""
    return a * (grad_a - np.dot(a, grad_a))


def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u * np.arange(1, n + 1) - cssv
    cond = rho_arr > 0
    if not np.any(cond):
        rho = u[-1]
    else:
        rho = np.max(rho_arr[cond])
    return np.maximum(v - rho, 0.0) * (s / max(np.maximum(v - rho, 0.0).sum(), 1e-30))


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a)]
        a_list = [max(0.0, min(1000.0, x)) for x in a_list]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def adam_softmax_optimize(a0, time_limit, verbose=True, beta_start=20.0, beta_end=500.0,
                          lr=0.01, warmup_steps=200, seed=42):
    """
    Optimize using log-sum-exp with Adam in softmax parameter space.
    a0: initial sequence (will be normalized to sum=1)
    Returns: (best_a_normalized_to_sum1, best_lse_value)
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize x from a (log-parameters)
    x = np.log(a + 1e-30)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_val = lse_eval(fft_conv_self(a), beta_end)  # use final beta for true metric
    
    t0 = time.monotonic()
    t = 0
    total_steps = max(1, int(time_limit / 0.003))  # estimate
    
    for t in range(1, total_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal beta
        frac = min(1.0, t / max(1, int(0.5 * total_steps)))
        beta = beta_start + (beta_end - beta_start) * frac
        
        # Compute a from x
        a = softmax_from_x(x)
        
        # Compute convolution and gradient
        conv = fft_conv_self(a)
        grad_a = lse_gradient(a, conv, beta)
        grad_x = grad_through_softmax(grad_a, a)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Learning rate with warmup
        lr_t = lr * min(1.0, t / warmup_steps)
        
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic evaluation with final beta
        if t % 500 == 0:
            a_check = softmax_from_x(x)
            conv_check = fft_conv_self(a_check)
            val = lse_eval(conv_check, beta_end)
            if val < best_val:
                best_val = val
                best_a = a_check.copy()
            
            if verbose and t % 5000 == 0:
                elapsed = time.monotonic() - t0
                max_conv = conv_check.max()
                obj = 2.0 * n * max_conv  # with sum=1
                print(f"      t={t} ({elapsed:.1f}s): beta={beta:.0f}, max_conv={max_conv:.8f}, obj={obj:.6f}")
    
    # Final check
    a_final = softmax_from_x(x)
    conv_final = fft_conv_self(a_final)
    val_final = lse_eval(conv_final, beta_end)
    if val_final < best_val:
        best_val = val_final
        best_a = a_final.copy()
    
    return best_a, best_val

def lp_best_response(a, M, n):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0.
    Uses scipy linprog as fallback."""
    from scipy.optimize import linprog
    # Variables: b (non-negative)
    # Constraints: conv(a,b)_k <= M for k=0..2n-2
    # conv(a,b)_k = sum_j a[j]*b[k-j] (valid indices)
    # This is a large LP; only use for small n or as verification
    # For large n, skip and return None
    if n > 2000:
        return None
    try:
        # Build constraint matrix A: A[k, j] = a[j] if 0<=k-j<n else a[k-j] if 0<=j-k<n else 0
        # Actually conv(a,b)_k = sum_j a[j] * b[k-j] where valid
        # For fixed k: sum over j where 0<=j<n and 0<=k-j<n: a[j] * b[k-j]
        # Let i = k-j, so j = k-i: sum over i where 0<=k-i<n and 0<=i<n: a[k-i] * b[i]
        # = sum_i a[k-i] * b[i] (where valid)
        # So A[k, i] = a[k-i] if 0<=k-i<n else 0
        
        A = np.zeros((2 * n - 1, n))
        for i in range(n):
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                A[k, i] = a[k - i]
        
        c = -np.ones(n)  # minimize -sum(b)
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': 10})
        if res.status == 0 and res.fun is not None:
            b = np.maximum(res.x, 0)
            if b.sum() > 1e-15:
                return b
    except Exception:
        pass
    return None

    y = a.copy()
    x = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fft_conv_self(y)
        M = conv_y.max()
        
        b = lp_best_response(y, M, n)
        if b is None:
            # Fallback: use subgradient direction
            break
        
        g = b / b.sum()
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = proj_simplex(y_new, 1.0)
        
        if t == 1:
            x_new = g.copy()
        else:
            beta_mom = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_mom * (y_new - y)
            x_new = proj_simplex(x_new, 1.0)
        
        y = y_new
        x = x_new
        
        m_y = fft_conv_self(y).max()
        if m_y < best_m:

            obj = 2.0 * n * best_m
            print(f"      LP t={t} ({elapsed:.1f}s): obj={obj:.6f}")

def coordinate_polish(a0, time_limit, verbose=False):

        conv = fft_conv_self(a)

        # For the max lag, find which pair (i, k_max-i) contributes most
        i_start = max(0, k_max - n + 1)

            i_start, i_end = 0, n - 1

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape. Normalizes to sum 1."""
    old_n = len(f)
    if old_n == new_n:
        f = np.asarray(f, dtype=np.float64).copy()
        f = np.maximum(f, 0)
        s = f.sum()
        return f / s if s > 1e-15 else np.ones(new_n) / new_n
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    n = len(a)

    mx = bc.max()

    return (mx / beta) + np.log(np.exp(bc - mx).sum()) / beta
