    # Nested optimization - one last adaptive sweep
    improved = False
    for iteration in range(5):
        # Skip if time expired
        if time.time() - start_time > time_budget:
            break

        # Limit mutation exploration to ensure no degradation
        if improved:
            candidate_improved = safe_parallel_optimization(best_attempt)
            if evaluate_adaptive_success(candidate_improved):
                best_attempt = candidate_improved
                improved = False

        times_tried = 2
        for _ in range(times_tried):
            # Explore the space with refined gradients
            candidate_variation = evaluate_sequence(best_attempt)
            if candidate_variation < best_score:
                best_score = candidate_variation
                best_attempt = candidate_variation
                times_tried -= 1
                improved = False
                break
        else:
            improved = True

    return best_attempt
