def compute_lse_gradient_fft(b, a, T):
    """Compute gradient of LSE_T(b) with respect to a, where b = a*a (autocorrelation).
    Returns grad of LSE w.r.t. a (length n).
    """
    n = len(a)
    max_b = b.max()
    if max_b < 1e-30:
        return np.zeros(n)
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    # grad_i = 2 * sum_{d} w[i+d] * a[d] for valid i+d

    A = rfft(a, fft_size)
    # We need: g[i] = sum_d w[i+d] * a[d] for i=0..n-1
    # This is the cross-correlation of w and a, shifted
    # corr[i] = sum_d w[i+d] * a[d]
    # = (a * w_flipped)[i + (2n-2) - (n-1)] ... let me use a different approach

    # We want g[i] = sum_d a[d] * w[i+d]
    # Set 2n-2-m = i, so m = 2n-2-i
    # (a * w_flip)[2n-2-i] = sum_d a[d] * w[i+d] = g[i]
    w_flip = w[::-1]
    CONN = irfft(A * W, fft_size)
    # Wait, that gives a * w, not a * w_flip. Let me redo.
    # We want g[i] = sum_d a[d] * w[i+d]
    # Let h = w reversed: h[j] = w[2n-2-j]
    # (a * h)[m] = sum_j a[j] h[m-j] = sum_j a[j] w[2n-2-(m-j)] = sum_j a[j] w[2n-2-m+j]
    # For m = 2n-2-i: (a*h)[2n-2-i] = sum_j a[j] w[i+j] = g[i]
    H = rfft(w_flip, fft_size)

def lse_objective_and_grad(v, n, T):
    """Compute LSE objective and gradient w.r.t. v (softmax parameters).
    v has length n, with v[0] effectively fixed (we optimize v[1:]).
    """
    # Softmax to get a
    v_safe = v.copy()
    v_safe -= v_safe.max()  # numerical stability
    exp_v = np.exp(v_safe)
    p = exp_v / exp_v.sum()
    a = p * np.sqrt(2.0 * n)
    # Compute autocorrelation
    b = fast_autocorr(a)
    # Compute LSE
    max_b = b.max()
    if max_b < 1e-30:
        return 0.0, np.zeros(n)
    log_sum = max_b / T + np.log(np.sum(np.exp((b - max_b) / T)))
    obj = T * log_sum
    # Gradient w.r.t. a
    grad_a = compute_lse_gradient_fft(b, a, T)
    # Gradient w.r.t. v via chain rule
    # a_j = sqrt(2n) * p_j where p = softmax(v)
    # dp_j/dv_i = p_j * (delta_ji - p_i)
    # dL/dv_i = sum_j dL/da_j * da_j/dv_i
    # da_j/dv_i = sqrt(2n) * p_j * (delta_ji - p_i) = a_j * (delta_ji - p_i)
    # dL/dv_i = a_i * grad_a_i - p_i * sum_j a_j * grad_a_j
    # sum_j a_j * grad_a_j:
    inner = np.sum(a * grad_a)
    grad_v = a * grad_a - p * inner
    return obj, grad_v

                # Convert back to a-space
                v_safe = v - v.max()
                p = np.exp(v_safe) / np.exp(v_safe).sum()
                a_new = p * norm
                score = fast_autocorr(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    if verbose:
                        print(f"      LSE-LBFGS T={T:.4f}: score={score:.8f}")
        except Exception as e:
            if verbose:
                print(f"      LSE-LBFGS error at T={T:.4f}: {e}")

        # Also check the exact (non-smooth) score
        v_safe = v - v.max()
        p = np.exp(v_safe) / np.exp(v_safe).sum()
        a_curr = p * norm
        exact_score = fast_autocorr(a_curr).max()
        if exact_score < best_score:
            best_score = exact_score
            best_a = a_curr.copy()

    if verbose:
        print(f"    LSE-LBFGS done: best={best_score:.8f}, time={time.monotonic()-t0:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_best_response_reduced(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    for _ in range(3):
        circ = T @ b_val
        tol = 1e-8 * M
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= size - 5 or n_active < 20:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-6):
            b_val = b_new
        else:
            break
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_restarts=3):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    total_restarts = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        if t < 50:
            b = lp_best_response(y, M_curr, solver)
        else:
            b = lp_best_response_reduced(y, M_curr, solver)

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

        if stall > 300 and total_restarts < max_restarts:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            stall = 0
            total_restarts += 1
            if verbose:
                print(f"    [Restart at iter {n_iters}, total={total_restarts}]")
            if total_restarts >= max_restarts:
                stall = 0  # stop restarting but continue

    return best_a, best_score


def asymmetric_init_exponential(n, rate, seed=42):
    """Create an asymmetric exponential initialization."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * j)
    # Add slight noise for asymmetry
    rng = np.random.default_rng(seed)
    a += 0.01 * a.mean() * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * np.sqrt(2.0 * n) / s
    return a


def asymmetric_init_powerlaw(n, power, seed=42):
    """Create an asymmetric power-law initialization."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** (-power)
    rng = np.random.default_rng(seed)
    a += 0.01 * a.mean() * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * np.sqrt(2.0 * n) / s
    return a


def asymmetric_init_gaussian(n, mu_frac, sigma_frac, seed=42):
    """Create an asymmetric Gaussian initialization (off-center)."""
    j = np.arange(n, dtype=np.float64)
    mu = mu_frac * n
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((j - mu) / sigma) ** 2)
    rng = np.random.default_rng(seed)
    a += 0.005 * a.mean() * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * np.sqrt(2.0 * n) / s
    return a


def asymmetric_init_bimodal(n, seed=42):
    """Create a bimodal asymmetric initialization."""
    j = np.arange(n, dtype=np.float64)
    a1 = np.exp(-0.5 * ((j - 0.25 * n) / (0.1 * n)) ** 2)
    a2 = 0.5 * np.exp(-0.5 * ((j - 0.75 * n) / (0.08 * n)) ** 2)
    a = a1 + a2
    rng = np.random.default_rng(seed)
    a += 0.005 * a.mean() * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    a = a * np.sqrt(2.0 * n) / s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def greedy_peak_shave(a0, time_limit, seed=42, verbose=True):
    """Aggressive greedy peak shaving with predictive evaluation."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        peak_k = np.argmax(b)

        # Find contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-18:
            break

        # Try reducing the top contributor
        src = idx[np.argmax(contrib)]
        partner = peak_k - src

        # Find good destination: minimize increase in other peaks
        # Score each candidate destination by predicted max increase
        # Use a quick heuristic: prefer positions with low value
        candidates = np.argsort(a)[:min(20, n)]
        best_trial = None
        best_trial_score = peak_val

        for dst in candidates:
            if dst == src:
                continue
            frac = 0.1
            amt = a[src] * frac
            if amt < 1e-18:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < -1e-12:
                continue
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial * norm / st
            score = fast_autocorr(a_trial).max()
            if score < best_trial_score:
                best_trial_score = score
                best_trial = a_trial

        if best_trial is not None and best_trial_score < best_score:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            n_improve += 1
            n_stall = 0
        elif best_trial is not None and best_trial_score < peak_val * 0.99:
            # Accept if it reduces the specific peak even if global max doesn't improve
            a = best_trial
            n_stall += 1
        else:
            n_stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial * norm / st
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1

            if n_stall > 100:
                # Larger perturbation
                n_pert = max(3, n // 40)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.2, 0.5)
                    dst = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial * norm / st
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.02:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                n_stall = 0

    if verbose:
        print(f"    PeakShave: best={best_score:.8f}, improvements={n_improve}, time={time.monotonic()-t0:.1f}s")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        best_seq = asymmetric_init_exponential(n_base, 0.002, seed=42)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: Multi-start LSE-LBFGS with asymmetric initializations
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-start LSE-LBFGS (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        inits = [
            ("exp_fast", asymmetric_init_exponential(n_base, 0.005, seed=1)),

            ("gauss_right", asymmetric_init_gaussian(n_base, 0.7, 0.15, seed=7)),
            ("bimodal", asymmetric_init_bimodal(n_base, seed=8)),
        ]
        # Also include parent as one start
        inits.append(("parent", best_seq.copy()))

        t_per_init = t1 / len(inits)
        for label, a_init in inits:
            remaining_check = time_budget - (time.monotonic() - t_global)
            if remaining_check < 30:
                break
            t_here = min(t_per_init, remaining_check - 10)
            if t_here < 10:
                break
            print(f"  LSE-LBFGS [{label}]: t={t_here:.0f}s")
            a_out, sc = lse_lbfgs_optimize(a_init, t_here, verbose=False)
            if a_out is not None:
                a_out = a_out / a_out.sum()
                sc_eval = evaluate_safe(a_out)
                if np.isfinite(sc_eval) and sc_eval < best_score:

                    best_seq = a_out.copy()

                elif np.isfinite(sc_eval):
                    print(f"    -> score={sc_eval:.6f}")

    # Phase 2: LP Nesterov on best from Phase 1

        do_try_update(a_out, f"pd_n{n_cur}")

    # Phase 4: Try longer length n+200
