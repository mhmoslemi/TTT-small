```python
def spectral_flatten(a, iterations=100):
    """Try to flatten the power spectrum of a."""
    n = len(a)
    a = a.copy()
    N = 1 << (2*n).bit_length()
    for _ in range(iterations):
        A = np.fft.fft(a, N)
        P = np.abs(A)**2
        # Target: flat spectrum
        P_target = np.ones(N) * P.mean()
        # Update: a_new = ifft(sqrt(P_target) * A/|A|)
        phase = A / np.abs(A + 1e-30)
        a_new = np.fft.ifft(np.sqrt(P_target) * phase).real[:n]
        # Project onto simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            break
        if np.allclose(a_new, a):
            break
        a = a_new
    return a
```

```python
def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_obj_grad(z, T):
    """Full LSE objective and gradient in z-space."""
    a = softmax_fn(z)
    n = len(a)
    c = fast_autocorr(a)  # length 2n-1
    
    # LSE value
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()  # weights, length 2n-1

    # grad_a[i] = (2/T) * sum_{m=0}^{n-1} a[m] * w[m+i]
    grad_a = np.zeros(n)
    for i in range(n):
        # sum_m a[m] * w[m+i] where 0 <= m+i <= 2n-2
        js = max(0, i)  # i is always >= 0
        je = min(n, 2*n-1 - i)  # m+i <= 2n-2 => m <= 2n-2-i
        if js < je:

    # Gradient in z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return lse_val, grad_z
```

This should be efficient. Each iteration is O(n log n) for the FFT computations.

For n=1600, each iteration should take maybe 0.5-1ms. So in 1000 seconds, we can do about 1-2 million iterations. That should be more than enough.

Let me also think about the temperature schedule. The LSE approximation to max is:
  LSE_T(c) ≈ max(c) when T -> infinity
  LSE_T(c) ≈ mean(c) + std(c) * sqrt(2 log(K) / pi) when T is small (Gaussian approximation)

For a good approximation, we want T * (max(c) - second_max(c)) >> 1. If the gap between the top two peaks is about 0.001 (relative), then T >> 1000 for a good approximation.

But very large T makes the optimization hard (the LSE becomes non-smooth). So a good strategy is:
- Start with T ~ 10-50 (smooth, easy to optimize)
- Gradually decrease to T ~ 100-1000 (sharper, more accurate)
- Finally, do a few iterations with the exact max (subgradient)

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full LogSumExp (LSE) objective with Adam in softmax z-space
2. Multi-stage temperature annealing with warm restarts
3. Subgradient-based peak shaving for final refinement
4. Multi-length exploration with interpolation
5. Spectral domain post-processing
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_weights(c, T):
    """Compute softmax weights for LSE: w_k = exp(T*c_k) / sum(exp(T*c))."""
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    return w


def compute_grad_a(a, w, T):
    """Compute gradient of LSE w.r.t. a: (2/T) * cross_corr(a, w)."""
    n = len(a)
    # cross[i] = sum_m a[m] * w[m+i] via FFT
    rev_a = a[::-1]
    N = 1 << (n + len(w) - 1).bit_length()
    cross = irfft(rfft(rev_a, N) * rfft(w, N), N)
    return (2.0 / T) * cross[n - 1: 2 * n - 1]


def lse_adam_optimize(a0, time_limit, T_values, lr_base=0.01, 
                      b1=0.9, b2=0.999, eps=1e-8,
                      check_interval=10, stall_reset=2000):
    """
    Optimize LSE objective in z-space using Adam.
    a0: initial point (normalized, non-negative)
    T_values: list of temperatures to try (high to low)
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    print(f"    LSE-Adam: n={n}, starting score={best_score:.8f}")
    
    for T_idx, T in enumerate(T_values):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Adaptive learning rate
        lr = lr_base * np.sqrt(1.0 / (1.0 + 0.1 * T_idx))
        lr = min(lr, 0.05)
        
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        stall = 0
        local_best = best_score
        
        remaining = time_limit - (time.monotonic() - t0)
        max_iters = max(100, int(remaining * 200))  # aim for ~200 iters/sec
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            a = softmax_fn(z)
            c = fast_autocorr(a)
            w = lse_weights(c, T)
            grad_a = compute_grad_a(a, w, T)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Adam update
            t_adam += 1
            m = b1 * m + (1 - b1) * grad_z
            v = b2 * v + (1 - b2) * grad_z ** 2
            m_hat = m / (1 - b1 ** t_adam)
            v_hat = v / (1 - b2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Periodic check
            if (it + 1) % check_interval == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < local_best:
                    local_best = score
                    stall = 0
                    if score < best_score:
                        best_score = score
                        best_a = a_check.copy()
                else:
                    stall += 1
                
                # Warm restart
                if stall > stall_reset // check_interval:
                    z = np.log(np.maximum(best_a, 1e-30))
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t_adam = 0
                    stall = 0
                    local_best = best_score
        
        # End of stage
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    LSE-Adam stage {T_idx+1}/{len(T_values)} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def subgradient_peak_shave(a0, time_limit, seed=42, max_rounds=50000):
    """
    Subgradient-based peak shaving: at each step, push down the current max peak
    using a subgradient step, then project onto the simplex.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Start with a large step, decrease over time
    step_start = 0.05
    step_end = 0.001
    decay = (step_end / step_start) ** (1.0 / max_rounds)
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        b = fast_autocorr(a)
        bmax = b.max()
        
        # Get subgradient of max: uniform over all argmax positions
        argmax_mask = (b >= bmax - 1e-12)
        k_indices = np.where(argmax_mask)[0]
        n_peak = len(k_indices)
        
        # Subgradient w.r.t. a: for each peak k, grad = 2*a[k-i] (by symmetry)
        # Average over all peak positions
        subgrad = np.zeros(n)
        for k in k_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            subgrad[idx] += 2.0 * a[k - idx]
        subgrad /= n_peak
        
        # Step size
        step = step_start * (decay ** rnd)
        # Normalize subgradient
        sg_norm = np.sqrt(np.dot(subgrad, subgrad))
        if sg_norm < 1e-15:
            break
        step = min(step, 0.1)  # cap step size
        
        # Projected gradient step
        a_new = a - step * subgrad / sg_norm
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = project_simplex(a_new)
        
        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score - 1e-12:
            a = a_new
            best_score = score_new
            best_a = a_new.copy()
        else:
            # Try random perturbation occasionally
            if rnd % 100 == 0:
                a = best_a.copy()
                eps_vec = rng.normal(0, 0.005, n)
                a_pert = a + eps_vec
                a_pert = project_simplex(a_pert)
                score_pert = 2.0 * n * fast_autocorr(a_pert).max()
                if score_pert < best_score:
                    best_score = score_pert
                    best_a = a_pert.copy()
                    a = a_pert
    
    return best_a, best_score


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def peak_shaving_heuristic(a0, time_limit, seed=777):
    """
    Heuristic peak shaving: try to move mass from the most contributing
    pair at the top peak to a less dangerous location.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 1:
        b = fast_autocorr(a)
        bmax = b.max()
        
        # Find top peaks
        top_k = min(20, len(b))
        peak_indices = np.argsort(b)[::-1][:top_k]
        
        improved = False
        for k in peak_indices[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find safe destinations (positions not in any top peak's danger zone)
            danger = np.zeros(n, dtype=bool)
            for pk in peak_indices[:10]:
                dj = max(0, pk - n + 1)
                de = min(n - 1, pk)
                danger[dj:de + 1] = True
            
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            
            # Try moving mass to the lowest valued safe position
            safe_sorted = safe[np.argsort(a[safe])]
            dst = safe_sorted[0] if len(safe_sorted) > 0 else None
            if dst is None:
                continue
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-12:
                    continue
                a_trial = a_trial / st
                new_max = fast_autocorr(a_trial).max()
                if new_max < bmax * (1 - 1e-10):
                    a = a_trial
                    new_score = 2.0 * n * new_max
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Random perturbation
                n_pert = rng.integers(2, 6)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_refine(a0, n_iter=100):
    """
    Spectral domain refinement: try to flatten the periodogram to reduce
    peak autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    N = 1 << (2 * n).bit_length()
    alpha = 0.3  # blending parameter
    
    for _ in range(n_iter):
        A = rfft(a, N)
        P = np.abs(A) ** 2
        # Try to flatten: compute target as geometric mean
        log_P = np.log(P + 1e-30)
        P_target = np.exp(np.mean(log_P)) * np.ones(N)
        # Blend
        P_new = (1 - alpha) * P + alpha * P_target
        # Take square root (maintain phase)
        mag = np.sqrt(np.maximum(P_new, 0))
        phase = A / (np.abs(A) + 1e-30)
        A_new = mag * phase
        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        score_new = 2.0 * n * fast_autocorr(a_new).max()
        score_cur = 2.0 * n * fast_autocorr(a).max()
        if score_new < score_cur:
            a = a_new
        else:
            alpha *= 0.5
            if alpha < 0.01:
                break
    
    return a
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full LogSumExp (LSE) objective with Adam in softmax z-space
2. Multi-stage temperature annealing with warm restarts
3. Subgradient-based peak shaving for final refinement
4. Heuristic peak shaving with danger-zone awareness
5. Spectral domain post-processing
6. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_weights(c, T):
    """Compute softmax weights for LSE: w_k = exp(T*c_k) / sum(exp(T*c))."""
    Tc = T * np.asarray(c, dtype=np.float64)
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    return w


def compute_grad_a(a, w, T):
    """Compute gradient of LSE w.r.t. a: (2/T) * cross_corr(a, w).
    cross[i] = sum_m a[m] * w[m+i] for valid indices.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    rev_a = a[::-1]
    N = 1 << (n + len(w) - 1).bit_length()
    cross = irfft(rfft(rev_a, N) * rfft(w, N), N)
    return (2.0 / T) * cross[n - 1: 2 * n - 1]


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def lse_adam_optimize(a0, time_limit, T_values, lr_base=0.01,
                      b1=0.9, b2=0.999, eps=1e-8,
                      check_interval=5, stall_reset=2000):
    """
    Optimize LSE objective in z-space using Adam.
    a0: initial point (will be normalized)
    T_values: list of temperatures to try (typically high to low)
    Returns (best_a, best_score) where score = 2*n*max(autocorr).
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    print(f"    LSE-Adam: n={n}, starting score={best_score:.8f}")

    for T_idx, T in enumerate(T_values):
        if time.monotonic() - t0 > time_limit - 3:
            break

        # Adaptive learning rate: smaller as temperature decreases
        lr = lr_base * (1.0 / (1.0 + 0.05 * T_idx))
        lr = min(lr, 0.05)

        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        stall = 0
        local_best = best_score

        max_iters = 1000000
        iters_done = 0

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit - 1:
                break

            a = softmax_fn(z)
            c = fast_autocorr(a)
            w = lse_weights(c, T)
            grad_a = compute_grad_a(a, w, T)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            # Adam update
            t_adam += 1
            m = b1 * m + (1.0 - b1) * grad_z
            v = b2 * v + (1.0 - b2) * grad_z ** 2
            m_hat = m / (1.0 - b1 ** t_adam)
            v_hat = v / (1.0 - b2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            iters_done += 1

            # Periodic check
            if (it + 1) % check_interval == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < local_best:
                    local_best = score
                    stall = 0
                    if score < best_score:
                        best_score = score
                        best_a = a_check.copy()
                else:
                    stall += 1

                # Warm restart if stuck
                if stall > stall_reset // check_interval:
                    z = np.log(np.maximum(best_a, 1e-30))
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t_adam = 0
                    stall = 0
                    local_best = best_score

        # End of stage: check final point
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        elapsed = time.monotonic() - t0
        print(f"    LSE-Adam stage {T_idx+1}/{len(T_values)} (T={T:.2f}, {iters_done} iters, {elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_peak_shave(a0, time_limit, seed=42, max_rounds=100000):
    """
    Subgradient-based peak shaving: push down the current max peak
    using a subgradient step, then project onto the simplex.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    step_start = 0.03
    step_end = 0.0005
    decay = (step_end / step_start) ** (1.0 / max(1, max_rounds))

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit - 1:
            break

        b = fast_autocorr(a)
        bmax = b.max()

        # Get subgradient of max: average over all argmax positions
        tol = max(1e-12, bmax * 1e-9)
        argmax_mask = (b >= bmax - tol)
        k_indices = np.where(argmax_mask)[0]
        n_peak = len(k_indices)

        # Subgradient w.r.t. a
        subgrad = np.zeros(n)
        for k in k_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            subgrad[idx] += 2.0 * a[k - idx]
        subgrad /= n_peak

        # Step size with decay
        step = step_start * (decay ** rnd)
        step = min(step, 0.05)
        sg_norm = np.sqrt(np.dot(subgrad, subgrad))
        if sg_norm < 1e-15:
            break

        # Projected gradient step
        a_new = a - step * subgrad / sg_norm
        a_new = project_simplex(a_new)

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score - 1e-12:
            a = a_new
            best_score = score_new
            best_a = a_new.copy()
        else:
            # Random restart occasionally
            if rnd % 200 == 0:
                a = best_a.copy()
                noise = rng.normal(0, 0.003, n)
                a_pert = project_simplex(a + noise)
                score_pert = 2.0 * n * fast_autocorr(a_pert).max()
                if score_pert < best_score:
                    best_score = score_pert
                    best_a = a_pert.copy()
                    a = a_pert

    return best_a, best_score


def peak_shaving_heuristic(a0, time_limit, seed=777):

        top_k = min(15, len(b))
        peak_indices = np.argsort(b)[::-1][:top_k]

        improved = False
        for k in peak_indices[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            src = idx[np.argmax(contrib)]

            # Danger zone: positions involved in any top peak
            danger = np.zeros(n, dtype=bool)
            for pk in peak_indices[:8]:
                dj = max(0, pk - n + 1)
                de = min(n - 1, pk)
                danger[dj:de + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            if len(safe) == 0:
                continue

            # Try least-valued safe positions
            safe_sorted = safe[np.argsort(a[safe])]
            for dst in safe_sorted[:3]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1.0 - 1e-10):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 30 == 0:

def spectral_refine(a0, n_iter=100, alpha_init=0.3):
    """
    Spectral domain refinement: flatten the periodogram to reduce peak
    autocorrelation. Uses phase-preserving magnitude equalization.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    N = 1 << (2 * n).bit_length()
    alpha = alpha_init

    for it in range(n_iter):
        A = rfft(a, N)
        P = np.abs(A) ** 2
        # Target: flat spectrum
        log_P = np.log(P + 1e-30)
        P_target = np.exp(np.mean(log_P))
        # Blend
        P_new = (1.0 - alpha) * P + alpha * P_target
        # Reconstruct with original phase
        mag = np.sqrt(np.maximum(P_new, 0))
        phase = A / (np.abs(A) + 1e-30)
        A_new = mag * phase
        a_new = irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        score_cur = 2.0 * n * fast_autocorr(a).max()
        if score_new < score_cur:
            a = a_new
        else:
            alpha *= 0.7
            if alpha < 0.01:
                break

    return a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        x = np.linspace(0, 2 * np.pi, n_init)

        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Initialized: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)

    # Helper: update best if improved (inline to avoid closure)
    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LSE Adam (main optimization, generous time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 700.0)
    print(f"\n=== Phase 1: LSE Adam n={n_base} (budget {t1:.0f}s) ===")
    if t1 > 60:
        T_sched1 = [60.0, 40.0, 30.0, 20.0, 15.0, 12.0, 10.0, 8.0, 6.0,
                    5.0, 4.0, 3.0, 2.5, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6, 0.5]
        a_out, _ = lse_adam_optimize(best_seq, t1, T_sched1,
                                      lr_base=0.015, b1=0.9, b2=0.999,
                                      check_interval=5, stall_reset=3000)
        _try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: LSE Adam v2 (different hyperparams) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.18, 300.0)
    print(f"\n=== Phase 2: LSE Adam v2 n={n_base} (budget {t2:.0f}s) ===")
    if t2 > 30:
        T_sched2 = [15.0, 10.0, 8.0, 6.0, 5.0, 4.0, 3.0, 2.5, 2.0, 1.5,
                    1.2, 1.0, 0.8, 0.6, 0.5, 0.4, 0.3]
        a_out, _ = lse_adam_optimize(best_seq, t2, T_sched2,
                                      lr_base=0.005, b1=0.85, b2=0.999,
                                      check_interval=3, stall_reset=2000)
        _try_update(a_out, f"lse2_n{n_base}")

    # === Phase 3: Subgradient peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 200.0)
    print(f"\n=== Phase 3: Subgradient peak shaving n={n_base} (budget {t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = subgradient_peak_shave(best_seq, t3, seed=42)
        _try_update(a_out, f"subgrad_n{n_base}")

    # === Phase 4: Heuristic peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Heuristic peak shaving n={n_base} (budget {t4:.0f}s) ===")
    if t4 > 15:
        a_out, _ = peak_shaving_heuristic(best_seq, t4, seed=777)
        _try_update(a_out, f"heuristic_n{n_base}")

    # === Phase 5: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 5: Mirror descent n={n_base} (budget {t5:.0f}s) ===")
    if t5 > 15:
        a_out, _ = mirror_descent_refine(best_seq, t5, eta=0.008, seed=555)
        _try_update(a_out, f"mirror_n{n_base}")

    # === Phase 6: Spectral refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Spectral refinement n={n_base} ===")
    if remaining > 20:
        a_out = spectral_refine(best_seq, n_iter=150, alpha_init=0.2)
        _try_update(a_out, f"spectral_n{n_base}")

    # === Phase 7: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_long = n_base + 150
        print(f"\n=== Phase 7: LSE Adam n={n_long} ===")
        t7 = min(remaining * 0.20, 250.0)
        if t7 > 40:
            init_long = interpolate_sequence(best_seq, n_long)
            T_sched3 = [40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.5, 1.0, 0.7, 0.5, 0.3]
            a_out, _ = lse_adam_optimize(init_long, t7, T_sched3,
                                          lr_base=0.01, check_interval=5, stall_reset=2500)
            _try_update(a_out, f"lse_n{n_long}")

            # Quick shave on longer
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t7b = min(remaining * 0.08, 80.0)
                if t7b > 15:
                    a_out, _ = peak_shaving_heuristic(best_seq, t7b, seed=1001)
                    _try_update(a_out, f"heuristic_n{n_long}")

    # === Phase 8: Final LSE polish (very low temp, high precision) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"\n=== Phase 8: Final LSE polish ===")
        t8 = min(remaining * 0.25, 150.0)
        if t8 > 15:
            T_final = [3.0, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]
            a_out, _ = lse_adam_optimize(best_seq, t8, T_final,
                                          lr_base=0.003, b1=0.9, b2=0.999,
                                          check_interval=3, stall_reset=1500)
            _try_update(a_out, "final_lse")

    # === Phase 9: Final subgradient + heuristic shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 9: Final shaving ===")
        t9 = min(remaining * 0.50, 100.0)
        if t9 > 10:
            a_out, _ = subgradient_peak_shave(best_seq, t9 * 0.5, seed=3000)
            _try_update(a_out, "final_subgrad")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 10:
                a_out, _ = peak_shaving_heuristic(best_seq, min(remaining, 60.0), seed=4000)
                _try_update(a_out, "final_shave")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
