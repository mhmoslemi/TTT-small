```python
        self.size = 2 * n - 1

        self.cols = np.zeros(n * n, dtype=np.int32)
```

3. The "anti-peak" coordinates are those with low g[i] (they contribute little to the peak).
4. Move mass from high-g[i] coordinates to low-g[i] coordinates.
5. Use line search to find the optimal step size.

This is essentially what the LP does, but in a more targeted way. The LP finds the optimal redistribution simultaneously for all peaks, which is better, but the LP might be slow.

**Combined approach**:
- Use LP for the "global" direction (best response)
- Use targeted coordinate descent for the "local" refinements
- Alternate between them

Let me also consider: what if I use a different normalization? The objective is `2n * max(b) / S^2` where S = sum(a). If I set S = sqrt(2n), then the objective is just max(b). The LP constraint is conv(a, b) <= max(b_current). 

Actually, here's another thought: what if I formulate the problem differently?

Minimize t subject to:
- conv(a, a)[k] <= t for all k
- sum(a) = 1
- a >= 0

This is a quadratically constrained program (QCP). It's non-convex, but we can solve it with:
- Semidefinite relaxation (SDR): relax a*a^T (outer product) to a PSD matrix
- Or use the alternating directions method (which is what the LP best-response does)

The SDR approach might give a better bound but is much more expensive to solve. Let's stick with the AD approach.

**Final algorithm plan:**

```

Phase 2 (15% time): LSE smooth optimization (scipy)

Phase 4 (20% time): Aggressive coordinate descent with line search
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration (primary global optimizer)
2. LSE smooth surrogate + L-BFGS-B (complementary trajectory)
3. Dual-guided coordinate descent with line search (new: uses LP dual for targeting)
4. Aggressive multi-peak coordinate descent with simulated annealing
5. Multi-length exploration
Key differences from previous: precomputed sparse matrix structure, dual-guided targeting,
line search in coordinate descent, simulated annealing for escape, and better time allocation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize
from scipy import sparse

# Global cache for solver and precomputed structures
_global_solver = None
_conv_cache = {}


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Fast aperiodic autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the sequence using the global evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def get_conv_matrix(n, a):
    """Build sparse convolution matrix T where T @ b = conv(a, b).
    Uses precomputed indices for efficiency."""
    global _conv_cache
    if n not in _conv_cache:
        size = 2 * n - 1
        rows = np.zeros(n * n, dtype=np.int32)
        cols = np.zeros(n * n, dtype=np.int32)
        for i in range(n):
            start = i * n
            rows[start:start + n] = np.arange(i, i + n)
            cols[start:start + n] = i
        _conv_cache[n] = (rows, cols, size)
    rows, cols, size = _conv_cache[n]
    vals = np.tile(a, n)
    T = sparse.csc_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def lp_best_response_with_dual(a, M, solver):
    """Solve LP best-response and return (b, dual_values)."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = get_conv_matrix(n, a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    dual_val = None
    try:
        dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
        if dual_val is not None:
            dual_val = np.maximum(dual_val, 0.0)
    except Exception:
        pass
    return b_val, dual_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve_time = t0
    stall_count = 0
    best_dual = None

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b, dual = lp_best_response_with_dual(y, M_curr, solver)
        if b is None:
            continue
        if dual is not None:
            best_dual = dual

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        n_iters += 1

        # Adaptive step size
        ratio = M_curr / best_score if best_score > 0 else 1.0
        alpha = 2.0 / (t + 2.0)
        alpha = min(alpha * min(max(ratio, 0.5), 2.0), 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve_time = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve_time = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve_time = time.monotonic()

        # Stall detection
        if time.monotonic() - last_improve_time > 45.0:
            stall_count += 1
            if stall_count > 15:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, best_dual


def lse_objective_and_grad(x, n, T_temp):
    """Compute LSE objective and gradient."""
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T_temp

    # Gradient of LSE w.r.t. a: need d(b[k])/da[i] weighted by w[k]
    # b[k] = sum_j a[j]*a[k-j], so db[k]/da[i] = a[k-i]*[valid] + a[i-k]*[valid]
    # Gradient of LSE = sum_k w[k] * db[k]/da[i] / T_temp
    # Actually: LSE = max_b + log(sum(exp(T*(b-max_b))))/T
    # dLSE/da[i] = sum_k w[k] * T * db[k]/da[i] / T = sum_k w[k] * db[k]/da[i]
    # Wait: d/dx log(sum(exp(T*(x-max)))) = T * exp(T*(x-max)) / sum(exp(T*(x-max))) = T*w
    # So dLSE/da[i] = sum_k T*w[k] * db[k]/da[i] / T = sum_k w[k] * db[k]/da[i]
    
    # Compute sum_k w[k] * db[k]/da[i] for all i
    # db[k]/da[i] = a[k-i] + a[i-k] (with bounds)
    # sum_k w[k] * (a[k-i] + a[i-k])
    # = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
    # First term: for fixed i, sum over k where 0<=k-i<n, i.e., k in [i, i+n-1]
    #   = sum_{j=0}^{n-1} w[i+j] * a[j] (with bounds on i+j < 2n-1)
    # Second term: for fixed i, sum over k where 0<=i-k<n, i.e., k in [max(0,i-n+1), min(i, 2n-2)]
    #   = sum_{j=0}^{n-1} w[i-j] * a[j] (with bounds)
    
    # This is a cross-correlation of w and a!
    # sum_k w[k]*a[k-i] = (w * a)[i] (sliding dot product)
    # sum_k w[k]*a[i-k] = (w ~ a)[i] (cross-correlation)
    
    # Use FFT for efficiency
    size_full = 2 * n - 1
    fft_size = 1 << (size_full - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:size_full] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    W_fft = rfft(w_padded, fft_size)
    A_fft = rfft(a_padded, fft_size)
    
    # Cross-correlation: correlation[w, a][i] = sum_k w[k+i] * a[k]
    # And: sum_k w[k] * a[k-i] = correlation at lag i
    # sum_k w[k] * a[i-k] = this is the "reverse correlation"
    
    # Let me use a different approach: direct computation via FFT
    # sum_k w[k] * db[k]/da[i] where db[k]/da[i] = a[k-i]*[0<=k-i<n] + a[i-k]*[0<=i-k<n]
    # = sum_{j=0}^{n-1} w[i+j]*a[j] + sum_{j=0}^{n-1} w[i-j]*a[j] (with w out-of-range = 0)
    
    # First sum: (w * a) in a sliding window sense = cross-correlation
    # conv(w, a)[i] style but with the specific indexing
    
    # Let me just do it with the full cross-correlation via FFT
    # c1[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for i=0,...,n-1 (w out of [0,2n-2] is 0)
    # This is: c1[i] = sum_j w[i+j]*a[j] = (flip(w) conv a) at appropriate index
    # Or: c1 = corr(w, a) at lags 0 to n-1
    
    # c2[i] = sum_{j=0}^{n-1} w[i-j] * a[j] for i=0,...,n-1
    # This is: c2[i] = sum_j w[i-j]*a[j] = (w conv a) at appropriate index (cross-correlation)
    
    # Using FFT:
    # c2[i] = sum_j w[i-j]*a[j] = (w * a)[i] where * is cross-correlation
    # In FFT: c2 = ifft(conj(FFT(w)) * FFT(a))
    # But we need to be careful with the aperiodic lengths.
    
    # Simpler: just compute directly for n up to ~2000
    # c1[i] = dot(w[i:i+n], a) but w has length 2n-1, so w[i:i+n] is valid for i=0,...,n-1
    # c2[i] = dot(reversed w[0:i+n-1 reversed], a[0:n-1-?]) ... this is getting complicated
    
    # Let me just use the direct O(n^2) approach since n is at most ~2000 and this is called
    # only a limited number of times in the LSE phase.
    
    grad_b = np.zeros(n)
    for i in range(n):
        # db[k]/da[i] = a[k-i] + a[i-k] for valid k
        # sum_k w[k] * db[k]/da[i]
        # = sum_k w[k] * a[k-i] (where 0<=k-i<n) + sum_k w[k] * a[i-k] (where 0<=i-k<n)
        # First: k ranges from i to i+n-1 (clamped to [0, 2n-2])
        k_start1 = i
        k_end1 = min(i + n, size_full)
        if k_end1 > k_start1:
            grad_b[i] += np.dot(w[k_start1:k_end1], a[:k_end1 - k_start1])
        # Second: k ranges from max(0, i-n+1) to i
        k_start2 = max(0, i - n + 1)
        k_end2 = i + 1
        if k_end2 > k_start2:
            grad_b[i] += np.dot(w[k_start2:k_end2], a[i - k_start2:i - k_start2 + 1][::-1])
    
    # Actually this is getting too complex and slow. Let me use a simpler but correct approach.
    # The gradient of LSE w.r.t. a[i] is:
    # sum_k w[k] * (a[k-i] + a[i-k]) where indices are valid
    # This equals: sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
    # First term with j=k-i: sum_j w[i+j]*a[j] for j=0..n-1 (w[i+j] = 0 if i+j >= 2n-1)
    # Second term with j=i-k: sum_j w[i-j]*a[j] for j=0..n-1 (w[i-j] = 0 if i-j < 0 or i-j >= 2n-1)
    
    # I'll compute these using np.correlate or manual sliding dot products
    # For efficiency with n~1600, O(n^2) = 2.5M ops per gradient evaluation is fine
    # since we only do this a few hundred times.
    
    obj_val = 2.0 * n * lse / (s * s)
    # Full gradient of objective:
    # d(obj)/da[i] = 2n/(s^2) * grad_b[i] - 2*obj_val*a[i]/s
    grad_obj = 2.0 * n * grad_b / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad_obj.ravel()
```

```python
def lse_grad_direct(a, w):
    """Compute sum_k w[k] * db[k]/da[i] for all i, using O(n^2) direct method."""
    n = len(a)
    size = 2 * n - 1
    grad = np.zeros(n)
    for i in range(n):
        # db[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # First part: k from i to min(i+n-1, size-1)
        k1_start = i
        k1_end = min(i + n, size)
        if k1_end > k1_start:
            grad[i] += np.dot(w[k1_start:k1_end], a[:k1_end - k1_start])
        # Second part: k from max(0, i-n+1) to i
        k2_start = max(0, i - n + 1)
        k2_end = min(i + 1, size)
        if k2_end > k2_start:
            # w[k] * a[i-k] for k from k2_start to k2_end-1
            # a[i-k] where k ranges from k2_start to k2_end-1
            # i-k ranges from i-k2_start down to i-(k2_end-1)
```

```python
def get_conv_matrix(n, a):
    global _conv_cache
    if n not in _conv_cache:
        size = 2 * n - 1
        rows = np.zeros(n * n, dtype=np.int32)
        cols = np.zeros(n * n, dtype=np.int32)
        for i in range(n):

            rows[start:start + n] = np.arange(i, i + n)
            cols[start:start + n] = i
        _conv_cache[n] = (rows, cols, size)
    rows, cols, size = _conv_cache[n]

    T = sparse.csc_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def lp_best_response_with_dual(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = get_conv_matrix(n, a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    dual_val = None
    try:
        dv = prob.constraints[0].dual_value
        if dv is not None:
            dual_val = np.asarray(dv, dtype=np.float64)
            dual_val = np.maximum(dual_val, 0.0)
    except Exception:
        pass
    return b_val, dual_val

def lse_obj_and_grad(x, n, T_temp):
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    size = 2 * n - 1
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    lse_val = max_b + np.log(exp_s.sum()) / T_temp
    fft_size = 1 << (n + size - 1).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:size] = w
    A_fft = rfft(a_pad, fft_size)
    W_fft = rfft(w_pad, fft_size)
    conv_wa = irfft(A_fft * W_fft, fft_size)
    corr_aw = irfft(np.conj(A_fft) * W_fft, fft_size)
    grad_lse = corr_aw[:n] + conv_wa[:n]
    obj_val = 2.0 * n * lse_val / (s * s)
    grad_obj = 2.0 * n * grad_lse / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad_obj.ravel()

def peak_gradient(a, k_star):
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        v = 0.0
        j1 = k_star - i
        if 0 <= j1 < n:
            v += a[j1]
        j2 = i - k_star
        if 0 <= j2 < n:
            v += a[j2]
        grad[i] = v
    return grad

            src_candidates = np.where(grad_src > max_b / n)[0]
            if len(src_candidates) == 0:
                continue
            src_candidates = src_candidates[np.argsort(grad_src[src_candidates])[::-1]][:5]

            avail_mask = np.ones(n, dtype=bool)
            for k2 in top_k:
                k2 = int(k2)
                js = max(0, k2 - n + 1)
                je = min(n - 1, k2)
                avail_mask[js:je + 1] = False

            dst_candidates = np.where(avail_mask)[0]
            if len(dst_candidates) == 0:
                dst_candidates = np.where(np.zeros(n, dtype=bool))[0]

            for src in src_candidates:
                if a[src] < 1e-12:
                    continue
                dst_scored = []
                for k_dst in top_k[1:]:
                    k_dst = int(k_dst)
                    grad_dst = peak_gradient(a, k_dst)
                    for dst in dst_candidates[:20]:
                        total_gain = grad_src[src] / max(grad_dst[dst] + 1e-30, 1e-30)
                        dst_scored.append((total_gain, dst, k_dst))
                dst_scored.sort(reverse=True)

                for gain_val, dst, k_dst in dst_scored[:6]:
                    found = False
                    for frac in [0.4, 0.25, 0.15, 0.08, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm_val / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 30:
                for _ in range(40):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
            if stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    t0_scan = time.monotonic()
    scan_budget = time_budget * 0.12
    scan_lengths = [1400, 1600, 1800, 2000]
    scan_results = {}

    for n_scan in scan_lengths:
        elapsed = time.monotonic() - t0_scan
        if elapsed > scan_budget - 30:
            break
        scan_time = min(scan_budget - elapsed - 10, 80.0)
        if scan_time < 15:
            break
        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)
        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out, _ = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} ({(time.monotonic()-t0_scan):.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Deep LP at best length
    elapsed_total = time.monotonic() - t_global
    remaining = time_budget - elapsed_total
    print(f"\n=== Phase 1: Deep LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _, dual_out = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
    do_try_update(a_out, f"lp_deep_n{n_base}")
    best_dual = dual_out

    # Phase 2: LSE smooth optimization
    elapsed_total = time.monotonic() - t_global
    remaining = time_budget - elapsed_total
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=40.0, T_end=2.0, n_stages=5)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 3: Dual-guided coordinate descent
    elapsed_total = time.monotonic() - t_global
    remaining = time_budget - elapsed_total
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Dual-guided CD n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = dual_guided_cd(a_norm, t3, dual_hint=best_dual, seed=42)
        do_try_update(a_out, f"dual_cd_n{n_best}")

    # Phase 4: Multi-peak CD
    elapsed_total = time.monotonic() - t_global
    remaining = time_budget - elapsed_total
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Multi-peak CD n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = multi_peak_cd(a_norm, t4, seed=777)
        do_try_update(a_out, f"mp_cd_n{n_best}")

    # Phase 5: Try a slightly different length
    elapsed_total = time.monotonic() - t_global
    remaining = time_budget - elapsed_total
    n_alt = len(best_seq) + 200
    print(f"\n=== Phase 5: LP alt length n={n_alt} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 20:
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"lp_alt_n{n_alt}")

    # Phase 6: Final coordinate descent polish
    elapsed_total = time.monotonic() - t_global
    remaining = time_budget - elapsed_total
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = remaining * 0.5
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = dual_guided_cd(a_norm, min(t6, 200.0), dual_hint=best_dual, seed=99999)
        do_try_update(a_out, "final_cd")

    # Phase 7: Remaining time - LSE with low temperature
    elapsed_total = time.monotonic() - t_global
    remaining = time_budget - elapsed_total
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 7: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
        t7 = min(remaining * 0.6, 150.0)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t7, T_start=8.0, T_end=0.5, n_stages=4)
        do_try_update(a_out, "lse_final")

    # Final coordinate descent with whatever time is left
    elapsed_total = time.monotonic() - t_global
    remaining = time_budget - elapsed_total
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 8: Last CD n={n_best} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = dual_guided_cd(a_norm, remaining * 0.8, dual_hint=best_dual, seed=31415)
        do_try_update(a_out, "last_cd")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
