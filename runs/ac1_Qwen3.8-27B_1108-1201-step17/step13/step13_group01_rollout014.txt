def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_constraint_matrix(a, n):
    """Build the (2n-1) x n constraint matrix T where (T@b)[k] = (a*b)(k)."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T

def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def lse_objective(a_flat, n, T_temp):
    """Compute LSE-smoothed objective and its gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    # Shift for numerical stability
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()  # weights for gradient
    
    # Gradient of LSE w.r.t. a
    # d(b_k)/da_i = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    # Using FFT: if we pad w to length 2n-1, then the gradient contribution
    # is the cross-correlation of a with w
    fft_size = 1 << (2 * n - 1).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    # cross-correlation: (a * w) in the "correlation" sense
    # We need sum_k w_k * (a[k-i] + a[i-k]) for each i
    # This is: sum_k w_k * a[k-i] + sum_k w_k * a[i-k]
    # = (w * a reversed correlation) which is the auto-conv structure
    
    # Actually, let me think more carefully.
    # b_k = sum_i a[i] * a[k-i] (with bounds)
    # db_k / da_i = a[k-i] (if valid) 
    # dLSE/da_i = sum_k w_k * db_k/da_i
    #            = sum_k w_k * a[k-i] (over valid k)
    
    # This is the cross-correlation of w with a (reversed)
    # = sum_k w_k * a[k-i] = (w cross-convolve a)[i] in a sense
    # Actually it's: for fixed i, sum over k of w[k] * a[k-i]
    # This is the "sliding dot product" = cross-correlation(w, a) at lag i
    # = fft-based: IFFT(FFT(w) * conj(FFT(a))) but with proper indexing
    
    # Let's use the direct FFT approach:
    # conv(w, a) gives c[m] = sum_j w[j] * a[m-j]
    # We want for each i: sum_k w[k] * a[k-i] = conv(w, a)[2*size-n+i]... 
    
    # Simpler: just use the fact that sum_k w[k]*a[k-i] is the cross-correlation
    # which can be computed as IFFT(FFT(w_zero_padded) * conj(FFT(a_zero_padded)))
    
    w_full = np.zeros(fft_size)
    w_full[:2*n-1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a
    
    W_full_fft = rfft(w_full, fft_size)
    A_full_fft = rfft(a_full, fft_size)
    
    # Cross-correlation: corr[i] = sum_k w[k] * a[k-i] 
    # = IFFT(FFT(w) * conj(FFT(a)))[i] for appropriate indexing
    # Actually for circular correlation:
    # circ_corr[i] = sum_k w[k] * a[(k-i) mod N]
    # For our case with zero-padding, this gives the linear cross-correlation
    
    cross = irfft(W_full_fft * np.conj(A_full_fft), fft_size)[:n]
    
    # So dLSE/da_i = cross[i]
    # And the full objective is: 2*n*LSE / s^2
    # d(obj)/da_i = 2*n*cross[i]/s^2 - 2*obj*a[i]/s
    
    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

def lse_scipy_optimize(a0, time_limit, T_start=50.0, T_end=2.0, n_stages=5):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / (n_stages - si), 150.0)

        def obj_func(x):
            val, _ = lse_objective(x, n, T)
            return val

        def grad_func(x):
            _, g = lse_objective(x, n, T)
            return g

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func, 
                                  options={'maxiter': 300, 'maxfun': 1000,
                                           'gtol': 1e-8, 'ftol': 1e-10,
                                           'maxls': 50})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate.copy()
                        print(f"    LSE T={T:.1f}: score={score_c:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.1f} error: {e}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with tight M target and Nesterov acceleration (primary)
2. LSE (log-sum-exp) smoothed gradient with L-BFGS-B for continuous refinement
3. Targeted coordinate descent focusing on peak-contributing elements
4. Multi-length exploration with interpolation
Key improvements: tighter M targets (forcing peak reduction), adaptive step sizes,
peak-targeted coordinate moves, and improved time allocation.

import numpy as np

import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():

            prob.solve(solver=s)

                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:

    _global_solver = cp.ECOS

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]

def build_constraint_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad


def lse_scipy_optimize(a0, time_limit, T_start=80.0, T_end=3.0, n_stages=5):

def targeted_coord_descent(a0, time_limit, seed=42):

        k_peak = int(np.argmax(b))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

            non_peak = np.arange(n)
        non_peak_sorted = non_peak[np.argsort(a[non_peak])]
        low_val_targets = non_peak_sorted[:15]

        improved = False
        for src in top_contrib:
            if a[src] < 1e-10:
                continue
            for dst in low_val_targets:
                if dst == src:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for _ in range(80):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.12)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            stall += 1
            if stall > 150:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_state, candidate_a, label, t_global):
    """Try to update best state with candidate. Returns updated best_state tuple."""
    best_seq, best_score = best_state
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: LP best-response with tight M, LSE gradient, targeted coord descent."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov with tight M (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest tight M n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_tight_M=True)
    best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_tight_n{n_best}", t_global)

    # Phase 2: LSE gradient optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE gradient n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 200.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=80.0, T_end=3.0, n_stages=5)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lse_n{n_best}", t_global)

    # Phase 3: Targeted coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, t3, seed=42)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"coord1_n{n_best}", t_global)

    # Phase 4: LP Nesterov round 2 from improved point
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP-Nest round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=False, use_tight_M=True)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_r2_n{n_best}", t_global)

    # Phase 5: Try longer sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_best + 400
    print(f"\n=== Phase 5: LP-Nest longer n={n_long} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, use_tight_M=True)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_long_n{n_long}", t_global)

    # Phase 6: Final polish - alternate coord descent and LSE
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        third = remaining / 3.0

        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_coord_descent(a_norm, third, seed=99999)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, "coord_final1", t_global)

        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 15:
            a_norm2 = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
            a_out, _ = lse_scipy_optimize(a_norm2, min(remaining2, 100.0), T_start=40.0, T_end=1.0, n_stages=3)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "lse_final", t_global)

        remaining3 = time_budget - (time.monotonic() - t_global)
        if remaining3 > 15:
            n_best3 = len(best_seq)
            a_norm3 = best_seq * np.sqrt(2.0 * n_best3) / best_seq.sum()
            a_out, _ = targeted_coord_descent(a_norm3, remaining3, seed=777)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "coord_final2", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
