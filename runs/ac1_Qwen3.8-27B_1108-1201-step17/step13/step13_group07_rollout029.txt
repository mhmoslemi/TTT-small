def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-14:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target_sum)
    rho_val = np.max(rho[cond]) if np.any(cond) else 0
    theta = (cssv[rho_val - 1] - target_sum) / rho_val if rho_val > 0 else 0
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        result = result / max(result.sum(), 1e-30) * target_sum
        result = np.maximum(result, 0)
    return result

    b_self = fast_autocorr(a_fixed)

            je = min(n - 1, k)
            for j in range(js, je + 1):

        prob = cp.Problem(cp.Minimize(M_var), constraints)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Column-generation LP game with Nesterov acceleration (primary optimizer)
2. LogSumExp-smoothed gradient descent with adaptive temperature (secondary)
3. Multi-strategy peak balancing coordinate descent (polish)
4. Multi-length exploration with interpolation transfer
Key differences from previous: column generation for faster LP, 
better gradient method, more aggressive peak balancing.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Full convolution a*b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    b = fast_autocorr(a)
    return float(2.0 * n * b.max())

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-14:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho_val = int(np.max(rho[cond]))
    theta = (cssv[rho_val - 1] - target_sum) / rho_val
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 1e-30:
            result = result * (target_sum / s)
        else:
            result = np.full(n, target_sum / n)
    return result


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name, max_iter=100)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response_cg(a_fixed, solver, max_constraints=150, max_cg_iters=8):
    """Solve LP best response with column generation (cutting plane)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    b_self = fast_autocorr(a_fixed)
    active_lags = list(np.argsort(b_self)[::-1][:max_constraints])

    b_val = None
    M_val = float('inf')

    for cg_iter in range(max_cg_iters):
        m = len(active_lags)
        T = np.zeros((m, n))
        for i, k in enumerate(active_lags):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                T[i, j] = a_fixed[k - j]

        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        try:
            prob.solve(solver=solver, max_iter=2000)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
        except Exception:
            pass

        if not solved:
            try:
                prob.solve(solver=cp.SCS, max_iter=10000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
            except Exception:
                pass

        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=5000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
            except Exception:
                pass

        if not solved:
            return None, None

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        M_val = float(M_var.value)

        # Check if all constraints are satisfied
        cross_conv = fast_cross_conv(a_fixed, b_val)
        max_cross = float(cross_conv.max())

        if max_cross <= M_val * (1.0 + 1e-5) + 1e-10:
            return b_val, M_val

        # Add the most violated constraint
        violated_lag = int(np.argmax(cross_conv))
        if violated_lag not in active_lags:
            active_lags.append(violated_lag)
            if len(active_lags) > max_constraints + 50:
                vals = np.array([cross_conv[l] for l in active_lags])
                order = np.argsort(vals)[::-1][:max_constraints]
                active_lags = [active_lags[i] for i in order]

    return b_val, M_val


def nesterov_lp_optimize_cg(a0, time_limit, solver, max_constraints=150, verbose=True):
    """Nesterov-accelerated LP best-response with column generation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response_cg(y, solver, max_constraints)
        if b_star is None:
            stall_count += 50
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        score_a = internal_score(a)
        score_y = internal_score(y)
        current_best = min(score_a, score_y)

        if current_best < best_score - 1e-10:
            best_score = current_best
            if score_a <= score_y:
                best_a = a.copy()
            else:
                best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        avg_a += a
        n_avg += 1
        if n_avg >= 100 and n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score - 1e-10:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if stall_count > 2000:
            break

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL-CG t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def lse_gradient(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*b_k)) where b=conv(a,a)."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # grad_j = 2 * sum_k w_k * a_{k-j} (with valid index handling)
    # This is 2 * (reverse(w) convolve a) appropriately indexed
    # Actually: sum_k w_k * a_{k-j} for valid k-j in [0,n-1]
    # = (w_reversed * a_reversed) cross-correlation... 
    # Let's use FFT: convolve w with reversed a, then extract the right part
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1].copy(), fft_size)
    conv = irfft(W * A_rev, fft_size)
    # conv[k] = sum_j w[j] * a_rev[k-j] = sum_j w[j] * a[n-1-(k-j)]
    # We want sum_k w[k] * a[k-j] for each j
    # = sum_k w[k] * a[k-j]
    # Let's think: if we define c[j] = sum_k w[k] * a[k-j] (valid k-j)
    # This is the cross-correlation of w and a at lag j
    # cross_corr(j) = sum_k w[k+j] * a[k] = sum_m w[m] * a[m-j] (sub m=k+j)
    # So c[j] = cross_corr(w, a)[j]
    # Using FFT: cross_corr(w,a)[j] = IFFT(FFT(w) * conj(FFT(a)))[j+n-1] (with appropriate size)
    
    # Simpler approach: 
    # c[j] = sum_k w[k] * a[k-j] where 0<=k-j<n means j <= k < j+n
    # = sum_{k=j}^{j+n-1} w[k] * a[k-j] (with w[k]=0 for k>=2n-1)
    # This is (w * a_padded) at position... 
    # Let's just use the direct FFT approach:
    # c = conv(w_full, a) where w_full has length 2n-1 and a has length n
    # conv(w_full, a)[j + n - 1]... hmm this is getting confusing.
    
    # Let me just do it directly with a small FFT trick:
    # We want g[j] = sum_k w[k] * a[k-j] for j = 0,...,n-1
    # = sum_k w[k] * a[k-j]
    # Define a_ext of length 2n-1: a_ext[m] = a[m] if 0<=m<n else 0
    # Then g[j] = sum_{k=j}^{2n-2} w[k] * a_ext[k-j]
    # = (w convolve a_ext_ext)[...] 
    # Actually g[j] = sum_k w[k] * a_ext[k-j] = conv(w, a_ext_flipped)... no.
    
    # The cleanest: g = cross_correlation(w, a) where cross_correlation[w,a][j] = sum_k w[k] * a[k-j]
    # Using FFT: cross_correlation = IFFT(conj(FFT(w)) * FFT(a)) but with care about lengths.
    
    # Actually the simplest correct approach:
    # g[j] = sum_k w[k] * a[k-j] for 0<=k-j<n
    # Let's create a padded array: a_pad of length (2n-1) + n - 1 = 3n-2
    # No, let's just use the correlation via FFT:
    # corr[j] = sum_k w[k] * a[k-j] 
    # = sum_k w[k] * a[k-j]
    # If we reverse a: a_r[i] = a[n-1-i]
    # Then a[k-j] = a_r[n-1-k+j]
    # g[j] = sum_k w[k] * a_r[n-1-k+j]
    # Let m = k: g[j] = sum_m w[m] * a_r[n-1-m+j]
    # = conv(w, a_r)[n-1+j] (where conv is standard linear convolution)
    
    a_r = a[::-1].copy()
    fft_size2 = 1 << (len(w) + len(a_r) - 1 - 1).bit_length()
    W2 = rfft(w, fft_size2)
    AR = rfft(a_r, fft_size2)
    conv_full = irfft(W2 * AR, fft_size2)
    # conv_full[n-1+j] = sum_m w[m] * a_r[n-1+j-m] = sum_m w[m] * a[j-m+n-1-(n-1)] ... 
    # Hmm let me just verify: conv_full[k] = sum_m w[m] * a_r[k-m]
    # g[j] = sum_m w[m] * a_r[n-1+j-m] = conv_full[n-1+j]
    g = conv_full[n-1:n-1+n]
    
    grad = 2.0 * g
    return grad, w


def gradient_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=15, lr=0.05, verbose=True):
    """LogSumExp-smoothed gradient descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    velocity = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Determine iterations for this stage
        iters = max(20, min(300, int(remaining * 80 / (n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            grad, w = lse_gradient(a, stage_T)
            
            t_adam += 1
            m_vec = beta1 * m_vec + (1 - beta1) * grad
            v_vec = beta2 * v_vec + (1 - beta2) * grad ** 2
            m_hat = m_vec / (1 - beta1 ** t_adam)
            v_hat = v_vec / (1 - beta2 ** t_adam)

            lr_t = lr / (1 + 0.005 * it)
            velocity = 0.9 * velocity + 0.1 * m_hat / (np.sqrt(v_hat) + 1e-8)

            a_new = a - lr_t * velocity
            a_new = project_simplex(a_new)

            # Check improvement every few iterations
            if it % 5 == 0 or it == iters - 1:
                score = internal_score(a_new)
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
            a = a_new

        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-grad stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_balancing(a0, time_limit, seed=42, verbose=True):
    """Aggressive peak balancing via coordinate descent."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        peak_lags = np.where(b > 0.95 * peak_val)[0]
        
        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(b)])
        
        improved = False
        
        # Strategy 1: Move mass from elements contributing to the highest peak
        # to elements that don't contribute to any peak
        for _ in range(20):
            k_main = int(peak_lags[rng.integers(len(peak_lags))])
            
            # Find elements that contribute most to this peak
            js = max(0, k_main - n + 1)
            je = min(n - 1, k_main)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_main - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])
            
            # Find destination: element not in any peak's "influence zone"
            peak_zones = set()
            for k2 in peak_lags[:min(20, len(peak_lags))]:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                peak_zones.update(range(js2, je2 + 1))
            
            candidates = np.array([j for j in range(n) if j not in peak_zones])
            if len(candidates) == 0:
                candidates = np.array([j for j in range(n) if j != src])
            if len(candidates) == 0:
                continue
            
            # Prefer low-value candidates
            cand_vals = a[candidates]
            sorted_idx = np.argsort(cand_vals)
            dst_pool = candidates[sorted_idx[:max(1, len(sorted_idx) // 4)]]
            dst = int(dst_pool[rng.integers(len(dst_pool))])
            
            # Try different transfer fractions
            best_frac = None
            best_trial_score = internal_score(a)
            for frac in [0.001, 0.003, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                trial_score = internal_score(a_trial)
                if trial_score < best_trial_score - 1e-12:
                    best_trial_score = trial_score
                    best_frac = frac
                    best_trial = a_trial.copy()
            
            if best_frac is not None:
                a = best_trial
                if best_trial_score < best_score:
                    best_score = best_trial_score
                    best_a = a.copy()
                    stall = 0
                improved = True
        
        # Strategy 2: Random perturbation (escape local minima)
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-15:
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                    elif rng.uniform() < 0.01:
                        # Occasional random accepted move (simulated annealing-like)
                        a = a_trial
                        a = a / a.sum()

            if stall > 300:
                # Larger perturbation
                n_perturb = max(5, n // 50)
                perturb_idx = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                amounts = rng.uniform(0, 0.02, size=n_perturb)
                for pi in range(n_perturb):
                    idx_from = int(perturb_idx[pi])
                    idx_to = int(rng.integers(n))
                    if idx_from != idx_to and a_trial[idx_from] > amounts[pi]:
                        a_trial[idx_from] -= amounts[pi]
                        a_trial[idx_to] += amounts[pi]
                a_trial = np.maximum(a_trial, 0)
                s_trial = a_trial.sum()
                if s_trial > 1e-15:
                    a_trial = a_trial / s_trial
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 1000:
                break

        if verbose:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 30 == 0 and int(elapsed) != getattr(peak_balancing, '_last_print', -1):
                peak_balancing._last_print = int(elapsed)
                print(f"    PeakBal ({elapsed:.0f}s): best={best_score:.6f}, stall={stall}")

    return best_a, best_score


def spectral_shaping(a, n_iter=300, alpha=0.25):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_direct(a0, time_limit, verbose=True):
    """Direct subgradient optimization with adaptive smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    v = np.zeros(n)
    beta = 0.85
    base_lr = 0.08
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)

        # Adaptive temperature: start high (smooth), end low (sharp)
        progress = min(1.0, t / max(1, int(time_limit * 20)))
        T_smooth = 1.0 + 200.0 * (1.0 - progress) ** 2

        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Gradient via FFT
        a_r = a[::-1].copy()
        fft_size = 1 << (len(w) + len(a_r) - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_r, fft_size)
        conv_full = irfft(W * AR, fft_size)
        grad = 2.0 * conv_full[n - 1:n - 1 + n]

        lr_t = base_lr / (1 + 0.005 * t)
        v = beta * v + (1 - beta) * grad
        a_new = a - lr_t * v
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization targeting score < 1.5027.
    Uses column-generation LP game, LSE gradient descent, peak balancing,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 1 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    n_base = 1600
    if best_seq is None:
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Spectral shaping (quick initial improvement)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n=n_base (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        for alpha_val in [0.3, 0.15, 0.1]:
            a_out = spectral_shaping(best_seq, n_iter=200, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 1: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 2: Nesterov LP game with column generation (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: NL-CG n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.38, 650.0)
    if t2 > 30:
        a_out, sc_out = nesterov_lp_optimize_cg(best_seq, t2, solver, max_constraints=150)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP-CG]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 2: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 3: LSE gradient descent (complementary direction)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LSE-grad n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 10:
        a_out, _ = gradient_lse_optimize(best_seq, t3, T_start=50.0, T_end=0.1, n_stages=12, lr=0.04)
        sc = evaluate_safe(a_out)

    # Phase 4: Direct subgradient (different dynamics)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: SubGrad n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 10:
        a_out, _ = subgradient_direct(best_seq, t4)

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()

            best_seq = np.maximum(a_out, 0)

            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_r2]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 7: best={best_score:.6f} (t={elapsed:.0f}s)")

    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final polish n={len(best_seq)} (remaining={remaining:.0f}s) ===")
