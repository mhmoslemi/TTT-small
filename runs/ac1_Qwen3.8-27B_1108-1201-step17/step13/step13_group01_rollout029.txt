    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T

                a_new = g.copy()
            else:
                momentum = (n_iters - 1.0) / (n_iters + 2.0)
                a_new = y_new + momentum * (y_new - y)
                a_new = np.maximum(a_new, 0.0)
                s_a = a_new.sum()
                if s_a < 1e-15:
                    a_new = y_new.copy()
                else:
                    a_new = a_new * norm / s_a
        else:
            a_new = y_new.copy()

        y = y_new
        a = a_new

        polyak_buf[polyak_idx % 200] = a
        polyak_idx += 1
        n_polyak = min(polyak_idx, 200)

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        if n_polyak >= 50 and n_polyak % 40 == 0:
            avg = polyak_buf[:n_polyak].mean(axis=0)
            s_avg = avg.sum()
            if s_avg > 1e-15:
                avg = avg * norm / s_avg
                score_avg = fast_autocorr(avg).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg.copy()
                    last_improve = time.monotonic()

        # Stall detection with restart
        if time.monotonic() - last_improve > 25.0:
            stall_count += 1
            if stall_count > 12:
                if verbose:
                    print(f"    NLP stall at iter={n_iters}, best={best_score:.8f}, restarting")
                # Restart from best
                y = best_a.copy()
                a = best_a.copy()
                momentum_reset = True
                T = build_convolution_matrix(y, n)
                T_rebuild_t = t
                stall_count = 0
                last_improve = time.monotonic()
                # Add small perturbation
                rng_r = np.random.default_rng(t)
                pert = rng_r.normal(0, 0.01, n)
                y = np.maximum(y + pert, 0)
                s_y2 = y.sum()
                if s_y2 > 1e-15:
                    y = y * norm / s_y2
                a = y.copy()
                if n_iters > restart_every * 3:
                    if verbose:
                        print(f"    NLP giving up at iter={n_iters}")
                    break
        else:
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pocs_spectral_optimize(a0, time_limit, verbose=True):
    """POCS: alternate between non-negativity projection and flat-spectrum projection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall = 0

    # FFT size
    fft_size = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        # Step 1: Compute spectrum
        A = fft(a, fft_size)

        # Step 2: Project toward flat spectrum (in magnitude)
        mag = np.abs(A)
        # Target: make |A|^2 as flat as possible (in the first n-1 bins where signal lives)
        # Use a soft flattening: mix current mag with its mean
        mag_mean = mag[:n].mean()
        # Blend factor: start aggressive, reduce over time
        n_iters += 1
        blend = max(0.1, 1.0 / (1.0 + 0.01 * n_iters))
        new_mag = (1 - blend) * mag + blend * mag_mean
        # Keep phase
        phase = A / (np.abs(A) + 1e-30)
        A_new = new_mag * phase
        a_new = np.real(ifft(A_new, fft_size))[:n]

        # Step 3: Project to non-negativity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            last_improve = time.monotonic()
        else:
            # Keep old a if new is worse
            a = a
            stall += 1
            if stall > 100:
                break

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    POCS iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multiplicative_weight_optimize(a0, time_limit, verbose=True, eta_start=0.5, eta_end=0.01, n_peaks=5):
    """Multiplicative weight updates targeting top peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        M_curr = b.max()

        # Find top peaks
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        # Compute pressure on each position
        pressure = np.zeros(n)
        for k in top_peaks:
            weight = max(0.0, b[k] - 0.5 * M_curr)  # Only weight peaks above half-max
            # Pressure contribution: for lag k, position i contributes a[i]*a[k-i]
            # The subgradient of conv(a,a)[k] w.r.t. a[i] is a[k-i] + a[i-k] (with bounds)
            for i in range(n):
                j1 = k - i
                j2 = i - k
                val = 0.0
                if 0 <= j1 < n:
                    val += a[j1]
                if 0 <= j2 < n:
                    val += a[j2]
                pressure[i] += weight * val

        p_max = pressure.max()
        if p_max < 1e-15:
            break

        # Adaptive eta
        progress = (time.monotonic() - t0) / time_limit
        eta = eta_start * (eta_end / eta_start) ** progress
        eta = max(eta, 0.001)

        # Multiplicative update
        a_new = a * np.exp(-eta * pressure / p_max)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        n_iters += 1
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            last_improve = time.monotonic()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """Targeted coordinate descent on top peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    fracs = [0.005, 0.01, 0.02, 0.05, 0.1]

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:5]

        improved = False
        for k_peak in top_peaks:
            # Find positions that contribute most to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_src = idx[np.argsort(contrib)[::-1][:8]]

            # Build peak mask (positions in any top peak)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_src:
                if a[src] < 1e-10:
                    continue
                # Try moving mass from src to non-peak positions
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]

                # Sort by current value (move to lower positions first)
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(4, len(candidates) // 4)]

                for dst in candidates_sorted[:4]:
                    for frac in fracs:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation on stall
            if stall % 5 == 0:
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break

            if stall > 300:
                break

    return best_a, best_score


def lse_objective(a_flat, n, T_temp):
    a = np.maximum(a_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    shifted = T_temp * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    lse = max_b + np.log(np.exp(shifted).sum()) / T_temp
    return 2.0 * n * lse / (s * s)


def lse_gradient(a_flat, n, T_temp):
    a = np.maximum(a_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    shifted = T_temp * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp

    # Gradient of lse w.r.t. a: (1/T) * sum_k w[k] * d(conv(a,a)[k])/da
    # d(conv(a,a)[k])/da[i] = a[k-i] + a[i-k] (with bounds)
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 2.0 * cross / T_temp
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad


def lse_scipy_optimize(a0, time_limit, T_start=20.0, T_end=1.0, n_stages=5):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / (n_stages - si), 100.0)

        x0 = a.copy()

        def obj_func(x):
            return lse_objective(x, n, T)

        def grad_func(x):
            return lse_gradient(x, n, T)

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                 jac=grad_func, options={'maxiter': 150,
                                 'maxfun': 400, 'ftol': 1e-10, 'gtol': 1e-8})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 4)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(6 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif choice == 2:
        a = np.exp(-6.0 * (x - 0.2) ** 2) + np.exp(-6.0 * (x - 0.8) ** 2) + 0.1
    else:
        a = np.sin(np.pi * x) ** 0.8 + 0.05
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final coord n={n_best} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, remaining * 0.7, seed=12345)
        do_try_update(a_out, "coord_final2")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
