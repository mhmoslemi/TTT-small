  **Theoretical consideration:**
  The problem is equivalent to: find a non-negative sequence a of length n that minimizes the "peak-to-mean ratio" of its autocorrelation. The mean of the autocorrelation (averaged over all lags) is S^2/(2n-1) (by Parseval). So we want to minimize max_b / mean_b = max_b * (2n-1) / S^2.

  The score is 2n * max_b / S^2 = 2n/(2n-1) * max_b/mean_b ≈ max_b/mean_b for large n.

  So we want the autocorrelation to be as flat as possible.

  **Connection to "Golomb rulers" and "Costas arrays":**

                    frac = rng.uniform(0.05, 0.2)
                    a_trial = a.copy()
                    a_trial[i] *= (1 - frac)
                    a_trial[j] += a[i] * frac
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 1e-10:
                        a_trial = a_trial / st
                        if obj_value(a_trial) < best_score:
                            a = a_trial
                            best_score = obj_value(a)
                            best_a = a.copy()
                            stall = 0
            
            if stall > 200:
                break
        
        if verbose and iteration % 10 == 0:
            print(f"    PairCD iter={iteration}: score={best_score:.8f} (t={time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for i in range(n):
        start = i
        end = min(i + n, size)
        for m in range(start, end):
            rows.append(m)
            cols.append(i)
            vals.append(a[m - i])
    
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    
    best_a = a.copy()
    best_score = obj_value(a)
    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_curr = fast_autocorr(y)
        M_curr = b_curr.max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        n_iters = t
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y = y_new
        a = a_new
        
        score_a = obj_value(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = obj_value(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                score_avg = obj_value(avg_check)
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
