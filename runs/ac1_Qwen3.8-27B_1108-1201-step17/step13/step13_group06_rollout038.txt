    # sum_j a[j] * p[i+j] = cross-correlation of a and p (reversed) at position i

    # Using FFT: cross = ifft(fft(a, L) * conj(fft(p, L)))  ... 
    
    # Actually, the standard cross-correlation theorem says:
    # ifft(fft(x) * conj(fft(y))) gives sum_n x[n+m] * y[n]  (circular)
    # Which is the cross-correlation (x ⋆ y)[m] = sum_n x[n+m] * conj(y[n])
    
    # What I want: sum_j a[j] * p[j+k]
    # Let me define a_rev[j] = a[n-1-j]

    # = sum_m a_rev[m] * p[(n-1+k)-m]
    # = (a_rev * p)[n-1+k] (convolution!)

    # = sum_j a[j] * p[i-j]
    # Let m = i-j: j = i-m
    # = sum_m a[i-m] * p[m]
    # = (a * p)[i] (convolution!)
    
    # Wait, that's not right either. Let me be more careful.

    # = sum_m a[i-m] * p[m]  (where m ranges so that i-m is valid)

    # = sum_m a[m-i] * p[m]

            # slope[2i] = 2*x (derivative of x^2 at x)

            k_start = i
            k_end = min(i + n, 2 * n - 1)
            
            # For affected k (k != 2i): b_k(x') = b_k + (x'-x) * 2*a[k-i]
            # For k = 2i (if in range): b_k(x') = b_k + x'^2 - x^2
            
            # Compute slopes for affected k
            affected = np.arange(k_start, k_end)
            offsets = affected - i  # = 0, 1, ..., min(n-1, 2n-2-i)
            slopes = 2.0 * a[offsets]
            
            # Handle k = 2i specially
            if 2 * i < k_end and 2 * i >= k_start:
                idx_2i = 2 * i - k_start
                # At k=2i: b_{2i}(x') = b_{2i} + x'^2 - x^2
                # This is NOT linear in x'. We'll handle it separately.
                slopes[idx_2i] = 0  # Will handle separately

            mask_affected = np.zeros(2 * n - 1, dtype=bool)
            mask_affected[k_start:k_end] = True
            max_unaffected = b[~mask_affected].max() if (~mask_affected).any() else 0.0
            
            # Bisection to find optimal x'
            # Objective: f(x') = max(b_k + (x'-x)*slopes_k for affected k) and b_{2i} + x'^2 - x^2
            # This is convex in x'. Find minimum by bisection on derivative.
            
            def f_of_x(xp):
                """Compute max b_k when a[i] = xp."""
                # Affected linear parts
                vals = b[k_start:k_end] + (xp - x) * slopes
                # Special case k=2i
                if 2 * i < k_end and 2 * i >= k_start:
                    idx_2i = 2 * i - k_start
                    vals[idx_2i] = b[2 * i] + xp * xp - x * x
                return max(vals.max(), max_unaffected)

            # x' can be in [0, some_upper]

            f_hi = f_of_x(x_hi)

                    elif f_right < f_xmid:

                    else:
                        break

    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = fast_autocorr(a)
    best_a = a.copy()
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    cycle = 0
    stall = 0

    while time.monotonic() - t0 < time_limit - 10:
        if cycle % 7 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            if time.monotonic() - t0 > time_limit - 10:
                break

            x = a[i]
            if x < 1e-14:
                continue

            k_start = i
            k_end = min(i + n, 2 * n - 1)
            if k_start >= k_end:
                continue

            affected = np.arange(k_start, k_end)
            offsets = affected - i
            slopes = 2.0 * a[offsets].copy()

            has_2i = (2 * i >= k_start) and (2 * i < k_end)
            if has_2i:
                idx_2i = 2 * i - k_start
                slopes[idx_2i] = 0.0

            mask_aff = np.zeros(2 * n - 1, dtype=bool)
            mask_aff[k_start:k_end] = True
            if (~mask_aff).any():
                max_unaff = b[~mask_aff].max()
            else:
                max_unaff = 0.0

            b_aff = b[k_start:k_end].copy()
            b_2i_val = b[2 * i] if has_2i else 0.0

            def f_opt(xp):
                vals = b_aff + (xp - x) * slopes
                if has_2i:
                    vals[idx_2i] = b_2i_val + xp * xp - x * x
                local_max = vals.max()
                if local_max > max_unaff:
                    return local_max
                return max_unaff

            f_at_x = f_opt(x)
            f_at_0 = f_opt(0.0)
            if f_at_0 <= f_at_x:
                a[i] = 0.0
                continue

            x_hi = min(1.0, 3.0 * x + 2.0 / n)
            f_at_hi = f_opt(x_hi)
            if f_at_hi <= f_at_x:
                a[i] = x_hi
                continue

            # Ternary search on [0, x_hi]
            lo = 0.0
            hi = x_hi
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = f_opt(m1)
                f2 = f_opt(m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            a[i] = 0.5 * (lo + hi)

        s = a.sum()
        if s > 1e-15:
            a = a / s

        b = fast_autocorr(a)
        cur_score = 2.0 * n * b.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        cycle += 1
        if cycle % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      BCD cycle {cycle} ({elapsed:.0f}s): score={best_score:.6f}")

        if stall > 30:
            pert = rng.random(n) * 0.005
            a_pert = np.maximum(a + pert, 0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert = a_pert / sp
                sp_score = 2.0 * n * fast_autocorr(a_pert).max()
                if sp_score < best_score:
                    a = a_pert
                    best_score = sp_score
                    best_a = a.copy()
                    stall = 0
                else:
                    a = best_a.copy()
            if stall > 100:
                break

    return best_a, best_score


def lse_adam_optimize(a0, time_limit, T_start=30.0, T_end=0.3, n_stages=25, seed=123):
    """Full LSE Adam with adaptive temperature and Nesterov momentum."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        lr = 2e-3 * min(1.0, 80.0 / n)
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        max_it = max(30, int(min(remaining, 45.0) * 8))
        stall_local = 0

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            a = softmax_fn(z)
            obj, grad_a = lse_objective_and_gradient(a, stage_T)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 8 == 0:
                a_check = softmax_fn(z)
                sc = 2.0 * n * fast_autocorr(a_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
                    stall_local = 0
                else:
                    stall_local += 1

                if stall_local > 15:
                    z = np.log(np.maximum(best_a, 1e-30))
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0
                    stall_local = 0

        a_stage = softmax_fn(z)
        sc_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if sc_stage < best_score:
            best_score = sc_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver):
    """Nesterov-accelerated LP zero-sum game."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    warm_b = None
    stall = 0

    avg_buf = []
    AVG_WINDOW = 50
    avg_sum = np.zeros(n)

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit - 5:
            break

        T = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T, solver, warm_b)
        if b_star is None:
            warm_b = None
            T = build_toeplitz(y)
            b_star, M_val = solve_lp_best_response(y, T, solver, None)
            if b_star is None:
                continue
        warm_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        avg_sum += a
        if len(avg_buf) >= AVG_WINDOW:
            avg_sum -= avg_buf[0]
        avg_buf.append(a.copy())

        sc_a = 2.0 * n * fast_autocorr(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if len(avg_buf) >= AVG_WINDOW and t % 20 == 0:
            avg_a = avg_sum / AVG_WINDOW
            sc_avg = 2.0 * n * fast_autocorr(avg_a).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_a.copy()
                stall = 0

        if stall > 400 and t > 200:
            a = best_a.copy()
            y = best_a.copy()
            avg_buf = []
            avg_sum = np.zeros(n)
            stall = 0
            warm_b = None

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): score={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Systematic peak shaving with smart destination selection."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 5:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.98
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            danger = np.zeros(n, dtype=bool)
            for kk in top_peaks[:12]:
                djs = max(0, kk - n + 1)
                dje = min(n - 1, kk)
                danger[djs:dje + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(8, len(safe_sorted) // 3)]
            if len(dst_pool) == 0:
                continue

            found = False
            for dst in dst_pool[:10]:
                for frac in [0.003, 0.008, 0.015, 0.03, 0.05, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 15 == 0:
                n_pert = rng.integers(1, 6)
                a_trial = a.copy()
                for _ in range(n_pert):
                    ii = rng.integers(n)
                    jj = rng.integers(n)
                    if ii != jj and a_trial[ii] > 1e-10:
                        frac = rng.uniform(0.001, 0.04)
                        amt = a_trial[ii] * frac
                        a_trial[ii] -= amt
                        a_trial[jj] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 5)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-10 * (x - 0.3) ** 2) + 0.5 * np.exp(-10 * (x - 0.7) ** 2)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (1 + 0.3 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 6:
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - np.abs(x) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(time_budget=1800.0):
    """
    Multi-phase optimization:
    Phase 1: BCD at parent length (45%)
    Phase 2: LSE Adam at parent length (20%)
    Phase 3: LP game at parent length (15%)
    Phase 4: BCD at longer length (12%)
    Phase 5: Final polish (8%)
    """
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 60.0, 1740.0)
    rng = np.random.default_rng(2024)

    solver = find_solver()
    print(f"Solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    # Phase 1: BCD at parent length
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 750.0)
    print(f"\n=== Phase 1: BCD n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = block_coordinate_descent(best_seq, t1, seed=42)
        try_update(a_out, f"bcd_n{n_base}")

    # Phase 2: LSE Adam
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 330.0)
    print(f"=== Phase 2: LSE Adam n={n_base} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lse_adam_optimize(best_seq, t2, T_start=40.0, T_end=0.5, n_stages=20, seed=77)
        try_update(a_out, f"lse_n{n_base}")

    # Phase 3: LP Game
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 250.0)
    print(f"=== Phase 3: LP Game n={n_base} (budget={t3:.0f}s) ===")
    if t3 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver)
        try_update(a_out, f"lp_n{n_base}")

    # Phase 4: BCD at longer length
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 200.0)
    print(f"=== Phase 4: BCD longer length (budget={t4:.0f}s) ===")
    if t4 > 60:
        n_long = n_base + 400
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = block_coordinate_descent(init_long, t4, seed=99)
        try_update(a_out, f"bcd_n{n_long}")

    # Phase 5: Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.08, 120.0)
    print(f"=== Phase 5: Polish (budget={t5:.0f}s) ===")
    if t5 > 20:
        # Quick BCD
        t5a = min(t5 * 0.6, 70.0)
        a_out, _ = block_coordinate_descent(best_seq, t5a, seed=555)
        try_update(a_out, "bcd_polish")

        # Peak shaving
        t5b = max(5.0, t5 - t5a - 5)
        a_out, _ = peak_shaving(best_seq, t5b, seed=1337)
        try_update(a_out, "peak_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
