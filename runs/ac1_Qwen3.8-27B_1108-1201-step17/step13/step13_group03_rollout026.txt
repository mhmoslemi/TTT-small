import numpy as np
import time
from scipy.fft import rfft, irfft


def _fast_autocorr(a):
    """Full convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _objective_normalized(a):
    """Compute 2n*max(conv)/sum^2 for non-negative a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = _fast_autocorr(a)
    return 2.0 * n * conv.max() / (s * s)


def _proj_simplex(v):
    """Project vector v onto the probability simplex {x: sum=1, x>=0}."""
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_gradient(y, T):
    """Gradient of LSE-smoothed max autocorrelation at y with temperature T."""
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = np.clip((ac - ac_max) / T, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def _lse_fista(a0, time_limit, verbose=False, T_list=None):
    """FISTA on LSE-smoothed max autocorrelation with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if T_list is None:
        T_list = [15.0, 5.0, 1.5, 0.4, 0.1, 0.03]

    y = a.copy()
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_list):
        if time.monotonic() - t0 > time_limit:
            break
        L = max(n / (T * 2.0), 10.0)
        t_fista = 1
        prev_a = a.copy()
        max_iters = 20000

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            grad = _lse_gradient(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - prev_a)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            prev_a = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 1000 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.3f} ({elapsed:.0f}s): max_conv={best_score:.8f}")

    sc = _fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _coord_descent(a, time_limit, verbose=False, n_cycles_max=500):
    """
    Exact coordinate descent: for each coordinate i, solve the 1D subproblem
    min_x 2n * max(M_unaff, max_k(x*s_k + c_k)) / (D+x)^2
    using upper envelope computation and critical point search.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # normalize to sum=1

    best_a = a.copy()
    best_score = 2.0 * n * _fast_autocorr(a).max()
    t0 = time.monotonic()
    conv = _fast_autocorr(a)
    n_conv = 2 * n - 1
    stall = 0

    for cycle in range(n_cycles_max):
        if time.monotonic() - t0 > time_limit:
            break
        cycle_improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            if old_val < 1e-15 and a.sum() > 0.01:
                # Still try to add some mass here
                pass

            D = s - old_val  # sum of all other coordinates
            if D < 1e-15:
                continue

            # Affected convolution indices: k where i is valid, i.e., k-i in [0,n-1]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            if k_lo > k_hi:
                continue

            # For affected k: conv_k = x * a[k-i] + (conv_k - old_val * a[k-i])
            indices = np.arange(k_lo, k_hi + 1)
            slopes = np.array([a[k - i] for k in indices])
            intercepts = conv[indices] - old_val * slopes

            # Unaffected max
            mask = np.zeros(n_conv, dtype=bool)
            mask[k_lo:k_hi + 1] = True
            if (~mask).any():
                M_unaff = conv[~mask].max()
            else:
                M_unaff = 0.0

            # Find optimal x >= 0
            # f(x) = 2n * max(M_unaff, max_k(x*s_k + c_k)) / (D + x)^2
            # Try: x=0, x=old_val, and critical points

            candidates = [0.0, old_val]

            # For each line k: critical point x_k = D - 2*c_k/s_k (if s_k > 0)
            pos_slopes = slopes > 1e-15
            if pos_slopes.any():
                crit_x = D - 2.0 * intercepts[pos_slopes] / slopes[pos_slopes]
                crit_x = crit_x[crit_x > 0]
                # Add a few of the most promising critical points
                if len(crit_x) > 0:
                    # Evaluate f at each and keep top candidates
                    crit_x_sorted = crit_x[np.argsort(crit_x)]
                    # Sample at most 20 critical points
                    if len(crit_x_sorted) > 20:
                        idxs = np.linspace(0, len(crit_x_sorted) - 1, 20, dtype=int)
                        candidates.extend(crit_x_sorted[idxs].tolist())
                    else:
                        candidates.extend(crit_x_sorted.tolist())

            # Also try some geometric spacing
            if D > 0:
                candidates.append(D / n)
                candidates.append(2.0 * D / n)
                candidates.append(D / (n * 2.0))

            # Evaluate f at candidates
            best_x = old_val
            best_f = 2.0 * n * max(M_unaff, (old_val * slopes + intercepts).max()) / (D + old_val) ** 2

            for x in candidates:
                if x < 0:
                    continue
                new_max_aff = (x * slopes + intercepts).max()
                new_max = max(M_unaff, new_max_aff)
                denom = (D + x) ** 2
                if denom < 1e-30:
                    continue
                f_val = 2.0 * n * new_max / denom
                if f_val < best_f:
                    best_f = f_val
                    best_x = x

            # Refine with local search around best_x
            if best_x > 0:
                for delta_frac in [0.5, 0.1, 0.01, -0.5, -0.1, -0.01, 1.5, 2.0]:
                    x_try = best_x * (1.0 + delta_frac) if delta_frac > -1 else best_x + delta_frac
                    if x_try < 0 or x_try > 100 * D / n:
                        continue
                    new_max_aff = (x_try * slopes + intercepts).max()
                    new_max = max(M_unaff, new_max_aff)
                    denom = (D + x_try) ** 2
                    if denom < 1e-30:
                        continue
                    f_val = 2.0 * n * new_max / denom
                    if f_val < best_f:
                        best_f = f_val
                        best_x = x_try

            if abs(best_x - old_val) > 1e-14:
                a[i] = best_x
                cycle_improved = True

        # Renormalize
        s_new = a.sum()
        if s_new > 1e-15:
            a = a / s_new
        else:
            break

        # Check score
        conv = _fast_autocorr(a)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and (cycle % 10 == 0 or stall >= 3):
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent cycle={cycle} ({elapsed:.0f}s): score={best_score:.10f} stall={stall}")

        if stall >= 5:
            break

    return best_a, best_score / (2.0 * n)  # return normalized (sum=1) best and max_conv/(2n)


def _interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _symmetrize(a):
    """Make sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    sym = 0.5 * (a + a[::-1])
    return sym


def _peak_shave(a, time_limit, seed=42):
    """Targeted peak reduction via mass redistribution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = _fast_autocorr(a)
        best_score_cur = 2.0 * n * conv.max()
        top_k = np.argsort(conv)[::-1][:6]
        improved = False

        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find destination: low values far from peak contributions
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            sorted_cands = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 3)]
            dst = sorted_cands[rng.integers(len(sorted_cands))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                score = 2.0 * n * _fast_autocorr(a_trial).max()
                if score < best_score_cur - 1e-14:
                    a = a_trial
                    best_score_cur = score
                    improved = True
                    break
            if improved:
                best_a = a.copy()
                best_score = best_score_cur
                stall = 0
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * _fast_autocorr(a_trial).max()
                    if score < best_score_cur:
                        a = a_trial
                        best_score_cur = score
                        best_a = a.copy()
                        best_score = best_score_cur
                        stall = 0

            if stall > 200:
                break

    return best_a, best_score / (2.0 * n)


def _subgradient_descent(a, time_limit, verbose=False):
    """
    Subgradient descent on 2n*max(conv) with Polyak step size.
    Works on the simplex (sum=1, a>=0).
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * _fast_autocorr(a).max()
    t0 = time.monotonic()

    # Lower bound: 2n/(2n-1) (achievable only in limit)
    f_star = 2.0 * n / (2.0 * n - 1.0)

    y = a.copy()
    t = 1
    step_hist = []

    while time.monotonic() - t0 < time_limit:
        conv = _fast_autocorr(y)
        f_t = 2.0 * n * conv.max()
        k_star = int(np.argmax(conv))

        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            gi = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                gi += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                gi += y[j2]
            grad[i] = 2.0 * n * gi

        # Polyak step size
        if f_t <= f_star + 1e-15:
            break
        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq < 1e-30:
            break
        eta = (f_t - f_star) / (g_norm_sq + 1e-30)
        eta = min(eta, 0.5 / t ** 0.5)  # cap for stability

        a_new = y - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = np.ones(n) / n

        if t == 1:
            y_new = a_new.copy()
        else:
            mom = (t - 1.0) / (t + 2.0)
            y_new = a_new + mom * (a_new - y)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new / s_y
            else:
                y_new = a_new.copy()

        y = y_new
        t += 1

        if t % 500 == 0:
            sc = 2.0 * n * _fast_autocorr(y).max()
            if sc < best_score:
                best_score = sc
                best_a = y.copy()
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    SubGrad t={t} ({elapsed:.0f}s): score={best_score:.10f}")

            # Also check a_new
            sc2 = 2.0 * n * _fast_autocorr(a_new).max()
            if sc2 < best_score:
                best_score = sc2
                best_a = a_new.copy()

    sc = 2.0 * n * _fast_autocorr(y).max()
    if sc < best_score:
        best_score = sc
        best_a = y.copy()
    return best_a, best_score

    m = int(taper * n)
    for i in range(m):
        f = (i + 1) / m
        sh[i] = f
        sh[n - 1 - i] = f
    shapes.append(sh)

    # Parabolic
    shapes.append(1.0 - (2 * x - 1) ** 2)

    # Power law edge decay
    for alpha in [0.5, 1.0, 2.0]:
        edge = np.minimum(x, 1 - x)
        sh = edge ** alpha
        shapes.append(sh)

    # Normalize all
    norms = []
    for sh in shapes:
        s = sh.sum()
        if s > 1e-15:
            norms.append(sh / s)
        else:
            norms.append(np.ones(n) / n)

    return norms


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-method, multi-length optimization with coordinate descent."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Get parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            best_seq = parent.copy()
            best_score = parent_score
    except Exception:
        parent = None

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if best_seq is None:
        n_base = 1600
        best_seq = np.sin(np.pi * np.linspace(0, 1, n_base)) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # ============================================================
    # Phase 1: Coordinate descent at base length (fast, new method)
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 20:
        print(f"\n=== Phase 1: CoordDescent n={n_base} (budget={t1:.0f}s) ===")
        a_out, _ = _coord_descent(best_seq, t1, verbose=True)
        _try_update(a_out, f"coord1_n{n_base}")

    # ============================================================
    # Phase 2: LSE-FISTA at base length
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        print(f"\n=== Phase 2: LSE-FISTA n={n_cur} (budget={t2:.0f}s) ===")
        T_sched = [10.0, 3.0, 0.8, 0.2, 0.05, 0.01]
        a_out, _ = _lse_fista(best_seq, t2, verbose=True, T_list=T_sched)
        _try_update(a_out, f"lse1_n{n_cur}")

    # ============================================================
    # Phase 3: Coordinate descent at n+300
    # ============================================================
    n2 = n_cur + 300
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: CoordDescent n={n2} (budget={t3:.0f}s) ===")
        init_long = _interpolate_to_length(best_seq, n2)
        a_out, _ = _coord_descent(init_long, t3, verbose=True)
        _try_update(a_out, f"coord2_n{n2}")

    # ============================================================
    # Phase 4: LSE-FISTA at n+300
    # ============================================================
    n2_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 160.0)
    if t4 > 20:
        print(f"\n=== Phase 4: LSE-FISTA n={n2_best} (budget={t4:.0f}s) ===")
        T_sched = [5.0, 1.5, 0.4, 0.1, 0.03]
        a_out, _ = _lse_fista(best_seq, t4, verbose=True, T_list=T_sched)
        _try_update(a_out, f"lse2_n{n2_best}")

    # ============================================================
    # Phase 5: Try different length n+500 with coordinate descent
    # ============================================================
    n3 = n_cur + 500
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 160.0)
    if t5 > 20:
        print(f"\n=== Phase 5: CoordDescent n={n3} (budget={t5:.0f}s) ===")
        init_long3 = _interpolate_to_length(best_seq, n3)
        a_out, _ = _coord_descent(init_long3, t5, verbose=True)
        _try_update(a_out, f"coord3_n{n3}")

    # ============================================================
    # Phase 6: Subgradient descent (different landscape exploration)
    # ============================================================
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 15:
        print(f"\n=== Phase 6: SubGrad n={n_cur} (budget={t6:.0f}s) ===")
        a_out, _ = _subgradient_descent(best_seq, t6, verbose=True)
        _try_update(a_out, f"subgrad_n{n_cur}")

    # ============================================================
    # Phase 7: Symmetric coordinate descent
    # ============================================================
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.08, 100.0)
    if t7 > 15:
        print(f"\n=== Phase 7: Symmetric CoordDescent n={n_cur} (budget={t7:.0f}s) ===")
        a_sym = _symmetrize(best_seq)
        a_sym = a_sym / a_sym.sum()
        a_out, _ = _coord_descent(a_sym, t7 * 0.7, verbose=True)
        # Enforce symmetry on result
        a_out_sym = _symmetrize(a_out)
        a_out_sym = a_out_sym / a_out_sym.sum()
        _try_update(a_out_sym, f"symcoord_n{n_cur}")

    # ============================================================
    # Phase 8: Multi-start with different shapes at best length
    # ============================================================
    n_cur = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.12, 150.0)
    if t8 > 30:
        print(f"\n=== Phase 8: Multi-start shapes n={n_cur} (budget={t8:.0f}s) ===")
        shapes = _generate_initial_shapes(n_cur, rng)
        # Pick top 3 most different shapes from current best
        scored = []
        for i, sh in enumerate(shapes):
            dist = np.linalg.norm(sh - best_seq)
            scored.append((dist, i, sh))
        scored.sort(key=lambda x: -x[0])

        n_starts = min(3, len(scored))
        time_per_start = t8 / n_starts
        for si in range(n_starts):
            if time.monotonic() - t_global > time_budget - 60:
                break
            _, _, sh = scored[si]
            # Blend with current best for warm start
            blend = 0.3 * sh + 0.7 * best_seq
            blend = blend / blend.sum()
            a_out, _ = _coord_descent(blend, time_per_start, verbose=False)
            _try_update(a_out, f"multi_start_{si}_n{n_cur}")

    # ============================================================
    # Phase 9: Longer length exploration
    # ============================================================
    n4 = len(best_seq) + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t9 = min(remaining * 0.10, 130.0)
    if t9 > 20:
        print(f"\n=== Phase 9: CoordDescent n={n4} (budget={t9:.0f}s) ===")
        init_long4 = _interpolate_to_length(best_seq, n4)
        a_out, _ = _coord_descent(init_long4, t9 * 0.6, verbose=True)
        _try_update(a_out, f"coord4_n{n4}")

        # LSE on the longer one
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t9b = min(remaining * 0.4, 100.0)
            a_out, _ = _lse_fista(best_seq, t9b, verbose=True,
                                  T_list=[3.0, 1.0, 0.3, 0.08])
            _try_update(a_out, f"lse3_n{len(best_seq)}")

    # ============================================================
    # Phase 10: Final coordinate descent polish
    # ============================================================
    n_final = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t10 = min(remaining * 0.15, 180.0)
    if t10 > 20:
        print(f"\n=== Phase 10: Final CoordDescent n={n_final} (budget={t10:.0f}s) ===")
        a_out, _ = _coord_descent(best_seq, t10, verbose=True)
        _try_update(a_out, f"coord_final_n{n_final}")

    # ============================================================
    # Phase 11: Peak shaving
    # ============================================================
    n_final = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t11 = min(remaining * 0.20, 150.0)
    if t11 > 10:
        print(f"\n=== Phase 11: PeakShave n={n_final} (budget={t11:.0f}s) ===")
        a_out, _ = _peak_shave(best_seq, t11, seed=99999)
        _try_update(a_out, f"peak_shave_n{n_final}")

    # ============================================================
    # Phase 12: Final LSE micro-polish
    # ============================================================
    n_final = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t12 = min(remaining * 0.30, 200.0)
    if t12 > 15:
        print(f"\n=== Phase 12: Final LSE n={n_final} (budget={t12:.0f}s) ===")
        T_sched = [2.0, 0.5, 0.1, 0.02, 0.005]
        a_out, _ = _lse_fista(best_seq, t12, verbose=True, T_list=T_sched)
        _try_update(a_out, f"lse_final_n{n_final}")

    # Final verification
    if best_seq is not None:
        final_score = _evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
