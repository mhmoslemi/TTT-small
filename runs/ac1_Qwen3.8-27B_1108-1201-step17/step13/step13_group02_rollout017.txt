    """Solve the LP best-response problem using dual formulation for better conditioning.
    Dual: min M * sum(y) s.t. T^T y >= 1, y >= 0
    Then b = T^T y - 1 gives the primal... actually no.
    Let's use the primal: max 1^T b s.t. T b <= M*1, b >= 0
    But solve via the dual for potential speedup.
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    if warm_start is not None:
        b_var.value = warm_start
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    return b_val, prob


def lp_best_response_reduced(a, M, solver, warm_start=None, max_rounds=5):
    """LP with iterative constraint reduction using active set."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Initial solve with all constraints
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Iteratively reduce to active constraints
    for _ in range(max_rounds):
        circ = T @ b_val
        tol = 1e-6 * M + 1e-12
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= 2 * n - 1:
            break
        if n_active < 10:
            break
        b_var = cp.Variable(n, nonneg=True)
        constraints = [T[active, :] @ b_var <= M[active]]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-5) + 1e-12:
            b_val = b_new
        else:
            break
    
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, alpha_mode='constant', alpha_val=0.15, verbose=True):
    """LP Nesterov with configurable alpha schedule."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    warm_b = None
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    consec_improve = 0

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        # Choose LP method based on iteration count
        if n_iters < 30:
            b = lp_best_response_reduced(y, M_curr, solver, warm_start=warm_b)
        else:
            b = lp_best_response_reduced(y, M_curr, solver, warm_start=warm_b, max_rounds=4)

        if b is None:
            # Try with relaxed constraint
            b = lp_best_response_reduced(y, M_curr * 1.01, solver, warm_start=warm_b)
            if b is None:
                continue
            M_curr *= 1.01

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        warm_b = b.copy()
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        if alpha_mode == 'standard':
            alpha = 2.0 / (n_iters + 2.0)
        else:
            # Decay slightly: start at alpha_val, slowly decrease
            alpha = alpha_val / (1.0 + 0.001 * n_iters)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if n_iters == 1:
            a_new = g.copy()
        else:
            # Nesterov extrapolation
            momentum = (n_iters - 1.0) / (n_iters + 2.0)
            if alpha_mode == 'standard':
                momentum = (n_iters - 1.0) / (n_iters + 2.0)
            else:
                momentum = alpha / (2.0 * alpha + 1.0)  # Adjusted for constant alpha
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            consec_improve += 1
        else:
            stall += 1
            consec_improve = max(0, consec_improve - 1)

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            consec_improve += 1

        # Adaptive restart
        if stall > 800 and consec_improve == 0:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            stall = 0
            warm_b = None
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    [Restart at iter {n_iters}, t={elapsed:.0f}s]")

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    """Direct subgradient descent using the exact subgradient of max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    step_size = norm * 0.05 / n

    # We use the subgradient of max_k (a*a)_k
    # Subgrad at a: for the set of maximizing indices K, 
    # subgrad_j = sum_{k in K} lambda_k * 2 * a[k-j] (if 0 <= k-j < n)
    # where lambda is a probability distribution over K
    
    # For efficiency, use uniform distribution over top few peaks
    
    T_anneal_start = best_score * 0.1
    T_anneal_end = best_score * 0.005

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_anneal_start * (T_anneal_end / T_anneal_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        # Compute subgradient using top peaks with softmax_weights
        # For the exact subgradient, use uniform over maximizing set
        # For smooth approximation, use softmax
        if T > 1e-10:
            w = np.exp((b - max_b) / T)
            w_sum = w.sum()
            if w_sum < 1e-30:
                break
            w = w / w_sum
            # Filter to top peaks for efficiency
            top_idx = np.argsort(b)[::-1][:32]
            w_sub = w[top_idx]
            w_sub = w_sub / w_sub.sum()
        else:
            active = b >= max_b - 1e-10
            if active.sum() == 0:
                active = np.argmax(b)
                active = np.array([active])
            w_sub = np.ones(len(active)) / len(active)
            top_idx = np.where(active)[0]

        # Compute gradient: grad_j = 2 * sum_k w_k * a[k-j] (if 0<=k-j<n)
        # This is 2 * (w_sub_extended * a_rev)[shift]
        # Efficient: for each k in top_idx, grad += 2*w_k * a[k-j] for valid j
        # = 2 * sum_k w_k * a[max(0, k-j) ... ] - use correlation
        
        # grad_j = 2 * sum_{k in top_idx} w_sub[k_idx] * a[j'] where j' = k - j, 0<=j'<n
        # = 2 * sum_{k} w_k * a[k-j] * 1_{0<=k-j<n}
        # For j in [0,n-1]: sum_k w_k * a[k-j] = (w_ext * a_rev)[j + n - 1]
        # where w_ext has w at positions k and 0 elsewhere, length 2n-1
        # a_rev = a[::-1], length n
        # conv(w_ext, a_rev) has length 3n-2
        # element [j + n - 1] = sum_i w_ext[i] * a_rev[n-1-(j+n-1-i)] = sum_i w_ext[i] * a_rev[i-j]
        # = sum_i w_ext[i] * a[n-1-i+j]... this is getting confusing.
        # Let me just use direct computation for small top_idx
        
        grad = np.zeros(n)
        for ki, k in enumerate(top_idx):
            wk = w_sub[ki]
            if wk < 1e-15:
                continue
            # grad_j += 2 * wk * a[k-j] for j where 0 <= k-j < n
            # j ranges from max(0, k-n+1) to min(n-1, k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # a[k-j] for j in [j_start, j_end]: indices k-j_end to k-j_start
            a_idx = np.arange(k - j_end, k - j_start + 1)
            grad[j_start:j_end + 1] += 2.0 * wk * a[a_idx]

        # Gradient descent step
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # Decay step size slowly
            step_size *= 0.9999
        else:
            step_size *= 0.998

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): best={best_score:.8f}, step={step_size:.6f}")

    return best_a, best_score


def peak_shave_analytical(a0, time_limit, seed=42, verbose=True):
    """Peak shaving with analytical redistribution."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Get current max peak and its contributing indices
        max_k = int(np.argmax(b))
        max_val = b[max_k]
        
        js = max(0, max_k - n + 1)
        je = min(n - 1, max_k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contrib = a[idx] * a[max_k - idx]

        if contrib.max() < 1e-20:
            break

        src = idx[np.argmax(contrib)]
        src_val = a[src]
        
        if src_val < 1e-15:
            # Try a different peak
            sim_idx = np.argsort(b)[::-1][:5]
            found = False
            for sk in sim_idx:
                js2 = max(0, sk - n + 1)
                je2 = min(n - 1, sk)
                idx2 = np.arange(js2, je2 + 1)
                if len(idx2) == 0:
                    continue
                contrib2 = a[idx2] * a[sk - idx2]
                src2 = idx2[np.argmax(contrib2)]
                if a[src2] > 1e-10 * a.mean():
                    src = src2
                    src_val = a[src]
                    max_k = sk
                    js = js2
                    je = je2
                    idx = idx2
                    found = True
                    break
            if not found:
                break

        # Find good destinations: positions that don't contribute to top peaks
        top_peaks = np.argsort(b)[::-1][:10]
        peak_mask = np.zeros(n, dtype=bool)
        for pk in top_peaks:
            js3 = max(0, pk - n + 1)
            je3 = min(n - 1, pk)
            peak_mask[js3:je3 + 1] = True
        
        non_peak = np.where(~peak_mask)[0]
        non_peak = non_peak[non_peak != src]
        if len(non_peak) == 0:
            non_peak = np.where(np.arange(n) != src)[0]

        # Select destination: prefer low values in non-peak regions
        n_try = min(8, len(non_peak))
        candidates = non_peak[np.argsort(a[non_peak])[:n_try * 2]]
        
        improved = False
        for dst in candidates[:n_try]:
            # Try different fractions
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                amt = src_val * frac
                if amt < 1e-18:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t

                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    n_improve += 1
                    improved = True
                    break

            if improved:
                break

        if not improved:
            n_stall += 1
            # Random perturbation
            if n_stall % 20 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10 * a.mean():
                    frac = rng.uniform(0.05, 0.2)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] = max(0, a[i] - amt)
                    a_trial[j] += amt
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                            n_stall = 0
                        
            if n_stall > 500:
                # Larger perturbation
                n_pert = max(5, n // 100)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.2, 0.5)
                    a_trial[idx] -= amt
                    j2 = rng.integers(n)
                    a_trial[j2] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.005:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0

        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length, preserving the sum-normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_diverse_init(base_a, n, rng, diversity=0.3):
    """Create a diverse initialization by perturbing the base shape."""
    shape = base_a / base_a.sum()
    # Add random modulation
    modulation = 1.0 + diversity * (rng.random(n) - 0.5)
    # Smooth it a bit to avoid too much noise
    kernel = np.ones(5) / 5
    modulation_smooth = np.convolve(modulation, kernel, mode='same')
    new_shape = shape * modulation_smooth
    new_shape = np.maximum(new_shape, 0.0)
    ss = new_shape.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return new_shape / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-phase optimization with LP Nesterov, subgradient polishing, and peak shaving."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1710.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')

    # Load parent
    parent_available = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            best_seq = p.copy()
            best_score = evaluate_safe(best_seq)
            parent_available = True
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None or best_score >= float('inf'):

        best_seq = np.sin(np.pi * x) ** 1.5

        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    remaining = time_budget - (time.monotonic() - t_global)

    n_cur = len(best_seq)

    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        # Add small perturbation to escape potential local optimum in the iteration
        rng_pert = np.random.default_rng(12345)
        a_perturbed = a_norm + 0.01 * a_norm.mean() * rng_pert.normal(0, 1, n_cur)
        a_perturbed = np.maximum(a_perturbed, 0)
        a_perturbed = a_perturbed * np.sqrt(2.0 * n_cur) / a_perturbed.sum()
        a_out, _ = nesterov_lp_optimize(a_perturbed, t2, solver, alpha_mode='constant', alpha_val=0.25, verbose=True)
        do_try_update(a_out, f"lp_a25_n{n_cur}")

    # Phase 3: Subgradient polishing
    remaining = time_budget - (time.monotonic() - t_global)

    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t3, verbose=True)

    # Phase 4: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)

        do_try_update(a_out, f"peak_n{n_cur}")
        # Another round with different seed

        t4b = min(remaining * 0.08, 80.0)
        if t4b > 10:
            n_cur = len(best_seq)

            a_out, _ = peak_shave_analytical(a_norm, t4b, seed=7777, verbose=True)
            do_try_update(a_out, f"peak2_n{n_cur}")

    # Phase 5: Try slightly longer length
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n_long} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 30:
        init_long = interpolate_to_length(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, alpha_mode='constant', alpha_val=0.15, verbose=True)
        do_try_update(a_out, f"lp_n{n_long}")
        
        # Subgradient on the longer one
        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.08, 60.0)
        if t5b > 10:
            n_best_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_best_cur) / best_seq.sum()
            a_out, _ = subgradient_polish(a_norm, t5b, verbose=False)
            do_try_update(a_out, f"subgrad2_n{n_best_cur}")

    # Phase 6: Final peak shaving polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final PeakShave n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shave_analytical(a_norm, t6, seed=31415, verbose=True)
        do_try_update(a_out, f"peak_final_n{n_best}")

    # Phase 7: Final subgradient + LP combo on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP polish n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.30, 150.0)
    if t7 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, alpha_mode='constant', alpha_val=0.10, verbose=True)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # Final: subgradient on absolute best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.30, 80.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_polish(a_norm, t8, verbose=False)
        do_try_update(a_out, f"subgrad_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
