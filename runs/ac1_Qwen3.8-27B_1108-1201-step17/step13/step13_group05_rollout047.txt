def fast_autocorr(a):

    n = len(a)

    A = rfft(a.astype(np.float64), fft_size)

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def lse_smooth_grad(a, ac, temperature):
    """Compute gradient of log-sum-exp smoothed max of ac with respect to a."""
    n = len(a)
    shifted = ac - ac.max()
    weights = np.exp(shifted / temperature)
    weights /= weights.sum()
    # g[i] = 2 * sum_k weights[k] * a[k-i] for valid k-i
    # = 2 * np.correlate(weights, a, mode='valid')
    g = 2.0 * np.correlate(weights, a, mode='valid')
    return g

        a = a * c / s0

    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    decay = 0.001  # temperature decay rate
    
    base_lr = 0.1
    momentum = 0.9
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Adaptive temperature: start high, decrease over time
        T = T_0 * M / (1.0 + decay * t)
        T = max(T, 0.01 * M + 1e-15)
        
        # Compute smooth gradient
        g = lse_smooth_grad(a, ac, T)
        
        # Step size: diminishing
        step = base_lr / (1.0 + 0.001 * t)

        v = momentum * v + (1.0 - momentum) * g
        a_new = a - step * v

        if M_new < best_m:

def lp_best_response_sparse(a, M, solver, top_m=50):
    """Solve LP with only top-M constraints for speed."""
    n = len(a)
    ac = fast_autocorr(a)
    # Use only top-m constraints
    top_indices = np.argsort(ac)[::-1][:top_m]
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix for top-m lags only
    T = np.zeros((top_m, n))
    for ii, k in enumerate(top_indices):
        k_int = int(k)
        for j in range(n):
            kj = k_int - j
            if 0 <= kj < n:
                T[ii, j] = a[kj]
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_polish(a0, time_limit, solver, verbose=True):
    """LP-based polish: alternate between LP best response and coordinate descent."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        M_curr = fast_autocorr(a).max()
        
        # Solve LP
        b = lp_best_response_sparse(a, M_curr, solver, top_m=30)
        if b is None:
            break
        
        # Scale b to match normalization
        b_sum = b.sum()
        g = b * c / b_sum
        
        # Blend: a = (1-alpha) * a + alpha * g
        alpha = 2.0 / (iteration + 2.0)
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * c / s_new
        
        M_new = fast_autocorr(a_new).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={iteration} ({elapsed:.1f}s): J={J:.6f}")

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
