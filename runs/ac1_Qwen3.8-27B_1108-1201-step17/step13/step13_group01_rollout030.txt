def normalize_seq(a, norm_target):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a * norm_target / s

def build_conv_matrix(a, n):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T

def lp_best_response(a, M, solver, T=None):
    n = len(a)
    if T is None:
        T = build_conv_matrix(a, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    dual_val = None
    try:
        dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration and adaptive M tightening
2. LSE smooth surrogate with gradient-based optimization  
3. Peak-targeted coordinate descent with trial moves
4. Spectral diversification for escape from local minima
"""

import numpy as np
import cvxpy as cp
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def normalize_seq(a, norm_target):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * norm_target / len(a)
    return a * norm_target / s

def build_conv_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T

def lp_best_response(a, M, solver, T=None):
    n = len(a)
    if T is None:
        T = build_conv_matrix(a, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall_count = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        # Adaptive tightening
        tighten = min(0.03, 0.003 * math.log1p(t / 100.0))
        M_eff = M_curr * (1.0 - tighten)
        M_eff = max(M_eff, 0.95 * M_curr)
        
        T_mat = build_conv_matrix(y, n)
        b = lp_best_response(y, M_eff, solver, T_mat)
        if b is None:
            b = lp_best_response(y, M_curr, solver, T_mat)
            if b is None:
                continue
        
        g = b * norm / b.sum()
        n_iters += 1
        
        # Step size
        ratio = M_curr / max(best_score, 1e-10)
        alpha = 2.0 / (t + 2.0)
        alpha = min(alpha * min(ratio, 3.0), 1.0)
        
        y_new = np.maximum((1.0 - alpha) * y + alpha * g, 0.0)
        y_new = normalize_seq(y_new, norm)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = normalize_seq(a_new, norm)
        
        y = y_new
        a = a_new
        
        # Track best
        for cand in [y, a]:
            sc = fast_autocorr(cand).max()
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = cand.copy()
                last_improve = time.monotonic()
        
        # Polyak average
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = normalize_seq(avg_a / n_avg, norm)
            sc_avg = fast_autocorr(avg_check).max()
            if sc_avg < best_score - 1e-10:
                best_score = sc_avg
                best_a = avg_check.copy()
                last_improve = time.monotonic()
        
        # Stall detection
        if time.monotonic() - last_improve > 90.0:
            stall_count += 1
            if stall_count > 20:
                if verbose:
                    print(f"    NLP stall at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0
        
        if verbose and n_iters % 50 == 0:
            print(f"    NLP iter={n_iters} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

# LSE gradient (FFT-based)
def lse_obj_grad(x, n, T):
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    exp_vals = np.exp(T * (b - max_b))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    lse = max_b + np.log(sum_exp) / T
    obj = 2.0 * n * lse / (s * s)
    
    # grad_conv via FFT convolution
    a_rev = a[::-1].copy()
    fft_size = 1 << (3 * n - 1).bit_length()
    A_fft = rfft(a, fft_size)
    AREV_fft = rfft(a_rev, fft_size)
    W_fft = rfft(w, fft_size)
    
    # conv(a_rev, w) -> irfft(AREV_fft * W_fft)
    conv_arev_w = irfft(AREV_fft * W_fft, fft_size)
    # conv(a, w) -> irfft(A_fft * W_fft)
    conv_a_w = irfft(A_fft * W_fft, fft_size)
    
    grad_conv = conv_arev_w[n-1:2*n-1] + conv_a_w[0:n]
    
    grad = (2.0 * n / (s * s)) * grad_conv - (4.0 * n * lse / (s ** 3)) * np.ones(n)
    return obj, grad.ravel()

def lse_scipy(a0, time_limit, T_start=20.0, T_end=1.0, n_stages=5):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_seq(a0, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / (n_stages - si), 80.0)
        
        def obj_func(x):
            return lse_obj_grad(x, n, T)[0]
        
        def grad_func(x):
            return lse_obj_grad(x, n, T)[1]
        
        try:
            res = scipy_minimize(obj_func, a, method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 200, 'maxfun': 800,
                                           'time_limit': stage_time})
            if res.x is not None:
                a_c = normalize_seq(res.x, norm)
                sc = fast_autocorr(a_c).max()
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a_c.copy()
                    a = a_c
                    print(f"    LSE T={T:.1f}: score={best_score:.8f}")
        except Exception:
            pass
    
    return best_a, best_score
