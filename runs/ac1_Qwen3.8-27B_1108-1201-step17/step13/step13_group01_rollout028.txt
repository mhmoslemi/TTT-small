4. Peak-targeted coordinate descent for final refinement
Key: fast LP via active-set cutting planes enables more iterations,
adaptive momentum, and multi-phase polish for final improvement.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv(a, b):
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_full_constraint_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                T[k, j] = a[i]
    return T


def lp_best_response_cp(a, M, T, solver, max_active=400, tol=1e-9):
    n = len(a)
    size = 2 * n - 1

    aa = fast_autocorr(a)
    k_peak = int(np.argmax(aa))
    active = sorted(set(list(range(max(0, k_peak - 8), min(size, k_peak + 9)))))

    b_var = cp.Variable(n, nonneg=True)
    b_solution = None
    prev_b = None

    for cp_iter in range(max_active):
        T_active = T[active]
        constraints = [T_active @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if b_solution is not None:
                return b_solution
            return None

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if b_solution is not None:
                return b_solution
            return None

        b_val = np.asarray(b_var.value, dtype=np.float64)
        if b_val is None:
            return None
        b_val = np.maximum(b_val, 0.0)

        if prev_b is not None:
            diff = np.max(np.abs(b_val - prev_b))
            if diff < 1e-12 * max(1.0, prev_b.sum()):
                b_solution = b_val
                break
        prev_b = b_val.copy()
        b_solution = b_val

        if cp_iter >= 3:
            ab = fast_conv(a, b_val)
            violations = ab - M
            max_viol = float(np.max(violations))
            if max_viol < tol * max(abs(M), 1e-10):
                break

            k_worst = int(np.argmax(violations))
            if k_worst not in active:
                active.append(k_worst)
                active.sort()

    if b_solution is None or b_solution.sum() < 1e-15:
        return None
    return b_solution


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    T = build_full_constraint_matrix(a)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    restart_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response_cp(y, M_curr, T, solver)
        if b is None:
            stall_count += 1
            if stall_count > 5:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        g = b * norm / b_sum
        n_iters += 1

        ratio = M_curr / max(best_score, 1e-12)
        alpha = 2.0 / (t + 2.0)
        alpha = min(alpha * min(ratio, 1.5) + 0.01, 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
            stall_count = 0

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
            stall_count = 0

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()
                    stall_count = 0

        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count >= 3:
                if restart_count < 8:
                    restart_count += 1
                    rng = np.random.default_rng(hash((t, n)) % 2**31)
                    kick = 0.05 * rng.standard_normal(n)
                    y_kicked = y + kick
                    y_kicked = np.maximum(y_kicked, 0.0)
                    s_k = y_kicked.sum()
                    if s_k > 1e-10:
                        y = y_kicked * norm / s_k
                    a = y.copy()
                    avg_a = np.zeros(n)
                    n_avg = 0
                    stall_count = 0
                    last_improve = time.monotonic()
                    if verbose:
                        print(f"    Restart {restart_count} at iter={n_iters}, best={best_score:.8f}")
                else:
                    if verbose:
                        print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                    break
        else:
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_objective(x, n, T_temp):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    shifted = T_temp * (b - max_b)
    shifted = np.clip(shifted, -500, 0)
    lse = max_b + np.log(np.exp(shifted).sum()) / T_temp
    obj = 2.0 * n * lse / (s * s)
    return obj


def lse_gradient(x, n, T_temp):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    shifted = T_temp * (b - max_b)
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    lse = max_b + np.log(w_sum) / T_temp

    a_padded = np.zeros(2 * n - 1)
    a_padded[:n] = a
    w_padded = w
    cross = fast_conv(a_padded, w_padded)
    conv_grad = 2.0 * cross[:n]

    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    mask = x < 0
    grad[mask] = 0.0
    return grad


def lse_scipy_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=5):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        x0 = a.copy()

        def obj_func(x):
            return lse_objective(x, n, T)

        def grad_func(x):
            return lse_gradient(x, n, T)

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                 jac=grad_func,
                                 options={'maxiter': 300, 'maxfun': 800,
                                          'time_limit': min(remaining / max(1, n_stages - si), 150.0)})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def coord_descent_polish(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:10]

        improved = False
        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:6]]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if js2 <= je2:
                    peak_mask[js2:je2 + 1] = True

            for src in top_contrib_idx:
                if a[src] < 1e-12:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                if len(candidates) == 0:
                    continue
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]

                for dst in candidates_sorted[:4]:
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 5 == 0:
                for _ in range(30):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break

            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = int(rng.integers(0, 5))
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 1:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 2:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 3:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.4 + 0.3 * np.sin(4 * np.pi * x) + 0.15 * np.sin(8 * np.pi * x)
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    print(f"\n=== Phase 7: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_descent_polish(a_norm, min(remaining * 0.6, 200.0), seed=123456)
        do_try_update(a_out, f"coord_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
