import numpy as np

from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    try:
        import cvxpy as cp
        for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                x = cp.Variable(4, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])

def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    """
    Compute gradient of LSE surrogate: T*log(sum_k exp(b_k/T)) where b = autocorr(a).
    Uses FFT for efficiency. Returns gradient w.r.t. a.
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2


def nesterov_lse_optimize(a0, time_limit, T_start=10.0, T_end=0.3, n_stages=12, verbose=True):
    """Nesterov-accelerated LSE gradient descent with geometric cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            if it == 0:
                grad = lse_gradient(y, T)
                y_new = y - (1.0 / 3.0) * grad / max(grad.max(), 1e-15)
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s_ex = y_extrap.sum()
                if s_ex < 1e-15:
                    y_extrap = y.copy()
                else:
                    y_extrap = y_extrap * norm / s_ex
                grad = lse_gradient(y_extrap, T)
                y_new = y_extrap - (1.0 / (it + 3.0)) * grad / max(grad.max(), 1e-15)

            y_prev = y.copy()

            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y = y_new
            it_count += 1

            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if it_count % 100 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"    LSE stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, verbose=True):
    """Direct subgradient peak shaving: minimize the actual max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    it_count = 0

    eta = 0.01
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        j_star = int(np.argmax(b))

        # Gradient of A_{j_star} w.r.t. a[m]
        # dA_{j*}/da[m] = 2*a[j*-m] if 0 <= j*-m < n, else 0
        # (with factor 1 instead of 2 when j* is even and m = j*/2)
        grad = np.zeros(n)
        m_min = max(0, j_star - (n - 1))
        m_max = min(n - 1, j_star)
        for m in range(m_min, m_max + 1):
            grad[m] = 2.0 * a[j_star - m]

        # Normalize gradient
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        grad = grad / grad_max

        # Take step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score - 1e-12:
            a = a_new
            best_score = score
            best_a = a.copy()
            stall = 0
            it_count += 1
        else:
            stall += 1
            eta *= 0.95
            if eta < 1e-6:
                break

        if stall > 500:
            # Reset with a random perturbation
            if it_count > 10:
                break
            eta = 0.01
            idx = np.random.randint(n)
            a[idx] *= 0.95
            a[(idx + n // 3) % n] *= 1.05
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
            stall = 0

        if it_count % 200 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    PeakShave it={it_count} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.6f}")

    return best_a, best_score


def lp_best_response(a, M):
    """Solve LP: max sum(b) s.t. (a*b)_j <= M for all j, b >= 0."""
    solver = get_solver()
    if solver is None:
        return None
    import cvxpy as cp
    from scipy.sparse import coo_matrix

    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for j in range(2 * n - 1):
        for k in range(n):
            i = j - k
            if 0 <= i < n and a[i] > 1e-15:
                rows.append(j)
                cols.append(k)
                vals.append(a[i])

    if len(rows) == 0:
        return None

    T = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_optimize(a0, time_limit, verbose=True):
    """LP best-response with averaging."""
    n = len(a0)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr)
        if b is None:

        else:

            best_a = y.copy()

        if it_count % 50 == 0:
            elapsed = time.monotonic() - t0

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_asymmetric_window(n, seed=0):
    """Create an asymmetric window initialization."""
    rng = np.random.default_rng(seed)
    choice = seed % 5
    x = np.linspace(0.0, 1.0, n)

    if choice == 0:
        # Skewed raised cosine
        shift = rng.uniform(-0.15, 0.15)
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - shift))) ** 2
    elif choice == 1:
        # Asymmetric exponential
        alpha = rng.uniform(1.5, 4.0)
        beta = rng.uniform(1.5, 4.0)
        a = np.exp(-alpha * x) + np.exp(-beta * (1 - x))
    elif choice == 2:
        # Power window with asymmetry
        p1 = rng.uniform(1.0, 3.0)
        p2 = rng.uniform(1.0, 3.0)
        center = rng.uniform(0.35, 0.65)
        a = np.power(np.maximum(x / center, 0), p1) * np.power(np.maximum((1 - x) / (1 - center), 0), p2)
    elif choice == 3:
        # Multi-bump asymmetric
        centers = [rng.uniform(0.2, 0.4), rng.uniform(0.6, 0.8)]
        widths = [rng.uniform(0.08, 0.2), rng.uniform(0.08, 0.2)]
        amps = [rng.uniform(0.7, 1.0), rng.uniform(0.7, 1.0)]
        a = np.zeros(n)
        for c, w, amp in zip(centers, widths, amps):
            a += amp * np.exp(-0.5 * ((x - c) / w) ** 2)
    else:
        # Sine with phase shift (asymmetric)
        phase = rng.uniform(-np.pi / 4, np.pi / 4)
        a = np.sin(np.pi * x + phase) ** 2
        a = np.maximum(a, 0)

    # Ensure zero at edges for better initialization
    a[:max(1, n // 200)] = 0
    a[-max(1, n // 200):] = 0

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        x2 = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x2) ** 2
        s = a.sum()
    return a / s

def two_scale_optimize(target_n, time_budget, verbose=True):
    """
    Two-scale optimization:
    1. Optimize at a smaller scale for speed
    2. Interpolate to target scale
    3. Refine at target scale
    """
    t0 = time.monotonic()
    best_a = None
    best_score = float('inf')

    # Small scale
    small_n = min(target_n // 4, 400)
    small_n = max(small_n, 50)
    remaining = time_budget - (time.monotonic() - t0)
    t_small = remaining * 0.3
    print(f"  Small scale: n={small_n}, budget={t_small:.0f}s")

    for seed in range(5):
        if time.monotonic() - t0 > time_budget:
            break
        a_init = make_asymmetric_window(small_n, seed)
        t_rem = time_budget - (time.monotonic() - t0)
        t_alloc = max(5.0, t_small / max(1, 5 - seed))
        a_out, sc = nesterov_lse_optimize(a_init, min(t_alloc, t_rem), T_start=5.0, T_end=0.2, n_stages=8, verbose=False)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                print(f"    Small seed={seed}: score={sc:.8f}")

    if best_a is None:
        return make_asymmetric_window(target_n, 0), float('inf')

    # Interpolate to target
    a_target = interpolate_sequence(best_a, target_n)
    if verbose:
        print(f"  Interpolated to n={target_n}")

    return a_target, best_score

        a_out, _ = nesterov_lse_optimize(a_work, t5, T_start=10.0, T_end=0.3, n_stages=10, verbose=True)
        try_update(a_out, f"lse_n{n_long}")

    # ============ Phase 6: Final peak shaving polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final Peak Shaving n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 150.0)
    if t6 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shaving(a_work, t6, verbose=True)
        try_update(a_out, "peak_polish")

    # ============ Phase 7: Final LSE low-T polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: LSE low-T polish n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 100.0)
    if t7 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lse_optimize(a_work, t7, T_start=2.0, T_end=0.05, n_stages=6, verbose=True)
        try_update(a_out, "lse_lowT")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
