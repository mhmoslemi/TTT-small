"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage LogSumExp smoothed gradient descent with Adam + cosine LR schedule
2. Spectral flattening regularizer
3. LP-based best response (Nesterov game) for global moves
4. Peak shaving and coordinate descent polish
5. Multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lse_objective_and_grad(a, T, topk=None):
    """
    Compute LogSumExp smoothed max of autocorrelation and its gradient.
    
    Returns (lse_value, gradient_wrt_a)
    """
    n = len(a)
    c = fast_autocorr(a)
    if topk is not None and topk < len(c):
        # Use top-k for computational savings
        top_idx = np.argsort(c)[::-1][:topk]
        top_vals = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        top_vals = c
    
    max_val = top_vals.max()
    exp_vals = np.exp(top_vals - max_val)
    lse_val = max_val + np.log(exp_vals.sum())
    w = exp_vals / exp_vals.sum()
    
    # Gradient: d/d a_j [sum_k w_k * c_k] where c_k = sum_i a_i * a_{k-i}
    # dc_k / da_j = a_{k-j} + a_{j-k} (for valid indices)
    # So grad_a[j] = sum_k w_k * (a_{k-j} + a_{j-k})
    grad_a = np.zeros(n)
    for idx in range(len(top_idx)):
        K = top_idx[idx]
        wk = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += wk * a[K - idx_range]
            # Also the symmetric part
            symmetric_idx = K - idx_range
            valid = (symmetric_idx >= 0) & (symmetric_idx < n)
            if valid.any():
                grad_a[symmetric_idx[valid]] += wk * a[idx_range[valid]]
    
    return lse_val, grad_a


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=40, topk=30, lr_base=0.005):
    """
    Multi-stage Adam LSE optimization with cosine annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    # Work in log-space for non-negativity
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Cosine learning rate schedule within each stage
        base_lr = lr_base * min(1.0, 80.0 / n)
        max_iter = max(30, int(min(remaining, 30.0) * 8))
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        
        for it in range(max_iter):
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            a = softmax_fn(z)
            obj, grad_a = lse_objective_and_grad(a, stage_T, topk)
            
            # Transform gradient to z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Cosine annealing learning rate
            progress = it / max(1, max_iter - 1)
            lr = base_lr * 0.5 * (1 + np.cos(np.pi * progress))
            lr = max(lr, base_lr * 0.01)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 8 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
        
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
    
    return best_a, best_score


def spectral_flatness_optimize(a0, time_limit, lam=0.1):

    The idea: minimize the variance of |FFT(a)|^2 to make autocorrelation flat.
    """

    v_z = np.zeros(n)
    t_adam = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    max_iter = int(time_limit * 10)
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        a = softmax_fn(z)
        # Compute power spectrum
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Spectral flatness: minimize variance of power spectrum
        mean_power = power.mean()
        if mean_power < 1e-15:
            break
        normalized_power = power / mean_power
        
        # Objective: entropy of normalized power (maximize = flatten)
        # Equivalent to minimizing sum(p * log(p)) where p = normalized_power / sum
        p = normalized_power / normalized_power.sum()
        p = np.maximum(p, 1e-30)
        entropy = -np.sum(p * np.log(p))
        
        # Gradient w.r.t. a: use finite differences or analytical
        # For simplicity, use the LSE gradient as a proxy
        obj_lse, grad_a = lse_objective_and_grad(a, 5.0, topk=20)
        
        # Combine: minimize LSE but also encourage flatness
        # The gradient of -entropy wrt a is complex, so we use a simpler approach:
        # penalize positions that contribute to high power spectral peaks
        fft_peaks = np.where(power > 2 * mean_power)[0]
        if len(fft_peaks) > 0:
            # These frequency peaks correspond to time-domain structure
            # We want to reduce them by flattening a
            penalty_grad = np.zeros(n)
            for freq in fft_peaks[:5]:
                # The contribution to this frequency is sum_j a_j * cos(freq * j * 2π/n)
                phase = 2 * np.pi * freq * np.arange(n) / n
                contrib = a * np.cos(phase)
                penalty_grad += 0.1 * contrib
            grad_a = grad_a + penalty_grad
        
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        t_adam += 1
        m_z = beta1 * m_z + (1 - beta1) * grad_z
        v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
        m_hat = m_z / (1 - beta1 ** t_adam)
        v_hat = v_z / (1 - beta2 ** t_adam)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if (it + 1) % 10 == 0:
            a_check = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a_check).max()
            if score < best_score:
                best_score = score
                best_a = a_check.copy()
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """
    Systematic peak shaving with more aggressive search.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        
        # Find all peaks above 99.5% of max
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]
        
        improved = False
        for k in top_peaks[:12]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find safe destinations: positions that don't contribute much to any top peak
            danger = np.zeros(n, dtype=bool)
            for kp in top_peaks[:15]:
                jp_s = max(0, kp - n + 1)
                jp_e = min(n - 1, kp)
                danger[jp_s:jp_e + 1] = True
            
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 3)]
            if len(dst_pool) == 0:
                continue
            
            found = False
            for dst in dst_pool[:10]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 300:
                break
    
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):
    """
    Nesterov-accelerated LP best-response with adaptive restart.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_long = np.zeros(n)
    n_long = 0
    stall_count = 0
    warm_b = None
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        T = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T, solver, warm_b)
        if b_star is None:
            warm_b = None
            continue
        warm_b = b_star
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        avg_long += a
        n_long += 1
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if n_long % 20 == 0 and n_long >= 40:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0
        
        if stall_count > 400 and t > 100:
            a = best_a.copy()
            y = best_a.copy()

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def coordinate_descent(a0, time_limit, seed=123):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    coord_order = rng.permutation(n)
    
    while time.monotonic() - t0 < time_limit - 2:
        improved = False
        for j in coord_order:
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            # Current autocorrelation
            b = fast_autocorr(a)
            bmax = b.max()
            
            # Try adjusting a[j]
            # The contribution of a[j] to b[k] is a[j] * (a[k-j] + a[j-k]) for valid indices
            # We want to find a[j] that minimizes max_k b[k]
            
            # Binary search on a[j]
            a_j_old = a[j]
            lo = 0.0
            hi = a_j_old * 3 + 1e-6
            
            best_val = a_j_old
            best_bmax = bmax
            
            for _ in range(12):
                mid = (lo + hi) / 2
                a_trial = a.copy()
                a_trial[j] = mid
                new_b = fast_autocorr(a_trial)
                new_bmax = new_b.max()
                
                if new_bmax < best_bmax:
                    best_bmax = new_bmax
                    best_val = mid
                    lo = mid
                else:
                    hi = mid
            
            if best_val != a_j_old:
                a[j] = best_val
                if best_bmax < bmax * (1 - 1e-10):
                    improved = True
                    st = a.sum()
                    if st > 1e-12:
                        a = a / st
                        score = 2.0 * n * fast_autocorr(a).max()
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP zero-sum game (primary global optimizer)
2. Multi-stage LogSumExp smoothed Adam gradient descent with cosine LR
3. Peak shaving with danger-zone awareness
4. Coordinate descent for fine polish
5. Multi-length exploration with interpolation
6. Recombination of best candidates from different methods
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lse_grad_of_autocorr(a, T, topk=None):
    """
    Compute LogSumExp smoothed max of autocorrelation and its gradient w.r.t. a.
    Returns (lse_value, gradient).
    """
    n = len(a)
    c = fast_autocorr(a)
    if topk is not None and topk < len(c):
        top_idx = np.argsort(c)[::-1][:topk]
        top_vals = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        top_vals = c

    max_val = top_vals.max()
    exp_vals = np.exp(top_vals - max_val)
    lse_val = max_val + np.log(exp_vals.sum())
    w = exp_vals / exp_vals.sum()

    # Gradient: dc_k/da_j = a[k-j] + a[j-k] (for valid indices in full convolution)
    # Full convolution: c[k] = sum_i a[i] * a[k-i] where 0<=i<n, 0<=k-i<n
    # dc_k/da_j = a[k-j] (when k-j is in [0,n-1])
    grad_a = np.zeros(n)
    for idx in range(len(top_idx)):
        K = top_idx[idx]
        wk = w[idx]
        # c[K] = sum_j a[j] * a[K-j] for j in [max(0,K-n+1), min(n-1,K)]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += wk * a[K - idx_range]

    return lse_val, grad_a


def adam_lse_optimize(a0, time_limit, T_start=60.0, T_end=0.1, n_stages=35, topk=35, lr_base=0.004):
    """
    Multi-stage Adam LSE optimization with cosine annealing learning rate.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        base_lr = lr_base * min(1.0, 80.0 / n)
        max_iter = max(20, int(min(remaining, 40.0) * 6))

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        for it in range(max_iter):
            if time.monotonic() - t0 > time_limit - 2:
                break

            a = softmax_fn(z)
            obj, grad_a = lse_grad_of_autocorr(a, stage_T, topk)

            # Transform to z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            # Cosine annealing
            progress = it / max(1, max_iter - 1)
            lr = base_lr * 0.5 * (1 + np.cos(np.pi * progress))
            lr = max(lr, base_lr * 0.01)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

    return best_a, best_score

    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_long = np.zeros(n)
    n_long = 0
    WINDOW_MED = 40
    med_buf = [None] * WINDOW_MED
    avg_med = np.zeros(n)
    n_med = 0
    med_idx = 0
    stall_count = 0
    warm_b = None

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        T = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T, solver, warm_b)
        if b_star is None:
            warm_b = None
            continue
        warm_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        avg_long += a
        n_long += 1
        if med_buf[med_idx] is not None:
            avg_med -= med_buf[med_idx]
        med_buf[med_idx] = a.copy()
        avg_med += a
        n_med += 1
        med_idx = (med_idx + 1) % WINDOW_MED

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if n_long % 20 == 0 and n_long >= 40:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if n_med >= WINDOW_MED and n_long % 20 == 0:
            avg_m = avg_med / WINDOW_MED
            score_m = 2.0 * n * fast_autocorr(avg_m).max()
            if score_m < best_score:
                best_score = score_m
                best_a = avg_m.copy()
                stall_count = 0

        if stall_count > 400 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            avg_med = np.zeros(n)
            n_med = 0
            med_buf = [None] * WINDOW_MED
            med_idx = 0
            stall_count = 0
            warm_b = None

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """
    Systematic peak shaving: for each top peak, find the most contributing
    pair and try to reduce it while checking global peak doesn't increase.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        danger = np.zeros(n, dtype=bool)
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            danger[js:je + 1] = True

        improved = False
        for k in top_peaks[:12]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 3)]
            if len(dst_pool) == 0:
                continue

            found = False
            for dst in dst_pool[:10]:
                for frac in [0.003, 0.007, 0.015, 0.03, 0.05, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 15 == 0:
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 250:
                break

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=123):
    """
    Coordinate descent: for each coordinate j, find optimal a[j] via
    ternary search (the objective is convex in a[j] for fixed others).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    coord_order = rng.permutation(n)
    no_improve_rounds = 0

    while time.monotonic() - t0 < time_limit - 2:
        round_improved = False
        for j in coord_order:
            if time.monotonic() - t0 > time_limit - 2:
                break

            b = fast_autocorr(a)
            bmax = b.max()

            # Ternary search on a[j] in [0, 3*a[j] + 0.01]
            a_j = a[j]
            lo = 0.0
            hi = max(a_j * 3.0, 0.01)

            # If current value is already at boundary, expand
            if a_j < 1e-12:
                lo = 0.0
                hi = 0.01
            elif a_j > hi * 0.9:
                hi = a_j * 3.0

            best_val = a_j
            best_bmax = bmax

            for _ in range(10):
                if hi - lo < 1e-10:
                    break
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                a[m[j]] = m1
                b1 = fast_autocorr(a).max()
                a[m[j]] = m2
                b2 = fast_autocorr(a).max()

                if b1 < b2:
                    hi = m2
                else:
                    lo = m1

                if b1 < best_bmax:
                    best_bmax = b1
                    best_val = m1
                if b2 < best_bmax:
                    best_bmax = b2
                    best_val = m2

            a[j] = best_val
            if best_bmax < bmax * (1 - 1e-10):
                round_improved = True
                st = a.sum()
                if st > 1e-12:
                    a = a / st
                    score = 2.0 * n * fast_autocorr(a).max()
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()

        if not round_improved:
            no_improve_rounds += 1
            if no_improve_rounds >= 2:
                break
        else:
            no_improve_rounds = 0
            coord_order = rng.permutation(n)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def recombine(candidates, time_limit):
    """Try convex combinations of candidate sequences."""
    if len(candidates) < 2:
        return None, float('inf')

    best_combo = None
    best_score = float('inf')
    t0 = time.monotonic()

    n = len(candidates[0])
    k = len(candidates)

    # Try all pairs with various weights
    for i in range(k):
        for j in range(i + 1, k):
            if time.monotonic() - t0 > time_limit:
                return best_combo, best_score
            for alpha in [0.3, 0.4, 0.5, 0.6, 0.7]:
                combo = alpha * candidates[i] + (1 - alpha) * candidates[j]
                combo = np.maximum(combo, 0)
                s = combo.sum()
                if s < 1e-12:
                    continue
                combo = combo / s
                score = 2.0 * n * fast_autocorr(combo).max()
                if score < best_score:
                    best_score = score
                    best_combo = combo.copy()

    # Try uniform mixture of all
    if k >= 3:
        for alpha_set in [(0.4, 0.3, 0.3), (0.5, 0.25, 0.25), (0.6, 0.2, 0.2)]:
            if len(alpha_set) <= k:
                combo = np.zeros(n)
                for idx in range(len(alpha_set)):
                    combo += alpha_set[idx] * candidates[idx]
                combo = np.maximum(combo, 0)
                s = combo.sum()
                if s > 1e-12:
                    combo = combo / s
                    score = 2.0 * n * fast_autocorr(combo).max()
                    if score < best_score:
                        best_score = score
                        best_combo = combo.copy()

    return best_combo, best_score


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining LP game, LSE gradient, peak shaving,
    coordinate descent, recombination, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 100.0, 1700.0)

    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    candidates = []

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            candidates.append(parent.copy())
    except Exception:
        parent = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            candidates.append(a.copy())
            if len(candidates) > 8:
                candidates.pop(0)
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    # === Phase 1: LP Nesterov at parent length (strongest global method) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.38, 550.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: LSE gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: LSE grad n={n_base} ===")
    t2 = min(remaining * 0.22, 300.0)
    if t2 > 30:
        a_out, _ = adam_lse_optimize(best_seq, t2, T_start=80.0, T_end=0.05, n_stages=35, topk=40, lr_base=0.003)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: Peak shaving n={n_base} ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 15:
        a_out, _ = peak_shaving(best_seq, t3, seed=777)
        try_update(a_out, "peak_shave_1")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: Coord descent n={n_base} ===")
    t4 = min(remaining * 0.08, 80.0)
    if t4 > 10:
        a_out, _ = coordinate_descent(best_seq, t4, seed=456)
        try_update(a_out, "coord_1")

    # === Phase 5: Longer length LP ===
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 5: LP Nesterov n={n_long} ===")
    t5 = min(remaining * 0.10, 130.0)
    if t5 > 40:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t5, solver)
        try_update(a_out, f"lp_n{n_long}")

    # === Phase 6: Even longer LSE ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_long2 = n_base + 350
        t6 = min(remaining * 0.07, 90.0)
        if t6 > 20:
            print(f"=== Phase 6: LSE n={n_long2} ===")
            init_long2 = interpolate_sequence(best_seq, n_long2)
            a_out, _ = adam_lse_optimize(init_long2, t6, T_start=50.0, T_end=0.5, n_stages=20, topk=30, lr_base=0.003)
            try_update(a_out, f"lse_n{n_long2}")

    # === Phase 7: Recombination ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10 and len(candidates) >= 2:
        print(f"=== Phase 7: Recombination ===")
        # Only recombine candidates of the same length
        same_len_candidates = [c for c in candidates if len(c) == n_base]
        if len(same_len_candidates) >= 2:
            combo, combo_score = recombine(same_len_candidates, min(remaining * 0.03, 20.0))
            if combo is not None:
                try_update(combo, "recombine")

    # === Phase 8: Final LSE polish (low temperature) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.12, 60.0)
        if t8 > 10:
            print(f"=== Phase 8: Final LSE polish ===")
            a_out, _ = adam_lse_optimize(best_seq, t8, T_start=5.0, T_end=0.03, n_stages=12, topk=8, lr_base=0.001)
            try_update(a_out, "final_lse")

    # === Phase 9: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t9 = min(remaining * 0.30, 50.0)
        if t9 > 5:
            print(f"=== Phase 9: Final peak shave ===")
            a_out, _ = peak_shaving(best_seq, t9, seed=1001)
            try_update(a_out, "peak_shave_final")

    # === Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 5:
        t10 = min(remaining * 0.20, 30.0)
        if t10 > 3:
            print(f"=== Phase 10: Final coord descent ===")
            a_out, _ = coordinate_descent(best_seq, t10, seed=999)
            try_update(a_out, "final_coord")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
