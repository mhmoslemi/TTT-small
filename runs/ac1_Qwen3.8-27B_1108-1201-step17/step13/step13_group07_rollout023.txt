import numpy as np

def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * float(fast_autocorr(a).max())


def evaluate_safe(arr):

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Water-filling algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    # Fix numerical issues
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 0:
        result = result * target_sum / s
    else:
        result = np.full(n, target_sum / n)
    return result


def build_conv_matrix_sparse(a_fixed):
    """Build sparse (2n-1) x n matrix T where T @ b = a_fixed * b (linear conv)."""
    n = len(a_fixed)
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    # For column j: T[j:j+n, j] = a_fixed
    # Use sparse diagonal structure
    rows = []
    cols = []
    vals = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        vals.extend(a_fixed)
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, max_iter=500):
    """Solve: min M s.t. (a_fixed*b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=max_iter)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=max_iter)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_iter_lp=500):
    """Nesterov-accelerated LP best-response for the zero-sum game."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver, max_iter_lp)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  NL t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score

        if time.monotonic() - t0 > time_limit:

        n_iters = max(50, int((time_limit - (time.monotonic() - t0)) * 5 / max(1, len(T_schedule) - si)))

        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            b = fast_autocorr(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
            # This is 2 * (w * a_reversed) convolved, taking the middle n entries
            fft_size = 1 << (2 * n - 1 - 1 + n - 1).bit_length()
            # Actually, conv of w (length 2n-1) with a_reversed (length n) has length 3n-2
            # We want entries [n-1 : 2n-2] of this conv
            a_rev = a[::-1]
            W = rfft(w, fft_size)
            A_rev = rfft(a_rev, fft_size)
            conv = irfft(W * A_rev, fft_size)
            grad = 2.0 * conv[n-1:2*n-1]

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = project_simplex(a_new)

            score = internal_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent / peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find the peak
        k_peak = np.argmax(b)

        # Find which pair contributes most to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        src = idx[np.argmax(contrib)]

        # Try moving mass from src to other positions
        # Choose a random target
        improved = False
        for _ in range(20):
            dst = rng.integers(n)
            if dst == src:
                continue
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.001, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                if a_trial.sum() > 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score + 0.001:  # Allow slight increase for exploration
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                        continue
            stall += 1
            if stall > 200:
                break

        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD ({elapsed:.0f}s): best={best_score:.6f}, stall={stall}")

    return best_a, best_score


def spectral_shaping(a0, n_iter=300, alpha=0.3):
    """Spectral shaping to flatten the power spectrum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s

    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        # Target: flat spectrum (exclude DC)
        non_dc = B[1:]
        target = np.median(non_dc)

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a

def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def internal_score(a):

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} using water-filling."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s > 1e-15:
        result = result * target_sum / s
    else:
        result = np.full(n, target_sum / n)

def build_conv_T(a_fixed):
    """Build (2n-1) x n sparse matrix T where T @ b = a_fixed * b (linear conv)."""
    n = len(a_fixed)
    rows_parts = []
    cols_parts = []
    data_parts = []
    for j in range(n):
        rows_parts.append(np.arange(j, j + n))
        cols_parts.append(np.full(n, j))
        data_parts.append(a_fixed)
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    data = np.concatenate(data_parts)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


class SparseConvT:
    """Sparse convolution matrix with in-place data updates."""

    def __init__(self, a):
        self.n = len(a)
        self.rows, self.cols = self._build_indices()
        self.data = np.concatenate([a] * self.n)
        self.T = None
        self.update(a)

    def _build_indices(self):
        n = self.n
        rows_parts = []
        cols_parts = []
        for j in range(n):
            rows_parts.append(np.arange(j, j + n))
            cols_parts.append(np.full(n, j))
        return np.concatenate(rows_parts), np.concatenate(cols_parts)

    def update(self, a):
        a = np.asarray(a, dtype=np.float64)
        self.data[:] = np.concatenate([a] * self.n)
        self.T = csr_matrix((self.data, (self.rows, self.cols)),
                            shape=(2 * self.n - 1, self.n))

        return self.T

    b_val = np.asarray(b_var.value, dtype=np.float64)

        if time.monotonic() - t0 > time_limit:
            break

        # Periodic restart with small noise to escape local minima
        if t - last_restart >= restart_interval and t > 100:
            noise = np.random.default_rng(t).normal(0, 0.01, n)
            y_perturbed = y + noise
            y_perturbed = np.maximum(y_perturbed, 0)
            sp = y_perturbed.sum()
            if sp > 1e-15:
                y = y_perturbed / sp
            last_restart = t

        b_star, M_val = solve_lp_best_response(y, conv_T, solver, max_iter_lp)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        # Track best
        score_a = internal_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Time averaging
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def lse_optimize(a0, time_limit, lr=0.01, verbose=True):
    """LogSumExp-smoothed gradient optimization with Adam on simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    T_schedule = np.geomspace(80.0, 0.05, 40)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break

        remaining_frac = (time_limit - (time.monotonic() - t0)) / max(1, time_limit)
        n_iters = max(30, int(200 * remaining_frac))

        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            b = fast_autocorr(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient computation via FFT
            fft_size = 1 << (3 * n - 2 - 1).bit_length()
            a_rev = a[::-1]
            W = rfft(w, fft_size)
            A_rev = rfft(a_rev, fft_size)
            conv_wa = irfft(W * A_rev, fft_size)
            grad = 2.0 * conv_wa[n - 1:2 * n - 1]

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = project_simplex(a_new)

            score = internal_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and si % 8 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent peak shaving with adaptive step sizes."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        b = fast_autocorr(a)

        # Top 5 peaks
        top_k = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Try multiple targets
            for trial in range(10):
                dst = rng.integers(n)
                if dst == src:
                    continue
                for frac in [0.002, 0.01, 0.03, 0.08, 0.15]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = internal_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = float(rng.uniform(0.001, 0.03))
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        continue
            stall += 1
            if stall > 300:
                break

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iter_count} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def spectral_shaping(a0, n_iter=400, alpha=0.25):
    """Spectral shaping to flatten power spectrum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s

    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        if np.any(non_dc <= 0):
            break
        target = np.median(non_dc)
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimizer: Nesterov LP (main) + LSE gradient + coordinate descent.
    Uses time-averaged iterates, periodic restarts, and adaptive time allocation.
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 50.0, 1720.0)

    solver = get_lp_solver()
    print(f"Solver: {solver}, budget: {time_budget:.0f}s")

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
        else:
            raise ValueError("Invalid parent")
    except Exception:
        n = 1600
        best_seq = np.ones(n) / n
        best_score = internal_score(best_seq)
        print(f"Uniform init: n={n}, score={best_score:.6f}")

    n = len(best_seq)

    # Phase 1: Quick spectral shaping (10s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral (rem={remaining:.0f}s) ===")
    a_spec = spectral_shaping(best_seq, n_iter=300, alpha=0.2)
    sc = internal_score(a_spec)
    if sc < best_score:
        best_score = sc
        best_seq = a_spec.copy()
        print(f"  Spectral: {best_score:.6f}")

    # Phase 2: Nesterov LP - main workhorse (55% of time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    t2 = remaining * 0.55
    print(f"\n=== Phase 2: Nesterov LP ({t2:.0f}s budget) ===")

        best_score = sc
        best_seq = np.maximum(a_out, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"  *** LP: {best_score:.6f}")
    elif sc_out < best_score:
        best_score = sc_out
        best_seq = np.maximum(a_out, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"  LP (internal): {best_score:.6f}")

    # Phase 3: LSE optimization (15% of remaining)
    remaining = time_budget - (time.monotonic() - t_global_start)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: LSE ({t3:.0f}s budget) ===")
    a_out, sc_out = lse_optimize(best_seq, t3, lr=0.008, verbose=True)
    sc = evaluate_safe(a_out)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = np.maximum(a_out, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"  *** LSE: {best_score:.6f}")
    elif sc_out < best_score:
        best_score = sc_out
        best_seq = np.maximum(a_out, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s

    # Phase 4: Second LP round from improved point (15% of remaining)
    remaining = time_budget - (time.monotonic() - t_global_start)
    t4 = remaining * 0.15
    if t4 > 30:
        print(f"\n=== Phase 4: LP round 2 ({t4:.0f}s) ===")

            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** LP2: {best_score:.6f}")
        elif sc_out < best_score:
            best_score = sc_out
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s

    # Phase 5: Coordinate descent polish (8% of remaining)
    remaining = time_budget - (time.monotonic() - t_global_start)

    if t5 > 5:
        print(f"\n=== Phase 5: Coord descent ({t5:.0f}s) ===")
