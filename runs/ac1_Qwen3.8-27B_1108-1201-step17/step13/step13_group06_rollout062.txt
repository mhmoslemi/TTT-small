import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if sum(a_list) < 0.01:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_toeplitz_sparse(a_fixed):
    """
    Build the Toeplitz matrix structure for computing T @ b = convolve(a_fixed, b).
    Returns a function that computes T @ x efficiently.
    """
    n = len(a_fixed)
    
    def apply(x):
        # convolve(a_fixed, x) using FFT for speed
        return fast_autocorr_joint(a_fixed, x)
    
    return apply


def fast_autocorr_joint(a, b):
    """Compute convolve(a, b) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

def solve_lp_best_response(y, solver, warm_b=None):
    """
    Solve LP: min M s.t. convolve(y, b)_k <= M for all k, sum(b)=1, b>=0.
    Uses cvxpy with the Toeplitz structure.
    """
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    
    # convolve(y, b) = T @ b where T is (2n-1) x n Toeplitz
    # Instead of building T explicitly, use the convolution structure
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    T = T  # This is dense but fast to build for n=1600
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve(a, b):
    """Compute convolve(a, b) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def solve_lp_best_response(y, solver, warm_b=None):

def coordinate_descent(a0, time_limit, max_sweeps=10000):

            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            
            if src_val < 1e-12:
                continue
            
            # Find destinations that don't contribute to top peaks
            danger = np.zeros(n, dtype=bool)
            for pk in top_peaks[:10]:
                pjs = max(0, pk - n + 1)
                pje = min(n - 1, pk)
                danger[pjs:pje + 1] = True
            danger[src] = True
            
            safe = np.where(~danger)[0]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(3, len(safe_sorted) // 5)]
            
            found = False
            for dst in dst_pool[:5]:
                for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_init(n, rng):
    """
    Generate initialization based on spectral design.
    Try to create a sequence with approximately flat autocorrelation.
    """
    choice = rng.integers(0, 6)
    
    if choice == 0:
        # Raised cosine (sin^2)
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        # Half-sine (concentrated in middle)
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x / 2) ** 2
    elif choice == 2:
        # Two-bump
        x = np.linspace(0, 1, n)
        a = np.exp(-30 * (x - 0.3) ** 2) + np.exp(-30 * (x - 0.7) ** 2)
    elif choice == 3:
        # Edge-concentrated with smooth falloff
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * x ** 2) + np.exp(-20 * (1 - x) ** 2)
    elif choice == 4:
        # Broad Gaussian
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
    else:
        # Flat-top window approximation
        x = np.linspace(0, 1, n)
        a = 0.4 + 0.36 * np.cos(2 * np.pi * (x - 0.48)) + 0.15 * np.cos(4 * np.pi * (x - 0.48))
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization for minimum peak autocorrelation sequences.
    
    Phase 1: LP Game (45% time) - Nesterov-accelerated zero-sum game
    Phase 2: Subgradient refinement (15% time)
    Phase 3: Multi-length exploration (20% time)
    Phase 4: Peak shaving + coordinate descent (15% time)
    Phase 5: Final polish (5% time)
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 60.0, 1740.0)
    
    solver = find_solver()
    print(f"LP solver: {solver}, total budget: {time_budget:.0f}s")
    
    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None
    
    n_base = len(best_seq) if best_seq is not None else 1600
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: LP Game ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Game n={n_base} ===")
    t1 = min(remaining * 0.45, 750.0)
    if t1 > 60:
        # Run LP game from parent
        a_out, _ = nesterov_lp_game(best_seq, t1 * 0.7, solver)
        try_update(a_out, f"lp_main")
        
        # If time remains, run from spectral init
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 60:
            t1b = min(remaining * 0.25, 200.0)
            if t1b > 30:
                spec_init = spectral_init(n_base, rng)
                a_out2, _ = nesterov_lp_game(spec_init, t1b, solver)
                try_update(a_out2, "lp_spec")
    
    # === Phase 2: Subgradient refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: Subgradient n={len(best_seq)} ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_out, _ = subgradient_nesterov(best_seq, t2, topk=5, eta_scale=1.0)
        try_update(a_out, "subgrad_1")
        
        # Try different topk
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t2b = min(remaining * 0.10, 120.0)
            if t2b > 15:
                a_out2, _ = subgradient_nesterov(best_seq, t2b, topk=3, eta_scale=0.5)
                try_update(a_out2, "subgrad_2")
    
    # === Phase 3: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: Multi-length exploration ===")
    lengths_to_try = [n_base + 100, n_base + 200, n_base + 400, n_base + 600]
    # Remove duplicates and sort
    lengths_to_try = sorted(set(l for l in lengths_to_try if l > 100))
    
    for i, n_try in enumerate(lengths_to_try[:3]):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        t3_each = min(remaining * 0.25, 200.0)
        if t3_each < 30:
            break
        
        print(f"  Trying n={n_try} (t={t3_each:.0f}s)")
        init = interpolate_sequence(best_seq, n_try)
        # Add some randomization
        noise = rng.normal(0, 0.02, n_try)
        init = init * (1 + noise)
        init = np.maximum(init, 0)
        init = init / init.sum()
        
        a_out, _ = nesterov_lp_game(init, t3_each, solver)
        try_update(a_out, f"lp_n{n_try}")
    
    # === Phase 4: Peak shaving + coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: Peak shaving + coord descent ===")
    t4a = min(remaining * 0.30, 200.0)
    if t4a > 10:
        a_out, _ = peak_shaving(best_seq, t4a, seed=999)
        try_update(a_out, "peak_shave_1")
    
    remaining = time_budget - (time.monotonic() - t_global)
    t4b = min(remaining * 0.25, 150.0)
    if t4b > 10:
        a_out, _ = coordinate_descent(best_seq, t4b)
        try_update(a_out, "coord_desc")
    
    # === Phase 5: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 5: Final polish ===")
    if remaining > 20:
        t5a = min(remaining * 0.4, 100.0)
        if t5a > 10:
            # Short LP game from best
            a_out, _ = nesterov_lp_game(best_seq, t5a, solver)
            try_update(a_out, "lp_final")
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t5b = min(remaining * 0.4, 80.0)
        if t5b > 5:
            a_out, _ = peak_shaving(best_seq, t5b, seed=7777)
            try_update(a_out, "peak_final")
    
    # Final subgradient polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t5c = min(remaining * 0.5, 60.0)
        if t5c > 10:
            a_out, _ = subgradient_nesterov(best_seq, t5c, topk=3, eta_scale=0.3)
            try_update(a_out, "subgrad_final")
    
    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
    
    return [float(x) for x in best_seq]
