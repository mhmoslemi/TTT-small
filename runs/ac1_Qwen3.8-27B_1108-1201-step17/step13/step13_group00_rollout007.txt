            break
        remaining = time_limit - (time.monotonic() - t0)

            break

            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()
            
            # Gradient: grad_i = 2 * (w ⊛ a)[i] = 2 * sum_k w[k] * a[k-i]
            # Using cross-correlation: (w ⊛ a)[i] = sum_k w[k+i] * a[k]
            # Wait, I need: sum_k w[k] * a[k-i]
            # Let's use: cross_corr_rev[i] = sum_k w[k] * a[k-i]
            # This is the cross-correlation of a with w at lag i: (a ⊛ w)[i] = sum_k a[k] * w[k-i]
            # Hmm, let me just compute it directly via FFT.
            # 
            # sum_k w[k] * a[k-i] for k range [0, 2n-2], a[j] = 0 for j not in [0,n-1]
            # = sum_j a[j] * w[j+i] (substituting j = k-i, k = j+i)
            # = (a ⊛ w)[i] where (a ⊛ w)[i] = sum_j a[j] * w[j+i]
            #
            # This is the cross-correlation of a and w, which can be computed as:
            # ifft(FFT(a) * conj(FFT(w))) gives (a ⊛ w)[m] = sum_j a[j] * w[j+m]
            # for m = 0, 1, ... (with wrap-around if not zero-padded)
            
            W = rfft(w, fft_size)
            cross = irfft(A * np.conj(W), fft_size)
            grad = 2.0 * cross[:n]
            
            # Projected gradient step
            lr = base_lr / (1.0 + 0.1 * it)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new
            
            if si == 0 or it % 50 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  LSE T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_fast(a0, time_limit, rng, verbose=True):
    """
    Fast pair coordinate descent with gradient-informed pair selection.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    # Precompute top peaks (these are the ones that matter)
    n_top = min(50, size)
    
    passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False
        
        # Get top peak indices
        top_peaks = np.argpartition(conv, -n_top)[-n_top:]
        
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            
            if a[i] < norm * 1e-9:
                continue
            
            # Try 2 random j's
            for attempt in range(2):
                j = int(rng.integers(n))
                if j == i:
                    continue
                
                # Compute gain vector g[k] = a[k-j] + a[j-k] - a[k-i] - a[i-k]
                # (first order change in conv when moving delta from i to j)
                g = np.zeros(size)
                
                # -a[k-i] for k in [i, i+n-1]
                lo, hi = i, min(i + n, size)
                if hi > lo:
                    g[lo:hi] -= a[:hi - lo]
                
                # -a[i-k] for k in [max(0,i-n+1), i]... wait
                # a[i-k] is valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i
                lo2, hi2 = max(0, i - n + 1), min(i + 1, size)
                if hi2 > lo2:
                    k_range = np.arange(lo2, hi2)
                    a_idx = i - k_range
                    g[lo2:hi2] -= a[a_idx]
                
                # +a[k-j] for k in [j, j+n-1]
                lo3, hi3 = j, min(j + n, size)
                if hi3 > lo3:
                    g[lo3:hi3] += a[:hi3 - lo3]
                
                # +a[j-k] for k in [max(0,j-n+1), j]
                lo4, hi4 = max(0, j - n + 1), min(j + 1, size)
                if hi4 > lo4:
                    k_range = np.arange(lo4, hi4)
                    a_idx = j - k_range
                    g[lo4:hi4] += a[a_idx]
                
                # Find optimal delta using a few samples
                ai = a[i]
                best_d = 0.0
                best_val = conv.max()
                
                # Sample delta values (geometric spacing for better coverage)
                deltas = np.geomspace(ai * 0.01, ai * 0.99, 12) if ai > 0 else np.array([0.0])
                deltas = np.concatenate([[0.0], deltas, [ai]])
                
                for d in deltas:
                    if d == 0:
                        continue
                    # New conv (first order)
                    new_conv = conv + d * g
                    # Second order corrections
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                    # Cross term at k = i+j (if valid): adds d*d... actually
                    # (ai-d)(aj+d) + (aj+d)(ai-d) at k=i+j... this is second order, skip for speed
                    
                    new_max = new_conv.max()
                    if new_max < best_val - 1e-12:
                        best_val = new_max
                        best_d = d
                
                if best_d > 1e-10 * ai:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[i] += best_d
                        a[j] -= best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"  PCD pass={passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
