def _active_set_lp(a, M, K, solver, warm_start=None):
    """Solve LP with only K most binding constraints."""
    n = len(a)
    b = np.convolve(a, a)  # full convolution
    # Find K most binding constraints (closest to M)
    idx = np.argsort(b)[::-1][:K]
    
    # Build constraint matrix for only these K constraints
    # Constraint k: sum_i a[max(0,k-n+1):min(n,k+1)] * b[k-i] <= M
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        row = np.zeros(n)
        for i in range(js, je+1):
            row[i] = a[k - i]
        constraints.append(row @ b_var <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    if warm_start is not None:
        b_var.value = warm_start
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

def _autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # First project onto sum=1, then scale
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out = out * target_sum / s
    else:
        out = np.full(n, target_sum / n)
    return out


def _active_set_lp(a, M, K, solver, warm_start=None):
    """Solve LP with only K most binding constraints."""
    n = len(a)
    b_ac = _autocorr(a)
    
    # Find K most binding constraints
    if K < len(b_ac):
        # Use the top-K by autocorrelation value as proxy for binding constraints
        idx = np.argsort(b_ac)[::-1][:K]
    else:
        idx = np.arange(len(b_ac))
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint rows
    rows = []
    for k in idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        row = np.zeros(n)
        for i in range(js, je + 1):
            row[i] = a[k - i]
        rows.append(row)
    
    if not rows:
        return None
    
    A_mat = np.array(rows)
    constraints = [A_mat @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    if warm_start is not None:
        b_var.value = warm_start
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def _full_lp(a, M, solver, warm_start=None):
    """Solve the full LP with all constraints."""
    n = len(a)
    # Build the full Toeplitz constraint matrix
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            T[k, i] = a[k - i]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    if warm_start is not None:
        b_var.value = warm_start
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def _subgradient_nesterov(a0, time_limit, verbose=True):
    """Subgradient method with Nesterov acceleration (no LP needed)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = _autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = _autocorr(y)
        k_star = int(np.argmax(ac))
        M = ac[k_star]
        
        # Subgradient at k_star
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            grad[i] = 2.0 * y[k_star - i]
        
        # Find best vertex (Frank-Wolfe)
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new * norm / s_y
        else:
            y_new = y.copy()
        
        if t == 1:
            a_new = v.copy()
        else:
            mom = (t - 1.0) / (t + 2.0)
            a_new = y_new + mom * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a > 1e-15:
                a_new = a_new * norm / s_a
            else:
                a_new = y_new.copy()
        
        y = y_new
        a = a_new
        
        # Check improvement
        score_a = _autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Polyak average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 50000 == 0:
            print(f"  Subgrad t={t} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

                _solver = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.CLARABEL
    print(f"  Solver (fallback): CLARABEL")
    return _solver


def _autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v, target=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 1:
        return np.array([target])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out = out * target / s
    else:
        out = np.full(n, target / n)
    return out


def _lse_grad(y, T):
    n = len(y)
    ac = _autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def _lse_fista(a_init, time_limit, T_schedule=None, verbose=True):
    """LSE-smoothed FISTA with backtracking line search."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if T_schedule is None:
        T_schedule = [50.0, 10.0, 3.0, 1.0, 0.4, 0.15, 0.05, 0.02]
    
    best_a = a.copy()
    best_score = float(_autocorr(a).max())
    t0 = time.monotonic()
    
    for stage_i, T in enumerate(T_schedule):
        y = a.copy()
        z = a.copy()
        t_fista = 1
        L = 4.0 / T  # Initial Lipschitz estimate
        
        # Estimate L more carefully
        # The gradient is 2 * cross_corr(y, w) where w is softmax(ac/T)
        # The operator norm is bounded by 2 * ||w||_2 * ||y||_2 * 2 <= 8
        # But let's use backtracking
        
        while time.monotonic() - t0 < time_limit:
            grad = _lse_grad(y, T)
            
            # Backtracking line search
            step = 1.0 / L
            for _ in range(20):
                z_new = y - step * grad
                a_new = _proj_simplex(z_new)
                
                # Check sufficient decrease (using LSE objective)
                ac_new = _autocorr(a_new)
                lse_new = T * np.log(np.sum(np.exp((ac_new - ac_new.max()) / T))) + ac_new.max()
                ac_y = _autocorr(y)
                lse_y = T * np.log(np.sum(np.exp((ac_y - ac_y.max()) / T))) + ac_y.max()
                
                pred = lse_y + grad @ (a_new - y) + 0.5 / step * np.sum((a_new - y) ** 2)
                
                if lse_new <= pred + 1e-10:
                    break
                step *= 0.5
                L *= 2
            
            # Nesterov extrapolation
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - z)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()
            
            z = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            
            # Check every 100 iterations
            if t_fista % 100 == 1:
                sc = float(_autocorr(a).max())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    if verbose:
                        print(f"    FISTA stage={stage_i} T={T:.2f} t={t_fista} ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
            
            # Time check
            if time.monotonic() - t0 > time_limit:
                break
        
        sc = float(_autocorr(a).max())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score
