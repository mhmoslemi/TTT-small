        if verbose and iters % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSG iter={iters} ({elapsed:.1f}s): M={best_M:.6f} score={best_M:.6f}")
    
    return best_a, best_M


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve(a, M, solver, top_k_constraints=400):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0.
    Uses top-K constraints for speed."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    
    # Select top-K constraints (those with largest M values... wait, M is a scalar)
    # Actually, all constraints are (a*b)[k] <= M for the same M.
    # The "most important" constraints are those that will be active.
    # Heuristic: use constraints where the diagonal contribution is largest.
    # (a*b)[k] includes a[j]*b[k-j], so constraints where a has large values matter most.
    
    # For now, just use all constraints if n is small, or top-K by some criterion
    if size <= top_k_constraints * 2:
        rows = np.arange(size)
    else:
        # Use a spread of constraints: all if not too many, else sample
        # Pick constraints near the peak of the auto-correlation of a
        C = fast_autocorr(a)
        # Rank constraints by how much they matter (proportional to local a values)
        importance = np.zeros(size)
        for k in range(size):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            importance[k] = np.sum(a[js:je+1])
        idx = np.argpartition(importance, -top_k_constraints)[-top_k_constraints:]
        rows = idx
    
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[rows, :] @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    # Verify feasibility with ALL constraints
    full_C = T @ b_val
    if full_C.max() > M * 1.01:
        # Try with all constraints
        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
        try:
            prob2.solve(solver=solver, verbose=False)
        except Exception:
            pass
        if prob2.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            if b_val.sum() < 1e-15:
                return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True, top_k=400):
    """LP-based Nesterov optimization."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    iters = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        t = iters + 1.0
        M_curr = fast_autocorr(y).max()
        if M_curr < best_M:
            best_M = M_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        b = lp_solve(y, M_curr, solver, top_k_constraints=top_k)
        if b is None:
            continue
        
        b_sum = b.sum()
        g = b * S / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * S / sy
        
        momentum = (t - 2.0) / (t + 1.0) if t > 2 else 0.0
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * S / sa
        
        y = y_new
        a = a_new
        
        if a.max() < best_M:  # Check a too (cheaper than full autocorr... actually not)
            Ma = fast_autocorr(a).max()
            if Ma < best_M:
                best_M = Ma
                best_a = a.copy()
                stall = 0
        
        if stall > 500:
            rng = np.random.default_rng(iters * 7 + 3)
            noise = rng.normal(0, 0.02 * a.mean(), n)
            y = np.maximum(y + noise, 0)
            y = y * S / y.sum()
            a = y.copy()
            stall = 0
        
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={iters} ({elapsed:.0f}s): best_M={best_M:.6f}")
    
    return best_a, best_M


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Local search: move mass from peak-contributing positions to non-peak positions."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    T_start = max(0.01, best_M * 0.01)
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (0.01) ** elapsed_frac
        
        C = fast_autocorr(a)
        top_k = np.argsort(C)[::-1][:5]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-18:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find good destinations (low value, not contributing to peaks)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True
            
            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(np.arange(n) != src)[0]
            if len(non_peak) == 0:
                continue
            
            dst = non_peak[rng.integers(len(non_peak))]
            
            frac = rng.choice([0.05, 0.1, 0.15, 0.2])
            amt = a[src] * frac
            if amt < 1e-18:
                continue
            
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial * S / st
            
            score = fast_autocorr(a_trial).max()
            
            if score < best_M:
                a = a_trial
                best_M = score
                best_a = a.copy()
                n_improve += 1
                improved = True
                break
            elif rng.random() < np.exp(-(score - best_M) / max(T, 1e-15)):
                a = a_trial
                improved = True
                break
        
        if not improved:
            n_stall += 1
            if n_stall > 100:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > S * 1e-10:
                    frac = rng.uniform(0.1, 0.3)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 1e-15:
                        a_trial = a_trial * S / st
                        score = fast_autocorr(a_trial).max()
                        if score < best_M * 1.005:
                            a = a_trial
                            if score < best_M:
                                best_M = score
                                best_a = a.copy()
                                n_improve += 1
                n_stall = 0
        
        if verbose and n_improve % 5000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PS improve={n_improve} ({elapsed:.0f}s): M={best_M:.6f}")
    
    return best_a, best_M


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def spectral_init(n, seed=0):
    """Generate a non-negative sequence with approximately flat spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    # Random phases with conjugate symmetry
    A = np.zeros(fft_size, dtype=complex)
    A[0] = rng.uniform(0.5, 1.0)
    for k in range(1, fft_size // 2 + 1):
        mag = rng.uniform(0.3, 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        A[k] = mag * np.exp(1j * phase)
        A[fft_size - k] = np.conj(A[k])
    a = np.real(ifft(A))[:n]
    a = np.abs(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
