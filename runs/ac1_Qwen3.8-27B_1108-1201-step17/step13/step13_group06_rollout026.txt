```python
def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_objective_and_grad_z(z, T):
    """
    Compute the log-sum-exp smoothed objective and its gradient in z-space.
    z is the log-parameterization: a = softmax(z).
    Returns (f_smooth, grad_z) where f_smooth = 2n * logsumexp(conv(a,a)/T).
    """
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    # log-sum-exp of b/T
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    log_sum_exp = b_max + np.log(exp_b.sum())
    f_val = 2.0 * n * log_sum_exp
    
    # Gradient: w_k = exp(b_k/T) / sum(exp(b_l/T))
    w = exp_b / exp_b.sum()
    # grad_a_j = (1/T) * sum_k w_k * a_{k-j} (valid indices)
    # This is correlate(w, a, 'valid') / T
    # w has length 2n-1, a has length n
    # correlate(w, a, 'valid')[j] = sum_i w[i+j] * a[i] for i=0..n-1
    # But we need sum_k w_k * a_{k-j}, which is the same as:
    # Let i = k-j, k = i+j: sum_i w[i+j] * a[i]
    # This is exactly correlate(w, a, 'valid')[j]
    grad_a = np.correlate(w, a, mode='valid') / T
    # grad_a should have length 2n-1-n+1 = n. Good.
    
    # Convert to z-space gradient
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return f_val, grad_z
```

Wait, I need to double-check the correlate computation. `np.correlate(w, a, 'valid')` where w has length 2n-1 and a has length n:

The 'valid' mode returns the correlation only for lags where the signals completely overlap. Since len(w) > len(a), the output has length len(w) - len(a) + 1 = (2n-1) - n + 1 = n.

The output is: out[j] = sum_i w[i+j] * a[i] for j = 0 to n-1, where i = 0 to n-1.

So out[j] = sum_{i=0}^{n-1} w[i+j] * a[i] = sum_{k=j}^{j+n-1} w[k] * a[k-j]

And we want grad_a_j = (1/T) * sum_k w_k * a_{k-j} where k-j is in [0, n-1], i.e., k is in [j, j+n-1].

Since w has length 2n-1, valid k values are 0 to 2n-2. For j=0: k in [0, n-1]. For j=n-1: k in [n-1, 2n-2]. All valid.

So grad_a = correlate(w, a, 'valid') / T.

```python
def adam_lse_optimize(a0, time_limit, T_schedule=None, lr=0.005, seed=42):
    """Adam optimization with LSE-smoothed objective in log-space."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    if T_schedule is None:
        T_schedule = [50.0, 20.0, 8.0, 3.0, 1.0, 0.3, 0.1]
    
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for stage_T in T_schedule:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        lr_current = lr * min(1.0, 500.0 / n)
        
        max_iters = max(500, int((time_limit - (time.monotonic() - t0)) * 3))
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit - 3:
                break
            
            f_val, grad_z = lse_obj_grad_z(z, stage_T)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr_current * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 100 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        # End of stage: reset Adam state
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
            z = np.log(np.maximum(best_a, 1e-30))
    
    return best_a, best_score


def build_toeplitz(y):
    """Build (2n-1) x n matrix where T@b = conv(y, b)."""
    n = len(y)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    return T


def find_solver():
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    warm_b = None
    stall_count = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        T = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T, solver, warm_b)
        if b_star is None:
            warm_b = None
            continue
        warm_b = b_star
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 300 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            stall_count = 0
            warm_b = None
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Peak shaving: reduce the highest peaks by redistributing mass."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]
        
        danger = np.zeros(n, dtype=bool)
        for k in top_peaks[:20]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            danger[js:je + 1] = True
        
        improved = False
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue
            
            found = False
            for dst in dst_pool[:10]:
                for frac in [0.003, 0.005, 0.01, 0.02, 0.04, 0.06, 0.09]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                n_pert = rng.integers(1, 6)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.04)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 300:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full log-sum-exp smoothed Adam gradient descent in log-space (multi-temperature annealing)
2. LP Nesterov zero-sum game for global structural optimization
3. Peak shaving for fine local improvements
4. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad_z(z, T):
    """
    Compute LSE-smoothed objective and gradient in z-space.
    a = softmax(z), objective = 2n * logsumexp(conv(a,a) / T)
    """
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    log_sum_exp = b_max + np.log(exp_b.sum())
    f_val = 2.0 * n * log_sum_exp
    w = exp_b / exp_b.sum()
    # grad_a_j = (1/T) * sum_{k} w_k * a_{k-j} for valid indices
    # = correlate(w, a, 'valid') / T
    grad_a = np.correlate(w, a, mode='valid') / T
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return f_val, grad_z


def adam_lse_optimize(a0, time_limit, T_schedule=None, lr=0.005, seed=42):
    """Adam optimization with full LSE-smoothed objective in log-space."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    if T_schedule is None:
        T_schedule = [50.0, 20.0, 8.0, 3.0, 1.0, 0.4, 0.15]

    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for stage_idx, stage_T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit - 5:
            break

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        lr_current = lr * min(1.0, 500.0 / n) * (0.8 ** stage_idx)

        # More iters at higher temperature (exploration), fewer at low (exploitation)
        iters_budget = max(200, int(40 * n * (1.0 / max(stage_T, 0.5)) ** 0.3))
        iters_budget = min(iters_budget, int((time_limit - (time.monotonic() - t0)) * 2))

        for it in range(iters_budget):
            if time.monotonic() - t0 > time_limit - 3:
                break

            f_val, grad_z = lse_obj_grad_z(z, stage_T)

            # Gradient clipping for stability
            g_norm = np.linalg.norm(grad_z)
            if g_norm > 10.0:
                grad_z = grad_z * (10.0 / g_norm)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z_update = lr_current * m_hat / (np.sqrt(v_hat) + eps)
            # Clip per-step update magnitude
            update_norm = np.linalg.norm(z_update)
            if update_norm > 5.0:
                z_update = z_update * (5.0 / update_norm)
            z = z - z_update

            if (it + 1) % 200 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    # Reset optimizer to avoid momentum drift
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0
                    z = np.log(np.maximum(best_a, 1e-30))

        # End of stage check
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        elapsed = time.monotonic() - t0
        if stage_T <= 3.0:
            print(f"    LSE stage T={stage_T:.1f} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def build_toeplitz(y):
    """Build (2n-1) x n matrix where T@b = conv(y, b)."""
    n = len(y)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    sv = b_val.sum()
    if sv < 1e-15:
        return None, None
    b_val = b_val / sv
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """Nesterov-accelerated LP best-response iteration with restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    warm_b = None
    stall_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        T = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T, solver, warm_b)
        if b_star is None:
            warm_b = None
            stall_count += 1
            if stall_count > 50:
                a = best_a.copy()
                y = best_a.copy()
                warm_b = None
                stall_count = 0
            continue

        warm_b = b_star
        stall_count = 0

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={t} ({elapsed:.0f}s): best={best_score:.6f}")

        # Restart if stalled
        if t % 200 == 0:
            current_gap = best_score - 2.0 * n * fast_autocorr(a).max()
            if current_gap < 1e-8 and t > 100:
                # Try restarting from best
                a = best_a.copy()
                y = best_a.copy()
                warm_b = None

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Systematic peak shaving with global peak verification."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        # Danger zones: positions that contribute to any top peak
        danger = np.zeros(n, dtype=bool)
        for k in top_peaks[:25]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            danger[js:je + 1] = True

        improved = False
        for k in top_peaks[:20]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destinations (not in danger zone, sorted by value)
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(8, len(safe_sorted) // 3)]
            if len(dst_pool) == 0:
                continue

            found = False
            for dst in dst_pool[:12]:
                for frac in [0.002, 0.005, 0.01, 0.015, 0.025, 0.04, 0.06, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-10):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 15 == 0:
                n_pert = rng.integers(1, 8)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.05)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a0, rng, scale=0.05):
    """Create a perturbed version of a sequence."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    noise = rng.normal(0, scale, n)
    a_new = a + noise * (a + 1e-8)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def propose_candidate():
    """
    Multi-phase optimization combining LSE-Adam, LP Nesterov game,
    peak shaving, and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = 1700.0

    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    if best_seq is None:
        n_default = 1600
        x = np.linspace(0, 1, n_default)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)
    rng = np.random.default_rng(2024)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={time.monotonic()-t_global:.0f}s)")

    # === Phase 1: Quick LSE-Adam to establish good starting point ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 1: LSE-Adam init n={n_base}, t={t1:.0f}s ===")
    if t1 > 30:
        a_out, _ = adam_lse_optimize(best_seq, t1,
                                      T_schedule=[30.0, 10.0, 4.0, 1.5, 0.5, 0.2],
                                      lr=0.01)
        try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: LP Nesterov (main global optimization) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.35, 550.0)
    print(f"=== Phase 2: LP Nesterov n={n_base}, t={t2:.0f}s ===")
    if t2 > 50 and n_base <= 4000:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 3: LSE-Adam polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 180.0)
    print(f"=== Phase 3: LSE-Adam polish n={n_base}, t={t3:.0f}s ===")
    if t3 > 20:
        a_out, _ = adam_lse_optimize(best_seq, t3,
                                      T_schedule=[8.0, 3.0, 1.0, 0.4, 0.15, 0.05],
                                      lr=0.006)
        try_update(a_out, f"lse2_n{n_base}")

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: Multi-length, remaining={remaining:.0f}s ===")

    # Try longer lengths
    for n_trial in [n_base + 150, n_base + 250, n_base + 350]:
        if remaining < 40:
            break
        t4_each = min(remaining * 0.20, 100.0)
        if t4_each < 20:
            break
        init_long = interpolate_sequence(best_seq, n_trial)
        a_out, _ = adam_lse_optimize(init_long, t4_each,
                                      T_schedule=[15.0, 5.0, 1.5, 0.5, 0.15],
                                      lr=0.006)
        try_update(a_out, f"multi_long_n{n_trial}")
        remaining = time_budget - (time.monotonic() - t_global)

    # Try shorter length (might be better since 2n factor is smaller)
    if remaining > 40 and n_base > 800:
        n_short = n_base - 200
        t4_short = min(remaining * 0.15, 80.0)
        if t4_short > 15:
            init_short = interpolate_sequence(best_seq, n_short)
            a_out, _ = adam_lse_optimize(init_short, t4_short,
                                          T_schedule=[15.0, 5.0, 1.5, 0.5, 0.15],
                                          lr=0.006)
            try_update(a_out, f"multi_short_n{n_short}")
        remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.20, 120.0)
    print(f"=== Phase 5: Peak shaving n={len(best_seq)}, t={t5:.0f}s ===")
    if t5 > 10:
        a_out, _ = peak_shaving(best_seq, t5, seed=7777)
        try_update(a_out, "peak_1")

    # === Phase 6: Second LSE-Adam pass (different T schedule) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 25:
        t6 = min(remaining * 0.25, 120.0)
        if t6 > 15:
            print(f"=== Phase 6: LSE-Adam deep polish ===")
            a_out, _ = adam_lse_optimize(best_seq, t6,
                                          T_schedule=[5.0, 2.0, 0.8, 0.3, 0.1, 0.03],
                                          lr=0.004)
            try_update(a_out, "lse3_deep")

    # === Phase 7: Multi-start with perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t7_total = min(remaining * 0.20, 80.0)
        if t7_total > 20:
            print(f"=== Phase 7: Multi-start ===")
            for si in range(3):
                if time.monotonic() - t_global > t_global + time_budget - 50:
                    break
                t7_each = t7_total / 3
                pert = perturb_sequence(best_seq, rng, scale=0.04 * (si + 1))
                a_out, _ = adam_lse_optimize(pert, t7_each,
                                              T_schedule=[10.0, 3.0, 1.0, 0.3, 0.1],
                                              lr=0.005)
                try_update(a_out, f"multistart_{si}")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 8:
            print(f"=== Phase 8: Final peak shave ===")
            a_out, _ = peak_shaving(best_seq, t8, seed=31415)
            try_update(a_out, "peak_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
