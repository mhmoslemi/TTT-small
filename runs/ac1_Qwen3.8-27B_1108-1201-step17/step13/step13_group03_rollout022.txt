def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _build_conv_matrix(a, n):

    for j in range(n):

def _lp_best_response(a, M, solver):

    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _frank_wolfe(a0, time_limit, verbose=False):
    """Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = _fast_autocorr(y)
        k_star = int(np.argmax(b))
        
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += 2.0 * y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += 2.0 * y[j2]
        
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def _lse_fista(a0, T_schedule, iters_per_stage, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 50.0

    for stage_i, T in enumerate(T_schedule):
        L = max(n / (T * 2.0), n / 2000.0)
        t_fista = 1
        staging = a.copy()
        max_iters = max(int(iters_per_stage * (T_schedule[0] / T) ** 0.3), 3000)

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            grad = _lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 300 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    sc = _fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T


def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def _lp_nesterov_restarts(a0, time_limit, solver, verbose=True, n_restarts=3):
    """LP best-response with Nesterov acceleration and periodic restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for restart in range(n_restarts):
        y = best_a.copy()
        avg_a = np.zeros(n)
        n_avg = 0
        n_iters = 0
        stall = 0
        prev_best = best_score

        for t in range(1, 200000):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            M_curr = _fast_autocorr(y).max()
            if M_curr < 1e-14:
                break
            b = _lp_best_response(y, M_curr, solver)
            if b is None:
                stall += 5
                continue

            b_sum = b.sum()
            if b_sum < 1e-15:
                stall += 5
                continue

            g = b * norm / b_sum
            n_iters += 1

            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            if t == 1:
                a_new = g.copy()
            else:
                momentum = (t - 1.0) / (t + 2.0)
                a_new = y_new + momentum * (y_new - y)
                a_new = np.maximum(a_new, 0.0)
                s_a = a_new.sum()
                if s_a < 1e-15:
                    a_new = y_new.copy()
                else:
                    a_new = a_new * norm / s_a

            y = y_new
            a = a_new

            score_y = _fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall = 0
            else:
                stall += 1

            score_a = _fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall = 0

            avg_a += a
            n_avg += 1
            if n_avg % 30 == 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    score_avg = _fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()
                        stall = 0

            if verbose and n_iters % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP-R restart={restart} iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

            if stall > 200:
                break

        # If we still have time, perturb and restart
        if time.monotonic() - t0 < time_limit and best_score < prev_best:
            y = best_a * (1 + 0.01 * rng.standard_normal(n))
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y

    return best_a, best_score


def _frank_wolfe(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = _fast_autocorr(y)
        k_star = int(np.argmax(b))

        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += 2.0 * y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += 2.0 * y[j2]

        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _peak_shave(a, time_limit, seed=42):
    """Aggressive multi-strategy peak shaving."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]
        peak_k = top_k[0]
        improved = False

        # Strategy 1: Targeted move from peak-contributing source
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]

        if contrib.max() > 1e-18:
            top_src = idx[np.argsort(contrib)[::-1][:3]]
            src = top_src[0]

            # Build peak mask
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2: je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]

            # Try moving to least-occupied non-peak positions
            candidates_sorted = candidates[np.argsort(a[candidates])][:30]
            for dst in candidates_sorted[:10]:
                for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = _fast_autocorr(a_trial).max()
                    if score < best_score - 1e-16:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        # Strategy 2: Safeguarded line search along gradient direction
        if not improved:
            grad = _lse_grad(a / a.sum(), 0.1)
            # Project gradient to not increase sum
            grad_centered = grad - grad.mean()
            for step_frac in [0.001, 0.005, 0.01, 0.02]:
                a_trial = a - step_frac * grad_centered * a
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-16:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        # Strategy 3: Random perturbation to escape
        if not improved:
            stall += 1
            if stall % 50 == 0:
                n_pert = rng.integers(2, 8)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > norm * 1e-8:
                        frac = rng.uniform(0.01, 0.08)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = _fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 1000:
                break

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _exp_family_optimize(a0, time_limit, verbose=True):
    """Optimize in exponential family: a_i proportional to exp(B @ theta)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    s = a.sum()
    a = a / s

    x = np.linspace(0, 1, n)
    basis = []
    basis.append(np.ones(n))
    basis.append(x)
    basis.append(x ** 2)
    basis.append((x - 0.5) ** 2)
    basis.append(x ** 3)
    basis.append((x - 0.5) ** 3)
    basis.append(np.sin(2 * np.pi * x))
    basis.append(np.sin(4 * np.pi * x))
    basis.append(np.cos(2 * np.pi * x))
    basis.append(np.cos(4 * np.pi * x))
    basis.append(np.exp(-10 * (x - 0.5) ** 2))
    basis.append(np.exp(-5 * (x - 0.3) ** 2))
    basis.append(np.exp(-5 * (x - 0.7) ** 2))

    B = np.column_stack(basis)
    n_params = B.shape[1]

    log_a = np.log(a + 1e-15)
    theta, _, _, _ = np.linalg.lstsq(B, log_a, rcond=None)

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    lr = 0.02
    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        log_a_pred = B @ theta
        log_a_pred -= log_a_pred.max()
        a_curr = np.exp(log_a_pred)
        a_curr = np.maximum(a_curr, 1e-15)
        s_curr = a_curr.sum()
        a_curr = a_curr / s_curr

        T = max(0.1, 1.0 * (1.0 - min(it / 50000.0, 1.0)))
        grad_a = _lse_grad(a_curr, T)

        # da_curr/dtheta_j = B_j * a_curr - a_curr * (sum_i B_ij * a_curr_i)
        # (softmax-like chain rule)
        grad_theta = np.zeros(n_params)
        for j in range(n_params):
            grad_theta[j] = np.sum(B[:, j] * a_curr * grad_a) / s_curr
            grad_theta[j] -= a_curr.sum() * np.mean(B[:, j] * a_curr * grad_a) / s_curr

        # Simplified: just use B.T @ grad_a
        grad_theta = B.T @ (grad_a / s_curr)

        lr = max(0.005, 0.05 * (1.0 / (1.0 + it * 0.0002)))
        theta = theta - lr * grad_theta
        theta = np.clip(theta, -30, 30)

        if it % 200 == 0:
            sc = _fast_autocorr(a_curr).max()
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()

            if verbose and it % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    ExpFam iter={it} ({elapsed:.0f}s): best={best_score:.8f}")

    sc = _fast_autocorr(a_curr).max()
    if sc < best_score:
        best_score = sc
        best_a = a_curr.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase, multi-length optimization with restarts."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1680.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.01)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LSE-FISTA on parent (quick check) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 20:
        print(f"\n=== Phase 1: LSE-FISTA n={len(best_seq)} ({t1:.0f}s) ===")
        T_sched = [15.0, 5.0, 1.5, 0.5, 0.15, 0.05]
        a_out, _ = _lse_fista(best_seq, T_sched, 8000, t1, verbose=True)
        _try_update(a_out, "lse1")

    # === Phase 2: LP + Nesterov with restarts (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov+restarts n={n_cur} ({t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov_restarts(a_norm, t2, solver, verbose=True, n_restarts=3)
        _try_update(a_out, "lp1")

    # === Phase 3: Exponential family (dimensionality reduction) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.08, 120.0)
    if t3 > 15:
        print(f"\n=== Phase 3: ExpFamily n={len(best_seq)} ({t3:.0f}s) ===")
        a_out, _ = _exp_family_optimize(best_seq, t3, verbose=True)
        _try_update(a_out, "expfam")

    # === Phase 4: LP on longer sequence (n+300) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq) + 300
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 60 and n4 <= 3000:
        print(f"\n=== Phase 4: LP n={n4} ({t4:.0f}s) ===")
        init4 = _interpolate_sequence(best_seq, n4)
        a_norm = init4 * np.sqrt(2.0 * n4) / init4.sum()
        a_out, _ = _lp_nesterov_restarts(a_norm, t4, solver, verbose=True, n_restarts=2)
        _try_update(a_out, f"lp_n{n4}")

    # === Phase 5: LSE-FISTA on current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 20:
        print(f"\n=== Phase 5: LSE-FISTA n={n_cur} ({t5:.0f}s) ===")
        T_sched2 = [8.0, 2.0, 0.5, 0.15, 0.05]
        a_out, _ = _lse_fista(best_seq, T_sched2, 12000, t5, verbose=True)
        _try_update(a_out, "lse2")

    # === Phase 6: FW polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.06, 80.0)
    if t6 > 10:
        print(f"\n=== Phase 6: FW n={n_cur} ({t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _frank_wolfe(a_norm, t6, verbose=True)
        _try_update(a_out, "fw")

    # === Phase 7: Aggressive peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.12, 180.0)
    if t7 > 15:
        print(f"\n=== Phase 7: PeakShave n={n_cur} ({t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, t7, seed=777)
        _try_update(a_out, "peak1")

    # === Phase 8: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 30:
        print(f"\n=== Phase 8: LP final n={n_cur} ({t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov_restarts(a_norm, t8, solver, verbose=True, n_restarts=2)
        _try_update(a_out, "lp_final")

    # === Phase 9: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 20:
        t9 = min(remaining * 0.7, 200.0)
        if t9 > 10:
            print(f"\n=== Phase 9: PeakShave final n={n_cur} ({t9:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = _peak_shave(a_norm, t9, seed=424242)
            _try_update(a_out, "peak_final")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
