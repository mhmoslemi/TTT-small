def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    """Build the convolution constraint matrix T where (T @ b)[k] = (a*b)(k)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver, T=None):
    """Solve LP: maximize sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if T is None:
        T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    dual_val = None
    try:
        dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    return b_val, dual_val

    # the gradient of max(b) w.r.t. a[i] is a[k*-i] + a[i-k*] where k* = argmax
    # For LSE: grad[i] = sum_k w[k]*(a[k-i] if 0<=k-i<n else 0 + a[i-k] if 0<=i-k<n else 0)

    # Use FFT: define a_ext of length fft_size with a in first n positions
    # w_ext of length fft_size with w in first (2n-1) positions
    # Then the "cross-correlation" we want is:
    # g[i] = sum_k w[k]*a[k-i] (valid part) + sum_k w[k]*a[i-k] (valid part)
    # The first term is (w convolved with reversed a) evaluated appropriately
    # Let's just compute it directly for small n or use the FFT trick

    # For efficiency with n up to ~2000, direct computation of the gradient
    # would be O(n * (2n-1)) which is ~4M for n=1000, ~16M for n=2000.
    # FFT approach: 
    # term1[i] = sum_{k} w[k] * a[k-i] where 0<=k-i<n and 0<=k<2n-1
    #   = sum_{j=0}^{n-1} w[i+j] * a[j]  (where i+j < 2n-1)
    #   This is a convolution of w with a, evaluated at specific positions
    #   Let's define: (w * a_full)[m] = sum_j w[m-j]*a[j] where a_full has a in [0,n-1]
    #   Then term1[i] = (w * a_full)[i] ... no wait
    #   term1[i] = sum_j w[i+j]*a[j] = sum_j w_rev[n-1-j+1+n-1-i]*a[j]... 
    #   Let me just use: term1 = ifft(fft(w_ext) * fft(a_ext)) appropriately shifted

    # Actually, the cleanest FFT approach:
    # b = ifft(|fft(a)|^2) gives autocorrelation
    # db/da[i] for the LSE is: sum_k w[k] * a[k-i] (valid) 
    # Let's compute via: create signal u of length fft_size where u[:n]=a, u[n:2n-1]=w[:n-1]
    # Hmm this is getting complicated. Let me just do O(n^2) for the gradient
    # since n is at most ~2000 and we're calling this maybe 200 times total.

    # Direct gradient computation
    grad_direct = np.zeros(2 * n - 1)
    # b[k] = sum_i a[i]*a[k-i], db[k]/da[i] = a[k-i] (if valid) + a[i-k] (if valid)
    # d(lse)/da[i] = sum_k w[k] * (a[k-i] if 0<=k-i<n else 0)
    #              + sum_k w[k] * (a[i-k] if 0<=i-k<n else 0)
    # Note the second sum: let j = k-i, then = sum_j w[j+i] * a[-j] ... 
    # Actually by symmetry of autocorrelation, b is symmetric: b[k] = b[2n-2-k] (approximately)
    # So we can just compute one side

    # Efficient: use the cross-correlation via FFT
    # d(lse)/da[i] = sum_k w[k] * a[k-i] (valid part only)
    # Define: c[m] = sum_k w[k] * a[k-m] for m = 0,...,n-1
    # c = correlation of w and a (not circular)
    # c[m] = sum_{k=m}^{m+n-1} w[k] * a[k-m] = sum_{j=0}^{n-1} w[m+j] * a[j]
    # This is the cross-correlation, computable as:
    # c = ifft(fft(w, N) * conj(fft(a, N))) for appropriate N >= 3n-2

    fft_size2 = 1 << (3 * n - 2).bit_length()
    a_ext = np.zeros(fft_size2)
    a_ext[:n] = a
    w_ext = np.zeros(fft_size2)
    w_ext[:2*n-1] = w
    corr = irfft(rfft(w_ext, fft_size2) * np.conj(rfft(a_ext, fft_size2)), fft_size2)
    # corr[m] = sum_k w_ext[k] * a_ext[k-m] (valid part)
    # We want c[i] = sum_{j=0}^{n-1} w[i+j] * a[j] = corr[i] (from the above)
    # But corr is circular, so for i in [0, n-1], corr[i] should give us what we want
    # if fft_size2 is large enough (which it is: 3n-2 > 2n-1 + n - 1)
    
    # Actually, the standard cross-correlation: (w ⋆ a)[i] = sum_j w[i+j] * a[j]
    # In FFT: ifft(fft(w) * conj(fft(a))) gives circular cross-correlation
    # For linear (non-circular), we need padding, which we have
    # So corr[:n] gives us d(lse)/da (the first part)
    
    # But wait, b[k] = sum_i a[i]*a[k-i], so db[k]/da[i] = a[k-i] if 0<=k-i<n
    # d(lse)/da[i] = sum_k w[k]*a[k-i] (valid) 
    # Let j = k-i: sum_j w[i+j]*a[j] (j from 0 to n-1, i+j from i to i+n-1, need i+j < 2n-1)
    # This is exactly corr[i] if our FFT is set up correctly
    # corr computed as ifft(fft(w_ext) * conj(fft(a_ext)))[m] = sum_k w_ext[k]*conj(a_ext[k-m])
    # For real: = sum_k w_ext[k]*a_ext[k-m]
    # We want: sum_{j=0}^{n-1} w[i+j]*a[j], let k=i+j, j=k-i:
    # = sum_{k=i}^{i+n-1} w[k]*a[k-i] = sum_k w_ext[k]*a_ext[k-i] (since a_ext[k-i]=a[k-i] for 0<=k-i<n)
    # And this equals (circular corr)[i] = corr[i] when no wraparound (ensured by padding)
    
    # So d(lse)/da[i] = corr[i] (for i=0,...,n-1) as long as no wraparound
    # But we also need the "other side": db[k]/da[i] also has contribution from a[i-k]
    # Wait no. b[k] = sum_i a[i]*a[k-i]. The derivative w.r.t. a[i] is a[k-i] (single term).
    # There's no "other side" because b is defined as sum over i, not sum over i and j separately.
    # Actually: b[k] = sum_i a[i] * a[k-i]. db[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    # Wait, that's only if we expand: b[k] = sum_i a[i]*a[k-i]. When we differentiate w.r.t. a[i]:
    # The term a[i]*a[k-i] contributes: if the index of the second a is also i (i.e., k-i=i, k=2i), 
    # then we get 2*a[i]. Otherwise: a[k-i] from the first factor, and a[i-k] from... 
    # No wait. b[k] = sum_j a[j]*a[k-j]. db[k]/da[i] = a[k-i] (from j=i in first factor) 
    # + a[i-k] (from k-j=i, i.e., j=k-i, in second factor)
    # So db[k]/da[i] = a[k-i] + a[i-k] (each term only if the index is in [0,n-1])
    
    # So d(lse)/da[i] = sum_k w[k]*(a[k-i] if 0<=k-i<n else 0 + a[i-k] if 0<=i-k<n else 0)
    # = sum_k w[k]*a[k-i] (valid) + sum_k w[k]*a[i-k] (valid)
    # First sum = corr[i] (as computed above)
    # Second sum: let m = i-k, then k = i-m, sum_m w[i-m]*a[m] (valid m in [0,n-1], i-m in [0,2n-2])
    # = sum_m w[i-m]*a[m] = this is also a cross-correlation but different direction
    # = (a ⋆ w)[i] = corr2[i] where corr2 = ifft(fft(a)*conj(fft(w)))
    # By symmetry of autocorrelation (b is symmetric), w is approximately symmetric too
    # But not exactly. Let me compute both.
    
    corr2 = irfft(rfft(a_ext, fft_size2) * np.conj(rfft(w_ext, fft_size2)), fft_size2)
    
    d_lse_da = (corr[:n] + corr2[:n]).real
    # Hmm wait, I think corr[:n] already captures both contributions due to the symmetry
    # Let me just verify with a small example mentally...
    # For n=3, a=[a0,a1,a2], b=[a0^2, 2*a0*a1, a1^2+2*a0*a2, 2*a1*a2, a2^2] (length 5)
    # db[0]/da[0] = 2*a0 (from k=0: a[0-0]=a0, and a[0-0]=a0, so yes 2*a0)
    # db[1]/da[0] = a[1] (from k=1: a[1-0]=a1, and a[0-1] invalid, so just a1)
    # db[2]/da[0] = a[2] (from k=2: a[2-0]=a2, and a[0-2] invalid)
    # So d(lse)/da[0] = sum_k w[k]*(a[k] if k<3 + a[-k] if -k>=0)
    # = w[0]*2*a0 + w[1]*a1 + w[2]*a2 (for the first part: a[k-0] for k=0,1,2)
    # + (second part: a[0-k] for valid k: only k=0 gives a[0], already counted)
    # Hmm, so for i=0: first sum gives w[0]*a[0]+w[1]*a[1]+w[2]*a[2], second gives w[0]*a[0]
    # Total = 2*w[0]*a[0] + w[1]*a[1] + w[2]*a[2] ✓
    
    # So we do need both. The gradient is:
    # grad_obj[i] = (2*n/(s^2)) * d_lse_da[i] - (2*obj/s) * a[i]
    # (the second term comes from differentiating 1/(sum)^2)
    
    grad_obj = (2.0 * n / (s * s)) * d_lse_da - (2.0 * obj / s) * a
    return obj, grad_obj.ravel()


def lse_scipy_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=5):
    """Optimize using LSE-smoothed objective with L-BFGS-B."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 120.0)

        def obj_wrapper(x):
            val, _ = lse_objective_and_grad(x, n, T)
            return val

        def grad_wrapper(x):
            _, g = lse_objective_and_grad(x, n, T)
            return g

        try:
            res = scipy_minimize(obj_wrapper, a.copy(), method='L-BFGS-B',
                                 jac=grad_wrapper,
                                 options={'maxiter': 300, 'maxfun': 1000,
                                           'time_limit': stage_time, 'ftol': 1e-12})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def multi_peak_polish(a0, time_limit, seed=42):
    """Reduce top peaks simultaneously by targeted mass redistribution."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:6]
        
        # Build peak mask: positions that contribute to top peaks
        peak_contrib = np.zeros(n)
        for k in top_peaks:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    peak_contrib[i] += a[j]
                j2 = i - k
                if 0 <= j2 < n:
                    peak_contrib[i] += a[j2]
        
        # Find positions with high peak contribution (sources) and low (destinations)
        high_contrib = np.where(peak_contrib > np.median(peak_contrib))[0]
        low_contrib = np.where(peak_contrib < np.median(peak_contrib))[0]
        
        improved = False
        # Try moving mass from high-contribution to low-contribution positions
        for src in high_contrib[:10]:
            if a[src] < 1e-10:
                continue
            for dst in low_contrib[:10]:
                if src == dst:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 5 == 0:
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.15)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def subgrad_nesterov(a0, time_limit):
    """Subgradient Nesterov method as a complementary trajectory."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        k_star = int(np.argmax(b))

        # Subgradient: direction to concentrate all mass
        i_min = int(np.argmin(np.arange(n)))
        v = np.zeros(n)
        # Find the index that minimizes the subgradient
        grad_points = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            val = 0.0
            if 0 <= j1 < n:
                val += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                val += y[j2]
            grad_points[i] = val
        i_min = int(np.argmin(grad_points))
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a = 0.98 * avg_a + 0.02 * a
        if t % 500 == 0:
            s_avg = avg_a.sum()
            if s_avg > 1e-15:
                avg_check = avg_a * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    rng = np.random.default_rng(seed)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Quick multi-length scan to find best length
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = [1200, 1600, 2000]
    scan_results = {}
    phase0_budget = time_budget * 0.12
    for n_scan in scan_lengths:
        if time.monotonic() - t_global > t_global + phase0_budget:
            break
        per_scan = (t_global + phase0_budget - time.monotonic()) / max(1, len(scan_lengths) - scan_lengths.index(n_scan))
        if per_scan < 15:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, per_scan, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} (t={time.monotonic()-t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a = scan_results[best_scan_n][0]
        do_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Deep LP-Nesterov at base length (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Deep LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, adaptive_M=True, verbose=True)
    do_update(a_out, f"lp_deep_n{n_base}")

    # Phase 2: LSE gradient optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=30.0, T_end=2.0, n_stages=5)
        do_update(a_out, f"lse_scipy_n{n_best}")

    # Phase 3: Try a longer length
    n_long = min(n_base + 400, 2400)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP n={n_long} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, adaptive_M=True, verbose=True)
        do_update(a_out, f"lp_long_n{n_long}")

    # Phase 4: Multi-peak polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Multi-peak polish n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = multi_peak_polish(a_norm, t4, seed=12345)
        do_update(a_out, "multi_peak_polish")

    # Phase 5: LSE with lower temperature (finer polishing)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LSE low-T n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t5, T_start=8.0, T_end=0.5, n_stages=4)
        do_update(a_out, "lse_lowT")

    # Phase 6: Subgradient Nesterov (different trajectory)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: SubGrad n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgrad_nesterov(a_norm, t6)
        do_update(a_out, "subgrad")

    # Phase 7: Final multi-peak polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.5, 300.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = multi_peak_polish(a_norm, t7, seed=99999)
        do_update(a_out, "final_polish")

    # Phase 8: One more LP round if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        print(f"\n=== Phase 8: Final LP n={n_best} (budget={remaining:.0f}s) ===")
        t8 = remaining * 0.7
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, adaptive_M=True, verbose=True)
        do_update(a_out, "lp_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
