def build_conv_matrix_rows(a, k_indices):
    """Build rows of the convolution matrix for specific k indices."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    for k in k_indices:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # (a*b)_k = sum_{i=lo}^{hi-1} a_i * b_{k-i}
        # For LP: we need T[k, j] = a_j if 0 <= k-j < n
        for j in range(lo, hi):
            idx = k - j
            if 0 <= idx < n:
                row[j] = a[idx]
        rows.append(row)
    return np.array(rows)


def build_conv_matrix_full(a):
    """Build full convolution matrix T where (T@b)_k = (a*b)_k."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, active_set=None, warm_start=None):
    """Solve the LP best-response: maximize sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and len(active_set) > 0:
        T = build_conv_matrix_rows(a, active_set)
        constraints = [T @ b_var <= M]
    else:
        T = build_conv_matrix_full(a)
        active_set = np.arange(size)
        constraints = [T @ b_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set, None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set, None

    # Try to get dual variables
    duals = None
    try:
        if isinstance(constraints[0], cp.constraints.Inequality):
            dual_val = constraints[0].dual_value
            if dual_val is not None:
                duals = np.asarray(dual_val, dtype=np.float64)
    except Exception:
        pass

    return b_val, active_set, duals

def pair_coordinate_descent(a0, time_limit, rng, verbose=True):
    """Pair swap coordinate descent with refined line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            # Try a few random partners
            for attempt in range(4):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Compute first-order gain vector
                g = np.zeros(size)
                # Contribution from position i (removing): -a shifted
                lo1 = max(0, i)
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    g[lo1:hi1] -= a[max(0, lo1 - i):hi1 - i]
                lo2 = max(0, n - 1 - i)
                hi2 = min(size, n + i)
                if hi2 > lo2:
                    src2 = n - 1 - np.arange(lo2, hi2)
                    valid = (src2 >= 0) & (src2 < n)
                    if any(valid):
                        g[lo2:hi2][valid] -= a[src2[valid]]

                # Contribution from position j (adding): +a shifted
                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    g[lo3:hi3] += a[max(0, lo3 - j):hi3 - j]
                lo4 = max(0, n - 1 - j)
                hi4 = min(size, n + j)
                if hi4 > lo4:
                    src4 = n - 1 - np.arange(lo4, hi4)
                    valid4 = (src4 >= 0) & (src4 < n)
                    if any(valid4):
                        g[lo4:hi4][valid4] += a[src4[valid4]]

                # Find optimal delta using refined search
                # First-order: minimize max_k(conv_k + delta * g_k) over delta in [0, ai]
                # Evaluate at strategic points
                candidates = [0.0, ai * 0.1, ai * 0.25, ai * 0.5, ai * 0.75, ai * 0.9, ai]
                best_d = 0.0
                best_val = conv.max()

                for d in candidates:
                    if d <= 0:
                        continue
                    new_max = (conv + d * g).max()
                    # Second-order correction for diagonal terms
                    di = 2 * i
                    if di < size:
                        new_max = max(new_max, conv[di] - 2 * a[i] * d + d * d)
                    dj = 2 * j
                    if dj < size:
                        new_max = max(new_max, conv[dj] + 2 * a[j] * d + d * d)
                    if new_max < best_val - 1e-13:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:

        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
    else:
        x = np.linspace(0.0, 1.0, n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    """Add multiplicative noise and renormalize."""
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def subgrad_fw(a0, time_limit, verbose=False):
    """Frank-Wolfe with subgradient from the peak."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            v = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                v += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                v += y[j2]
            grad[i] = v
        grad *= 2.0

        # FW step
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def propose_candidate():
    rng = np.random.default_rng(12345)
    t_global = time.monotonic()
    time_budget = 1740.0  # Safety margin from 1800s

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 1: LP Nesterov at base length ===
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.28, 420.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        do_try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Pair CD polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Pair CD n={n_best} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t2, rng, verbose=True)
        do_try_update(a_out, f"pcd1_n{n_best}")

    # === Phase 3: Perturbation restart + LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Perturbation restarts ===")
    t3_total = min(remaining * 0.25, 350.0)
    n_restarts = 3
    t3_each = t3_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t3 = min(t3_each, remaining - 60)
        if t3 < 30:
            break
        n_cur = len(best_seq)
        pert = perturb_sequence(best_seq, rng, scale=0.03 + 0.02 * r)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=False, seed=100 + r)
        do_try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 4: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 300
    print(f"\n=== Phase 4: LP Nesterov n={n2} ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, seed=7)
        do_try_update(a_out, f"lp_n{n2}")

    # === Phase 5: Pair CD on best so far ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Pair CD n={n_best} ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t5, rng, verbose=True)
        do_try_update(a_out, f"pcd2_n{n_best}")

    # === Phase 6: LP Nesterov on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP Nesterov n={n_best} (final) ===")
    t6 = min(remaining * 0.15, 180.0)
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, seed=42)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 7: Final pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Pair CD n={n_best} ===")
    t7 = min(remaining * 0.30, 150.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t7, rng, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n_best}")

    # === Phase 8: Shorter length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_short = max(1000, len(best_seq) - 300)
    if remaining > 80:
        print(f"\n=== Phase 8: LP Nesterov n={n_short} ===")
        t8 = min(remaining * 0.4, 100.0)
        if t8 > 20:
            init_short = interpolate_sequence(best_seq, n_short)
            a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=False, seed=99)
            do_try_update(a_out, f"lp_n{n_short}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
