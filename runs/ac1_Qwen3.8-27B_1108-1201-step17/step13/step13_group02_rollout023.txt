def compute_conv_gradient(a, w):
    """Compute gradient of sum_k w[k]*c[k] where c = a*a (full conv).
    grad[i] = sum_k w[k] * d(c[k])/da[i] = 2 * (w_ref * a)[i] for symmetric w
    More precisely: c[k] = sum_j a[j]*a[k-j], so dc[k]/da[i] = a[k-i] + a[i-(k-(n-1))]
    Actually for full conv: c[k] = sum_j a[j]*a[k-j] (j from max(0,k-n+1) to min(n-1,k))
    dc[k]/da[i] = a[k-i] if k-i is in [0,n-1], else 0.
    So grad[i] = sum_k w[k] * a[k-i] (where k-i in [0,n-1])
    This is the cross-correlation of w with a: grad = corr(w, a)
    Which equals (w_rev * a)[n-1 : 2n-1] where * is full convolution.
    """
    n = len(a)
    w_rev = w[::-1]
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    W = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:2 * n - 1]


def variance_aware_lse_gd(a0, time_limit, verbose=True):
    """LSE gradient descent with variance penalty on the convolution."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.5, best_score * 0.05)
    T_end = max(0.01, best_score * 0.003)
    lam_start = 0.1
    lam_end = 0.5

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac
        lam = lam_start + (lam_end - lam_start) * elapsed_frac

        c = fast_autocorr(a)
        max_c = c.max()
        if max_c < 1e-15:
            break

        # Softmax weights for LSE
        w = np.exp((c - max_c) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient of LSE approx of max
        grad_peak = compute_conv_gradient(a, w)

        # Gradient of variance penalty: var(c) = mean(c^2) - mean(c)^2
        # d(var(c))/da[i] = (2/(2n-1)) * sum_k (c[k] - mean(c)) * dc[k]/da[i]
        c_mean = c.mean()
        w_var = (c - c_mean) / (2 * n - 1)
        grad_var = compute_conv_gradient(a, w_var)

        # Combined gradient
        grad = grad_peak + lam * grad_var

        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-30:
            break
        step = 0.5 / (grad_norm + 1e-10)

        # Try the step with backtracking
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial * norm / s_t

        score_trial = fast_autocorr(a_trial).max()

        # Backtracking if needed
        trial_step = step
        for _ in range(5):
            if score_trial < max_c or score_trial < best_score:
                break
            trial_step *= 0.5
            a_trial = a - trial_step * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                a_trial = a.copy()
                break
            a_trial = a_trial * norm / s_t
            score_trial = fast_autocorr(a_trial).max()

        a = a_trial
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a > 1e-15:
            a = a * norm / s_a
        steps += 1

        if score_trial < best_score:
            best_score = score_trial
            best_a = a.copy()

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    VarLSE step={steps} ({elapsed:.0f}s): T={T:.4f}, lam={lam:.3f}, best={best_score:.8f}")

    return best_a, best_score


def pair_move_peak_shave(a0, time_limit, seed=42, verbose=True):

        elapsed_frac = (time.monotonic() - t0) / time_limit
        T_curr = max(0.01, best_score * 0.02 * (1.0 - elapsed_frac))

        c = fast_autocorr(a)
        top_peaks = np.argsort(c)[::-1][:4]

        improved = False
        for attempt in range(10):
            # Pick 1 or 2 source positions (contributing to peaks)
            k = top_peaks[rng.integers(len(top_peaks))]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            src = idx[np.argmax(contrib)]

            # Find destination: position with low value not contributing to peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)

            if len(non_peak) < 2:
                non_peak = np.where(np.arange(n) != src)[0]

                a = a_trial
                improved = True
                break

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def smooth_interpolate(f, new_n, kernel_width=5):
    """Interpolate with a Gaussian smoothing kernel to avoid sharp transitions."""
    base = interpolate_sequence(f, new_n)
    if kernel_width > 1:
        kernel = np.exp(-0.5 * (np.arange(-kernel_width, kernel_width + 1) / kernel_width) ** 2)
        kernel = kernel / kernel.sum()
        base = np.convolve(base, kernel, mode='same')
    s = base.sum()
    if s < 1e-15:
        return np.ones(new_n)
    return base / s

        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc = pair_move_peak_shave(a_norm, t3, seed=99, verbose=True)
        do_try_update(a_out, f"ps_n{n_cur}")

    # === Phase 4: Try length 1800 ===
    n2 = max(n_base + 200, 1800)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Optimize n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        init_long = smooth_interpolate(best_seq, n2, kernel_width=3)
        a_out, sc = nesterov_lp_optimize(init_long, t4, solver, verbose=False)
        if do_try_update(a_out, f"lp_n{n2}"):
            # Polish with LSE
            remaining = time_budget - (time.monotonic() - t_global)
            t4b = min(remaining * 0.05, 60.0)
            if t4b > 10:
                n_cur2 = len(best_seq)
                a_norm2 = best_seq * np.sqrt(2.0 * n_cur2) / best_seq.sum()
                a_out2, _ = variance_aware_lse_gd(a_norm2, t4b, verbose=False)
                do_try_update(a_out2, f"varlse_n{n2}")

    # === Phase 5: Second LP round on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP Round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # === Phase 6: Multiplicative weights variant ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: MW variant n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc = multiplicative_weights_optimize(a_norm, t6, solver, verbose=False)
        do_try_update(a_out, f"mw_n{n_best}")

    # === Phase 7: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final PS n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc = pair_move_peak_shave(a_norm, t7, seed=31415, verbose=True)
        do_try_update(a_out, f"psfinal_n{n_best}")

    # === Phase 8: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LSE n={n_best} (budget={remaining:.0f}s) ===")
    t8 = max(remaining, 20.0)
    t8 = min(remaining * 0.20, 150.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc = variance_aware_lse_gd(a_norm, t8, verbose=True)
        do_try_update(a_out, f"lsefinal_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
