from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full linear autocorrelation (convolution with itself) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_gradient_corrected(a, T):
    """
    Correct gradient of LSE surrogate: T*log(sum_k exp(b_k/T)).
    Result: dL/da[m] = 2 * sum_k w[k] * a[k-m] (with a[j]=0 for j out of range)
           = 2 * (a_rev ⊛ w)[n-1+m]
    """

    b = fast_autocorr(a)
    max_b = b.max()
    exp_vals = np.exp((b - max_b) / T)
    w_sum = exp_vals.sum()

        w[n - 1] = 1.0
    else:
        w = exp_vals / w_sum

    A_rev = rfft(a[::-1], fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)

    return 2.0 * conv[n-1 : 2*n-1]

3. DECREASE those positions (move mass from them).

5. INCREASE those positions (add mass to them).
