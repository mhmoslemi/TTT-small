    A = rfft(a, fft_size)

def build_sparse_conv_matrix(a, n):
    """Build sparse matrix T where (T @ b)[k] = sum_j a_j * b_{k-j} (valid indices)"""
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # b_j contributes to conv positions k where a_j * b_{k-j} is valid
        # k-j must be in [0, n-1], so k in [j, j+n-1]
        # a_{k-j} for k in [j, j+n-1] is a[0], a[1], ..., a[n-1]
        for i in range(n):

            if k < size:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T

import numpy as np

def _autocorr(a):
    """Full linear autocorrelation of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = rfft(a, fs)
    return irfft(A * A, fs)[:size]

def _cross_conv(a, b):
    """Full linear convolution of a and b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fs = 1 << (size - 1).bit_length()
    A = rfft(a, fs)
    B = rfft(b, fs)
    return irfft(A * B, fs)[:size]

def _eval_local(a):
    """Evaluate the objective locally (no external call needed)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    conv = _autocorr(a)
    return float(2.0 * n * conv.max() / (s * s))

def _normalize(a, target_sum):
    """Normalize a to be nonnegative with the given sum."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) * target_sum / len(a)
    return a * target_sum / s

def _build_constraints(y, active_set, n):
    """Build dense constraint submatrix for LP."""
    n_act = len(active_set)
    T_sub = np.zeros((n_act, n), dtype=np.float64)
    for idx, k in enumerate(active_set):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            T_sub[idx, j_lo:j_hi+1] = y[k - j_lo : k - j_hi + 1]
    return T_sub

def _lp_solve(y, M, active_set, n):
    """Solve LP: max sum(b) s.t. T @ b <= M, b >= 0 using HiGHS."""
    T_sub = _build_constraints(y, active_set, n)
    c = -np.ones(n)
    b_ub = np.full(len(active_set), M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    
    result = linprog(c, A_ub=T_sub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        return None
    b = np.maximum(result.x, 0.0)
    if b.sum() < 1e-15:
        return None
    return b

def _nesterov_lp(y0, time_limit, verbose=True):

    n = len(y0)

    y = _normalize(y0, norm)
    
    conv = _autocorr(y)
    best_score = conv.max()
    best_y = y.copy()
    
    active_set = None

    n_iters = 0

def _structured_init(n, rng):
    """Generate a random structured initialization."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)
    elif choice == 1:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(x) + 0.15 * np.cos(2.5 * x) + 0.05 * np.cos(4.3 * x)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - np.abs(2 * x - 1)) ** 0.7
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 5:
        x = np.linspace(0, 1, n)
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.3)
        a = np.exp(-((x - center) / width) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.cos(2 * np.pi * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** rng.uniform(0.3, 1.5)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    return a / s

def _interpolate(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n)
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_a = np.interp(new_x, old_x, a_norm)
    new_a = np.maximum(new_a, 0.0)
    ss = new_a.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return new_a / ss
