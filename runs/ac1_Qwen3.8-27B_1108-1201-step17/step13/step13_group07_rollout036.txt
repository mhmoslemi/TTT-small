import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def internal_score(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def project_nonnegative_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_vals = v_sorted * np.arange(1, n + 1) - (cssv - target_sum)
    rho = np.nonzero(rho_vals > 0)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        result += diff / n
        result = np.maximum(result, 0)
        result = result * (target_sum / result.sum())
    return result

def softmax_weights(b, T):
    """Compute softmax weights for the convolution peaks."""
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    s = exp_Tb.sum()
    if s < 1e-300:
        return np.ones(len(b)) / len(b)
    return exp_Tb / s

def compute_gradient(a, w):
    """Compute gradient of weighted sum of convolution values w.r.t. a."""
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    # The gradient of sum_k w_k * (a*a)_k w.r.t. a_j is 2 * sum_k w_k * a_{k-j}
    # conv[n-1 : 2n-1] gives sum_k w_k * a[k-j] for j=0..n-1
    grad = 2.0 * conv[n - 1 : 2 * n - 1]
    return grad

def mirror_descent_step(a, grad, eta):
    """Multiplicative weight update."""
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-300)  # prevent underflow
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

def accelerated_mirror_descent(a0, time_limit, T_start=200.0, T_end=0.5, n_stages=30):
    """Nesterov-accelerated mirror descent with adaptive temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    eta_base = 2.0
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # Number of iterations for this stage
        iters = max(50, int(remaining * 30 / (n_stages - si)))
        
        # Nesterov momentum
        y = a.copy()
        gamma = 1.0  # momentum parameter
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            b = fast_autocorr(y)
            w = softmax_weights(b, T)
            grad = compute_gradient(y, w)
            
            # Mirror descent step
            eta = eta_base / (1 + it * 0.05)
            a_new = mirror_descent_step(y, grad, eta)
            
            # Nesterov update
            gamma = 2.0 / (it + 3.0)
            y = (1 - gamma) * y + gamma * a_new
            
            # Check improvement
            if it % 10 == 0:
                score = internal_score(a_new)
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                a = a_new
                score_y = internal_score(y)
                if score_y < best_score:
                    best_score = score_y
                    best_a = y.copy()
                    y = a_new.copy()
        
        # End of stage: update a
        a = a_new
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      AMD stage {si}/{n_stages} (T={T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

1. Multi-temperature subgradient descent with Nesterov acceleration and BB step size

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def softmax_weights(b, T):
    """Compute softmax weights on convolution values."""
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    s = exp_Tb.sum()
    if s < 1e-300:
        return np.ones(len(b)) / len(b)
    return exp_Tb / s


def compute_gradient(a, w):
    """Gradient of sum_k w_k * (a*a)_k w.r.t. a."""
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_vals = v_sorted * np.arange(1, n + 1) - (cssv - target_sum)
    rho_indices = np.nonzero(rho_vals > 0)[0]
    if len(rho_indices) == 0:
        return np.full(n, target_sum / n)
    rho = rho_indices[-1]
   theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    # Fix numerical issues
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        result = result + diff / n
        result = np.maximum(result, 0)
        ss = result.sum()
        if ss > 1e-300:
            result = result * (target_sum / ss)
    return result

def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Spectral shaping to flatten the power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = internal_score(a)
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def softmax_weights(b, T):
    """Compute softmax weights on convolution values."""
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    s = exp_Tb.sum()
    if s < 1e-300:
        return np.ones(len(b)) / len(b)
    return exp_Tb / s


def compute_gradient(a, w):
    """Gradient of sum_k w_k * (a*a)_k w.r.t. a."""
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_vals = v_sorted * np.arange(1, n + 1) - (cssv - target_sum)
    rho_indices = np.nonzero(rho_vals > 0)[0]
    if len(rho_indices) == 0:
        return np.full(n, target_sum / n)
    rho = rho_indices[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        result = result + diff / n
        result = np.maximum(result, 0)
        ss = result.sum()
        if ss > 1e-300:
            result = result * (target_sum / ss)
    return result

                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-30:
                continue

            # Find top contributing coordinate
            top_contrib_idx = np.argsort(contrib)[::-1][:5]
            for tc_i in top_contrib_idx:
                src = idx[tc_i]
                partner = k - src

                # Find good destination: far from src and partner, small value
                dist_to_src = np.abs(np.arange(n) - src)
                dist_to_partner = np.abs(np.arange(n) - partner)
                min_dist = np.minimum(dist_to_src, dist_to_partner)

                # Candidates: far from both, small value
                threshold = max(n // 4, 20)
                candidates_mask = (min_dist > threshold) & (a < np.mean(a)) & (np.arange(n) != src)
                candidates = np.where(candidates_mask)[0]
                if len(candidates) == 0:
                    candidates_mask = (min_dist > threshold // 2) & (np.arange(n) != src)
                    candidates = np.where(candidates_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]

                # Pick from smallest values
                cand_vals = a[candidates]
                sort_order = np.argsort(cand_vals)[:min(20, len(candidates))]
                dst_pool = candidates[sort_order]
                dst = dst_pool[rng.integers(len(dst_pool))]

                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04, 0.06]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = internal_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break

                if improved:
                    break

            if improved:
                break

        if not improved:
            # Random small perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.001, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            stall += 1
            if stall > 800:
                break

        if iteration % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakShave iter {iteration} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Spectral shaping to flatten the power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Spectral shaping (fast initialization)
    Phase 2: Subgradient Nesterov-BB (primary optimizer)
    Phase 3: Mirror descent (complementary direction)
    Phase 4: Gradient-guided coordinate descent (refinement)
    Phase 5: Peak shaving (targeted final refinement)
    Phase 6: Second optimization round with different parameters
    Phase 7: Try alternative length
    Phase 8: Final polish
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        # Use a raised cosine as default
        x = np.linspace(0, 1, n_base)
        best_seq = (0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Phase 1: Spectral shaping (fast, multiple alphas)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={len(best_seq)} ===")
    for alpha_val in [0.5, 0.3, 0.15]:
        a_out = spectral_shaping(best_seq, n_iter=250, alpha=alpha_val)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  Spectral alpha={alpha_val}: {best_score:.6f}")
    print(f"  After Phase 1: best={best_score:.6f}")

    # Phase 2: Subgradient Nesterov-BB (primary)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: SubGrad-Nesterov-BB (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.28, 480.0)
    if t2 > 30:
        a_out, _ = subgradient_nesterov_bb(best_seq, t2, T_start=200.0, T_end=0.05, n_stages=35)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [SubGrad-BB]: {best_score:.6f}")
    print(f"  After Phase 2: best={best_score:.6f}")

    # Phase 3: Mirror descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Mirror Descent (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_out, _ = mirror_descent_optimize(best_seq, t3, T_start=80.0, T_end=0.05, n_stages=25)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [MirrorDesc]: {best_score:.6f}")
    print(f"  After Phase 3: best={best_score:.6f}")

    # Phase 4: Gradient-guided coordinate descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Gradient Coord Descent (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 15:
        a_out, _ = gradient_coord_descent(best_seq, t4, n_rounds=300)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [CoordDesc]: {best_score:.6f}")
    print(f"  After Phase 4: best={best_score:.6f}")

    # Phase 5: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Peak Shaving (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 10:
        a_out, _ = peak_shaving_optimize(best_seq, t5, seed=123)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [PeakShave1]: {best_score:.6f}")
    print(f"  After Phase 5: best={best_score:.6f}")

    # Phase 6: Second round of subgradient (lower temperatures)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: SubGrad Round 2 (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.18, 280.0)
    if t6 > 20:
        a_out, _ = subgradient_nesterov_bb(best_seq, t6, T_start=50.0, T_end=0.01, n_stages=30)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [SubGrad2]: {best_score:.6f}")
    print(f"  After Phase 6: best={best_score:.6f}")

    # Phase 7: Try alternative length
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_base = len(best_seq)
    n_alt = int(round(n_base * 1.15))
    print(f"\n=== Phase 7: Try n={n_alt} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 280.0)
    if t7 > 30:
        init_alt = interpolate_sequence(best_seq, n_alt)
        # Quick spectral at new length
        a_spec = spectral_shaping(init_alt, n_iter=150, alpha=0.25)
        sc = internal_score(a_spec)
        if sc < best_score:
            best_score = sc
            best_seq = a_spec.copy()
            print(f"  *** NEW BEST [spectral_n{n_alt}]: {best_score:.6f}")
        # Subgradient at new length
        t7b = t7 * 0.6
        if t7b > 20:
            a_out, _ = subgradient_nesterov_bb(best_seq, t7b, T_start=100.0, T_end=0.05, n_stages=25)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [SubGrad_n{n_alt}]: {best_score:.6f}")
    print(f"  After Phase 7: best={best_score:.6f}")

    # Phase 8: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        print(f"\n=== Phase 8: Final Peak Shaving n={len(best_seq)} ===")
        t8 = min(remaining * 0.35, 100.0)
        if t8 > 5:
            a_out, _ = peak_shaving_optimize(best_seq, t8, seed=789)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [PeakShave2]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
