    A = rfft(a, fft_size)

If I think of the problem in the frequency domain: the autocorrelation is the inverse FFT of |FFT(a)|^2. So minimizing the peak of the autocorrelation is equivalent to finding a sequence whose power spectrum is as flat as possible (white noise). But a is non-negative, which constrains the possible power spectra.

For a non-negative sequence, the power spectrum is |sum a[k] e^{-2πikf/n}|^2 = |A(f)|^2 where A(f) is the DTFT of a. We want |A(f)|^2 to be as flat as possible.

For a uniform sequence, A(f) is a sinc function, which has large sidelobes. For an optimal sequence, we want A(f) to be close to constant.

This is related to the "flat top" filter design problem. The optimal non-negative sequence with the flattest spectrum might be achievable through iterative spectrum flattening:
1. Compute A(f) = FFT(a)
2. Compute the desired magnitude |A(f)|_target = sqrt(mean(|A(f)|^2)) = constant
3. Update: a_new = IFFT(A(f) * |A(f)|_target / |A(f)|) = IFFT(A(f) * c/|A(f)|) where c is the target magnitude
4. Project a_new to be non-negative
5. Repeat

This is the "spectral factorization" or "CEM" (Correlation Error Minimization) approach used in radar wavelet design!

The CEM algorithm:
1. Initialize a (non-negative)
2. Compute the correlation error: e[k] = b[k] - d[k] where d is the desired autocorrelation (e.g., delta at 0, small elsewhere)
3. Update: a_new = a + mu * IFFT(FFT(a) * conj(FFT(e))) (or something like that)
4. Project to non-negative and normalize

Actually, the standard CEM is:
1. a_0 = random or structured init
2. For each iteration:
   a. Compute b = autocorr(a)
   b. Compute error: e = b - b_target (where b_target is the desired autocorrelation)
   c. a_new = a + mu * ifft(fft(a) * conj(fft(e)))
   d. "Cem" step: minimize ||a_new - a_old||^2 subject to the autocorrelation constraint
   e. a = a_new (or a = projected a_new)

The "CEM" step solves: min ||a - a_init||^2 s.t. autocorr(a) <= b_target
This is a QP that can be solved with CVXPY!

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_peak_gradient(a, k_star):
    """
    Compute subgradient of b[k_star] w.r.t. a.
    db[k]/da[m] = 2*a[k-m] if 0<=k-m<n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    k = k_star
    if k < n:
        # m = 0..k: grad[m] = 2*a[k-m]
        grad[0:k+1] = 2.0 * a[k::-1]
    else:
        # m = k-n+1..n-1: grad[m] = 2*a[k-m]
        start = k - n + 1
        if start < n:
            # k-m ranges from k-start=n-1 down to k-(n-1)
            grad[start:] = 2.0 * a[k-(n-1):n][::-1]
    return grad


def multiplicative_subgrad(a0, time_limit, eta0=None, decay=0.9995,
                           use_multi_peak=False, n_peaks=3, verbose=True):
    """
    Multiplicative subgradient descent using exact peak (or top-K peaks).
    Update: a[m] *= exp(-eta * grad[m]), then normalize.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if eta0 is None:
        eta0 = 1.0 / max(1.0, a.max() * n * 0.1)

    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    eta = eta0
    it = 0
    patience = 0
    patience_limit = 5000

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)

        if use_multi_peak and n_peaks > 1:
            # Use top-K peaks with equal weighting
            top_k = np.argsort(b)[::-1][:n_peaks]
            grad = np.zeros(n)
            for k in top_k:
                grad += compute_peak_gradient(a, k)
            grad /= len(top_k)
        else:
            k_star = int(np.argmax(b))
            grad = compute_peak_gradient(a, k_star)

        # Multiplicative update
        log_a = np.log(a + 1e-30) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        b_new = fast_autocorr(a_new)
        peak_new = b_new.max()

        if peak_new < best_peak:
            best_peak = peak_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new
        eta *= decay
        it += 1

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MulSubGrad it={it} eta={eta:.6f} ({elapsed:.0f}s): best={best_peak:.8f}")

        if patience > patience_limit and it > 1000:
            # Restart with smaller eta
            eta *= 0.5
            patience = 0

    return best_a, best_peak


def lse_multiplicative(a0, time_limit, T_start=5.0, T_end=0.1, n_stages=15,
                       eta_scale=1.0, verbose=True):

    Multiplicative descent on LSE (log-sum-exp) surrogate.
    Gradient: sum_k w[k] * 2*a[k-m] where w ~ softmax(b/T)
    Computed via FFT: grad = 2 * ifft(fft(a) * fft(w_padded))  (appropriately shifted)

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    it = 0

    temps = np.geomspace(T_start, T_end, n_stages)
    sizes = 2 * n - 1

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_iters_stage = max(100, int(remaining * 2.5 / max(1, n_stages - si)))

        for it_s in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            b = fast_autocorr(a)
            max_b = b.max()
            w = np.exp((b - max_b) / T)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(sizes)
                w[n - 1] = 1.0
            else:
                w /= w_sum

            # grad[m] = sum_k w[k] * 2*a[k-m] = 2 * (w * a)[m] (cross-correlation)
            # (w * a)[m] = sum_k w[k] * a[k-m]

            # = convolve(w, a[::-1])[m + n - 1]... let me use FFT directly.

            # = ifft(fft(w, N) * fft(a, N))[m] for appropriate N
            fft_size = 1 << (sizes + n - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # We want sum_k w[k]*a[k-m]: this is (w * a_flipped)[m + n - 1]
            # Actually: sum_k w[k]*a[k-m] = sum_j w[j+m]*a[j] (substituting j=k-m)
            # This is the cross-correlation of w and a, which equals conv(w, a[::-1]) shifted.
            # Let me just compute it directly using the formula:
            # grad[m] = 2 * sum_k w[k] * a[k-m] (where a valid in [0,n))
            # For m in [0,n): k-m in [0,n) means k in [m, m+n)

            # Let v[m] = sum_j w[m+j]*a[j] for j=0..n-1 (with w[k]=0 for k>=2n-1)
            # v = ifft(fft(w, N) * conj(fft(a, N))) ... no, that's cross-correlation
            # Actually v[m] = sum_j a[j]*w[m+j] = (a * w)[m] in the "sliding window" sense
            # = cross_correlation(a, w, mode='valid' reversed)...
            # Simplest: v = irfft(rfft(w, fft_size) * rfft(a, fft_size)) gives (w*a)[k] = sum w[j]*a[k-j]
            # We want sum_k w[k]*a[k-m] = sum_j w[m+j]*a[j]
            # Let u[j] = a[j], v[j] = w[m+j]. This is <u, v_shifted>
            # = conv(u, v_rev)[m + n - 1] where v_rev is reversed...
            # OK let me just use the direct FFT approach:
            # sum_k w[k]*a[k-m] = (ifft(fft(w)*fft(a)))[offset]...
            # Actually the cleanest: define h[k] = w[k] for k in [0, 2n-2], 0 otherwise.
            # Then sum_k h[k]*a[k-m] = (h * a_flipped)[m + n - 1]
            # where a_flipped = a[::-1], and * denotes linear convolution.
            A_rev = rfft(a[::-1], fft_size)
            conv = irfft(W * A_rev, fft_size)
            # conv[m + n - 1] = sum_j h[j]*a_rev[n-1-j] = sum_j h[j]*a[j-(n-1-j)]... 
            # This is getting confusing. Let me just use np.correlate for clarity (slow but correct)

            conj_A = np.conj(rfft(a, fft_size))
            cross = irfft(W * conj_A, fft_size)
            grad = 2.0 * cross[:n]

            # Multiplicative update
            step_eta = eta_scale / (it_s + 10) ** 0.3
            log_a = np.log(a + 1e-30) - step_eta * grad

            a = a_new
            it += 1

            peak = fast_autocorr(a).max()
            if peak < best_peak:
                best_peak = peak
                best_a = a.copy()

            if verbose and it % 3000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE-Mul stage={si}/{n_stages} T={T:.3f} it={it} ({elapsed:.0f}s): best={best_peak:.8f}")

def ica_peak_shave(a0, time_limit, verbose=True):
    """
    ICA-style peak shaving: iteratively find peak-contributing pairs and reduce them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 200:
        b = fast_autocorr(a)
        peak = b.max()
        k_star = int(np.argmax(b))

        # Find pairs (i, j) with i+j = k_star (or i-j = k_star for negative lag contribution)
        # b[k_star] = sum_i a[i]*a[k_star-i]
        # The main contributors are where a[i]*a[k_star-i] is large
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        pairs = a[idx] * a[k_star - idx]
        
        # Find the top contributing pair
        top_idx = idx[np.argmax(pairs)]
        top_j = k_star - top_idx
        top_val = pairs.max()
        
        if top_val < 1e-15:
            break

        # Try reducing the larger of the two
        if a[top_idx] >= a[top_j]:
            src = top_idx
            dst = top_j
        else:
            src = top_j
            dst = top_idx

        # Try several reduction fractions
        improved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            new_peak = fast_autocorr(a_trial).max()
            if new_peak < best_peak - 1e-12:
                a = a_trial
                best_peak = new_peak
                best_a = a.copy()
                improved = True
                stall = 0
                it += 1
                break

        if not improved:
            stall += 1
            # Try random perturbation
            if stall % 10 == 0:
                for _ in range(5):
                    i = np.random.randint(n)
                    j = np.random.randint(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = np.random.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    new_peak = fast_autocorr(a_trial).max()
                    if new_peak < best_peak:
                        a = a_trial
                        best_peak = new_peak
                        best_a = a.copy()
                        stall = 0
                        break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ICA it={it} ({elapsed:.0f}s): best={best_peak:.8f}, stall={stall}")

    return best_a, best_peak


def coordinate_descent(a0, time_limit, n_sweeps=None, verbose=True):
    """
    Cyclic coordinate descent with 1D line search.
    For each m, find optimal a[m] given all others fixed.
    Uses the quadratic form: b[k] changes by 2*d*a[k-m] + d^2*delta(k-2m)
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        if n_sweeps is not None and sweep >= n_sweeps:
            break
            
        b = fast_autocorr(a)
        peak = b.max()
        
        for m in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            
            x = a[m]
            if x < 1e-12:
                a[m] = 0 if x < 1e-14 else x
                continue

            # Try reducing a[m] (since subgradient is positive when a[k*-m] > 0)
            # Quick check: what's the gradient at m for the current peak?
            k_star = int(np.argmax(b))
            km = k_star - m
            if 0 <= km < n:
                grad_val = 2.0 * a[km]
            else:
                # Check other peaks too
                grad_val = 0.0
                for k in np.argsort(b)[::-1][:5]:
                    km2 = k - m
                    if 0 <= km2 < n:
                        grad_val += 2.0 * a[km2]
                    km3 = m - k  # Not needed, we showed it's just 2*a[k-m]
                    # Actually we only need k-m, not m-k
                    break
                if grad_val == 0:
                    continue

            if grad_val <= 0:
                continue

            # Try several reduction amounts
            best_d = 0
            best_local_peak = peak
            # The change in b[k] for delta d at position m:
            # b[k] -> b[k] + 2*d*a[k-m] + d^2*delta(k-2m)
            # We want to minimize max_k b_new[k]
            # Try negative d (reduction)
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
                d = -frac * x
                # Compute new peak efficiently
                # Only lags k where a[k-m] > 0 are affected
                # b_new[k] = b[k] + 2*d*a[k-m] + d^2*delta(k-2m)
                # For the peak k_star: b_new[k_star] = b[k_star] + 2*d*a[k_star-m]
                # Since d < 0 and a[k_star-m] > 0, this reduces b[k_star]
                # But might increase other lags... unlikely since d<0 reduces all affected b[k]
                # (because a[k-m] >= 0)
                # The only concern is the d^2 term at k=2m
                if d < 0:
                    # All affected b[k] decrease (since 2*d*a[k-m] < 0)
                    # except possibly the d^2 term at k=2m which adds d^2 > 0
                    # Net change at k=2m: 2*d*a[m] + d^2 = d*(2*a[m] + d)
                    # Since d = -frac*x and 2*a[m]+d = 2x - frac*x = x(2-frac) > 0 for frac<2
                    # So net change at k=2m is d*x*(2-frac) = -frac*x^2*(2-frac) < 0
                    # So all affected lags decrease!
                    # The new peak is max of: (old peak - reduction) and (unaffected lags)
                    # Simple bound: new_peak <= old_peak (since all affected lags decrease)
                    # For efficiency, just check the top few lags
                    a_trial = a.copy()
                    a_trial[m] = max(0, x + d)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    new_peak = fast_autocorr(a_trial).max()
                    if new_peak < best_local_peak:
                        best_local_peak = new_peak
                        best_d = d

            if best_d != 0 and best_local_peak < best_peak:
                a[m] = max(0, x + best_d)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                best_peak = best_local_peak
                best_a = a.copy()
                b = fast_autocorr(a)  # Update b for next iterations

        sweep += 1
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc sweep={sweep} ({elapsed:.0f}s): best={best_peak:.8f}")

    return best_a, best_peak


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 4
    elif choice == 5:
        # Raised cosine with taper
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 6:
        # Product of two Gaussians
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-10*(x-0.25)**2) * np.exp(-10*(x-0.75)**2) + 0.1
    else:
        # Random smooth
        a = np.ones(n) * 0.3
        for _ in range(rng.integers(3, 7)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.25)
            amp = rng.uniform(-0.2, 0.4)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
