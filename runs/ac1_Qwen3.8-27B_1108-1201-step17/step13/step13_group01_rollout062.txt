"""

2. LSE (log-sum-exp) smooth surrogate with L-BFGS-B (polishing with corrected gradient)

def fast_autocorr(a):
    """Compute full autocorrelation (self-convolution) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:

        a_list = [float(x) for x in a_arr]

def normalize_to_sum(a, target_sum):
    """Normalize a to have the given sum."""
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s


def lp_best_response(a, M, solver):
    """Solve LP best response: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    b_var = cvx.Variable(n, nonneg=True)
    
    # Build constraint matrix T: T[k, i] = a[k-i] if valid
    T = np.zeros((size, n), dtype=np.float64)
    for i in range(n):
        start = i
        end = min(i + n, size)
        T[start:end, i] = a[i:end-i]
    
    constraints = [T @ b_var <= M]
    prob = cvx.Problem(cvx.Maximize(cvx.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iterations."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stall_count += 1
            if stall_count > 50:
                break
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = normalize_to_sum(b, norm)
        n_iters += 1
        stall_count = 0
        
        # Adaptive step size
        alpha = 2.0 / (t + 2.0)
        
        # Nesterov momentum
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to_sum(y_new, norm)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_sum(a_new, norm)
        
        y = y_new
        a = a_new
        
        # Track best
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
        
        # Check average
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = normalize_to_sum(avg_a / n_avg, norm)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg

                last_improve = time.monotonic()
        
        # Stall detection

            stall_count = 0
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0

    return best_a, best_score


def lse_objective(a_flat, n, T):
    """LSE smooth approximation of the objective."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    exp_vals = np.exp(T * (b - max_b))
    lse = max_b + np.log(exp_vals.sum()) / T
    return 2.0 * n * lse / (s * s)


def lse_gradient(a_flat, n, T):
    """Corrected gradient of the LSE objective."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    
    exp_vals = np.exp(T * (b - max_b))
    w = exp_vals / exp_vals.sum()  # softmax weights
    lse = max_b + np.log(exp_vals.sum()) / T
    
    # Compute X[m] = sum_k w[k] * a[k-m] via FFT correlation
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    
    # X[m] = sum_k a[k] * w[k+m] = correlation
    cross = irfft(np.conj(A_fft) * W_fft, fft_size)[:n]
    
    # dLSE/da[m] = 2 * X[m]
    # df/da[m] = 4*n*X[m]/S^2 - 4*n*LSE*a[m]/S^3
    conv_grad = 2.0 * cross  # dLSE/da[m]
    
    grad = 2.0 * n * conv_grad / (s * s) - 4.0 * n * lse * a / (s * s * s)
    return grad.ravel()


def lse_lbfgs_optimize(a0, time_limit, T_stages=None):
    """Optimize LSE objective with L-BFGS-B, multi-stage temperature."""
    if T_stages is None:

    t0 = time.monotonic()
    
    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, len(T_stages) - si), 120.0)
        
        x0 = a.copy()
        
        def obj_func(x):
            return lse_objective(x, n, T)
        
        def grad_func(x):
            return lse_gradient(x, n, T)
        
        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, 1000)] * n,
                                  options={'maxiter': 300, 'maxfun': 1000,
                                           'ftol': 1e-15, 'gtol': 1e-12})
            if res.x is not None:

def subgradient_polish(a0, time_limit, n_iter=50000):
    """Polish using exact subgradient of the max."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    
    for t in range(1, n_iter + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(y)
        # Find all lags achieving (near) the max
        max_b = b.max()
        # Use top few peaks for robustness
        threshold = max_b * 0.999
        peak_indices = np.where(b >= threshold)[0]
        
        # Average subgradient over peak lags
        grad = np.zeros(n)
        for k_star in peak_indices:
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad[i] += 2.0 * y[j]
        grad /= len(peak_indices)
        
        # Normalize gradient step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Step size (Polyak-like)
        alpha = 1.0 / (t ** 0.5) * 0.1
        
        # Take step in negative gradient direction, project to simplex-like
        v = y - alpha * grad / grad_norm
        v = np.maximum(v, 0.0)
        v = normalize_to_sum(v, norm)
        
        # Nesterov extrapolation
        if t == 1:
            a_new = v
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = v + beta * (v - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_sum(a_new, norm)
        
        y = v
        a = a_new
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
    
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """Coordinate descent focused on reducing peak autocorrelation."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, norm)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        # Get top peaks
        top_peaks = np.argsort(b)[::-1][:6]
        
        improved = False
        
        # Try moving mass from high-contribution positions to low-peak positions
        for k_peak in top_peaks:
            # Positions that contribute to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = np.zeros(n)
            for i in idx:
                contrib[i] = a[i] * a[k_peak - i]
            
            # Source: position with highest contribution
            src_candidates = idx[np.argsort(contrib[idx])[::-1][:3]]
            
            # Destination: positions that don't contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True
            
            dst_candidates = np.where(~peak_mask)[0]
            if len(dst_candidates) == 0:
                dst_candidates = np.where(~peak_mask | (np.arange(n) != src_candidates[0]))[0]
            
            # Try moving from src to dst
            for src in src_candidates:
                if a[src] < 1e-10:
                    continue
                # Sort destinations by how "safe" they are (low autocorr contribution)
                for dst in dst_candidates[np.argsort(a[dst_candidates])][:5]:
                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        a_trial = normalize_to_sum(a_trial, norm)
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 5 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = normalize_to_sum(a_trial, norm)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
            if stall > 300:
                break
    
    return best_a, best_score
