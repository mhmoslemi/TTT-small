def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = np.max(u[np.where(cond)[0]])
    return np.maximum(v - rho, 0.0)


def compute_excess_subgrad(a, ac, target, top_k=10, beta=2.0):
    """Subgradient weighted by excess above target (focused on violators)."""
    n = len(a)
    excess = np.maximum(ac - target, 0.0)
    if excess.sum() < 1e-30:
        # Fall back to top-k of ac
        k = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k]
        w = ac[top_idx].copy()
        w = np.maximum(w, 1e-30)
        w /= w.sum()
    else:
        # Use top-k excess lags
        k = min(top_k, len(ac))
        sorted_excess = np.sort(excess)[::-1][:k]
        if sorted_excess[0] < 1e-30:
            top_idx = np.argsort(ac)[::-1][:k]
            w = ac[top_idx].copy()
            w = np.maximum(w, 1e-30)
            w /= w.sum()
        else:
            threshold = sorted_excess[-1]
            top_idx = np.where(excess >= threshold)[0]
            w = excess[top_idx] ** beta
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(len(top_idx)) / len(top_idx)
            else:
                w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = ki - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad


def compute_softmax_subgrad(a, ac, top_k=8, temperature=0.01):
    """Subgradient using softmax weighting over top-k lags."""
    n = len(a)
    k = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k]
    top_vals = ac[top_idx].copy()
    # Softmax with temperature
    scaled = (top_vals - top_vals.max()) / max(temperature, 1e-10)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        ki = int(k)
        i_start = max(0, ki - n + 1)
        i_end = min(n, ki + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = ki - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad


def mirror_descent_optimize(a0, time_limit, verbose=True, top_k=8, beta_start=3.0, beta_end=1.0):

            break

        # Adaptive temperature: anneal from high to low

        beta = beta_start + (beta_end - beta_start) * progress
        temp = 0.05 * (1.0 - 0.8 * progress)  # Temperature decreases

        step = eta / (np.sqrt(t) + 1.0)

        ac = fast_autocorr(a)
        M = ac.max()
        target = best_m  # Use current best as target for excess

        # Use excess-based subgradient
        g = compute_excess_subgrad(a, ac, target, top_k=top_k, beta=beta)

        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:

        ac_new = fast_autocorr(a)
        M_new = ac_new.max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg

            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    # Build constraint matrix efficiently
    # (a*b)_k = sum_j a_max(0,k-n+1,min(n-1,k)) ... a_{k-j} * b_j
    # Use convolution structure
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j - n + 1)  # not needed for conv
        # (conv(a,b))[k] = sum_j a[j] * b[k-j] for valid k-j
        # So T[k,j] = a[j] if 0 <= k-j < n, else 0
        valid_k = np.arange(max(0, j), min(2*n-1, j+n))
        T[valid_k, j] = a[j]

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_best_response_fft(a, M, solver):
    """LP best response using sparse constraint formulation."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    # (conv(a,b))[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j]*b[k-j]
    # For each k, the constraint is: sum_j a[j]*b[k-j] <= M
    constraints = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        coeffs = a[js]
        constraints.append(coeffs @ b_var[js] <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        a = a * c / s0

    x = a.copy()

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-15:
            break

            b = lp_best_response(y, M_curr, solver)

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:

        else:

            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x

        y = y_new
        x = x_new

        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()

                    break
                ki = int(k)
                js = max(0, ki - n + 1)
                je = min(n - 1, ki)
                idx = np.arange(js, je + 1)
                contrib = a[idx] * a[ki - idx]
                if contrib.max() < 1e-20:
                    continue
                src = idx[np.argmax(contrib)]

                # Find good destinations: positions NOT heavily involved in top peaks
                peak_count = np.zeros(n)

                    js2 = max(0, k2i - n + 1)

                # Candidates: low peak_count AND low current value
                scores = peak_count * (a + 1e-10)
                mask = np.ones(n, dtype=bool)
                mask[src] = False
                candidates = np.where(mask)[0]

                # Sort by score (ascending) - prefer low-impact positions

                            improved = True

                        break

            if not improved:

        print(f"    Sym polish ({elapsed:.1f}s): best_J={J:.6f}")

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def generate_init(n, seed, mode):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    if mode == 0:  # Cosine window
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif mode == 1:  # U-shaped
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.5 * x ** 2
    elif mode == 2:  # W-shaped
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x)
    elif mode == 3:  # Two-peak
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.2) ** 2) / 0.02) + np.exp(-((x - 0.8) ** 2) / 0.02) + 0.3
    elif mode == 4:  # Random smooth
        a = np.ones(n)
        for f in range(3):
            freq = rng.uniform(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.4)
            x = np.linspace(0, 1, n)
            a += amp * np.cos(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()

    except Exception:

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = generate_init(n_base, 42, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)
    c0 = np.sqrt(2.0 * n0)

    # === Phase 1: Multi-start mirror descent at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-start MD n={n0} (remaining={remaining:.0f}s) ===")
    if remaining > 50:
        t1_total = min(remaining * 0.12, 180.0)
        t1_each = t1_total / 4
        if t1_each > 10:
            # Start from parent
            a_norm = best_seq * c0 / best_seq.sum()
            a_out, _ = mirror_descent_optimize(a_norm, t1_each, verbose=True)
            try_update(a_out, f"md_parent_n{n0}")

            # Start from cosine
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_init = generate_init(n0, 1, 0)
                a_init = a_init * c0 / a_init.sum()
                a_out, _ = mirror_descent_optimize(a_init, t1_each * 0.7, verbose=False)
                try_update(a_out, f"md_cosine_n{n0}")

            # Start from W-shaped
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_init = generate_init(n0, 2, 2)
                a_init = a_init * c0 / a_init.sum()
                a_out, _ = mirror_descent_optimize(a_init, t1_each * 0.7, verbose=False)
                try_update(a_out, f"md_wshaped_n{n0}")

            # Start from random smooth
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 15:
                a_init = generate_init(n0, 3, 4)
                a_init = a_init * c0 / a_init.sum()
                a_out, _ = mirror_descent_optimize(a_init, t1_each * 0.5, verbose=False)
                try_update(a_out, f"md_random_n{n0}")

    # === Phase 2: LP Nesterov at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP Nesterov n={n0} (remaining={remaining:.0f}s) ===")
    if remaining > 40:
        t2 = min(remaining * 0.20, 350.0)
        if t2 > 30:
            c0 = np.sqrt(2.0 * len(best_seq))
            a_norm = best_seq * c0 / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
            try_update(a_out, f"lp_n{n0}")

    # === Phase 3: Coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Coord polish n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    if remaining > 20:
        t3 = min(remaining * 0.15, 250.0)
        if t3 > 15:
            c_best = np.sqrt(2.0 * len(best_seq))
            a_norm = best_seq * c_best / best_seq.sum()
            a_out, _ = coordinate_polish(a_norm, t3, verbose=True)
            try_update(a_out, f"coord_n{len(best_seq)}")

    # === Phase 4: Symmetric polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Sym polish n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    if remaining > 15:
        t4 = min(remaining * 0.08, 120.0)
        if t4 > 10:
            c_best = np.sqrt(2.0 * len(best_seq))
            a_norm = best_seq * c_best / best_seq.sum()
            a_out, _ = symmetric_polish(a_norm, t4, verbose=True)
            try_update(a_out, f"sym_n{len(best_seq)}")

    # === Phase 5: Scale up to n=2400 ===
    n1 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Scale to n={n1} (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        a_long = interpolate_sequence(best_seq, n1)
        c1 = np.sqrt(2.0 * n1)
        a_norm = a_long * c1 / a_long.sum()

        # Mirror descent at n1
        t5a = min(remaining * 0.12, 200.0)
        if t5a > 15:
            a_out, _ = mirror_descent_optimize(a_norm, t5a, verbose=True)
            try_update(a_out, f"md_n{n1}")

        # LP at n1
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            t5b = min(remaining * 0.12, 200.0)
            if t5b > 30:
                c1 = np.sqrt(2.0 * n1)
                a_norm = best_seq * c1 / best_seq.sum() if len(best_seq) == n1 else interpolate_sequence(best_seq, n1) * c1 / interpolate_sequence(best_seq, n1).sum()
                a_long = interpolate_sequence(best_seq, n1)
                a_norm = a_long * c1 / a_long.sum()
                a_out, _ = nesterov_lp_optimize(a_norm, t5b, solver, verbose=True)
                try_update(a_out, f"lp_n{n1}")

        # Coord at n1
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t5c = min(remaining * 0.10, 150.0)
            if t5c > 10:
                c1 = np.sqrt(2.0 * n1)
                a_norm = best_seq * c1 / best_seq.sum() if len(best_seq) == n1 else interpolate_sequence(best_seq, n1) * c1 / interpolate_sequence(best_seq, n1).sum()
                a_out, _ = coordinate_polish(a_norm, t5c, verbose=True)
                try_update(a_out, f"coord_n{n1}")

    # === Phase 6: Final intensive polish at best length ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final polish n={n_best} (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()

        # More MD with different params
        t6a = min(remaining * 0.20, 200.0)
        if t6a > 15:
            a_out, _ = mirror_descent_optimize(a_norm, t6a, top_k=12, beta_start=2.0, beta_end=0.5, verbose=True)
            try_update(a_out, f"md_aggr_n{n_best}")

        # Final coord
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t6b = min(remaining * 0.40, 300.0)
            if t6b > 10:
                c_best = np.sqrt(2.0 * n_best)
                a_norm = best_seq * c_best / best_seq.sum()
                a_out, _ = coordinate_polish(a_norm, t6b, verbose=True, num_seeds=3)
                try_update(a_out, f"coord_final_n{n_best}")

        # Final sym
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t6c = min(remaining * 0.20, 100.0)
            if t6c > 8:
                c_best = np.sqrt(2.0 * n_best)
                a_norm = best_seq * c_best / best_seq.sum()
                a_out, _ = symmetric_polish(a_norm, t6c, verbose=True)
                try_update(a_out, f"sym_final_n{n_best}")

        # Very final coord (last resort)
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t6d = min(remaining * 0.50, 100.0)
            if t6d > 5:
                c_best = np.sqrt(2.0 * n_best)
                a_norm = best_seq * c_best / best_seq.sum()
                a_out, _ = coordinate_polish(a_norm, t6d, verbose=True, num_seeds=2)
                try_update(a_out, f"coord_last_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
