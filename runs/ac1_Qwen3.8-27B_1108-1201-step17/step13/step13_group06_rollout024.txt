def build_constraint_matrix(y, constraint_indices):

    n = len(y)
    S = len(constraint_indices)
    T = np.zeros((S, n))
    for i, k in enumerate(constraint_indices):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            T[i, m_start:m_end+1] = y[k:m_start-1:-1]
    return T

            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start > m_end:
                continue
            positions = np.arange(m_start, m_end + 1)
            contrib = 2.0 * a[k - positions]
            
            # Find the most contributing positions (sources)
            top_src = positions[np.argsort(contrib)[::-1][:10]]
            
            # Find safe destinations (low contribution to ALL top peaks)
            danger = np.zeros(n)
            for k2 in top_k[:15]:
                ms = max(0, k2 - n + 1)
                me = min(n - 1, k2)
                if ms <= me:
                    danger[ms:me+1] += a[k2 - np.arange(ms, me+1)]
            
            # Safe positions: low danger and != source
            for src in top_src[:5]:
                # Find best destination
                dest_scores = danger.copy()
                dest_scores[src] = np.inf
                # Prefer low positions
                dest_sorted = np.argsort(dest_scores)
                
                for dst in dest_sorted[:10]:
                    if dst == src:
                        continue
                    # Compute delta R[k] = 2*delta*(a[k-dst] - a[k-src])
                    # We need: R[k] + 2*delta*(a[k-dst] - a[k-src]) <= M for all k
                    # delta_max = min over k where a[k-dst] > a[k-src] of (M - R[k]) / (2*(a[k-dst] - a[k-src]))
                    
                    # Compute the constraint
                    # For each k, the coefficient is 2*(a[k-dst] - a[k-src])
                    # This is a vector of length 2n-1
                    coeff = np.zeros(2*n-1)
                    # a[k-dst] is non-zero when 0<=k-dst<n, i.e., dst<=k<=dst+n-1
                    if dst < 2*n-1:

                    # a[k-src] is non-zero when 0<=k-src<n, i.e., src<=k<=src+n-1  
                    if src < 2*n-1:
                        coeff[src:min(src+n, 2*n-1)] -= a[0:min(n, 2*n-1-src)]
                    
                    # For the peak to decrease: coeff[k] < 0 at the max peaks
                    if coeff[k] >= 0:
                        continue  # This move won't reduce peak k
                    
                    # Find delta_max
                    positive_mask = coeff > 1e-15
                    if np.any(positive_mask):
                        delta_max = ((M - R[positive_mask]) / (2.0 * coeff[positive_mask])).min()
                    else:
                        delta_max = a[src]  # Can move all mass
                    
                    delta_max = min(delta_max, a[src] * 0.5)  # Limit to 50% of source mass
                    if delta_max <= 1e-12:
                        continue
                    
                    # Try the move
                    a_trial = a.copy()
                    a_trial[src] -= delta_max
                    a_trial[dst] += delta_max
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-12:
                        continue
                    a_trial = a_trial / s_t
                    
                    R_trial = fast_autocorr(a_trial)
                    M_trial = R_trial.max()

                        R = R_trial

                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_a, best_score

Actually, let me reconsider the constraint generation approach more carefully.

In the Nesterov game, at iteration t:
- We have y_t (the "row player's" strategy).
- We solve the LP: min M s.t. conv(y_t, b)[k] <= M for k in S_t, sum(b) = 1, b >= 0.
- The optimal b_t is the best response.
- We update y_{t+1} = (1-α) y_t + α b_t.

The constraint set S_t should be the set of "active" constraints, i.e., those k where conv(y_t, b_t)[k] is close to M_t (the optimal value). In the full LP, all 2n-1 constraints are present, and the optimal M is the maximum of conv(y_t, b_t)[k]. With constraint generation, we only use a subset, so the LP optimum M_cg might be smaller than the true optimum (since we're ignoring some constraints).

To handle this:
1. After solving the LP, compute conv(y_t, b_t)[k] for ALL k (via FFT).
2. If max_k conv(y_t, b_t)[k] > M_cg + tolerance, then we have violated constraints.
3. Add the most violated constraints to S_{t+1}.
4. The next LP iteration will be tighter.

This is the standard constraint generation / cutting plane approach. It should converge to the same solution as the full LP, but with faster individual LP solves.

The key question is: how many constraints do we need? In practice, the number of binding constraints at the optimum is at most n (the number of variables) + 1 (the M variable) = n+1. So we should never need more than n+1 active constraints. In practice, the active set is much smaller (maybe 10-50 constraints).

def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n1, n2 = len(a), len(b)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    mask = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(mask):
        return np.ones(n) / n
    rho = np.nonzero(mask)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_constraint_matrix(y, constraint_indices):
    """Build |S| x n sparse-ish matrix where row k has y[k-m] at column m."""
    n = len(y)

        m_end = min(n - 1, k)
        if m_start <= m_end:

    return T

        M = R.max()

        # Find top peaks
        n_peaks = min(20, 2 * n - 1)
        top_peaks = np.argsort(R)[::-1][:n_peaks]

        # Compute "danger" score for each position (contribution to top peaks)
        danger = np.zeros(n)
        for k in top_peaks[:15]:
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start <= m_end:
                pos = np.arange(m_start, m_end + 1)
                danger[pos] += a[k - pos]

        # Find candidate sources (high danger) and destinations (low danger)
        src_candidates = np.argsort(danger)[::-1][:30]
        dst_candidates = np.argsort(danger)[:50]

        for src in src_candidates:
            if a[src] < 1e-10:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue

                # Compute coefficient vector: coeff[k] = a[k-dst] - a[k-src]
                # (with 0 for out-of-bounds)
                # coeff[k] is non-zero when 0<=k-dst<n or 0<=k-src<n
                lo = min(max(0, src - n + 1, dst - n + 1), 2 * n - 2)
                hi = max(min(2 * n - 2, src + n - 1, dst + n - 1), 0)
                if lo > hi:
                    continue

                # Evaluate at peak positions first (quick rejection)
                k_star = top_peaks[0]
                val_src = a[k_star - src] if 0 <= k_star - src < n else 0.0
                val_dst = a[k_star - dst] if 0 <= k_star - dst < n else 0.0
                if val_dst >= val_src:
                    continue  # Won't reduce the main peak

                # Full check: find delta_max
                # Need: R[k] + 2*delta*coeff[k] <= M for all k
                # For k where coeff[k] > 0: delta <= (M - R[k]) / (2*coeff[k])

                # Compute coeff only where needed (where R[k] is close to M)
                # Use threshold: only check k where R[k] > M * 0.95
                check_mask = R > M * 0.95
                if not np.any(check_mask):
                    check_mask = np.zeros(2 * n - 1, dtype=bool)
                    check_mask[:5] = True

                check_indices = np.where(check_mask)[0]
                delta_max = a[src]  # Upper bound

                feasible = True
                for k in check_indices:
                    v_dst = a[k - dst] if 0 <= k - dst < n else 0.0
                    v_src = a[k - src] if 0 <= k - src < n else 0.0
                    c = v_dst - v_src
                    if c > 1e-15:
                        allowed = (M - R[k]) / (2.0 * c)
                        if allowed < delta_max:
                            delta_max = allowed
                        if delta_max <= 1e-12:
                            feasible = False
                            break

                if not feasible or delta_max <= 1e-12:
                    continue

                # Limit step size
                delta_max = min(delta_max, a[src] * 0.4)
                if delta_max <= 1e-12:
                    continue

                # Apply move and verify
                a_trial = a.copy()
                a_trial[src] -= delta_max
                a_trial[dst] += delta_max
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-12:
                    continue
                a_trial = a_trial / s_t

                R_trial = fast_autocorr(a_trial)
                M_trial = R_trial.max()

                if M_trial < M * (1 - 1e-9):
                    a = a_trial
                    R = R_trial
                    M = M_trial
                    new_score = 2.0 * n * M
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try random perturbation to escape
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            n_pert = rng.integers(2, 6)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-10:
                a_trial = a_trial / s_t
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                else:
                    break  # Stuck, stop

    return best_a, best_score


def subgradient_refine(a0, time_limit):
    """
    Projected subgradient method with adaptive step size.
    Minimizes max_k R[k] where R = conv(a,a).
    """
    n = len(a0)
    a = normalize(a0)
    R = fast_autocorr(a)
    M = R.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()

    eta = 0.005
    patience = 0

    while time.monotonic() - t0 < time_limit - 2:
        R = fast_autocorr(a)
        M = R.max()

        # Find the peak and possibly multiple near-peaks
        threshold = M * 0.99
        active_peaks = np.where(R >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([np.argmax(R)])

        # Compute gradient: weighted average of gradients at active peaks
        grad = np.zeros(n)
        weights = R[active_peaks] - M * 0.99 + 0.01  # Emphasize the max
        weights = weights / weights.sum()

        for idx, k in enumerate(active_peaks):
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start <= m_end:
                pos = np.arange(m_start, m_end + 1)
                grad[pos] += weights[idx] * 2.0 * a[k - pos]

        # Projected gradient step
        a_new = a - eta * grad
        a_new = project_simplex(a_new)

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score * (1 - 1e-10):
            a = a_new
            best_score = score_new
            best_a = a.copy()
            patience = 0
            eta = min(eta * 1.02, 0.02)
        else:
            patience += 1
            eta *= 0.95

        if patience > 200:
            # Reset to best and try smaller eta
            a = best_a.copy()
            eta = 0.002
            patience = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, choice=0):
    """Generate structured initializations."""
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.25) ** 2) + 0.6 * np.exp(-8 * (x - 0.75) ** 2)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 6:
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - np.abs(x) ** 1.5)
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * x ** 2) + np.exp(-50 * (x - 1) ** 2)
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    return normalize(a)


def perturbed_init(a0, rng, scale=0.05):
    """Create a perturbed version of a sequence."""
    a = normalize(a0)
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def local_search_detailed(a0, time_limit):
    """
    More detailed local search trying all top source-dest pairs systematically.
    """
    n = len(a0)
    a = normalize(a0)
    R = fast_autocorr(a)
    M = R.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()

    iteration = 0
    while time.monotonic() - t0 < time_limit - 2:
        iteration += 1
        R = fast_autocorr(a)
        M = R.max()

        # Compute danger for all positions
        top_peaks = np.argsort(R)[::-1][:min(25, 2 * n - 1)]
        danger = np.zeros(n)
        for k in top_peaks[:20]:
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start <= m_end:
                pos = np.arange(m_start, m_end + 1)
                danger[pos] += a[k - pos]

        # Get sorted candidates
        src_sorted = np.argsort(danger)[::-1]
        dst_sorted = np.argsort(danger)

        best_move = None
        best_move_delta = 0

        # Try top 40 sources x top 40 destinations
        for src in src_sorted[:40]:
            if a[src] < 1e-8:
                continue
            for dst in dst_sorted[:40]:
                if dst == src:
                    continue
                # Quick check at main peak
                k0 = top_peaks[0]
                v_src = a[k0 - src] if 0 <= k0 - src < n else 0.0
                v_dst = a[k0 - dst] if 0 <= k0 - dst < n else 0.0
                if v_dst >= v_src:
                    continue

                # Compute delta_max efficiently
                # Only check positions where R is high
                high_R = np.where(R > M * 0.97)[0]
                delta_max = a[src]

                feasible = True
                for k in high_R:
                    v_d = a[k - dst] if 0 <= k - dst < n else 0.0
                    v_s = a[k - src] if 0 <= k - src < n else 0.0
                    c = v_d - v_s
                    if c > 1e-14:
                        allowed = (M - R[k]) / (2.0 * c)
                        if allowed < delta_max:
                            delta_max = allowed
                        if delta_max <= 1e-12:
                            feasible = False
                            break

                if not feasible or delta_max <= 1e-12:
                    continue

                delta_max = min(delta_max, a[src] * 0.3)
                if delta_max > best_move_delta:
                    best_move = (src, dst, delta_max)
                    best_move_delta = delta_max

        if best_move is None:
            break

        src, dst, delta = best_move
        a[src] -= delta
        a[dst] += delta
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-12:
            break
        a = a / s

        R = fast_autocorr(a)
        M = R.max()
        new_score = 2.0 * n * M
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
        elif new_score > best_score * 1.001:
            a = best_a.copy()  # Revert

    return best_a, best_score


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization:
    1. LP Nesterov with constraint generation (main global optimizer)
    2. Detailed local search
    3. Subgradient refinement
    4. Multi-length exploration
    5. Final polish
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 120.0, 1650.0)

    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    if best_seq is None:
        # Fallback: structured init
        best_seq = structured_init(1600, rng, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Fallback init: n={len(best_seq)}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov with constraint generation ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov-CG n={n_base} ===")
    t1 = min(remaining * 0.45, 700.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_cg(best_seq, t1, solver, max_active=400, initial_active=100)
        try_update(a_out, "nlp_cg_1")

    # === Phase 2: Second LP Nesterov round (refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: LP Nesterov-CG round 2 ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 40:
        a_out, _ = nesterov_lp_cg(best_seq, t2, solver, max_active=300, initial_active=80)
        try_update(a_out, "nlp_cg_2")

    # === Phase 3: Detailed local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: Local search (detailed) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, _ = local_search_detailed(best_seq, t3)
        try_update(a_out, "local_detailed")

    # === Phase 4: Subgradient refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: Subgradient ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        a_out, _ = subgradient_refine(best_seq, t4)
        try_update(a_out, "subgrad")

    # === Phase 5: Efficient local search (second pass) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 5: Local search (efficient) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 15:
        a_out, _ = local_search_efficient(best_seq, t5)
        try_update(a_out, "local_eff")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        for n_try in [n_base + 100, n_base - 100, n_base + 200]:
            if remaining < 30:
                break
            if n_try < 100:
                continue
            print(f"=== Phase 6: Length exploration n={n_try} ===")
            t6 = min(remaining * 0.15, 120.0)
            if t6 < 20:
                break
            init_long = interpolate_sequence(best_seq, n_try)
            a_out, _ = nesterov_lp_cg(init_long, t6, solver, max_active=300, initial_active=80)
            try_update(a_out, f"len_{n_try}")

            # Quick local search at new length
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20 and a_out is not None:
                t6b = min(remaining * 0.05, 50.0)
                a_out2, _ = local_search_efficient(a_out, t6b)
                try_update(a_out2, f"len_{n_try}_local")

            remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 7: Final polish - LP Nesterov on best found ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"=== Phase 7: Final LP polish ===")
        t7 = min(remaining * 0.3, 100.0)
        if t7 > 20:
            a_out, _ = nesterov_lp_cg(best_seq, t7, solver, max_active=500, initial_active=150)
            try_update(a_out, "final_lp")

    # === Phase 8: Final local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t8 = min(remaining * 0.4, 60.0)
        if t8 > 5:
            print(f"=== Phase 8: Final local search ===")
            a_out, _ = local_search_efficient(best_seq, t8)
            try_update(a_out, "final_local")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
