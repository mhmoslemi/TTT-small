"""

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:

def normalize_to_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def normalize_to_sqrt2n(a):
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * np.sqrt(2.0 * n) / n
    return a * np.sqrt(2.0 * n) / s


def subgradient_at_peak(a, k_star):
    n = len(a)
    sg = np.zeros(n)
    # sg[j] = 2 * a[k_star - j] if 0 <= k_star - j < n
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    for j in range(j_min, j_max + 1):
        sg[j] = 2.0 * a[k_star - j]
    return sg


def mirror_descent_step(a, sg, eta):
    # BR[j] = exp(-eta * sg[j]) / sum(exp(-eta * sg[i]))
    # For numerical stability, subtract max
    vals = -eta * sg
    vals -= vals.max()
    w = np.exp(vals)
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return w / s


def accelerated_mirror_descent(a0, time_limit, eta=None, verbose=False):
    """Fast accelerated mirror descent for minimizing max autocorrelation."""
    n = len(a0)
    a = normalize_to_sum1(a0)
    if eta is None:
        eta = n * 0.5
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    prev_a = a.copy()
    count = 0
    last_improve = t0
    
    while time.monotonic() - t0 < time_limit:
        t = count + 1
        
        # Compute subgradient at y
        b = fast_autocorr(y)
        k_star = int(np.argmax(b))
        sg = subgradient_at_peak(y, k_star)
        
        # Mirror descent step
        BR = mirror_descent_step(y, sg, eta)
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * BR
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Momentum update for y
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            y = a_new + momentum * (a_new - prev_a)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = a_new.copy()
            else:
                y = y / s_y
        else:
            y = a_new.copy()
        
        prev_a = a
        a = a_new
        count += 1
        
        # Check improvement
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            last_improve = time.monotonic()
        
        if verbose and count % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirDesc iter={count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def lp_best_response(a, M, solver):
    """Solve the LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0"""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    
    # Build the convolution matrix T where (T@b)[k] = (a*b)[k]
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_best_response_dual(a, M, solver):
    """Solve LP and return dual values."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)

    try:

        return None, None, None

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    a = normalize_to_sum1(a0)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    prev_a = a.copy()

    stall_count = 0

        t = count + 1
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b = b / b.sum()
        count += 1
        
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            y = a_new + momentum * (a_new - prev_a)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = a_new.copy()
            else:
                y = y / s_y
        else:
            y = a_new.copy()
        
        prev_a = a
        a = a_new
        
        # Track best
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()
        
        # Stall detection
        if time.monotonic() - last_improve > 60.0:
            stall_count += 1
            if stall_count > 20:
                break
        else:
            stall_count = 0
        
        if verbose and count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={count} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def lse_objective_and_grad(a_flat, n, T):
    """Compute LSE smooth surrogate objective and gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    # LSE approximation
    shifted = T * (b - max_b)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T
    lse_val = log_sum
    
    # Weights (softmax)
    w = exp_vals / exp_vals.sum()
    
    # Gradient of LSE w.r.t. a
    # d(LSE)/da[j] = (1/T) * sum_k w[k] * d(b[k])/da[j]
    # d(b[k])/da[j] = 2*a[k-j] if 0 <= k-j < n (roughly)
    # More precisely: d(b[k])/da[j] = a[k-j] (if valid) + a[j-k] (if valid)
    
    # Efficient computation using FFT:
    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) (with bounds)
    # = (w * a_reversed_shifted)[j] + (w * a_shifted)[j]
    
    # Actually, let's use the convolution structure:
    # b = a * a (convolution)
    # db/da = 2 * (a convolved with delta) ... this is getting complex
    # Let's use the direct formula:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_to_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def subgradient_full(a, k_star):
    n = len(a)
    sg = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    for j in range(j_min, j_max + 1):
        sg[j] = 2.0 * a[k_star - j]
    return sg


def weighted_subgradient(a, b, T):
    """Compute weighted subgradient using softmax over autocorrelation peaks."""
    n = len(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    shifted = T * (b - max_b)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-300:
        top_k = int(np.argmax(b))
        return subgradient_full(a, top_k)
    w = exp_vals / exp_sum
    # gradient = 2 * (a cross-correlation w), first n values
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:2 * n - 1] = w
    A = rfft(a_pad, fft_size)
    W = rfft(w_pad, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)[:n]
    return 2.0 * cross


def accelerated_mirror_descent(a0, time_limit, eta_scale=1.0, verbose=False):
    n = len(a0)
    a = normalize_to_sum1(a0)
    eta = n * 0.5 * eta_scale
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    prev_a = a.copy()
    count = 0
    last_improve = t0
    avg_a = np.zeros(n)
    n_avg = 0

    while time.monotonic() - t0 < time_limit:
        t = count + 1
        b = fast_autocorr(y)
        k_star = int(np.argmax(b))
        sg = subgradient_full(y, k_star)
        # Mirror step
        vals = -eta * sg
        vals -= vals.max()
        w = np.exp(vals)
        s_w = w.sum()
        if s_w < 1e-300:
            BR = np.ones(n) / n
        else:
            BR = w / s_w
        alpha = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha) * a + alpha * BR
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            y = a_new + momentum * (a_new - prev_a)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = a_new.copy()
            else:
                y = y / s_y
        else:
            y = a_new.copy()
        prev_a = a
        a = a_new
        count += 1
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            last_improve = time.monotonic()
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:

                    best_a = avg_check.copy()
                    last_improve = time.monotonic()
        if verbose and count % 5000 == 0:
            print(f"    MirDesc iter={count} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    return best_a, best_score


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def scp_optimize(a0, time_limit, solver, verbose=True):
    """Sequential Convex Programming: linearize autocorrelation, solve LP for update."""
    n = len(a0)
    a = normalize_to_sum1(a0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    count = 0
    last_improve = t0
    stall_count = 0
    step = 1.0
    b = fast_autocorr(a)

    while time.monotonic() - t0 < time_limit:
        # Build linearization matrix C where C[k,j] = 2*a[k-j] if 0<=k-j<n
        size = 2 * n - 1
        C = np.zeros((size, n))
        for i in range(n):
            j_min = max(0, -i)
            j_max = min(n - 1, size - 1 - i)
            for j in range(j_min, j_max + 1):
                C[i + j, j] = 2.0 * a[i]
        # LP: minimize M s.t. C@delta + b <= M, sum(delta)=0, delta>=-a
        delta = cp.Variable(n)
        M_var = cp.Variable()
        constraints = [
            C @ delta + b <= M_var,
            cp.sum(delta) == 0,
            delta >= -a
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        delta_val = np.asarray(delta.value, dtype=np.float64)
        if delta_val is None:
            continue
        # Feedback step size
        a_trial = a + step * delta_val
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            step *= 0.5
            continue
        a_trial = a_trial / s_t
        score_trial = fast_autocorr(a_trial).max()
        if score_trial < best_score:
            a = a_trial.copy()
            best_a = a_trial.copy()
            best_score = score_trial
            last_improve = time.monotonic()
            step = min(step * 1.1, 1.0)
            b = fast_autocorr(a)
            count += 1
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_try2 = a + step * frac * delta_val
                a_try2 = np.maximum(a_try2, 0.0)
                s2 = a_try2.sum()
                if s2 < 1e-15:
                    break
                a_try2 = a_try2 / s2
                sc2 = fast_autocorr(a_try2).max()
                if sc2 < best_score:
                    a = a_try2.copy()
                    best_a = a_try2.copy()
                    best_score = sc2
                    last_improve = time.monotonic()
                    b = fast_autocorr(a)
                    count += 1
                    break
            step *= 0.5
            if step < 0.01:
                step = 1.0
        # Stall detection
        if time.monotonic() - last_improve > 40.0:
            stall_count += 1
            if stall_count > 20:
                break
        else:
            stall_count = 0
        # Also try mirror descent steps between SCP iterations
        if count % 5 == 0:
            k_star = int(np.argmax(b))
            sg = subgradient_full(a, k_star)
            vals = -n * 0.3 * sg
            vals -= vals.max()
            w_md = np.exp(vals)
            s_w = w_md.sum()
            if s_w > 1e-300:
                BR = w_md / s_w
                a_md = 0.95 * a + 0.05 * BR
                a_md = np.maximum(a_md, 0.0)
                s_md = a_md.sum()
                if s_md > 1e-15:
                    a_md = a_md / s_md
                    sc_md = fast_autocorr(a_md).max()
                    if sc_md < best_score:
                        a = a_md.copy()
                        best_a =
