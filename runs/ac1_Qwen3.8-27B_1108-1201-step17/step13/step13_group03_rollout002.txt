def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

        t_fista = 1
        y = a.copy()
        staging = a.copy()
        max_iters = max(3000, int(5000 * (T_schedule[0] / T) ** 0.25))

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            grad = _lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 400 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T


def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 15 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _spectral_polish(a0, time_limit, verbose=False):
    """Work in frequency domain to reduce specific autocorrelation peaks."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        ac = _fast_autocorr(a)
        peak_k = int(np.argmax(ac))
        peak_val = ac[peak_k]

        if peak_val <= ac.mean() * 1.01:
            break

        A = rfft(a, fft_size)
        # In frequency domain, ac_k is related to the inverse FFT of |A|^2
        # To reduce a specific lag, we can adjust the phase of A
        # For real a, A is conjugate symmetric
        # Small perturbation: a' = a + delta, where delta has a specific frequency structure
        # The autocorrelation of delta at lag k: (delta*delta)_k
        # We want to choose delta to reduce (a*delta)_k + (delta*delta)_k at the peak lag

        # Simple approach: add a small sinusoidal perturbation that is negative at the peak lag
        # delta_j = eps * cos(2*pi*peak_k*j/n + phi) (approximately)
        improved = False
        for eps in [0.001, 0.002, 0.005, 0.01, 0.02]:
            for phi in [0, np.pi / 2, np.pi, 3 * np.pi / 2]:
                x = np.arange(n)
                delta = eps * np.cos(2 * np.pi * peak_k * x / n + phi)
                a_trial = a + delta
                if np.any(a_trial < 0):
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-10:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try random small perturbations
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            found = False
            for _ in range(200):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                frac = rng.uniform(0.0005, 0.005)
                amt = a[i] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = _fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        found = True
                        break
            if not found:
                if verbose:
                    print(f"    Spectral polish converged at score={best_score:.8f}")
                break

    return best_a, best_score


def _pairwise_local_search(a0, time_limit, seed=42, verbose=False):
    """Careful pairwise move local search for final polish."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall_count = 0
    max_stall = 200

    # Priority: target positions that contribute most to the peak
    ac = _fast_autocorr(a)
    peak_k = int(np.argmax(ac))
    js = max(0, peak_k - n + 1)
    je = min(n - 1, peak_k)
    peak_contrib = a[js:je+1] * a[peak_k - js:peak_k - je + 1]
    hot_indices = np.argsort(peak_contrib)[::-1][:max(10, n // 20)]
    hot_set = set(hot_indices.tolist())

    while time.monotonic() - t0 < time_limit and stall_count < max_stall:
        improved = False

        # Determine source: positions with high contribution to peak
        # Determine dest: positions with low contribution to all peaks
        ac = _fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:5]

        src_candidates = []
        for pk in top_peaks:
            js_p = max(0, pk - n + 1)
            je_p = min(n - 1, pk)
            idx = np.arange(js_p, je_p + 1)
            contrib = a[idx] * a[pk - idx]
            src_candidates.append((idx[np.argmax(contrib)], contrib.max()))

        # Sort by contribution (highest first)
        src_candidates.sort(key=lambda x: x[1], reverse=True)

        # Destination: positions far from peaks (low autocorrelation contribution)
        all_contrib = np.zeros(n)
        for pk in top_peaks:
            js_p = max(0, pk - n + 1)
            je_p = min(n - 1, pk)
            idx = np.arange(js_p, je_p + 1)
            contrib = a[idx] * a[pk - idx]
            all_contrib[js_p:je_p+1] += contrib

        dst_candidates = np.argsort(all_contrib)[:max(5, n // 10)]

        for src, src_contrib in src_candidates[:5]:
            if a[src] < norm * 1e-10:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                for frac in [0.0005, 0.001, 0.002, 0.005, 0.01]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = _fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall_count = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation to escape local minimum
            stall_count += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall_count = 0

        if verbose and stall_count % 50 == 0:
            print(f"    Pairwise search stall={stall_count}, score={best_score:.8f}")

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _generate_initializations(n, seed=42):
    """Generate diverse initialization candidates."""
    rng = np.random.default_rng(seed)
    candidates = []

    # Sine-based with various exponents
    x = np.linspace(0.0, 1.0, n)
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        c = np.sin(np.pi * x) ** alpha
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 1e-10:
            candidates.append(c / s)

    # Raised cosine
    c = 0.5 * (1 + np.cos(2 * np.pi * x))
    c = np.maximum(c, 0.0)
    s = c.sum()
    if s > 1e-10:
        candidates.append(c / s)

    # Triangular
    c = 1 - 2 * np.abs(x - 0.5)
    c = np.maximum(c, 0.0)
    s = c.sum()
    if s > 1e-10:
        candidates.append(c / s)

    # Random positive
    c = rng.dirichlet(np.ones(n))
    candidates.append(c)

    # Gaussian-like
    c = np.exp(-20 * (x - 0.5) ** 2)
    s = c.sum()
    if s > 1e-10:
        candidates.append(c / s)

    # Bimodal
    c = np.exp(-30 * (x - 0.3) ** 2) + np.exp(-30 * (x - 0.7) ** 2)
    s = c.sum()
    if s > 1e-10:
        candidates.append(c / s)

    return candidates


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization with adaptive time allocation."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1710.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Quick LSE-FISTA at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 150.0)
    if t1 > 15:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        a_out, _ = _lse_fista(best_seq, t1, n_stages=4, verbose=True)
        _try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: LP-Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.22, 350.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-Nesterov n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t2, solver, verbose=True)
        _try_update(a_out, f"lp1_n{n_best}")

    # === Phase 3: Try larger length n+400 ===
    n2 = len(best_seq) + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP-Nesterov n={n2} (budget={t3:.0f}s) ===")
        init_long = _interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = _lp_nesterov(a_norm, t3, solver, verbose=True)
        _try_update(a_out, f"lp2_n{n2}")

    # === Phase 4: LSE-FISTA at current best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 10:
        print(f"\n=== Phase 4: LSE-FISTA n={n_cur} (budget={t4:.0f}s) ===")
        a_out, _ = _lse_fista(best_seq, t4, n_stages=4, verbose=True)
        _try_update(a_out, f"lse2_n{n_cur}")

    # === Phase 5: Try even larger length n+800 ===
    n3 = len(best_seq) + 800
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.14, 200.0)
    if t5 > 25:
        print(f"\n=== Phase 5: LP-Nesterov n={n3} (budget={t5:.0f}s) ===")
        init_long3 = _interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = _lp_nesterov(a_norm, t5, solver, verbose=True)
        _try_update(a_out, f"lp3_n{n3}")

    # === Phase 6: LP polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.15, 180.0)
    if t6 > 20:
        print(f"\n=== Phase 6: LP polish n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t6, solver, verbose=True)
        _try_update(a_out, f"lp_polish_n{n_best}")

    # === Phase 7: Spectral polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.10, 80.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Spectral polish n={n_best} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _spectral_polish(a_norm, t7, verbose=False)
        _try_update(a_out, f"spec_n{n_best}")

    # === Phase 8: Pairwise local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.15, 120.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Pairwise search n={n_best} (budget={t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _pairwise_local_search(a_norm, t8, seed=12345, verbose=False)
        _try_update(a_out, f"pair1_n{n_best}")

    # === Phase 9: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t9 = min(remaining * 0.18, 150.0)
    if t9 > 20:
        print(f"\n=== Phase 9: Final LP n={n_best} (budget={t9:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t9, solver, verbose=True)
        _try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 10: Final pairwise search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t10 = min(remaining * 0.4, 80.0)
        if t10 > 5:
            print(f"\n=== Phase 10: Final pairwise n={n_best} (budget={t10:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = _pairwise_local_search(a_norm, t10, seed=99999, verbose=False)
            _try_update(a_out, f"pair_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
