**Fundamentally different approach: "Fourier domain optimization"**

The convolution in the time domain is multiplication in the frequency domain. So `conv(a,a) = idft(|dft(a)|^2)` (with appropriate zero-padding).

If we want `conv(a,a)` to be flat, we want `|dft(a)|^2` to be flat, which means `|dft(a)|` is constant. This is the "CAZAC" (Constant Amplitude Zero Auto-Correlation) property.

The question is: does there exist a non-negative sequence with this property? Probably not exactly, but we can get close.

The optimization becomes: minimize the variance of `|A_k|^2` (or some other measure of non-flatness), subject to `a >= 0, sum(a) = 1`.

**Peak-Directed Coordinated Refinement (PDCR):**

The idea is:
1. Identify the top K peaks of conv(a,a).
2. For each peak k, identify the "bottleneck" elements - those j where a[j] * a[k-j] is large.
3. Compute a "pressure" map: for each element j, sum up its contribution to all top peaks.
4. Move mass from high-pressure elements to low-pressure elements in a coordinated way.
5. Use a line search to find the best step size.

The key difference from simple peak shaving is that we move mass from ALL contributing elements simultaneously, not just one at a time.

```python
def peak_directed_refine(a0, time_limit, K=30, seed=42):
    n = len(a0)

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:K]
        
        # Compute pressure for each element
        pressure = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            pressure[idx] += contrib
        
        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        
        # Move mass from top-pressure to bottom-pressure elements
        n_move = max(10, n // 20)
        src_idx = np.argsort(pressure)[::-1][:n_move]
        dst_idx = np.argsort(pressure)[:n_move]
        
        src_total = a[src_idx].sum()
        if src_total < 1e-15:
            break
        
        # Try different fractions
        improved = False
        for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]:
            amt = src_total * frac
            a_trial = a.copy()
            # Distribute removal proportionally
            src_frac = a[src_idx] / src_total
            a_trial[src_idx] -= amt * src_frac
            # Distribute addition to lowest pressure elements
            a_trial[dst_idx] += amt / len(dst_idx)

            if ss > 1e-15:
```

```python
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    """Full convolution a*b using FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-14:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.ones(n) * target_sum / n
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        ss = result.sum()
        if ss > 0:
            result = result / ss * target_sum
    return result

    for k in row_indices:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            row[js:je + 1] = a_fixed[k - je:k - js + 1][::-1]
        rows.append(row)
    return np.array(rows)


def solve_lp_column_gen(a_fixed, solver, max_active=100, tol=1e-12):
    """Solve LP: min M s.t. conv(a_fixed,b)_k <= M, sum(b)=1, b>=0 using column generation."""
    n = len(a_fixed)
    conv_len = 2 * n - 1
    # Initialize active set with top peaks
    init_b = np.ones(n) / n
    init_conv = fast_conv(a_fixed, init_b)
    active = np.argsort(init_conv)[::-1][:min(max_active, conv_len)]

    b_val = init_b.copy()
    M_val = init_conv.max()

    for outer in range(40):
        T_active = build_conv_rows(a_fixed, active)
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T_active @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, max_iter=500)
        except Exception:
            pass
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Try fallback solver
            for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt, max_iter=500)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return b_val, M_val

        b_val_new = np.asarray(b_var.value, dtype=np.float64)
        b_val_new = np.maximum(b_val_new, 0)
        s = b_val_new.sum()
        if s < 1e-15:
            return b_val, M_val
        b_val_new = b_val_new / s
        M_val_new = max(float(M_var.value), 0.0)

        # Separation: check if any inactive constraint is violated
        conv_vals = fast_conv(a_fixed, b_val_new)
        if len(active) < conv_len:
            inactive_mask = np.ones(conv_len, dtype=bool)
            inactive_mask[active] = False
            if inactive_mask.any():
                max_inactive = conv_vals[inactive_mask].max()
                if max_inactive > M_val_new + tol:
                    new_idx = int(np.argmax(np.where(inactive_mask, conv_vals, -np.inf)))
                    active = np.append(active, new_idx)
                    if len(active) > max_active:
                        active = np.argsort(conv_vals)[::-1][:max_active]
                    b_val = b_val_new
                    M_val = max(M_val_new, max_inactive)
                    continue

        b_val = b_val_new
        M_val = M_val_new
        break

    return b_val, M_val


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response for the zero-sum game with column generation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0
    stall_threshold = 3000
    best_lp_M = float('inf')

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_column_gen(y, solver, max_active=100)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        # Check scores
        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-9:
                stall_count = 0
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            if score_y < best_score - 1e-9:
                stall_count = 0
            best_score = score_y
            best_a = y.copy()

        if M_val < best_lp_M - 1e-8:
            best_lp_M = M_val
            stall_count = 0

        avg_a += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        stall_count += 1
        if stall_count > stall_threshold:
            break

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NL t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def compute_lse_grad_a(a, T_param):
    """Compute gradient of (1/T)*log(sum exp(T*conv(a,a)_k)) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T_param * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(np.asarray(a[::-1], dtype=np.float64), fft_size)
    W = rfft(np.asarray(w, dtype=np.float64), fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=30, lr=0.004):
    """LogSumExp-smoothed optimization with Adam optimizer."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(20, min(150, int(remaining * 80 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            a = softmax_fn(z)
            grad_a, _ = compute_lse_grad_a(a, stage_T)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.005 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 15 == 0:
                a = softmax_fn(z)
                score = internal_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 6 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_directed_refine(a0, time_limit, K=30, seed=42):
    """
    Peak-Directed Coordinated Refinement:
    Simultaneously reduce mass from elements contributing to ALL top peaks,
    and redistribute to low-pressure elements.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:K]

        # Compute pressure for each element (contribution to top peaks)
        pressure = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            pressure[idx] += contrib

        # Identify source and destination sets
        n_move = max(10, n // 15)
        src_idx = np.argsort(pressure)[::-1][:n_move]
        dst_idx = np.argsort(pressure)[:n_move]

        src_total = a[src_idx].sum()
        if src_total < 1e-15:
            break

        improved = False
        for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.04, 0.06, 0.08]:
            amt = src_total * frac
            a_trial = a.copy()
            src_frac = a[src_idx] / (src_total + 1e-30)
            a_trial[src_idx] -= amt * src_frac
            a_trial[dst_idx] += amt / len(dst_idx)
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial = a_trial / ss
            score = internal_score(a_trial)
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > 300:
                break
            # Random perturbation to escape
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.002, 0.015)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

    return best_a, best_score


def subgradient_optimize(a0, time_limit, verbose=True):
    """
    Subgradient optimization with adaptive smoothing.
    Uses top-K peaks with softmax weights.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    v = np.zeros(n)
    beta = 0.85
    base_lr = 0.08
    t = 0
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)

        # Adaptive smoothing temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T_smooth = max(1.0, 40.0 * (1.0 - progress) + 1.0)
        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Gradient via FFT
        A_rev = rfft(np.asarray(a[::-1], dtype=np.float64), fft_size)
        W = rfft(np.asarray(w, dtype=np.float64), fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        lr_t = base_lr / (1 + 0.005 * t)
        v = beta * v + (1 - beta) * grad
        a_new = a - lr_t * v
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_shaving_final(a0, time_limit, seed=123):
    """Final coordinate-descent peak shaving with random restarts."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = int(idx[np.argmax(contrib)])

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j in range(js2, je2 + 1):
                    peak_areas.add(j)

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = int(dst_pool[rng.integers(len(dst_pool))])

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 400:
                break

    return best_a, best_score

    n_long = int(round(n_base * 1.3))
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 400.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral at longer length
        t5a = t5 * 0.1
        a_spec = spectral_shaping(init_long, n_iter=150, alpha=0.25)

        if np.isfinite(sc) and sc < best_score:

            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s

        # LP at longer length

            sc = evaluate_safe(a_out)

                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [PDCR_n{n_long}]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start

    # Phase 6: Adam LSE polish at best length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Adam LSE n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 120.0)

        a_out, _ = adam_lse_optimize(best_seq, t6, T_start=30.0, T_end=0.03, n_stages=20, lr=0.003)

            if s > 0:

            print(f"  *** NEW BEST [Adam_LSE]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 6: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 7: PDCR + peak shave final
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final refinement n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 100.0)
    if t7 > 10:
        a_out, _ = peak_directed_refine(best_seq, t7 * 0.5, K=20, seed=999)

        t7b = min(remaining * 0.15, 60.0)
        if t7b > 5:
            a_out, _ = peak_shaving_final(best_seq, t7b, seed=444)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:

                best_seq = np.maximum(a_out, 0)

                if s > 0:

                print(f"  *** NEW BEST [peak_shave_final]: {best_score:.6f}")

    # Phase 8: One more LP round if time remains
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: More LP n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t8 = min(remaining * 0.35, 150.0)
    if t8 > 20:

            print(f"  *** NEW BEST [LP_round2]: {best_score:.6f}")

    # Final peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)

        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()

        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
