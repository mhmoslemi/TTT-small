import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

# Global solver
_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.SCS
    print("  LP solver (fallback): SCS")
    return _global_solver

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def build_conv_matrix(a):
    """Build the convolution matrix T where T @ b = conv(a, b) (first 2n-1 lags)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_max_sum(a, M, solver):
    """LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_min_peak(a, solver):
    """LP: min t s.t. conv(a,b)_k <= t for all k, sum(b) = 1, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    b_val = b_val / b_val.sum()
    return b_val

def nesterov_lp(a0, time_limit, solver, use_min_peak=False, restart_interval=80, verbose=True, seed=42):
    """Nesterov-accelerated LP best-response with periodic restarts."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    a = np.maximum(a, 0.0)
    a = a * c / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    t = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        if use_min_peak:
            b = lp_min_peak(y, solver)
        else:
            b = lp_max_sum(y, M_curr, solver)
        
        if b is None:
            continue
        
        b = np.maximum(b, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        
        y = y_new
        x = x_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        # Periodic restart
        if t % restart_interval == 0:
            # Perturb the best point and restart
            perturbation = rng.normal(0, 0.01 * c / n, n)
            y = best_a + perturbation
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * c / s_y
            x = y.copy()
            t = 0  # Reset Nesterov counter
        
        if verbose and t % 20 == 0:

            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def proj_simplex(v, s):

    if n == 0 or s < 1e-15:

    x = a.copy()
    t = 0
    iter_count = 0

        elapsed = time.monotonic() - t0

        if M_curr < 1e-14:

            continue

            continue

        if s_y < 1e-15:

        else:

        x = x_new

            x = x_new

            print(f"    LP1 {J:.6f} ({elapsed:.0f}s, iter={iter_count})")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        if verbose and t % 40 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP1 J={J:.6f} ({elapsed:.0f}s, iter={iter_count})")
    return best_a, best_m

def nesterov_lp_minpeak(a0, time_limit, solver, restart_interval=50, seed=123, verbose=True):
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    a = np.maximum(a, 0.0)
    a = a * c / a.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    y = a.copy()
    x = a.copy()
    t = 0
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        iter_count += 1
        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = y
        constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if b_val.sum() < 1e-15:
            continue
        g = b_val * c / b_val.sum()
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        y = y_new
        x = x_new
        sy_m = fast_autocorr(y).max()
        if sy_m < best_m:
            best_m = sy_m
            best_a = y.copy()
        sx_m = fast_autocorr(x).max()
        if sx_m < best_m:
            best_m = sx_m
            best_a = x.copy()
        if t >= restart_interval and t % restart_interval == 0:
            perturbation = rng.normal(0, 0.03 * c / np.sqrt(n), n)
            y_new = best_a + perturbation
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new * c / s_y
            x = y_new.copy()
            y = y_new
            t = 0
        if verbose and t % 40 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP2 J={J:.6f} ({elapsed:.0f}s, iter={iter_count})")
    return best_a, best_m

def interpolated_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def coordinate_polish(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 400:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:12]
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[int(k) - idx]
            max_idx = np.argmax(contrib)
            src = idx[max_idx]
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                peak_mask[js2:je2 + 1] = True
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 3:
                candidates = np.where(np.arange(n) != src)[0]
            cands_sorted = candidates[np.argsort(a[candidates])][:25]
            for dst in cands_sorted:
                if dst == src:
                    continue
                for frac in [0.01, 0.02, 0.04, 0.08, 0.15]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        break
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        break
                    a_trial = a_trial * c / s_t
                    ac_trial = fast_autocorr(a_trial)
                    M_trial = ac_trial.max()
                    if M_trial < best_m - 1e-12:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                if a[i] >= 0:
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a * c / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = a * c / a.sum()
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m / (c * c)
        print(f"    Coord polish ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def mirror_descent_step(a0, time_limit, top_k=5, beta=3.0, seed=99, verbose=True):
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    a = np.maximum(a, 1e-15)
    a = a * c / a.sum()
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    avg_a = a.copy()
    n_avg = 1
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        step = 1.0 / (np.sqrt(t) + 1.0)
        ac = fast_autocorr(a)
        top_idx = np.argsort(ac)[::-1][:top_k]
        top_vals = np.maximum(ac[top_idx], 0)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w = w / w_sum
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            i_start = max(0, int(k) - n + 1)
            i_end = min(n, int(k) + 1)
            idx = np.arange(i_start, i_end)
            j = int(k) - idx
            grad[idx] += 2.0 * wi * a[j]
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.0f}s): J={J:.6f}")
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    best_seq = None
    best_score = float('inf')
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    # Phase 1: LP1 (max sum) at parent length
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP1 maxsum n={n0} ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 20:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = nesterov_lp_maxsum(a_norm, t1, solver, restart_interval=50, seed=42, verbose=True)
        try_update(a_out, f"lp1_n{n0}")
    # Phase 2: LP2 (min peak) at parent length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP2 minpeak n={n0} ===")
    t2 = min(remaining * 0.12, 200.0)
    if t2 > 20:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = nesterov_lp_minpeak(a_norm, t2, solver, restart_interval=50, seed=123, verbose=True)
        try_update(a_out, f"lp2_n{n0}")
    # Phase 3: LP1 at n=2000
    n1 = 2000
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP1 maxsum n={n1} ===")
    t3 = min(remaining * 0.15, 280.0)
    if t3 > 30:
        a_long = interpolated_sequence(best_seq, n1)
        c1 = np.sqrt(2.0 * n1)
        a_norm = a_long * c1 / a_long.sum()
        a_out, _ = nesterov_lp_maxsum(a_norm, t3, solver, restart_interval=40, seed=77, verbose=True)
        try_update(a_out, f"lp1_n{n1}")
    # Phase 4: LP1 at n=2400
    n2 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP1 maxsum n={n2} ===")
    t4 = min(remaining * 0.20, 350.0)
    if t4 > 30:
        a_long2 = interpolated_sequence(best_seq, n2)
        c2 = np.sqrt(2.0 * n2)
        a_norm = a_long2 * c2 / a_long2.sum()
        a_out, _ = nesterov_lp_maxsum(a_norm, t4, solver, restart_interval=35, seed=55, verbose=True)
        try_update(a_out, f"lp1_n{n2}")
    # Phase 5: Coordinate polish at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord polish n={n_best} ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = coordinate_polish(a_norm, t5, seed=42, verbose=True)
        try_update(a_out, f"coord_n{n_best}")
    # Phase 6: Final LP1 on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final LP1 n={n_best} ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 15:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = nesterov_lp_maxsum(a_norm, t6, solver, restart_interval=30, seed=200, verbose=True)
        try_update(a_out, f"lp1_final_n{n_best}")
    # Phase 7: Final coord polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        t7 = min(remaining * 0.4, 80.0)
        if t7 > 5:
            c_best = np.sqrt(2.0 * n_best)
            a_norm = best_seq * c_best / best_seq.sum()
            a_out, _ = coordinate_polish(a_norm, t7, seed=88, verbose=True)
            try_update(a_out, f"coord_final_n{n_best}")
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0, 1.0, 1.0]
    return [float(x) for x in best_seq]
