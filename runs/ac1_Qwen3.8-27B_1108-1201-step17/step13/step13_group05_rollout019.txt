            w = exp_bw / exp_bw.sum()
            
            # Compute smooth gradient via FFT
            g = compute_smooth_grad_fft(y, w, fft_size_cc)
            
            # Step size (adaptive)
            step = C_step / np.sqrt(t)
            
            # Nesterov extrapolation
            if use_nesterov and t > 1:
                beta_nest = (t - 1.0) / (t + 2.0)
                y_extrap = y + beta_nest * (y - x_prev)
                y_extrap = np.maximum(y_extrap, 1e-15)
                y_extrap = y_extrap / y_extrap.sum()
            else:
                y_extrap = y.copy()
            
            # Recompute gradient at extrapolated point if different
            if use_nesterov and t > 1 and not np.allclose(y_extrap, y, atol=1e-10):
                ac_e = fast_autocorr(y_extrap)
                M_e = ac_e.max()
                shifted_e = ac_e - M_e
                bw_e = beta * shifted_e
                max_bw_e = bw_e.max()
                exp_bw_e = np.exp(bw_e - max_bw_e)
                w_e = exp_bw_e / exp_bw_e.sum()
                g = compute_smooth_grad_fft(y_extrap, w_e, fft_size_cc)
            
            # Mirror descent update: a_new[i] prop to a[i] * exp(-step * g[i])
            log_a = np.log(y_extrap + 1e-15)
            log_a_new = log_a - step * g
            max_log = log_a_new.max()
            a_new = np.exp(log_a_new - max_log)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            
            x_prev = y_extrap.copy()
            y = a_new
            a = a_new
            
            # Evaluate and track best
            M_new = fast_autocorr(a).max()
            if M_new < best_f:
                best_f = M_new
                best_a = a.copy()
                no_improve_count = 0
            
            # Also check Nesterov accelerated point
            if use_nesterov and t > 3:
                beta_nest2 = (t - 1.0) / (t + 2.0)
                x_nest = a + beta_nest2 * (a - x_prev)
                x_nest = np.maximum(x_nest, 1e-15)
                x_nest = x_nest / x_nest.sum()
                M_nest = fast_autocorr(x_nest).max()
                if M_nest < best_f:
                    best_f = M_nest
                    best_a = x_nest.copy()
                    no_improve_count = 0
            
            # Verbose output
            if verbose and t % 2000 == 0:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_f
                print(f"      ASMD iter={global_iter} ({elapsed:.0f}s): J={J:.6f}, beta={beta:.1f}")
        
        # End of cycle: check for improvement
        if best_f >= f_before_cycle - 1e-10:
            no_improve_count += 1
        else:
            no_improve_count = 0
        
        # Update starting point for next cycle
        a = best_a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_f
            print(f"    ASMD cycle done ({elapsed:.0f}s): best J={J:.6f}, f={best_f:.8f}")
    
    return best_a, best_f

def topk_md_optimize(a_init, time_limit, top_k=5, beta=3.0, verbose=True):
    """Mirror descent with top-k soft subgradient (from previous SOTA, kept as complement)."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_f = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    H = np.log(n)
    C_step = 1.0 / np.sqrt(2.0 * H + 1.0)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        g = compute_soft_subgrad_topk(a, ac, top_k=top_k, beta=beta)
        
        step = C_step / np.sqrt(t)
        
        log_a = np.log(a + 1e-15)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_f:
            best_f = M_new
            best_a = a.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_f
            print(f"      TopK-MD t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_f

def lp_best_response(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0."""
    n = len(a)
    solver = get_solver()
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix: T[i,j] such that (T @ b)[k] = conv(a,b)[k]
    # conv(a,b)[k] = sum_j a[j] * b[k-j] for valid j, k-j
    # So T[k,j] = a[j] if 0 <= k-j < n, else 0
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    constraints = [T @ b_var <= M + 1e-10]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a_init, time_limit, verbose=True):
    """Nesterov-accelerated LP best-response method."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_f = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x_prev = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-15:
            break
        
        b = lp_best_response(y, M_curr)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum  # normalize to sum=1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / s_x
        
        y = y_new
        x_prev = y_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_f:
            best_f = score_y
            best_a = y.copy()
        
        if t > 2:
            beta2 = (t - 1.0) / (t + 2.0)
            x_nest = y + beta2 * (y - x_prev)
            x_nest = np.maximum(x_nest, 0.0)
            s_xn = x_nest.sum()
            if s_xn > 1e-15:
                x_nest = x_nest / s_xn
                score_xn = fast_autocorr(x_nest).max()
                if score_xn < best_f:
                    best_f = score_xn
                    best_a = x_nest.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_f
            print(f"      LP iter={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a, best_f

def coordinate_polish(a_init, time_limit, verbose=True):
    """Coordinate descent: shift mass to reduce max autocorrelation."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init, fast_autocorr(a_init).max()
    a = a / s
    
    best_a = a.copy()
    best_f = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        top_count = min(10, 2 * n - 1)
        top_k = np.argsort(ac)[::-1][:top_count]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[ki - idx]
            src = idx[np.argmax(contrib)]
            
            # Find best destination: minimize increase in max
            # Try all positions not in any top peak's support
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Pick a few candidates with low values
            cands_sorted = candidates[np.argsort(a[candidates])][:20]
            if len(cands_sorted) == 0:
                continue
            
            dst = cands_sorted[rng.integers(len(cands_sorted))]
            if dst == src:
                continue
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                ac_trial = fast_autocorr(a_trial)
                M_trial = ac_trial.max()
                if M_trial < best_f - 1e-14:
                    a = a_trial
                    best_f = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 50 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    if a[i] >= 0:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if M_new < best_f:
                            best_f = M_new
                            best_a = a.copy()
                            stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_f
        print(f"    Coord polish ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_f

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape and normalizing to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def generate_diverse_inits(n, rng):
    """Generate multiple diverse initial sequences of length n, all normalized to sum=1."""
    inits = []
    x = np.linspace(0.0, 1.0, n)
    
    # Half-sine
    s1 = np.sin(np.pi * x) ** 2
    s1 = np.maximum(s1, 1e-10)
    inits.append(s1 / s1.sum())
    
    # Raised cosine
    s2 = 0.5 * (1 + np.cos(2 * np.pi * x))
    s2 = np.maximum(s2, 1e-10)
    inits.append(s2 / s2.sum())
    
    # Triangle (hat)
    s3 = 1.0 - np.abs(2.0 * x - 1.0)
    s3 = np.maximum(s3, 1e-10)
    inits.append(s3 / s3.sum())
    
    # Exponential bell from center
    center = (n - 1) / 2.0
    s4 = np.exp(-((np.arange(n) - center) ** 2) / (2.0 * (n / 4.0) ** 2))
    inits.append(s4 / s4.sum())
    
    # Random smooth
    base = rng.random(n)
    # Smooth with a window
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    s5 = np.convolve(base, kernel, mode='same')
    s5 = np.maximum(s5, 1e-10)
    inits.append(s5 / s5.sum())
    
    # Bi-modal
    c1, c2 = n * 0.25, n * 0.75
    s6 = np.exp(-((np.arange(n) - c1) ** 2) / (2.0 * (n/12.0)**2)) + \
         np.exp(-((np.arange(n) - c2) ** 2) / (2.0 * (n/12.0)**2))
    inits.append(s6 / s6.sum())
    
    return inits

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

    """
    Main search function: multi-phase optimization with diverse methods and lengths.
    Phase 1: Accelerated smooth-max MD at n=1600 (from parent)
    Phase 2: Top-k MD at n=1600 (complementary)
    Phase 3: Scale to n=2400, accelerated smooth-max MD
    Phase 4: Scale to n=3200, accelerated smooth-max MD
    Phase 5: LP polish on best candidate
    Phase 6: Coordinate descent micro-optimize

    rng = np.random.default_rng(42)
    t_global = time.monotonic()

    print(f"Time budget: {time_budget:.0f}s")
    get_solver()
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()

    parent = None

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy() / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")

        pass
    
    if best_seq is None:

    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final ASMD n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 15:
        a_out, _ = accelerated_md_optimize(best_seq, t8, beta_start=30.0, beta_end=100.0,
                                            cycle_len=2000, use_nesterov=True, verbose=True)
        try_update(a_out, f"final_n{n_best}")
    
    # ===== Phase 9: Final coordinate polish =====
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.40, 150.0)
    if t9 > 10:
        a_out, _ = coordinate_polish(best_seq, t9, verbose=True)
        try_update(a_out, f"finalcoord_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback
        n_fb = 1600
        x = np.linspace(0, 1, n_fb)
        fb = np.sin(np.pi * x) ** 2
        fb = fb / fb.sum()
        return [float(x) for x in fb]
    
    return [float(x) for x in best_seq]
