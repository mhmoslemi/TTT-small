def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def exact_subgrad(a, ac, top_k=10, power=2.0):
    """Compute subgradient using top-k lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].copy()
    top_vals = np.maximum(top_vals, 0)
    w = top_vals ** power
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    a = np.maximum(a, 1e-15)
    a = a * c / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = a.copy()
    n_avg = 1
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = 1.0 / (np.sqrt(t) * (best_m / c + 1e-10))
        
        ac = fast_autocorr(a)
        g = exact_subgrad(a, ac, top_k=top_k, power=power)
        
        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def lp_best_response(a, M, solver, reduced=True):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix using banded structure
    # Only need constraints for k = 0, ..., n-1 (by symmetry)
    if reduced:
        n_cons = n
    else:
        n_cons = 2 * n - 1
    
    # Use sparse matrix for efficiency
    from scipy.sparse import lil_matrix
    T = lil_matrix((n_cons, n))
    for j in range(n):
        for ki in range(n_cons):
            val = a[ki - j] if 0 <= ki - j < n else 0
            if val > 0:
                T[ki, j] = val
    
    T = T.tocsr()
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver, reduced=True)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        
        y = y_new
        x = x_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
