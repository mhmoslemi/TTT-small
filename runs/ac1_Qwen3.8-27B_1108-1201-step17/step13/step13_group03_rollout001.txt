def _fast_autocorr(a):
    """Compute full linear convolution a * a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v):
    """Project vector v onto the probability simplex {x: sum=1, x>=0}."""
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_gradient(y, T):
    """Gradient of log-sum-exp smoothing of max autocorrelation."""
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -80.0, 80.0)
    exp_s = np.exp(shifted)
    wsum = exp_s.sum()
    if wsum < 1e-300:
        w = np.zeros_like(ac)
        w[np.argmax(ac)] = 1.0
    else:
        w = exp_s / wsum
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def _lse_fista_adaptive(a0, time_limit, T_start, T_end, verbose=True):
    """
    FISTA on LSE-smoothed max autocorrelation with continuous temperature annealing.
    T decreases exponentially over the time budget.
    Returns (best_a, best_score) where best_score = max(conv(a,a)) with sum(a)=1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    # Enforce small boundary to reduce boundary peaks
    a[0] *= 0.1
    a[-1] *= 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    staging = a.copy()
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    t_fista = 1.0
    total_iters = 0

    log_ratio = np.log(T_start / T_end)
    elapsed_at_start = time.monotonic() - t0

    while True:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Continuous exponential temperature schedule
        progress = min(elapsed / time_limit, 1.0)
        T = T_start * np.exp(-log_ratio * progress)
        T = max(T, 0.001)

        # Lipschitz constant estimate: L = max eigenvalue of Hessian ~ n/T (rough)
        L = max(n / (T * 5.0), n * 0.5)

        # Adaptive: run bursts of iterations
        burst_size = max(500, int(20000 * (T_end / T) ** 0.5))
        burst_size = max(burst_size, 200)

        for _ in range(burst_size):
            if time.monotonic() - t0 > time_limit:
                break

            grad = _lse_gradient(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1.0:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1.0
            total_iters += 1

        # Evaluate periodically
        score = _fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Also check y
        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if verbose and total_iters % 5000 < burst_size:
            print(f"    LSE-FISTA adapt: iters={total_iters}, T={T:.4f}, score={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")

    score = _fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def _build_conv_matrix(a, n):
    """Build the convolution matrix T where T @ b = conv(a, b)."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T


def _lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def _lp_homotopy(a0, time_limit, solver, verbose=True):
    """
    LP best-response with homotopy M-tightening.
    Start with M = max(conv(a,a)), solve LP, update a, then tighten M.
    Uses Nesterov acceleration on the iterates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    M_homotopy = best_score
    stall_count = 0
    stall_threshold = 25

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP with current M
        M_curr = min(M_homotopy, _fast_autocorr(y).max() * 1.01)
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            # Try with looser M
            M_curr = _fast_autocorr(y).max() * 1.05
            b = _lp_best_response(y, M_curr, solver)
            if b is None:
                continue

        n_iters += 1
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = b.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
            # Tighten homotopy M
            M_homotopy = best_score * 1.02
        else:
            stall_count += 1

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
            M_homotopy = best_score * 1.02

        # If stalled, relax M to allow progress
        if stall_count > stall_threshold:
            M_homotopy = _fast_autocorr(y).max() * 1.03
            stall_count = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-homotopy iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_homotopy:.8f}")

    return best_a, best_score


def _coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find the best value that
    minimizes the max of the convolution. Uses bisection on each coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Precompute: for coordinate i, conv_k changes only for k in [i, i+n-1]
    # (since conv[k] = sum_j a[j]*a[k-j], changing a[i] affects k where i is in range)

    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        # Cycle through coordinates in random order
        order = rng.permutation(n)
        improved_this_pass = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            if old_val < 1e-12:
                continue

            # The contribution of a[i] to conv[k] is a[i] * (sum of a[j] where j = k-i, valid)
            # conv[k] contains a[i]*a[k-i] for valid k-i
            # We want to find new_val that minimizes max conv

            # Determine which conv positions are affected
            # conv[k] gets contribution a[i]*a[k-i] for k in [i, i+n-1]
            # Also conv[k] gets a[k-i]*a[i] for k in [i, i+n-1] (same thing by symmetry)
            # Actually conv[k] = sum_j a[j]*a[k-j], the term with j=i is a[i]*a[k-i]
            # for k-i in [0, n-1], i.e., k in [i, i+n-1]
            # But also the term with k-j=i, i.e., j=k-i, gives a[k-i]*a[i] (same)
            # So conv[k] = ... + a[i]*a[k-i] + ... for k in [i, i+n-1]
            # But wait, if k-i = i (i.e., k=2i), then it's a[i]*a[i] counted once
            # The full conv[k] = sum_j a[j]*a[k-j], and the j=i term is a[i]*a[k-i]
            # The j=k-i term (if k-i != i) is a[k-i]*a[i], same value
            # So if i != k-i (i.e., k != 2i), the contribution is 2*a[i]*a[k-i]
            # If k == 2i, contribution is a[i]^2

            # For efficiency, let's just try a few candidate values
            # and check the max of the full convolution

            candidates = []
            # Try reducing
            candidates.append(old_val * 0.5)
            candidates.append(old_val * 0.8)
            candidates.append(old_val * 0.95)
            # Try increasing slightly
            candidates.append(old_val * 1.2)
            # Try setting to 0
            if old_val > 0.01:
                candidates.append(0.0)

            best_trial = None
            best_trial_score = best_score

            for cv in candidates:
                if cv < 0 or cv > 1.0:
                    continue
                a[i] = cv
                # Renormalize
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s_new
                sc = _fast_autocorr(a_norm).max()
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = a_norm.copy()

            a[i] = old_val
            if best_trial is not None:
                a = best_trial
                if best_trial_score < best_score:
                    best_score = best_trial_score
                    best_a = a.copy()
                    improved_this_pass = True

        iter_count += 1
        if verbose and iter_count % 10 == 0:
            print(f"    CoordDescent pass={iter_count}: score={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")

        if not improved_this_pass and iter_count > 5:
            break

    return best_a, best_score


def _interp_sequence(f, new_n):
    """Interpolate sequence f to new length new_n, preserving the shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _perturb_escape(a, rng, strength=0.1):
    """Add random perturbation to escape local minima."""
    n = len(a)
    noise = rng.normal(0, strength, n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800.0):

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        # Initialize with a raised-cosine shape (zero at boundaries)
        x = np.linspace(0, np.pi, n_base)
        best_seq = 0.5 * (1 + np.cos(x))
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init (raised cosine): n={n_base}, score={best_score:.6f}")

    def try_update(a, label):

            return

        if s < 0.01:
            return

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()

    remaining = time_budget - (time.monotonic() - t_global)

    remaining = time_budget - (time.monotonic() - t_global)

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 10: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        t10 = min(remaining * 0.4, 120.0)
        print(f"\n=== Phase 10: Final LP polish n={n_cur} (budget={t10:.0f}s) ===")
        a_out, _ = _lp_homotopy(best_seq, t10, solver, verbose=True)

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 15:
        t11 = min(remaining * 0.5, 60.0)

        a_out, _ = _coordinate_descent(best_seq, t11, verbose=True)

    # Final verification
    if best_seq is not None:

        if np.isfinite(fs):

        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    # Ensure all values are valid floats
    result = [float(x) for x in best_seq]
    return result
